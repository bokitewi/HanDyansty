# Reuse the stable release key, with the original vanilla transaction pattern.
# Never change government as a side effect of independence.
hd_qxzl_release_ruler_effect = {
	if = {
		limit = { hd_hb_live_vassal_trigger = yes }
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = hd_hb_independence_change
			add_claim_on_loss = no
		}
		becomes_independent = { change = scope:hd_hb_independence_change }
		resolve_title_and_vassal_change = scope:hd_hb_independence_change
		if = {
			limit = { is_independent_ruler = yes }
			hd_hb_clear_player_pending_effect = yes
			debug_log = "HD_HB INDEPENDENCE_CONFIRMED"
		}
		else = { debug_log = "HD_HB INDEPENDENCE_NOT_CONFIRMED" }
	}
}

# Decide everyone BEFORE applying any result. Temporary lists stay in this
# execution chain, so no persistent per-character score/decision survives it.
hd_hb_decide_ruler_effect = {
	# Culture scopes have no liege accessor. Save the character's direct liege
	# before the score's culture-heritage comparison changes scope.
	liege = { save_temporary_scope_as = hd_hb_round_liege }
	if = {
		limit = {
			hd_hb_force_independence_trigger = yes
			NOT = { hd_hb_block_independence_trigger = yes }
		}
		add_to_temporary_list = hd_hb_release_targets
	}
	else_if = {
		limit = {
			hd_hb_block_independence_trigger = yes
			NOT = { hd_hb_force_independence_trigger = yes }
		}
		# A D3 single-sided block also invalidates any unanswered D0 choice.
		hd_hb_clear_player_pending_effect = yes
	}
	else = {
		if = {
			limit = { is_ai = yes }
			random = {
				chance = hd_hb_independence_chance
				add_to_temporary_list = hd_hb_release_targets
			}
		}
		else_if = {
			limit = { NOT = { has_character_flag = hd_hb_choice_pending } }
			add_to_temporary_list = hd_hb_player_targets
		}
	}
}

# Current round's snapshot. The duchy batch goes first, followed by higher
# ranks, then counties. All outcomes already exist, including descendants of
# released parents: never re-filter them by top_liege or current Han membership.
hd_hb_apply_round_effect = {
	every_in_list = {
		list = hd_hb_release_targets
		limit = { highest_held_title_tier = tier_duchy }
		hd_qxzl_release_ruler_effect = yes
	}
	every_in_list = {
		list = hd_hb_release_targets
		limit = { highest_held_title_tier > tier_duchy }
		hd_qxzl_release_ruler_effect = yes
	}
	every_in_list = {
		list = hd_hb_release_targets
		limit = { highest_held_title_tier = tier_county }
		hd_qxzl_release_ruler_effect = yes
	}
	every_in_list = {
		list = hd_hb_player_targets
		limit = {
			hd_hb_live_vassal_trigger = yes
			NOT = { has_character_flag = hd_hb_choice_pending }
		}
		# The original direct liege can itself now be independent. It remains
		# the player's direct liege unless this player was already released.
		liege = { save_scope_as = hd_hb_choice_liege }
		set_variable = { name = hd_hb_choice_pending_liege value = scope:hd_hb_choice_liege }
		save_temporary_scope_as = hd_hb_pending_player
		title:h_china = {
			add_to_variable_list = { name = hd_hb_pending_players target = scope:hd_hb_pending_player }
		}
		add_character_flag = hd_hb_choice_pending
		trigger_event = hd_qxzl.0001
	}
}

# Stable public entry called only by the founded coalition's terminal cleanup.
# h_china owns timers: emperor death/deposition cannot cancel them.
hd_qxzl_begin_effect = {
	if = {
		limit = {
			exists = global_var:hd_anti_regent_chain_resolved
			NOT = { exists = global_var:hd_hb_started }
			exists = title:h_china.holder
		}
		set_global_variable = hd_hb_started
		# Keep the public start marker used by unrelated submission/person chains.
		set_global_variable = hd_qxzl_started
		set_global_variable = { name = hd_qxzl_start_date value = current_date }
		title:h_china = {
			if = { limit = { has_variable_list = hd_hb_duchy_titles } clear_variable_list = hd_hb_duchy_titles }
		}
		title:h_china.holder = {
			every_vassal_or_below = {
				limit = { hd_hb_live_vassal_trigger = yes }
				add_to_temporary_list = hd_hb_round_targets
				# Spatial tickets include each initial vassal's territorial titles,
				# so split inheritance and D0/D3 independent parents stay covered.
				every_held_title = {
					limit = { tier >= tier_county is_landless_type_title = no }
					save_temporary_scope_as = hd_hb_spatial_initial_title
					title:h_china = {
						add_to_variable_list = { name = hd_hb_spatial_titles target = scope:hd_hb_spatial_initial_title }
					}
				}
				if = {
					limit = { highest_held_title_tier = tier_duchy }
					# Capture every original territorial duchy, not just the primary:
					# splitting two duchies between heirs must not hide either heir.
					every_held_title = {
						limit = { tier = tier_duchy is_landless_type_title = no }
						save_temporary_scope_as = hd_hb_initial_duchy
						title:h_china = {
							add_to_variable_list = { name = hd_hb_duchy_titles target = scope:hd_hb_initial_duchy }
						}
					}
				}
			}
		}
		every_in_list = { list = hd_hb_round_targets hd_hb_decide_ruler_effect = yes }
		hd_hb_apply_round_effect = yes
		title:h_china = {
			trigger_event = { on_action = hd_hb_recheck_after_three_days days = 3 }
			trigger_event = { on_action = hd_hb_legacy_after_thirty_days days = 30 }
		}
		debug_log = "HD_HB D0_COMPLETED_D3_QUEUED"
	}
}

# Title scope h_china. Only the bounded initial duchy-title list is consumed.
# A successor/new direct liege is evaluated live; an already independent holder
# or an unheld title is skipped. Deduplicate holders before any new decisions.
hd_hb_recheck_duchies_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_hb_d3_done } }
		set_global_variable = hd_hb_d3_done
		if = {
			limit = { has_variable_list = hd_hb_duchy_titles }

				variable = hd_hb_duchy_titles
				holder ?= {
					if = {
						limit = {
							hd_hb_live_vassal_trigger = yes
							NOT = { is_in_list = hd_hb_round_targets }
						}
						add_to_temporary_list = hd_hb_round_targets
					}
				}
			}
			clear_variable_list = hd_hb_duchy_titles
		}
		every_in_list = { list = hd_hb_round_targets hd_hb_decide_ruler_effect = yes }
		hd_hb_apply_round_effect = yes
		debug_log = "HD_HB D3_COMPLETED"
		set_global_variable = hd_hb_d3_applied
		title:h_china = { hd_hb_maybe_spatial_effect = yes }
	}
}

# One additional spatial phase, after D3 AND all still-valid player choices.
# Territorial-title tickets are captured at D0 from vassals, never Han's direct domain.
# Timed retries below only wait for choices; they do not repeat spatial judgment.
hd_hb_clear_player_pending_effect = {
	remove_character_flag = hd_hb_choice_pending
	remove_variable = hd_hb_choice_pending_liege
}

# Title scope h_china. No global ruler scan; inheritance resolves territorial-title tickets.
hd_hb_maybe_spatial_effect = {
	if = {
		limit = {
			exists = global_var:hd_hb_d3_applied
			NOT = { exists = global_var:hd_hb_spatial_started }
		}
		# Character receipts survive death, inheritance and trading away the
		# initial counties. Connectivity tickets must not gate player choices.
		if = {
			limit = { has_variable_list = hd_hb_pending_players }
			every_in_list = {
				variable = hd_hb_pending_players
				if = {
					limit = { has_character_flag = hd_hb_choice_pending }
					if = {
						limit = {
							hd_hb_live_vassal_trigger = yes
							is_ai = no
							exists = var:hd_hb_choice_pending_liege
							liege = var:hd_hb_choice_pending_liege
						}
						add_to_temporary_list = hd_hb_spatial_waiting_players
					}
					else = { hd_hb_clear_player_pending_effect = yes }
				}
			}
		}
		if = {
			limit = { any_in_list = { list = hd_hb_spatial_waiting_players always = yes } }
			if = {
				limit = { NOT = { has_variable = hd_hb_spatial_wait_queued } }
				set_variable = hd_hb_spatial_wait_queued
				trigger_event = { on_action = hd_hb_spatial_wait_for_choices days = 1 }
			}
		}
		else = { hd_hb_spatial_cleanup_effect = yes }
	}
}

# Character scope. The actor scope is saved anew before each preflight/apply.
# Preserve native interaction legality; do not rely on an engine bypass.
# Character scope. Save the entire original tuple before any actor is released.
# Any connected COUNTY IN THE WHOLE SUB-REALM preserves this actor's whole realm.
# The reference capital is the original DIRECT liege's capital, even when tribute
# later falls back to the original independent top liege.
hd_hb_spatial_snapshot_actor_effect = {
	save_temporary_scope_as = hd_hb_spatial_actor
	if = {
		limit = {
			hd_hb_live_vassal_trigger = yes
			is_tributary = no
			exists = liege.capital_county
			exists = top_liege
			any_sub_realm_title = { tier = tier_county is_landless_type_title = no }
		}
		set_variable = { name = hd_hb_spatial_direct value = liege }
		set_variable = { name = hd_hb_spatial_top value = top_liege }
		set_variable = { name = hd_hb_spatial_capital value = liege.capital_county }
		if = {
			limit = { liege = { is_independent_ruler = yes } }
			set_variable = { name = hd_hb_spatial_suzerain value = liege }
		}
		else = { set_variable = { name = hd_hb_spatial_suzerain value = top_liege } }
		var:hd_hb_spatial_capital = { save_temporary_scope_as = hd_hb_spatial_capital }
		if = {
			limit = {
				NOT = {
					any_sub_realm_title = {
						tier = tier_county
						is_landless_type_title = no
						is_connected_to = { target = scope:hd_hb_spatial_capital }
					}
				}
			}
			if = {
				limit = { hd_hb_spatial_suzerain_valid_trigger = yes }
				add_to_temporary_list = hd_hb_spatial_release_targets
			}
			else = { debug_log = "HD_HB SPATIAL_SKIPPED_NO_LEGAL_SUZERAIN" }
		}
	}
}

# Restore the original direct relationship if the native tributary effect fails.
# This is a failure rollback only, using the mod's existing swear_fealty pattern.
hd_hb_spatial_rollback_effect = {
	if = {
		limit = {
			is_independent_ruler = yes
			is_tributary = no
			exists = var:hd_hb_spatial_direct
			var:hd_hb_spatial_direct = {
				is_alive = yes
				is_ruler = yes
				highest_held_title_tier > scope:hd_hb_spatial_actor.highest_held_title_tier
				NOT = { is_vassal_or_below_of = scope:hd_hb_spatial_actor }
			}
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = hd_hb_spatial_rollback
			add_claim_on_loss = no
		}
		change_liege = { liege = var:hd_hb_spatial_direct change = scope:hd_hb_spatial_rollback }
		resolve_title_and_vassal_change = scope:hd_hb_spatial_rollback
		if = {
			limit = { exists = liege liege = var:hd_hb_spatial_direct }
			debug_log = "HD_HB SPATIAL_TRIBUTE_FAILED_ORIGINAL_LIEGE_RESTORED"
		}
		else = { debug_log = "HD_HB SPATIAL_ROLLBACK_NOT_CONFIRMED" }
	}
	else = { debug_log = "HD_HB SPATIAL_TRIBUTE_FAILED_ROLLBACK_UNAVAILABLE" }
}

hd_hb_spatial_release_actor_effect = {
	save_temporary_scope_as = hd_hb_spatial_actor
	if = {
		limit = {
			hd_hb_live_vassal_trigger = yes
			is_tributary = no
			exists = var:hd_hb_spatial_direct
			liege = var:hd_hb_spatial_direct
			hd_hb_spatial_suzerain_valid_trigger = yes
		}
		hd_qxzl_release_ruler_effect = yes
		if = {
			limit = { is_independent_ruler = yes is_tributary = no }
			# Recheck after independence: native on_actions may invalidate the saved suzerain.
			if = {
				limit = { hd_hb_spatial_suzerain_valid_trigger = yes }
				# Contract validity is checked against the SAVED suzerain, not
				# the actor's now-self top_liege. No forced government conversion.
				if = {
					limit = {
						var:hd_hb_spatial_suzerain = {
							highest_held_title_tier = tier_hegemony
							has_government = celestial_government
						}
					}
					start_tributary = { contract_group = tributary_celestial suzerain = var:hd_hb_spatial_suzerain }
				}
				else = {
					# Generic existing contract has no celestial/hegemony condition.
					start_tributary = { contract_group = tributary_subjugated suzerain = var:hd_hb_spatial_suzerain }
				}
			}
			if = {
				limit = { exists = var:hd_hb_spatial_suzerain is_tributary_of = var:hd_hb_spatial_suzerain }
				debug_log = "HD_HB SPATIAL_TRIBUTE_CONFIRMED"
			}
			else = { hd_hb_spatial_rollback_effect = yes }
		}
	}
}

		limit = { NOT = { exists = global_var:hd_hb_spatial_started } }

		if = {
			limit = { has_variable_list = hd_hb_spatial_titles }

				variable = hd_hb_spatial_titles
				holder ?= {
					if = {
						limit = {
							hd_hb_live_vassal_trigger = yes
							NOT = { is_in_list = hd_hb_spatial_candidates }
						}
						add_to_temporary_list = hd_hb_spatial_candidates
					}
				}
			}
			clear_variable_list = hd_hb_spatial_titles
		}

		if = { limit = { has_variable_list = hd_hb_pending_players } clear_variable_list = hd_hb_pending_players }
		set_global_variable = hd_hb_spatial_done
		debug_log = "HD_HB SPATIAL_ONCE_COMPLETED"
	}
}

  while (i < ts.length && ts[i] !== '}') {
   let key = ts[i++], op = ts[i++], value = ts[i++];
   assert(['=', '?=', '>', '<', '>=', '<=', '!='].includes(op), `operator ${key}`);
   if (value === '{') value = block(true);

  }

 }
 const out = block(false); assert.equal(i, ts.length); return out;
}

function property(a, key) {
 if (a == null) return undefined;
 if (key === 'top_liege') {let z = a; const seen = new Set(); while (z.liege) {assert(!seen.has(z)); seen.add(z); z = z.liege;} return z;}
 if (key === 'top_suzerain') {let z = a.suz; const seen = new Set(); while (z?.suz) {assert(!seen.has(z)); seen.add(z); z = z.suz;} return z;}
 if (key === 'is_independent_ruler') return !a.liege;
 if (key === 'is_tributary') return !!a.suz;
 return a[key];
}
function scalar(s, a, c) {
 if (s in tiers) return tiers[s];
 if (s === 'yes' || s === 'no') return s === 'yes';
 if (/^-?\d+(\.\d+)?$/.test(s)) return Number(s);
 if (s.startsWith('flag:')) return s.slice(5);
 const parts = s.split('.'); let z = a;
 if (parts[0] === 'this') parts.shift();
 else if (parts[0].startsWith('scope:')) z = c.scopes[parts.shift().slice(6)];
 else if (parts[0].startsWith('global_var:')) z = c.globals[parts.shift().slice(11)];
 else if (parts[0].startsWith('title:')) z = c.titles[parts.shift().slice(6)];
 for (const part of parts) z = part.startsWith('var:') ? z?.vars?.[part.slice(4)] : property(z, part);
 return z;
}
function compare(x, op, y) {
 return ({'=':()=>x===y, '?=':()=>x===y, '!=':()=>x!==y, '>':()=>x>y, '<':()=>x<y, '>=':()=>x>=y, '<=':()=>x<=y})[op]();
}
function cond(ns, a, c) {
 return ns.every(n => {
  const {key:k, value:v, op} = n;
  if (k === 'OR') return v.some(z => cond([z], a, c));
  if (k === 'AND') return cond(v, a, c);
  if (k === 'NOT') return !cond(v, a, c);
  if (k === 'trigger_if') {const l = v.find(z => z.key === 'limit'); return !cond(l.value, a, c) || cond(v.filter(z => z.key !== 'limit'), a, c);}
  if (k === 'always') return v === 'yes';
  if (k === 'exists') {const z = scalar(v, a, c); return z !== undefined && z !== null;}
  if (k === 'has_variable') return Object.hasOwn(a.vars, v);
  if (k === 'has_variable_list') return Array.isArray(a.vars[v]);
  if (k === 'has_character_flag') return a.flags.has(v);
  if (k === 'has_trait') return a.traits.has(v);
  if (k.startsWith('has_relation_')) return a.relations.has(k.slice(13));
  if (k === 'is_obedient_to') return a.obedient;
  if (k === 'opinion') return cond(v.filter(z => z.key === 'value'), {value:a.opinion}, c);
  if (k === 'has_realm_law_flag') return a.laws.has(v);
  if (k === 'government_has_flag') return a.gflags.has(v);
  if (k === 'has_government') return a.gov === v;
  if (k === 'is_tributary_of') return a.suz === scalar(v, a, c);
  if (k === 'is_in_list') return (c.lists[v] || []).includes(a);
  if (k === 'is_vassal_or_below_of') {const target = scalar(v, a, c); let z = a.liege; while (z) {if (z === target) return true; z = z.liege;} return false;}
  if (k === 'any_in_list') {const name = v.find(z => z.key === 'list').value; return (c.lists[name] || []).some(z => cond(v.filter(q => q.key !== 'list'), z, c));}
  if (k === 'any_sub_realm_title') return a.counties.some(z => cond(v, z, c));
  if (k === 'is_connected_to') return a.component === scalar(v.find(z => z.key === 'target').value, a, c).component;
  if (defs[k]) return cond(defs[k], a, c) === (v === 'yes');
  if (Array.isArray(v)) {const z = scalar(k, a, c); return !!z && cond(v, z, c);}
  return compare(scalar(k, a, c), op, scalar(v, a, c));
 });
}
function exec(ns, a, c) {
 let selected = false;
 for (const n of ns) {
  const {key:k, value:v} = n;
  if (['if','else_if','else'].includes(k)) {
   if (k === 'if') selected = false;
   const l = v.find(z => z.key === 'limit');
   if (!selected && (!l || cond(l.value, a, c))) {selected = true; exec(v.filter(z => z.key !== 'limit'), a, c);} continue;
  }
  if (k === 'every_in_list') {
   const vn = v.find(z => z.key === 'variable'), ln = v.find(z => z.key === 'list'), l = v.find(z => z.key === 'limit');
   const list = vn ? a.vars[vn.value] : c.lists[ln.value];
   for (const z of [...(list || [])]) if (!l || cond(l.value, z, c)) exec(v.filter(q => !['variable','list','limit'].includes(q.key)), z, c);
   continue;
  }
  if (k === 'add_to_temporary_list') {c.lists[v] ??= []; if (!c.lists[v].includes(a)) c.lists[v].push(a); continue;}
  if (k === 'save_temporary_scope_as' || k === 'save_scope_as') {c.scopes[v] = a; continue;}
  if (k === 'set_variable') {if (Array.isArray(v)) a.vars[v.find(z => z.key === 'name').value] = scalar(v.find(z => z.key === 'value').value, a, c); else a.vars[v] = true; continue;}
  if (k === 'remove_variable' || k === 'clear_variable_list') {delete a.vars[v]; continue;}
  if (k === 'add_character_flag') {a.flags.add(v); continue;}
  if (k === 'remove_character_flag') {a.flags.delete(v); continue;}
  if (k === 'set_global_variable') {c.globals[v] = true; continue;}
  if (k === 'trigger_event') {c.queue.push({action:v.find(z => z.key === 'on_action')?.value, days:v.find(z => z.key === 'days')?.value}); continue;}
  if (k === 'create_title_and_vassal_change') {c.scopes[v.find(z => z.key === 'save_scope_as').value] = {}; continue;}
  if (k === 'becomes_independent') {
   a.liege = null; c.releases.push(a.id);
   if (a.killSuzOnRelease) a.vars.hd_hb_spatial_suzerain.is_alive = false;
   if (a.removeSuzLawOnRelease) a.vars.hd_hb_spatial_suzerain.laws.clear();
   continue;
  }
  if (k === 'resolve_title_and_vassal_change') continue;
  if (k === 'start_tributary') {
   const suz = scalar(v.find(z => z.key === 'suzerain').value, a, c);
   assert(!a.liege); assert(suz && suz !== a && suz.is_alive && !suz.liege && suz.laws.has('can_have_tributaries'));
   const group = v.find(z => z.key === 'contract_group').value;
   if (group === 'tributary_celestial') assert(suz.gov === 'celestial_government' && suz.highest_held_title_tier === 6);
   if (!a.tributeFails) a.suz = suz; c.tributes.push([a.id, suz.id, group]); continue;
  }
  if (k === 'change_liege') {a.liege = scalar(v.find(z => z.key === 'liege').value, a, c); continue;}
  if (k === 'debug_log') {c.logs.push(v); continue;}
  if (defs[k]) {exec(defs[k], a, c); continue;}
  if (Array.isArray(v)) {const z = scalar(k, a, c); if (n.op === '?=' && !z) continue; assert(z, `invalid scope ${k}`); exec(v, z, c); continue;}
  throw new Error(`unsupported effect ${k}`);
 }
}
