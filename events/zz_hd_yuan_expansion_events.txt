namespace = hd_yuan_expansion

# Manual repair entry: event hd_yuan_expansion.0000. It does not grant land,
# reset completed rounds, force an immediate regional war or touch Cao state.
hd_yuan_expansion.0000 = {
	type = character_event
	hidden = yes
	immediate = {
		character:yuan_shao = {
			hd_yuan_expansion_ensure_heartbeat_effect = yes
			hd_yuan_expansion_bootstrap_effect = yes
		}
	}
}

# One menu for every normal round-end, not one menu per lifetime.
hd_yuan_expansion.1000 = {
	type = character_event
	title = hd_yuan_expansion.1000.t
	desc = hd_yuan_expansion.1000.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }
	trigger = {
		this = character:yuan_shao
		is_alive = yes
		has_character_flag = hd_yuan_expansion_menu_pending
		NOT = { has_character_flag = hd_yuan_expansion_menu_open }
	}
	immediate = {
		remove_character_flag = hd_yuan_expansion_menu_queued
		add_character_flag = hd_yuan_expansion_menu_open

	option = {
		name = hd_yuan_expansion.1000.a

		ai_chance = {
			base = 100
			modifier = { add = 100 hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_jizhou } }
			# Avoid selecting a strategically empty direction when another is viable.
			modifier = {
				factor = 0
				NOT = { hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_jizhou } }
				OR = {
					hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_youzhou }
					hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_qingzhou }
				}
			}
		}
	}
	option = {
		name = hd_yuan_expansion.1000.b
		trigger = { NOT = { hd_yuan_expansion_controls_region_trigger = { REGION = title:k_uuii_youzhou } } }

		ai_chance = {
			base = 80
			modifier = { add = 100 hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_youzhou } }
			# Avoid selecting a strategically empty direction when another is viable.
			modifier = {
				factor = 0
				NOT = { hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_youzhou } }
				OR = {
					hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_jizhou }
					hd_yuan_expansion_region_has_target_trigger = { REGION = title:k_uuii_qingzhou }
				}
			}
		}
	}
	option = {
		name = hd_yuan_expansion.1000.c
		trigger = { NOT = { hd_yuan_expansion_controls_region_trigger = { REGION = title:k_uuii_qingzhou } } }
