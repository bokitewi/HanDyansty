WJ_yinshi.3001 = {
	override_background = { reference = courtyard }
	type = character_event
	title = WJ_yinshi.3001.t
	desc = WJ_yinshi.3001.desc
	theme = diplomacy
	left_portrait = { character = root animation = thinking }

	trigger = {
		WJ_yinshi_able_trigger = yes
		NOT = {
			has_character_flag = wj_yinshi_ev_3001
		}
	}
	immediate = {
		save_scope_as = wj_yinshi_subject
		add_character_flag = {
			flag = wj_yinshi_ev_3001
			years = 10
		}
		location ?= {
			county ?= {
				save_scope_as = wj_yinshi_scene_county
			}
		}
	}
	option = {
		name = WJ_yinshi.3001.a
		ai_chance = {
			base = 50
			modifier = {
				add = 20
				has_trait = content
			}
			modifier = {
				add = 20
				has_trait = craven
			}
		}
		trigger = {
			WJ_yinshi_scene_valid_trigger = yes
		}
		if = {
			limit = {
				WJ_yinshi_scene_valid_trigger = yes
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = WJ_yinshi.3001.safe_result
				left_icon = root
				change_merit = 25
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = WJ_yinshi_scene_lapsed
				left_icon = root
				custom_tooltip = WJ_yinshi_scene_lapsed_tt
			}
		}
	}
	option = {
		name = WJ_yinshi.3001.b
		ai_chance = {
			base = 50
			modifier = {
				add = 20
				has_trait = ambitious
			}
			modifier = {
				add = 20
				has_trait = brave
			}
		}
		trigger = {
			WJ_yinshi_scene_valid_trigger = yes
			WJ_yinshi_scene_local_valid_trigger = yes
		}
		if = {
			limit = {
				WJ_yinshi_scene_valid_trigger = yes
				WJ_yinshi_scene_local_valid_trigger = yes
			}
			random_list = {
				0 = {
					modifier = {
						add = WJ_yinshi_chance_stewardship_value
						always = yes
					}
					desc = WJ_yinshi.3001.success_result
					send_interface_toast = {
						type = event_toast_effect_good
						title = WJ_yinshi.3001.success_result
						left_icon = root
						change_merit = 75
						add_stewardship_lifestyle_xp = 200
						scope:wj_yinshi_scene_county ?= {
							if = {
								limit = {
									has_county_modifier = WJ_yinshi_local_accord_modifier
								}
								remove_county_modifier = WJ_yinshi_local_accord_modifier
							}
							add_county_modifier = {
								modifier = WJ_yinshi_local_accord_modifier
								years = 5
							}
						}
					}
				}
				0 = {
					modifier = {
						add = WJ_yinshi_fail_stewardship_value
						always = yes
					}
					desc = WJ_yinshi.3001.failure_result
					send_interface_toast = {
						type = event_toast_effect_bad
						title = WJ_yinshi.3001.failure_result
						left_icon = root
						add_prestige = -30
					}
				}
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = WJ_yinshi_scene_lapsed
				left_icon = root
				custom_tooltip = WJ_yinshi_scene_lapsed_tt
			}
		}
	}
	option = {
		name = WJ_yinshi.3001.c
		trigger = {
			has_trait = just
			WJ_yinshi_scene_valid_trigger = yes
		}
		trait = just
		ai_chance = {
			base = 100
		}
		if = {
			limit = {
				has_trait = just
				WJ_yinshi_scene_valid_trigger = yes
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = WJ_yinshi.3001.special_result
				left_icon = root
				change_merit = 50
				add_stewardship_lifestyle_xp = 200
				add_stress = -10
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = WJ_yinshi_scene_lapsed
				left_icon = root
				custom_tooltip = WJ_yinshi_scene_lapsed_tt
			}
		}
	}
	option = {
		name = WJ_yinshi_scene_close
		trigger = {

			}
		}

			type = event_toast_effect_neutral

			left_icon = root
			custom_tooltip = WJ_yinshi_scene_lapsed_tt
