# 拉票：为本人所投之人拉票（每轮一次）
WJ_qingyi.3102 = {
	type = activity_event
	title = WJ_qingyi.3102.t
	desc = WJ_qingyi.3102.desc
	theme = feast_activity
	left_portrait = { character = scope:wj_yd_target animation = admiration }
	right_portrait = { character = root animation = personality_bold }

	trigger = {
		is_ai = no
		has_variable = wj_yd_voted_for
	}

	immediate = {
		var:wj_yd_voted_for = { save_scope_as = wj_yd_target }
	}

	# 花威望 75：+2 票
	option = {
		name = WJ_qingyi.3102.a
		trigger = { prestige >= minor_prestige_value }
		add_prestige = { value = minor_prestige_value multiply = -1 }
		WJ_qingyi_yuedan_canvass_add_effect = { VOTES = 2 }
		custom_tooltip = WJ_qingyi.3102.plus2_tt
	}

	# 花影响力 30：+2 票
	option = {
		name = WJ_qingyi.3102.b
		trigger = {
			government_has_flag = government_has_influence
			influence >= minor_influence_value
		}
		change_influence = { value = minor_influence_value multiply = -1 }
		WJ_qingyi_yuedan_canvass_add_effect = { VOTES = 2 }
		custom_tooltip = WJ_qingyi.3102.plus2_tt
	}

	# 外交辩论：成功 +3，失败 -1
	option = {
		name = WJ_qingyi.3102.c
		duel = {
			skill = diplomacy
			value = WJ_qingyi_yd_duel_target
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				min = 5
				desc = WJ_qingyi.3102.c.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = WJ_qingyi.3102.c.success
					left_icon = scope:wj_yd_target
					WJ_qingyi_yuedan_canvass_add_effect = { VOTES = 3 }
					custom_tooltip = WJ_qingyi.3102.plus3_tt
				}
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
				min = 5
				desc = WJ_qingyi.3102.c.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = WJ_qingyi.3102.c.failure
					left_icon = scope:wj_yd_target
					WJ_qingyi_yuedan_canvass_add_effect = { VOTES = -1 }
					custom_tooltip = WJ_qingyi.3102.minus1_tt
				}
			}
		}
	}

	option = {
		name = WJ_qingyi.3102.d
	}
}
