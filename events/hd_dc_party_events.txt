hd_dc_par.1201 = {
	type = character_event
	title = hd_dc_par.1201.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:hd_dc_policy_trial.var:hd_dc_policy_result = flag:fulfilled }
				desc = hd_dc_par.1201.fulfilled
			}
			triggered_desc = {
				trigger = { scope:hd_dc_policy_trial.var:hd_dc_policy_result = flag:authority_lost }
				desc = hd_dc_par.1201.authority_lost
			}
			desc = hd_dc_par.1201.interrupted
		}
	}
	theme = stewardship
	trigger = {
		hd_dc_policy_notice_recipient_trigger = yes
		scope:hd_dc_policy_trial = { has_variable = hd_dc_policy_result }
	}
	option = {
		name = hd_dc_par.1201.a
		if = {
			limit = { scope:hd_dc_policy_trial.var:hd_dc_policy_payment = { var:hd_dc_q_committed = 5 } }
			custom_tooltip = hd_dc_par.1201.q.tt
		}
		else = { custom_tooltip = hd_dc_par.1201.no_q.tt }
	}
}
