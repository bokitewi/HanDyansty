############################
# 4101 起义求援
# Scope：受邀的汉人政权。须有 scope:hd_yiguan_qy_rebel（起义者）、scope:hd_yiguan_qy_def（胡人领主），由 hd_yiguan_qiyi_start_effect 存
# 战争可能在作答前已经结束，所以不用存下的战争作用域，改为从起义者身上重找
############################
hd_yiguan.4101 = {
	type = character_event
	title = hd_yiguan.4101.t
	desc = hd_yiguan.4101.desc
	theme = war
	override_background = { reference = army_camp }
	left_portrait = {
		character = scope:hd_yiguan_qy_rebel
		animation = war_attacker
	}
	right_portrait = {
		character = scope:hd_yiguan_qy_def
		animation = anger
	}

	trigger = {
		exists = scope:hd_yiguan_qy_rebel
		exists = scope:hd_yiguan_qy_def
		scope:hd_yiguan_qy_rebel = {
			is_alive = yes
			any_character_war = {
				using_cb = hd_yiguan_qiyi_cb
				primary_attacker = scope:hd_yiguan_qy_rebel
			}
		}
	}

	immediate = {
		scope:hd_yiguan_qy_rebel = {
			random_character_war = {
				limit = {
					using_cb = hd_yiguan_qiyi_cb
					primary_attacker = scope:hd_yiguan_qy_rebel
				}
				save_scope_as = hd_yiguan_qy_cur_war
			}
		}
	}

	option = {	# 出兵相助
		name = hd_yiguan.4101.a
		trigger = {
			exists = scope:hd_yiguan_qy_cur_war
			NOT = { is_allied_to = scope:hd_yiguan_qy_def }
			NOT = { is_at_war_with = scope:hd_yiguan_qy_rebel }
			NOT = {
				scope:hd_yiguan_qy_cur_war = { is_participant = root }
			}
		}
		custom_tooltip = hd_yiguan.4101.a.tt
		hidden_effect = {
			scope:hd_yiguan_qy_cur_war ?= { add_attacker = root }	# 先例：原版 07_ep3_wars.txt:139
		}
		add_prestige = minor_prestige_gain
		scope:hd_yiguan_qy_rebel = {
			add_opinion = {	# 原版通用好感"在我需要时给予过帮助"（+30，10 年递减）；资助北还好感 hd_yiguan_funded_opinion 专指北还，不借用
				target = root
				modifier = helped_me_in_time_of_need
			}
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_good
					title = hd_yiguan_qiyi_helper_joined_toast
					left_icon = root
					custom_tooltip = hd_yiguan_qiyi_helper_joined_toast_desc
				}
			}
		}
		ai_chance = {
			base = 25
			modifier = {
				add = 30
				opinion = {
					target = scope:hd_yiguan_qy_rebel
					value >= 20
				}
			}
			modifier = {
				add = 20
				opinion = {
					target = scope:hd_yiguan_qy_def
					value <= -20
				}
			}
			modifier = {
				add = -30
				opinion = {
					target = scope:hd_yiguan_qy_rebel
					value < 0
				}
			}
			modifier = {	# 自顾不暇
				add = -20
				is_at_war = yes
			}
			ai_value_modifier = {
				ai_boldness = 0.3
				ai_honor = 0.2
			}
		}
	}

	option = {	# 坐观成败
		name = hd_yiguan.4101.b
		scope:hd_yiguan_qy_rebel = {
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = hd_yiguan_qiyi_helper_declined_toast
					left_icon = root
					custom_tooltip = hd_yiguan_qiyi_helper_declined_toast_desc
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				add = 20
				has_trait = craven
			}
			ai_value_modifier = {
				ai_boldness = -0.3
			}
		}
	}
}
