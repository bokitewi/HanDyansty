# 议伐吴：279.11 起天子为玩家时。Root：天子
hd263.1000 = {
	type = character_event
	title = hd263.1000.t
	desc = hd263.1000.desc
	theme = war
	left_portrait = { character = root animation = war_attacker }
	right_portrait = {
		character = title:e_jiangdong.holder
		animation = worry
	}
	trigger = {
		hd263_miewu_ready_trigger = yes
		this = title:h_china.holder
	}
	# 大举伐吴
	option = {
		name = hd263.1000.a
		hd263_miewu_start_effect = yes
		ai_chance = { base = 100 }
	}
	# 暂缓，一年后再议
	option = {
		name = hd263.1000.b
		set_global_variable = { name = hd263_miewu_wait value = yes days = 365 }
		if = {
			limit = { exists = global_var:hd263_miewu_asked }
			remove_global_variable = hd263_miewu_asked
		}
		ai_chance = { base = 0 }
	}
}
