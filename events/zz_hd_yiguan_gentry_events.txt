# 目的地收到投奔请求。root = 目的地；scope:hd_yiguan_gt_migrant 迁徙者，scope:hd_yiguan_gt_dest = root
hd_yiguan.4001 = {
	type = character_event
	title = hd_yiguan.4001.t
	desc = hd_yiguan.4001.desc
	theme = diplomacy
	override_background = { reference = tgp_asia_throne_room }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:hd_yiguan_gt_migrant
		animation = worry
	}
	trigger = {
		exists = scope:hd_yiguan_gt_migrant
		scope:hd_yiguan_gt_migrant = { is_alive = yes }
	}
	on_trigger_fail = {	# 先例 原版 coronation_events.txt:3942
		scope:hd_yiguan_gt_migrant ?= {
			if = {
				limit = { is_alive = yes }
				trigger_event = hd_yiguan.4004
			}
		}
	}

	option = {	# 纳之
		name = hd_yiguan.4001.a
		custom_tooltip = hd_yiguan.4001.a.tt
		if = {
			limit = {
				scope:hd_yiguan_gt_migrant = { hd_yiguan_gt_nanqian_still_valid_trigger = yes }
			}
			scope:hd_yiguan_gt_migrant = { trigger_event = hd_yiguan.4002 }
		}
		else = {
			scope:hd_yiguan_gt_migrant = { trigger_event = hd_yiguan.4004 }
		}
		ai_chance = {
			base = 50
			opinion_modifier = {	# 写法见原版 coronation_events_1.txt:1203
				opinion_target = scope:hd_yiguan_gt_migrant
				multiplier = 0.5
			}
			modifier = {	# 才具（§6 第 18 条：本人能力）
				add = 15
				scope:hd_yiguan_gt_migrant = {
					OR = {
						diplomacy >= 14
						stewardship >= 14
						learning >= 14
						martial >= 14
					}
				}
			}
			modifier = {	# 天朝制得一世族封臣
				add = 15
				government_has_flag = government_is_celestial
				scope:hd_yiguan_gt_migrant = { any_held_title = { is_noble_family_title = yes } }
			}
			modifier = {	# 宿敌
				add = -100
				OR = {
					has_relation_rival = scope:hd_yiguan_gt_migrant
					has_relation_nemesis = scope:hd_yiguan_gt_migrant
				}
			}
			ai_value_modifier = {
				ai_compassion = 0.3
			}
		}
	}
	option = {	# 婉拒
		name = hd_yiguan.4001.b
		custom_tooltip = hd_yiguan.4001.b.tt
		scope:hd_yiguan_gt_migrant = { trigger_event = hd_yiguan.4003 }
		ai_chance = {
			base = 20
			ai_value_modifier = {
				ai_compassion = -0.3
				ai_greed = 0.2
			}
		}
	}
}

# AI 流亡者起兵复核（隐藏）。受资助后发起；一年一查，至多 10 次
hd_yiguan.4015 = {
	type = character_event
	hidden = yes
	trigger = {
		is_ai = yes
		hd_yiguan_gt_exile_base_trigger = yes
	}
	immediate = {
		remove_variable = hd_yiguan_gt_ai_chk
		if = {
			limit = {
				hd_yiguan_gt_beihuan_can_start_trigger = yes
				prestige >= medium_prestige_value
			}
			random = {
				chance = hd_yiguan_gt_beihuan_ai_chance_value	# 具名数值作几率，先例 原版 00_bastard_effects.txt:410
				add_prestige = {
					value = medium_prestige_value
					multiply = -1
				}
				hd_yiguan_gt_beihuan_begin_effect = yes
			}
		}
		if = {
			limit = {
				NOT = { has_variable = hd_yiguan_gt_beihuan_done }
				NOT = { has_variable = hd_yiguan_gt_bh_paid }
				OR = {
					NOT = { has_variable = hd_yiguan_gt_ai_chk_n }
					var:hd_yiguan_gt_ai_chk_n < 10
				}
			}
			if = {
				limit = { NOT = { has_variable = hd_yiguan_gt_ai_chk_n } }
				set_variable = { name = hd_yiguan_gt_ai_chk_n value = 0 }
			}
			change_variable = { name = hd_yiguan_gt_ai_chk_n add = 1 }
			set_variable = { name = hd_yiguan_gt_ai_chk value = yes days = 400 }
			trigger_event = {
				id = hd_yiguan.4015
				days = 365
			}
		}
	}
}

# 北还功成·归附之议。root = 已得地独立的流亡者（C 的 hd_yiguan_beihuan_victory_effect 3 天后发来）
# 有仍合格的资助者才发生；无资助者即已独立，不另作选择（§6 第 21 条）
hd_yiguan.4021 = {
	type = character_event
	title = hd_yiguan.4021.t
	desc = hd_yiguan.4021.desc
	theme = war
	override_background = { reference = tgp_asia_throne_room }
	left_portrait = {
		character = root
		animation = war_over_win
	}
	right_portrait = {
		character = scope:hd_yiguan_gt_patron
		animation = thinking
	}
	trigger = {
		is_alive = yes
		is_landed = yes
		is_independent_ruler = yes
		has_variable_list = hd_yiguan_gt_funders
		any_in_list = {	# 先例 原版 00_scripted_triggers.txt:507
			variable = hd_yiguan_gt_funders
			hd_yiguan_gt_patron_valid_trigger = { EXILE = root }
		}
	}
	immediate = {
		hd_yiguan_gt_pick_patron_effect = yes
	}

	option = {	# 奉表来归（玩家可能把事件开着不点，资助者须此刻仍合格；失格时灰显，只能据地自立）
		name = hd_yiguan.4021.a
		trigger = {
			scope:hd_yiguan_gt_patron ?= { hd_yiguan_gt_patron_valid_trigger = { EXILE = root } }
		}
		show_as_unavailable = { exists = scope:hd_yiguan_gt_patron }	# 写法同原版 chariot_ongoing_events_jp.txt:992
		hd_yiguan_gt_submit_to_patron_effect = yes
		hidden_effect = { hd_yiguan_gt_clear_funding_effect = yes }
		ai_chance = {
			base = 50
			opinion_modifier = {
				opinion_target = scope:hd_yiguan_gt_patron
				multiplier = 0.5
			}
			modifier = {
				add = 20
				has_trait = content
			}
			modifier = {
				add = 15
				has_trait = humble
			}
			modifier = {
				add = 10
				has_trait = just
			}
			modifier = {
				add = -30
				has_trait = ambitious
			}
			modifier = {
				add = -20
				has_trait = arrogant
			}
		}
	}
	option = {	# 据地自立（资助者已失格时也走这里，不记"负资不归"）
		name = hd_yiguan.4021.b
		scope:hd_yiguan_gt_patron ?= {
			if = {
				limit = { hd_yiguan_gt_patron_valid_trigger = { EXILE = root } }
				add_opinion = {
					target = root
					modifier = hd_yiguan_gt_spurned_opinion
				}
				trigger_event = hd_yiguan.4023
			}
		}
		add_prestige = minor_prestige_gain
		hidden_effect = { hd_yiguan_gt_clear_funding_effect = yes }
		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = ambitious
			}
			modifier = {
				add = 20
				has_trait = arrogant
			}
			ai_value_modifier = {
				ai_boldness = 0.3
			}
		}
	}
}
