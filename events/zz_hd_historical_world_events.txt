namespace = hd_historical_world

# Guandu: Cao chooses whether the world opportunity becomes a real war.
hd_historical_world.0100 = {
	type = character_event
	title = hd_historical_world.0100.t
	desc = hd_historical_world.0100.desc
	theme = war
	right_portrait = { character = character:yuan_shao }

	option = {
		name = hd_guandu_decide_battle
		ai_chance = { base = 78 modifier = { add = 12 has_trait = brave } modifier = { add = -20 has_trait = craven } }
		character:yuan_shao = {
			if = {
				limit = { title:d_yinchuan.holder.top_liege = root }
				start_war = { cb = hd_historical_war_cb target = root target_title = title:d_yinchuan }
			}
			else_if = {
				limit = { title:d_chenliu_yuzhou.holder.top_liege = root }
				start_war = { cb = hd_historical_war_cb target = root target_title = title:d_chenliu_yuzhou }
			}
			random_character_war = {
				limit = { primary_attacker = character:yuan_shao primary_defender = root using_cb = hd_historical_war_cb }
				set_variable = { name = hd_world_war_kind value = flag:guandu }
				save_scope_as = hd_guandu_war
			}
		}
		if = {
			limit = { exists = scope:hd_guandu_war }
			set_global_variable = hd_guandu_world_active
			remove_global_variable = hd_guandu_world_open
			if = {
				limit = {
					character:xu_you = { is_alive = yes is_imprisoned = no }
					character:xu_you.top_liege = character:yuan_shao
				}
				scope:hd_guandu_war = { set_variable = hd_guandu_xu_you_defection_available }
				character:yuan_shao = { add_character_modifier = { modifier = hd_guandu_xu_you_wuchao_modifier years = 2 } }
			}
			else = {
				scope:hd_guandu_war = { set_variable = hd_guandu_scout_fallback }
				character:yuan_shao = { add_character_modifier = { modifier = hd_guandu_scout_wuchao_modifier years = 2 } }
			}
		}
		else = {
			set_global_variable = hd_guandu_world_resolved
			set_global_variable = hd_chain_terminal_alternate_failure
			remove_global_variable = hd_guandu_world_open
		}
	}
	option = {
		name = hd_guandu_defend_yellow_river
		ai_chance = { base = 12 modifier = { add = 15 has_trait = patient } }
		add_character_flag = hd_guandu_yellow_river_defense
		set_global_variable = hd_guandu_world_resolved
		set_global_variable = hd_chain_terminal_alternate_success
		remove_global_variable = hd_guandu_world_open
	}
	option = {
		name = hd_guandu_compromise
		ai_chance = { base = 4 modifier = { add = 12 has_trait = craven } }
		add_prestige = medium_prestige_loss
		set_global_variable = hd_guandu_world_resolved
		set_global_variable = hd_chain_terminal_nonhistorical
		remove_global_variable = hd_guandu_world_open
	}
	option = {
		name = hd_guandu_clear_other_enemies
		ai_chance = { base = 6 modifier = { add = 12 has_trait = cautious_leader } }
		add_character_flag = hd_guandu_delayed_for_other_enemies
		set_global_variable = hd_guandu_world_resolved
		set_global_variable = hd_chain_terminal_alternate_failure
		remove_global_variable = hd_guandu_world_open
	}
}

# Jingzhou is resolved through its current holder, not through a date-forced Liu Cong result.
hd_historical_world.0210 = {
	type = character_event
	title = hd_historical_world.0210.t
	desc = hd_historical_world.0210.desc
	theme = diplomacy_family_focus
	right_portrait = { character = character:cao_cao }

	option = {
		name = hd_chibi_jingzhou_surrender
		ai_chance = {
			base = 68
			modifier = { add = 18 this = character:liu_cong_2 }
			modifier = { add = -25 has_trait = brave }
			modifier = { add = 15 has_trait = craven }
		}
		create_title_and_vassal_change = { type = conquest save_scope_as = hd_chibi_jingzhou_change }
		title:k_uuii_jingzhou = {
			change_title_holder_include_vassals = {
				holder = character:cao_cao
				change = scope:hd_chibi_jingzhou_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_chibi_jingzhou_change
		set_global_variable = hd_chibi_jingzhou_surrendered
		remove_global_variable = hd_chibi_risk_jingzhou_loyalty
		character:cao_cao = { trigger_event = hd_historical_world.0200 }
	}
	option = {
		name = hd_chibi_jingzhou_resist
		ai_chance = { base = 32 modifier = { add = 25 has_trait = brave } }
		set_global_variable = hd_chibi_jingzhou_resisted
		character:cao_cao = { trigger_event = hd_historical_world.0200 }
	}
}

# Chibi: alliance and southern-expedition hard gates were checked before this event.
hd_historical_world.0200 = {
	type = character_event
	title = hd_historical_world.0200.t
	desc = hd_historical_world.0200.desc
	theme = war
	right_portrait = { character = character:sun_quan }

	option = {
		name = hd_chibi_continue_southeast
		ai_chance = { base = 76 modifier = { add = 12 has_trait = ambitious } }
		character:cao_cao = {
			if = { limit = { exists = global_var:hd_chibi_risk_disease } add_character_modifier = { modifier = hd_chibi_risk_disease_modifier years = 3 } }
			if = { limit = { exists = global_var:hd_chibi_risk_expedition } add_character_modifier = { modifier = hd_chibi_risk_expedition_modifier years = 3 } }
			if = { limit = { exists = global_var:hd_chibi_risk_naval } add_character_modifier = { modifier = hd_chibi_risk_naval_modifier years = 3 } }
			if = { limit = { exists = global_var:hd_chibi_risk_jingzhou_loyalty } add_character_modifier = { modifier = hd_chibi_risk_jingzhou_loyalty_modifier years = 3 } }
			if = { limit = { exists = global_var:hd_chibi_risk_tied_ships } add_character_modifier = { modifier = hd_chibi_risk_tied_ships_modifier years = 3 } }
			if = { limit = { exists = global_var:hd_chibi_risk_logistics } add_character_modifier = { modifier = hd_chibi_risk_logistics_modifier years = 3 } }
			if = { limit = { exists = global_var:hd_chibi_risk_huang_gai } add_character_modifier = { modifier = hd_chibi_risk_huang_gai_modifier years = 3 } }
			if = {
				limit = { title:d_danyang.holder.top_liege = character:sun_quan }
				start_war = { cb = hd_historical_war_cb target = character:sun_quan target_title = title:d_danyang }
			}
			else_if = {
				limit = { title:d_yuzhang.holder.top_liege = character:sun_quan }
				start_war = { cb = hd_historical_war_cb target = character:sun_quan target_title = title:d_yuzhang }
			}
			random_character_war = {
				limit = { primary_attacker = character:cao_cao primary_defender = character:sun_quan using_cb = hd_historical_war_cb }
				set_variable = { name = hd_world_war_kind value = flag:chibi }
				save_scope_as = hd_chibi_war
			}
		}
		if = {
			limit = { exists = scope:hd_chibi_war }
			set_global_variable = hd_chibi_world_active
			remove_global_variable = hd_chibi_world_open
		}
		else = {
			hd_clear_chibi_risks_effect = yes
			set_global_variable = hd_chibi_world_resolved
			set_global_variable = hd_chain_terminal_alternate_failure
			remove_global_variable = hd_chibi_world_open
		}
	}
	option = {
		name = hd_chibi_prepare_fleet
		ai_chance = { base = 14 modifier = { add = 15 has_trait = patient } }
		remove_global_variable = hd_chibi_risk_naval
		remove_global_variable = hd_chibi_risk_tied_ships
		remove_global_variable = hd_chibi_risk_logistics
		trigger_event = { id = hd_historical_world.0200 days = 30 }
	}
	option = {
		name = hd_chibi_withdraw
		ai_chance = { base = 6 modifier = { add = 15 has_trait = cautious_leader } }
		hd_clear_chibi_risks_effect = yes
		set_global_variable = hd_chibi_world_resolved
		set_global_variable = hd_chain_terminal_alternate_success
		remove_global_variable = hd_chibi_world_open
	}
	option = {
		name = hd_chibi_hold_jingzhou
		ai_chance = { base = 4 }
		remove_global_variable = hd_chibi_risk_expedition
		set_global_variable = hd_chibi_world_resolved
		set_global_variable = hd_chain_terminal_nonhistorical
		remove_global_variable = hd_chibi_world_open
	}
}
