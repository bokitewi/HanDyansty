# 拒纳作乱（隐藏判定）：苏峻型一半、其余一成半起兵（守方是拒纳者本人，不算沦陷区起义）
hd_yiguan.1110 = {
	type = character_event
	hidden = yes
	trigger = {
		is_landed = yes
		has_variable = hd_yiguan_spurned_shuai
		var:hd_yiguan_spurned_shuai = {
			is_alive = yes
			is_landed = no
			is_imprisoned = no
		}
	}
	immediate = {
		var:hd_yiguan_spurned_shuai = { save_scope_as = hd_yiguan_shuai }
		remove_variable = hd_yiguan_spurned_shuai
		random = {
			chance = 15
			modifier = {
				add = 35
				scope:hd_yiguan_shuai = { has_trait = arrogant }
			}
			clear_saved_scope = hd_yiguan_revolt_county
			random_realm_county = {
				limit = { NOT = { this = root.capital_county } }
				save_scope_as = hd_yiguan_revolt_county
			}
			if = {
				limit = { exists = scope:hd_yiguan_revolt_county }
				hd_yiguan_spawn_uprising_effect = {
					LEADER = scope:hd_yiguan_shuai
					COUNTY = scope:hd_yiguan_revolt_county
					MEN = 1500
					BIG = no
				}
				if = {
					limit = { exists = scope:hd_yiguan_up_war }
					trigger_event = hd_yiguan.1111
				}
			}
		}
	}
}

scripted_trigger hd_yiguan_ev_has_qiaojun_trigger = {
	any_in_global_list = {
		variable = hd_yiguan_qiaojun_list
		has_county_modifier = hd_yiguan_qiaojun_county_modifier
		holder ?= {
			OR = {
				this = root
				target_is_liege_or_above = root
			}
		}
	}
}

# 侨吴之争中的侨姓一方：家族带侨姓标记、本人为北人，或华夏传承而本人文化与首府县文化不同；侨吴联姻的家族不卷入
scripted_trigger hd_yiguan_ev_qiao_member_trigger = {
	NOT = { house ?= { has_variable = hd_yiguan_qiaowu_married } }
	OR = {
		hd_yiguan_is_qiaoxing_trigger = yes
		has_variable = hd_yiguan_northerner
		AND = {
			hd_yiguan_is_huaxia_trigger = yes
			exists = capital_county
			NOT = { culture = capital_county.culture }
		}
	}
}

# 侨吴之争中的吴姓一方：吴姓、非北人、本人文化与首府县文化相同；侨吴联姻的家族不卷入
scripted_trigger hd_yiguan_ev_wu_member_trigger = {
	NOT = { house ?= { has_variable = hd_yiguan_qiaowu_married } }
	NOT = { has_variable = hd_yiguan_northerner }
	hd_yiguan_is_wuxing_trigger = yes
	exists = capital_county
	culture = capital_county.culture
}
