# 灾异示警
hd_hanfl.1101 = {
	type = character_event
	title = hd_hanfl.1101.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:hd_hanfl_zaiyi_kind ?= flag:disaster }
				desc = hd_hanfl.1101.desc_disaster
			}
			random_valid = {
				desc = hd_hanfl.1101.desc_eclipse
				desc = hd_hanfl.1101.desc_comet
			}
		}
	}
	theme = dynastic_cycle
	override_background = { reference = tgp_asia_throne_room }
	left_portrait = { character = root animation = worry }
	right_portrait = { character = scope:hd_hanfl_sangong animation = stress }
	trigger = {
		hd_hanfl_is_emperor_trigger = yes
		is_imprisoned = no
	}
	immediate = {
		add_character_flag = { flag = hd_hanfl_zaiyi_cd years = 3 }
		save_scope_as = hd_hanfl_emperor
		ordered_vassal = {
			limit = {
				is_alive = yes
				hd_asp_court_sangong_trigger = yes
			}
			order_by = {
				value = 100
				subtract = "opinion(root)"
			}
			save_scope_as = hd_hanfl_sangong
		}
		if = {
			limit = { NOT = { exists = scope:hd_hanfl_sangong } }
			ordered_courtier = {
				limit = {
					is_alive = yes
					hd_asp_court_sangong_trigger = yes
				}
				order_by = {
					value = 100
					subtract = "opinion(root)"
				}
				save_scope_as = hd_hanfl_sangong
			}
		}
		ordered_courtier = {
			limit = {
				is_adult = yes
				is_imprisoned = no
			}
			order_by = learning
			save_scope_as = hd_hanfl_zhijian
		}
		debug_log = "HD_HANFL ZAIYI FIRED"
	}

	# 下诏罪己
	option = {
		name = hd_hanfl.1101.a
		add_prestige = minor_prestige_loss
		add_legitimacy = minor_legitimacy_gain
		add_character_modifier = { modifier = hd_hanfl_zuiji_modifier years = 2 }
		hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_reconciliation ACTOR = root }
		custom_tooltip = hd_hanfl.1101.src
		ai_chance = {
			base = 25
			modifier = { factor = 2 has_trait = just }
			modifier = { factor = 2 has_trait = humble }
			modifier = { factor = 0.5 has_trait = arrogant }
		}
	}

	# 策免三公
	option = {
		name = hd_hanfl.1101.b
		trigger = {
			exists = scope:hd_hanfl_sangong
			scope:hd_hanfl_sangong = {
				is_alive = yes
				hd_asp_court_sangong_trigger = yes
			}
		}
		scope:hd_hanfl_sangong = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_hanfl.1103
			}
			add_opinion = { target = root modifier = hd_hanfl_scapegoat_opinion }
			hd_dc_kit_ten_strip_office_effect = yes
		}
		add_legitimacy = minor_legitimacy_gain
		custom_tooltip = hd_hanfl.1101.b.src
		ai_chance = {
			base = 25
			modifier = { factor = 2 has_trait = paranoid }
			modifier = { factor = 2 has_trait = callous }
		}
	}

	# 下诏求直言
	option = {
		name = hd_hanfl.1101.c
		trigger = { exists = scope:hd_hanfl_zhijian }
		custom_tooltip = hd_hanfl.1101.c.tt
		trigger_event = { id = hd_hanfl.1102 days = 30 }
		ai_chance = {
			base = 25
			modifier = { factor = 2 has_trait = diligent }
			modifier = { factor = 1.5 has_trait = scholar }
		}
	}

	# 置之不理
	option = {
		name = hd_hanfl.1101.d
		add_legitimacy = minor_legitimacy_loss
		ai_chance = {
			base = 15
			modifier = { factor = 2 has_trait = arrogant }
			modifier = { factor = 2 has_trait = lazy }
		}
	}
}

# 祥瑞上报
hd_hanfl.1104 = {
	type = character_event
	title = hd_hanfl.1104.t
	desc = {
		desc = hd_hanfl.1104.desc_intro
		random_valid = {
			desc = hd_hanfl.1104.desc_dragon
			desc = hd_hanfl.1104.desc_ganlu
			desc = hd_hanfl.1104.desc_phoenix
			desc = hd_hanfl.1104.desc_jiahe
		}
		desc = hd_hanfl.1104.desc_end
	}
	theme = dynastic_cycle
	override_background = { reference = tgp_asia_throne_room }
	left_portrait = { character = root animation = happiness }
	right_portrait = { character = scope:hd_hanfl_reporter animation = personality_bold }
	trigger = {
		hd_hanfl_is_emperor_trigger = yes
		is_at_war = no
		any_vassal = { government_has_flag = government_is_celestial }
	}
	immediate = {
		add_character_flag = { flag = hd_hanfl_xiangrui_cd years = 5 }
		save_scope_as = hd_hanfl_emperor
		random_vassal = {
			limit = { government_has_flag = government_is_celestial }
			weight = {
				base = 1
				modifier = { add = 2 has_trait = deceitful }
				modifier = { add = 1 has_trait = ambitious }
			}
			save_scope_as = hd_hanfl_reporter
		}
		debug_log = "HD_HANFL XIANGRUI FIRED"
	}

	# 受贺
	option = {
		name = hd_hanfl.1104.a
		hd_hanfl_xiangrui_accept_effect = yes
		ai_chance = {
			base = 50
			modifier = { factor = 2 has_trait = arrogant }
		}
	}

	# 因瑞改元
	option = {
		name = hd_hanfl.1104.b
		trigger = {
			WJ_era_adopter_trigger = yes
			NOT = { has_character_flag = gaiyuan_flag }
		}
		hd_hanfl_xiangrui_accept_effect = yes
		add_character_flag = { flag = gaiyuan_flag years = { 15 20 } }
		trigger_event = SG_Nianhao_event.002
		ai_chance = {
			base = 25
			modifier = { factor = 2 has_trait = ambitious }
		}
	}

	# 斥为虚饰
	option = {
		name = hd_hanfl.1104.c
		add_prestige = minor_prestige_gain
		scope:hd_hanfl_reporter = {
			add_opinion = { target = root modifier = disappointed_opinion opinion = -15 }
		}
		hd_hanfl_scholars_opinion_effect = { MODIFIER = respect_opinion OPINION = 10 }
		custom_tooltip = hd_hanfl.1104.src
		ai_chance = {
			base = 20
			modifier = { factor = 2 has_trait = humble }
			modifier = { factor = 1.5 has_trait = just }
		}
	}
}

# 祥瑞造假（1104 受贺且上报人欺诈时，40% 于 1~3 年后）
hd_hanfl.1105 = {
	type = character_event
	title = hd_hanfl.1105.t
	desc = hd_hanfl.1105.desc
	theme = intrigue
	override_background = { reference = tgp_asia_throne_room }
	left_portrait = { character = root animation = anger }
	right_portrait = { character = scope:hd_hanfl_reporter animation = fear }
	trigger = {
		exists = var:hd_hanfl_xiangrui_reporter
		var:hd_hanfl_xiangrui_reporter = { is_alive = yes }
	}
	immediate = {
		var:hd_hanfl_xiangrui_reporter = { save_scope_as = hd_hanfl_reporter }
	}

	# 问罪
	option = {
		name = hd_hanfl.1105.a
		add_legitimacy = minor_legitimacy_loss
		if = {
			limit = { scope:hd_hanfl_reporter = { is_imprisoned = no } }
			imprison = {
				target = scope:hd_hanfl_reporter
				type = dungeon
			}
		}
		remove_variable = hd_hanfl_xiangrui_reporter
		ai_chance = {
			base = 50
			modifier = { factor = 2 has_trait = wrathful }
		}
	}

	# 隐忍
	option = {
		name = hd_hanfl.1105.b
		add_legitimacy = medium_legitimacy_loss
		remove_variable = hd_hanfl_xiangrui_reporter
		ai_chance = {
			base = 30
			modifier = { factor = 2 has_trait = craven }
		}
	}
}
