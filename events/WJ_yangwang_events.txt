WJ_yangwang.0101 = {
 type = character_event
 title = WJ_yangwang.0101.t
 desc = WJ_yangwang.0101.desc
 theme = learning
 left_portrait = { character = scope:wj_yangwang_actor }
 trigger = {
  WJ_yangwang_able_trigger = yes
  NOT = { has_character_flag = wj_yangwang_type_1 }
  WJ_yangwang_mufu_invitation_valid_trigger = yes
 }
 immediate = {
  add_character_flag = { flag = wj_yangwang_type_1 months = 179 }
  scope:actor = {
   set_variable = { name = wj_yangwang_mufu_pending value = root }
   WJ_mufu_register_annual_pending = yes
  }
  
 }
 option = {
  name = WJ_yangwang.0101.a
  if = {
   limit = { WJ_yangwang_able_trigger = yes WJ_yangwang_mufu_reply_valid_trigger = yes }
   WJ_yangwang_mufu_accept_effect = yes
  }
  ai_chance = { base = 100 }
 }
 option = {
  name = WJ_yangwang.0101.b
  if = {
   limit = { WJ_yangwang_able_trigger = yes WJ_yangwang_mufu_reply_valid_trigger = yes }
   WJ_yangwang_mufu_refuse_effect = yes
   WJ_yangwang_periodic_refusal_reward_effect = yes
  }
  ai_chance = { base = 100 }
 }
 after = {
  scope:actor ?= {
   if = {
    limit = { var:wj_yangwang_mufu_pending ?= root }
    WJ_mufu_clear_annual_pending = yes
   }
  }
 }
}
