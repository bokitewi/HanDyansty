# 郡国制·军阀契约事件（2026-10-24）
# hd_warlord.0001：军阀被召参战（朝廷「派遣军事援助」或都督区「部署辖区援军」选中了军阀），由军阀自行决定应召与否。
# Root = 军阀；scope:hd_wl_summoner = 召唤者；scope:hd_wl_beneficiary = 被援者（战争领袖）；
# scope:hd_wl_summon_kind = flag:celestial（朝廷派遣，影响力已按三倍扣过）/ flag:district（都督区派遣，另有 scope:wj_dd_* 三个作用域）
# 拒绝：召唤者好感 -20（五年衰减），影响力不退（用户拍板）

namespace = hd_warlord

hd_warlord.0001 = {
	type = character_event
	title = hd_warlord.0001.t
	desc = hd_warlord.0001.desc
	theme = war
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:hd_wl_summoner
		animation = schadenfreude
	}

	trigger = {
		exists = scope:hd_wl_summoner
		exists = scope:hd_wl_beneficiary
	}

	# 应召
	option = {
		name = hd_warlord.0001.a
		if = {
			limit = { scope:hd_wl_summon_kind ?= flag:district }
			WJ_dd_join_aid_wars_effect = yes
		}
		else = {
			# 照搬原版 celestial_send_military_aid_interaction 的参战写法
			scope:hd_wl_beneficiary = {
				every_character_war = {
					limit = {
						scope:hd_wl_beneficiary = { is_leader_in_war = prev }
						is_attacker = scope:hd_wl_beneficiary
						NOT = { is_participant = root }
					}
					add_attacker = root
				}
				every_character_war = {
					limit = {
						scope:hd_wl_beneficiary = { is_leader_in_war = prev }
						is_defender = scope:hd_wl_beneficiary
						NOT = { is_participant = root }
					}
					add_defender = root
				}
			}
			scope:hd_wl_summoner = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = hd_warlord_summon_accepted_msg
					left_icon = root
					right_icon = scope:hd_wl_beneficiary
				}
			}
		}
		ai_chance = {
			base = 0
			modifier = { add = hd_warlord_summon_accept_chance }
		}
	}

	# 拒绝
	option = {
		name = hd_warlord.0001.b
		scope:hd_wl_summoner = {
			add_opinion = {
				target = root
				modifier = hd_warlord_refused_summon_opinion
			}
			send_interface_toast = {
				type = event_toast_effect_bad
				title = hd_warlord_summon_refused_msg
				left_icon = root
			}
		}
		ai_chance = {
			base = 0
			modifier = { add = hd_warlord_summon_refuse_chance }
		}
	}
}
