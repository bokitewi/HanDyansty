# 收事件者：·胡 = 胡人政权；·汉 = 有地的沦陷区汉人；无地的沦陷区汉人（世族）只收不依赖封地的 2011、2211、2213
# 义军事件任何阶段都会发生；"沦陷区汉人起义"催化剂由战争侧的义军起兵效果在战争确实开始时投，本文件不重复投
# 事件义军开战后，胡人宗主 4 年内不再收事件义军（变量 hd_yiguan_evt_uprising_cd），给举兵起义、起兵北还留出空档

# 胡人政权，领内有都城以外的沦陷县
scripted_trigger hd_yiguan_evn_hu_with_occ_trigger = {
	hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_hu }
	hd_yiguan_has_occupied_county_trigger = yes
}

# 有地的沦陷区汉人（有主君）
scripted_trigger hd_yiguan_evn_occupied_landed_trigger = {
	hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_occupied }
	is_landed = yes
	exists = liege
}

# 沦陷区汉人，有地无地都算（有主君）。只给选项不依赖封地的事件用：坞堡选项须有直辖县，无地者自动不显示
scripted_trigger hd_yiguan_evn_occupied_member_trigger = {
	hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_occupied }
	exists = liege
}

# 宗主治下另有可起事的沦陷县（不在你领内），且宗主眼下没有在应对义军、不在事件义军冷却中
scripted_trigger hd_yiguan_evn_uprising_site_trigger = {
	top_liege = {
		hd_yiguan_no_active_uprising_trigger = yes
		NOT = { has_variable = hd_yiguan_evt_uprising_cd }
	}
	top_liege = {
		any_realm_county = {
			has_variable = hd_yiguan_occ
			holder ?= {
				NOT = { this = root }
				NOT = { target_is_liege_or_above = root }
			}
		}
	}
}

scripted_trigger hd_yiguan_evn_has_kin_to_send_trigger = {
	any_courtier = {
		is_close_or_extended_family_of = root
		NOT = { root.player_heir ?= this }
		is_ruler = no
		is_imprisoned = no
	}
}

# 有可归附的南方政权（归南之议）：与本人国境接壤、实领爵级和最高头衔都高于本人、未与本人交战
# 与战争侧 hd_yiguan_guifu_pick_lord_effect、决议 hd_yiguan_guifu_nanchao_decision 的候选条件一致，三处须同步
scripted_trigger hd_yiguan_evn_guifu_lord_available_trigger = {
	any_neighboring_top_liege_realm_owner = {
		hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_south }
		hd_yiguan_war_landed_tier_value > root.hd_yiguan_war_landed_tier_value
		highest_held_title_tier > root.highest_held_title_tier
		NOT = { is_at_war_with = root }
	}
}

# 事件义军开战后记冷却：胡人宗主 4 年内不再收事件义军。须在战争侧起兵效果之后调用（scope:hd_yiguan_up_defender 由该效果重存）
scripted_effect hd_yiguan_evn_uprising_cd_effect = {
	if = {
		limit = {
			exists = scope:hd_yiguan_up_war
			exists = scope:hd_yiguan_up_defender
		}
		scope:hd_yiguan_up_defender = {
			set_variable = { name = hd_yiguan_evt_uprising_cd value = yes years = 4 }
		}
	}
}

# 事件结束（after）时清理没有起事的义军首领（赎买遣散、起兵失败），免得留下孤立的角色和新王朝；
# 已招安得县（有地）或战争已开者不动。写法同原版 07_dlc_ep3_scripted_effects.txt 的 death_vanished
scripted_effect hd_yiguan_evn_cleanup_leader_effect = {
	if = {
		limit = {
			exists = scope:hd_yiguan_new_leader
			NOT = { exists = scope:hd_yiguan_up_war }
			scope:hd_yiguan_new_leader = {
				is_alive = yes
				is_landed = no
			}
		}
		scope:hd_yiguan_new_leader = {
			death = { death_reason = death_vanished }
		}
	}
}
