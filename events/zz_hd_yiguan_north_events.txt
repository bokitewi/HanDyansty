scripted_trigger hd_yiguan_hu_lord_with_occ_trigger = {
	hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_hu_lord }
	hd_yiguan_has_occupied_county_trigger = yes
}

scripted_trigger hd_yiguan_han_vassal_trigger = {
	hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_han_vassal }
	exists = liege
}

# 宗主治下另有可起事的沦陷县（不在你领内）
scripted_trigger hd_yiguan_han_vassal_uprising_site_trigger = {
	top_liege = { hd_yiguan_no_active_uprising_trigger = yes }
	top_liege = {
		any_realm_county = {
			has_variable = hd_yiguan_occ
			holder ?= {
				NOT = { this = root }
				NOT = { target_is_liege_or_above = root }
			}
		}
	}
}

scripted_trigger hd_yiguan_has_kin_to_send_trigger = {
	any_courtier = {
		is_close_or_extended_family_of = root
		NOT = { root.player_heir ?= this }
		is_ruler = no
		is_imprisoned = no
	}
}

hd_yiguan.2001 = {	# 汉民南奔
	type = character_event
	title = hd_yiguan.2001.t
	desc = hd_yiguan.2001.desc
	theme = realm
	override_background = { reference = mpo_rural_village_asian }
	left_portrait = {
		character = root
		animation = anger
	}
	trigger = {
		hd_yiguan_hu_lord_with_occ_trigger = yes
	}
	weight_multiplier = {
		base = 1
		modifier = {
			factor = 2
			hd_yiguan_phase_is_trigger = { PHASE = 1 }
		}
		modifier = {
			factor = 1.5
			hd_yiguan_phase_is_trigger = { PHASE = 2 }
		}
	}
	immediate = {
		hd_yiguan_event_cd_effect = yes
		hd_yiguan_pick_own_occupied_county_effect = yes
	}
	option = {	# 发兵追捕
		name = hd_yiguan.2001.a
		add_prestige = minor_prestige_gain
		every_vassal = {
			limit = { hd_yiguan_is_huaxia_trigger = yes }
			add_opinion = {
				target = root
				modifier = hd_yiguan_chase_opinion
			}
		}
		ai_chance = {
			base = 35
			ai_value_modifier = {
				ai_boldness = 0.3
				ai_vengefulness = 0.3
			}
		}
	}
	option = {	# 听之任之
		name = hd_yiguan.2001.b
		scope:hd_yiguan_occ_county ?= {
			add_county_modifier = {
				modifier = hd_yiguan_liusan_county_modifier
				years = 10
			}
		}
		ai_chance = {
			base = 30
			ai_value_modifier = { ai_energy = -0.3 }
		}
	}
	option = {	# 安抚留民
		name = hd_yiguan.2001.c
		trigger = { gold >= medium_gold_value }
		remove_short_term_gold = medium_gold_value
		scope:hd_yiguan_occ_county ?= { change_county_control = 20 }
		ai_chance = {
			base = 35
			ai_value_modifier = {
				ai_compassion = 0.3
				ai_greed = -0.3
			}
		}
	}
}
