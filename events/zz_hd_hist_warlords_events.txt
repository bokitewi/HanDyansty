namespace = hd_hw

# 玩家是清单首领、且尚未成型时，自选是否在史实郡县起兵。
hd_hw.0001 = {
	type = character_event
	title = hd_hw.0001.t
	desc = hd_hw.0001.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }
	trigger = {
		is_alive = yes
		has_character_flag = hd_hw_offer
		NOT = { has_character_flag = hd_hw_keep }
	}
	on_trigger_fail = { remove_character_flag = hd_hw_offer }
	option = {
		name = hd_hw.0001.a
		hidden_effect = {
			hd_hw_spawn_self_effect = yes
			hd_hw_alliances_effect = yes
		}
	}
	option = {
		name = hd_hw.0001.b
		remove_character_flag = hd_hw_offer
	}
}

# 清单首领是玩家（非天子）的封臣或廷臣时，问玩家是否放他割据。
hd_hw.0002 = {
	type = character_event
	title = hd_hw.0002.t
	desc = hd_hw.0002.desc
	theme = realm
	left_portrait = { character = scope:hd_hw_petitioner animation = personality_bold }
	trigger = {
		exists = scope:hd_hw_petitioner
		scope:hd_hw_petitioner = {
			is_alive = yes
			has_character_flag = hd_hw_offer
			NOT = { has_character_flag = hd_hw_keep }
		}
	}
	on_trigger_fail = {
		scope:hd_hw_petitioner ?= { remove_character_flag = hd_hw_offer }
	}
	option = {
		name = hd_hw.0002.a
		hidden_effect = {
			scope:hd_hw_petitioner = { hd_hw_spawn_self_effect = yes }
			hd_hw_alliances_effect = yes
		}
	}
	option = {
		name = hd_hw.0002.b
		scope:hd_hw_petitioner = { remove_character_flag = hd_hw_offer }
	}
}
