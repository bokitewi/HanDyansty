namespace = hd_lubu_exile

# Narrative only: the victory callback already creates the camp and gives troops.
# A player cannot accidentally choose out of the mandatory historical settlement.
hd_lubu_exile.0001 = {
	type = character_event
	title = hd_lubu_exile.0001.t
	desc = hd_lubu_exile.0001.desc
	theme = war
	left_portrait = { character = root }
	trigger = {
		this = character:lv_bu
		is_alive = yes
		has_character_flag = hd_lbx_camp_prepared
		has_character_flag = hd_lbx_event_troops_received
		NOT = { has_character_flag = hd_lbx_intro_shown }
	}
	immediate = {
		remove_character_flag = hd_lbx_intro_queued
		add_character_flag = hd_lbx_intro_shown
	}
	option = { name = hd_lubu_exile.0001.a }
}

# Chen Gong arrives one day AFTER the first two followers have been processed.
# Temporary failure leaves his own status uncommitted; the personal story retries.
hd_lubu_exile.0002 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:lv_bu
		is_alive = yes
		has_character_flag = hd_lbx_exile_pending
		has_character_flag = hd_lbx_camp_prepared
		has_government = landless_adventurer_government
		exists = domicile.domicile_location
		NOT = { has_character_flag = hd_lbx_chen_resolved }
	}
	immediate = {
		remove_character_flag = hd_lbx_chen_event_queued
		hd_lbx_chen_arrival_effect = yes
	}
}
