namespace = WJ_yinshi

WJ_yinshi.6001 = {
	override_background = { reference = study }
	type = character_event
	title = WJ_yinshi.6001.t
	desc = WJ_yinshi.6001.desc
	theme = learning
	left_portrait = { character = root animation = thinking }
	trigger = {
		WJ_yinshi_able_trigger = yes
		NOR = {
			has_character_flag = wj_yinshi_ev_6001
			has_character_modifier = WJ_yinshi_zhushu_capacity_modifier
		}
	}
	immediate = {
		save_scope_as = wj_yinshi_subject
		add_character_flag = {
			flag = wj_yinshi_ev_6001
			years = 10
		}
	}
	option = {
		name = WJ_yinshi.6001.a
		ai_chance = {
			base = 70
		}
		trigger = {
			WJ_yinshi_scene_valid_trigger = yes
			NOT = {
				has_character_modifier = WJ_yinshi_zhushu_capacity_modifier
			}
		}
		if = {
			limit = {
				WJ_yinshi_scene_valid_trigger = yes
				NOT = {
					has_character_modifier = WJ_yinshi_zhushu_capacity_modifier
				}
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = WJ_yinshi.6001.success_result
				left_icon = root
				add_character_modifier = {
					modifier = WJ_yinshi_zhushu_capacity_modifier
				}
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = WJ_yinshi_scene_lapsed
				left_icon = root
				custom_tooltip = WJ_yinshi_scene_lapsed_tt
			}
		}
	}
	option = {
		name = WJ_yinshi.6001.b
		ai_chance = {
			base = 30
		}
		trigger = {
			WJ_yinshi_scene_valid_trigger = yes
		}
		if = {
			limit = {
				WJ_yinshi_scene_valid_trigger = yes
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = WJ_yinshi.6001.safe_result
				left_icon = root
				change_merit = 75
				add_learning_lifestyle_xp = 150
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = WJ_yinshi_scene_lapsed
				left_icon = root
				custom_tooltip = WJ_yinshi_scene_lapsed_tt
			}
		}
	}
	option = {
		name = WJ_yinshi_scene_close
		trigger = {
			NOT = {
				WJ_yinshi_scene_valid_trigger = yes
			}
		}
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = WJ_yinshi_scene_lapsed
			left_icon = root
			custom_tooltip = WJ_yinshi_scene_lapsed_tt
		}
		ai_chance = {
			base = 100
		}
	}
}
