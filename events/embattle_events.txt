namespace = embattle_events

# Legacy event IDs retained. Current movement/combat entry points call effects directly.
embattle_events.0001 = {
    hidden = yes
    trigger = { exists = scope:army }
    immediate = { scope:army = { hd_embattle_maintain_army_effect = yes } }
}
embattle_events.0002 = {
    hidden = yes
    trigger = { exists = scope:army }
    immediate = { scope:army = { hd_embattle_refresh_army_bonus_effect = yes } }
}

# Character-only safety check. At most one outstanding event per buffed commander.
# No candidate selection, no province iteration, no AI-only-army polling.
embattle_events.0100 = {
    hidden = yes
    immediate = {
        remove_character_flag = hd_embattle_watch_pending
        if = {
            limit = { has_character_flag = hd_embattle_buff_active }
            if = {
                limit = { hd_embattle_bonus_eligible_trigger = yes }
                commanding_army = { hd_embattle_normalize_army_effect = yes }
                hd_embattle_sync_bonus_effect = yes
            }
            else = { hd_embattle_clear_bonus_effect = yes }
        }
    }
}
