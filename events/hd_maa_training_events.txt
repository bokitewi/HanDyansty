hd_maa_training.1000 = {
	type = character_event
	title = hd_maa_training.1000.t
	desc = hd_maa_training.1000.desc
	theme = war
	override_background = { reference = army_camp }

	left_portrait = {
		character = scope:hd_maa_trainer
		animation = personality_zealous
	}

	trigger = {
		exists = scope:hd_maa_trainer
		scope:hd_maa_trainer = { is_alive = yes }
	}

	option = { # 因粮就饷，择勇而练：100金 150威望，训练修正 +0
		name = hd_maa_training.1000.a
		add_gold = -100
		add_prestige = -150
		hd_resolve_xianzhenying_training_effect = { INVESTMENT = 0 }
		ai_chance = {
			base = 50
			modifier = { factor = 0 gold < 100 }
			modifier = { factor = 0.5 gold < 300 }
		}
	}

	option = { # 厚赐军士，广募骁勇：250金 300威望，训练修正 +10
		name = hd_maa_training.1000.b
		trigger = {
			gold >= 250
			prestige >= 300
		}
		add_gold = -250
		add_prestige = -300
		hd_resolve_xianzhenying_training_effect = { INVESTMENT = 10 }
		ai_chance = {
			base = 100
			modifier = { factor = 0 gold < 250 }
		}
	}

	option = { # 不惜府库，尽选天下锐士：500金 500威望，训练修正 +20

			prestige >= 500
		}

			base = 150

			modifier = { factor = 2 gold >= 1000 }
		}
	}
