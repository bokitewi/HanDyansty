# S1 琅邪王出镇建邺。Root：天子
hd_304.0110 = {
	type = character_event
	title = hd_304.0110.t
	desc = hd_304.0110.desc
	theme = realm

	right_portrait = { character = character:si_ma_rui animation = personality_honorable }

		is_alive = yes
		character:si_ma_rui ?= { is_alive = yes }
	}

		name = hd_304.0110.b

	}
}

# S2 石勒归汉。Root：石勒
hd_304.0120 = {
	type = character_event
	title = hd_304.0120.t
	desc = hd_304.0120.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }
	immediate = {
		hd_304_save_han_ruler_effect = yes
	}
	trigger = { is_alive = yes }
	option = {
		name = hd_304.0120.a
		trigger = { exists = scope:hd_304_han }
		if = {
			limit = { scope:hd_304_han = { is_ai = no } }
			scope:hd_304_han = { trigger_event = hd_304.0121 }
		}
		else = {
			hd_304_shile_join_effect = yes
		}
		ai_chance = { base = 100 }
	}
	option = {
		name = hd_304.0120.b
		debug_log = "HD_304 S2 SHILE DECLINED"
		ai_chance = { base = 0 }
	}
}

# S2 玩家汉主收不收石勒。Root：汉主；石勒是 character:shangdang_shi_003
hd_304.0121 = {
	type = character_event
	title = hd_304.0121.t
	desc = hd_304.0121.desc
	theme = war
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = character:shangdang_shi_003 animation = personality_bold }
	trigger = {
		is_alive = yes
		character:shangdang_shi_003 ?= { is_alive = yes }
	}
	option = {
		name = hd_304.0121.a
		# 效果按 root = 石勒写，转一个以石勒为 root 的隐藏事件执行（执行后复核必改 1）
		character:shangdang_shi_003 = { trigger_event = hd_304.0122 }
	}
	option = {
		name = hd_304.0121.b
		debug_log = "HD_304 S2 SHILE REFUSED_BY_HAN"
	}
}

# S2 玩家汉主收下石勒后执行归汉。Root：石勒（隐藏）
hd_304.0122 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
	}
	immediate = {
		hd_304_save_han_ruler_effect = yes
		if = {
			limit = { exists = scope:hd_304_han }
			hd_304_shile_join_effect = yes
		}
	}
}
