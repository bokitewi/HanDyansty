# ============================================================================
# 邀请类与授官事件。
# scope:wj_yinshi_via 标记来源：
#   flag:story       —— 故事抽中（10 年不重复）
#   flag:interaction —— 他人对隐士发起"辟召/举孝廉/举茂才"交互，自动受理后转来
#   flag:ai_route    —— AI 府主经 WJ_ai_bizhao_01 辟召
# scope:actor = 发邀者，scope:recipient = 隐士。
# 接受即出山（入幕经 WJ_mufu_enter、孝廉/茂才经授予效果调用 WJ_yangwang_exit_effect）；
# 拒绝得贤望 200 + 威望 100。
# ============================================================================

# 辟书叩衡门（入幕）
WJ_yinshi.5001 = {
 type = character_event
 title = WJ_yinshi.5001.t
 desc = WJ_yinshi.5001.desc
 theme = learning
 left_portrait = { character = scope:actor animation = admiration }
 trigger = {
  WJ_yinshi_able_trigger = yes
  exists = scope:actor
  exists = scope:wj_yinshi_via
 }
 immediate = {
  if = {
   limit = { scope:wj_yinshi_via = flag:story }
   add_character_flag = { flag = wj_yinshi_ev_5001 years = 10 }
   WJ_yinshi_mufu_lock_effect = yes
  }
 }
 option = {
  name = WJ_yinshi.5001.a
  if = {
   limit = { scope:wj_yinshi_via = flag:story WJ_yinshi_mufu_story_reply_valid_trigger = yes }
   WJ_yinshi_mufu_story_accept_effect = yes
   hidden_effect = { WJ_yangwang_exit_effect = yes } # 入幕内部校验未过时仍出山；入幕成功已出山则无作用。
  }
  else_if = {
   # 故事邀约在应邀时已失效（府主身故、幕位已满等）：仍按选项出山。
   limit = { scope:wj_yinshi_via = flag:story }
   custom_tooltip = WJ_yinshi_invite_lapsed_tt
   show_as_tooltip = { remove_trait = WJ_yangwang_trait }
   hidden_effect = { WJ_yangwang_exit_effect = yes }
  }
  else_if = {
   limit = { scope:wj_yinshi_via = flag:interaction }
   WJ_yinshi_mufu_interaction_accept_effect = yes
  }
  else_if = {
   limit = { scope:wj_yinshi_via = flag:ai_route }
   WJ_yinshi_mufu_ai_accept_effect = yes
  }
  ai_chance = {
   base = 30
   modifier = { add = 40 has_trait = ambitious }
   modifier = { add = 10 has_trait = greedy }
  }
 }
 option = {
  name = WJ_yinshi.5001.b
  WJ_yinshi_mufu_refuse_effect = yes
  WJ_yinshi_refusal_reward_effect = yes
  if = {
   limit = { scope:wj_yinshi_via = flag:interaction scope:actor = { is_ai = no } }
   scope:actor = { trigger_event = WJ_bizhao_reply.0002 }
  }
  ai_chance = {
   base = 70
   modifier = { add = 30 has_trait = content }
   modifier = { add = 20 has_trait = humble }
  }
 }
 after = { WJ_yinshi_mufu_unlock_effect = yes }
}
