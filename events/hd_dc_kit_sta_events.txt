# hd_dc_sta.0101~0103：三种策反阴谋成功；hd_dc_sta.0109：策反失败

namespace = hd_dc_sta

# 策反独立：成功
hd_dc_sta.0101 = {
	type = character_event
	title = hd_dc_sta.0101.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:hd_dc_sta_sub_done }
				desc = hd_dc_sta.0101.desc
			}
			desc = hd_dc_sta.subversion_target_gone.desc
		}
	}
	theme = generic_intrigue_scheme
	left_portrait = {
		character = root
		animation = schadenfreude
	}
	right_portrait = {
		character = scope:target
		animation = personality_bold
	}

	immediate = {
		if = {
			limit = {
				scope:target = {
					is_alive = yes
					is_imprisoned = no
					is_independent_ruler = no
					exists = liege
					NOT = { is_at_war_with = liege }
				}
			}
			save_scope_value_as = {
				name = hd_dc_sta_sub_done
				value = yes
			}
			hd_dc_sta_incite_independence_success_effect = yes
			hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_subversion_success ACTOR = scope:owner }
		}
	}

	option = {
		name = hd_dc_sta.0101.a
		scope:scheme ?= { end_scheme = yes }
	}
}

	theme = generic_intrigue_scheme
	left_portrait = {
		character = root

	immediate = {
		if = {
			limit = {

		}
	}

	option = {
