# 汉代风味内容 D9：毁淫祀结果（2026-10-25）。计划：/mnt/project-files/汉代风味内容/汉代风味内容_计划.md §4.3
# 入口：决议 hd_hanfl_destroy_yinsi_decision；刚毁祠的县带 30 天变量 hd_hanfl_yinsi_just_destroyed
namespace = hd_hanfl

hd_hanfl.1201 = {
	type = character_event
	title = hd_hanfl.1201.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					any_held_title = {
						has_variable = hd_hanfl_yinsi_just_destroyed
						has_variable = hd_hanfl_yinsi_chengyang
					}
				}
				desc = hd_hanfl.1201.desc_chengyang
			}
			desc = hd_hanfl.1201.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = { scope:hd_hanfl_yinsi_result ?= flag:curse }
				desc = hd_hanfl.1201.desc_curse
			}
			triggered_desc = {
				trigger = { scope:hd_hanfl_yinsi_result ?= flag:resent }
				desc = hd_hanfl.1201.desc_resent
			}
			desc = hd_hanfl.1201.desc_calm
		}
	}
	theme = faith
	override_background = { reference = tgp_temple_asia }
	left_portrait = { character = root animation = personality_bold }
	immediate = {
		random_list = {
			70 = { save_scope_value_as = { name = hd_hanfl_yinsi_result value = flag:calm } }
			20 = { save_scope_value_as = { name = hd_hanfl_yinsi_result value = flag:curse } }
			10 = { save_scope_value_as = { name = hd_hanfl_yinsi_result value = flag:resent } }
		}
	}

	option = {
		name = hd_hanfl.1201.a
		if = {
			limit = { scope:hd_hanfl_yinsi_result = flag:curse }
			add_stress = medium_stress_gain
		}
		else_if = {
			limit = { scope:hd_hanfl_yinsi_result = flag:resent }
			every_held_title = {
				limit = { has_variable = hd_hanfl_yinsi_just_destroyed }
				add_county_modifier = { modifier = hd_hanfl_yinsi_resent_modifier years = 5 }
			}
		}
		else = {
			add_stewardship_lifestyle_xp = medium_lifestyle_xp
		}
		custom_tooltip = hd_hanfl.1201.src
		ai_chance = { base = 50 }
	}

	# 乡里生怨时：花钱抚恤，免掉"毁祠之怨"
	option = {
		name = hd_hanfl.1201.b
		trigger = { scope:hd_hanfl_yinsi_result = flag:resent }
		remove_treasury_or_gold = minor_treasury_or_gold_value
		ai_chance = {
			base = 50
			modifier = { factor = 2 has_trait = compassionate }
			modifier = { factor = 0.5 has_trait = greedy }
		}
	}

	after = {
		every_held_title = {
			limit = { has_variable = hd_hanfl_yinsi_just_destroyed }
			remove_variable = hd_hanfl_yinsi_just_destroyed
		}
	}
}
