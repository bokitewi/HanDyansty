namespace = hd_dong_aftermath

# The guaranteed historical plot is resolved by Wang Yun, not a random scheme roll.
hd_dong_aftermath.0001 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:wang_yun
		is_alive = yes
		exists = global_var:hd_dong_murder_pending
		exists = global_var:hd_dong_war_chain_ended
		character:dong_zhuo_2 = { is_alive = yes }
		exists = title:h_china.holder
	}
	on_trigger_fail = {
		remove_global_variable = hd_dong_murder_pending
		debug_log = "HD_DONG MURDER_ABORTED_ACTOR_MISSING"
	}
	immediate = {
		if = {
			limit = { exists = global_var:hd_dong_murder_waiting }
			trigger_event = { id = hd_dong_aftermath.0001 days = 1 }
		}
		else = {
			remove_global_variable = hd_dong_murder_pending
			character:dong_zhuo_2 = {
				hd_dong_queue_death_settlement_effect = yes
				remove_character_modifier = hd_dong_zhuo_timeline_protection
				remove_character_modifier = hd_historical_actor_protection
				hd_dc_clear_protection_reason_effect = yes
				# 2026-10-17：吕布在场时由吕布动手。
				if = {
					limit = {
					character:lv_bu ?= {
						is_alive = yes
						is_imprisoned = no
						hd_dong_inside_realm_trigger = { RULER = title:h_china.holder }
					}
				}
					death = { death_reason = death_murder killer = character:lv_bu }
				}
				else = { death = { death_reason = death_murder killer = character:wang_yun } }
			}
			set_global_variable = hd_hev_184_wangyun_kill_dong
			debug_log = "HD_DONG WANG_YUN_MURDER_COMPLETED"
		}
	}
}

# The Han ruler receives this after Dong's title succession has completed.
hd_dong_aftermath.0002 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		this = title:h_china.holder
		exists = global_var:hd_dong_settlement_pending
		NOT = { exists = global_var:hd_dong_settlement_processed }
		character:dong_zhuo_2 = { is_alive = no }
	}
	on_trigger_fail = { hd_dong_abort_settlement_effect = yes }
	immediate = {
		hd_dong_total_breakup_before_settlement_effect = yes
		hd_dong_settle_death_effect = yes
		if = {
			limit = { character:wang_yun = { is_alive = yes } character:lv_bu = { is_alive = yes } }
			trigger_event = hd_dong_aftermath.0010
		}
	}
}

hd_dong_aftermath.0003 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:li_jue
		is_alive = yes
		is_landed = yes
		is_independent_ruler = yes
		exists = global_var:hd_li_jue_war_pending
		NOT = { exists = global_var:hd_li_jue_war_resolved }
		title:d_jingzhao.holder = { is_alive = yes NOT = { this = character:li_jue } }
	}
	on_trigger_fail = {
		hd_li_jue_end_war_effect = yes
		debug_log = "HD_DONG LI_JUE_WAR_ABORTED_INVALID_ACTOR_OR_TITLE"
	}
	immediate = {
		start_war = { cb = hd_li_jue_changan_cb target = title:d_jingzhao.holder }
		if = {
			limit = { any_character_war = { using_cb = hd_li_jue_changan_cb primary_attacker = root } }
			remove_global_variable = hd_li_jue_war_pending
			set_global_variable = hd_li_jue_war_active
			set_global_variable = hd_hev_184_lijue_war
			# 全局通报：西凉兵反攻长安
			save_scope_as = hd_news_a
			title:d_jingzhao.holder = { save_scope_as = hd_news_b }
			title:b_uuii_p_867.title_province = { save_scope_as = hd_news_place }
			hd_world_news_send_all_effect = { EVENT = hd_world_news.0010 }
			debug_log = "HD_DONG LI_JUE_WAR_CONFIRMED"
		}
		else = {
			hd_li_jue_end_war_effect = yes
			debug_log = "HD_DONG LI_JUE_WAR_START_FAILED"
		}
	}
}

# A bounded event-driven repair; no realm scans or permanent global pulse.
hd_dong_aftermath.0005 = {
	type = character_event
	hidden = yes
	immediate = {
		remove_character_flag = hd_dong_regency_repair_queued
		if = {
			limit = {
				this = title:h_china.holder
				hd_dong_historical_regent_active_trigger = yes
				OR = {
					NOT = { has_diarchy_type = grand_secretariat }
					NOT = { diarch = var:hd_dong_fixed_regent }
					AND = { var:hd_dong_fixed_regent = character:wang_yun NOT = { has_realm_law = grand_chancellor_law } }
					AND = { NOT = { var:hd_dong_fixed_regent = character:wang_yun } NOT = { has_realm_law = WJ_grand_chengxiang_law } }
				}
			}
			var:hd_dong_fixed_regent = { hd_dong_install_regent_effect = yes }
		}
	}
}
