# Source pregnancy policy: empress guaranteed check; consort choice/contraception through native effect.
# Both participants are adults >=18; pregnancy remains subject to native fertility mechanics.
WJ_harem.005 = {
 type = character_event title = WJ_harem_intimacy_title desc = WJ_harem_intimacy_desc theme = family
 right_portrait = { character = scope:ShiQie animation = personality_rational }
 trigger = { WJ_harem_intimacy_allowed = yes }
 immediate = { save_scope_as = HuangDi }
 option = {
  name = WJ_harem_allow_birth
  if = { limit = { WJ_harem_intimacy_allowed = yes }
   scope:ShiQie = { add_character_flag = { flag = ShiQin_flag days = 150 } add_stress = -55 add_prestige = 50 }
   add_stress = -55
   if = { limit = { scope:ShiQie = { has_trait = SanGong_GongWei_1 } }
    add_legitimacy = 50 had_sex_with_effect = { CHARACTER = scope:ShiQie PREGNANCY_CHANCE = 100 }
   }
   else = {
    add_legitimacy = -30 had_sex_with_effect = { CHARACTER = scope:ShiQie PREGNANCY_CHANCE = pregnancy_chance }
    if = { limit = { any_consort = { has_trait = SanGong_GongWei_1 NOT = { any_child = { father = scope:HuangDi } } } }
     add_prestige = -50 add_legitimacy = -50 remove_treasury_or_gold = 50
    }
   }
  }
  ai_chance = { base = 100 modifier = { factor = 0 root = { any_consort = { has_trait = SanGong_GongWei_1 NOT = { any_child = { father = root } } } } scope:ShiQie = { NOT = { has_trait = SanGong_GongWei_1 } } } }
 }
 option = {
  name = WJ_harem_prevent_birth
  if = { limit = { WJ_harem_intimacy_allowed = yes }
   scope:ShiQie = { add_character_flag = { flag = ShiQin_flag days = 150 } add_stress = -55 add_prestige = 50 }
   add_stress = -55 had_sex_with_effect = { CHARACTER = scope:ShiQie PREGNANCY_CHANCE = 0 }
  }
  ai_chance = { base = 100 modifier = { factor = 0 scope:ShiQie = { has_trait = SanGong_GongWei_1 } } }
 }
 option = { name = cancel ai_chance = { base = 0 } }
}
