hd_nbc.0110 = {
	type = character_event
	title = hd_nbc.0110.t
	desc = hd_nbc.0110.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }
	option = {
		name = hd_nbc.0110.a
		add_prestige = major_prestige_gain
		title:h_china.holder ?= { save_scope_as = hd_nbc_jin_emperor }
		# 成汉收尾（boki 10-26）：胜利转移只转县，成汉之主仍持巴蜀帝国、益州与各郡头衔，只靠渡口相连的南平、枳两县也打不进目标。
		# 余县、郡级头衔与封臣交桓温；益州交周抚（周抚不在则交天子）；巴蜀帝国销毁；成汉之主无地后送建康
		title:e_bashu.holder ?= {
			if = {
				limit = {
					NOT = { this = root }
					NOT = { title:h_china.holder ?= this }
				}
				save_scope_as = hd_nbc_cheng_lord
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_nbc_cheng_lord } }
			character:bacong_li_026 ?= {
				if = {
					limit = {
						is_alive = yes
						NOT = { this = root }
					}
					save_scope_as = hd_nbc_cheng_lord
				}
			}
		}
		if = {
			limit = { exists = scope:hd_nbc_cheng_lord }
			create_title_and_vassal_change = { type = conquest save_scope_as = hd_nbc_bashu_change add_claim_on_loss = no }
			scope:hd_nbc_cheng_lord = {
				every_held_title = {
					limit = {
						is_noble_family_title = no
						tgp_is_minister_title = no
						is_landless_type_title = no
						tier >= tier_county
						tier <= tier_duchy
					}
					change_title_holder = { holder = root change = scope:hd_nbc_bashu_change take_baronies = yes }
				}
				every_vassal = {
					limit = { NOT = { this = root } }
					change_liege = { liege = root change = scope:hd_nbc_bashu_change }
				}
			}
			resolve_title_and_vassal_change = scope:hd_nbc_bashu_change
			if = {
				limit = { title:k_uuii_yizhou.holder ?= scope:hd_nbc_cheng_lord }
				if = {
					limit = { character:lujiang_zhou_004 ?= { is_alive = yes is_landed = yes } }
					character:lujiang_zhou_004 = { save_scope_as = hd_nbc_yizhou_holder }
				}
				else = {
					scope:hd_nbc_jin_emperor ?= { save_scope_as = hd_nbc_yizhou_holder }
				}
				if = {
					limit = { exists = scope:hd_nbc_yizhou_holder }
					create_title_and_vassal_change = { type = granted save_scope_as = hd_nbc_yizhou_change add_claim_on_loss = no }
					title:k_uuii_yizhou = { change_title_holder = { holder = scope:hd_nbc_yizhou_holder change = scope:hd_nbc_yizhou_change } }
					resolve_title_and_vassal_change = scope:hd_nbc_yizhou_change
				}
			}
			if = {
				limit = { title:e_bashu.holder ?= scope:hd_nbc_cheng_lord }
				scope:hd_nbc_cheng_lord = { destroy_title = title:e_bashu }
			}
			# 成汉之主出降，送建康
			scope:hd_nbc_cheng_lord = {
				if = {
					limit = {
						is_alive = yes
						is_landed = no
						exists = scope:hd_nbc_jin_emperor
					}
					scope:hd_nbc_jin_emperor = { add_courtier = scope:hd_nbc_cheng_lord }
				}
			}
			debug_log = "HD_NBC 346 FASHU SETTLED"
		}
		save_scope_as = hd_news_a
		if = {
			limit = { exists = scope:hd_nbc_cheng_lord }
			scope:hd_nbc_cheng_lord = { save_scope_as = hd_news_b }
		}
		else = {
			character:bacong_li_026 ?= { save_scope_as = hd_news_b }
		}
		hd_world_news_send_effect = { EVENT = hd_nbc.0901 }
		ai_chance = { base = 100 }
	}
}
