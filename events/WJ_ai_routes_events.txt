WJ_ai_routes.0001 = {
 type = letter_event opening = char_interaction.0001.opening
 sender = scope:actor
 desc = {
  desc = WJ_ai_bizhao_letter
  triggered_desc = { trigger = { scope:mufu_role_1 = yes } desc = WJ_ai_bizhao_role_1 }
  triggered_desc = { trigger = { scope:mufu_role_2 = yes } desc = WJ_ai_bizhao_role_2 }
  triggered_desc = { trigger = { scope:mufu_role_3 = yes } desc = WJ_ai_bizhao_role_3 }
  triggered_desc = { trigger = { scope:mufu_role_4 = yes } desc = WJ_ai_bizhao_role_4 }
  triggered_desc = { trigger = { scope:mufu_role_5 = yes } desc = WJ_ai_bizhao_role_5 }
  triggered_desc = { trigger = { scope:wj_ai_offer = 1 } desc = WJ_ai_offer_gold }
  triggered_desc = { trigger = { scope:wj_ai_offer = 2 } desc = WJ_ai_offer_hook }
  desc = WJ_ai_bizhao_strong_hook
 }
 option = {
  name = ACCEPT trigger = { WJ_ai_bizhao_reply_valid = yes }
  if = { limit = { WJ_ai_bizhao_reply_valid = yes } WJ_ai_bizhao_complete = yes }
 }
 option = {
  name = DECLINE
  if = {
   limit = { WJ_ai_bizhao_reply_valid = yes }
   scope:recipient = {
    add_to_variable_list = { name = wj_mufu_rejected_owners target = scope:actor years = 3 }
    WJ_mufu_refusal_identity_effect = yes
   }
  }
 }
 after = { WJ_ai_bizhao_clear = yes }
}
