# 招魂 · 魂不守舍者（1001-1006）／招魂 · 三年内亡故者（1011-1012）
# 入口：on_action zgrc_flavor_zhaohun_0 / _1

# ------------------------------------------------------------
# 1001 —— 巫觋叩门（帮助者）
# ------------------------------------------------------------
zgrc_custom_flavor.1001 = {
	type = character_event
	title = zgrc_custom_flavor.1001.t
	desc = zgrc_custom_flavor.1001.desc

	theme = study_faith
	# override_background = { reference = zgrc_custom_flavor_zhaohun }

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:zgrc_hun_target
		animation = incapable
	}

	immediate = {
		save_scope_as = zgrc_hun_helper

		# 择一魂不守舍者：亲人 → 朋友 → 至交 → 情人 → 灵魂伴侣
		if = {
			limit = { any_close_family_member = { has_trait = incapable } }
			random_close_family_member = {
				limit = { has_trait = incapable }
				save_scope_as = zgrc_hun_target
			}
		}
		else_if = {
			limit = { any_relation = { type = friend this = { has_trait = incapable } } }
			random_relation = {
				type = friend
				limit = { this = { has_trait = incapable } }
				this = { save_scope_as = zgrc_hun_target }
			}
		}
		else_if = {
			limit = { any_relation = { type = best_friend this = { has_trait = incapable } } }
			random_relation = {
				type = best_friend
				limit = { this = { has_trait = incapable } }
				this = { save_scope_as = zgrc_hun_target }
			}
		}
		else_if = {
			limit = { any_relation = { type = lover this = { has_trait = incapable } } }
			random_relation = {
				type = lover
				limit = { this = { has_trait = incapable } }
				this = { save_scope_as = zgrc_hun_target }
			}
		}
		else_if = {
			limit = { any_relation = { type = soulmate this = { has_trait = incapable } } }
			random_relation = {
				type = soulmate
				limit = { this = { has_trait = incapable } }
				this = { save_scope_as = zgrc_hun_target }
			}
		}
		# 空安全：控制台直触发等无合适目标时以自身充当，避免空作用域
		if = {
			limit = { NOT = { exists = scope:zgrc_hun_target } }
			save_scope_as = zgrc_hun_target
		}
	}

	# —— A 许之：出香火之资，请巫觋行法 ——
	option = {
		name = zgrc_custom_flavor.1001.a
		trigger = { gold >= zgrc_zhaohun_cost }
		show_as_unavailable = { always = yes }

		remove_short_term_gold = zgrc_zhaohun_cost
		add_stress = major_stress_loss

		trigger_event = { id = zgrc_custom_flavor.1002 days = { 1 5 } }

		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 20 has_trait = compassionate }
			modifier = { add = -20 has_trait = cynical }
		}
	}

	# —— B 斥之：一派胡言 ——
	option = {
		name = zgrc_custom_flavor.1001.b

		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = cynical }
			modifier = { add = -20 has_trait = zealous }
		}
	}
}

# ------------------------------------------------------------
# 1002 —— 魂兮归来（招魂者）★循环体★
#   只知一遍遍呼喊，不知魂在何方、是否将归
#   再唤 → 魂那边走一段（1003）；四方已遍则只余空喊（重入本事件）；作罢 → 结束
# ------------------------------------------------------------
zgrc_custom_flavor.1002 = {
	type = character_event
	title = zgrc_custom_flavor.1002.t
	desc = zgrc_custom_flavor.1002.desc

	theme = faith
	override_background = { reference = zgrc_custom_flavor_zhaohun }

	left_portrait = {
		character = root
		animation = lantern
	}

	# —— A 再唤一声（反复呼喊不生慰藉：慰藉只在起念、作罢、魂归时）——
	option = {
		name = zgrc_custom_flavor.1002.a

		if = {
			limit = {
				scope:zgrc_hun_target = {
					OR = {
						NOT = { has_character_flag = zgrc_zhaohun_visited_dong }
						NOT = { has_character_flag = zgrc_zhaohun_visited_nan }
						NOT = { has_character_flag = zgrc_zhaohun_visited_xi }
						NOT = { has_character_flag = zgrc_zhaohun_visited_bei }
					}
				}
			}
			scope:zgrc_hun_target = {
				trigger_event = { id = zgrc_custom_flavor.1003 days = { 1 7 } }
			}
		}
		else = {
			trigger_event = { id = zgrc_custom_flavor.1002 days = { 5 15 } }
		}

		ai_chance = {
			base = 90
			# 四方已遍、魂仍不归：AI 不再空唤，就此收手
			modifier = {
				factor = 0
				scope:zgrc_hun_target = {
					has_character_flag = zgrc_zhaohun_visited_dong
					has_character_flag = zgrc_zhaohun_visited_nan
					has_character_flag = zgrc_zhaohun_visited_xi
					has_character_flag = zgrc_zhaohun_visited_bei
				}
			}
		}
	}

	# —— B 作罢 ——
	option = {
		name = zgrc_custom_flavor.1002.b
		add_stress = medium_stress_loss

		ai_chance = { base = 10 }

		# 呼唤自此止住：他若仍未醒来，就只剩一片死寂
		scope:zgrc_hun_target = {
			trigger_event = { id = zgrc_custom_flavor.1008 days = { 3 10 } }
		}
	}
}

# 1007 —— 重逢（恢复者）●一体两面·一面●
#   与 1006 同一件事：团聚，两处各表一面
# ------------------------------------------------------------
zgrc_custom_flavor.1007 = {
	type = character_event
	title = zgrc_custom_flavor.1007.t
	desc = zgrc_custom_flavor.1007.desc

	theme = friendly

	left_portrait = {
		character = root
		animation = ecstasy
	}
	right_portrait = {
		character = scope:zgrc_hun_helper
		animation = ecstasy
	}

	immediate = {
		# 团聚：关系更进一步（情人→灵魂伴侣／朋友→至交／无关系→朋友）
		if = {
			limit = {
				is_adult = yes
				has_relation_lover = scope:zgrc_hun_helper
				NOT = { has_relation_soulmate = scope:zgrc_hun_helper }
			}
			set_relation_soulmate = { target = scope:zgrc_hun_helper }
		}
		else_if = {
			limit = {
				is_adult = yes
				has_relation_friend = scope:zgrc_hun_helper
				NOT = { has_relation_best_friend = scope:zgrc_hun_helper }
			}
			set_relation_best_friend = { target = scope:zgrc_hun_helper }
		}
		else_if = {
			limit = {
				is_adult = yes
				NOT = { has_relation_friend = scope:zgrc_hun_helper }
				NOT = { has_relation_lover = scope:zgrc_hun_helper }
			}
			set_relation_friend = { target = scope:zgrc_hun_helper }
		}

		# 恩重难还：招魂者自此握他一份薄情（弱牵制，十年为期）
		scope:zgrc_hun_helper = {
			add_hook = {
				target = scope:zgrc_hun_target
				type = indebted_hook
			}
		}
	}

	option = {
		name = zgrc_custom_flavor.1007.a
	}
}

# 1011 —— 巫觋叩门·招亡者（招魂者）
#   亡者不现身

zgrc_custom_flavor.1011 = {
	type = character_event
	title = zgrc_custom_flavor.1011.t
	desc = zgrc_custom_flavor.1011.desc

	theme = study_faith
	override_background = { reference = zgrc_custom_flavor_zhaohun }

	left_portrait = {
		character = root
		animation = thinking
	}

	immediate = {
		save_scope_as = zgrc_hun_helper

		# 取三年内亡故的挚亲挚友（爱人 / 灵魂伴侣 / 朋友 / 至交 / 近亲 / 配偶）
		random_memory = {
			limit = {
				memory_age_years < 3
				OR = {
					has_memory_type = lover_died
					has_memory_type = soulmate_died
					has_memory_type = friend_died
					has_memory_type = best_friend_died
					has_memory_type = relative_died
					has_memory_type = spouse_died
				}
			}
			memory_participant:dead_relation = { save_scope_as = zgrc_hun_target }
		}

		# 空安全：控制台直触发等无合适目标时以自身充当，避免空作用域
		if = {
			limit = { NOT = { exists = scope:zgrc_hun_target } }
			save_scope_as = zgrc_hun_target
		}
	}

	# —— A 许之：仍要招 ——
	option = {
		name = zgrc_custom_flavor.1011.a
		trigger = { gold >= zgrc_zhaohun_cost }
		show_as_unavailable = { always = yes }

		remove_short_term_gold = zgrc_zhaohun_cost
		add_stress = major_stress_loss

		trigger_event = { id = zgrc_custom_flavor.1012 days = { 1 5 } }

		ai_chance = {
			base = 70
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 20 has_trait = compassionate }
			modifier = { add = -20 has_trait = cynical }
		}
	}

	# —— B 斥之 ——
	option = {
		name = zgrc_custom_flavor.1011.b

		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = cynical }
			modifier = { add = -20 has_trait = zealous }
		}

zgrc_custom_flavor.3001 = {
	type = character_event
	title = zgrc_custom_flavor.3001.t
	desc = zgrc_custom_flavor.3001.desc
	theme = stewardship_wealth_focus

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:swordsman
		animation = personality_content
	}

		target = scope:sword_scope
		position = lower_left_portrait
	}

		create_character = {
			template = wandering_knight_character
			location = root.location
			culture = root.culture
			faith = root.faith

		}
		scope:swordsman = {
			zgrc_create_wuyue_sword_effect = yes
		}
		# 宝剑与剑士跨事件传递（3002 结果事件用）
		set_variable = { name = zgrc_wuyue_sword value = scope:sword_scope }
		set_variable = { name = zgrc_swordsman value = scope:swordsman }
	}

	# —— 1 购买（高价）——
	option = {
		name = zgrc_custom_flavor.3001.a
		trigger = {
			gold >= major_gold_value
		}

		remove_short_term_gold = major_gold_value
		scope:sword_scope = { set_owner = root }

		stress_and_fulfillment_impact = {
			generous = medium_stress_impact_loss
			greedy = medium_stress_impact_gain
		}

		ai_chance = {
			base = 15
			modifier = {
				add = 10
				prowess >= 12
			}
			modifier = {
				factor = 0.5
				OR = {
					has_trait = greedy
					has_trait = callous
				}
			}
		}
	}

	# —— 2 决斗（原版单挑流程）——
	option = {
		name = zgrc_custom_flavor.3001.b
		custom_tooltip = zgrc_custom_flavor.3001.b_tt
		trigger = {
			can_start_single_combat_trigger = yes
		}
		# 决斗场景：清理交给 3002，after 跳过早清理
		add_character_flag = zgrc_wuyue_duel

		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = root
			SC_DEFENDER = scope:swordsman
			FATALITY = possible
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = zgrc_custom_flavor.3002
			INVALIDATION_EVENT = zgrc_custom_flavor.3002
		}

		ai_chance = {
			base = 5
			modifier = { add = 10  has_trait = brave }
			modifier = { factor = 0  has_trait = craven }
		}
	}

	# —— 3 招募（宝剑仍归剑士）——
	option = {
		name = zgrc_custom_flavor.3001.c
		root = { add_courtier = scope:swordsman }

		ai_chance = {
			base = 10
			modifier = { add = 5  has_trait = gregarious }
		}
	}

	# —— 4 赶走 ——
	option = {
		name = zgrc_custom_flavor.3001.d

		ai_chance = {
			base = 70
			modifier = { add = 10  has_trait = callous }
		}
	}

	after = {
		hidden_effect = {
			# 决斗场景：变量留给 3002 处理
			if = {
				limit = { has_character_flag = zgrc_wuyue_duel }
				remove_character_flag = zgrc_wuyue_duel
			}
			else = {
				# 非决斗：剑士没被招募 → 宝剑（若没到你手上）销毁 + 剑士消失
				if = {
					limit = { NAND = { exists = var:zgrc_swordsman var:zgrc_swordsman = { employer = root } } }	# 本 mod：剑士缺失时也算「未被招募」
					if = {
						limit = { exists = var:zgrc_wuyue_sword var:zgrc_wuyue_sword = { artifact_owner != root } }	# 本 mod：?= 在变量缺失时为真，会对空变量 destroy_artifact
						destroy_artifact = var:zgrc_wuyue_sword
					}
					var:zgrc_swordsman ?= { silent_disappearance_effect = yes }
				}
				remove_variable = zgrc_wuyue_sword
				remove_variable = zgrc_swordsman
			}
		}
	}
}

# ============================================================
# 吴越宝剑 · 决斗结果（3002）—— 单挑结束后处理
# 入口：3001 选项 2 → configure_start_single_combat_effect → OUTPUT_EVENT
# 作用域：root = 发起者，scope:sc_victor / scope:sc_loser
# ============================================================
zgrc_custom_flavor.3002 = {
	type = character_event
	hidden = yes

	immediate = {
		# root 胜 → 得剑；败 → 扣威望
		if = {
			limit = { scope:sc_victor = root }
			var:zgrc_wuyue_sword ?= { set_owner = root }
			send_interface_toast = {
				title = zgrc_custom_flavor.3001.win_t
				left_icon = root
				right_icon = var:zgrc_wuyue_sword
				custom_tooltip = zgrc_custom_flavor.3001.win_tt
			}
		}
		else = {
			add_prestige = major_prestige_loss
			send_interface_toast = {
				title = zgrc_custom_flavor.3001.lose_t
				left_icon = root
				custom_tooltip = zgrc_custom_flavor.3001.lose_tt
			}
		}

		# 剑士：决斗落败/未招募 → 消失
		var:zgrc_swordsman ?= { silent_disappearance_effect = yes }

		# 宝剑若没到你手上 → 销毁
		if = {
			limit = { exists = var:zgrc_wuyue_sword var:zgrc_wuyue_sword = { artifact_owner != root } }	# 本 mod：?= 在变量缺失时为真，会对空变量 destroy_artifact
			destroy_artifact = var:zgrc_wuyue_sword
		}

		# 清理变量
		remove_variable = zgrc_wuyue_sword
		remove_variable = zgrc_swordsman
	}
}
