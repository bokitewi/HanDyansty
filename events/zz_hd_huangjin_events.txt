# ==========================================================================
# HD 招降青州黄巾 三连鉴定链(原入口决议 hd_subdue_qingzhou_huangjin_decision 已于2026-10-11删除，
#   .1~.3 现无入口，仅保留备用；青州军改由曹操专属事件 .10 获得)
# .1 军事鉴定(阅其众) → .2 管理鉴定(安其食) → .3 外交鉴定(结其心)+总结算
#
# 鉴定公式(调平衡只改 immediate 里两处阈值):
#   score = max(root能力, 对应内阁能力) × 2 + 随机0~40   [.3 仅用root外交]
#   ≥72 大成功(计为成功,并当场永久关闭决议) / ≥50 成功 / 其余失败
# 结算: 每次成功得青州兵(hd_qingzhou_bing)事件兵4000;
#        三验全过 → 另增10000(共22000,一支军队)+ 直辖县发展度增长+30%(20年),
#        并永久关闭决议
# ==========================================================================

hd_huangjin.1 = {
	type = character_event
	title = hd_huangjin.1.t
	theme = war
	override_background = { reference = army_camp }

	left_portrait = {
		character = root
		animation = personality_zealous
	}

	trigger = {
		has_character_flag = hd_hj_done
		is_alive = yes
	}

	immediate = {
		hidden_effect = {
			# 军事鉴定: 取root与元帅中较高者
			set_variable = { name = hd_hj_score value = root.martial }
			if = {
				limit = {
					exists = root.cp:councillor_marshal
					root.cp:councillor_marshal = { martial > root.martial }
				}
				set_variable = { name = hd_hj_score value = root.cp:councillor_marshal.martial }
			}
			change_variable = { name = hd_hj_score multiply = 2 }
			change_variable = { name = hd_hj_score add = { integer_range = { min = 0 max = 40 } } }
			if = {
				limit = { var:hd_hj_score >= 72 }
				change_variable = { name = hd_hj_successes add = 1 }
				set_variable = { name = hd_hj_crit value = 1 }
				set_global_variable = { name = hd_hj_closed value = root }
			}
			else_if = {
				limit = { var:hd_hj_score >= 50 }
				change_variable = { name = hd_hj_successes add = 1 }
			}
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:hd_hj_score >= 72 }
				desc = hd_huangjin.1.desc_crit
			}
			triggered_desc = {
				trigger = { var:hd_hj_score >= 50 }
				desc = hd_huangjin.1.desc_success
			}
			desc = hd_huangjin.1.desc_fail
		}
	}

	option = {
		name = hd_huangjin.1.a
		trigger_event = { id = hd_huangjin.2 days = 1 }
	}
}

				set_variable = { name = hd_hj_crit value = 1 }
				set_global_variable = { name = hd_hj_closed value = root }
			}
			else_if = {
				limit = { var:hd_hj_score >= 50 }
				change_variable = { name = hd_hj_successes add = 1 }
			}
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:hd_hj_score >= 72 }
				desc = hd_huangjin.3.desc_crit
			}
			triggered_desc = {
				trigger = { var:hd_hj_score >= 50 }
				desc = hd_huangjin.3.desc_success
			}
			desc = hd_huangjin.3.desc_fail
		}
	}

	option = {
		name = hd_huangjin.3.a
		if = {
			limit = { var:hd_hj_successes >= 3 }
			# 文本重写 2026-10-09：去掉手写数值提示，让引擎直接显示兵马与屯田修正
			spawn_army = {
				name = hd_hj_qingzhou_army
				men_at_arms = {
					type = hd_qingzhou_bing
					men = 22000
				}
				location = capital_province
				uses_supply = yes
				inheritable = yes
			}
			every_held_county = {
				add_county_modifier = {
					modifier = hd_hj_qingzhou_reclaim_county_modifier
					years = 20
				}
			}
			hidden_effect = {
				set_global_variable = { name = hd_hj_closed value = root }
			}
		}
		else_if = {
			limit = { var:hd_hj_successes >= 2 }
			spawn_army = {
				name = hd_hj_qingzhou_army
				men_at_arms = {
					type = hd_qingzhou_bing
					men = 8000
				}
				location = capital_province
				uses_supply = yes
				inheritable = yes
			}
		}
		else_if = {
			limit = { var:hd_hj_successes >= 1 }
			spawn_army = {
				name = hd_hj_qingzhou_army
				men_at_arms = {
					type = hd_qingzhou_bing
					men = 4000
				}
				location = capital_province
				uses_supply = yes
				inheritable = yes
			}
		}
		else = {
			custom_tooltip = hd_huangjin.3.reward_none
		}
		# 决议关闭状态提示(纯展示)
		if = {
			limit = { var:hd_hj_crit >= 1 }
			custom_tooltip = hd_huangjin.3.closed_crit
		}
		else_if = {
			limit = { var:hd_hj_successes >= 3 }
			custom_tooltip = hd_huangjin.3.closed_full
		}
		else_if = {
			limit = { global_var:hd_hj_click_count >= 3 }
			custom_tooltip = hd_huangjin.3.closed_third
		}
		hidden_effect = {
			remove_variable = hd_hj_score
			remove_variable = hd_hj_successes
			remove_variable = hd_hj_crit
		}
	}
}
