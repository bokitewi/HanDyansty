hd_liu_bei_rise.0100 = {
	type = character_event
	title = hd_liu_bei_rise.0100.t
	desc = hd_liu_bei_rise.0100.desc
	theme = war
	left_portrait = { character = root animation = personality_rational }
	trigger = { this = character:liu_bei
		hd_lb_actor_trigger = yes
		has_character_flag = hd_lb_yellow_pending
		NOT = { has_character_flag = hd_lb_yellow_seen }
	}
	on_trigger_fail = { remove_character_flag = hd_lb_yellow_pending }
	option = {
		name = hd_liu_bei_rise.0100.a
		custom_tooltip = hd_lb_yellow_join_tt
		remove_character_flag = hd_lb_yellow_pending
		add_character_flag = hd_lb_yellow_seen
		if = {
			limit = { hd_lb_actor_trigger = yes }
			add_character_flag = hd_lb_yellow_accepted
			hd_lb_join_yellow_effect = yes
			hd_lb_sync_story_effect = yes
		}
		ai_chance = { base = 100 }
	}
	option = {
		name = hd_liu_bei_rise.0100.b
		custom_tooltip = hd_lb_no_effect_tt
		remove_character_flag = hd_lb_yellow_pending
		add_character_flag = hd_lb_yellow_seen

		ai_chance = { base = 0 }
	}
}

			add_character_flag = hd_lb_mi_visit_done
			hd_lb_mi_visit_effect = yes
			character:mi_zhu ?= { hd_lb_mi_recruit_pending_effect = yes }
			character:mi_fang ?= { hd_lb_mi_recruit_pending_effect = yes }
			hd_lb_sync_story_effect = yes
		}
		ai_chance = { base = 100 }
	}
	option = {
		name = hd_liu_bei_rise.0600.b
		custom_tooltip = hd_lb_mi_recruit_only_tt
		remove_character_flag = hd_lb_mi_pending
		if = {
			limit = { hd_lb_in_xuzhou_trigger = yes character:mi_zhu = { is_alive = yes } }
			add_character_flag = hd_lb_mi_visit_done
			# Recruitment without marriage, chosen by the player.
			character:mi_zhu ?= { hd_lb_mi_recruit_pending_effect = yes }
			character:mi_fang ?= { hd_lb_mi_recruit_pending_effect = yes }
			hd_lb_sync_story_effect = yes
		}
		ai_chance = { base = 0 }
	}
}

hd_liu_bei_rise.0700 = {
	type = character_event
	title = hd_liu_bei_rise.0700.t
	desc = hd_liu_bei_rise.0700.desc
	theme = stewardship
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = character:liu_bei animation = personality_rational }
	trigger = { this = character:tao_qian
		is_alive = yes
		exists = global_var:hd_lb_bequest_pending
		hd_lb_bequest_ready_trigger = yes
	}
	on_trigger_fail = { remove_global_variable = hd_lb_bequest_pending
		remove_global_variable = hd_lb_bequest_authorized }
	option = {
		name = hd_liu_bei_rise.0700.a
		custom_tooltip = hd_lb_bequest_tao_tt
		if = {
			limit = { hd_lb_bequest_ready_trigger = yes }
			set_global_variable = hd_lb_bequest_authorized
			if = {
				limit = { character:liu_bei = { is_ai = no } }
				character:liu_bei = { trigger_event = hd_liu_bei_rise.0701 }
			}
			else = { hd_lb_commit_bequest_effect = yes }
		}
		else = { set_global_variable = hd_lb_bequest_resolved
			remove_global_variable = hd_lb_bequest_pending
			remove_global_variable = hd_lb_bequest_authorized }
		ai_chance = { base = 100 }
	}
	option = {
		name = hd_liu_bei_rise.0700.b
		custom_tooltip = hd_lb_bequest_decline_tt
		set_global_variable = hd_lb_bequest_resolved
		remove_global_variable = hd_lb_bequest_pending
		remove_global_variable = hd_lb_bequest_authorized
		ai_chance = { base = 0 }
	}
}
