namespace = hd_world_news

# 大历史事件全局通报。ROOT = 收到通报的玩家。
# 由 hd_world_news_send_effect 发出，需事先存好 scope：
#   hd_news_a 主事人、hd_news_b 对手（无对手时同 a）、hd_news_place 事发地（省份）。
# 当事双方不收（他们有自己的事件或战争通知）。

hd_world_news.0001 = {
	type = character_event
	title = hd_world_news.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:hd_news_a
					exists = scope:hd_news_place
					exists = scope:hd_news_c
					exists = scope:hd_news_place_c
					exists = scope:hd_news_d
					exists = scope:hd_news_place_d
				}
				desc = hd_world_news.0001.desc
			}
			desc = hd_world_news.0001.missing_details_desc
		}
	}
	theme = war
	override_background = { reference = army_camp }
	left_portrait = { character = scope:hd_news_a animation = personality_zealous }
	right_portrait = {
		character = scope:hd_news_b
		animation = personality_rational
		trigger = { NOT = { scope:hd_news_b = scope:hd_news_a } }
	}
	lower_left_portrait = { character = scope:hd_news_c trigger = { exists = scope:hd_news_c } }
	lower_right_portrait = { character = scope:hd_news_d trigger = { exists = scope:hd_news_d } }
	option = {
		name = hd_world_news.a
		ai_chance = { base = 100 }
	}
}
