						is_alive = yes
						is_landed = yes

						highest_held_title_tier < tier_kingdom
						NOT = { has_title = title:h_china }
						NOT = { has_title = title:e_han }

		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_palace_sili_change
			add_claim_on_loss = no
		}
		title:k_uuii_sili = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:hd_palace_sili_change
			}
		}
		every_in_list = {
			list = hd_palace_sili_duchy_holders
			limit = {
				is_alive = yes
				is_landed = yes
				NOT = { this = root }
				primary_title = { tier < tier_kingdom }
				highest_held_title_tier < tier_kingdom
				NOT = { has_title = title:h_china }
				NOT = { has_title = title:e_han }
				NOT = { liege = root }
			}

				liege = root
				change = scope:hd_palace_sili_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_palace_sili_change
		random_owned_story = {
			type = hd_story_dong_zhuo_life
			set_variable = { name = hd_dong_zhuo_life_phase value = flag:hd_dong_zhuo_life_sili }

# Historical 189.5.13 transition. Titles and direct vassals are transferred
# before death so vanilla succession law never decides the Han inheritance.
hd_palace_crisis.0090 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:liu_hong_3
		is_alive = yes
		has_character_flag = hd_liu_hong_189_death_scheduled
		NOT = { has_character_flag = hd_liu_hong_189_death_completed }
		NOT = { exists = global_var:hd_yellow_turban_han_fell }
	}
	immediate = {
		remove_character_modifier = hd_liu_hong_timeline_protection

		if = {
			limit = { character:liu_bian ?= { is_alive = yes } }
			every_held_title = {
				limit = { tier >= tier_county }
				add_to_list = hd_liu_hong_held_titles
			}
			every_vassal = { add_to_list = hd_liu_hong_direct_vassals }
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = hd_liu_hong_succession_change
				add_claim_on_loss = no
			}
			every_in_list = {
				list = hd_liu_hong_held_titles
				change_title_holder = {
					holder = character:liu_bian
					change = scope:hd_liu_hong_succession_change
				}
			}
			every_in_list = {
				list = hd_liu_hong_direct_vassals
				limit = { NOT = { this = character:liu_bian } }
				change_liege = {
					liege = character:liu_bian
					change = scope:hd_liu_hong_succession_change
				}
			}
			resolve_title_and_vassal_change = scope:hd_liu_hong_succession_change
		}
		if = {
			limit = { NOT = { exists = global_var:hd_yellow_turban_han_fell } }
			add_character_flag = hd_palace_crisis_liu_hong_death_signal
			hd_force_start_palace_crisis_effect = yes
		}
		remove_character_flag = hd_liu_hong_189_death_scheduled
		add_character_flag = hd_liu_hong_189_death_completed
		if = { limit = { NOT = { has_trait = ill } } add_trait = ill }
		death = { death_reason = death_ill }
	}
}

# Lu Bu accepts Dong Zhuo's bargain, kills Ding Yuan, and transfers the whole
# Bingzhou grant (including its direct county) to Dong.
hd_palace_crisis.0022 = {
	type = character_event
	title = hd_palace_crisis.0022.t
	desc = hd_palace_crisis.0022.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_bold }

	option = {
		name = hd_palace_crisis.0022.a
		every_vassal = {
			limit = {
				is_alive = yes
				is_landed = yes
				primary_title = { tier >= tier_county tier < tier_kingdom }
				highest_held_title_tier < tier_kingdom
				NOT = { has_title = title:h_china }
				NOT = { has_title = title:e_han }
			}
			add_to_list = hd_ding_yuan_direct_vassals
		}
		every_held_title = {
			limit = { tier >= tier_county }
			add_to_list = hd_ding_yuan_titles
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_ding_yuan_transfer
			add_claim_on_loss = no
		}
		# Transfer the kingdom first so its title vassals follow before Ding Yuan's
		# remaining personal titles are processed.
		if = {
			limit = { has_title = title:k_uuii_bingzhou }
			title:k_uuii_bingzhou = {
				change_title_holder_include_vassals = {
					holder = character:dong_zhuo_2
					change = scope:hd_ding_yuan_transfer
				}
			}
		}
		every_in_list = {
			list = hd_ding_yuan_titles
			limit = {
				this != title:k_uuii_bingzhou
				tier >= tier_county
			}
			change_title_holder_include_vassals = {
				holder = character:dong_zhuo_2
				change = scope:hd_ding_yuan_transfer
			}
		}
		# Administrative title transfer does not reliably rewrite every direct
		# character liege. Reattach any remainder without flattening sub-vassals.
		every_in_list = {
			list = hd_ding_yuan_direct_vassals
			limit = {
				is_alive = yes
				is_landed = yes
				primary_title = { tier >= tier_county tier < tier_kingdom }
				highest_held_title_tier < tier_kingdom
				NOT = { has_title = title:h_china }
				NOT = { has_title = title:e_han }
				NOT = { liege = character:dong_zhuo_2 }
			}
			change_liege = {
				liege = character:dong_zhuo_2
				change = scope:hd_ding_yuan_transfer
			}
		}
		resolve_title_and_vassal_change = scope:hd_ding_yuan_transfer
		character:dong_zhuo_2 = {
			random_owned_story = {
				type = hd_story_dong_zhuo_life
				set_variable = { name = hd_dong_zhuo_life_phase value = flag:hd_dong_zhuo_life_bingzhou }
			}
		}
		character:lv_bu = {
			remove_character_flag = hd_ding_yuan_adoptive_son
			set_employer = character:dong_zhuo_2
			set_location = { location = title:b_uuii_p_867.title_province stick_to_location = yes }
		}
		character:dong_zhuo_2 = {
			if = {
				limit = { character:lv_bu = { NOT = { father = character:dong_zhuo_2 } } }
				adopt = character:lv_bu
			}
		}
		death = { death_reason = death_murder killer = character:lv_bu }
		character:dong_zhuo_2 = {
			remove_character_flag = hd_palace_ding_lu_branch_open
			remove_character_flag = hd_palace_deposition_aftermath
		}
		ai_chance = { base = 100 }
	}
}

# The on_death callback runs before title succession.  One day later the real
# Sili holder is landed and can safely inherit both the warlord government and
# the Han regency without invalidating an active anti-regent war.
hd_palace_crisis.0024 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		is_landed = yes
		has_character_flag = hd_han_regent_succession_pending
		this = title:k_uuii_sili.holder
		exists = title:e_han.holder
		is_vassal_or_below_of = title:e_han.holder
	}
	on_trigger_fail = {
		if = {
			limit = { is_alive = yes }
			remove_character_flag = hd_han_regent_succession_pending
		}
		if = {
			limit = {
				exists = title:e_han.holder
				title:k_uuii_sili.holder ?= {
					is_alive = yes
					is_landed = yes
					is_vassal_or_below_of = title:e_han.holder
					NOT = { has_character_flag = hd_han_regent_succession_pending }
				}
			}
			title:k_uuii_sili.holder = {
				add_character_flag = hd_han_regent_succession_pending
				trigger_event = { id = hd_palace_crisis.0024 days = 1 }
			}
		}
		else = {
			if = {
				limit = { exists = global_var:hd_anti_regent_league_leader }
				global_var:hd_anti_regent_league_leader = { hd_anti_regent_failure_effect = yes }
			}
			else = {
				title:e_han.holder ?= {
					end_diarchy = yes
					remove_trait = kuzhu_puppet_ruler
				}
			}
		}
	}
	immediate = {
		remove_character_flag = hd_han_regent_succession_pending
		save_scope_as = hd_han_regent_successor
		hd_force_warlord_effect = yes
		add_trait = kuzhu_empire_overseer
		add_character_flag = hd_current_han_regent
		title:e_han.holder = {
			add_trait = kuzhu_puppet_ruler
			try_start_diarchy = regency
			designate_diarch = scope:hd_han_regent_successor
			set_diarch = scope:hd_han_regent_successor
			set_diarchy_swing = 100
		}
		if = {
			limit = { exists = global_var:hd_anti_regent_league_leader }
			global_var:hd_anti_regent_league_leader = {
				save_scope_as = hd_anti_regent_new_leader
				set_variable = { name = hd_anti_regent_target value = scope:hd_han_regent_successor }
				scope:hd_han_regent_successor = {
					set_variable = { name = hd_anti_regent_leader value = scope:hd_anti_regent_new_leader }
				}
				if = {
					limit = {
						has_character_flag = hd_anti_regent_invitations_dispatched
						NOT = { has_variable = hd_anti_regent_pending_invites }
						NOT = { has_character_flag = hd_anti_regent_war_ready }
						NOT = { has_character_flag = hd_anti_regent_first_war_started }
						NOT = { has_character_flag = hd_anti_regent_second_war_started }
					}
					scope:hd_han_regent_successor = {
						add_character_flag = hd_anti_regent_invitations_complete
						trigger_event = { id = hd_palace_crisis.0023 days = 1 }
					}
				}
			}
		}
	}
}
