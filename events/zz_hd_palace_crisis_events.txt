	title = hd_palace_crisis.0001.t
	desc = hd_palace_crisis.0001.desc

		has_character_flag = hd_palace_crisis_controller

	immediate = {
		set_global_variable = hd_palace_crisis_started
		remove_global_variable = hd_palace_crisis_queued
		remove_global_variable = hd_palace_crisis_queue_retry_used

	option = {
		name = hd_palace_crisis.0001.a
		trigger_event = { id = hd_palace_crisis.0002 days = { 20 50 } }
		ai_chance = { base = 100 }
	}

hd_palace_crisis.0002 = {
	type = character_event
	title = hd_palace_crisis.0002.t
	desc = hd_palace_crisis.0002.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_rational }

	trigger = {
		exists = global_var:hd_palace_crisis_started
		is_alive = yes
	}
	on_trigger_fail = { hd_palace_recover_started_controller_effect = yes }

	option = {
		name = hd_palace_crisis.0002.a
		if = {
			limit = { title:k_uuii_liangzhou.holder ?= { is_alive = yes } }
			title:k_uuii_liangzhou.holder = {
				trigger_event = { id = hd_palace_crisis.0003 days = { 10 20 } }
			}
		}
		else = {
			if = {
				limit = { character:he_jin ?= { is_alive = yes } }
				character:he_jin = {
					trigger_event = { id = hd_palace_crisis.0004 days = { 10 20 } }
				}
			}
			else = {
				set_global_variable = hd_palace_he_jin_killed
				if = {
					limit = { character:yuan_shao ?= { is_alive = yes } }
					character:yuan_shao = {
						trigger_event = { id = hd_palace_crisis.0005 days = { 20 50 } }
					}
				}
				else_if = {
					limit = { character:cao_cao ?= { is_alive = yes } }
					character:cao_cao = {
						trigger_event = { id = hd_palace_crisis.0005 days = { 20 50 } }
					}
				}
				else = {
					hd_palace_purge_effect = yes
					hd_palace_crisis_cleanup_effect = yes
				}
			}
		}
		ai_chance = { base = 100 }
	}
}

hd_palace_crisis.0004 = {
	type = character_event
	title = hd_palace_crisis.0004.t
	desc = hd_palace_crisis.0004.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_coward }

	trigger = {
		exists = global_var:hd_palace_crisis_started
		this = character:he_jin
		is_alive = yes
		NOT = { exists = global_var:hd_palace_he_jin_killed }
	}
	on_trigger_fail = { hd_palace_continue_after_he_jin_loss_effect = yes }

	immediate = {
		character:zhang_rang ?= {
			if = {
				limit = { is_alive = yes }
				save_scope_as = hd_palace_killer
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_palace_killer } }
			character:zhao_zhong ?= {
				if = {
					limit = { is_alive = yes }
					save_scope_as = hd_palace_killer
				}
			}
		}
	}

	option = {
		name = hd_palace_crisis.0004.a
		set_global_variable = hd_palace_he_jin_killed
		remove_character_flag = hd_palace_crisis_controller
		if = {
			limit = { exists = scope:hd_palace_killer }
			death = {
				death_reason = death_murder
				killer = scope:hd_palace_killer
			}
		}
		else = { death = { death_reason = death_murder } }

		if = {
			limit = { character:yuan_shao ?= { is_alive = yes } }
			character:yuan_shao = {
				trigger_event = { id = hd_palace_crisis.0005 days = { 20 50 } }
			}
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes } }
			character:cao_cao = {
				trigger_event = { id = hd_palace_crisis.0005 days = { 20 50 } }
			}
		}
		else = {
			hd_palace_purge_effect = yes
			if = {
				limit = {
					title:k_uuii_liangzhou = { has_variable = hd_palace_liangzhou_invited }
					title:k_uuii_liangzhou.holder ?= { is_alive = yes has_character_flag = hd_palace_liangzhou_marching }
				}
				title:k_uuii_liangzhou.holder = { hd_palace_try_finalize_liangzhou_effect = yes }
			}
			else = { hd_palace_crisis_cleanup_effect = yes }
		}
		ai_chance = { base = 100 }
	}
}

# Sent to every landed Liangzhou vassal (including sub-vassals) after the main
# army begins its march.
hd_palace_crisis.0010 = {
	type = character_event
	title = hd_palace_crisis.0010.t
	desc = hd_palace_crisis.0010.desc
	theme = diplomacy
	left_portrait = { character = root animation = personality_rational }

	trigger = {
		is_alive = yes
		has_variable = hd_palace_liangzhou_inviter
		title:k_uuii_liangzhou = { has_variable = hd_palace_liangzhou_invited }
		var:hd_palace_liangzhou_inviter = { is_alive = yes }
	}
	on_trigger_fail = { remove_variable = hd_palace_liangzhou_inviter }

	immediate = {
		var:hd_palace_liangzhou_inviter = { save_scope_as = hd_liangzhou_march_leader }
	}

	option = {
		name = hd_palace_crisis.0010.a
		if = {
			limit = { exists = domicile }
			domicile = { move_domicile = title:b_uuii_p_787.title_province }
		}
		add_opinion = {
			target = scope:hd_liangzhou_march_leader
			modifier = obedience_opinion
			years = 15
		}
		if = {
			limit = { NOT = { has_trait = loyal } }
			add_trait = loyal
		}
		add_character_flag = hd_palace_liangzhou_followed
		remove_variable = hd_palace_liangzhou_inviter
		ai_chance = {
			base = 45
			modifier = { add = 30 opinion = { target = scope:hd_liangzhou_march_leader value >= 50 } }
			modifier = { add = 20 has_trait = loyal }
			modifier = { add = 15 has_trait = brave }
			modifier = { add = -30 has_trait = craven }
			modifier = { add = -40 opinion = { target = scope:hd_liangzhou_march_leader value <= -30 } }
		}
	}

	option = {
		name = hd_palace_crisis.0010.b
		remove_variable = hd_palace_liangzhou_inviter
		ai_chance = { base = 55 }
	}
}

hd_palace_crisis.0015 = {
	type = character_event
	title = hd_palace_crisis.0015.t
	desc = hd_palace_crisis.0015.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		is_alive = yes
		has_character_flag = hd_palace_liangzhou_settled
		exists = title:h_china.holder
	}
	on_trigger_fail = { hd_palace_crisis_cleanup_effect = yes }

	option = {
		name = hd_palace_crisis.0015.a
		if = {
			limit = { character:liu_xie ?= { is_alive = yes } }
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = hd_palace_deposition_change
				add_claim_on_loss = no
			}
			title:h_china = {
				change_title_holder = { holder = character:liu_xie change = scope:hd_palace_deposition_change }
			}
			title:e_han = {
				change_title_holder = { holder = character:liu_xie change = scope:hd_palace_deposition_change }
			}
			resolve_title_and_vassal_change = scope:hd_palace_deposition_change
		}
		# Rebind the regency to the new Han emperor.  Title-holder changes can
		# clear a diarchy, and the later anti-regent chain needs an explicit
		# current regent target even after Liu Xie takes the throne.
		title:e_han.holder = {
			try_start_diarchy = regency
			set_diarch = root
			set_diarchy_swing = 20
		}
		if = {
			limit = { character:yuan_shao ?= { is_alive = yes } }
			character:yuan_shao = {
				set_variable = { name = hd_palace_deposition_regent value = root }
				set_global_variable = {
					name = hd_anti_regent_pending_leader
					value = this
				}
				trigger_event = { id = hd_palace_crisis.0016 days = { 20 50 } }
			}
		}
		else = { hd_anti_regent_select_fallback_leader_effect = yes }
		ai_chance = { base = 100 }
	}
}

hd_palace_crisis.0016 = {
	type = character_event
	title = hd_palace_crisis.0016.t
	desc = hd_palace_crisis.0016.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		this = character:yuan_shao
		is_alive = yes
		has_variable = hd_palace_deposition_regent
		exists = global_var:hd_anti_regent_pending_leader
		global_var:hd_anti_regent_pending_leader = root
	}
	on_trigger_fail = {
		remove_variable = hd_palace_deposition_regent
		if = {
			limit = {
				is_alive = no
				exists = global_var:hd_anti_regent_pending_leader
				global_var:hd_anti_regent_pending_leader = root
			}
			hd_anti_regent_select_fallback_leader_effect = yes
		}
		else = {
			if = {
				limit = {
					exists = global_var:hd_anti_regent_pending_leader
					global_var:hd_anti_regent_pending_leader = root
				}
				hd_anti_regent_clear_pending_dispatch_effect = yes
			}
			hd_palace_crisis_cleanup_effect = yes
		}
	}

	option = {
		name = hd_palace_crisis.0016.a
		remove_variable = hd_palace_deposition_regent
		remove_character_flag = hd_palace_crisis_controller
		# Only the deposition branch sends Yuan Shao away from Sili.  The
		# earlier "风平浪静？" branch deliberately leaves him there.
		hd_yuan_move_to_bohai_effect = yes
		hd_palace_crisis_cleanup_effect = yes
		ai_chance = { base = 100 }
	}
}
