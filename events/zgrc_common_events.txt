# ============================================================
# 入乡随俗 —— 可见选择
# 检测已在 on_action zgrc_ruxiang_pulse 当场判定（五年度脉冲）
# 选项A：入乡随俗（改从当地风俗）；选项B：安土重迁（保留故俗，冷却50年）
# ============================================================
zgrc_common.0002 = {
	type = character_event
	title = zgrc_common.0002.t
	desc = zgrc_common.0002.desc

	theme = culture_change

	left_portrait = {
		character = root
		animation = thinking
	}

	immediate = {
		hidden_effect = {
			add_character_flag = { flag = zgrc_ruxiang_cooldown years = 50 }  # 冷却50年
		}
	}

	option = {
		name = zgrc_common.0002.a

		if = {
			limit = {
				# 校验：首都文化仍为汉家风俗（有华夏礼乐）才改从当地
				capital_county = {
					culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
				}
			}
			set_culture = capital_county.culture
		}

		ai_chance = {
			base = 60
			modifier = {
				add = 20
				OR = {
					has_trait = adventurer
					has_trait = curious
					has_trait = gregarious
					has_trait = humble
					has_trait = compassionate
				}
			}
			modifier = {
				add = -20
				OR = {
					has_trait = stubborn
					has_trait = arrogant
					has_trait = zealous
				}
			}
		}
	}

	option = {
		name = zgrc_common.0002.b

		ai_chance = {
			base = 40
			modifier = {
				add = -20
				OR = {
					has_trait = adventurer
					has_trait = curious
					has_trait = gregarious
				}
			}
		}
	}
}

# ============================================================
# 厚生慎死·极端爆发 —— 非常时期人物响应（场景表态）
# 入口：on_war_started → zgrc_tianxia_xiangying_effect → 本事件
# 传参：var zgrc_extreme_war（战争）/ zgrc_extreme_liege（领主）/ zgrc_extreme_attacker（来犯者）
# 选项：A 食汉禄勤王 / B 王侯将相宁有种乎（独立起义）/ C 苟全无事
# ============================================================

# 1001 —— 场景1：农民起义（四方民变）
zgrc_common.1001 = {
	type = character_event
	title = zgrc_common.1001.t
	desc = zgrc_common.1001.desc
	theme = war

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = zgrc_common.1001.a
		zgrc_extreme_loyal_response_effect = yes
		remove_variable = zgrc_extreme_war
		remove_variable = zgrc_extreme_liege
		remove_variable = zgrc_extreme_attacker
		ai_chance = {
			base = 50
		}
	}

	option = {
		name = zgrc_common.1001.b
		trigger = {
			highest_held_title_tier >= tier_kingdom
			NOT = { government_allows = administrative }
			NOT = { is_at_war_with = root.var:zgrc_extreme_liege }
		}
		zgrc_extreme_rebel_response_effect = yes
		remove_variable = zgrc_extreme_war
		remove_variable = zgrc_extreme_liege
		remove_variable = zgrc_extreme_attacker
		ai_chance = {
			base = 50
		}
	}

	option = {
		name = zgrc_common.1001.c
		add_character_modifier = {
			modifier = zgrc_gouquan_modifier
			years = 5
		}
		remove_variable = zgrc_extreme_war
		remove_variable = zgrc_extreme_liege
		remove_variable = zgrc_extreme_attacker
		ai_chance = {
			base = 0
		}
	}
}

# 乡勇义军 —— 汉人起义 / 收复故土 时，义军来投
# 入口：on_war_started → zgrc_jiduan_baofa_effect → 本事件
# 传参：var zgrc_yijun_war（战争）
# ============================================================

# 2001 —— 义军来投（收复故土，唯一事件）
zgrc_common.2001 = {
	type = character_event
	title = zgrc_common.2001.t
	desc = zgrc_common.2001.desc
	theme = war

	left_portrait = {
		character = root
		animation = thinking
	}

	# —— 选项1：驱之锋镝（事件兵，义军随行，不编入常备）—— AI 默认
	option = {
		name = zgrc_common.2001.a
		spawn_army = {
			name = zgrc_yijun_army_name
			men_at_arms = {
				type = zgrc_hanjia_yiyong
				stacks = {
					value = 8
				}
			}
			location = root.location
			origin = root.location
			inheritable = no
		}
		ai_chance = {
			base = 100
		}
	}
	# —— 选项2：收为麾下（正常招募 2 折，编入常备 800 人）——
	option = {
		name = zgrc_common.2001.b
		trigger = {
			gold >= zgrc_hanjia_yiyong_event_cost
		}
		remove_short_term_gold = zgrc_hanjia_yiyong_event_cost

		create_maa_regiment = {
			type = zgrc_hanjia_yiyong
			check_can_recruit = no
			size = 8
		}
		ai_chance = {
			base = 0
		}
	}
	# —— 选项3：仁德君子（温言安抚，安居乡里）——
	option = {
		name = zgrc_common.2001.c
		add_prestige = 100
		add_character_modifier = {
			modifier = zgrc_rende_modifier
			years = 5
		}
		ai_chance = {
			base = 0
		}
	}
}
