hd_dc_adv.0024 = {
	type = character_event
	title = hd_dc_adv.0024.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:hd_dc_adv_reform_ok }
				desc = hd_dc_adv.0024.desc_success
			}
			desc = hd_dc_adv.0024.desc_failure
		}
	}
	theme = administrative
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { exists = scope:hd_dc_adv_reform_ok }
			animation = happiness
		}
		animation = shame
	}
	right_portrait = {
		character = scope:hd_dc_adv_reform_minister_scope
		animation = steward
	}
	trigger = {
		has_variable = hd_dc_adv_reform_years
	}
	immediate = {
		var:hd_dc_adv_reform_minister ?= { save_scope_as = hd_dc_adv_reform_minister_scope }
		save_scope_value_as = {
			name = hd_dc_adv_reform_roll
			value = hd_dc_kit_adv_reform_chance_value
		}
		random = {
			chance = hd_dc_kit_adv_reform_chance_value
			save_scope_value_as = {
				name = hd_dc_adv_reform_ok
				value = yes
			}
		}
	}
	option = {
		name = {
			trigger = { exists = scope:hd_dc_adv_reform_ok }
			text = hd_dc_adv.0024.a_success
		}
		name = {
			trigger = { NOT = { exists = scope:hd_dc_adv_reform_ok } }
			text = hd_dc_adv.0024.a
		}
		if = {
			limit = { exists = scope:hd_dc_adv_reform_ok }
			hd_dc_kit_adv_reform_success_effect = yes
		}
		else = {
			hd_dc_kit_adv_reform_failure_effect = yes
		}
	}
}
