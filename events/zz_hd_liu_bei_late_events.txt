# LB-22 success adapter: factual success offers the approved Jingzhou appointment.
hd_liu_bei_late.2200 = {
	type = character_event
	title = hd_liu_bei_late.2200.t
	desc = hd_liu_bei_late.2200.desc
	theme = diplomacy_family_focus
	right_portrait = { character = character:guan_yu }

	option = {
		name = hd_lb22_appoint_guan_yu_jingzhou
		trigger = {
			character:guan_yu = { is_alive = yes }
			title:k_uuii_jingzhou.holder = root
		}
		ai_chance = { base = 86 modifier = { add = 8 has_trait = generous } }
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_guan_yu_jingzhou_change
			add_claim_on_loss = no
		}
		title:k_uuii_jingzhou = {
			change_title_holder = {
				holder = character:guan_yu
				change = scope:hd_guan_yu_jingzhou_change
				take_baronies = no
			}
		}
		resolve_title_and_vassal_change = scope:hd_guan_yu_jingzhou_change
		character:guan_yu = { add_character_flag = hd_guan_yu_highest_ruler_jingzhou }
	}
	option = {
		name = hd_lb22_keep_jingzhou_arrangement
		ai_chance = { base = 14 modifier = { add = 12 has_trait = paranoid } }
	}
}

# LB-25 opens only after Guan Yu is dead and Jingzhou is in Sun hands.
hd_liu_bei_late.2500 = {
	type = character_event
	title = hd_liu_bei_late.2500.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { NOT = { hd_lb25_historical_war_possible_trigger = yes } } desc = hd_liu_bei_late.2500.alt_desc }
			desc = hd_liu_bei_late.2500.desc
		}
	}
	theme = war
	right_portrait = { character = character:sun_quan }

	option = {
		name = hd_yiling_attack_wu
		trigger = { hd_lb25_historical_war_possible_trigger = yes }
		ai_chance = { base = 84 modifier = { add = 10 has_trait = vengeful } modifier = { add = -18 has_trait = patient } }
		hd_set_yiling_risks_effect = yes
		if = {
			limit = { title:k_uuii_jingzhou.holder.top_liege = character:sun_quan }
			start_war = { cb = hd_historical_war_cb target = character:sun_quan target_title = title:k_uuii_jingzhou }
		}
		else_if = {
			limit = { title:d_nanjun.holder.top_liege = character:sun_quan }
			start_war = { cb = hd_historical_war_cb target = character:sun_quan target_title = title:d_nanjun }
		}
		random_character_war = {
			limit = { primary_attacker = root primary_defender = character:sun_quan using_cb = hd_historical_war_cb }
			set_variable = { name = hd_world_war_kind value = flag:lb25 }
			save_scope_as = hd_lb25_war
		}
		if = {
			limit = { NOT = { exists = scope:hd_lb25_war } }
			hd_clear_yiling_risks_effect = yes
			set_global_variable = hd_lb25_world_resolved
			set_global_variable = hd_chain_terminal_alternate_failure
			remove_global_variable = hd_lb25_world_open
		}
	}
	option = {
		name = hd_yiling_keep_alliance
		ai_chance = { base = 6 modifier = { add = 20 has_trait = patient } }
		set_global_variable = hd_lb25_world_resolved
		set_global_variable = hd_chain_terminal_nonhistorical
		remove_global_variable = hd_lb25_world_open
	}
	option = {
		name = hd_yiling_demand_jingzhou
		ai_chance = { base = 6 modifier = { add = 15 has_trait = just } }
		character:sun_quan = { trigger_event = hd_liu_bei_late.2510 }
	}
	option = {
		name = hd_yiling_delay_revenge
		ai_chance = { base = 4 modifier = { add = 16 has_trait = cautious_leader } }
		set_global_variable = hd_lb25_world_resolved
		set_global_variable = hd_chain_terminal_alternate_success
		remove_global_variable = hd_lb25_world_open
	}
}
