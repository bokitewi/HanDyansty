namespace = hd_yuan_jizhou

# Timer callbacks do the work immediately; notification buttons cannot delay it.
hd_yuan_jizhou.0001 = {
	type = character_event
	hidden = yes
	trigger = { this = character:yuan_shao is_alive = yes exists = global_var:hd_yuan_jizhou_pending NOT = { exists = global_var:hd_yuan_jizhou_war_attempted } }
	immediate = {
		if = { limit = { exists = global_var:hd_yuan_jizhou_invite_waiting } trigger_event = { id = hd_yuan_jizhou.0001 days = 1 } }
		else = { hd_yuan_jizhou_start_effect = yes }
	}
}

hd_yuan_jizhou.0002 = {
	type = character_event
	hidden = yes
	trigger = { this = character:yuan_shao is_alive = yes exists = global_var:hd_yuan_jizhou_pending NOT = { exists = global_var:hd_yuan_jizhou_handover_started } }
	immediate = {
		if = { limit = { exists = global_var:hd_yuan_jizhou_handover_waiting } trigger_event = { id = hd_yuan_jizhou.0002 days = 1 } }
		else_if = { limit = { hd_yuan_jizhou_handover_ready_trigger = yes } hd_yuan_jizhou_handover_effect = yes }
		else = { debug_log = "HD_YUAN HANDOVER_ABORTED_ACTOR_TARGET_OR_ORIGINAL_WAR_INVALID" hd_yuan_jizhou_end_effect = yes }
	}
}

hd_yuan_jizhou.0003 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				exists = scope:hd_yuan_original_war
				exists = global_var:hd_yuan_jizhou_war
				scope:hd_yuan_original_war = { this = global_var:hd_yuan_jizhou_war using_cb = hd_yuan_jizhou_cb primary_attacker = character:gong_sun_zan primary_defender = character:yuan_shao }
			}
			set_global_variable = hd_yuan_jizhou_handover_complete
			debug_log = "HD_YUAN ORIGINAL_WAR_PRIMARY_DEFENDER_CONFIRMED"
			if = {
				limit = {
					character:han_fu = { is_alive = yes is_courtier_of = character:yuan_shao is_ruler = no }
					exists = global_var:hd_yuan_jizhou_realm_transferred
				}
				trigger_event = hd_yuan_jizhou.0011
			}
			else = { debug_log = "HD_YUAN HAN_REALM_OR_COURT_INCOMPLETE" }
		}
		else = { debug_log = "HD_YUAN ORIGINAL_WAR_PRIMARY_DEFENDER_NOT_CONFIRMED_NO_REPLACEMENT_WAR" }
	}
}
