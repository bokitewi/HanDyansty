hd_regional_decisions.310 = {
	type = character_event
	title = hd_regional_decisions.310.t
	desc = hd_regional_decisions.310.desc
	theme = realm

	# 候选：自己或直属封臣持有的、非其首都的伯爵领，位于该族安置区；没有则取与该族伯爵领接壤者。
	# 按发展度从低到高列出前 8 个。
	immediate = {
		save_scope_as = hd_migration_ruler
		if = {
			limit = {
				hd_wuhu_decision_has_region_candidate_trigger = { TRIBE = scope:hd_wuhu_tribe }
			}
			every_held_title = {
				limit = { hd_wuhu_decision_region_candidate_county_trigger = { TRIBE = scope:hd_wuhu_tribe } }
				add_to_list = hd_migration_candidates
			}
			every_vassal = {
				every_held_title = {
					limit = { hd_wuhu_decision_region_candidate_county_trigger = { TRIBE = scope:hd_wuhu_tribe } }
					add_to_list = hd_migration_candidates
				}
			}
		}
		else = {
			every_held_title = {
				limit = { hd_wuhu_decision_border_candidate_county_trigger = { TRIBE = scope:hd_wuhu_tribe } }
				add_to_list = hd_migration_candidates
			}
			every_vassal = {
				every_held_title = {
					limit = { hd_wuhu_decision_border_candidate_county_trigger = { TRIBE = scope:hd_wuhu_tribe } }
					add_to_list = hd_migration_candidates
				}
			}
		}
		ordered_in_list = {
			list = hd_migration_candidates
			order_by = hd_wuhu_negative_development
			position = 0
			check_range_bounds = no
			save_scope_as = hd_migration_county_1
		}
		if = { # 候选不足 2 个时不存，避免越界取到同一个县造成重复选项
			limit = { list_size = { name = hd_migration_candidates value >= 2 } }
			ordered_in_list = {
				list = hd_migration_candidates
				order_by = hd_wuhu_negative_development
				position = 1
				save_scope_as = hd_migration_county_2
			}
		}
		if = { # 候选不足 3 个时不存，避免越界取到同一个县造成重复选项
			limit = { list_size = { name = hd_migration_candidates value >= 3 } }
			ordered_in_list = {
				list = hd_migration_candidates
				order_by = hd_wuhu_negative_development
				position = 2
				save_scope_as = hd_migration_county_3
			}
		}
		if = { # 候选不足 4 个时不存，避免越界取到同一个县造成重复选项
			limit = { list_size = { name = hd_migration_candidates value >= 4 } }
			ordered_in_list = {
				list = hd_migration_candidates
				order_by = hd_wuhu_negative_development
				position = 3
				save_scope_as = hd_migration_county_4
			}
		}
		if = { # 候选不足 5 个时不存，避免越界取到同一个县造成重复选项
			limit = { list_size = { name = hd_migration_candidates value >= 5 } }
			ordered_in_list = {
				list = hd_migration_candidates
				order_by = hd_wuhu_negative_development
				position = 4
				save_scope_as = hd_migration_county_5
			}
		}
		if = { # 候选不足 6 个时不存，避免越界取到同一个县造成重复选项
			limit = { list_size = { name = hd_migration_candidates value >= 6 } }
			ordered_in_list = {
				list = hd_migration_candidates
				order_by = hd_wuhu_negative_development
				position = 5
				save_scope_as = hd_migration_county_6
			}
		}
		if = { # 候选不足 7 个时不存，避免越界取到同一个县造成重复选项
			limit = { list_size = { name = hd_migration_candidates value >= 7 } }

				list = hd_migration_candidates
				order_by = hd_wuhu_negative_development
				position = 6
				save_scope_as = hd_migration_county_7

			limit = { list_size = { name = hd_migration_candidates value >= 8 } }
			ordered_in_list = {

				order_by = hd_wuhu_negative_development
				position = 7
				save_scope_as = hd_migration_county_8
			}
		}
	}

		hd_complete_inward_migration_effect = { TITLE = scope:hd_migration_county_1 }
