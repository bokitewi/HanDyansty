# 通告分发：参与者玩家（及结局时已标记者）收事件，其余天下玩家收弹窗
# 角色变量 hd_yiguan_notice 由 hd_yiguan_notify_effect 写入
############################
hd_yiguan.0009 = {
	type = character_event
	hidden = yes
	trigger = {
		has_variable = hd_yiguan_notice
	}
	immediate = {
		if = {
			limit = { exists = global_var:hd_yiguan_ender }
			global_var:hd_yiguan_ender = { save_scope_as = hd_yiguan_ender }
		}
		# 通告视角（按族属与宗主判断，不按组：结局后已查不到组），随 trigger_event 传给 0001–0014
		#   flag:hu     胡人（非汉化），独立宗主为都城在北方的胡人政权
		#   flag:north  其余非汉人（汉化王朝成员等），独立宗主为都城在北方的胡人政权或汉化政权
		#   不存        汉人视角（各事件原文）
		if = {
			limit = {
				hd_yiguan_is_hu_trigger = yes
				NOT = { hd_yiguan_is_sinicized_trigger = yes }
				top_liege = {
					hd_yiguan_is_hu_regime_trigger = yes
					hd_yiguan_capital_in_north_trigger = yes
				}
			}
			save_scope_value_as = { name = hd_yiguan_view value = flag:hu }
		}
		else_if = {
			limit = {
				NOT = { hd_yiguan_is_han_trigger = yes }
				top_liege = {
					OR = {
						hd_yiguan_is_hu_regime_trigger = yes
						hd_yiguan_is_sinicized_regime_trigger = yes
					}
					hd_yiguan_capital_in_north_trigger = yes
				}
			}
			save_scope_value_as = { name = hd_yiguan_view value = flag:north }
		}
		# 衣冠沦丧的宣布者本人一律按胜者（胡人）视角，不论都城在何处
		if = {
			limit = {
				var:hd_yiguan_notice = 13
				exists = scope:hd_yiguan_ender
				scope:hd_yiguan_ender = root
			}
			save_scope_value_as = { name = hd_yiguan_view value = flag:hu }
		}
		if = {
			limit = {
				OR = {
					has_variable = hd_yiguan_notice_full
					top_participant_group:hd_yiguan_nandu ?= { always = yes }
				}
			}
			if = {
				limit = { var:hd_yiguan_notice = 1 }
				trigger_event = hd_yiguan.0001
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 2 }
				trigger_event = hd_yiguan.0002
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 3 }
				trigger_event = hd_yiguan.0003
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 4 }
				trigger_event = hd_yiguan.0004
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 11 }
				trigger_event = hd_yiguan.0011
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 12 }
				trigger_event = hd_yiguan.0012
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 13 }
				trigger_event = hd_yiguan.0013
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 14 }
				trigger_event = hd_yiguan.0014
			}
		}
		else = {
			if = {
				limit = { var:hd_yiguan_notice = 1 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0001.t
					custom_tooltip = hd_yiguan_toast_start
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 2 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0002.t
					custom_tooltip = hd_yiguan_toast_taowang
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 3 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0003.t
					custom_tooltip = hd_yiguan_toast_chongtu
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 4 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0004.t
					custom_tooltip = hd_yiguan_toast_tingzhi
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 11 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0011.t
					custom_tooltip = hd_yiguan_toast_restored
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 12 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0012.t
					custom_tooltip = hd_yiguan_toast_sinicized
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 13 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_yiguan.0013.t
					custom_tooltip = hd_yiguan_toast_lost
				}
			}
			else_if = {
				limit = { var:hd_yiguan_notice = 14 }
				if = {
					limit = {
						exists = global_var:hd_yiguan_outcome
						global_var:hd_yiguan_outcome = flag:guangfu
					}
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = hd_yiguan.0014.t
						custom_tooltip = hd_yiguan_toast_fallback_guangfu
					}
				}
				else_if = {
					limit = {
						exists = global_var:hd_yiguan_outcome
						global_var:hd_yiguan_outcome = flag:heliu
					}
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = hd_yiguan.0014.t
						custom_tooltip = hd_yiguan_toast_fallback_heliu
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = hd_yiguan.0014.t
						custom_tooltip = hd_yiguan_toast_fallback_lunsang
					}
				}
			}
		}
		remove_variable = hd_yiguan_notice
		if = {
			limit = { has_variable = hd_yiguan_notice_full }
			remove_variable = hd_yiguan_notice_full
		}
	}
}
