# 两晋十六国统一史实线（j16）机制事件，号段 hd_j16.0001–0099（08 §2）。
# 命名空间 hd_j16 在每个 j16 事件文件开头各声明一次（先例：hd_hanfl 在 8 个文件里各声明一次）。
namespace = hd_j16

# 大限（A1-F6）：被杀类卒年的当事人是玩家时不杀，改弹本事件。Root：当事人。
# 死因、凶手由 hd_j16_death_violent_effect 记在 var:hd_j16_dx_reason、var:hd_j16_dx_killer 上。
# 强撑每人只能一次（D10），与病笃 hd_nbc.0100 共用 hd_j16_endured
hd_j16.0001 = {
	type = character_event
	title = hd_j16.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = hd_j16_endured }
				desc = hd_j16.0001.desc_again
			}
			desc = hd_j16.0001.desc
		}
	}
	theme = death
	left_portrait = { character = root animation = fear }
	right_portrait = {
		trigger = { exists = scope:hd_j16_dx_k }
		character = scope:hd_j16_dx_k
		animation = schadenfreude
	}
	trigger = { is_alive = yes }
	immediate = {
		if = {
			limit = { var:hd_j16_dx_killer ?= { is_alive = yes } }
			var:hd_j16_dx_killer = { save_scope_as = hd_j16_dx_k }
		}
	}

	# 坦然赴死
	option = {
		name = hd_j16.0001.a
		hd_j16_doom_die_effect = yes
		ai_chance = { base = 100 }
	}
	# 强撑一时（只能一次）
	option = {
		name = hd_j16.0001.b
		trigger = { NOT = { has_variable = hd_j16_endured } }
		hd_j16_doom_clear_effect = yes
		set_variable = hd_j16_endured
		if = {
			limit = { NOT = { has_trait = wounded_1 } }
			add_trait = wounded_1
		}
		add_stress = major_stress_impact_gain
		set_variable = { name = hd_nbc_death_delay years = 1 }
		ai_chance = { base = 0 }
	}
}
