	title = hd_sc_bawang.0001.t
	desc = hd_sc_bawang.0001.desc
	theme = war
	left_portrait = {

		animation = war_over_win

			name = hd_sc_bawang_army
			men_at_arms = { type = light_footmen men = 3000 }

			uses_supply = yes

	}

		name = hd_sc_bawang.0001.a
	}
