hd_dc_debug.0050 = {
	type = character_event title = hd_dc_debug.results theme = stewardship
	trigger = { debug_only = yes is_ai = no exists = situation:dynastic_cycle }
	immediate = {
		clear_saved_scope = hd_dc_debug_record
		situation:dynastic_cycle = {
			ordered_in_list = {
				variable = hd_dc_records
				limit = { OR = { NOT = { exists = scope:hd_dc_debug_before } var:hd_dc_record_id < scope:hd_dc_debug_before } }
				order_by = { value = var:hd_dc_record_id multiply = -1 } max = 1 check_range_bounds = no
				save_scope_as = hd_dc_debug_record
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = { trigger = { exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.desc }
			desc = hd_dc_debug.receipt.empty
		}
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { var:hd_dc_record_kind = flag:instance } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.instance }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_actual_debit } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.payment }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_unreturned_debit } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.refund }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_protection_started } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.protection }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_protection_wait } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.protection_wait }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_protection_audit_pending } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.audit_pending }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_pause_started } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.pause }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_audit_first_funded } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.trial }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { var:hd_dc_node = 19 } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.letters }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_ordinary_result } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.closed }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { has_variable = hd_dc_protected_person } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.subject }
		triggered_desc = { trigger = { scope:hd_dc_debug_record ?= { var:hd_dc_record_kind = flag:point_history } exists = scope:hd_dc_debug_record } desc = hd_dc_debug.receipt.points }
	}
	option = {
		name = hd_dc_debug.details
		trigger = { exists = scope:hd_dc_debug_record }
		trigger_event = { id = hd_dc_debug.0051 }
	}
	option = {
		name = hd_dc_debug.older
		trigger = { exists = scope:hd_dc_debug_record }
		save_scope_value_as = { name = hd_dc_debug_before value = scope:hd_dc_debug_record.var:hd_dc_record_id }
		trigger_event = { id = hd_dc_debug.0050 }
	}
	option = { name = hd_dc_debug.latest clear_saved_scope = hd_dc_debug_before trigger_event = { id = hd_dc_debug.0050 } }
	option = { name = hd_dc_debug.back clear_saved_scope = hd_dc_debug_before trigger_event = { id = hd_dc_debug.0001 } }
}
