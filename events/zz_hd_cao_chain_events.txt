hd_cao_chain.0001 = {
	type = character_event
	title = hd_cao_chain.0001.t
	desc = hd_cao_chain.0001.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_rational }

	trigger = {
		this = character:cao_cao
		is_alive = yes
		has_character_flag = hd_cao_early_story_pending
		var:hd_cao_early_story_phase = flag:hd_cao_phase_opening
	}
	on_trigger_fail = {
		if = {
			limit = { is_alive = yes has_character_flag = hd_cao_early_story_pending var:hd_cao_early_story_phase = flag:hd_cao_phase_opening }
			trigger_event = { id = hd_cao_chain.0001 months = 1 }
		}
	}

	immediate = {
		remove_character_flag = hd_cao_early_story_pending
		add_character_flag = hd_cao_early_story_started
		set_variable = { name = hd_cao_early_story_phase value = flag:hd_cao_phase_prepare }
		character:wu_fu ?= {
			if = {
				limit = { is_alive = yes }
				death = { death_reason = death_execution }
			}
		}
	}

	option = {
		name = hd_cao_chain.0001.a
		trigger_event = { id = hd_cao_chain.0002 months = 1 }
		ai_chance = { base = 100 }
	}
}

hd_cao_chain.0003 = {
	type = character_event
	title = hd_cao_chain.0003.t
	desc = hd_cao_chain.0003.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_bold }

	trigger = { this = character:cao_cao is_alive = yes has_character_flag = hd_cao_early_story_started var:hd_cao_early_story_phase = flag:hd_cao_phase_sword }
	on_trigger_fail = { if = { limit = { is_alive = yes has_character_flag = hd_cao_early_story_started var:hd_cao_early_story_phase = flag:hd_cao_phase_sword } trigger_event = { id = hd_cao_chain.0003 months = 1 } } }

	option = {
		name = hd_cao_chain.0003.a
		set_variable = { name = hd_cao_early_story_phase value = flag:hd_cao_phase_assassination }
		add_character_flag = hd_cao_qixing_baodao_obtained
		# The startup artifact belongs to Wang Yun in the 184 bookmark.  Hand
		# that historical sword to Cao Cao; if Wang Yun is absent, the startup
		# fallback may already have placed it on Cao Cao.
		character:wang_yun ?= {
			if = {
				limit = { is_alive = yes }
				random_character_artifact = {
					limit = { has_variable = hd_qixing_baodao }
					save_scope_as = hd_cao_qixing_artifact
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_cao_qixing_artifact } }
			character:cao_cao = {
				random_character_artifact = {
					limit = { has_variable = hd_qixing_baodao }
					save_scope_as = hd_cao_qixing_artifact
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_cao_qixing_artifact } }
			character:cao_cao = {
				create_artifact = {
					name = hd_qixing_baodao_name
					description = hd_qixing_baodao_description
					type = sword
					template = hd_legendary_weapon_template
					visuals = hd_qixing_baodao
					modifier = hd_qixing_baodao_weapon_modifier
					decaying = no
					save_scope_as = hd_cao_qixing_artifact
				}
			}
		}
		if = {
			limit = { exists = scope:hd_cao_qixing_artifact }
			scope:hd_cao_qixing_artifact = {
				set_variable = { name = hd_qixing_baodao value = yes }
			}
		}
		if = {
			limit = { exists = scope:hd_cao_qixing_artifact }
			scope:hd_cao_qixing_artifact = {
				set_owner = character:cao_cao
				equip_artifact_to_owner_replace = yes
			}
		}
		trigger_event = { id = hd_cao_chain.0004 days = 7 }
		ai_chance = { base = 100 }
	}
}

hd_cao_chain.0006 = {
	type = character_event
	title = hd_cao_chain.0006.t
	desc = hd_cao_chain.0006.desc
	theme = intrigue
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		this = character:cao_cao
		is_alive = yes
		has_title = title:d_chenliu
		has_character_flag = hd_cao_chenliu_memory_pending
		NOT = { has_character_flag = hd_cao_chenliu_memory_received }
	}
	on_trigger_fail = {
		if = {
			limit = { is_alive = yes has_character_flag = hd_cao_chenliu_memory_pending NOT = { has_character_flag = hd_cao_chenliu_memory_received } }
			trigger_event = { id = hd_cao_chain.0006 months = 1 }
		}
	}

	immediate = {
		if = {
			limit = { character:lv_bo_she ?= { is_alive = yes } }
			character:lv_bo_she = {
				every_close_family_member = {
					if = {
						limit = { is_alive = yes }
						death = { death_reason = death_murder killer = root }
					}
				}
				if = { limit = { is_alive = yes } death = { death_reason = death_murder killer = root } }
			}
		}
		if = {
			limit = { character:chen_gong ?= { is_alive = yes } }
			set_relation_rival = { target = character:chen_gong reason = rival_claimed_throne }
		}
		remove_character_flag = hd_cao_chenliu_memory_pending
		add_character_flag = hd_cao_chenliu_memory_received
		random_owned_story = {
			type = hd_story_cao_life
			set_variable = { name = hd_cao_life_phase value = flag:hd_cao_life_escape }
		}
	}

	option = {
		name = hd_cao_chain.0006.a
		trigger_event = { id = hd_cao_chain.0094 days = 1 }
		ai_chance = { base = 100 }
	}
}

hd_cao_chain.0090 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:cao_cao
		is_alive = yes
		has_character_flag = hd_cao_chenliu_migration_pending
		NOT = { has_character_flag = hd_cao_chenliu_migration_completed }
		var:hd_cao_early_story_phase = flag:hd_cao_phase_migration
	}
	on_trigger_fail = {
		if = {
			limit = { is_alive = yes has_character_flag = hd_cao_chenliu_migration_pending NOT = { has_character_flag = hd_cao_chenliu_migration_completed } var:hd_cao_early_story_phase = flag:hd_cao_phase_migration }
			trigger_event = { id = hd_cao_chain.0090 months = 1 }
		}
		else = { remove_character_flag = hd_cao_chenliu_migration_pending }
	}
	immediate = {
		remove_character_flag = hd_cao_chenliu_migration_pending
		add_character_flag = hd_cao_chenliu_migration_started
		if = { limit = { NOT = { has_trait = callous } } add_trait = callous }
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_cao_chenliu_change
			add_claim_on_loss = no
		}
		title:d_chenliu = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:hd_cao_chenliu_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_cao_chenliu_change
		hd_force_independent_warlord_effect = yes
		remove_character_flag = hd_cao_chenliu_migration_started
		add_character_flag = hd_cao_chenliu_migration_completed
		random_owned_story = {
			type = hd_story_cao_life
			set_variable = { name = hd_cao_life_phase value = flag:hd_cao_life_chenliu }
		}
		hd_anti_regent_add_cao_effect = yes
		if = {
			limit = {
				NOT = { has_character_flag = hd_cao_chenliu_reorganization_pending }
				NOT = { has_character_flag = hd_cao_chenliu_reorganization_received }
			}
			add_character_flag = hd_cao_chenliu_reorganization_pending
			set_variable = { name = hd_cao_early_story_phase value = flag:hd_cao_phase_reorganization }
			add_character_flag = hd_cao_chenliu_memory_pending
			trigger_event = { id = hd_cao_chain.0006 days = 1 }
		}
	}
}

	trigger = {

		is_alive = yes
		has_character_flag = hd_cao_chenliu_reorganization_pending
		NOT = { has_character_flag = hd_cao_chenliu_reorganization_received }
		var:hd_cao_early_story_phase = flag:hd_cao_phase_reorganization
	}
	on_trigger_fail = {
		if = {
			limit = { this = character:cao_cao is_alive = yes has_character_flag = hd_cao_chenliu_reorganization_pending NOT = { has_character_flag = hd_cao_chenliu_reorganization_received } var:hd_cao_early_story_phase = flag:hd_cao_phase_reorganization }
			trigger_event = { id = hd_cao_chain.0094 months = 1 }
		}
	}

	option = {
		name = hd_cao_chain.0094.a
		character:xia_hou_dun ?= {
			if = {
				limit = { is_alive = yes is_landed = yes }
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_cao_xiahou_fealty_change }
				change_liege = { liege = root change = scope:hd_cao_xiahou_fealty_change }
				resolve_title_and_vassal_change = scope:hd_cao_xiahou_fealty_change
			}
		}

		character:xia_hou_yuan ?= { add_to_list = hd_cao_chenliu_courtiers }
		character:cao_ren ?= { add_to_list = hd_cao_chenliu_courtiers }
		character:cao_hong ?= { add_to_list = hd_cao_chenliu_courtiers }
		character:cao_chun ?= { add_to_list = hd_cao_chenliu_courtiers }
		every_in_list = {
			list = hd_cao_chenliu_courtiers
			if = { limit = { is_alive = yes is_landed = no } set_employer = root }
			every_spouse = {
				limit = { is_alive = yes is_landed = no }
				set_employer = root
			}
			every_child = {
				limit = { is_alive = yes is_adult = no is_landed = no }
				set_employer = root
			}
		}

		add_gold = 6000
		spawn_army = {
			name = hd_cao_chenliu_danyang_army
			men_at_arms = { type = hd_danyang_jingbing men = 4000 }
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
		remove_character_flag = hd_cao_chenliu_reorganization_pending
		add_character_flag = hd_cao_chenliu_reorganization_received
		remove_character_flag = hd_cao_early_story_started
		add_character_flag = hd_cao_early_story_completed
		remove_variable = hd_cao_early_story_phase
		ai_chance = { base = 100 }
	}
}
