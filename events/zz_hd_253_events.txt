# 魏将来降：北伐得胜后，守方一名无地将领来投（没有合适的人就不显示）。Root：攻方
hd253.0120 = {
	type = character_event
	title = hd253.0120.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:hd253_defector }
				desc = hd253.0120.desc
			}
			desc = hd253.0120.desc_none
		}
	}
	theme = war
	left_portrait = { character = root animation = personality_honorable }
	right_portrait = {
		character = scope:hd253_defector
		animation = personality_rational
		trigger = { exists = scope:hd253_defector }
	}
	trigger = { is_alive = yes }
	immediate = {
		# 先在战败守将宫中挑，再在其主君宫中挑；司马氏、邓艾、钟会等史实上不会入蜀的人除外
		scope:hd253_beifa_loser ?= {
			random_courtier = {
				limit = { hd253_defector_trigger = yes }
				save_scope_as = hd253_defector
			}
			if = {
				limit = { NOT = { exists = scope:hd253_defector } }
				liege ?= {
					random_courtier = {
						limit = { hd253_defector_trigger = yes }
						save_scope_as = hd253_defector
					}
				}
			}
		}
	}
	option = {
		name = hd253.0120.a
		trigger = { exists = scope:hd253_defector }
		scope:hd253_defector = {
			set_employer = root
			add_opinion = { target = root modifier = grateful_opinion opinion = 20 }
		}
		ai_chance = { base = 100 }
	}
	option = {
		name = hd253.0120.b
		trigger = { NOT = { exists = scope:hd253_defector } }
		add_prestige = minor_prestige_gain
		ai_chance = { base = 100 }
	}
}

# 毌丘俭起兵。Root：毌丘俭。电脑起兵，玩家可隐忍
hd253.0400 = {
	type = character_event
	title = hd253.0400.t
	desc = hd253.0400.desc
	theme = war
	left_portrait = { character = root animation = war_attacker }
	right_portrait = {
		character = character:wen_qin
		animation = personality_bold
		trigger = { character:wen_qin ?= { is_alive = yes } }
	}
	trigger = {
		is_alive = yes
		NOT = { has_global_variable = hd253_guanqiu_done }
	}
	# 矫太后诏，起兵讨司马（10-18 加固 D：确认开战后才上锁、发通报；没开出来记一次失败，下次再问）
	option = {
		name = hd253.0400.a
		if = {
			limit = {
				NOT = { has_global_variable = hd253_guanqiu_done }
				is_at_war = no
			}
			liege ?= { save_scope_as = hd_news_b }
			hd253_rebel_effect = yes
		}
		if = {
			limit = { exists = scope:hd253_rebel_war }
			set_global_variable = hd253_guanqiu_done
			set_global_variable = hd253_guanqiu_war_ok
			if = {
				limit = { has_global_variable = hd253_guanqiu_pending }
				remove_global_variable = hd253_guanqiu_pending
			}
			save_scope_as = hd_news_a
			hd_world_news_send_effect = { EVENT = hd253.0550 }
		}
		else_if = {
			limit = { NOT = { has_global_variable = hd253_guanqiu_done } }
			hd253_rebel_fail_effect = { KEY = guanqiu }
		}
		ai_chance = { base = 100 }
	}
	# 隐忍不发
	option = {
		name = hd253.0400.b
		set_global_variable = hd253_guanqiu_done
		if = {
			limit = { has_global_variable = hd253_guanqiu_pending }
			remove_global_variable = hd253_guanqiu_pending
		}
		add_stress = medium_stress_impact_gain
		ai_chance = { base = 0 }
	}
}
