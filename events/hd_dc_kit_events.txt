# 天朝循环·时代特色：通用隐藏事件

# A2 无国库开战（AUHCO 移植）：跳过军费检查，直接尝试一场最佳战争。
# 目标限制照 AUHCO：和平时目标不强于自己；已在战时目标不超过自己 1.5 倍。
hd_dc_kit.0001 = {
	type = character_event
	hidden = yes
	trigger = {
		is_ai = yes
		is_independent_ruler = yes
		hd_dc_phase_is_luanshi_trigger = yes
	}
	immediate = {
		save_scope_value_as = { name = hd_dc_kit_at_war value = { value = 0 if = { limit = { is_at_war = yes } value = 1 } } }
		ai_start_best_war = {
			cb = {
				claim_cb
				de_jure_cb
				chinese_consolidation_cb
				hd_strategic_annex_cb
				hd_kingdom_seize_cb
				duchy_conquest_cb
				county_conquest_cb
				individual_duchy_de_jure_cb
				individual_county_de_jure_cb
				vassalization_cb
			}
			is_valid = {
				scope:target_character = {
					in_diplomatic_range = root
					trigger_if = {
						limit = { scope:hd_dc_kit_at_war = 1 }
						current_military_strength <= root.hd_dc_kit_150_percent_strength_value
					}
					trigger_else = {
						current_military_strength <= root.current_military_strength
					}
				}
			}
		}
	}
}
