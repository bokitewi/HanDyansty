namespace = hd_qxzl

hd_qxzl.0001 = {
	type = character_event
	title = hd_qxzl.0001.t
	desc = hd_qxzl.0001.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		exists = global_var:hd_qxzl_scan_done
		has_character_flag = hd_qxzl_pending
		NOT = { has_character_flag = hd_qxzl_seen }
		NOT = { has_character_flag = hd4_tao_qxzl_protected }
			hd_qxzl_ticket_eligible_trigger = yes
	}
	on_trigger_fail = {
		remove_character_flag = hd_qxzl_pending
		debug_log = "HD_QXZL VISIBLE_EVENT_SKIPPED_INELIGIBLE"
	}
	immediate = { add_character_flag = hd_qxzl_seen }

	option = {
		name = hd_qxzl.0001.a
		if = {
			limit = { NOT = { has_character_flag = hd4_tao_qxzl_protected }
			hd_qxzl_ticket_eligible_trigger = yes }
			custom_tooltip = hd_qxzl_independence_tt
			hidden_effect = { hd_qxzl_release_ruler_effect = yes }
		}
		else = { custom_tooltip = hd_qxzl_ineligible_tt }
		hidden_effect = { remove_character_flag = hd_qxzl_pending }
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = ambitious }
			# 奸佞 is the DISLOYAL trait, not deceitful (狡诈).
			modifier = { add = 20 has_trait = disloyal }
			modifier = { add = 20 has_relation_rival = liege }
			# 不忠顺 is the obedience state toward the CURRENT direct liege.
			modifier = { add = 100 NOT = { is_obedient_to = liege } }
		}
	}

	option = {
		name = hd_qxzl.0001.b
		custom_tooltip = hd_qxzl_wait_tt
		hidden_effect = { remove_character_flag = hd_qxzl_pending }
		ai_chance = {
			base = 20
			# County base = 20 + 20 = 40; duke base stays 20.
			modifier = { add = 20 highest_held_title_tier = tier_county }
			modifier = {
				add = 30
				OR = {
					has_relation_friend = liege
					has_relation_best_friend = liege
				}
			}
			modifier = { add = 10 has_trait = content }
			# 忠信 is the LOYAL trait, not honest (诚实).
			modifier = { add = 100 has_trait = loyal }
			modifier = { add = 100 is_obedient_to = liege }
		}
	}
}

# before installing this patch. Not attached to any pulse or normal event.
# Console: event hd_qxzl.9000 <living-character-id>
# Normal future coalition endings do not require this command.
hd_qxzl.9000 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		exists = global_var:hd_anti_regent_chain_resolved
		NOT = { exists = global_var:hd_qxzl_started }
	}
	immediate = { hd_qxzl_begin_effect = yes }
}
