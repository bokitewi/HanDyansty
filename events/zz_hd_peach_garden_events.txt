namespace = hd_peach_garden

hd_peach_garden.0001 = {
	type = character_event
	title = hd_peach_garden.0001.t
	desc = hd_peach_garden.0001.desc
	theme = alliance
	override_background = {
		reference = garden_location
	}

	left_portrait = {
		character = root
		animation = gongshou
		camera = camera_event_left_away_3_4
	}

	right_portrait = {
		character = scope:guan_yu
		animation = gongshou
		camera = camera_event_right_to_the_right
	}

	lower_center_portrait = {
		character = scope:zhang_fei
		animation = gongshou
	}

	trigger = {
		this = character:liu_bei
		is_alive = yes
		game_start_date = 184.1.1
		current_date >= 184.1.1
		current_date <= 184.12.31
		has_character_flag = hd_peach_garden_event_pending
		NOT = { has_character_flag = hd_peach_garden_oath_completed }
		character:guan_yu ?= {
			is_alive = yes
			is_landed = no
		}
		character:zhang_fei ?= {
			is_alive = yes
			is_landed = no
		}
	}

	immediate = {
		remove_character_flag = hd_peach_garden_event_pending
		add_character_flag = hd_peach_garden_event_opened
		character:guan_yu ?= {
			save_scope_as = guan_yu
		}
		character:zhang_fei ?= {
			save_scope_as = zhang_fei
		}
	}

	option = {
		name = hd_peach_garden.0001.a

		character:guan_yu = {
			if = {
				limit = { employer != root }
				set_employer = root
			}
			set_relation_blood_brother = {
				target = root
			}
		}

		character:zhang_fei = {
			if = {
				limit = { employer != root }
				set_employer = root
			}
			set_relation_blood_brother = {
				target = root
			}
		}

		character:guan_yu = {
			set_relation_blood_brother = {
				target = character:zhang_fei
			}
		}

		remove_character_flag = hd_peach_garden_event_opened
		add_character_flag = hd_peach_garden_oath_completed
		if = {
			limit = {
				NOT = { has_character_flag = hd_peach_garden_sponsorship_pending }
				NOT = { has_character_flag = hd_peach_garden_sponsorship_received }
			}
			add_character_flag = hd_peach_garden_sponsorship_pending
			trigger_event = { id = hd_peach_garden.0002 days = { 7 14 } }
		}

		ai_chance = {
			base = 100
		}
	}
}

hd_peach_garden.0002 = {
	type = character_event
	title = hd_peach_garden.0002.t
	desc = hd_peach_garden.0002.desc
	theme = stewardship
	left_portrait = { character = scope:zhang_shi_ping animation = personality_rational }
	right_portrait = { character = scope:su_shuang animation = personality_rational }

	trigger = {
		this = character:liu_bei
		is_alive = yes
		has_character_flag = hd_peach_garden_sponsorship_pending
		NOT = { has_character_flag = hd_peach_garden_sponsorship_received }
	}
	immediate = {
		character:zhang_shi_ping ?= { save_scope_as = zhang_shi_ping }
		character:su_shuang ?= { save_scope_as = su_shuang }
	}

	option = {
		name = hd_peach_garden.0002.a
		add_gold = 3000
		spawn_army = {
			name = hd_liu_bei_sponsored_army
			men_at_arms = { type = light_footmen men = 3000 }
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
		remove_character_flag = hd_peach_garden_sponsorship_pending
		add_character_flag = hd_peach_garden_sponsorship_received
		ai_chance = { base = 100 }
	}
}
