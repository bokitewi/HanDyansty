namespace = hds

# 2026-10-10 孙家线重制。
# hds.5100 岘山之死：孙坚征刘表，中黄祖伏兵阵亡（191.1–192.12；由 hds_sj_xianshan_schedule_effect 排期）。
# hds.4100 孙权接续：孙策死后，孙权接掌兄长的部曲与江东之志。

hds.5100 = {
	type = character_event
	title = hds.5100.t
	desc = hds.5100.desc
	theme = death
	left_portrait = { character = root }
	right_portrait = { character = scope:hds_xianshan_killer }
	trigger = {
		this = character:sun_jian
		is_alive = yes
		is_imprisoned = no
		NOT = { has_character_flag = hds_sj_departed }
		exists = character:liu_biao
		character:liu_biao ?= { is_alive = yes is_imprisoned = no }
	}
	# 刘表已死：取消，交给 200 年的兜底剧情；孙坚或刘表被囚：撤销排期，下次 60 天检查重排。
	on_trigger_fail = {
		remove_character_flag = hds_sj_xianshan_scheduled
		if = {
			limit = {
				OR = {
					NOT = { exists = character:liu_biao }
					character:liu_biao ?= { is_alive = no }
				}
			}
			add_character_flag = hds_sj_xianshan_cancelled
		}
	}
	immediate = {
		character:huang_zu ?= {
			if = { limit = { is_alive = yes } save_scope_as = hds_xianshan_killer }
		}
		if = {
			limit = { NOT = { exists = scope:hds_xianshan_killer } }
			character:liu_biao = { save_scope_as = hds_xianshan_killer }
		}
	}
	option = {
		name = hds.5100.a
		ai_chance = { base = 100 }
		hds_sj_xianshan_death_effect = yes
	}
}
