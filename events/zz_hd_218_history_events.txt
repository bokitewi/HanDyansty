	type = character_event
	title = hd_sg218.0010.t
	desc = hd_sg218.0010.desc
	theme = martial
	left_portrait = { character = root animation = personality_bold }
	trigger = {
		game_start_date = 218.1.1
		current_date = 218.1.1
		this = character:liu_bei
		is_alive = yes
		is_landed = yes
		exists = scope:hd_sg218_war_hanzhong
		NOT = { has_character_flag = hd218_liu_bei_mobilization_done }

	immediate = {
		add_character_flag = hd218_liu_bei_mobilization_done

		name = hd_sg218.0010.a

			limit = { exists = scope:hd_sg218_war_hanzhong }

			hidden_effect = {
				spawn_army = {
					name = hd218_hanzhong_mobilized_army

					men_at_arms = { type = hd_bandun_congbing men = 1500 }

					origin = root.capital_province
					uses_supply = yes
					inheritable = no
					war = scope:hd_sg218_war_hanzhong
				}
			}
		}
		add_character_modifier = { modifier = hd_sg218_levy_modifier years = 2 }
