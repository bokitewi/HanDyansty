SanGong_DianZhang.015 = {
 type = character_event hidden = yes
 trigger = {
  is_ai = yes WJ_inner_emperor_trigger = yes is_adult = yes
  primary_spouse = { has_trait = SanGong_GongWei_1 }
  NOT = { any_child = { is_alive = yes has_trait = SanGong_Taizi } }
  any_child = { WJ_royal_prince_candidate = { EMPEROR = root } }
 }
 immediate = { 
        save_scope_as = HuangDi 
        ordered_child = {
            max = 1
            limit = {
                WJ_royal_prince_candidate = { EMPEROR = scope:HuangDi }
            }
            order_by = {
                value = age
                if = {
                    limit = { mother = { has_trait = SanGong_GongWei_1 } }
                    add = 50
                }
                
                if = {
                    limit = { has_trait = intellect_good_3 }
                    add = 30
                }
                if = {
                    limit = { is_male = yes }
                    add = 100
                }
                if = {
                    limit = { has_trait = beauty_good_3 }
                    add = 30
                }
                if = {
                    limit = { 
                        OR = {
                            has_trait = beauty_bad_1 
                            has_trait = beauty_bad_2
                            has_trait = beauty_bad_3
                            has_trait = intellect_bad_1 
                            has_trait = intellect_bad_2
                            has_trait = intellect_bad_3
                            has_trait = great_pox
                            has_trait = early_great_pox
                            has_trait = lovers_pox
                            has_trait = leper
                            has_trait = maimed
                            has_trait = disfigured
                            has_trait = eunuch
                            has_trait = blind
                            has_trait = one_eyed
                            has_trait = one_legged
                            has_trait = clubfooted
                            has_trait = hunchbacked
                            has_trait = lisping
                            has_trait = stuttering
                        }
                        }
                    add = -100
                }
            }
            save_scope_as = taizi
        }
        
    
  if = { limit = { WJ_royal_can_install_prince = yes } scope:taizi = { add_gold = 300 } WJ_royal_install_prince_effect = yes }
 }
}
