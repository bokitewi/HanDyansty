# D24：攻下京城次日结算。Root：占领者；scope:hd_court_occ_war：该场战争。
# 挟天子 CB 的战争以攻方胜利结束（胜利效果即挟天子）；其他战争以白和结束，再立占领者为权臣。
hd_court.0002 = {
	type = character_event
	hidden = yes
	trigger = {
		exists = scope:hd_court_occ_war
		exists = title:h_china.holder
		is_independent_ruler = yes
		hd_court_regent_eligible_trigger = yes
		scope:hd_court_occ_war = {
			primary_attacker = root
			primary_defender = title:h_china.holder
		}
	}
	immediate = {
		if = {
			limit = { scope:hd_court_occ_war = { using_cb = hd_court_xietianzi_cb } }
			scope:hd_court_occ_war = { end_war = attacker }
		}
		else = {
			scope:hd_court_occ_war = { end_war = white_peace }
			hd_court_takeover_effect = yes
		}
	}
}
