# 火井煮盐（蜀郡）
hd_hanfl.2001 = {
	type = character_event
	title = hd_hanfl.2001.t
	desc = hd_hanfl.2001.desc
	theme = stewardship
	override_background = { reference = tgp_farm_asia }
	left_portrait = { character = root animation = admiration }
	trigger = {
		NOT = { has_character_flag = hd_hanfl_region_2001_seen }
		hd_hanfl_capital_in_duchy_trigger = { DUCHY = d_shujun }
	}
	immediate = { add_character_flag = hd_hanfl_region_2001_seen }

	option = {
		name = hd_hanfl.2001.a
		capital_county = {
			add_county_modifier = { modifier = hd_hanfl_region_saltwell_modifier years = 5 }
		}
		custom_tooltip = hd_hanfl.2001.src
	}
	option = {
		name = hd_hanfl.2001.b
		add_stress = minor_stress_loss
	}
}
