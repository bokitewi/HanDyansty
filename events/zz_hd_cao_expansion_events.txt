namespace = hd_cao_expansion

# Repeatable by completed ROUND, not a once-in-a-lifetime event. Pending stays
# set while the player reads the event, so no other pulse starts underneath it.
hd_cao_expansion.0001 = {
	type = character_event
	title = hd_cao_expansion.0001.t
	desc = hd_cao_expansion.0001.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }
	trigger = {
		this = character:cao_cao
		is_alive = yes
		has_character_flag = hd_cao_expansion_menu_pending
	}
	immediate = {
		if = {
			limit = { has_variable = hd_cao_expansion_last_region }
			var:hd_cao_expansion_last_region = { save_scope_as = hd_cao_expansion_last_region_scope }
		}
	}
	option = {
		name = hd_cao_expansion.0001.a
		trigger = {
			NOT = { hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_yuzhou } }
		}
		custom_tooltip = hd_cao_expansion.yuzhou_tt
		hidden_effect = { hd_cao_expansion_start_region_effect = { REGION = title:k_uuii_yuzhou } }
		ai_chance = {
			base = 50
			modifier = {
				factor = 4
				hd_cao_expansion_region_has_target_trigger = { REGION = title:k_uuii_yuzhou }
			}
		}
	}
	option = {
		name = hd_cao_expansion.0001.b
		trigger = {
			NOT = { hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_qingzhou } }
		}
		custom_tooltip = hd_cao_expansion.qingzhou_tt
		hidden_effect = { hd_cao_expansion_start_region_effect = { REGION = title:k_uuii_qingzhou } }
		ai_chance = {
			base = 50
			modifier = {
				factor = 4
				hd_cao_expansion_region_has_target_trigger = { REGION = title:k_uuii_qingzhou }
			}
		}
	}
	option = {
		name = hd_cao_expansion.0001.c
		trigger = {
			OR = {
				hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_yuzhou }
				hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_qingzhou }
			}
			NOT = { hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_xuzhou } }
		}
		custom_tooltip = hd_cao_expansion.xuzhou_tt
		hidden_effect = { hd_cao_expansion_start_region_effect = { REGION = title:k_uuii_xuzhou } }
		ai_chance = {
			base = 50
			modifier = {
				factor = 4
				hd_cao_expansion_region_has_target_trigger = { REGION = title:k_uuii_xuzhou }
			}
		}
	}
	# If all three directions are controlled, still show this final occurrence
	# of the event with one valid exit instead of generating an empty window.
	option = {
		name = hd_cao_expansion.0001.d
		trigger = {
			hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_yuzhou }
			hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_qingzhou }
			hd_cao_expansion_controls_region_trigger = { REGION = title:k_uuii_xuzhou }
		}
		remove_character_flag = hd_cao_expansion_menu_pending
		add_character_flag = hd_cao_expansion_finished
		custom_tooltip = hd_cao_expansion.finished_tt
		ai_chance = { base = 100 }
	}
}
