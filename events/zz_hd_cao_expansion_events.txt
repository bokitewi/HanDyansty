namespace = hd_cao_expansion
hd_cao_expansion.0001 = {
	type = character_event
	title = hd_cao_expansion.0001.t
	desc = hd_cao_expansion.0001.desc
	theme = war
	left_portrait = {
		character = root
		animation = personality_bold
	}
	trigger = {
		this = character:cao_cao
		is_alive = yes
		has_character_flag = hd_cao_expansion_menu_pending
	}
	immediate = {
		if = {
			limit = {
				has_variable = hd_cao_expansion_last_region
			}
			var:hd_cao_expansion_last_region = {
				save_scope_as = hd_cao_expansion_last_region_scope
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = hd_cao_expansion_finished_round
				}
			}
			set_variable = {
				name = hd_cao_expansion_finished_round
				value = 0
			}
		}
		save_scope_value_as = {
			name = hd4_cao_menu_round
			value = var:hd_cao_expansion_finished_round
		}
		add_character_flag = {
			flag = hd4_cao_menu_lease
			days = 180
		}
	}
	option = {
		name = hd_cao_expansion.0001.a
		trigger = {
			NOT = {
				hd_cao_expansion_controls_region_trigger = {
					REGION = title:k_uuii_yuzhou
				}
			}
			has_character_flag = hd_cao_expansion_menu_pending
			exists = scope:hd4_cao_menu_round
			var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
		}
		ai_chance = {
			base = 50
			modifier = {
				factor = 4
				hd_cao_expansion_region_has_target_trigger = {
					REGION = title:k_uuii_yuzhou
				}
			}
		}
		if = {
			limit = {
				has_character_flag = hd_cao_expansion_menu_pending
				exists = scope:hd4_cao_menu_round
				var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
			}
			custom_tooltip = hd_cao_expansion.yuzhou_tt
			hidden_effect = {
				hd_cao_expansion_start_region_effect = {
					REGION = title:k_uuii_yuzhou
				}
			}
		}
	}
	option = {
		name = hd_cao_expansion.0001.b
		trigger = {
			NOT = {
				hd_cao_expansion_controls_region_trigger = {
					REGION = title:k_uuii_qingzhou
				}
			}
			has_character_flag = hd_cao_expansion_menu_pending
			exists = scope:hd4_cao_menu_round
			var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
		}
		ai_chance = {
			base = 50
			modifier = {
				factor = 4
				hd_cao_expansion_region_has_target_trigger = {
					REGION = title:k_uuii_qingzhou
				}
			}
		}
		if = {
			limit = {
				has_character_flag = hd_cao_expansion_menu_pending
				exists = scope:hd4_cao_menu_round
				var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
			}
			custom_tooltip = hd_cao_expansion.qingzhou_tt
			hidden_effect = {
				hd_cao_expansion_start_region_effect = {
					REGION = title:k_uuii_qingzhou
				}
			}
		}
	}
	option = {
		name = hd_cao_expansion.0001.c
		trigger = {
			OR = {
				hd_cao_expansion_controls_region_trigger = {
					REGION = title:k_uuii_yuzhou
				}
				hd_cao_expansion_controls_region_trigger = {
					REGION = title:k_uuii_qingzhou
				}
			}
			NOT = {
				hd_cao_expansion_controls_region_trigger = {
					REGION = title:k_uuii_xuzhou
				}
			}
			has_character_flag = hd_cao_expansion_menu_pending
			exists = scope:hd4_cao_menu_round
			var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
		}
		ai_chance = {
			base = 50
			modifier = {
				factor = 4
				hd_cao_expansion_region_has_target_trigger = {
					REGION = title:k_uuii_xuzhou
				}
			}
		}
		if = {
			limit = {
				has_character_flag = hd_cao_expansion_menu_pending
				exists = scope:hd4_cao_menu_round
				var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
			}
			custom_tooltip = hd_cao_expansion.xuzhou_tt
			hidden_effect = {
				hd_cao_expansion_start_region_effect = {
					REGION = title:k_uuii_xuzhou
				}
			}
		}
	}
	option = {
		name = hd_cao_expansion.0001.d
		trigger = {
			hd_cao_expansion_controls_region_trigger = {
				REGION = title:k_uuii_yuzhou
			}
			hd_cao_expansion_controls_region_trigger = {
				REGION = title:k_uuii_qingzhou
			}
			hd_cao_expansion_controls_region_trigger = {
				REGION = title:k_uuii_xuzhou
			}
			has_character_flag = hd_cao_expansion_menu_pending
			exists = scope:hd4_cao_menu_round
			var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
		}
		ai_chance = {
			base = 100
		}
		if = {
			limit = {
				has_character_flag = hd_cao_expansion_menu_pending
				exists = scope:hd4_cao_menu_round
				var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
			}
			remove_character_flag = hd_cao_expansion_menu_pending
			add_character_flag = hd_cao_expansion_finished
			custom_tooltip = hd_cao_expansion.finished_tt
		}
	}
	option = {
		name = hd4_close
		trigger = {
			NOT = {
				AND = {
					has_character_flag = hd_cao_expansion_menu_pending
					exists = scope:hd4_cao_menu_round
					var:hd_cao_expansion_finished_round = scope:hd4_cao_menu_round
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
}
