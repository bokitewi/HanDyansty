hd_hanfl.1001 = {
	type = character_event
	title = hd_hanfl.1001.t
	desc = hd_hanfl.1001.desc
	theme = administrative
	override_background = { reference = tgp_study_asia }
	left_portrait = { character = root animation = pondering }
	trigger = { hd_hanfl_kaoke_event_trigger = yes }
	immediate = { hd_hanfl_kaoke_snapshot_effect = yes }

	option = {
		name = hd_hanfl.1001.a
		show_as_unavailable = { always = yes }
		trigger = {
			hd_hanfl_kaoke_choice_valid_trigger = yes
			gold >= 100
		}
		if = {
			limit = {
				hd_hanfl_kaoke_choice_valid_trigger = yes
				gold >= 100
			}
			remove_short_term_gold = 100
			add_stress = 15
			scope:wj_kaoke_duty_county = {
				add_county_modifier = { modifier = hd_hanfl_kaoke_clean_modifier years = 3 }
			}
			custom_tooltip = hd_hanfl.kaoke.good.tt
			custom_tooltip = hd_hanfl.1001.src
			hidden_effect = { WJ_kaoke_choice_effect = { POINTS = 20 } }
		}
		ai_chance = {
			base = 25
			modifier = { factor = 2 has_trait = honest }
			modifier = { factor = 2 has_trait = just }
			modifier = { factor = 0 gold < 100 }
		}
	}

	option = {
		name = hd_hanfl.1001.b
		trigger = { hd_hanfl_kaoke_choice_valid_trigger = yes }
		if = {
			limit = { hd_hanfl_kaoke_choice_valid_trigger = yes }
			custom_tooltip = hd_hanfl.kaoke.normal.tt
			hidden_effect = { WJ_kaoke_choice_effect = { POINTS = 0 } }
		}
		ai_chance = {
			base = 40
			modifier = { factor = 2 has_trait = lazy }
		}
	}

	option = {
		name = hd_hanfl.1001.c
		trigger = { hd_hanfl_kaoke_choice_valid_trigger = yes }
		if = {
			limit = { hd_hanfl_kaoke_choice_valid_trigger = yes }
			add_gold = 30
			scope:wj_kaoke_duty_county = { change_county_control = -10 }
			custom_tooltip = hd_hanfl.kaoke.bad.tt
			hidden_effect = { WJ_kaoke_choice_effect = { POINTS = -20 } }
		}
		ai_chance = {
			base = 25
			modifier = { factor = 2 has_trait = deceitful }
			modifier = { factor = 2 has_trait = greedy }
			modifier = { factor = 0.25 has_trait = honest }
		}
	}

	option = {
		name = hd_hanfl.1001.d
		trigger = { hd_hanfl_kaoke_choice_valid_trigger = yes }
		if = {
			limit = { hd_hanfl_kaoke_choice_valid_trigger = yes }
			duel = {
				skills = { stewardship intrigue }
				value = decent_skill_rating
				50 = {
					desc = hd_hanfl.1001.d.success
					compare_modifier = { value = scope:duel_value multiplier = 3.5 }
					scope:wj_kaoke_duty_county = { change_county_control = 5 }
					custom_tooltip = hd_hanfl.kaoke.duel_success.tt
					hidden_effect = { WJ_kaoke_choice_effect = { POINTS = 10 } }
				}
				50 = {
					desc = hd_hanfl.1001.d.failure
					compare_modifier = { value = scope:duel_value multiplier = -3.5 }
					add_stress = 20
					custom_tooltip = hd_hanfl.kaoke.duel_failure.tt
					hidden_effect = { WJ_kaoke_choice_effect = { POINTS = -10 } }
				}
			}
