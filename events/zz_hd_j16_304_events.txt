# E1-H10 大陵之战：刘渊是玩家时。Root：刘渊。a 出兵 / b 暂缓（90 天后再问）
hd_j16.0300 = {
	type = character_event
	title = hd_j16.0300.t
	desc = hd_j16.0300.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }
	trigger = {
		is_alive = yes
		NOT = { has_global_variable = hd_j16_daling_done }
	}

	# 起五部之众，击腾于大陵
	option = {
		name = hd_j16.0300.a
		if = {
			limit = { has_global_variable = hd_j16_daling_asked }
			remove_global_variable = hd_j16_daling_asked
		}
		set_global_variable = hd_j16_daling_go
		save_scope_as = hd_j16_dl_liu
		if = {
			limit = { exists = scope:hd_j16_dl_teng }
			clear_saved_scope = hd_j16_dl_teng
		}
		hd_j16_resolve_effect = { ID = si_ma_teng AS = hd_j16_dl_teng }
		if = {
			limit = {
				exists = scope:hd_j16_dl_teng
				NOT = { has_global_variable = hd_j16_daling_done }    # 待答期内窗口已关、节点已跳过时不再开战
				NOT = { has_global_variable = hd_j16_daling_started }
				NOT = { is_at_war_with = scope:hd_j16_dl_teng }
			}
			hd_j16_daling_war_effect = yes
		}
		ai_chance = { base = 100 }
	}
	# 且按兵观望
	option = {
		name = hd_j16.0300.b
		if = {
			limit = { has_global_variable = hd_j16_daling_asked }
			remove_global_variable = hd_j16_daling_asked
		}
		set_global_variable = { name = hd_j16_daling_wait value = yes days = 90 }
		debug_log = "HD_J16 e1_h10 WAIT player_deferred"
		ai_chance = { base = 0 }
	}
}
