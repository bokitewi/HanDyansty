hd263.0100 = {
	type = character_event
	title = hd263.0100.t
	desc = hd263.0100.desc
	theme = war
	left_portrait = { character = root animation = personality_cynical }
	right_portrait = {
		character = character:yingchuan_zhong_024
		animation = personality_bold
		trigger = { character:yingchuan_zhong_024 ?= { is_alive = yes } }
	}
	trigger = { is_alive = yes }
	# 三道并进
	option = {
		name = hd263.0100.a

	}
}
