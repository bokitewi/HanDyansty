# Yuan Shao arrives in Bohai and opens the historical coalition chain.
hd_anti_regent.0005 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:yuan_shao
		is_alive = yes
		has_character_flag = hd_yuan_bohai_migration_pending
		NOT = { has_character_flag = hd_yuan_bohai_migration_completed }
		var:hd_yuan_early_story_phase = flag:hd_yuan_phase_migration
		OR = {
			NOT = { exists = global_var:hd_anti_regent_pending_leader }
			global_var:hd_anti_regent_pending_leader = root
		}
	}
	on_trigger_fail = {
		if = {
			limit = { is_alive = no }
			hd_anti_regent_select_fallback_leader_effect = yes
		}
		else_if = {
			limit = {
				has_character_flag = hd_yuan_bohai_migration_pending
				NOT = { has_character_flag = hd_yuan_bohai_migration_completed }
				var:hd_yuan_early_story_phase = flag:hd_yuan_phase_migration
			}
			trigger_event = { id = hd_anti_regent.0005 months = 1 }
		}
	}
	immediate = {
		remove_character_flag = hd_yuan_bohai_migration_pending
		add_character_flag = hd_yuan_bohai_migration_started
		if = {
			limit = { NOT = { exists = global_var:hd_anti_regent_pending_leader } }
			set_global_variable = {
				name = hd_anti_regent_pending_leader
				value = root
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_yuan_bohai_change
			add_claim_on_loss = no
		}
		title:d_bohai = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:hd_yuan_bohai_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_yuan_bohai_change
		hd_force_warlord_effect = yes
		remove_character_flag = hd_yuan_bohai_migration_started
		add_character_flag = hd_yuan_bohai_migration_completed
		add_character_flag = hd_yuan_bohai_arrived
		add_character_flag = hd_anti_regent_dispatch_ready
		add_character_flag = hd_anti_regent_creation_pending
		set_variable = { name = hd_yuan_early_story_phase value = flag:hd_yuan_phase_reorganization }
		if = {
			limit = {
				NOT = { has_character_flag = hd_yuan_bohai_reorganization_pending }
				NOT = { has_character_flag = hd_yuan_bohai_reorganization_received }
			}
			add_character_flag = hd_yuan_bohai_reorganization_pending
			trigger_event = { id = hd_anti_regent.0009 days = 1 }
		}
		if = {
			limit = { hd_anti_regent_forced_creation_ready_trigger = yes }
			add_character_flag = hd_anti_regent_creation_delay_scheduled
			trigger_event = { id = hd_anti_regent.0001 months = 3 }
		}
		else = { trigger_event = { id = hd_anti_regent.0001 months = 1 } }
	}
}

# Invitation dispatch one month after the alliance is announced.
hd_anti_regent.0002 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		has_character_flag = hd_anti_regent_league_leader
		has_variable_list = hd_anti_regent_members
		has_character_flag = hd_anti_regent_creation_started
		has_variable = hd_anti_regent_target
		var:hd_anti_regent_target = { is_alive = yes }
		has_character_flag = hd_anti_regent_invitations_pending
		NOT = { has_character_flag = hd_anti_regent_invitations_dispatched }
	}
	on_trigger_fail = {
		if = {
			limit = {
				is_alive = yes
				has_character_flag = hd_anti_regent_creation_started
				has_variable_list = hd_anti_regent_members
				has_variable = hd_anti_regent_target
				var:hd_anti_regent_target = { is_alive = yes }
				NOT = { has_character_flag = hd_anti_regent_invitations_dispatched }
			}
			add_character_flag = hd_anti_regent_invitations_pending
			trigger_event = { id = hd_anti_regent.0002 months = 1 }
		}
		else = { hd_anti_regent_failure_effect = yes }
	}
	immediate = {
		save_scope_as = hd_inviting_leader
		var:hd_anti_regent_target = { save_scope_as = hd_anti_regent_target_scope }
		set_variable = { name = hd_anti_regent_pending_invites value = 0 }
		remove_character_flag = hd_anti_regent_invitations_pending
		add_character_flag = hd_anti_regent_invitations_dispatched
		if = {
			limit = { title:e_han.holder ?= { is_alive = yes } }
			title:e_han.holder = { save_scope_as = hd_anti_regent_emperor }
		}
		else_if = {
			limit = { title:h_china.holder ?= { is_alive = yes } }
			title:h_china.holder = { save_scope_as = hd_anti_regent_emperor }
		}
		if = {
			limit = { exists = scope:hd_anti_regent_emperor }
			scope:hd_anti_regent_emperor = {
				every_vassal_or_below = {
				limit = {
					is_alive = yes
					is_landed = yes
					highest_held_title_tier >= tier_duchy
					capital_province = { geographical_region = uuii_empire_han_region }
					NOT = { this = scope:hd_inviting_leader }
					NOT = { this = scope:hd_anti_regent_target_scope }
					NOT = { this = character:cao_cao }
					NOT = { has_title = title:h_china }
					NOT = { has_title = title:e_han }
					NOT = { is_vassal_or_below_of = scope:hd_anti_regent_target_scope }
					NOT = { has_character_flag = hd_anti_regent_league_member }
				}
					scope:hd_inviting_leader = { change_variable = { name = hd_anti_regent_pending_invites add = 1 } }
					set_variable = { name = hd_anti_regent_inviter value = scope:hd_inviting_leader }
					trigger_event = { id = hd_anti_regent.0003 days = { 1 3 } }
				}
			}
		}
		every_independent_ruler = {
			limit = {
				is_alive = yes
				is_landed = yes
				highest_held_title_tier >= tier_duchy
				capital_province = { geographical_region = uuii_empire_han_region }
				NOT = { this = scope:hd_inviting_leader }
				NOT = { this = scope:hd_anti_regent_target_scope }
				NOT = { this = character:cao_cao }
				NOT = { has_title = title:h_china }
				NOT = { has_title = title:e_han }
				NOT = { is_vassal_or_below_of = scope:hd_anti_regent_target_scope }
				NOT = { has_character_flag = hd_anti_regent_league_member }
			}
			scope:hd_inviting_leader = { change_variable = { name = hd_anti_regent_pending_invites add = 1 } }
			set_variable = { name = hd_anti_regent_inviter value = scope:hd_inviting_leader }
			trigger_event = { id = hd_anti_regent.0003 days = { 1 3 } }
		}
		hd_anti_regent_try_finish_invites_effect = yes
	}
}

# Second-war victory.  Titles are distributed by the native war contribution
# value recorded by the CB before it opens this event.
hd_anti_regent.0200 = {
	type = character_event
	title = hd_anti_regent.0200.t
	desc = hd_anti_regent.0200.desc
	theme = crown
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		is_alive = yes
		has_character_flag = hd_anti_regent_league_leader
		has_variable_list = hd_anti_regent_members
		has_variable = hd_anti_regent_target
		exists = global_var:hd_anti_regent_league_leader
		global_var:hd_anti_regent_league_leader = root
	}
	on_trigger_fail = { hd_anti_regent_failure_effect = yes }

	immediate = {
		save_scope_as = hd_spoils_distributor
		var:hd_anti_regent_target = {
			every_held_title = {
				limit = {
					is_landless_type_title = no
					tier >= tier_county
					NOT = { this = title:e_han }
					NOT = { this = title:h_china }
					NOT = { is_noble_family_title = yes }
				}
				add_to_list = hd_anti_regent_spoils
			}
		}
		every_in_list = {
			variable = hd_anti_regent_members
			if = {
				limit = { NOT = { has_variable = hd_anti_regent_contribution } }
				set_variable = { name = hd_anti_regent_contribution value = 0 }
			}
		}
		every_in_list = {
			variable = hd_anti_regent_members
			remove_character_flag = hd_anti_regent_spoils_round
		}
		ordered_in_list = {
			list = hd_anti_regent_spoils
			order_by = tier
			save_temporary_scope_as = hd_spoils_title
			remove_variable = hd_spoils_recipient_found
			scope:hd_spoils_distributor = {
				ordered_in_list = {
					variable = hd_anti_regent_members
					order_by = var:hd_anti_regent_contribution
					max = 1
					limit = {
						is_alive = yes
						var:hd_anti_regent_contribution > 0
						NOT = { has_character_flag = hd_anti_regent_spoils_round }
						scope:hd_spoils_title = {
							NOT = { has_variable = hd_spoils_recipient_found }
						}
					}
					save_scope_as = hd_spoils_recipient
					scope:hd_spoils_title = { set_variable = { name = hd_spoils_recipient_found value = yes } }
				}
			}
			if = {
				limit = { NOT = { has_variable = hd_spoils_recipient_found } }
				scope:hd_spoils_distributor = {
					every_in_list = {
						variable = hd_anti_regent_members
						remove_character_flag = hd_anti_regent_spoils_round
					}
				}
				scope:hd_spoils_distributor = {
					ordered_in_list = {
						variable = hd_anti_regent_members
						order_by = var:hd_anti_regent_contribution
						max = 1
						limit = {
							is_alive = yes
							var:hd_anti_regent_contribution > 0
							scope:hd_spoils_title = {
								NOT = { has_variable = hd_spoils_recipient_found }
							}
						}
						save_scope_as = hd_spoils_recipient
						scope:hd_spoils_title = { set_variable = { name = hd_spoils_recipient_found value = yes } }
					}
				}
			}
			if = {
				limit = { has_variable = hd_spoils_recipient_found }
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_spoils_change }
				scope:hd_spoils_title = {
					change_title_holder_include_vassals = {
						holder = scope:hd_spoils_recipient
						change = scope:hd_spoils_change
						take_baronies = no
					}
				}
				resolve_title_and_vassal_change = scope:hd_spoils_change
			}
			else = {
				# If all league contributors have been exhausted, the selected
				# league leader receives the remainder so no spoil is lost.
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_spoils_fallback_change }
				scope:hd_spoils_title = {
					change_title_holder_include_vassals = {
						holder = scope:hd_spoils_distributor
						change = scope:hd_spoils_fallback_change
						take_baronies = no
					}
				}
				resolve_title_and_vassal_change = scope:hd_spoils_fallback_change
			}
			if = {
				limit = { has_variable = hd_spoils_recipient_found }
				scope:hd_spoils_recipient = { add_character_flag = hd_anti_regent_spoils_round }
			}
			remove_variable = hd_spoils_recipient_found
		}
		every_in_list = {
			variable = hd_anti_regent_members
			remove_character_flag = hd_anti_regent_spoils_round
		}
		hd_restore_han_after_anti_regent_victory_effect = yes
		set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
		hd_anti_regent_cleanup_effect = yes
	}

	option = { name = hd_anti_regent.0200.a ai_chance = { base = 100 } }
}
