hd_anti_regent.0003 = {
    type = character_event title = hd_anti_regent.0003.t desc = hd_anti_regent.0003.desc theme = war
    left_portrait = { character = root animation = personality_bold }
    trigger = {
        is_alive = yes is_landed = yes
        hd_anti_regent_invitation_rank_allowed_trigger = yes
        has_character_flag = hd_anti_regent_invitation_pending
        hd_anti_regent_invitation_liege_allowed_trigger = yes
        NOT = { exists = global_var:hd_anti_regent_chain_resolved }
        NOT = { has_character_flag = hd_anti_regent_invitation_accepted }
        NOT = { has_character_flag = hd_anti_regent_invitation_resolved }
        has_variable = hd_anti_regent_inviter
        var:hd_anti_regent_inviter = { is_alive = yes has_character_flag = hd_anti_regent_league_leader has_variable = hd_anti_regent_target }
    }
    on_trigger_fail = {
        if = {
            limit = { OR = { NOT = { has_character_flag = hd_anti_regent_invitation_accepted } hd_anti_regent_invitation_rank_allowed_trigger = no } }
            hd_anti_regent_invitation_resolved_effect = yes
        }
    }
    immediate = {
        var:hd_anti_regent_inviter = { save_scope_as = hd_anti_regent_inviter_scope }
        if = { limit = { var:hd_anti_regent_inviter = { has_variable = hd_anti_regent_target } } var:hd_anti_regent_inviter = { var:hd_anti_regent_target = { save_scope_as = hd_anti_regent_target_scope } } }
        if = { limit = { title:h_china.holder ?= { is_alive = yes } } title:h_china.holder = { save_scope_as = hd_anti_regent_emperor } }
    }
    option = {
        name = hd_anti_regent.0003.a
        if = { limit = { hd_anti_regent_invitation_rank_allowed_trigger = yes } add_character_flag = hd_anti_regent_invitation_accepted hd_anti_regent_try_join_effect = yes }
        else = { hd_anti_regent_invitation_resolved_effect = yes }
        ai_chance = {
            base = 40
            modifier = { add = 45 opinion = { target = scope:hd_anti_regent_inviter_scope value >= 50 } }
            modifier = { add = 30 has_trait = brave }
            modifier = { add = 25 has_trait = ambitious }
            modifier = { add = -50 has_trait = craven }
            modifier = { add = -40 opinion = { target = scope:hd_anti_regent_inviter_scope value <= -30 } }
            modifier = { add = 50 opinion = { target = scope:hd_anti_regent_target_scope value <= -50 } }
        }
    }
    option = {
        name = hd_anti_regent.0003.b hd_anti_regent_invitation_resolved_effect = yes
        ai_chance = {
            base = 35
            modifier = { add = 45 opinion = { target = scope:hd_anti_regent_target_scope value >= 40 } }
            modifier = { add = 30 opinion = { target = scope:hd_anti_regent_emperor value >= 40 } }
            modifier = { add = 20 has_trait = content }
            modifier = { add = 20 has_trait = loyal }
        }
    }
    option = {
        name = hd_anti_regent.0003.c hd_anti_regent_invitation_resolved_effect = yes
        ai_chance = {
            base = 25 modifier = { add = 30 has_trait = ambitious }
            modifier = { add = 25 has_trait = deceitful } modifier = { add = 25 has_trait = craven } modifier = { add = -30 has_trait = loyal }
        }
    }
}
