hd_anti_regent.0003 = {
	type = character_event
	title = hd_anti_regent.0003.t
	desc = hd_anti_regent.0003.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		is_alive = yes
		is_landed = yes
		has_character_flag = hd_anti_regent_invitation_pending
		NOT = { exists = global_var:hd_anti_regent_chain_resolved }
		NOT = { has_character_flag = hd_anti_regent_invitation_accepted }
		NOT = { has_character_flag = hd_anti_regent_invitation_resolved }
		has_variable = hd_anti_regent_inviter
		var:hd_anti_regent_inviter = {
			is_alive = yes
			has_character_flag = hd_anti_regent_league_leader
			has_variable = hd_anti_regent_target
		}
	}
	on_trigger_fail = {
		if = {
			limit = { NOT = { has_character_flag = hd_anti_regent_invitation_accepted } }
			hd_anti_regent_invitation_resolved_effect = yes
		}
	}

	immediate = {
		var:hd_anti_regent_inviter = { save_scope_as = hd_anti_regent_inviter_scope }
		if = {
			limit = { var:hd_anti_regent_inviter = { has_variable = hd_anti_regent_target } }
			var:hd_anti_regent_inviter = {
				var:hd_anti_regent_target = { save_scope_as = hd_anti_regent_target_scope }
			}
		}
		if = {
			limit = { title:e_han.holder ?= { is_alive = yes } }
			title:e_han.holder = { save_scope_as = hd_anti_regent_emperor }
		}
		else_if = {
			limit = { title:h_china.holder ?= { is_alive = yes } }
			title:h_china.holder = { save_scope_as = hd_anti_regent_emperor }
		}
	}

	option = {
		name = hd_anti_regent.0003.a
		add_character_flag = hd_anti_regent_invitation_accepted
		hd_anti_regent_try_join_effect = yes
		ai_chance = {
			base = 40
			modifier = { add = 45 opinion = { target = scope:hd_anti_regent_inviter_scope value >= 50 } }
			modifier = { add = 30 has_trait = brave }
			modifier = { add = 25 has_trait = ambitious }
			modifier = { add = -50 has_trait = craven }
			modifier = { add = -40 opinion = { target = scope:hd_anti_regent_inviter_scope value <= -30 } }
			modifier = { add = 50 opinion = { target = scope:hd_anti_regent_target_scope value <= -50 } }
		}
	}

	option = {
		name = hd_anti_regent.0003.b
		hd_anti_regent_invitation_resolved_effect = yes
		ai_chance = {
			base = 35
			modifier = { add = 45 opinion = { target = scope:hd_anti_regent_target_scope value >= 40 } }
			modifier = { add = 30 opinion = { target = scope:hd_anti_regent_emperor value >= 40 } }
			modifier = { add = 20 has_trait = content }
			modifier = { add = 20 has_trait = loyal }
		}
	}

	option = {
		name = hd_anti_regent.0003.c
		hd_anti_regent_invitation_resolved_effect = yes
		ai_chance = {
			base = 25
			modifier = { add = 30 has_trait = ambitious }
			modifier = { add = 25 has_trait = deceitful }
			modifier = { add = 25 has_trait = craven }
			modifier = { add = -30 has_trait = loyal }
		}
	}
}

# Second-war victory.  Titles are distributed by the native war contribution
# value recorded by the CB before it opens this event.
hd_anti_regent.0200 = {
	type = character_event
	title = hd_anti_regent.0200.t
	desc = hd_anti_regent.0200.desc
	theme = crown
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		global_var:hd_anti_regent_stage = 4
		NOT = { exists = global_var:hd_anti_regent_chain_resolved }
		is_alive = yes
		has_character_flag = hd_anti_regent_league_leader
		has_variable_list = hd_anti_regent_members
		is_member_of_confederation = global_var:hd_anti_regent_active_confederation
		has_variable = hd_anti_regent_target
		exists = global_var:hd_anti_regent_league_leader
		global_var:hd_anti_regent_league_leader = root
	}
	on_trigger_fail = {
		remove_character_flag = hd_anti_regent_settlement_queued
		hd_anti_regent_queue_maintenance_effect = yes
	}

	immediate = {
		save_scope_as = hd_spoils_distributor
		var:hd_anti_regent_target = {
			every_held_title = {
				limit = {
					is_landless_type_title = no
					tier >= tier_county
					NOT = { this = title:e_han }
					NOT = { this = title:h_china }
					NOT = { is_noble_family_title = yes }
				}
				add_to_list = hd_anti_regent_spoils
			}
		}
		every_in_list = {
			variable = hd_anti_regent_members
			if = {
				limit = { NOT = { has_variable = hd_anti_regent_contribution } }
				set_variable = { name = hd_anti_regent_contribution value = 0 }
			}
		}
		every_in_list = {
			variable = hd_anti_regent_members
			remove_character_flag = hd_anti_regent_spoils_round
		}
		ordered_in_list = {
			list = hd_anti_regent_spoils
			order_by = tier
			save_temporary_scope_as = hd_spoils_title
			remove_variable = hd_spoils_recipient_found
			scope:hd_spoils_distributor = {
				ordered_in_list = {
					variable = hd_anti_regent_members
					order_by = var:hd_anti_regent_contribution
					max = 1
					limit = {
						is_alive = yes
						var:hd_anti_regent_contribution > 0
						NOT = { has_character_flag = hd_anti_regent_spoils_round }
						scope:hd_spoils_title = {
							NOT = { has_variable = hd_spoils_recipient_found }
						}
					}
					save_scope_as = hd_spoils_recipient
					scope:hd_spoils_title = { set_variable = { name = hd_spoils_recipient_found value = yes } }
				}
			}
			if = {
				limit = { NOT = { has_variable = hd_spoils_recipient_found } }
				scope:hd_spoils_distributor = {
					every_in_list = {
						variable = hd_anti_regent_members
						remove_character_flag = hd_anti_regent_spoils_round
					}
				}
				scope:hd_spoils_distributor = {
					ordered_in_list = {
						variable = hd_anti_regent_members
						order_by = var:hd_anti_regent_contribution
						max = 1
						limit = {
							is_alive = yes
							var:hd_anti_regent_contribution > 0
							scope:hd_spoils_title = {
								NOT = { has_variable = hd_spoils_recipient_found }
							}
						}
						save_scope_as = hd_spoils_recipient
						scope:hd_spoils_title = { set_variable = { name = hd_spoils_recipient_found value = yes } }
					}
				}
			}
			if = {
				limit = { has_variable = hd_spoils_recipient_found }
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_spoils_change }
				scope:hd_spoils_title = {
					change_title_holder_include_vassals = {
						holder = scope:hd_spoils_recipient
						change = scope:hd_spoils_change
						take_baronies = no
					}
				}
				resolve_title_and_vassal_change = scope:hd_spoils_change
			}
			else = {
				# If all league contributors have been exhausted, the selected
				# league leader receives the remainder so no spoil is lost.
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_spoils_fallback_change }
				scope:hd_spoils_title = {
					change_title_holder_include_vassals = {
						holder = scope:hd_spoils_distributor
						change = scope:hd_spoils_fallback_change
						take_baronies = no
					}
				}
				resolve_title_and_vassal_change = scope:hd_spoils_fallback_change
			}
			if = {
				limit = { has_variable = hd_spoils_recipient_found }
				scope:hd_spoils_recipient = { add_character_flag = hd_anti_regent_spoils_round }
			}
			remove_variable = hd_spoils_recipient_found
		}
		every_in_list = {
			variable = hd_anti_regent_members
			remove_character_flag = hd_anti_regent_spoils_round
		}
		hd_restore_han_after_anti_regent_victory_effect = yes
		debug_log = "HD_COALITION TERMINAL_FINAL_VICTORY"
		set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
		random_owned_story = {
			type = hd_story_yuan_life
			set_variable = { name = hd_yuan_life_phase value = flag:hd_yuan_life_victory }
		}
		character:cao_cao ?= {
			random_owned_story = {
				type = hd_story_cao_life
				set_variable = { name = hd_cao_life_phase value = flag:hd_cao_life_victory }
			}
		}
		character:dong_zhuo_2 ?= {
			random_owned_story = {
				type = hd_story_dong_zhuo_life
				set_variable = { name = hd_dong_zhuo_life_phase value = flag:hd_dong_zhuo_life_defeated }
			}
		}
		hd_anti_regent_cleanup_effect = yes
	}

	option = { name = hd_anti_regent.0200.a ai_chance = { base = 100 } }
}
