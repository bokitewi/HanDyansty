# Yuan Shao arrives in Bohai and opens the historical coalition chain.
hd_anti_regent.0005 = {
	type = character_event
	hidden = yes
	immediate = {
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_yuan_bohai_change
			add_claim_on_loss = no
		}
		title:d_bohai = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:hd_yuan_bohai_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_yuan_bohai_change
		hd_force_independent_warlord_effect = yes
		add_character_flag = hd_yuan_bohai_arrived
		random_owned_story = {
			type = hd_story_yuan_life
			set_variable = { name = hd_yuan_life_phase value = flag:hd_yuan_life_bohai }
		}
		trigger_event = { id = hd_anti_regent.0009 days = 1 }
	}
}

hd_anti_regent.0003 = {
	type = character_event
	title = hd_anti_regent.0003.t
	desc = hd_anti_regent.0003.desc
	theme = war
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		is_alive = yes
		is_landed = yes
		has_character_flag = hd_anti_regent_invitation_pending
		NOT = { has_character_flag = hd_anti_regent_invitation_resolved }
		has_variable = hd_anti_regent_inviter
		var:hd_anti_regent_inviter = {
			is_alive = yes
			has_character_flag = hd_anti_regent_league_leader
			has_variable_list = hd_anti_regent_members
		}
	}
	on_trigger_fail = {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				has_character_flag = hd_anti_regent_invitation_pending
				character:cao_cao ?= {
					is_alive = yes
					has_character_flag = hd_anti_regent_takeover_pending
				}
			}
			set_variable = { name = hd_anti_regent_inviter value = character:cao_cao }
			trigger_event = { id = hd_anti_regent.0003 days = 3 }
		}
		else = { hd_anti_regent_invitation_resolved_effect = yes }
	}

	immediate = {
		var:hd_anti_regent_inviter = { save_scope_as = hd_anti_regent_inviter_scope }
		if = {
			limit = { var:hd_anti_regent_inviter = { has_variable = hd_anti_regent_target } }
			var:hd_anti_regent_inviter = {
				var:hd_anti_regent_target = { save_scope_as = hd_anti_regent_target_scope }
			}
		}
		if = {
			limit = { title:e_han.holder ?= { is_alive = yes } }
			title:e_han.holder = { save_scope_as = hd_anti_regent_emperor }
		}
		else_if = {
			limit = { title:h_china.holder ?= { is_alive = yes } }
			title:h_china.holder = { save_scope_as = hd_anti_regent_emperor }
		}
		if = {
			limit = { hd_is_forced_anti_regent_member_trigger = yes }
			if = {
				limit = { is_independent_ruler = no }
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = hd_anti_regent_forced_independence
					add_claim_on_loss = no
				}
				becomes_independent = { change = scope:hd_anti_regent_forced_independence }
				resolve_title_and_vassal_change = scope:hd_anti_regent_forced_independence
			}
			remove_character_flag = hd_anti_regent_invitation_pending
			add_character_flag = hd_anti_regent_invitation_resolved
			add_character_flag = hd_anti_regent_join_pending
			trigger_event = { id = hd_anti_regent.0010 days = 2 }
		}
	}

	option = {
		name = hd_anti_regent.0003.a
		if = {
			limit = { NOT = { hd_is_forced_anti_regent_member_trigger = yes } }
			if = {
				limit = { is_independent_ruler = no }
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = hd_anti_regent_member_independence
					add_claim_on_loss = no
				}
				becomes_independent = { change = scope:hd_anti_regent_member_independence }
				resolve_title_and_vassal_change = scope:hd_anti_regent_member_independence
			}
			remove_character_flag = hd_anti_regent_invitation_pending
			add_character_flag = hd_anti_regent_invitation_resolved
			add_character_flag = hd_anti_regent_join_pending
			trigger_event = { id = hd_anti_regent.0010 days = 2 }
		}
		ai_chance = {
			base = 40
			modifier = { add = 45 opinion = { target = scope:hd_anti_regent_inviter_scope value >= 50 } }
			modifier = { add = 30 has_trait = brave }
			modifier = { add = 25 has_trait = ambitious }
			modifier = { add = -50 has_trait = craven }
			modifier = { add = -40 opinion = { target = scope:hd_anti_regent_inviter_scope value <= -30 } }
			modifier = { add = 50 opinion = { target = scope:hd_anti_regent_target_scope value <= -50 } }
		}
	}

	option = {
		name = hd_anti_regent.0003.b
		trigger = { NOT = { hd_is_forced_anti_regent_member_trigger = yes } }
		add_character_flag = hd_anti_regent_invitation_opposed
		hd_anti_regent_invitation_resolved_effect = yes
		ai_chance = {
			base = 35
			modifier = { add = 45 opinion = { target = scope:hd_anti_regent_target_scope value >= 40 } }
			modifier = { add = 30 opinion = { target = scope:hd_anti_regent_emperor value >= 40 } }
			modifier = { add = 20 has_trait = content }
			modifier = { add = 20 has_trait = loyal }
		}
	}

	option = {
		name = hd_anti_regent.0003.c
		trigger = { NOT = { hd_is_forced_anti_regent_member_trigger = yes } }
		if = {
			limit = { is_independent_ruler = no }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = hd_anti_regent_neutral_independence
				add_claim_on_loss = no
			}
			becomes_independent = { change = scope:hd_anti_regent_neutral_independence }
			resolve_title_and_vassal_change = scope:hd_anti_regent_neutral_independence
		}
		add_character_flag = hd_anti_regent_invitation_neutral
		hd_anti_regent_invitation_resolved_effect = yes
		ai_chance = {
			base = 25
			modifier = { add = 30 has_trait = ambitious }
			modifier = { add = 25 has_trait = deceitful }
			modifier = { add = 25 has_trait = craven }
			modifier = { add = -30 has_trait = loyal }
		}
	}
}

# Accepted rulers become independent first.  This delayed node joins the real
# confederation only after the title-and-vassal transaction has resolved.
hd_anti_regent.0010 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		is_landed = yes
		is_independent_ruler = yes
		has_character_flag = hd_anti_regent_join_pending
		has_variable = hd_anti_regent_inviter
		var:hd_anti_regent_inviter = {
			is_alive = yes
			has_character_flag = hd_anti_regent_league_leader
			exists = confederation
			confederation = { confederation_type = confederation_type:hd_anti_regent_league_type }
		}
	}
	on_trigger_fail = {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				has_character_flag = hd_anti_regent_join_pending
				exists = global_var:hd_anti_regent_league_leader
				global_var:hd_anti_regent_league_leader = { is_alive = yes }
			}
			global_var:hd_anti_regent_league_leader = { save_temporary_scope_as = hd_anti_regent_retry_leader }
			set_variable = { name = hd_anti_regent_inviter value = scope:hd_anti_regent_retry_leader }
			trigger_event = { id = hd_anti_regent.0010 days = 3 }
		}
		else_if = {
			limit = {
				is_alive = yes
				is_landed = yes
				has_character_flag = hd_anti_regent_join_pending
				character:cao_cao ?= {
					is_alive = yes
					has_character_flag = hd_anti_regent_takeover_pending
				}
			}
			set_variable = { name = hd_anti_regent_inviter value = character:cao_cao }
			trigger_event = { id = hd_anti_regent.0010 days = 3 }
		}
		else = {
			remove_character_flag = hd_anti_regent_join_pending
			remove_variable = hd_anti_regent_inviter
		}
	}
	immediate = {
		var:hd_anti_regent_inviter = { save_scope_as = hd_anti_regent_join_leader }
		scope:hd_anti_regent_join_leader.confederation = { save_scope_as = hd_anti_regent_join_league }
		if = {
			limit = {
				exists = confederation
				confederation = { NOT = { this = scope:hd_anti_regent_join_league } }
			}
			confederation = { remove_confederation_member = root }
		}
		if = {
			limit = { NOT = { is_member_of_confederation = scope:hd_anti_regent_join_league } }
			scope:hd_anti_regent_join_league = { add_confederation_member = root }
		}
		scope:hd_anti_regent_join_leader = {
			if = {
				limit = {
					NOT = {
						is_target_in_variable_list = {
							name = hd_anti_regent_members
							target = root
						}
					}
				}
				add_to_variable_list = { name = hd_anti_regent_members target = root }
			}
		}
		add_character_flag = hd_anti_regent_league_member
		add_character_flag = hd_anti_regent_invitation_member
		remove_character_flag = hd_anti_regent_join_pending
		remove_variable = hd_anti_regent_inviter
		if = {
			limit = { this = character:cao_cao }
			random_owned_story = {
				type = hd_story_cao_life
				set_variable = { name = hd_cao_life_phase value = flag:hd_cao_life_coalition }
			}
		}
		if = {
			limit = {
				scope:hd_anti_regent_join_leader = {
					OR = {
						has_character_flag = hd_anti_regent_first_war_started
						has_character_flag = hd_anti_regent_second_war_started
					}
				}
			}
			scope:hd_anti_regent_join_leader = { hd_anti_regent_call_members_to_all_offensive_wars_effect = yes }
		}
	}
}

# Yuan Shao's pre-war death hands the historical coalition to Cao Cao.  The
# actual transfer waits until Cao Cao's guaranteed Chenliu settlement resolves.
hd_anti_regent.0011 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:cao_cao
		is_alive = yes
		has_character_flag = hd_anti_regent_takeover_pending
	}
	on_trigger_fail = { hd_anti_regent_silent_close_effect = yes }
	immediate = {
		if = {
			limit = { NOT = { has_title = title:d_chenliu } }
			remove_character_flag = hd_cao_chenliu_migration_completed
			set_variable = { name = hd_cao_early_story_phase value = flag:hd_cao_phase_migration }
			hd_cao_move_to_chenliu_effect = yes
			trigger_event = { id = hd_anti_regent.0011 days = 3 }
		}
		else = {
			hd_force_independent_warlord_effect = yes
			remove_character_flag = hd_anti_regent_takeover_pending
			if = {
				limit = {
					exists = confederation
					confederation = { NOT = { confederation_type = confederation_type:hd_anti_regent_league_type } }
				}
				confederation = { remove_confederation_member = root }
			}
			every_confederation = {
				limit = { confederation_type = confederation_type:hd_anti_regent_league_type }
				save_scope_as = hd_existing_anti_regent_league
			}
			if = {
				limit = {
					exists = scope:hd_existing_anti_regent_league
					NOT = { is_member_of_confederation = scope:hd_existing_anti_regent_league }
				}
				scope:hd_existing_anti_regent_league = { add_confederation_member = root }
			}
			else_if = {
				limit = { NOT = { exists = scope:hd_existing_anti_regent_league } }
				create_confederation = {
					name = HD_ANTI_REGENT_LEAGUE_NAME
					type = confederation_type:hd_anti_regent_league_type
				}
			}
			add_character_flag = hd_anti_regent_league_leader
			add_character_flag = hd_anti_regent_league_member
			set_global_variable = { name = hd_anti_regent_league_leader value = this }
			if = {
				limit = { NOT = { has_variable = hd_anti_regent_target } }
				set_variable = { name = hd_anti_regent_target value = character:dong_zhuo_2 }
			}
			if = {
				limit = {
					NOT = {
						is_target_in_variable_list = {
							name = hd_anti_regent_members
							target = root
						}
					}
				}
				add_to_variable_list = { name = hd_anti_regent_members target = root }
			}
			if = {
				limit = { NOT = { has_character_flag = hd_anti_regent_creation_started } }
				add_character_flag = hd_anti_regent_creation_pending
				add_character_flag = hd_anti_regent_creation_delay_scheduled
				trigger_event = { id = hd_anti_regent.0001 days = 1 }
			}
			else_if = {
				limit = { NOT = { has_character_flag = hd_anti_regent_invitations_dispatched } }
				add_character_flag = hd_anti_regent_invitations_pending
				trigger_event = { id = hd_anti_regent.0002 days = 1 }
			}
		}
	}
}

# Second-war victory.  Titles are distributed by the native war contribution
# value recorded by the CB before it opens this event.
hd_anti_regent.0200 = {
	type = character_event
	title = hd_anti_regent.0200.t
	desc = hd_anti_regent.0200.desc
	theme = crown
	left_portrait = { character = root animation = personality_bold }

	trigger = {
		is_alive = yes
		has_character_flag = hd_anti_regent_league_leader
		has_variable_list = hd_anti_regent_members
		has_variable = hd_anti_regent_target
		exists = global_var:hd_anti_regent_league_leader
		global_var:hd_anti_regent_league_leader = root
	}
	on_trigger_fail = { hd_anti_regent_failure_effect = yes }

	immediate = {
		save_scope_as = hd_spoils_distributor
		var:hd_anti_regent_target = {
			every_held_title = {
				limit = {
					is_landless_type_title = no
					tier >= tier_county
					NOT = { this = title:e_han }
					NOT = { this = title:h_china }
					NOT = { is_noble_family_title = yes }
				}
				add_to_list = hd_anti_regent_spoils
			}
		}
		every_in_list = {
			variable = hd_anti_regent_members
			if = {
				limit = { NOT = { has_variable = hd_anti_regent_contribution } }
				set_variable = { name = hd_anti_regent_contribution value = 0 }
			}
		}
		every_in_list = {
			variable = hd_anti_regent_members
			remove_character_flag = hd_anti_regent_spoils_round
		}
		ordered_in_list = {
			list = hd_anti_regent_spoils
			order_by = tier
			save_temporary_scope_as = hd_spoils_title
			remove_variable = hd_spoils_recipient_found
			scope:hd_spoils_distributor = {
				ordered_in_list = {
					variable = hd_anti_regent_members
					order_by = var:hd_anti_regent_contribution
					max = 1
					limit = {
						is_alive = yes
						var:hd_anti_regent_contribution > 0
						NOT = { has_character_flag = hd_anti_regent_spoils_round }
						scope:hd_spoils_title = {
							NOT = { has_variable = hd_spoils_recipient_found }
						}
					}
					save_scope_as = hd_spoils_recipient
					scope:hd_spoils_title = { set_variable = { name = hd_spoils_recipient_found value = yes } }
				}
			}
			if = {
				limit = { NOT = { has_variable = hd_spoils_recipient_found } }
				scope:hd_spoils_distributor = {
					every_in_list = {
						variable = hd_anti_regent_members
						remove_character_flag = hd_anti_regent_spoils_round
					}
				}
				scope:hd_spoils_distributor = {
					ordered_in_list = {
						variable = hd_anti_regent_members
						order_by = var:hd_anti_regent_contribution
						max = 1
						limit = {
							is_alive = yes
							var:hd_anti_regent_contribution > 0
							scope:hd_spoils_title = {
								NOT = { has_variable = hd_spoils_recipient_found }
							}
						}
						save_scope_as = hd_spoils_recipient
						scope:hd_spoils_title = { set_variable = { name = hd_spoils_recipient_found value = yes } }
					}
				}
			}
			if = {
				limit = { has_variable = hd_spoils_recipient_found }
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_spoils_change }
				scope:hd_spoils_title = {
					change_title_holder_include_vassals = {
						holder = scope:hd_spoils_recipient
						change = scope:hd_spoils_change
						take_baronies = no
					}
				}
				resolve_title_and_vassal_change = scope:hd_spoils_change
			}
			else = {
				# If all league contributors have been exhausted, the selected
				# league leader receives the remainder so no spoil is lost.
				create_title_and_vassal_change = { type = conquest save_scope_as = hd_spoils_fallback_change }
				scope:hd_spoils_title = {
					change_title_holder_include_vassals = {
						holder = scope:hd_spoils_distributor
						change = scope:hd_spoils_fallback_change
						take_baronies = no
					}
				}
				resolve_title_and_vassal_change = scope:hd_spoils_fallback_change
			}
			if = {
				limit = { has_variable = hd_spoils_recipient_found }
				scope:hd_spoils_recipient = { add_character_flag = hd_anti_regent_spoils_round }
			}
			remove_variable = hd_spoils_recipient_found
		}
		every_in_list = {
			variable = hd_anti_regent_members
			remove_character_flag = hd_anti_regent_spoils_round
		}
		hd_restore_han_after_anti_regent_victory_effect = yes
		set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
		if = {
			limit = { this = character:yuan_shao }
			random_owned_story = {
				type = hd_story_yuan_life
				set_variable = { name = hd_yuan_life_phase value = flag:hd_yuan_life_victory }
			}
		}
		else_if = {
			limit = { this = character:cao_cao }
			random_owned_story = {
				type = hd_story_cao_life
				set_variable = { name = hd_cao_life_phase value = flag:hd_cao_life_victory }
			}
		}
		character:cao_cao ?= {
			random_owned_story = {
				type = hd_story_cao_life
				set_variable = { name = hd_cao_life_phase value = flag:hd_cao_life_victory }
			}
		}
		character:dong_zhuo_2 ?= {
			random_owned_story = {
				type = hd_story_dong_zhuo_life
				set_variable = { name = hd_dong_zhuo_life_phase value = flag:hd_dong_zhuo_life_defeated }
			}
		}
		hd_anti_regent_cleanup_effect = yes
	}

	option = { name = hd_anti_regent.0200.a ai_chance = { base = 100 } }
}
