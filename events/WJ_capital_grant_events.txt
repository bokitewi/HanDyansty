namespace = WJ_capital_grant

WJ_capital_grant.1 = {
 type = character_event
 title = WJ_capital_appointment_title
 desc = WJ_capital_appointment_desc
 theme = administrative
 widget = { gui = "WJ_capital_selection" container = "custom_widgets_container" }
 immediate = {
  save_scope_as = actor
  clear_variable_list = wj_capital_candidates
  remove_variable = wj_capital_selected
  # Stored title is immutable for this appointment; migration invalidates it.
  set_variable = { name = wj_capital_pending_title value = scope:WJ_capital_grant_selected_title }
  every_courtier_or_guest = { WJ_capital_collect_candidate = yes }
  every_vassal_or_below = { WJ_capital_collect_candidate = yes }
 }
 option = {
  name = WJ_capital_appoint_confirm
  trigger = { exists = var:wj_capital_selected }
  show_as_unavailable = { always = yes }
  if = {
   limit = { exists = var:wj_capital_selected }
   var:wj_capital_selected = { save_scope_as = recipient }
   WJ_capital_grant_effect = yes
  }
 }
 option = { name = WJ_capital_appoint_cancel }
 after = {
  clear_variable_list = wj_capital_candidates
  remove_variable = wj_capital_selected
  remove_variable = wj_capital_pending_title
 }
}
