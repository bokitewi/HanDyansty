hd_party_debate.0010 = {
	type = activity_event
	title = hd_party_debate.0010.t
	desc = hd_party_debate.0010.desc
	theme = learning
	left_portrait = {
		character = root
		animation = scheme
	}
	trigger = {
		exists = involved_activity
	}
	immediate = {
		hd_party_debate_event_setup_effect = yes
	}
	option = { # 联络同党
		name = hd_party_debate.0010.a
		trigger = { gold >= 50 }
		remove_short_term_gold = 50
		custom_tooltip = hd_party_debate.0010.a.tt
		hd_party_debate_progress_effect = { VALUE = 5 }
		scope:activity = {
			every_guest_subset = {
				name = hd_debate_neutral
				random = {
					chance = 30
					hd_party_debate_set_side_effect = { SIDE = support }
				}
			}
		}
	}
	option = { # 拉拢中立
		name = hd_party_debate.0010.b
		duel = {
			skill = diplomacy
			value = 12
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 }
				desc = hd_party_debate.0010.b.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = hd_party_debate.0010.b.success
					scope:activity = {
						random_guest_subset = {
							name = hd_debate_neutral
							hd_party_debate_set_side_effect = { SIDE = support }
						}
					}
				}
			}
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 }
				desc = hd_party_debate.0010.b.fail
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = hd_party_debate.0010.b.fail
				}
			}
		}
	}
	option = { # 散布流言
		name = hd_party_debate.0010.c
		duel = {
			skill = intrigue
			value = 12
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 }
				desc = hd_party_debate.0010.c.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = hd_party_debate.0010.c.success
					scope:activity = {
						random_guest_subset = {
							name = hd_debate_oppose
							limit = { NOT = { this = scope:host } }
							hd_party_debate_set_side_effect = { SIDE = neutral }
						}
					}
				}
			}
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 }
				desc = hd_party_debate.0010.c.fail
				send_interface_toast = {
					type = event_toast_effect_bad
					title = hd_party_debate.0010.c.fail
					add_prestige = -100
				}
			}
		}
	}
}

				}
			}
		}
		else = {
			hd_party_debate_progress_effect = { VALUE = 10 }
		}
	}
}

# 决胜之辩：学识 + 外交
hd_party_debate.0030 = {
	type = activity_event
	title = hd_party_debate.0030.t
	desc = hd_party_debate.0030.desc
	theme = learning
	left_portrait = { character = root animation = debating }
	right_portrait = { character = scope:hd_party_debate_rival animation = debating }
	trigger = { exists = involved_activity }
	immediate = { hd_party_debate_event_setup_effect = yes }
	option = {
		name = hd_party_debate.0030.a
		if = {
			limit = { exists = scope:hd_party_debate_rival }
			save_scope_value_as = {
				name = hd_party_debate_edge
				value = {
					value = learning
					add = diplomacy
					subtract = scope:hd_party_debate_rival.learning
					subtract = scope:hd_party_debate_rival.diplomacy
				}
			}
			random_list = {
				50 = {
					modifier = { add = { value = scope:hd_party_debate_edge multiply = 3 } }
					desc = hd_party_debate.0030.success
					hd_party_debate_progress_effect = { VALUE = 15 }
				}
				50 = {
					modifier = { add = { value = scope:hd_party_debate_edge multiply = -3 } }
					desc = hd_party_debate.0030.fail
					hd_party_debate_progress_effect = { VALUE = -15 }
				}
			}
		}
		else = {
			hd_party_debate_progress_effect = { VALUE = 15 }
		}
	}
}
