hd_22_xiu_join_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_ai = yes
				is_ruler = no
				is_imprisoned = no
				NOT = { has_character_flag = hd_22_xiu_joined }
				NOT = { any_consort = { is_ruler = yes } }
				OR = {
					NOT = { exists = court_owner }
					court_owner = { is_ai = yes }
					court_owner ?= character:liu_xiu
				}
				character:liu_xiu ?= { is_alive = yes is_landed = yes }
			}
			add_character_flag = hd_22_xiu_joined
			if = {
				limit = { NOT = { court_owner ?= character:liu_xiu } }
				character:liu_xiu = { add_courtier = character:$WHO$ }
			}
			set_knight_status = force
			debug_log = "HD_22 XIU JOINED $WHO$"
		}
	}
}

# 夺郡、交还南阳诸县、改投、治所。scope 不限，人物一律用 character:ID
hd_22_xiu_north_land_effect = {
	character:liu_xiu = { save_scope_as = hd_22_xiu }
	# 夺郡：持有者及其最高主君都是电脑
	if = {
		limit = {
			exists = title:d_anping.holder
			title:d_anping.holder = {
				is_ai = yes
				NOT = { this = character:liu_xiu }
				top_liege = { is_ai = yes }
			}
		}
		title:d_anping.holder = { save_scope_as = hd_22_old_anping }
		hd_22_give_duchy_effect = { DUCHY = title:d_anping TO = scope:hd_22_xiu KEEP = title:c_uuii_p_1191 }
		title:d_anping = {
			if = {
				limit = { has_variable = hd_dc_taoni_rebel }
				remove_variable = hd_dc_taoni_rebel   # 讨逆标记不随郡转给刘秀（计划复核必改 2）
			}
		}
		scope:hd_22_old_anping = {
			if = {
				limit = { is_alive = yes is_landed = no is_ai = yes }
				scope:hd_22_xiu = { add_courtier = scope:hd_22_old_anping }
			}
		}
		if = {
			limit = { title:d_anping.holder ?= scope:hd_22_xiu }
			set_global_variable = hd_22_xiu_got_xindu
		}
	}
	if = {
		limit = { exists = global_var:hd_22_xiu_got_xindu }
		scope:hd_22_xiu = {
			if = {
				limit = { exists = liege NOT = { liege = this } }
				# 交还安平郡以外的县（湖阳、昆阳等）给直属主君
				liege = { save_scope_as = hd_22_xiu_old_liege }
				create_title_and_vassal_change = { type = granted save_scope_as = hd_22_xiu_return_change add_claim_on_loss = no }
				every_held_title = {
					limit = {
						tier = tier_county
						NOT = { de_jure_liege ?= title:d_anping }
					}
					change_title_holder = { holder = scope:hd_22_xiu_old_liege change = scope:hd_22_xiu_return_change take_baronies = yes }
				}
				resolve_title_and_vassal_change = scope:hd_22_xiu_return_change
				# 改投最高主君（同一国内挪动；愤然离去后不改投，免得投到刚结仇的更始手下）
				if = {
					limit = {
						NOT = { exists = global_var:hd_22_xiu_rage }
						exists = top_liege
						NOT = { top_liege = this }
						NOT = { liege = top_liege }
					}
					top_liege = { save_scope_as = hd_22_xiu_top }
					hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_22_xiu_top }
				}
				# 主君只有郡级（与刘秀同级）：不留同级封臣，提前自立（电脑立即；玩家下次心跳问 0220）
				if = {
					limit = {
						exists = liege
						NOT = { liege = this }
						liege = { highest_held_title_tier <= tier_duchy }
					}
					remove_global_variable = hd_22_xiu_g4_wait
					if = {
						limit = { is_ai = yes }
						hd_22_xiu_independence_effect = yes
					}
					debug_log = "HD_22 XIU G3 LIEGE_TIER_TOO_LOW_EARLY_INDEPENDENCE"
				}
			}
			if = {
				limit = { title:c_uuii_p_4226.holder ?= this }
				set_realm_capital = title:c_uuii_p_4226
			}
		}
		debug_log = "HD_22 XIU G3 NORTH_WITH_XINDU"
	}
	else = { debug_log = "HD_22 XIU G3 NORTH_NO_XINDU" }
}

	if = {
		limit = {

# 自立：放人 → 独立 → 通知或停战 → 清讨逆标记 → 萧王 → 征服者
hd_22_xiu_independence_effect = {
	set_global_variable = hd_22_xiu_g4_done
	set_global_variable = hd_22_xiu_independent
	hd_22_xiu_release_effect = yes
	character:liu_xiu = {
		save_scope_as = hd_22_xiu
		if = {
			limit = { exists = liege NOT = { liege = this } }
			liege = { save_scope_as = hd_22_xiu_was_liege }
			if = {
				limit = { exists = top_liege NOT = { top_liege = liege } }
				top_liege = { save_scope_as = hd_22_xiu_was_top }
			}
			hd_nbc_become_independent_effect = yes
		}
	}
	# 最高主君是玩家（直属主君另有其人）时也通知（执行后复核必改 2）
	scope:hd_22_xiu_was_top ?= {
		if = {
			limit = { is_ai = no }
			trigger_event = hd_22.0225
		}
	}
	if = {
		limit = { exists = scope:hd_22_xiu_was_liege }
		scope:hd_22_xiu_was_liege = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_22.0225   # 通知玩家主君：刘秀辞召自立
			}
			else_if = {
				# 电脑主君（多为更始）：互给停战约一年，史实上两人到 25 年才在河内交兵（计划复核建议 2）
				limit = { scope:hd_22_xiu = { is_ai = yes } NOT = { is_at_war_with = scope:hd_22_xiu } }
				add_truce_both_ways = {
					character = scope:hd_22_xiu
					days = 365
					name = hd_22_truce
				}
			}
		}
	}
	scope:hd_22_xiu = {
		if = {
			limit = { exists = primary_title }
			primary_title = {
				if = {
					limit = { has_variable = hd_dc_taoni_rebel }
					remove_variable = hd_dc_taoni_rebel
				}
			}
		}
		# 萧王：主头衔是安平郡时改名「萧」，称谓显示为「萧王」
		if = {
			limit = { primary_title ?= title:d_anping }
			title:d_anping = { set_title_name = hd_22_xiao_title }
			set_global_variable = hd_22_xiu_xiao_named
		}
		hd_nbc_style_effect = { WHO = liu_xiu STYLE = wang }
		hd_rise_ensure_story_effect = yes
		# 征服者（boki 要求自立即给）：电脑走原版 conqueror.0001（起步金与威望、特质、故事、极端规则修正）；
		# 玩家直接照该事件的接受选项给，不弹窗（计划复核拍板 1，按 boki 原话）
		if = {
			limit = {
				NOR = {
					has_trait = conqueror
					any_owned_story = { story_type = story_conqueror }
				}
			}
			if = {
				limit = { is_ai = yes }
				trigger_event = conqueror.0001
			}
			else = {
				if = {
					limit = { NOT = { has_game_rule = no_conquerors_bonuses } }
					add_trait = conqueror
				}
				create_story = story_conqueror
			}
		}
	}
	debug_log = "HD_22 XIU G4 INDEPENDENT_CONQUEROR"
}
