WJ_pr_begin = {
 custom_tooltip = WJ_pr_begin_tt
 hidden_effect = {
 if = { limit = { WJ_pr_can_petition = { KIND = $KIND$ } }
  save_scope_as = wj_pr_actor
  set_variable = { name = wj_pr_emperor value = top_liege }
  set_variable = { name = wj_pr_kind value = flag:$KIND$ }
  if = { limit = { NOT = { has_variable = wj_pr_ticket } } set_variable = { name = wj_pr_ticket value = 0 } }
  change_variable = { name = wj_pr_ticket add = 1 }
  save_scope_value_as = { name = wj_pr_ticket value = var:wj_pr_ticket }
  set_variable = { name = wj_pr_recent_emperor value = top_liege years = 1 }
  top_liege = { set_variable = { name = wj_pr_petitioner value = scope:wj_pr_actor } }
  WJ_pr_grant = yes
 }
}
}
WJ_pr_end = {
 save_scope_as = wj_pr_ending
 var:wj_pr_emperor ?= {
  if = { limit = { exists = var:wj_pr_petitioner var:wj_pr_petitioner = scope:wj_pr_ending } remove_variable = wj_pr_petitioner }
 }
 remove_variable = wj_pr_emperor remove_variable = wj_pr_kind
 remove_variable = wj_pr_method remove_variable = wj_pr_chance
 remove_variable = wj_pr_authorized_emperor remove_variable = wj_pr_authorized_kind
}
# Successful political consent/failed resistance uses the original award exactly once.
WJ_pr_grant = {
 if = { limit = { WJ_pr_live = yes WJ_pr_political_qualification = yes WJ_pr_item_requirements = yes }
  set_variable = { name = wj_pr_authorized_emperor value = var:wj_pr_emperor }
  set_variable = { name = wj_pr_authorized_kind value = var:wj_pr_kind }
  # No permanent control flag and no native puppet for a player.
  switch = {
   trigger = var:wj_pr_kind
   flag:duke = { save_scope_as = wj_rf_recipient WJ_rf_begin = yes }
   flag:hereditary = {
    title:e_WJ_minister_cheng_xiang = { add_title_law = noble_family_succession_law }
    send_interface_toast = { type = event_generic_good title = WJ_cx_hereditary_decision custom_tooltip = WJ_cx_hereditary_done_tt }
   }
   flag:king = { WJ_rf_promote = yes }
   flag:abdicate = { WJ_ab_begin = yes }
  }
 }
 else = { send_interface_toast = { type = event_generic_neutral title = WJ_pr_invalid_title custom_tooltip = WJ_pr_invalid_tt } }
 WJ_pr_end = yes
}
# Compatibility for already queued old resistance results: no probability or window.
WJ_pr_resolve = { WJ_pr_grant = yes }
