# $ATTACKER$ 对 $DEFENDER$ 宣战（hd_89_war_cb），主目标 $TARGET$，场次 $KIND$。成功则存为 scope:hd_89_war_$KIND$
hd_89_start_war_effect = {
	$ATTACKER$ = {
		if = {
			limit = { is_allied_to = $DEFENDER$ }
			break_alliance_due_to_attack_effect = { TARGET = $DEFENDER$ }
		}
		# 开局给的停战与上一场的停战都挡不住脚本宣战，这里再设一次破停战旗标（照 22 通则）
		add_character_flag = { flag = allowed_war_declaration_sans_truce_penalties days = 1 }
	}
	# 目标经连通筛选（boki 10-18，zz_hd_war_contiguity_effects.txt）
	hd_wct_add_input_effect = { TARGET = $TARGET$ }
	hd_wct_start_war_effect = { CB = hd_89_war_cb ATTACKER = $ATTACKER$ DEFENDER = $DEFENDER$ }
	hd_wct_clear_input_effect = yes
	$ATTACKER$ = {
		random_character_war = {
			limit = {
				using_cb = hd_89_war_cb
				primary_attacker = $ATTACKER$
				primary_defender = $DEFENDER$
				NOT = { has_variable = hd_89_war_kind }
			}
			set_variable = { name = hd_89_war_kind value = flag:$KIND$ }
			save_scope_as = hd_89_war_$KIND$
		}
	}
	if = {
		limit = { exists = scope:hd_89_war_$KIND$ }
		set_global_variable = hd_89_$KIND$_active
		debug_log = "HD_89 WAR_CONFIRMED $KIND$"
	}
	else = {
		debug_log = "HD_89 WAR_START_FAILED $KIND$"
	}
}

# 封臣出征（照 hd_nbc_vassal_war_effect，复核建议 9）：先由本人宣战，不成则由其最高领主宣战、本人参战。$WHO$ 是 saved scope
hd_89_vassal_war_effect = {
	hd_89_start_war_effect = { ATTACKER = $WHO$ DEFENDER = $DEFENDER$ TARGET = $TARGET$ KIND = $KIND$ }
	if = {
		limit = {
			NOT = { exists = scope:hd_89_war_$KIND$ }
			$WHO$ = { is_independent_ruler = no }
			$WHO$.top_liege = { is_ai = yes }
		}
		$WHO$.top_liege = { save_scope_as = hd_89_att_$KIND$ }
		hd_89_start_war_effect = { ATTACKER = scope:hd_89_att_$KIND$ DEFENDER = $DEFENDER$ TARGET = $TARGET$ KIND = $KIND$ }
		scope:hd_89_war_$KIND$ ?= {
			if = {
				limit = { NOT = { is_participant = $WHO$ } }
				add_attacker = $WHO$
			}
		}
	}
	else_if = {
		limit = {
			NOT = { exists = scope:hd_89_war_$KIND$ }
			$WHO$ = { is_independent_ruler = no }
		}
		# 最高领主是玩家：不替他宣战，下次心跳再试（执行后复核必改 2）
		debug_log = "HD_89 VASSAL_WAR_SKIPPED_PLAYER_LIEGE $KIND$"
	}
}

hd_89_war_result_kind_effect = {
	if = {
		limit = { var:hd_89_war_kind ?= flag:$KIND$ }
		set_global_variable = hd_89_$KIND$_$RESULT$
		if = {
			limit = { has_global_variable = hd_89_$KIND$_active }
			remove_global_variable = hd_89_$KIND$_active
		}
		debug_log = "HD_89 WAR_RESULT $KIND$ $RESULT$"
	}
}

# 被杀、自尽。电脑角色直接死；玩家角色不杀，只记日志
hd_89_die_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ }
			debug_log = "HD_89 DIED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_89 DIE_SKIPPED_PLAYER $WHO$"
		}
	}
}

# 把郡头衔 $DUCHY$ 与 $FROM$（scope）及其封臣手里法理属本郡的各县交给 $TO$（scope）
hd_89_give_duchy_effect = {
	create_title_and_vassal_change = { type = granted save_scope_as = hd_89_duchy_change add_claim_on_loss = no }
	$DUCHY$ = {
		if = {
			limit = { holder ?= $FROM$ }
			change_title_holder = { holder = $TO$ change = scope:hd_89_duchy_change }
		}
		every_de_jure_county = {
			limit = {
				holder ?= {
					OR = {
						this = $FROM$
						target_is_liege_or_above = $FROM$
					}
				}
			}
			change_title_holder = { holder = $TO$ change = scope:hd_89_duchy_change take_baronies = yes }
		}
	}
	resolve_title_and_vassal_change = scope:hd_89_duchy_change
}

	# 宋由（坐窦宪党自杀）、班固（死于洛阳狱中）：前提是 S4 诛窦已执行

		limit = {

				exists = global_var:hd_89_zhudou_executed
				exists = global_var:hd_89_zhudou_done
			}
			current_date >= 92.9.1
		}
		set_global_variable = hd_89_zhudou_deaths_done
		if = {
			limit = { exists = global_var:hd_89_zhudou_executed }
			hd_89_die_effect = { WHO = song_you REASON = death_suicide }
			hd_89_die_effect = { WHO = hd_89_ban_gu REASON = death_dungeon }
		}
		else = { debug_log = "HD_89 ZHUDOU_DEATHS SKIPPED_NOT_EXECUTED" }

		limit = {

			current_date >= 93.7.1
		}
		set_global_variable = hd_89_yuchujian_death_done

			limit = {
				exists = global_var:hd_89_yuchujian_tributary

			}

		}
		else = { debug_log = "HD_89 YUCHUJIAN_DEATH SKIPPED" }
	}

hd_89_yanran_executor_effect = {
	if = {
		limit = { character:hd_89_dou_xian ?= { is_alive = yes is_landed = yes } }
		character:hd_89_dou_xian = { save_scope_as = hd_89_yanran_exec }
	}
	else_if = {
		limit = { character:xingye_deng_007 ?= { is_alive = yes is_landed = yes } }
		character:xingye_deng_007 = { save_scope_as = hd_89_yanran_exec }
	}
	else = {
		title:h_china.holder ?= { save_scope_as = hd_89_yanran_exec }
	}
}
hd_89_check_yanran_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_89_yanran_started }
			NOT = { exists = global_var:hd_89_yanran_done }
			current_date >= 89.6.27
		}
		if = {
			limit = { current_date >= 90.1.19 }
			set_global_variable = hd_89_yanran_done
			debug_log = "HD_89 S1 YANRAN SKIPPED_DEADLINE"
		}
		else_if = {
			limit = {
				NOT = {
					title:d_gobi_ongi.holder.top_liege ?= {
						is_independent_ruler = yes
						NOT = { this = title:h_china.holder }
					}
				}
			}
			debug_log = "HD_89 S1 YANRAN WAIT_NO_TARGET"
		}
		else = {
			hd_89_yanran_executor_effect = yes
			if = {
				limit = { exists = scope:hd_89_yanran_exec }
				if = {
					limit = { scope:hd_89_yanran_exec = { is_ai = no } }
					# 玩家执行人：出塞或暂缓（只问一次，暂缓即跳过）
					if = {
						limit = { NOT = { exists = global_var:hd_89_yanran_asked } }
						set_global_variable = hd_89_yanran_asked
						scope:hd_89_yanran_exec = { trigger_event = hd_89.0110 }
					}
				}
				else = {
					hd_89_yanran_war_effect = yes
				}
			}
			else = { debug_log = "HD_89 S1 YANRAN WAIT_NO_EXECUTOR" }
		}
	}
	# 结算：已开战、且已没有这一场次的战争
	if = {
		limit = {
			exists = global_var:hd_89_yanran_started
			NOT = { exists = global_var:hd_89_yanran_done }
			NOT = { exists = global_var:hd_89_yanran_active }
		}
		set_global_variable = hd_89_yanran_done
		if = {
			limit = { exists = global_var:hd_89_yanran_won }
			hd_89_yanran_victory_effect = yes
		}
		else = { debug_log = "HD_89 S1 YANRAN NOT_WON" }
	}
}
