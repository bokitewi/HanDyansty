hd_370_game_start_effect = {
	# 衣冠南渡 317 年已经发生：与 346、404 一样以对峙阶段开局；第一行就设 hd_yiguan_started
	hd_nbc_yiguan_late_start_effect = yes
	# 南徐（京口）、南兖（广陵）侨郡：nbc 只在 346 分支里预置，370 自己补（南豫已由上面的效果预置）
	if = {
		limit = { exists = global_var:hd_yiguan_late_start }
		hd_nbc_qiaojun_effect = { COUNTY = c_uuii_p_2086 }
		hd_nbc_qiaojun_effect = { COUNTY = c_uuii_p_7062 }
	}
	# 五胡大单于：刘渊、石勒、李特、苻洪、姚弋仲都早已去世
	set_global_variable = hd_wuhu_chanyu_liuyuan_done
	set_global_variable = hd_wuhu_chanyu_shile_done
	set_global_variable = hd_wuhu_chanyu_lite_done
	set_global_variable = hd_wuhu_chanyu_fuhong_done
	set_global_variable = hd_wuhu_chanyu_yaoyizhong_done
	# 僚人入蜀 343–344 年已发生（随机规则下仍放行；617 线程补设）
	if = { limit = { NOT = { has_game_rule = hd_liao_rule_random } } set_global_variable = { name = hd_liao_migration_launched value = yes } }
	# 历史入侵：慕容儁已死（同 346）。慕容垂、姚苌、乞伏国仁、吕光、拓跋珪等淝水以后的规则保留
	set_global_variable = { name = hd_hi_done_murongjun value = yes }
	# 天意青睐"每月主动开战"要求讨董链已结束（184 专用变量），同 304、531
	set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
	# 天子京城：建康
	title:h_china = {
		set_capital_county = title:c_uuii_p_4389
		set_variable = { name = hd_court_capital_county value = title:c_uuii_p_4389 }
	}
	# 传国玉玺归海西公司马奕
	title:h_china.holder ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { exists = global_var:hd_han_treasures_initialized }
			}
			hd_han_create_imperial_seal_effect = { OWNER = this }
			set_global_variable = hd_han_treasures_initialized
		}
	}
	# 天朝循环：对峙阶段，晋金德（nbc 版本身没有日期分支，直接用）
	hd_nbc_dynastic_cycle_effect = yes
	hd_nbc_culture_effect = yes
	hd_370_exia_effect = yes
	hd_370_realm_display_effect = yes
	debug_log = "HD_370 STAGE GAME_START"
}

# 挟天子：桓温为权臣兼丞相，天子固定在建康（照 hd_531_regency_effect，天子取当时的 h_china 持有者，S5 废立后再调一次）。
# Scope：桓温
hd_370_regency_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			title:h_china.holder ?= { is_alive = yes }
			liege ?= title:h_china.holder
		}
		title:h_china.holder = {
			save_scope_as = hd_370_emperor
			set_location = {
				location = title:c_uuii_p_4389.title_province
				stick_to_location = yes
			}
		}
		# 玩家扮演时功名补到帝国级预期，否则丞相授不上（218 实测）
		if = {
			limit = {
				is_ai = no
				merit_level < merit_level_expected_empire_value
			}
			while = {
				limit = { merit_level < merit_level_expected_empire_value }
				change_merit_level = 1
			}
		}
		hd218_grant_chengxiang_effect = yes
		WJ_install_historical_secretariat_effect = { OFFICE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang LAW = WJ_grand_chengxiang_law }
		hd_court_install_regency_effect = yes
		hd_court_set_stance_effect = { STANCE = moni }
		scope:hd_370_emperor = {
			if = {
				limit = { has_diarchy_type = grand_secretariat diarch ?= character:longkang_huan_017 }
				set_diarchy_swing = 100
			}
		}
		# 终身辅政锁：天子罢免后自动补回（hd_dc_kit_ten_regent_watch）
		title:h_china = { set_variable = { name = hd_dc_ten_qjc_regent value = character:longkang_huan_017 } }
		# 权臣任期：标记在就不能废帝（hd_court_depose_ceremonial_decision）。照 253、531 设 12 年
		if = {
			limit = { NOT = { has_character_flag = hd_court_regent_tenure } }
			add_character_flag = { flag = hd_court_regent_tenure years = 12 }
		}
		if = {
			limit = { has_title = title:e_WJ_minister_cheng_xiang }
			debug_log = "HD_370 REGENCY HUAN_WEN OK"
		}
		else = { debug_log = "HD_370 REGENCY HUAN_WEN NO_CHENGXIANG" }
	}
	else = { debug_log = "HD_370 REGENCY SKIPPED" }
}

# S1 洛阳之战：太和四年十二月王猛率步骑三万攻洛阳，开局时正在围城。苻坚作攻方，目标河南郡（金墉）。
# 王猛所部 15,000 在陕（事件兵给宣战人苻坚，由他调遣）；燕乐安王慕容臧的援军 8,000 在荥阳
hd_370_luoyang_war_effect = {
	if = {
		limit = {
			character:lueyang_fu_020 ?= { is_alive = yes is_independent_ruler = yes }
			character:changli_murong_077 ?= { is_alive = yes is_independent_ruler = yes }
			title:d_henan.holder ?= {
				OR = {
					this = character:changli_murong_077
					top_liege = character:changli_murong_077
				}
			}
		}
		hd_nbc_start_war_effect = { ATTACKER = character:lueyang_fu_020 DEFENDER = character:changli_murong_077 TARGET = title:d_henan KIND = luoyang }
		if = {
			limit = { exists = scope:hd_nbc_war_luoyang }
			set_global_variable = hd_370_luoyang_started
			character:lueyang_fu_020 = {
				hd_nbc_spawn_army_effect = { SIZE = 15000 WHERE = title:c_uuii_p_842 NAME = hd_370_wangmeng_army_name }
			}
			character:changli_murong_077 = {
				hd_nbc_spawn_army_effect = { SIZE = 8000 WHERE = title:c_uuii_p_761 NAME = hd_370_murongzang_army_name }
			}
		}
		else = {
			set_global_variable = hd_370_luoyang_done
			debug_log = "HD_370 S1 LUOYANG WAR_START_FAILED"
		}
	}
	else = {
		set_global_variable = hd_370_luoyang_done
		debug_log = "HD_370 S1 LUOYANG SKIPPED"
	}
}
