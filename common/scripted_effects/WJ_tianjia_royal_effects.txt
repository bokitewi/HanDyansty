# Tianjia prince identity retained; no East Palace grant or succession alteration (2026-10-04).
# Remove only our marked East Palace, never unrelated landless/family titles.
WJ_royal_remove_east_palace_effect = {
 every_held_title = {
  limit = { has_variable = SanGong_DongGong_title is_noble_family_title = no }
  holder = { destroy_title = prev }
 }
}
WJ_royal_clear_prince_effect = {
 save_scope_as = wj_clearing_prince
 var:WJ_royal_patron ?= { if = { limit = { var:WJ_royal_heir = scope:wj_clearing_prince } remove_variable = WJ_royal_heir } }
 remove_trait = SanGong_Taizi remove_character_flag = taisun_flag
 remove_variable = WJ_royal_patron
 WJ_royal_remove_east_palace_effect = yes
}
WJ_royal_install_prince_effect = {
 if = {
  limit = { WJ_royal_can_install_prince = yes }
  save_scope_as = HuangDi
  scope:taizi = {
   add_trait = SanGong_Taizi
   set_variable = { name = WJ_royal_patron value = scope:HuangDi }
   create_character_memory = { type = mem_add_taizi }
  }
  set_variable = { name = WJ_royal_heir value = scope:taizi }
  every_vassal_or_below = { limit = { is_ai = no } trigger_event = SanGong_DianZhang.017 }
 }
}
WJ_royal_depose_prince_effect = {
 save_scope_as = HuangDi
 if = {
  limit = { WJ_inner_emperor_trigger = yes exists = scope:taizi scope:taizi = { WJ_royal_crown_prince = yes var:WJ_royal_patron = scope:HuangDi } }
  add_stress = 50 add_prestige = -300 change_influence = -300 add_legitimacy = -100 add_tyranny = 20
  scope:taizi = {
   WJ_royal_clear_prince_effect = yes
   add_prestige_level = -2 add_stress = 150 add_prestige = -300 change_influence = -300
   mother ?= { if = { limit = { is_alive = yes } add_prestige_level = -1 add_stress = 100 } }
  }
  trigger_event = SanGong_DianZhang.018
  every_vassal_or_below = { limit = { is_ai = no } trigger_event = SanGong_DianZhang.018 }
 }
}
