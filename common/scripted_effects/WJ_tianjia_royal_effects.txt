# Ported from Tianjia DianZhang .015/.016 and TaiZiDongGong: emperor -> prince grant.
# Remove only our marked East Palace, never unrelated landless/family titles.
WJ_royal_remove_east_palace_effect = {
 every_held_title = {
  limit = { has_variable = SanGong_DongGong_title is_noble_family_title = no }
  holder = { destroy_title = prev }
 }
}
WJ_royal_clear_prince_effect = {
 save_scope_as = wj_clearing_prince
 var:WJ_royal_patron ?= { if = { limit = { var:WJ_royal_heir = scope:wj_clearing_prince } remove_variable = WJ_royal_heir } }
 remove_trait = SanGong_Taizi remove_character_flag = taisun_flag
 remove_variable = WJ_royal_patron
 WJ_royal_remove_east_palace_effect = yes
}
WJ_royal_install_prince_effect = {
 if = {
  limit = { WJ_royal_can_install_prince = yes }
  save_scope_as = HuangDi
  scope:taizi = { set_variable = WJ_royal_installing }
  if = {
   limit = { scope:taizi = { is_landed = yes } }
   scope:taizi = { force_step_down_landed_titles = yes }
   add_courtier = scope:taizi
  }
  scope:taizi = { WJ_royal_remove_east_palace_effect = yes }
  if = { limit = { primary_title.tier = tier_hegemony } create_dynamic_title = { tier = empire name = TaiZiDG } }
  else = { create_dynamic_title = { tier = kingdom name = TaiZiDG } }
  scope:new_title = { save_scope_as = wj_east_palace }
  create_title_and_vassal_change = { type = created save_scope_as = wj_royal_change add_claim_on_loss = no }
  scope:wj_east_palace = {
   set_coa = scope:HuangDi.primary_title
   set_capital_county = scope:HuangDi.capital_county
   set_landless_title = yes set_destroy_on_succession = yes
   set_variable = SanGong_DongGong_title
   set_variable = { name = titular_title value = yes }
   set_no_automatic_claims = yes set_definitive_form = yes
   set_can_be_named_after_dynasty = no set_can_use_nomadic_naming = no
   change_title_holder = { holder = scope:HuangDi change = scope:wj_royal_change }
  }
  resolve_title_and_vassal_change = scope:wj_royal_change
  create_title_and_vassal_change = { type = granted save_scope_as = wj_royal_grant add_claim_on_loss = no }
  scope:wj_east_palace = { change_title_holder = { holder = scope:taizi change = scope:wj_royal_grant take_baronies = no } }
  resolve_title_and_vassal_change = scope:wj_royal_grant
  scope:taizi = {
   add_trait = SanGong_Taizi
   set_variable = { name = WJ_royal_patron value = scope:HuangDi }
   remove_variable = WJ_royal_installing
   set_primary_title_to = scope:wj_east_palace
   create_character_memory = { type = mem_add_taizi }
  }
  set_variable = { name = WJ_royal_heir value = scope:taizi }
  primary_title = { add_title_law = SanGong_zongfa_succession_law }
  every_vassal_or_below = { limit = { is_ai = no } trigger_event = SanGong_DianZhang.017 }
 }
}
WJ_royal_depose_prince_effect = {
 save_scope_as = HuangDi
 if = {
  limit = { WJ_inner_emperor_trigger = yes exists = scope:taizi scope:taizi = { WJ_royal_crown_prince = yes var:WJ_royal_patron = scope:HuangDi } }
  add_stress = 50 add_prestige = -300 change_influence = -300 add_legitimacy = -100 add_tyranny = 20
  scope:taizi = {
   WJ_royal_clear_prince_effect = yes
   add_prestige_level = -2 add_stress = 150 add_prestige = -300 change_influence = -300
   mother ?= { if = { limit = { is_alive = yes } add_prestige_level = -1 add_stress = 100 } }
  }
  trigger_event = SanGong_DianZhang.018
  every_vassal_or_below = { limit = { is_ai = no } trigger_event = SanGong_DianZhang.018 }
 }
}

# Source TaiSun_effect: a grandson succeeds the deceased prince's designation, not his lands.
WJ_royal_install_grandson = {
 if = { limit = { WJ_inner_emperor_trigger = yes exists = var:WJ_grandson_selected var:WJ_grandson_selected = { WJ_royal_grandson_candidate = { EMPEROR = root } } NOT = { any_child = { is_alive = yes has_trait = SanGong_Taizi } } NOT = { AND = { exists = var:WJ_royal_heir var:WJ_royal_heir = { WJ_royal_crown_prince = yes } } } }
  save_scope_as = HuangDi
  var:WJ_grandson_selected = {
   save_scope_as = taizi add_trait = SanGong_Taizi add_character_flag = taisun_flag
   set_variable = { name = WJ_royal_patron value = scope:HuangDi }
   create_character_memory = { type = mem_add_taizi }
  }
  set_variable = { name = WJ_royal_heir value = scope:taizi }
  primary_title = { add_title_law = SanGong_zongfa_succession_law }
  trigger_event = SanGong_DianZhang.017
 }
}
