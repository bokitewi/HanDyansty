### Character（收留者，事件 hd_wuhu.2000 的 ROOT）：选纠纷伯爵领、族群、胡首
hd_wuhu_frontier_setup_effect = {
	save_scope_as = hd_wuhu_host
	# 本国（含封臣）与胡族伯爵领接壤的伯爵领，优先位于安置区内
	random_realm_county = {
		limit = {
			hd_wuhu_frontier_border_county_trigger = { OWNER = scope:hd_wuhu_host }
			title_province = { geographical_region = hd_wuhu_all_settle_region }
		}
		alternative_limit = {
			hd_wuhu_frontier_border_county_trigger = { OWNER = scope:hd_wuhu_host }
		}
		save_scope_as = hd_wuhu_nomads_county
	}
	scope:hd_wuhu_nomads_county ?= {
		random_neighboring_county = {
			limit = { hd_wuhu_six_culture_trigger = yes }
			save_scope_as = hd_wuhu_tribe_county
		}
		if = {
			limit = {
				exists = holder
				NOT = { holder = scope:hd_wuhu_host }
			}
			holder = { save_scope_as = hd_wuhu_upset_vassal }
		}
	}
	scope:hd_wuhu_tribe_county ?= {
		hd_wuhu_save_tribe_from_culture_effect = yes
	}
	if = {
		limit = { exists = scope:hd_wuhu_tribe }
		hd_wuhu_save_tribe_culture_rite_effect = { COUNTY = scope:hd_wuhu_tribe_county }
		hd_wuhu_create_chief_effect = { LOCATION = scope:hd_wuhu_nomads_county.title_province }
		scope:hd_wuhu_chief = { save_scope_as = turkic_leader }
	}
}

### $A$ / $B$ 两个公爵领中取"受地者兵力较弱"的有效目标，存为 scope:hd_wuhu_petition_target / scope:hd_wuhu_receiver
hd_wuhu_pick_weaker_duchy_effect = {
	$A$.title_capital_county.holder.top_liege ?= { save_scope_as = hd_wuhu_rcv_a }
	$B$.title_capital_county.holder.top_liege ?= { save_scope_as = hd_wuhu_rcv_b }
	if = {
		limit = {
			exists = scope:hd_wuhu_rcv_a
			scope:hd_wuhu_rcv_a = { hd_wuhu_petition_receiver_trigger = yes }
			$A$ = { hd_wuhu_petition_duchy_valid_trigger = { RECEIVER = scope:hd_wuhu_rcv_a } }
		}
		$A$ = { save_scope_as = hd_wuhu_cand_a }
	}
	if = {
		limit = {
			exists = scope:hd_wuhu_rcv_b
			scope:hd_wuhu_rcv_b = { hd_wuhu_petition_receiver_trigger = yes }
			$B$ = { hd_wuhu_petition_duchy_valid_trigger = { RECEIVER = scope:hd_wuhu_rcv_b } }
		}
		$B$ = { save_scope_as = hd_wuhu_cand_b }
	}
	if = {
		limit = {
			exists = scope:hd_wuhu_cand_a
			OR = {
				NOT = { exists = scope:hd_wuhu_cand_b }
				scope:hd_wuhu_rcv_a.current_military_strength <= scope:hd_wuhu_rcv_b.current_military_strength
			}
		}
		scope:hd_wuhu_cand_a = { save_scope_as = hd_wuhu_petition_target }
		scope:hd_wuhu_rcv_a = { save_scope_as = hd_wuhu_receiver }
	}
	else_if = {
		limit = { exists = scope:hd_wuhu_cand_b }
		scope:hd_wuhu_cand_b = { save_scope_as = hd_wuhu_petition_target }
		scope:hd_wuhu_rcv_b = { save_scope_as = hd_wuhu_receiver }
	}
}

### 单个公爵领 $A$ 作为目标

		}

	}
}

### 从 hd_wuhu_duchy_candidates 列表中取受地者兵力最弱的有效公爵领
hd_wuhu_pick_weakest_from_list_effect = {
	# 先剔除无效候选（受地者 = 该公爵领首府持有者的最高宗主）
	every_in_list = {
		list = hd_wuhu_duchy_candidates
		if = {
			limit = { exists = title_capital_county.holder.top_liege }
			title_capital_county.holder.top_liege = { save_temporary_scope_as = hd_wuhu_tmp_receiver }
			if = {
				limit = {
					scope:hd_wuhu_tmp_receiver = { hd_wuhu_petition_receiver_trigger = yes }
					hd_wuhu_petition_duchy_valid_trigger = { RECEIVER = scope:hd_wuhu_tmp_receiver }
				}
				add_to_list = hd_wuhu_valid_duchies
			}
		}
	}
	ordered_in_list = {
		list = hd_wuhu_valid_duchies
		order_by = hd_wuhu_duchy_receiver_weakness
		save_scope_as = hd_wuhu_petition_target
	}
	scope:hd_wuhu_petition_target ?= {
		title_capital_county.holder.top_liege = { save_scope_as = hd_wuhu_receiver }
	}
}

### 同上，但候选列表为 hd_wuhu_region_duchies（通用兜底优先安置区）

					hd_wuhu_petition_duchy_valid_trigger = { RECEIVER = scope:hd_wuhu_tmp_receiver }
				}
				add_to_list = hd_wuhu_valid_duchies
			}
		}
	}
	ordered_in_list = {
		list = hd_wuhu_valid_duchies
		order_by = hd_wuhu_duchy_receiver_weakness
		save_scope_as = hd_wuhu_petition_target
	}
	scope:hd_wuhu_petition_target ?= {
		title_capital_county.holder.top_liege = { save_scope_as = hd_wuhu_receiver }
	}
}

	}
}

### 通用兜底：每年 3%，全局冷却 10 年
hd_wuhu_generic_petition_effect = {
	if = {
		limit = {
			hd_wuhu_enabled_trigger = yes
			hd_wuhu_ai_enabled_trigger = yes
			OR = {
				NOT = { exists = global_var:hd_wuhu_generic_next_year }
				current_year >= global_var:hd_wuhu_generic_next_year
			}
		}
		random = {
			chance = hd_wuhu_generic_petition_yearly_chance
			random_independent_ruler = {
				limit = {
					is_ai = yes
					is_landed = yes
					highest_held_title_tier >= tier_duchy
					hd_wuhu_six_culture_trigger = yes
					any_realm_county = {
						NOT = { target_is_de_jure_liege_or_above = title:h_china }
						any_neighboring_county = {
							target_is_de_jure_liege_or_above = title:h_china
							holder ?= {
								top_liege = { hd_wuhu_petition_receiver_trigger = yes }
							}
						}
					}
				}
				save_scope_as = hd_wuhu_petitioner
			}
			scope:hd_wuhu_petitioner ?= {
				hd_wuhu_save_tribe_from_culture_effect = yes
				# 候选公爵领：与求地者接壤的中原伯爵领所属公爵领
				every_realm_county = {
					limit = { NOT = { target_is_de_jure_liege_or_above = title:h_china } }
					every_neighboring_county = {
						limit = {
							target_is_de_jure_liege_or_above = title:h_china
							exists = duchy
						}
						duchy = {
							if = {
								limit = { NOT = { is_in_list = hd_wuhu_duchy_candidates } }
								add_to_list = hd_wuhu_duchy_candidates
							}
						}
					}
				}
			}
			if = {
				limit = { exists = scope:hd_wuhu_tribe }
				# 优先该族安置区内的
				if = {
					limit = {
						any_in_list = {
							list = hd_wuhu_duchy_candidates
							hd_wuhu_title_in_tribe_region_trigger = { TRIBE = scope:hd_wuhu_tribe }
						}
					}
					# 只保留安置区内的候选
					every_in_list = {
						list = hd_wuhu_duchy_candidates
						limit = {
							hd_wuhu_title_in_tribe_region_trigger = { TRIBE = scope:hd_wuhu_tribe }
						}
						add_to_list = hd_wuhu_region_duchies
					}
					hd_wuhu_pick_weakest_from_region_list_effect = yes
				}
				# 安置区内没有有效目标时，退回任意接壤的中原公爵领
				if = {
					limit = { NOT = { exists = scope:hd_wuhu_petition_target } }
					hd_wuhu_pick_weakest_from_list_effect = yes
				}
			}

hd_wuhu_chanyu_launch_effect = {
	save_scope_as = hd_wuhu_chanyu
	$KINGDOM$ = { save_scope_as = hd_wuhu_chanyu_kingdom }
	# 守方：目标王国内伯爵领最多的独立统治者
	$KINGDOM$ = {
		every_de_jure_county = {
			limit = { exists = holder }
			holder.top_liege = {
				if = {
					limit = { NOT = { is_in_list = hd_wuhu_chanyu_defenders } }
					add_to_temporary_list = hd_wuhu_chanyu_defenders
				}
			}
		}
	}
	ordered_in_list = {
		list = hd_wuhu_chanyu_defenders
		limit = { NOT = { this = scope:hd_wuhu_chanyu } }
		order_by = hd_wuhu_count_in_chanyu_kingdom
		save_scope_as = hd_wuhu_chanyu_defender
	}
	if = {
		limit = { exists = scope:hd_wuhu_chanyu_defender }
		if = {
			limit = { is_landed = no }
			save_scope_as = hd_revolt_leader
			hd_historical_revolt_ensure_adventurer_effect = { LOCATION = $HOME$ }
		}
		set_variable = { name = hd_wuhu_chanyu_name value = flag:$NAME$ }
		set_variable = { name = hd_wuhu_chanyu_kingdom value = $KINGDOM$ }
		start_war = {
			cb = hd_wuhu_chanyu_cb
			target = scope:hd_wuhu_chanyu_defender
			target_title = $KINGDOM$
		}
		random_character_war = {
			limit = {
				primary_attacker = scope:hd_wuhu_chanyu
				using_cb = hd_wuhu_chanyu_cb
			}
			save_scope_as = war
		}
		# 兜底：AI 大单于是守方领国内的封臣、宣战失败时，领地交给继承人，本人以冒险者身份重新起兵
		if = {
			limit = {
				NOT = { exists = scope:war }
				is_ai = yes
				is_landed = yes
				top_liege = scope:hd_wuhu_chanyu_defender
			}
			hd_wuhu_chief_abandon_lands_effect = yes
			save_scope_as = hd_revolt_leader
			hd_historical_revolt_ensure_adventurer_effect = { LOCATION = $HOME$ }
			start_war = {
				cb = hd_wuhu_chanyu_cb
				target = scope:hd_wuhu_chanyu_defender
				target_title = $KINGDOM$
			}
			random_character_war = {
				limit = {
					primary_attacker = scope:hd_wuhu_chanyu
					using_cb = hd_wuhu_chanyu_cb
				}
				save_scope_as = war
			}
		}
		if = {
			limit = { exists = scope:war }
			set_global_variable = { name = hd_wuhu_chanyu_$KEY$_done value = yes }
			if = {
				limit = { exists = capital_province }
				capital_province = { save_scope_as = hd_wuhu_chanyu_army_location }
			}
			else = {
				location = { save_scope_as = hd_wuhu_chanyu_army_location }
			}
			spawn_army = {
				name = hd_wuhu_chanyu_conquer_army_name
				levies = 5000
				men_at_arms = {
					type = horse_archers
					stacks = 15
				}
				men_at_arms = {
					type = light_horsemen
					stacks = 10
				}
				location = scope:hd_wuhu_chanyu_army_location
				uses_supply = no
				war = scope:war
				inheritable = no
				save_scope_as = hd_wuhu_chanyu_army
			}
			spawn_army = {
				name = hd_wuhu_chanyu_hold_army_name
				levies = 2500
				men_at_arms = {
					type = horse_archers
					stacks = 10
				}
				location = scope:hd_wuhu_chanyu_army_location
				uses_supply = no
				inheritable = no
			}
			scope:hd_wuhu_chanyu_army ?= { assign_commander = scope:hd_wuhu_chanyu }
		}
		else = {
			remove_variable = hd_wuhu_chanyu_name
			remove_variable = hd_wuhu_chanyu_kingdom
			# 玩家走决定但宣战失败：退还决定花费
			if = {

			}
		}
	}

		add_prestige = 1000
	}

### CB 胜利：取得/创建目标王国、改国号（ROOT = 大单于，在隐藏事件 hd_wuhu.4001 中执行）
hd_wuhu_chanyu_crown_effect = {
	var:hd_wuhu_chanyu_kingdom = { save_scope_as = hd_wuhu_chanyu_kingdom }
	if = {
		limit = { scope:hd_wuhu_chanyu_kingdom = { is_title_created = no } }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = hd_wuhu_crown_change
			add_claim_on_loss = no
		}
		scope:hd_wuhu_chanyu_kingdom = {
			change_title_holder = {
				holder = root
				change = scope:hd_wuhu_crown_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_wuhu_crown_change
	}
	else_if = {
		limit = {
			NOT = { scope:hd_wuhu_chanyu_kingdom.holder ?= root }
			scope:hd_wuhu_chanyu_kingdom.holder ?= { is_ai = yes }
		}
		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = hd_wuhu_crown_change
			add_claim_on_loss = yes
		}
		scope:hd_wuhu_chanyu_kingdom = {
			change_title_holder = {
				holder = root
				change = scope:hd_wuhu_crown_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_wuhu_crown_change
	}
	if = {
		limit = { has_title = scope:hd_wuhu_chanyu_kingdom }
		if = {
			limit = { highest_held_title_tier <= tier_kingdom }
			set_primary_title_to = scope:hd_wuhu_chanyu_kingdom
		}
		# 国号：直接改名，并写入王朝固定国号（天朝循环 hd_fixed_realm_name 体系）
		set_variable = {
			name = new_realm_title_name
			value = var:hd_wuhu_chanyu_name
		}
		scope:hd_wuhu_chanyu_kingdom = {
			set_title_name_dynamic = new_realm_title_name_from_var
		}
		dynasty ?= {
			remove_variable = hd_fixed_realm_title
			set_variable = {
				name = hd_fixed_realm_name
				value = root.var:hd_wuhu_chanyu_name
			}
		}
		remove_variable = new_realm_title_name
	}
	remove_variable = hd_wuhu_chanyu_name
	remove_variable = hd_wuhu_chanyu_kingdom
}
