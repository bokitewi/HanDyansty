WJ_dd_clear_realm_effect = {
 save_scope_as = wj_dd_cleared_realm
 every_in_list = { variable = wj_dd_realm_cores add_to_list = wj_dd_realm_cleanup }
 # Also discover districts created by the earlier test build before the registry.
 holder ?= {
  every_vassal_or_below = {
   every_held_title = {
    limit = { var:wj_dd_core = this var:wj_dd_realm = scope:wj_dd_cleared_realm }
    add_to_list = wj_dd_realm_cleanup
   }
  }
 }
 every_in_list = {
  list = wj_dd_realm_cleanup
  if = { limit = { var:wj_dd_core = this var:wj_dd_realm = scope:wj_dd_cleared_realm } WJ_dd_dissolve_core_effect = yes }
 }
 clear_variable_list = wj_dd_realm_cores
}
