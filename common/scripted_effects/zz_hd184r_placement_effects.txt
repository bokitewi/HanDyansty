# Scope：首领。
hd184r_hw_adjust_self_effect = {
	save_scope_as = hd_hw_actor
	save_scope_as = hd_hw_recv
	clear_saved_scope = hd_hw_home_liege
	hd184r_hw_build_table_effect = yes
	if = {
		limit = {
			has_variable_list = hd184r_hw_table
			is_alive = yes
			is_ai = yes
			is_landed = yes
			NOT = { hd184r_holds_kingdom_land_trigger = yes }
			NOT = { hd_hw_player_above_trigger = yes }
		}
		# 一、搬迁：首都不在表内郡、且表内首郡可拿时
		clear_saved_scope = hd184r_cap_duchy
		capital_county ?= { duchy ?= { save_scope_as = hd184r_cap_duchy } }
		clear_saved_scope = hd184r_first_duchy
		var:hd184r_hw_first ?= { save_scope_as = hd184r_first_duchy save_scope_as = hd_hw_duchy save_scope_as = hd_hw_cand }
		if = {
			limit = {
				exists = scope:hd184r_first_duchy
				# 2026-10-22 190 讨董：曹操开局在陈留起家（D7），AI 曹操在诛董后也不搬去东郡，保住陈留三县
				NOT = { has_character_flag = hd190_hw_no_relocate }
				OR = {
					NOT = { exists = scope:hd184r_cap_duchy }
					NOT = { is_target_in_variable_list = { name = hd184r_hw_table target = scope:hd184r_cap_duchy } }
				}
				scope:hd184r_first_duchy = {
					NOT = { hd_hw_duchy_owned_trigger = yes }
					hd184r_hw_duchy_usable_trigger = yes
				}
			}
			scope:hd184r_first_duchy = { hd184r_hw_take_duchy_effect = yes }
			if = {
				limit = { scope:hd184r_first_duchy = { hd_hw_duchy_owned_trigger = yes } }
				hd184r_hw_release_outlying_effect = yes
				debug_log = "HD184R P3 RELOCATED"
				debug_log_scopes = yes
			}
		}
		# 二、补足：按表的顺序，已拥有的计数，接壤且可拿的就拿，直到达到表中郡数
		every_in_list = {
			variable = hd184r_hw_table
			save_scope_as = hd_hw_duchy
			save_scope_as = hd_hw_cand
			if = {
				limit = { hd_hw_duchy_owned_trigger = yes }
				scope:hd_hw_actor = { change_variable = { name = hd184r_hw_left add = -1 } }
			}
			else_if = {
				limit = {
					scope:hd_hw_actor.var:hd184r_hw_left > 0
					hd184r_duchy_adjacent_to_actor_trigger = yes
					hd184r_hw_duchy_usable_trigger = yes
				}
				hd184r_hw_take_duchy_effect = yes
				if = {
					limit = { hd_hw_duchy_owned_trigger = yes }
					scope:hd_hw_actor = { change_variable = { name = hd184r_hw_left add = -1 } }
					debug_log = "HD184R P3 TOPUP_DUCHY"
					debug_log_scopes = yes
				}
			}
		}
		# 三、首都不在表内郡时移到表内实际持有的县
		clear_saved_scope = hd184r_cap_duchy
		capital_county ?= { duchy ?= { save_scope_as = hd184r_cap_duchy } }
		if = {
			limit = {
				OR = {
					NOT = { exists = scope:hd184r_cap_duchy }
					NOT = { is_target_in_variable_list = { name = hd184r_hw_table target = scope:hd184r_cap_duchy } }
				}
			}
			clear_saved_scope = hd184r_new_cap
			every_in_list = {
				variable = hd184r_hw_table
				save_scope_as = hd184r_scan_duchy
				if = {
					limit = { NOT = { exists = scope:hd184r_new_cap } }
					scope:hd_hw_actor = {
						random_held_title = {
							limit = { tier = tier_county is_landless_type_title = no duchy ?= scope:hd184r_scan_duchy }
							save_scope_as = hd184r_new_cap
						}
					}
				}
			}
			if = {
				limit = { exists = scope:hd184r_new_cap }
				set_realm_capital = scope:hd184r_new_cap
				set_location = { location = scope:hd184r_new_cap.title_province stick_to_location = yes }
				domicile ?= { move_domicile = scope:hd184r_new_cap.title_province }
			}
		}
		# 已拿地但还不是独立的（原本无地的首领）交由 hd_hw 规则处理；这里不改独立状态。
	}
	clear_variable_list = hd184r_hw_table
	remove_variable = hd184r_hw_left
	remove_variable = hd184r_hw_first
	clear_saved_scope = hd_hw_duchy
	clear_saved_scope = hd_hw_cand
}

# Scope：郡头衔（同时存为 scope:hd_hw_duchy、hd_hw_cand）。需 scope:hd_hw_actor / hd_hw_recv。
# 照 hd_hw_apply_effect 的一、三两步：接郡头衔，夺可夺的县，收编本郡真实县主。
hd184r_hw_take_duchy_effect = {
	title:h_china = { if = { limit = { exists = holder } set_landless_title = yes } }
	if = {
		limit = {
			hd184r_hw_duchy_takeable_trigger = yes
			NOT = { holder ?= scope:hd_hw_recv }
		}
		create_title_and_vassal_change = { type = granted save_scope_as = hd184r_hw_change_a add_claim_on_loss = no }
		change_title_holder = { holder = scope:hd_hw_recv change = scope:hd184r_hw_change_a }
		resolve_title_and_vassal_change = scope:hd184r_hw_change_a
	}
	create_title_and_vassal_change = { type = conquest save_scope_as = hd184r_hw_change_c add_claim_on_loss = no }
	every_in_de_jure_hierarchy = {
		limit = { tier = tier_county hd184r_hw_grab_county_trigger = yes }
		change_title_holder_include_vassals = { holder = scope:hd_hw_actor change = scope:hd184r_hw_change_c }
	}
	resolve_title_and_vassal_change = scope:hd184r_hw_change_c
	scope:hd_hw_actor = { clear_variable_list = hd184r_hw_vassalize }
	every_in_de_jure_hierarchy = {
		limit = {
			tier = tier_county
			holder ?= { hd_hw_vassalize_holder_trigger = yes }
		}
		holder = { save_temporary_scope_as = hd184r_vas_holder }
		scope:hd_hw_actor = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = hd184r_hw_vassalize target = scope:hd184r_vas_holder } } }
				add_to_variable_list = { name = hd184r_hw_vassalize target = scope:hd184r_vas_holder }
			}
		}
	}
	create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd184r_hw_change_d add_claim_on_loss = no }
	scope:hd_hw_actor = {
		every_in_list = {
			variable = hd184r_hw_vassalize
			limit = { hd_hw_vassalize_holder_trigger = yes }
			change_liege = { liege = scope:hd_hw_recv change = scope:hd184r_hw_change_d }
		}
		clear_variable_list = hd184r_hw_vassalize
	}
	resolve_title_and_vassal_change = scope:hd184r_hw_change_d
}

# Scope：首领。表外的县交给新立的本地首领并独立（照 hd_dong_localize_non_jingzhao_direct_counties_effect）；

hd184r_hw_release_outlying_effect = {
	clear_variable_list = hd184r_hw_out_counties
	clear_variable_list = hd184r_hw_out_duchies
	every_held_title = {
		limit = {
			tier = tier_county
			is_landless_type_title = no
			NOT = {
				duchy ?= {
					save_temporary_scope_as = hd184r_out_duchy
					scope:hd_hw_actor = { is_target_in_variable_list = { name = hd184r_hw_table target = scope:hd184r_out_duchy } }
				}
			}
		}
		save_temporary_scope_as = hd184r_out_county
		scope:hd_hw_actor = { add_to_variable_list = { name = hd184r_hw_out_counties target = scope:hd184r_out_county } }
	}
	every_in_list = {
		variable = hd184r_hw_out_counties
		hd184r_localize_county_effect = yes
		debug_log = "HD184R P3 RELOCATE_COUNTY_LOCALIZED"
	}
	every_held_title = {
		limit = {
			tier = tier_duchy
			is_landless_type_title = no
			NOT = { scope:hd_hw_actor = { is_target_in_variable_list = { name = hd184r_hw_table target = prev } } }
			title_capital_county.holder ?= { NOT = { this = scope:hd_hw_actor } is_ai = yes }
		}
		save_temporary_scope_as = hd184r_out_duchy_t
		scope:hd_hw_actor = { add_to_variable_list = { name = hd184r_hw_out_duchies target = scope:hd184r_out_duchy_t } }
	}
	every_in_list = {
		variable = hd184r_hw_out_duchies
		save_temporary_scope_as = hd184r_out_duchy_title
		create_title_and_vassal_change = { type = granted save_scope_as = hd184r_out_duchy_change add_claim_on_loss = no }
		change_title_holder = { holder = scope:hd184r_out_duchy_title.title_capital_county.holder change = scope:hd184r_out_duchy_change }
		resolve_title_and_vassal_change = scope:hd184r_out_duchy_change
	}
	clear_variable_list = hd184r_hw_out_counties
	clear_variable_list = hd184r_hw_out_duchies
}
