# 佛教东传（hd_fo，2026-10-25）：史实事件 E01–E58 的效果与年度调度。设计见 docs/佛教东传_设计草案.md §5、§10。
# 每个事件到年份、开局早于该年、还没做过才触发（hd_fo_due_trigger）；条件不满足时每年重试，满 10 年作废（§10.1 R2）。
# 状态改动都在效果里做完，弹出的事件只负责叙事；有选项的（E19、E32、E35、E42）在选项里改。

########## 共用 ##########

hd_fo_done_effect = {
	set_global_variable = { name = hd_fo_done_$E$ value = yes }
	debug_log = "hd_fo: 事件 $E$"
}

hd_fo_expire_effect = {
	if = {
		limit = { current_year >= $EXP$ }
		set_global_variable = { name = hd_fo_done_$E$ value = yes }
		debug_log = "hd_fo: 事件 $E$ 到期作废"
	}
}

# 主角：县持有者的最高领主
hd_fo_actor_place_effect = {
	if = {
		limit = { NOT = { exists = scope:hd_fo_actor } }
		title:$PLACE$.holder.top_liege ?= {
			if = {
				limit = { is_alive = yes }
				save_scope_as = hd_fo_actor
			}
		}
	}
}

# 主角：指定的史实人物（在世即可）
hd_fo_actor_char_effect = {
	if = {
		limit = { NOT = { exists = scope:hd_fo_actor } }
		character:$ID$ ?= {
			if = {
				limit = { is_alive = yes }
				save_scope_as = hd_fo_actor
			}
		}
	}
}

# 主角：汉天子，不在则洛阳持有者的最高领主
hd_fo_actor_tianzi_effect = {
	title:e_han.holder ?= {
		if = {
			limit = { is_alive = yes }
			save_scope_as = hd_fo_actor
		}
	}
	hd_fo_actor_place_effect = { PLACE = c_luoyang }
}

# 主角：某位高僧所在宫廷的最高领主
hd_fo_actor_monk_liege_effect = {
	global_var:hd_fo_p_$ID$ ?= {
		if = {
			limit = { is_alive = yes exists = court_owner }
			court_owner.top_liege = { save_scope_as = hd_fo_actor }
		}
	}
}

# 直接把高僧送进某人宫廷（史实接纳者，不论是否独立）
hd_fo_give_monk_to_effect = {
	global_var:hd_fo_p_$ID$ ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { is_courtier_of = $WHO$ }
			}
			save_temporary_scope_as = hd_fo_tmp_monk
			$WHO$ = { add_courtier = scope:hd_fo_tmp_monk }
		}
	}
}

# 灭佛（主角作用域，§10.2）：灭佛标记、毁寺降窟、高僧出逃、佛教人物好感 −30
hd_fo_persecute_effect = {
	save_scope_as = hd_fo_persecutor
	add_character_flag = hd_fo_miefo
	hd_fo_persecute_buildings_effect = yes
	every_vassal_or_below = {
		limit = { faith.religion = religion:buddhism_religion }
		add_opinion = { target = scope:hd_fo_persecutor modifier = hd_fo_persecutor_opinion }
	}
	every_courtier = {
		limit = {
			faith.religion = religion:buddhism_religion
			NOT = { has_character_flag = hd_fo_monk }
		}
		add_opinion = { target = scope:hd_fo_persecutor modifier = hd_fo_persecutor_opinion }
	}
	every_living_character = {
		limit = { has_character_flag = hd_fo_monk }
		add_opinion = { target = scope:hd_fo_persecutor modifier = hd_fo_persecutor_opinion }
		if = {
			limit = {
				exists = court_owner
				court_owner.top_liege ?= scope:hd_fo_persecutor
			}
			hd_fo_monk_flee_effect = yes
		}
	}
	debug_log = "hd_fo: 灭佛"
}

# 高僧出逃（高僧作用域）：按第 13 项退路去最近不敌视佛教的君主

	clear_saved_scope = hd_fo_kingdom
}
