    scope:wj_rf_picker = {

      NOT = { is_target_in_variable_list = { name = wj_rf_choices target = scope:wj_rf_list_title } }

 }
}

WJ_rf_transfer_duchy = {

  set_variable = { name = wj_rf_original_liege value = de_jure_liege }
 }

  if = { limit = { NOT = { is_target_in_variable_list = { name = wj_rf_duchies target = scope:wj_rf_transfer_duchy } } }
   add_to_variable_list = { name = wj_rf_duchies target = scope:wj_rf_transfer_duchy }
  }
 }
 if = { limit = { exists = holder holder.top_liege = scope:wj_rf_sovereign holder = { highest_held_title_tier < tier_kingdom NOT = { this = scope:wj_rf_beneficiary } } }
  holder = { change_liege = { liege = scope:wj_rf_beneficiary change = scope:wj_rf_change } }
 }
 else = { change_title_holder = { holder = scope:wj_rf_beneficiary change = scope:wj_rf_change government_base = scope:wj_rf_sovereign } }
 every_de_jure_county = {
  limit = { exists = holder holder.top_liege = scope:wj_rf_sovereign }
  if = { limit = { NOT = { this = scope:wj_rf_capital_county } NOT = { holder = scope:wj_rf_beneficiary } }
   if = { limit = { holder = { highest_held_title_tier < tier_kingdom } }
    # County officers remain below their own retained commandery, not flattened unnecessarily.
    if = { limit = { holder = { highest_held_title_tier < tier_duchy } exists = scope:wj_rf_transfer_duchy.holder
      scope:wj_rf_transfer_duchy.holder = { top_liege = scope:wj_rf_sovereign highest_held_title_tier < tier_kingdom NOT = { this = scope:wj_rf_beneficiary } } }
     holder = { change_liege = { liege = scope:wj_rf_transfer_duchy.holder change = scope:wj_rf_change } }
    }
    else = { holder = { change_liege = { liege = scope:wj_rf_beneficiary change = scope:wj_rf_change } } }
   }
   else = { change_title_holder = { holder = scope:wj_rf_beneficiary change = scope:wj_rf_change government_base = scope:wj_rf_sovereign } }
  }
 }
 set_de_jure_liege_title = scope:wj_rf_new_fief
}

  every_held_title = {
   limit = {
    is_noble_family_title = no tgp_is_minister_title = no
    NOT = { has_variable = wj_rf_rank }
    NOT = { this = scope:wj_rf_new_fief }
    NOT = { is_de_jure_liege_or_above_target = scope:wj_rf_new_fief }
    NOT = { scope:wj_rf_new_fief = { is_de_jure_liege_or_above_target = prev } }
   }
   clear_saved_scope = wj_rf_office_successor
   current_heir ?= {
    if = { limit = { is_alive = yes NOT = { this = scope:wj_rf_retiring_holder } }
     save_scope_as = wj_rf_office_successor
    }
   }
   if = { limit = { NOT = { exists = scope:wj_rf_office_successor } }
    scope:wj_rf_sovereign = { save_scope_as = wj_rf_office_successor }
   }
   change_title_holder = { holder = scope:wj_rf_office_successor change = scope:wj_rf_retirement }
  }
  resolve_title_and_vassal_change = scope:wj_rf_retirement
  set_primary_title_to = scope:wj_rf_new_fief
  if = { limit = { exists = scope:wj_rf_new_kingdom scope:wj_rf_new_kingdom.holder = scope:wj_rf_retiring_holder }
   set_primary_title_to = scope:wj_rf_new_kingdom
  }
  WJ_rf_refresh_holder = yes
 }
}

# on_death runs before succession: capture the direct lord, not an assumed top emperor.
WJ_rf_capture_succession_liege = {
 if = { limit = { exists = liege }
  save_scope_as = wj_rf_dying_holder
  every_held_title = { limit = { has_variable = wj_rf_rank }
   set_variable = { name = wj_rf_previous_liege value = scope:wj_rf_dying_holder.liege }
   set_variable = { name = wj_rf_previous_liege_title value = scope:wj_rf_dying_holder.liege.primary_title }
  }
 }
}
WJ_rf_restore_succession_liege = {
 clear_saved_scope = wj_rf_inherited_liege
 if = { limit = { exists = scope:title.var:wj_rf_previous_liege scope:title.var:wj_rf_previous_liege = { is_alive = yes } }
  scope:title.var:wj_rf_previous_liege = { save_scope_as = wj_rf_inherited_liege }
 }
 else_if = { limit = { exists = scope:title.var:wj_rf_previous_liege_title.holder }
  scope:title.var:wj_rf_previous_liege_title.holder = { save_scope_as = wj_rf_inherited_liege }
 }
 if = { limit = {
  exists = scope:wj_rf_inherited_liege
  scope:wj_rf_inherited_liege = { is_alive = yes NOT = { this = root } NOT = { is_vassal_or_below_of = root } }
  NOT = { liege = scope:wj_rf_inherited_liege }
 }
  create_title_and_vassal_change = { type = swear_fealty save_scope_as = wj_rf_succession_change add_claim_on_loss = no }
  change_liege = { liege = scope:wj_rf_inherited_liege change = scope:wj_rf_succession_change }
  resolve_title_and_vassal_change = scope:wj_rf_succession_change
 }
 scope:title = { remove_variable = wj_rf_previous_liege remove_variable = wj_rf_previous_liege_title }
}

# Called by the native destruction hook while the old title and its tracked duchies remain accessible.
WJ_rf_restore_destroyed_fief = {
 save_scope_as = wj_rf_destroyed_fief
 every_in_list = { variable = wj_rf_duchies
  limit = { de_jure_liege = scope:wj_rf_destroyed_fief }
  save_scope_as = wj_rf_restoring_duchy
  clear_saved_scope = wj_rf_restored_parent
  if = { limit = { exists = var:wj_rf_original_liege.holder var:wj_rf_original_liege = { tier = tier_kingdom NOT = { has_variable = wj_rf_rank } } }
   var:wj_rf_original_liege = { save_scope_as = wj_rf_restored_parent }
  }
  else = {
   clear_variable_list = wj_rf_restore_candidates
   every_de_jure_county = { every_county_province = { every_neighboring_province = {
    limit = {
     exists = county.kingdom.holder
     county.duchy != scope:wj_rf_restoring_duchy
     county.kingdom = { NOT = { has_variable = wj_rf_rank } }
     WJ_rf_physical_neighbor = { OTHER = prev }
    }
    county.kingdom = { save_scope_as = wj_rf_neighbor_parent }
    scope:wj_rf_restoring_duchy = {
     if = { limit = { NOT = { is_target_in_variable_list = { name = wj_rf_restore_candidates target = scope:wj_rf_neighbor_parent } } }
      add_to_variable_list = { name = wj_rf_restore_candidates target = scope:wj_rf_neighbor_parent }
     }
    }
   } } }
   random_in_list = { variable = wj_rf_restore_candidates
    limit = { exists = holder root.top_liege = holder.top_liege }
    save_scope_as = wj_rf_restored_parent
   }
   if = { limit = { NOT = { exists = scope:wj_rf_restored_parent } }
    random_in_list = { variable = wj_rf_restore_candidates save_scope_as = wj_rf_restored_parent }
   }
   clear_variable_list = wj_rf_restore_candidates
  }
  if = { limit = { exists = scope:wj_rf_restored_parent }
   set_de_jure_liege_title = scope:wj_rf_restored_parent
  }
  # No valid parent: native destruction detaches this duchy; do not invent a title.
  remove_variable = wj_rf_original_liege
  every_de_jure_county = {
   limit = { has_variable = wj_rf_capital_of var:wj_rf_capital_of = scope:wj_rf_destroyed_fief }
   remove_variable = wj_rf_capital_of
   remove_title_law = noble_family_succession_law
  }
 }
 clear_variable_list = wj_rf_duchies
}
