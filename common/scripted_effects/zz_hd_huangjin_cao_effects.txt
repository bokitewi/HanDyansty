# ==========================================================================
# HD 曹操专属：济北受降，收编青州军(2026-10-11)
# 挂在曹操的历史心跳 hd_cao_expansion_legacy_tick_effect 上(15~30天一次，玩家/AI皆跑)
#   讨董联盟结束(global_var:hd_anti_regent_chain_resolved)后首次心跳 → 开始一年倒计时
#   倒计时结束、已到191年、曹操在世且有地 → 弹 hd_huangjin.10，满额奖励
# 原「招降青州黄巾」决议已于同日删除，青州军只能由曹操经此事件获得
# hd_hj_done / hd_hj_closed 检查仅防旧存档里已点过旧决议的情形
# ==========================================================================

hd_hj_cao_qingzhou_tick_effect = {
	if = {
		limit = {
			this = character:cao_cao
			is_alive = yes
			exists = global_var:hd_anti_regent_chain_resolved
			NOT = { has_character_flag = hd_hj_cao_countdown_started }
		}
		add_character_flag = hd_hj_cao_countdown_started
		add_character_flag = { flag = hd_hj_cao_countdown days = 365 }
	}
	else_if = {
		limit = {
			this = character:cao_cao
			is_alive = yes
			is_landed = yes
			has_character_flag = hd_hj_cao_countdown_started
			current_date >= 191.1.1
			NOT = { has_character_flag = hd_hj_cao_countdown }
			NOT = { has_character_flag = hd_hj_cao_event_sent }
			NOT = { has_character_flag = hd_hj_done }
			NOT = { exists = global_var:hd_hj_closed }
		}
		add_character_flag = hd_hj_cao_event_sent
		trigger_event = hd_huangjin.10
	}
}
