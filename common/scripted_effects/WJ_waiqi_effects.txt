# Enumerate only the three relevant families. Mother/queen groups win overlapping houses.
WJ_waiqi_collect_house = {
 every_house_member = {
  limit = { is_alive = yes is_male = yes is_adult = yes top_liege = scope:wj_waiqi_emperor NOT = { this = scope:wj_waiqi_emperor } }
  save_scope_as = wj_waiqi_relative
  scope:wj_waiqi_emperor = {
   if = { limit = { NOT = { is_target_in_variable_list = { name = wj_waiqi_members target = scope:wj_waiqi_relative } } }
    add_to_variable_list = { name = wj_waiqi_members target = scope:wj_waiqi_relative }
    add_to_variable_list = { name = $GROUP$ target = scope:wj_waiqi_relative }
   }
  }
 }
}
WJ_waiqi_refresh_roster = {
 save_scope_as = wj_waiqi_emperor
 clear_variable_list = wj_waiqi_members
 clear_variable_list = wj_roster_waiqi_mother
 clear_variable_list = wj_roster_waiqi_empress
 clear_variable_list = wj_roster_waiqi_consorts
 mother ?= { house ?= { WJ_waiqi_collect_house = { GROUP = wj_roster_waiqi_mother } } }
 primary_spouse ?= { house ?= { WJ_waiqi_collect_house = { GROUP = wj_roster_waiqi_empress } } }
 every_consort = { house ?= { WJ_waiqi_collect_house = { GROUP = wj_roster_waiqi_consorts } } }
}
# Both entry points settle here; recheck stale dialogs and the shared 180-day recipient flag.
WJ_waiqi_reward = {
 if = { limit = { WJ_waiqi_can_reward = { EMPEROR = scope:actor } }
  add_character_flag = { flag = Tiba_Waiqi_flag days = 180 }
  if = { limit = { NOT = { has_character_modifier = WJ_maocai_modifier } }
   WJ_grant_road_maocai_effect = yes
   change_merit = 2000
  }
  add_gold = 300
  add_opinion = { modifier = pleased_opinion target = scope:actor opinion = 20 }
  save_scope_value_as = { name = increased_career_score value = { add = career_score_major_base_value multiply = 4 } }
  if = { limit = { is_military_candidate_trigger = yes }
   if = { limit = { has_variable = merit_military_career_score_bonus } change_variable = { name = merit_military_career_score_bonus add = scope:increased_career_score } }
   else = { set_variable = { name = merit_military_career_score_bonus value = scope:increased_career_score } }
  }
  else = {
   if = { limit = { has_variable = merit_civilian_career_score_bonus } change_variable = { name = merit_civilian_career_score_bonus add = scope:increased_career_score } }
   else = { set_variable = { name = merit_civilian_career_score_bonus value = scope:increased_career_score } }
  }
 }
}
