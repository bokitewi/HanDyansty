# 原版 appointment_timeout 只表达当前不接受任命；WJ 所有权旗标避免误清原版辞谢状态。
# 已存在非 WJ 任命超时时不窃取所有权。
WJ_set_owned_appointment_timeout_effect = {
	if = {
		limit = {
			OR = {
				NOT = { appointment_timeout_days > 0 }
				has_character_flag = wj_appointment_timeout_owned
			}
		}
		set_appointment_timeout = {
			years = 150
			desc = WJ_appointment_timeout_desc_provisional
		}
		add_character_flag = wj_appointment_timeout_owned
	}
}

# 只有 WJ 自己建立的超时才允许由 WJ 清理。
WJ_clear_owned_appointment_timeout_effect = {
	if = {
		# 养望仍有效时保留原生拒任；真正出山先由养望退出效果移除特质。
		limit = { has_character_flag = wj_appointment_timeout_owned NOT = { has_trait = WJ_yangwang_trait } }
		# 失官后WJ标记可能仍在，但原生任命数据已不存在。
		# 沿用原版examination_clear_timeout_effect的前置判断。
		if = {
			limit = { appointment_timeout_days > 0 }
			clear_appointment_timeout = yes
		}
		remove_character_flag ?= wj_appointment_timeout_owned
	}
}

# 进入唐制时转换；离开汉晋但未进入唐制时只清理现任职权，保留个人资格及四轨；仍在汉晋时校正地方任期。
# on_vassal_change 在死亡交接时可能传入死者；普通人物校正不能对其授资格或移除人物标记。
# 死亡任期封存仍由专用 on_death 入口处理，不跳过其存活下属／廷臣的逐人校正。
WJ_reconcile_han_jin_character_state_effect = {
# 	WJ_migrate_langguan_identity_effect = yes
	if = {
		limit = { is_alive = yes }
		if = {
			limit = {
				WJ_has_han_jin_conversion_state_trigger = yes
				WJ_tang_law_trigger = yes
			}
			WJ_convert_han_jin_character_to_tang_effect = yes
		}
		else_if = {
			limit = {
				WJ_has_han_jin_conversion_state_trigger = yes
				WJ_han_jin_law_trigger = no
			}
			WJ_clear_han_jin_character_state_effect = yes
		}
		if = {
			limit = { WJ_han_jin_law_trigger = yes }
			WJ_kaoke_reconcile_effect = yes
		}
	}
}

# 法律 on_pass 在新法生效前执行，因此该入口不再检查当前是否已经是唐制。
WJ_convert_han_jin_realm_to_tang_effect = {
	save_scope_as = wj_han_jin_conversion_realm_liege
	if = {
		limit = { WJ_has_han_jin_conversion_state_trigger = yes }
		WJ_convert_han_jin_character_to_tang_effect = yes
	}
	every_courtier_or_guest = {
		limit = {
			top_liege = scope:wj_han_jin_conversion_realm_liege.top_liege
			WJ_has_han_jin_conversion_state_trigger = yes
		}
		WJ_convert_han_jin_character_to_tang_effect = yes
	}
	every_vassal_or_below = {
		if = {
			limit = { WJ_has_han_jin_conversion_state_trigger = yes }
			WJ_convert_han_jin_character_to_tang_effect = yes
		}
		every_courtier_or_guest = {
			limit = {
				top_liege = scope:wj_han_jin_conversion_realm_liege.top_liege
				WJ_has_han_jin_conversion_state_trigger = yes
			}
			WJ_convert_han_jin_character_to_tang_effect = yes
		}
	}
}

# 将一次开局仕履贡献随机投入四条路线之一；已投入两次的路线不再入选。

		}

			trigger = { var:wj_starting_mingshi_wuxun_rank < 2 }

		}
	}
}

WJ_settle_recommendation_movement_options_effect = {
	if = {
		limit = { always = scope:gift }
		scope:actor = {
			pay_short_term_gold = {
				gold = wj_request_recommendation_gold_value
				target = scope:recipient
			}
			stress_impact = { greedy = minor_stress_impact_gain }
		}
	}
	if = {
		limit = { always = scope:offer_hook }
		scope:actor = {
			stress_impact = {
				base = minor_stress_impact_gain
				paranoid = major_stress_impact_gain
			}
		}
		scope:recipient = {
			add_hook = {
				type = favor_hook
				target = scope:actor
			}
		}
	}
	if = {
		limit = { always = scope:offer_hook_strong }
		scope:actor = { stress_impact = { base = major_stress_impact_gain } }
		scope:recipient = {
			add_hook = {
				type = strong_favor_hook
				target = scope:actor
			}
		}
	}
	if = {
		limit = { always = scope:hook }
		scope:actor = { use_hook = scope:recipient }
	}
	if = {
		limit = { always = scope:prestige }
		scope:actor = {
			add_prestige = {
				value = prestige_bribe_value
				multiply = -1
			}
		}
		scope:recipient = { add_prestige = prestige_bribe_value }
	}
	if = {
		limit = { always = scope:influence }
		scope:actor = {
			change_influence = {
				value = influence_bribe_value
				multiply = -1
			}
		}
		scope:recipient = { change_influence = influence_bribe_value }
	}
}
