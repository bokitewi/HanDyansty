# Character scope. Reuse the native family constructor after explicitly splitting non-heads.
WJ_peer_get_family = {
 clear_saved_scope = wj_peer_family
 every_held_title = { limit = { is_noble_family_title = yes NOT = { has_variable = wj_peer_rank } } save_scope_as = wj_peer_family }
 # Use the recipient's actual holdings, not an inherited saved scope from an earlier grant.
 if = { limit = { NOT = { any_held_title = { is_noble_family_title = yes } } WJ_peer_family_possible = yes }
  save_scope_value_as = { name = wj_peer_was_close_kin value = no }
  if = { limit = { WJ_peer_royal_kin = { EMPEROR = top_liege } } save_scope_value_as = { name = wj_peer_was_close_kin value = yes } }
  if = { limit = { NOT = { this = house.house_head } }
   house = { save_scope_as = wj_peer_original_house }
   create_cadet_branch = {}
   # Only this newly created branch: never replace the head of the original house.
   if = { limit = { NOT = { house = scope:wj_peer_original_house } } house = { set_house_head = prev set_house_name_from_house = scope:wj_peer_original_house } }
  }
  if = { limit = { scope:wj_peer_was_close_kin = yes } house = { add_house_modifier = { modifier = SanGong_SeparatedCloseKin_modifier } } }
  create_noble_family_effect = { GOVERNMENT_GIVER = top_liege }
  every_held_title = { limit = { is_noble_family_title = yes NOT = { has_variable = wj_peer_rank } } save_scope_as = wj_peer_family }
  clear_saved_scope = wj_peer_was_close_kin
 }
}

# Family title scope. This is the only authoritative record; character variables are display adapters.
WJ_peer_sync_holder = {
 save_scope_as = wj_peer_family

  var:wj_peer_bearer = { if = { limit = { var:wj_peer_family = scope:wj_peer_family } WJ_peer_clear_mirror = yes } }
 }
 set_variable = { name = wj_peer_bearer value = holder }
 holder = {
  set_variable = { name = wj_peer_family value = scope:wj_peer_family }
  set_variable = { name = wj_peer_rank value = scope:wj_peer_family.var:wj_peer_rank }
  set_variable = { name = wj_peer_name value = scope:wj_peer_family.var:wj_peer_name }
  set_variable = { name = wj_peer_food value = scope:wj_peer_family.var:wj_peer_food }
  set_variable = { name = wj_peer_realm value = scope:wj_peer_family.var:wj_peer_realm }
  set_variable = { name = wj_peer_last_paid value = scope:wj_peer_family.var:wj_peer_last_paid }
  remove_variable = wj_peer_heir
  if = { limit = { scope:wj_peer_family = { has_variable = wj_peer_princess } } set_variable = { name = wj_peer_princess value = yes } }
  else = { remove_variable = wj_peer_princess }
  add_trait = WJ_peerage_trait
 }
 var:wj_peer_name = { set_variable = { name = wj_peer_name_owner value = scope:wj_peer_family.holder } }
 var:wj_peer_food = { set_variable = { name = wj_peer_food_owner value = scope:wj_peer_family.holder } }

# Character scope; family record owns every cleanup operation.
WJ_peer_clear = {
 if = { limit = { has_variable = wj_peer_personal } WJ_peer_clear_personal = yes }
 else_if = { limit = { exists = var:wj_peer_family } var:wj_peer_family = { WJ_peer_clear_family = yes } }
 else = { WJ_peer_clear_mirror = yes }
}
WJ_peer_commit = {
 if = { limit = { WJ_peer_commit_valid = yes }
  save_scope_as = wj_peer_issuer
  if = { limit = { var:wj_peer_target = { dynasty = root.dynasty } }
   var:wj_peer_target = {
    set_variable = { name = wj_peer_personal value = yes }
    set_variable = { name = wj_peer_rank value = scope:wj_peer_issuer.var:wj_peer_pending_rank }
    set_variable = { name = wj_peer_name value = scope:wj_peer_issuer.var:wj_peer_pending_name }
    set_variable = { name = wj_peer_food value = scope:wj_peer_issuer.var:wj_peer_pending_food }
    set_variable = { name = wj_peer_realm value = scope:wj_peer_issuer.primary_title }
    set_variable = { name = wj_peer_last_paid value = current_year }
    if = { limit = { is_female = yes } set_variable = { name = wj_peer_princess value = yes } }
    WJ_peer_bind_personal = yes
   }
   WJ_peer_finish_grant = yes
  }
  else = {
   var:wj_peer_target = { WJ_peer_get_family = yes }
   if = { limit = { WJ_peer_commit_valid = yes exists = scope:wj_peer_family scope:wj_peer_family = { holder = root.var:wj_peer_target NOT = { has_variable = wj_peer_rank } } }
   # Native creation can trigger hooks; only charge once a real family title is available.
   scope:wj_peer_family = {
    set_variable = { name = wj_peer_rank value = scope:wj_peer_issuer.var:wj_peer_pending_rank }
    set_variable = { name = wj_peer_name value = scope:wj_peer_issuer.var:wj_peer_pending_name }
    set_variable = { name = wj_peer_food value = scope:wj_peer_issuer.var:wj_peer_pending_food }
    set_variable = { name = wj_peer_realm value = scope:wj_peer_issuer.primary_title }
    set_variable = { name = wj_peer_last_paid value = current_year }
    if = { limit = { scope:wj_peer_issuer.var:wj_peer_target = { is_female = yes } }
     set_variable = { name = wj_peer_princess value = yes }
     set_variable = { name = wj_peer_princess_dynasty value = scope:wj_peer_issuer.dynasty }
    }
    WJ_peer_bind_family = yes
   }
    WJ_peer_finish_grant = yes
   }
  }
 }
}

# Family title scope; preserve an existing different grant if two family titles converge.
WJ_peer_maintain_family = {
 # Death precedes native family-title transfer; do not destroy the grant in that interval.
 if = { limit = { exists = holder holder = { is_alive = no } } }
 else_if = { limit = { exists = var:wj_peer_realm.holder var:wj_peer_realm.holder = { is_alive = no } } }
 else_if = { limit = { WJ_peer_family_valid = no } WJ_peer_clear_family = yes }
 else_if = { limit = { holder = { has_variable = wj_peer_rank NOT = { var:wj_peer_family = prev } } } WJ_peer_clear_family = yes }
 else = {
  if = { limit = { NOT = { var:wj_peer_bearer = holder } } holder = { trigger_event = { id = WJ_peer.021 days = 1 } } }
  WJ_peer_sync_holder = yes
 }
}
