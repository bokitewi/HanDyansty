WJ_peer_choose = {
 if = { limit = { WJ_peer_rank_allowed = yes exists = scope:wj_peer_choice scope:wj_peer_choice = { WJ_peer_place_available = yes } }
  if = { limit = { scope:wj_peer_choice = { tier = tier_kingdom } var:wj_peer_pending_rank = 5 } set_variable = { name = wj_peer_pending_name value = scope:wj_peer_choice } }
  else_if = { limit = { scope:wj_peer_choice = { tier = tier_duchy } var:wj_peer_pending_rank >= 3 var:wj_peer_pending_rank <= 4 } set_variable = { name = wj_peer_pending_name value = scope:wj_peer_choice } }
  else_if = { limit = { scope:wj_peer_choice = { tier = tier_county } var:wj_peer_pending_rank = 2 } set_variable = { name = wj_peer_pending_name value = scope:wj_peer_choice } }
  else_if = { limit = { scope:wj_peer_choice = { tier = tier_barony } var:wj_peer_pending_rank = 1 } set_variable = { name = wj_peer_pending_name value = scope:wj_peer_choice } }
  if = { limit = { OR = { AND = { scope:wj_peer_choice = { tier = tier_county } var:wj_peer_pending_rank >= 2 } scope:wj_peer_choice = { tier = tier_barony } } }
   set_variable = { name = wj_peer_pending_food value = scope:wj_peer_choice } trigger_event = WJ_peer.011
  }
  else = { set_variable = { name = wj_peer_parent value = scope:wj_peer_choice } WJ_peer_build_choices = yes }
 }
}

}

WJ_peer_commit = {
 if = { limit = { WJ_peer_commit_valid = yes }
  save_scope_as = wj_peer_issuer
  if = { limit = { var:wj_peer_pending_rank >= 4 } remove_short_term_gold = 50 add_piety = -50 change_influence = -50 }
  var:wj_peer_target = {
   set_variable = { name = wj_peer_rank value = scope:wj_peer_issuer.var:wj_peer_pending_rank }
   set_variable = { name = wj_peer_name value = scope:wj_peer_issuer.var:wj_peer_pending_name }
   set_variable = { name = wj_peer_food value = scope:wj_peer_issuer.var:wj_peer_pending_food }
   set_variable = { name = wj_peer_realm value = scope:wj_peer_issuer.primary_title }
   # Annual entitlement starts next January; no up-front duplicate farming.
   set_variable = { name = wj_peer_last_paid value = current_year }
   WJ_peer_bind = yes
   trigger_event = WJ_peer.020
  }
  WJ_peer_end_workflow = yes
 }
}
WJ_peer_pay = {
 if = { limit = { WJ_peer_valid = yes OR = { NOT = { has_variable = wj_peer_last_paid } var:wj_peer_last_paid < current_year } }
  set_variable = { name = wj_peer_last_paid value = current_year }
  set_variable = { name = wj_peer_amount value = WJ_peer_annual_gold }
  add_gold = var:wj_peer_amount
  if = { limit = { is_ai = no } trigger_event = WJ_peer.100 }
 }
}

WJ_peer_inherit = {
 save_scope_as = wj_peer_deceased
 # A designation is attached to this grant, never a shared house-wide character flag.
 if = { limit = { exists = var:wj_peer_heir var:wj_peer_heir = { WJ_peer_heir_candidate = yes } }
  var:wj_peer_heir = { save_scope_as = wj_peer_successor }
 }
 else = {
  ordered_child = { limit = { WJ_peer_heir_candidate = yes } order_by = age max = 1 check_range_bounds = no save_scope_as = wj_peer_successor }
  if = { limit = { NOT = { exists = scope:wj_peer_successor } }
   every_child = { even_if_dead = yes every_child = { limit = { WJ_peer_heir_candidate = yes } add_to_list = wj_peer_grandsons } }
   ordered_in_list = { list = wj_peer_grandsons order_by = age max = 1 check_range_bounds = no save_scope_as = wj_peer_successor }
  }
  if = { limit = { NOT = { exists = scope:wj_peer_successor } }
   every_child = { even_if_dead = yes every_child = { even_if_dead = yes every_child = { limit = { WJ_peer_heir_candidate = yes } add_to_list = wj_peer_greatgrandsons } } }
   ordered_in_list = { list = wj_peer_greatgrandsons order_by = age max = 1 check_range_bounds = no save_scope_as = wj_peer_successor }
  }
  if = { limit = { NOT = { exists = scope:wj_peer_successor } }
   ordered_close_family_member = { limit = { WJ_peer_heir_candidate = yes } order_by = age max = 1 check_range_bounds = no save_scope_as = wj_peer_successor }
  }
 }
 if = { limit = { exists = scope:wj_peer_successor } WJ_peer_transfer = yes }
 else = { WJ_peer_clear = yes }
}
