# Cao Cao's Fengying takeover.  The caller is always Cao Cao.
#
# 2026-10-11 改版：宣战对象为挟持天子之人（剧本摄政或通用权臣），无人挟持则为天子本人。
# 胜利后：对方持有河南公国则连同其在河南境内的县一并交给曹操；
# 原权臣罢官独立；曹操成为天子封臣并以通用挟天子系统傀儡天子；
# 随即接管朝廷：不转汉帝国头衔、不走剧本摄政，天子直辖地与封臣转归曹操，庄园迁往曹操首都。
hd_cao_fengying_victory_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_fengying_world_resolved }
			has_character_flag = hd_cao_fengying_authorized
			hd_cao_fengying_actor_valid_trigger = yes
			hd_dong_transition_pending_trigger = no
			title:h_china.holder ?= { is_alive = yes }
		}
		save_scope_as = hd_cao_fengying_takeover_actor
		title:h_china.holder = { save_scope_as = hd_cao_fengying_han_emperor }

		# 对方持有河南公国：公国头衔与对方及其封臣在河南境内的县归曹操
		if = {
			limit = { scope:defender = { has_title = title:d_henan } }
			# 河南的县可能是天子最后的领地
			title:h_china = { set_landless_title = yes }
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = hd_cao_fengying_henan_change
				add_claim_on_loss = no
			}
			title:d_henan = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_county
						holder ?= {
							OR = {
								this = scope:defender
								is_vassal_or_below_of = scope:defender
							}
						}
					}
					change_title_holder = {
						holder = scope:hd_cao_fengying_takeover_actor
						change = scope:hd_cao_fengying_henan_change
						take_baronies = yes
					}
				}
				change_title_holder = {
					holder = scope:hd_cao_fengying_takeover_actor
					change = scope:hd_cao_fengying_henan_change
				}
			}
			resolve_title_and_vassal_change = scope:hd_cao_fengying_henan_change
		}

		# 对方是挟持天子之人：罢官，带剩余领地独立
		if = {
			limit = { NOT = { scope:defender = scope:hd_cao_fengying_han_emperor } }
			scope:defender = { remove_character_flag = hd_current_han_regent }
			scope:hd_cao_fengying_han_emperor = {
				if = {
					limit = { var:hd_dong_fixed_regent ?= scope:defender }
					remove_variable = hd_dong_fixed_regent
				}
			}
			if = {
				limit = { title:h_china.var:hd_court_regent ?= scope:defender }
				hd_court_release_effect = yes
			}
			scope:defender = { hd_court_strip_office_effect = yes }
			if = {
				limit = {
					scope:defender = {
						is_alive = yes
						is_ruler = yes
						liege ?= scope:hd_cao_fengying_han_emperor
					}
				}
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = hd_cao_fengying_old_regent_change
					add_claim_on_loss = no
				}
				scope:defender = {
					becomes_independent = { change = scope:hd_cao_fengying_old_regent_change }
				}
				resolve_title_and_vassal_change = scope:hd_cao_fengying_old_regent_change
			}
		}

		# 曹操成为天子封臣，傀儡天子（通用挟天子系统）
		if = {
			limit = { NOT = { liege ?= scope:hd_cao_fengying_han_emperor } }
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = hd_cao_fengying_liege_change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = scope:hd_cao_fengying_han_emperor
				change = scope:hd_cao_fengying_liege_change
			}
			resolve_title_and_vassal_change = scope:hd_cao_fengying_liege_change
		}
		hd_court_install_regency_effect = yes
		add_character_flag = { flag = hd_court_regent_tenure years = 3 }

		hd_cao_fengying_complete_takeover_effect = yes
	}
}

hd_cao_fengying_complete_takeover_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_fengying_world_resolved }
			has_character_flag = hd_cao_fengying_authorized
			title:h_china.holder ?= { is_alive = yes }
			liege ?= title:h_china.holder
		}

		save_scope_as = hd_cao_fengying_takeover_actor
		title:h_china.holder = { save_scope_as = hd_cao_fengying_han_emperor }
		capital_county = { save_scope_as = hd_cao_fengying_capital }

		# Snapshot the emperor's direct vassals and real titles.  h_china, e_han,
		# and landless noble-family/minister titles never enter these lists.
		scope:hd_cao_fengying_han_emperor = {
			every_vassal = {
				limit = {
					NOT = { this = scope:hd_cao_fengying_takeover_actor }
				}
				add_to_list = hd_cao_fengying_han_vassals
			}
			every_held_title = {
				limit = {
					tier >= tier_barony
					is_landless_type_title = no
					is_noble_family_title = no
					NOT = { this = title:h_china }
					NOT = { this = title:e_han }
				}
				add_to_list = hd_cao_fengying_han_real_titles
			}
		}

		# 霸权无地并定都曹操首都（须在天子交出最后一县之前）
		title:h_china = {
			set_landless_title = yes
			set_capital_county = scope:hd_cao_fengying_capital
			set_variable = { name = hd_court_capital_county value = scope:hd_cao_fengying_capital }
		}

		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_cao_fengying_domain_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = hd_cao_fengying_han_real_titles
			limit = {
				holder = scope:hd_cao_fengying_han_emperor
				NOT = { holder = scope:hd_cao_fengying_takeover_actor }
			}
			change_title_holder = {
				holder = scope:hd_cao_fengying_takeover_actor
				change = scope:hd_cao_fengying_domain_change
				take_baronies = yes
			}
		}
		every_in_list = {
			list = hd_cao_fengying_han_vassals
			limit = {
				is_alive = yes
				NOT = { this = scope:hd_cao_fengying_takeover_actor }
				# 封臣层级须低于曹操，否则留在天子名下
				highest_held_title_tier < scope:hd_cao_fengying_takeover_actor.highest_held_title_tier
				liege ?= scope:hd_cao_fengying_han_emperor
			}
			if = {
				limit = { is_ruler = yes }
				change_liege = {
					liege = scope:hd_cao_fengying_takeover_actor
					change = scope:hd_cao_fengying_domain_change
				}
			}
			else = { set_employer = scope:hd_cao_fengying_takeover_actor }
		}
		resolve_title_and_vassal_change = scope:hd_cao_fengying_domain_change

		# 天子庄园与本人迁往曹操首都
		scope:hd_cao_fengying_han_emperor = {
			# Existing celestial rulers normally already have an east Asian estate.
			# If a title-transfer edge case removed it, create the native noble
			# family title first, then use the native estate setup helper.
			if = {
				limit = {
					has_domicile = no
					NOT = { any_held_title = { is_noble_family_title = yes } }
				}
				if = {
					limit = { NOT = { has_game_rule = never_noble_family_title } }
					create_noble_family_effect = { GOVERNMENT_GIVER = this }
				}
				else = {
					# Only the emperor bypasses the optional family-title creation
					# rule. Preserve the native house-head prerequisite.
					if = {
						limit = {
							exists = house.house_head
							NOT = { house.house_head = scope:hd_cao_fengying_han_emperor }
						}
						if = {
							limit = {
								house.house_head = {
									is_ai = yes
									OR = {
										AND = {
											top_liege = scope:hd_cao_fengying_han_emperor.top_liege
											OR = {
												is_ruler = no
												highest_held_title_tier < scope:hd_cao_fengying_han_emperor.highest_held_title_tier
											}
										}
										NOT = { exists = liege }
									}
								}
							}
							house = { set_house_head = scope:hd_cao_fengying_han_emperor }
						}
						else_if = {
							limit = { culture = { hd_culture_uses_chinese_cadet_names_trigger = yes } }
							hd_create_chinese_cadet_branch_effect = yes
						}
						else = { create_cadet_branch = {} }
					}
					give_noble_family_title = {
						save_scope_as = hd_cao_fengying_emperor_family_title
						name = noble_family_name
						tier = county
						article = DEFAULT_TITLE_NAME_ARTICLE
						government = celestial_government
					}
				}
				domicile ?= { set_up_domicile_estate_effect = yes }
			}
			domicile ?= {
				move_domicile = scope:hd_cao_fengying_capital.title_province
			}
			set_location = {
				location = scope:hd_cao_fengying_capital.title_province
				stick_to_location = yes
			}
			if = {
				limit = { has_domicile = no }
				debug_log = "HD_CAO FENGYING EMPEROR_ESTATE_MISSING"
			}
		}

		set_global_variable = {
			name = hd_fengying_world_claimed
			value = scope:hd_cao_fengying_takeover_actor
		}
		set_global_variable = hd_fengying_world_resolved
		set_global_variable = hd_chain_terminal_historical
		remove_global_variable = hd_fengying_world_open
		scope:hd_cao_fengying_takeover_actor = {
			remove_character_flag = hd_cao_fengying_authorized
		}
		debug_log = "HD_CAO FENGYING TAKEOVER_COMPLETED"
	}
}
