# try_start（共用体）：$KEY$ 规则开着、未了结时，解析本人 $ID$，就绪即开始入侵
############################################################
hd_hi_try_start_effect = {
	if = {
		limit = {
			has_game_rule = hd_hi_$KEY$_on
			NOT = { has_global_variable = hd_hi_done_$KEY$ }
		}
		if = {
			limit = { exists = scope:hd_hi_ts_leader }
			clear_saved_scope = hd_hi_ts_leader
		}
		hd_j16_resolve_effect = { ID = $ID$ AS = hd_hi_ts_leader }
		scope:hd_hi_ts_leader ?= {
			if = {
				limit = { hd_hi_leader_ready_trigger = yes }
				hd_hi_begin_effect = { KEY = $KEY$ }
			}
		}
	}
}

# 了结共用体：$KEY$ 规则开、未了结，且本人 $ID$ 已死或秦况了结 → 设 done
hd_hi_qin_close_effect = {
	if = {
		limit = {
			has_game_rule = hd_hi_$KEY$_on
			NOT = { has_global_variable = hd_hi_done_$KEY$ }
			OR = {
				hd_j16_dead_trigger = { ID = $ID$ }
				hd_hi_qin_end_trigger = yes
			}
		}
		set_global_variable = { name = hd_hi_done_$KEY$ value = yes }
		debug_log = "HD_J16 hi_qin SKIP dead_or_end"
	}
}

# ① 慕容垂：E4-M25 许之（hd_j16_chui_north）即开始；E4-M25 不许（hd_j16_chui_north_denied）推迟到 M >= 6；否则 M >= 2。

	hd_hi_qin_close_effect = { KEY = murongchui ID = changli_murong_038 }

		limit = {
			has_game_rule = hd_hi_murongchui_on
			NOT = { has_global_variable = hd_hi_done_murongchui }
			hd_hi_qin_started_trigger = yes
			OR = {
				has_global_variable = hd_j16_chui_north
				AND = {
					has_global_variable = hd_j16_chui_north_denied
					hd_hi_qin_m_trigger = { M = 6 MW = 18 }
				}
				AND = {
					NOT = { has_global_variable = hd_j16_chui_north_denied }
					hd_hi_qin_m_trigger = { M = 2 MW = 14 }
				}
			}
		}
		hd_hi_try_start_murongchui_effect = yes
	}
}
