# Scope: dynastic_cycle. User-approved persistent W/H authority, independent of
# native phase progress. Native catalysts enter through hd_dc_catalyst_* effects.
# Starting a phase refreshes the display; it never imports or subtracts a gauge.
hd_dc_sync_pressure_effect = {
	if = {
		limit = { NOT = { has_variable = hd_dc_pressure_raw } }
		set_variable = { name = hd_dc_pressure_raw value = 0 }
		set_variable = { name = hd_dc_prosperity_raw value = 0 }
		set_variable = { name = hd_dc_point_sequence value = 0 }
		set_variable = { name = hd_dc_last_point_source value = flag:initialization }
		set_variable = { name = hd_dc_last_w_requested value = 0 }
		set_variable = { name = hd_dc_last_h_requested value = 0 }
		set_variable = { name = hd_dc_last_w_actual value = 0 }
		set_variable = { name = hd_dc_last_h_actual value = 0 }
		set_variable = { name = hd_dc_ordinary_year value = current_year }
		set_variable = { name = hd_dc_ordinary_w_net value = 0 }
	}
	set_variable = {
		name = hd_dc_pressure_percent
		value = { value = var:hd_dc_pressure_raw divide = hd_dc_pressure_max multiply = 100 }
	}
	set_variable = {
		name = hd_dc_prosperity_percent
		value = { value = var:hd_dc_prosperity_raw divide = hd_dc_prosperity_max multiply = 100 }
	}
	set_variable = { name = hd_dc_pressure_band value = hd_dc_pressure_band_value }
}

hd_dc_pressure_phase_end_effect = {
	hd_dc_sync_pressure_effect = yes
}

# Scope: participant character. Remove only this system's modifier source.
hd_dc_remove_character_pressure_effect = {
	remove_character_modifier = hd_dc_pressure_character_2
	remove_character_modifier = hd_dc_pressure_character_3
	remove_character_modifier = hd_dc_pressure_character_4
	remove_character_modifier = hd_dc_pressure_character_5
	remove_variable = hd_dc_character_pressure_band
}
hd_dc_apply_character_pressure_effect = {
	if = {
		limit = {
			situation:dynastic_cycle = { has_variable = hd_dc_pressure_band }
			OR = {
				NOT = { has_variable = hd_dc_character_pressure_band }
				NOT = { var:hd_dc_character_pressure_band = situation:dynastic_cycle.var:hd_dc_pressure_band }
			}
		}
		hd_dc_remove_character_pressure_effect = yes
		set_variable = { name = hd_dc_character_pressure_band value = situation:dynastic_cycle.var:hd_dc_pressure_band }
		switch = {
			trigger = var:hd_dc_character_pressure_band
			2 = { add_character_modifier = hd_dc_pressure_character_2 }
			3 = { add_character_modifier = hd_dc_pressure_character_3 }
			4 = { add_character_modifier = hd_dc_pressure_character_4 }
			5 = { add_character_modifier = hd_dc_pressure_character_5 }
		}
	}
}

# Scope: county. Leaving the region removes only the live opinion source;
# ownership and regional membership never erase the stock-loss account.
hd_dc_remove_county_pressure_effect = {
	remove_county_modifier = hd_dc_pressure_county_2
	remove_county_modifier = hd_dc_pressure_county_3
	remove_county_modifier = hd_dc_pressure_county_4
	remove_county_modifier = hd_dc_pressure_county_5
	remove_variable = hd_dc_county_pressure_band
	remove_variable = hd_dc_county_recovery_since
	remove_variable = hd_dc_county_recovery_wait
}

hd_dc_apply_county_pressure_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = hd_dc_county_pressure_band }
				NOT = { var:hd_dc_county_pressure_band = situation:dynastic_cycle.var:hd_dc_pressure_band }
			}
		}
		remove_county_modifier = hd_dc_pressure_county_2
		remove_county_modifier = hd_dc_pressure_county_3
		remove_county_modifier = hd_dc_pressure_county_4
		remove_county_modifier = hd_dc_pressure_county_5
		set_variable = { name = hd_dc_county_pressure_band value = situation:dynastic_cycle.var:hd_dc_pressure_band }
		switch = {
			trigger = var:hd_dc_county_pressure_band
			2 = { add_county_modifier = { modifier = hd_dc_pressure_county_2 } }
			3 = { add_county_modifier = { modifier = hd_dc_pressure_county_3 } }
			4 = { add_county_modifier = { modifier = hd_dc_pressure_county_4 } }
			5 = { add_county_modifier = { modifier = hd_dc_pressure_county_5 } }
		}
	}
	if = {
		limit = { var:hd_dc_county_pressure_band > 1 }
		remove_variable = hd_dc_county_recovery_since
		remove_variable = hd_dc_county_recovery_wait
		if = {
			limit = { NOT = { has_variable = hd_dc_loss_cycle } }
			set_variable = { name = hd_dc_loss_cycle value = yes }
			set_variable = { name = hd_dc_control_baseline value = county_control }
			set_variable = { name = hd_dc_development_baseline value = development_level }
			set_variable = { name = hd_dc_highest_loss_band value = 1 }
			set_variable = { name = hd_dc_control_processed value = 0 }
			set_variable = { name = hd_dc_development_processed value = 0 }
			set_variable = { name = hd_dc_control_paid value = 0 }
			set_variable = { name = hd_dc_development_paid value = 0 }
		}
		if = {
			limit = { var:hd_dc_county_pressure_band > var:hd_dc_highest_loss_band }
			set_variable = { name = hd_dc_control_loss_due value = hd_dc_control_loss_due_value }
			set_variable = { name = hd_dc_development_loss_due value = hd_dc_development_loss_due_value }
			change_county_control = { value = var:hd_dc_control_loss_due multiply = -1 }
			change_development_level = { value = var:hd_dc_development_loss_due multiply = -1 }
			change_variable = { name = hd_dc_control_paid add = var:hd_dc_control_loss_due }
			change_variable = { name = hd_dc_development_paid add = var:hd_dc_development_loss_due }
			set_variable = { name = hd_dc_control_processed value = hd_dc_control_loss_target_value }
			set_variable = { name = hd_dc_development_processed value = hd_dc_development_loss_target_value }
			set_variable = { name = hd_dc_highest_loss_band value = var:hd_dc_county_pressure_band }
			remove_variable = hd_dc_control_loss_due
			remove_variable = hd_dc_development_loss_due
		}
	}
	else_if = {
		limit = { has_variable = hd_dc_loss_cycle }
		if = {
			limit = { NOT = { has_variable = hd_dc_county_recovery_since } }
			set_variable = { name = hd_dc_county_recovery_since value = current_date }
			set_variable = { name = hd_dc_county_recovery_wait value = yes months = 60 }
		}
		else_if = {
			limit = { NOT = { has_variable = hd_dc_county_recovery_wait } }
			remove_variable = hd_dc_loss_cycle
			remove_variable = hd_dc_control_baseline
			remove_variable = hd_dc_development_baseline
			remove_variable = hd_dc_highest_loss_band
			remove_variable = hd_dc_control_processed
			remove_variable = hd_dc_development_processed
			remove_variable = hd_dc_control_paid
			remove_variable = hd_dc_development_paid
			remove_variable = hd_dc_county_recovery_since
		}
	}
}

hd_dc_apply_pressure_sources_effect = {
	if = {
		limit = { hd_dc_snapshot_valid_trigger = yes }
		every_situation_participant = { hd_dc_apply_character_pressure_effect = yes }
		every_in_list = {
			variable = hd_dc_pressure_counties
			limit = { NOT = { any_county_situation = { this = situation:dynastic_cycle } } }
			hd_dc_remove_county_pressure_effect = yes
		}
		clear_variable_list = hd_dc_pressure_counties
		every_situation_county = {
			save_scope_as = hd_dc_pressure_county
			situation:dynastic_cycle = {
				add_to_variable_list = { name = hd_dc_pressure_counties target = scope:hd_dc_pressure_county }
			}
			if = {
				limit = { situation:dynastic_cycle = { has_variable = hd_dc_pressure_interval_unproven } }
				remove_variable = hd_dc_county_recovery_since
				remove_variable = hd_dc_county_recovery_wait
			}
			hd_dc_apply_county_pressure_effect = yes
		}
		remove_variable = hd_dc_pressure_interval_unproven
	}
}
