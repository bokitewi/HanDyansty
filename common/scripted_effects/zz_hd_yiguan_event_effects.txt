# 取一个沦陷县作为北人籍贯（新生成角色的文化）。结果：scope:hd_yiguan_origin_county
hd_yiguan_pick_origin_county_effect = {
	clear_saved_scope = hd_yiguan_origin_county
	random_county_in_region = {
		region = hd_yiguan_north_region
		limit = { has_variable = hd_yiguan_occ }
		save_scope_as = hd_yiguan_origin_county
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_origin_county } }
		title:c_luoyang = { save_scope_as = hd_yiguan_origin_county }
	}
}

# Scope：南方宗主。取境内直辖一县（无同类修正者优先）。结果：scope:hd_yiguan_target_county
hd_yiguan_pick_domain_county_effect = {
	clear_saved_scope = hd_yiguan_target_county
	random_held_county = {
		limit = { NOT = { has_county_modifier = $MODIFIER$ } }
		weight = {
			base = 1
			modifier = {
				add = 4
				development_level >= 10
			}
		}
		save_scope_as = hd_yiguan_target_county
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_target_county } }
		random_held_county = { save_scope_as = hd_yiguan_target_county }
	}
}

# 生成北方士族（入宫廷）。$SKILL$：stewardship / learning。Scope：收留者。结果：scope:hd_yiguan_gentry
hd_yiguan_create_gentry_effect = {
	save_scope_as = hd_yiguan_host
	hd_yiguan_pick_origin_county_effect = yes
	scope:hd_yiguan_origin_county = {
		culture = { save_scope_as = hd_yiguan_gentry_culture }
		faith = { save_scope_as = hd_yiguan_gentry_faith }
		rite ?= { save_scope_as = hd_yiguan_gentry_rite }
	}
	create_character = {
		location = root.capital_province
		template = pool_repopulate_$SKILL$
		age = { 25 45 }
		gender_female_chance = 0
		dynasty = generate
		culture = scope:hd_yiguan_gentry_culture
		faith = scope:hd_yiguan_gentry_faith
		rite = scope:hd_yiguan_gentry_rite
		$SKILL$ = { 14 20 }
		save_scope_as = hd_yiguan_gentry
	}
	scope:hd_yiguan_gentry = {
		set_variable = hd_yiguan_northerner
	}
	add_courtier = scope:hd_yiguan_gentry
}

# Scope：南方参与者。取另一位南方一侧参与者（与你接壤者优先）。结果：scope:hd_yiguan_other_south
hd_yiguan_pick_other_south_effect = {
	save_scope_as = hd_yiguan_self
	clear_saved_scope = hd_yiguan_other_south
	situation:hd_yiguan_nandu ?= {
		every_situation_participant = {
			limit = {
				NOT = { this = scope:hd_yiguan_self }
				hd_yiguan_south_side_trigger = yes
			}
			add_to_list = hd_yiguan_south_pool
		}
	}
	random_in_list = {
		list = hd_yiguan_south_pool
		weight = {
			base = 1
			modifier = {
				add = 9
				any_realm_county = {
					any_neighboring_county = {
						holder ?= { top_liege = scope:hd_yiguan_self }
					}
				}
			}
		}
		save_scope_as = hd_yiguan_other_south
	}
}

# Scope：北方汉人封臣。取最近的南方宗主（与你接壤者优先）。结果：scope:hd_yiguan_near_south
hd_yiguan_pick_near_south_lord_effect = {

	situation:hd_yiguan_nandu ?= {

			limit = { hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_south_lord } }
			add_to_list = hd_yiguan_south_lords
		}
	}
	random_in_list = {
		list = hd_yiguan_south_lords
		weight = {
			base = 1
			modifier = {
				add = 9
				any_realm_county = {
					any_neighboring_county = {
						holder ?= { top_liege = scope:hd_yiguan_self.top_liege }
					}
				}
			}
		}
		save_scope_as = hd_yiguan_near_south
	}
}

# 授以前线一县：取宗主直辖、与沦陷县接壤的县（无则取非都城直辖县）
hd_yiguan_shuai_grant_county_effect = {
	clear_saved_scope = hd_yiguan_grant_county
	random_held_county = {
		limit = {
			NOT = { this = root.capital_county }
			any_neighboring_county = { has_variable = hd_yiguan_occ }
		}
		save_scope_as = hd_yiguan_grant_county
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_grant_county } }
		random_held_county = {
			limit = { NOT = { this = root.capital_county } }
			save_scope_as = hd_yiguan_grant_county
		}
	}
	if = {
		limit = { exists = scope:hd_yiguan_grant_county }
		add_courtier = scope:hd_yiguan_shuai
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_yiguan_grant_change
			add_claim_on_loss = no
		}
		scope:hd_yiguan_grant_county = {
			change_title_holder = {
				holder = scope:hd_yiguan_shuai
				change = scope:hd_yiguan_grant_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_yiguan_grant_change
	}
	else = {
		add_courtier = scope:hd_yiguan_shuai
	}
}

# 侨郡安户：境内直辖一县加侨郡，记入全局列表（局势结束 10 年后统一移除）

# 侨姓吴姓之争：$QIAO$ / $WU$ = up 或 down（扶侨姓：QIAO = up WU = down；扶吴姓反之）
# 侨姓：华夏传承、本人文化与其首府县文化不同（北来者）；吴姓：本人文化与首府县文化相同（土著）
hd_yiguan_qiao_wu_effect = {
	every_vassal = {
		limit = {
			hd_yiguan_is_huaxia_trigger = yes
			exists = capital_county
			NOT = { culture = capital_county.culture }
		}
		add_opinion = {
			target = root
			modifier = hd_yiguan_favor_$QIAO$_opinion
		}
	}
	every_vassal = {
		limit = {
			exists = capital_county
			culture = capital_county.culture
		}
		add_opinion = {
			target = root
			modifier = hd_yiguan_favor_$WU$_opinion
		}
	}
}

hd_yiguan_sinicization_step_effect = {
	save_scope_as = hd_dc_con_ruler
	set_variable = {
		name = hd_dc_con_sinicization_stage
		value = $STAGE$
	}
	every_vassal = {
		limit = { hd_dc_con_native_trigger = { CULTURE = scope:hd_dc_con_ruler.var:hd_dc_con_native_culture } }
		add_opinion = {
			target = scope:hd_dc_con_ruler
			modifier = hd_dc_con_sinicization_native_opinion
		}
	}
	every_vassal = {
		limit = { hd_dc_is_chinese_heritage_trigger = yes }
		add_opinion = {
			target = scope:hd_dc_con_ruler
			modifier = hd_dc_con_sinicization_han_opinion
		}
	}
	random = {
		chance = 20
		custom_tooltip = hd_dc_con_native_revolt_risk_tt
		hidden_effect = { hd_dc_con_native_revolt_effect = yes }
	}
	if = {
		limit = { var:hd_dc_con_sinicization_stage = 1 }
		trigger_event = {
			id = hd_yiguan.3002
			days = { 730 1095 }
		}
	}
	else_if = {
		limit = { var:hd_dc_con_sinicization_stage = 2 }
		trigger_event = {
			id = hd_yiguan.3003
			days = { 730 1095 }
		}
	}
	else_if = {
		limit = { var:hd_dc_con_sinicization_stage = 3 }
		trigger_event = {
			id = hd_yiguan.3004
			days = { 730 1095 }
		}
	}
	else = {
		remove_variable = hd_yiguan_north_reform
		hd_dc_con_complete_sinicization_effect = yes
	}
}
