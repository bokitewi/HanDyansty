# 事件冷却（本模块私有）。Scope：收事件的角色
# 设角色变量 hd_yiguan_event_cd（180 天），月度事件脉冲据此跳过此人
hd_yiguan_ev_cd_effect = {
	set_variable = { name = hd_yiguan_event_cd value = yes days = 180 }
}

# 取一个沦陷县作为北人籍贯（新生成角色的文化）。结果：scope:hd_yiguan_origin_county
hd_yiguan_pick_origin_county_effect = {
	clear_saved_scope = hd_yiguan_origin_county
	random_county_in_region = {
		region = hd_yiguan_north_region
		limit = { has_variable = hd_yiguan_occ }
		save_scope_as = hd_yiguan_origin_county
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_origin_county } }
		title:c_luoyang = { save_scope_as = hd_yiguan_origin_county }
	}
}

# Scope：新生成的北来角色（家族为新生成）。记"北人"，并把其家族记为侨姓（籍贯取 scope:hd_yiguan_origin_county）
# 侨姓家族变量见 §6 第 15 条；家族已有侨姓标记者不覆盖籍贯
hd_yiguan_ev_mark_qiao_effect = {
	set_variable = hd_yiguan_northerner
	house ?= {
		if = {
			limit = { NOT = { has_variable = hd_yiguan_qiao_house } }
			set_variable = { name = hd_yiguan_qiao_house value = yes }
			if = {
				limit = { exists = scope:hd_yiguan_origin_county }
				set_variable = { name = hd_yiguan_qiao_origin value = scope:hd_yiguan_origin_county }
			}
		}
	}
}

# Scope：收留者。取境内直辖一县（无同类修正者优先）。结果：scope:hd_yiguan_target_county
hd_yiguan_pick_domain_county_effect = {
	clear_saved_scope = hd_yiguan_target_county
	random_held_county = {
		limit = { NOT = { has_county_modifier = $MODIFIER$ } }
		weight = {
			base = 1
			modifier = {
				add = 4
				development_level >= 10
			}
		}
		save_scope_as = hd_yiguan_target_county
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_target_county } }
		random_held_county = { save_scope_as = hd_yiguan_target_county }
	}
}

# 生成北方士族（入宫廷，家族记为侨姓）。$SKILL$：stewardship / learning。Scope：收留者。结果：scope:hd_yiguan_gentry
hd_yiguan_create_gentry_effect = {

		faith = { save_scope_as = hd_yiguan_gentry_faith }
		rite ?= { save_scope_as = hd_yiguan_gentry_rite }
	}
	create_character = {
		location = root.capital_province
		template = pool_repopulate_$SKILL$
		age = { 25 45 }
		gender_female_chance = 0
		dynasty = generate
		culture = scope:hd_yiguan_gentry_culture
		faith = scope:hd_yiguan_gentry_faith
		rite = scope:hd_yiguan_gentry_rite
		$SKILL$ = { 14 20 }
		save_scope_as = hd_yiguan_gentry
	}
	scope:hd_yiguan_gentry = {
		hd_yiguan_ev_mark_qiao_effect = yes
	}
	add_courtier = scope:hd_yiguan_gentry
}

# Scope：南方政权或北方汉人。另取一位南方政权或北方汉人参与者（与你接壤者优先）。结果：scope:hd_yiguan_other_south
hd_yiguan_pick_other_south_effect = {
	save_scope_as = hd_yiguan_self
	clear_saved_scope = hd_yiguan_other_south
	situation:hd_yiguan_nandu ?= {
		every_situation_participant = {
			limit = {
				NOT = { this = scope:hd_yiguan_self }
				is_alive = yes
				OR = {
					hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_south }
					hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_north_han }
				}
			}
			add_to_list = hd_yiguan_south_pool
		}
	}
	random_in_list = {
		list = hd_yiguan_south_pool
		weight = {
			base = 1
			modifier = {
				add = 9
				any_realm_county = {
					any_neighboring_county = {
						holder ?= { top_liege = scope:hd_yiguan_self }
					}
				}
			}
		}
		save_scope_as = hd_yiguan_other_south
	}
}

# 授以前线一县：取直辖、与沦陷县接壤的县（无则取非都城直辖县）
hd_yiguan_shuai_grant_county_effect = {
	clear_saved_scope = hd_yiguan_grant_county
	random_held_county = {
		limit = {
			NOT = { this = root.capital_county }
			any_neighboring_county = { has_variable = hd_yiguan_occ }
		}
		save_scope_as = hd_yiguan_grant_county
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_grant_county } }
		random_held_county = {
			limit = { NOT = { this = root.capital_county } }
			save_scope_as = hd_yiguan_grant_county
		}
	}
	scope:hd_yiguan_shuai = { hd_yiguan_ev_mark_qiao_effect = yes }	# 落户：家族记侨姓
	if = {
		limit = { exists = scope:hd_yiguan_grant_county }
		add_courtier = scope:hd_yiguan_shuai
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_yiguan_grant_change
			add_claim_on_loss = no
		}
		scope:hd_yiguan_grant_county = {
			change_title_holder = {
				holder = scope:hd_yiguan_shuai
				change = scope:hd_yiguan_grant_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_yiguan_grant_change
	}
	else = {
		add_courtier = scope:hd_yiguan_shuai
	}
}

# 起事县的胡人宗主须眼下没有在应付义军 / 举兵起义 / 起兵北还（否则战争侧起兵效果会静默跳过）；

		}
		save_scope_as = hd_yiguan_north_target
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_north_target } }
		random_county_in_region = {
			region = hd_yiguan_north_region
			limit = {
				has_variable = hd_yiguan_occ
				holder ?= { top_liege = { hd_yiguan_no_active_uprising_trigger = yes } }
			}
			save_scope_as = hd_yiguan_north_target
		}
	}
	if = {
		limit = { exists = scope:hd_yiguan_north_target }
		hd_yiguan_spawn_uprising_effect = {
			LEADER = scope:hd_yiguan_shuai
			COUNTY = scope:hd_yiguan_north_target

		}
		else_if = {
			# 起兵效果先建冒险者营地再宣战，宣战失败时首领可能已是营地主，只收留仍是普通角色者

			scope:hd_yiguan_shuai = { hd_yiguan_ev_mark_qiao_effect = yes }
			add_courtier = scope:hd_yiguan_shuai
		}
	}

# 立侨州。Scope：君主；须先有 scope:hd_yiguan_qiao_title。结果：scope:hd_yiguan_qiao_holder
# 君主为王国级以上时授给有地的北人或侨姓封臣；君主只是公国级（侨州与其同级，封臣受封会与主君平级）或无合适封臣时自领
hd_yiguan_create_qiao_effect = {
	clear_saved_scope = hd_yiguan_qiao_holder
	if = {
		limit = {
			any_held_title = {	# 无地官职头衔不算（同 hd_yiguan_is_regime_base_trigger 的写法）
				tier >= tier_kingdom
				is_landless_type_title = no
			}
		}
		random_vassal = {
			limit = {
				is_landed = yes
				OR = {
					has_variable = hd_yiguan_northerner
					hd_yiguan_is_qiaoxing_trigger = yes
				}
			}
			save_scope_as = hd_yiguan_qiao_holder
		}
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_qiao_holder } }
		save_scope_as = hd_yiguan_qiao_holder
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = hd_yiguan_qiao_change
		add_claim_on_loss = no
	}
	scope:hd_yiguan_qiao_title = {
		change_title_holder = {
			holder = scope:hd_yiguan_qiao_holder
			change = scope:hd_yiguan_qiao_change
		}
	}
	resolve_title_and_vassal_change = scope:hd_yiguan_qiao_change
}

# 侨姓吴姓之争：$QIAO$ / $WU$ = up 或 down（扶侨姓：QIAO = up WU = down；扶吴姓反之）。Scope：南方政权
# 侨姓：家族带侨姓标记、本人为北人，或华夏传承而本人文化与其首府县文化不同（北来者）；封臣与成年廷臣都算
# 吴姓：吴姓判定成立、非北人，且本人文化与其首府县文化相同（土著）
# 家族带 hd_yiguan_qiaowu_married（侨吴联姻，士族侧设）者不卷入
hd_yiguan_qiao_wu_effect = {
	save_scope_as = hd_yiguan_qw_lord
	every_vassal = {
		limit = {
			NOT = { house ?= { has_variable = hd_yiguan_qiaowu_married } }
			OR = {
				hd_yiguan_is_qiaoxing_trigger = yes
				has_variable = hd_yiguan_northerner
				AND = {
					hd_yiguan_is_huaxia_trigger = yes
					exists = capital_county
					NOT = { culture = capital_county.culture }
				}
			}
		}
		add_to_list = hd_yiguan_qw_qiao
	}
	every_courtier = {
		limit = {
			is_adult = yes
			NOT = { house ?= { has_variable = hd_yiguan_qiaowu_married } }
			OR = {
				hd_yiguan_is_qiaoxing_trigger = yes
				has_variable = hd_yiguan_northerner
			}
		}
		add_to_list = hd_yiguan_qw_qiao
	}
	every_vassal = {
		limit = {
			NOT = { is_in_list = hd_yiguan_qw_qiao }
			NOT = { house ?= { has_variable = hd_yiguan_qiaowu_married } }
			NOT = { has_variable = hd_yiguan_northerner }
			hd_yiguan_is_wuxing_trigger = yes
			exists = capital_county
			culture = capital_county.culture
		}
		add_to_list = hd_yiguan_qw_wu
	}
	every_in_list = {
		list = hd_yiguan_qw_qiao
		add_opinion = {
			target = scope:hd_yiguan_qw_lord
			modifier = hd_yiguan_favor_$QIAO$_opinion
		}
	}
	every_in_list = {
		list = hd_yiguan_qw_wu
		add_opinion = {
			target = scope:hd_yiguan_qw_lord
			modifier = hd_yiguan_favor_$WU$_opinion
		}
