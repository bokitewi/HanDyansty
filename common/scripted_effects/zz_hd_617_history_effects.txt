# 被杀或病死。电脑角色直接死；玩家角色不杀，只记日志
hd_617_die_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ }
			debug_log = "HD_617 DIED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_617 DIE_SKIPPED_PLAYER $WHO$"
		}
	}
}
hd_617_kill_effect = {

}

# $NEW$ 接收 $OLD$ 的全部实地头衔（含 h_china）与封臣：同一个 change 里先转头衔、后改投（照 hd_22_take_realm_effect）
hd_617_take_realm_effect = {
	$NEW$ = { save_scope_as = hd_617_new_ruler }
	create_title_and_vassal_change = { type = granted save_scope_as = hd_617_take_change add_claim_on_loss = no }
	$OLD$ = {
		every_held_title = {
			limit = {
				is_noble_family_title = no
				tgp_is_minister_title = no
				is_landless_type_title = no
				tier >= tier_county
			}
			change_title_holder = {
				holder = scope:hd_617_new_ruler
				change = scope:hd_617_take_change
				take_baronies = yes
			}
		}
		every_vassal = {
			limit = { NOT = { this = scope:hd_617_new_ruler } }
			change_liege = { liege = scope:hd_617_new_ruler change = scope:hd_617_take_change }
		}
	}
	resolve_title_and_vassal_change = scope:hd_617_take_change
}

# 把郡头衔 $DUCHY$ 与其持有者手里的本郡各县交给 $TO$（scope）；$KEEP$ 是不交出的县（照 hd_22_give_duchy_effect，计划复核必改 7）
hd_617_give_duchy_effect = {
	if = {
		limit = { exists = $DUCHY$.holder }
		$DUCHY$.holder = { save_scope_as = hd_617_old_duke }
		create_title_and_vassal_change = { type = granted save_scope_as = hd_617_duchy_change add_claim_on_loss = no }
		$DUCHY$ = {
			if = {
				limit = { NOT = { holder ?= $TO$ } }
				change_title_holder = { holder = $TO$ change = scope:hd_617_duchy_change }
			}
			every_de_jure_county = {
				limit = {
					holder ?= scope:hd_617_old_duke
					NOT = { this = $KEEP$ }
					NOT = { holder ?= $TO$ }
				}
				change_title_holder = { holder = $TO$ change = scope:hd_617_duchy_change take_baronies = yes }
			}
		}
		resolve_title_and_vassal_change = scope:hd_617_duchy_change
	}
}

		}
		# 玩家扮演时功名补到帝国级预期，否则丞相授不上（531 执行后复核必改 5）

			limit = {

				merit_level < merit_level_expected_empire_value
			}
			while = {
				limit = { merit_level < merit_level_expected_empire_value }
				change_merit_level = 1
			}
		}
		hd218_grant_chengxiang_effect = yes
		WJ_install_historical_secretariat_effect = { OFFICE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang LAW = WJ_grand_chengxiang_law }
		hd_court_install_regency_effect = yes
		hd_court_set_stance_effect = { STANCE = moni }

			if = {
				limit = { has_diarchy_type = grand_secretariat diarch ?= character:hd_617_wang_shichong }
				set_diarchy_swing = 100
			}
		}
		title:h_china = { set_variable = { name = hd_dc_ten_qjc_regent value = character:hd_617_wang_shichong } }
		# 皇泰二年四月即篡，任期取 1 年
		add_character_flag = { flag = hd_court_regent_tenure years = 1 }

	}
	else = { debug_log = "HD_617 REGENCY WANG_SHICHONG SKIPPED" }
}
