#   ("legacy ministry-title transaction was removed", always = no),
#   导致授内阁职位不再授朝臣头衔(自动授予/移除链断裂)。
# 本文件 per-key 覆盖恢复原版正文(程序化提取自
#   game\common\scripted_effects\10_dlc_tgp_scripted_effects.txt), 并追加:
#     got/fired 末尾 -> 按实际资格维护原生开府自动档，保留独立手动授予。
#     辟召只读真实开府合同；旧开府修饰符及其生命周期已移除。
# 依赖: 原版 TGP DLC 的 e_minister_* 头衔 + $MINISTER_TITLE$/$MINISTER_POSITION$ 参数;

got_minister_position_effect = {
	save_scope_as = new_minister

	if = { # If your liege has a ministry, set up titles if needed
		limit = {
			scope:councillor_liege = {
				this != scope:new_minister
				tgp_has_access_to_ministry_trigger = yes
			}
			not = { exists = scope:swapped_position }
			# 只隔离 WJ 扩展三职，原版九职继续使用同一授职链。
			trigger_if = {
				limit = {
					OR = {
						title:$MINISTER_TITLE$ = title:e_WJ_minister_tai_fu
						title:$MINISTER_TITLE$ = title:e_WJ_minister_zong_zheng
						title:$MINISTER_TITLE$ = title:e_WJ_minister_cheng_xiang
						title:$MINISTER_TITLE$ = title:e_WJ_minister_guang_lu_xun
					}
				}
				scope:councillor_liege = { government_has_flag = government_is_celestial }
			}
		}
		# Grant the ministry title if councillor doesn't have it
		if = {
			limit = {
				NOT = { has_title = title:$MINISTER_TITLE$ }
			}

			if = {
				limit = {
					title:$MINISTER_TITLE$ = { is_title_created = no }
				}
				create_title_and_vassal_change = {
					type = created
					save_scope_as = minister_title_creation_change
					add_claim_on_loss = no
				}
				title:$MINISTER_TITLE$ = {
					change_title_holder = {
						holder = scope:councillor_liege
						change = scope:minister_title_creation_change
					}
				}
				resolve_title_and_vassal_change = scope:minister_title_creation_change
			}
			create_title_and_vassal_change = {
				type = appointment
				save_scope_as = minister_title_change
				add_claim_on_loss = no
			}

			if = {
				limit = {
					scope:councillor_liege = {
						tgp_has_access_to_ministry_trigger = yes
					}
				}
				# retire from other ministry primary titles
				scope:new_minister = {
					every_held_title = {
						limit = {
							this != title:$MINISTER_TITLE$
							tgp_is_minister_title = yes
						}
						change_title_holder = {
							holder = current_heir
							change = scope:minister_title_change
						}
					}
				}
			}

			title:$MINISTER_TITLE$ = {
				change_title_holder = {
					holder = scope:new_minister
					change = scope:minister_title_change
				}
			}
			resolve_title_and_vassal_change = scope:minister_title_change
		}
		# Add the correct title law if it doesn't have it already
		if = {
			limit = {
				exists = title:$MINISTER_TITLE$.holder
				title:$MINISTER_TITLE$ = title:e_minister_grand_marshal
				NOT = {
					title:$MINISTER_TITLE$ = {
						has_title_law = celestial_grand_marshal_appointment_succession_law
					}
				}
			}
			title:$MINISTER_TITLE$ = {
				add_title_law = celestial_grand_marshal_appointment_succession_law
			}
		}
		else_if = {
			limit = {
				exists = title:$MINISTER_TITLE$.holder
				NOR = {
					title:$MINISTER_TITLE$ = title:e_minister_grand_marshal
					title:$MINISTER_TITLE$ = {
						has_title_law = celestial_ministry_appointment_succession_law
					}
				}
			}
			title:$MINISTER_TITLE$ = {
				add_title_law = celestial_ministry_appointment_succession_law
			}
		}

		# Make them a minister if they don't have the council position
		if = {
			limit = {
				NOT = { has_council_position = $MINISTER_POSITION$ }
			}
			top_liege = {
				assign_councillor_type = {
					type = $MINISTER_POSITION$
					target = scope:new_minister
					remove_existing_councillor = yes
				}
			}
		}
		# If we have any vassals remaining, let's move them to where they are supposed to be
		if = {
			limit = {
				vassal_count > 0
				# A real duke/king keeps the actual fief when entering the imperial council.
				NOT = { WJ_rf_has_fief = yes }
				# 主包指定辅政保留已有辖区，不因接入真实朝臣职位拆走下属。
				trigger_if = {
					limit = { scope:councillor_liege = { has_variable = hd_dong_fixed_regent } }
					NOT = { this = scope:councillor_liege.var:hd_dong_fixed_regent }
				}
				trigger_if = {
					limit = { title:h_china = { has_variable = hd_dc_ten_qjc_regent } }
					NOT = { this = title:h_china.var:hd_dc_ten_qjc_regent }
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = minister_vassal_change
			}
			every_vassal = {
				change_liege = {
					liege = top_liege # I.e. the top liege
					change = scope:minister_vassal_change
				}
			}
			resolve_title_and_vassal_change = scope:minister_vassal_change
		}
		# If the new minister should be the diarch, make it so
		top_liege ?= { WJ_sync_grand_secretariat_effect = yes }

		# Save and track accomplishment for the Gate domicile building
		if = {
			limit = {
				title:e_minister_chancellor = {
					this = title:$MINISTER_TITLE$
					holder = scope:new_minister
				}
			}
			scope:new_minister.house ?= {
				house_head ?= {
					if = {
						limit = {
							exists = domicile
						}
						domicile ?= {
							set_family_accomplishment_effect = { ACCOMPLISHMENT = family_accomplishment_minister }
						}
					}
				}
			}
		}
		if = {
			limit = {
				title:e_minister_grand_marshal = {
					this = title:$MINISTER_TITLE$
					holder = scope:new_minister
				}
			}
			scope:new_minister.house ?= {
				house_head ?= {
					if = {
						limit = {
							exists = domicile
						}
						domicile ?= {
							set_family_accomplishment_effect = { ACCOMPLISHMENT = family_accomplishment_grand_marshal }
						}
					}
				}
			}
		}
		if = { # make sure the minister is participant in the Dynastic Cycle situation
			limit = {
				NOT = {
					any_character_situation = {
						situation_type = dynastic_cycle
					}
				}
				top_liege ?= {
					any_character_situation = {
						situation_type = dynastic_cycle
					}
				}
			}

			situation:dynastic_cycle = {
				add_manual_participant = prev
			}
		}
		if = {
			limit = {
				NOT = { has_character_modifier = minister_scheme_slots_modifier }
			}
			add_character_modifier = minister_scheme_slots_modifier
		}
		# Move capital to realm capital
		scope:councillor_liege.capital_county = {
			save_scope_as = capital_county
		}
		if = {
			limit = {
				scope:councillor_liege.capital_county != title:$MINISTER_TITLE$.title_capital_county
			}
			title:$MINISTER_TITLE$ = {
				set_capital_county = scope:capital_county
			}
		}
		every_held_title = {
			limit = {
				is_noble_family_title = yes
			}
			set_capital_county = scope:capital_county
		}
		if = {
			limit = { is_ai = yes }
			domicile ?= {
				move_domicile = scope:capital_county.title_province
			}
		}
		if = {
			limit = {
				is_in_army = no
				is_travelling = no
				NOT = { exists = involved_activity }
			}
			set_location_to_default = yes
		}
	}
	# Native obligation write, only after every title/vassal transaction above resolves.
	# Correct a stale opening/default level for a real Han/Jin imperial councillor.
	if = {
		limit = {
			is_alive = yes
			is_independent_ruler = no
			government_has_flag = government_is_celestial
			government_has_flag = government_is_celestial
			liege = top_liege
			liege = scope:councillor_liege
			scope:councillor_liege = { tgp_has_access_to_ministry_trigger = yes }
			has_title = title:$MINISTER_TITLE$
			has_council_position = $MINISTER_POSITION$
			is_councillor_of = scope:councillor_liege
			tgp_is_any_minister = yes
			NOT = { exists = scope:swapped_position }
		}
		if = {
			limit = {
				has_title = title:e_minister_grand_marshal
				NOT = { vassal_contract_obligation_level:celestial_ministry_obligations = 2 }
			}
			vassal_contract_set_obligation_level = { type = celestial_ministry_obligations level = 2 }
		}
		else_if = {
			limit = {
				NOT = { has_title = title:e_minister_grand_marshal }
				NOT = { vassal_contract_obligation_level:celestial_ministry_obligations = 1 }
			}
			vassal_contract_set_obligation_level = { type = celestial_ministry_obligations level = 1 }
		}
	}
	if = {
		limit = {
			title:$MINISTER_TITLE$ = title:e_WJ_minister_tai_fu
			scope:councillor_liege = { government_has_flag = government_is_celestial }
		}
		add_character_flag = officer_tai_fu
	}
	if = {
		limit = {
			title:$MINISTER_TITLE$ = title:e_WJ_minister_zong_zheng
			scope:councillor_liege = { government_has_flag = government_is_celestial }
		}
		add_character_flag = officer_zong_zheng
	}
	if = {
		limit = {
			title:$MINISTER_TITLE$ = title:e_WJ_minister_cheng_xiang
			scope:councillor_liege = { government_has_flag = government_is_celestial }
		}
		add_character_flag = officer_cheng_xiang
	}
	if = {
		limit = {
			title:$MINISTER_TITLE$ = title:e_WJ_minister_guang_lu_xun
			is_alive = yes
			has_title = title:e_WJ_minister_guang_lu_xun
			has_council_position = WJ_minister_guang_lu_xun
			is_councillor_of = scope:councillor_liege
			top_liege = scope:councillor_liege
			scope:councillor_liege = { government_has_flag = government_is_celestial }
		}
		add_character_flag = officer_guang_lu_xun
	}
	if = {
		limit = { title:$MINISTER_TITLE$ = title:e_minister_grand_marshal }
		add_character_flag = officer_da_jiang_jun
	}
	WJ_reconcile_contract_privileges_effect = yes
}

# 调用作用域：完成改制的内阁主人；不遍历全图，不改变其他头衔或转移财产。

# Native startup and office handover share the same real title/law linkage.
WJ_start_grand_secretariat_effect = {
 if = {
  limit = { NOT = { has_diarchy_type = grand_secretariat } }
  if = { limit = { has_active_diarchy = yes } end_diarchy = yes }
  try_start_diarchy = grand_secretariat
 }
}

# Root is the emperor; OFFICE is an existing ministry title, not an identity cache.
WJ_install_secretariat_office_effect = {
 if = {
  limit = {
   exists = title:$OFFICE$.holder
   title:$OFFICE$.holder ?= {
    liege = prev
    is_diarch_valid_trigger = yes
   }
   OR = {
    NOT = { exists = diarch }
    NOT = { diarch = title:$OFFICE$.holder }
   }
  }
  WJ_start_grand_secretariat_effect = yes
  if = {
   limit = { has_diarchy_type = grand_secretariat }
   set_diarch = title:$OFFICE$.holder
  }
 }
}

WJ_sync_grand_secretariat_effect = {
 if = {
  limit = {
   has_title = title:h_china
   government_has_flag = government_is_celestial
   has_dlc_feature = all_under_heaven
  }
  if = {
   limit = {
    NOR = {
     has_realm_law = grand_chancellor_law
     has_realm_law = grand_marshal_law
     has_realm_law = grand_censor_law
     has_realm_law = WJ_grand_taiwei_law
     has_realm_law = WJ_grand_taifu_law
     has_realm_law = WJ_grand_chengxiang_law
    }
   }
   add_realm_law_skip_effects = grand_marshal_law
  }
  WJ_start_grand_secretariat_effect = yes
  switch = {
   trigger = has_realm_law
   grand_chancellor_law = { WJ_install_secretariat_office_effect = { OFFICE = e_minister_chancellor } }
   grand_marshal_law = { WJ_install_secretariat_office_effect = { OFFICE = e_minister_grand_marshal } }
   grand_censor_law = { WJ_install_secretariat_office_effect = { OFFICE = e_minister_censor } }
   WJ_grand_taiwei_law = { WJ_install_secretariat_office_effect = { OFFICE = e_minister_of_war } }
   WJ_grand_taifu_law = { WJ_install_secretariat_office_effect = { OFFICE = e_WJ_minister_tai_fu } }
   WJ_grand_chengxiang_law = { WJ_install_secretariat_office_effect = { OFFICE = e_WJ_minister_cheng_xiang } }
  }
 }
}
