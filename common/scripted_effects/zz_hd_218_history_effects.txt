# 找守方：$TARGET$（头衔）持有者往上找到天子的直属封臣（如曹操）；持有者所在国不在天子之下的，取其最高领主。
# 存为 scope:hd_sg218_def_$KIND$
hd_sg218_find_defender_effect = {
	$TARGET$.holder ?= {
		if = {
			limit = {
				exists = title:h_china.holder
				liege ?= title:h_china.holder
			}
			save_scope_as = hd_sg218_def_$KIND$
		}
		else_if = {
			limit = {
				exists = title:h_china.holder
				any_liege_or_above = { liege ?= title:h_china.holder }
			}
			every_liege_or_above = {
				limit = { liege ?= title:h_china.holder }
				save_scope_as = hd_sg218_def_$KIND$
			}
		}
		else = {
			top_liege = { save_scope_as = hd_sg218_def_$KIND$ }
		}
	}
}

# 找进行中的某场 218 战争：$WHO$ 是参战者。存为 scope:hd_sg218_war_$KIND$、hd_sg218_att_$KIND$、hd_sg218_def_$KIND$
hd_sg218_find_war_effect = {
	$WHO$ ?= {
		random_character_war = {
			limit = { var:hd_sg218_war_kind ?= flag:$KIND$ }
			save_scope_as = hd_sg218_war_$KIND$
			primary_attacker = { save_scope_as = hd_sg218_att_$KIND$ }
			primary_defender = { save_scope_as = hd_sg218_def_$KIND$ }
		}
	}
}

# $ATTACKER$ 对 scope:hd_sg218_def_$KIND$ 宣战，主目标 $TARGET$（头衔）。
# 成功则战争存为 scope:hd_sg218_war_$KIND$，设战争种类和全局标记 hd_sg218_$KIND$_active；失败写日志。
# 战争变量写法照 zz_hd_lubu_exile_effects.txt 的 random_character_war，加防守方条件区分同时进行的几场。
hd_sg218_start_war_effect = {
	$ATTACKER$ = {
		# 孙刘开局结盟（boki 10-12 20:28）：对盟友开战前先背盟（原版背盟惩罚：威望、"攻击盟友"修正）
		if = {
			limit = { is_allied_to = scope:hd_sg218_def_$KIND$ }
			break_alliance_due_to_attack_effect = { TARGET = scope:hd_sg218_def_$KIND$ }
		}
		start_war = {
			cb = hd_sg218_war_cb
			target = scope:hd_sg218_def_$KIND$
			target_title = $TARGET$
		}
		random_character_war = {
			limit = {
				using_cb = hd_sg218_war_cb
				primary_attacker = $ATTACKER$
				primary_defender = scope:hd_sg218_def_$KIND$
				NOT = { has_variable = hd_sg218_war_kind }
			}
			set_variable = { name = hd_sg218_war_kind value = flag:$KIND$ }
			save_scope_as = hd_sg218_war_$KIND$
		}
	}
	if = {
		limit = { exists = scope:hd_sg218_war_$KIND$ }
		set_global_variable = hd_sg218_$KIND$_active
		debug_log = "HD218 WAR_CONFIRMED $KIND$"
	}
	else = {
		debug_log = "HD218 WAR_START_FAILED $KIND$"
	}
}

# 封臣作攻方宣战不成时，改由其最高领主宣战、封臣参战（计划第 4 步"攻守双方"）
hd_sg218_start_war_fallback_effect = {
	hd_sg218_start_war_effect = { ATTACKER = $ATTACKER$ KIND = $KIND$ TARGET = $TARGET$ }
	if = {
		limit = { NOT = { exists = scope:hd_sg218_war_$KIND$ } }
		$ATTACKER$.top_liege = { save_scope_as = hd_sg218_fallback_$KIND$ }
		if = {
			limit = { NOT = { scope:hd_sg218_fallback_$KIND$ = $ATTACKER$ } }
			debug_log = "HD218 WAR_FALLBACK $KIND$"
			hd_sg218_start_war_effect = { ATTACKER = scope:hd_sg218_fallback_$KIND$ KIND = $KIND$ TARGET = $TARGET$ }
			scope:hd_sg218_war_$KIND$ ?= {
				if = {
					limit = { NOT = { is_participant = $ATTACKER$ } }
					add_attacker = $ATTACKER$
				}
			}
		}
	}
}

	hd218_war_result_kind_effect = { KIND = wuhuan RESULT = $RESULT$ }
	hd218_war_result_kind_effect = { KIND = wan RESULT = $RESULT$ }

hd218_war_result_kind_effect = {
	if = {
		# CB 结算时 root 是战争理由不是战争，变量要走 root.war（照原版 00_event_war.txt、本 mod zz_hd_uprising_wars.txt）。
		# 10-14 前直接读 var: 永远为空，_won 等标记从未设上，汉中之后的史实链全部卡住
		limit = {
			exists = root.war
			root.war.var:hd_sg218_war_kind ?= flag:$KIND$
		}
		set_global_variable = hd_sg218_$KIND$_$RESULT$
		if = {
			limit = { has_global_variable = hd_sg218_$KIND$_active }
			remove_global_variable = hd_sg218_$KIND$_active
		}
		debug_log = "HD218 WAR_RESULT $KIND$ $RESULT$"
	}
}

# $WHO$ 脱离领主独立（叛乱、据城）。用法照原版 becomes_independent
hd_sg218_independence_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				is_independent_ruler = no
			}
			create_title_and_vassal_change = { type = independency save_scope_as = hd_sg218_indep_change add_claim_on_loss = no }
			becomes_independent = { change = scope:hd_sg218_indep_change }
			resolve_title_and_vassal_change = scope:hd_sg218_indep_change
			debug_log = "HD218 INDEPENDENT $WHO$"
		}
	}
}

# $WHO$ 改投 $LIEGE$（献城、归附）
hd_sg218_swear_fealty_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes

			}
			create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd_sg218_fealty_change add_claim_on_loss = no }
			change_liege = { liege = $LIEGE$ change = scope:hd_sg218_fealty_change }
			resolve_title_and_vassal_change = scope:hd_sg218_fealty_change
			debug_log = "HD218 FEALTY $WHO$"
		}
	}
}

	$DUCHY$ = {
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_county
				holder ?= {
					NOT = { this = scope:hd_sg218_annexer }
