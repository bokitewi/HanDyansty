# Head character scope. These lists are presentation only; interactions recheck.
WJ_dd_refresh_district_people_effect = {
 save_scope_as = wj_dd_people_head
 every_in_list = { list = wj_dd_warring_people remove_from_list = wj_dd_warring_people }
 primary_title = {
  clear_variable_list = wj_dd_ui_defenders
  clear_variable_list = wj_dd_ui_officers
  clear_variable_list = wj_dd_ui_officers_kingdom
  clear_variable_list = wj_dd_ui_officers_duchy
  clear_variable_list = wj_dd_ui_officers_county
 }
 top_liege = {
  every_vassal_or_below = {
   limit = { WJ_dd_member_of_core_trigger = { CORE = scope:wj_dd_people_head.primary_title } }
   save_scope_as = wj_dd_people_person
   if = {
    limit = { WJ_contract_sanjie_visible_trigger = yes this != scope:wj_dd_people_head }
    scope:wj_dd_people_head.primary_title = { add_to_variable_list = { name = wj_dd_ui_officers target = scope:wj_dd_people_person } }
    if = { limit = { highest_held_title_tier >= tier_kingdom }
     scope:wj_dd_people_head.primary_title = { add_to_variable_list = { name = wj_dd_ui_officers_kingdom target = scope:wj_dd_people_person } }
    }
    else_if = { limit = { highest_held_title_tier = tier_duchy }
     scope:wj_dd_people_head.primary_title = { add_to_variable_list = { name = wj_dd_ui_officers_duchy target = scope:wj_dd_people_person } }
    }
    else_if = { limit = { highest_held_title_tier = tier_county }
     scope:wj_dd_people_head.primary_title = { add_to_variable_list = { name = wj_dd_ui_officers_county target = scope:wj_dd_people_person } }
    }
   }
   clear_variable_list = wj_dd_ui_attack_wars
   clear_variable_list = wj_dd_ui_defend_wars
   clear_variable_list = wj_dd_ui_main_wars
   clear_variable_list = wj_dd_ui_support_wars
   if = {
    limit = { is_at_war = yes }
    add_to_list = wj_dd_warring_people
    every_character_war = {
     save_scope_as = wj_dd_people_war
     if = {
      limit = { is_war_leader = scope:wj_dd_people_person }
      scope:wj_dd_people_person = { add_to_variable_list = { name = wj_dd_ui_main_wars target = scope:wj_dd_people_war } }
     }
     else = {
      scope:wj_dd_people_person = { add_to_variable_list = { name = wj_dd_ui_support_wars target = scope:wj_dd_people_war } }
     }
     if = {
      limit = { is_attacker = scope:wj_dd_people_person }
      scope:wj_dd_people_person = { add_to_variable_list = { name = wj_dd_ui_attack_wars target = scope:wj_dd_people_war } }
     }
     else = {
      scope:wj_dd_people_person = { add_to_variable_list = { name = wj_dd_ui_defend_wars target = scope:wj_dd_people_war } }
     }
    }
   }
  }
 }
 # Sort only the war-card roster; leave officer management and war lists intact.
 ordered_in_list = {
  list = wj_dd_warring_people
  order_by = highest_held_title_tier
  max = { value = 0 every_in_list = { list = wj_dd_warring_people add = 1 } }
  check_range_bounds = no
  scope:wj_dd_people_head.primary_title = { add_to_variable_list = { name = wj_dd_ui_defenders target = prev } }
 }
 every_in_list = { list = wj_dd_warring_people remove_from_list = wj_dd_warring_people }
}
