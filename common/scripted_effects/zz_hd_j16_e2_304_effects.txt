# R-继统（02 §2.0）：$RULER$ 在世有地、$HEIR$ 在世且与之同一王朝时成立，存 scope:hd_j16_b1h_ok（＝君主，调用方据此判「条件已满足」）；
# 君主是 AI 且储君不是 $HEIR$ → set_designated_heir；玩家君主不改储（只记日志）。不在世、不同王朝 → 不存 scope、不动
hd_j16_b1_heir_effect = {
	if = {
		limit = { exists = scope:hd_j16_b1h_ok }
		clear_saved_scope = hd_j16_b1h_ok
	}
	hd_j16_304_res_effect = { ID = $RULER$ AS = hd_j16_b1h_r }
	hd_j16_304_res_effect = { ID = $HEIR$ AS = hd_j16_b1h_h }
	if = {
		limit = {
			scope:hd_j16_b1h_r ?= {
				is_landed = yes
				exists = dynasty
			}
			scope:hd_j16_b1h_h ?= {
				NOT = { this = scope:hd_j16_b1h_r }
				dynasty ?= scope:hd_j16_b1h_r.dynasty
			}
		}
		scope:hd_j16_b1h_r = {
			save_scope_as = hd_j16_b1h_ok
			if = {
				limit = {
					is_ai = yes
					NOT = { designated_heir ?= scope:hd_j16_b1h_h }
				}
				set_designated_heir = scope:hd_j16_b1h_h
				debug_log = "HD_J16 b1 FIRE heir_set_$HEIR$"
			}
			else_if = {
				limit = { is_ai = no }
				debug_log = "HD_J16 b1 WAIT heir_player_$RULER$"
			}
		}
	}
}

# 尚书令（若空）：$LIEGE$ 的尚书令席（minister_personnel）空着时，授 $WHO$（在世、不是君主本人；AI 无地者先入其宫廷）。
# 写法照 hd_nbc_minister_effect（它只收人物 ID，这里收 scope）。scope:councillor_liege 由本效果设
hd_j16_b1_minister_effect = {

		limit = {
			NOT = { exists = scope:councillor_liege.cp:minister_personnel }

				is_alive = yes
				is_imprisoned = no
				NOT = { this = scope:councillor_liege }
				NOT = { has_title = title:e_minister_of_personnel }

# 史实战争开战：$ATT$ 对 $DEF$ 以 $TARGET$ 开 $KIND$ 场次（hd_tb_hist_start_war_effect：灭国场次走吞并 CB，其余 hd_nbc_start_war_effect 经 wct）；$ATT$ 是封臣且自己开不成时，
# 由其最高领主宣战、$ATT$ 加入攻方（照 hd_nbc_vassal_war_effect，但收 scope 不收 ID，TEMPLATE §5-4）。成功判据：exists = scope:hd_nbc_war_$KIND$
hd_j16_b1_war_effect = {

			$ATT$ = { is_independent_ruler = no }
		}
		if = {

# 夺位拒之（收尾 pX-A，Q1；照 E3-Z20 李城 1011 b 与 hd_j16_licheng_rebel_effect）。Root：被夺者（玩家，事件 root）。
# $WHO$（夺位者，人物 scope）在世有地 → 独立、得 root 主头衔宣称；记下该头衔（hd_j16_$KIND$_claim，节点据其持有者认守方，被夺者死后由继承人接着当守方），

				is_imprisoned = no

# 夺位拒之后的内战（节点 _go 分支调用）：守方 = hd_j16_$KIND$_claim 的现持有者（独立有地），目标 = 守方首府所在郡（PREF），
#   经 hd_j16_b4_civil_war_go_effect（PREF 不与叛方接壤时改取接壤县所在郡、关键县随之改），成功给 $ATT$ 事件兵 $SIZE$；不成记一次 WAIT，下一跳重试
hd_j16_b1_usurp_civil_war_effect = {
	if = {
		limit = { exists = scope:hd_j16_b1u_def }
		clear_saved_scope = hd_j16_b1u_def
	}
	if = {
		limit = { exists = scope:hd_j16_b1u_pref }
		clear_saved_scope = hd_j16_b1u_pref
	}
	global_var:hd_j16_$KIND$_claim.holder ?= {

			limit = {
				is_alive = yes
				is_landed = yes
				is_independent_ruler = yes
			}
			save_scope_as = hd_j16_b1u_def
			capital_county.duchy ?= { save_scope_as = hd_j16_b1u_pref }
		}
	}
	if = {
		limit = {
			exists = scope:hd_j16_b1u_def
			exists = scope:hd_j16_b1u_pref
			$ATT$ ?= {
				is_alive = yes
				is_landed = yes
				is_independent_ruler = yes
				NOT = { this = scope:hd_j16_b1u_def }
			}
		}
		hd_j16_b4_civil_war_go_effect = { ATT = $ATT$ DEF = scope:hd_j16_b1u_def PREF = scope:hd_j16_b1u_pref KIND = $KIND$ LOG = $LOG$ }
		if = {
			limit = { exists = scope:hd_nbc_war_$KIND$ }
			hd_j16_304_army_effect = { WHO = $ATT$ SIZE = $SIZE$ NAME = $NAME$ }
		}
	}
	else = {
		hd_j16_304_wait_effect = { KEY = $KIND$_rb LOG = $LOG$ WHY = rebel_not_ready }
	}
}
