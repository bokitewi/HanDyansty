# ============================================================================
# WJ 资望四轨统一入口
#
# 运行期仅向已有宦海特质且正式入仕者发放；不提前授特质，不积攒或补发试守经验。
# 正向 VALUE 先减半，再限制为 1～3，保留小数；0 与负数原值传递。
# 轨道数据直接由 CK3 保存；历史／测试初始化直接 add_trait_xp，保留原始数值。
# ============================================================================

WJ_ziwang_qingyi_xp_effect = {
	if = {
		limit = {
			has_trait = WJ_mingshi_trait
			WJ_has_entry_road_trigger = yes
		}
		add_trait_xp = {
			trait = WJ_mingshi_trait
			track = qingtan
			value = {
				value = $VALUE$
				if = {
					limit = { $VALUE$ > 0 }
					multiply = 0.5
					min = 1
					max = 3
				}
			}
		}
	}
}
