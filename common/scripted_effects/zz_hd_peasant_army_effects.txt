# 农民义军：为合格者开启故事
hd_peasant_army_try_start_effect = {
	if = {
		limit = {
			hd_peasant_army_eligible_trigger = yes
			NOT = { hd_peasant_army_has_story_trigger = yes }
		}
		create_story = story_hd_peasant_army
	}
}

# 农民义军：起义首领获得农民领袖特质（已有则不动）
hd_peasant_army_grant_trait_effect = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_trait = peasant_leader }
		}
		add_trait = peasant_leader
		add_trait_xp = {
			trait = peasant_leader
			value = $XP$
		}
	}
}
