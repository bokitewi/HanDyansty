WJ_cr_queue_empire_settlement = {
 save_scope_value_as = { name = wj_cr_inherited_empire value = no }
 if = { limit = { exists = scope:transfer_type OR = { scope:transfer_type = flag:inheritance scope:transfer_type = flag:appointment_succession scope:transfer_type = flag:abdication } }
  save_scope_value_as = { name = wj_cr_inherited_empire value = yes }
 }
 trigger_event = WJ_cr.7
}
WJ_cr_establish_empire = {
 if = { limit = { WJ_cr_empire_unsettled = yes }
  if = { limit = { exists = scope:wj_cr_inherited_empire scope:wj_cr_inherited_empire = yes }
   # Inheriting an existing imperial throne is not a second foundation.
   set_variable = wj_cr_empire_settled
  }
  else_if = { limit = { WJ_cr_empire_in_china = yes }
   set_variable = wj_cr_empire_settled
   WJ_miaoshi_initialize_effect = yes
   if = { limit = { WJ_cr_royal_origin = yes }
    add_legitimacy = 1000
    send_interface_toast = { type = event_toast_effect_good title = WJ_cr_empire_found_t left_icon = root custom_tooltip = WJ_cr_empire_found_tt }
   }
   else = {
    add_legitimacy = -1000
    send_interface_toast = { type = event_toast_effect_bad title = WJ_cr_empire_usurp_t left_icon = root custom_tooltip = WJ_cr_empire_usurp_tt }
   }
  }
 }
}

}
WJ_cr_complete = {
 if = { limit = { WJ_cr_can_plan = yes scope:activity = { has_variable = wj_cr_investiture_done NOT = { has_variable = wj_cr_paid } } }
  scope:activity = { set_variable = wj_cr_paid }
  add_realm_law = crowned_emperor set_variable = crowned_emperor_var remove_variable = crowned_king_var
  WJ_miaoshi_initialize_effect = yes
  # Native rewards are paid once; realm-name options keep their native separate naming effects.
  if = { limit = { scope:activity = { has_variable = wj_cr_oath var:wj_cr_oath = 1 } } set_variable = { name = diplomacy_oath_1 value = 0 } }
  else_if = { limit = { scope:activity = { has_variable = wj_cr_oath var:wj_cr_oath = 2 } } set_variable = { name = stewardship_oath_1 value = 0 } }
  coronation_disburse_rewards_effect = yes
  coronation_upgrade_oath_value_effect = yes
  # HD 218 start: finish the king-to-emperor proclamation (no-op without hd218_wang_display)
  hd_sg218_jianzuo_complete_effect = yes
  WJ_mr_end_request = yes WJ_mr_clear_rank = yes remove_variable = WJ_mr_award_realm
  WJ_title_style_reconcile = yes
  # The naming event consumes this permit; an open choice must not expire with game time.
  set_variable = wj_cr_allow_naming
  tgp_fire_china_realm_name_event_effect = yes
  trigger_event = WJ_cr.4
 }
 WJ_cr_cleanup = yes
