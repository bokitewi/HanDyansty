# Assessment roster, read-only selection. Never calls assessment reconciliation or scoring.

WJ_ty_local_picker_init = {
 WJ_ty_local_picker_clear = yes
 set_variable = { name = wj_ty_local_filter_tier value = 0 }
 set_variable = { name = wj_ty_local_filter_grade value = 0 }
 WJ_ty_local_refresh = yes
}
WJ_ty_local_refresh = {
 if = { limit = { WJ_ty_local_picker_context = yes }
  save_scope_as = wj_ty_operator clear_variable_list = wj_ty_candidates
  var:wj_ty_emperor = { every_vassal_or_below = { WJ_ty_add_candidate = yes } }
  WJ_ty_local_filter = yes
 }
}
WJ_ty_local_filter = {
 if = { limit = { WJ_ty_local_picker_context = yes }
  clear_variable_list = wj_ty_local_filtered_roster
  save_scope_as = wj_ty_operator save_scope_as = wj_kaoke_filter_viewer
  every_in_list = { variable = wj_ty_candidates
   limit = {
    WJ_ty_removal_candidate = { OPERATOR = scope:wj_ty_operator }
    OR = { NOT = { scope:wj_ty_operator = { has_variable = wj_ty_local_filter_relations } } WJ_kaoke_close_relation_trigger = yes }
    OR = { scope:wj_ty_operator.var:wj_ty_local_filter_tier = 0 highest_held_title_tier = scope:wj_ty_operator.var:wj_ty_local_filter_tier }
    OR = { scope:wj_ty_operator.var:wj_ty_local_filter_grade = 0 WJ_kaoke_grade_value = scope:wj_ty_operator.var:wj_ty_local_filter_grade }
   }
   scope:wj_ty_operator = { add_to_variable_list = { name = wj_ty_local_filtered_roster target = prev } }
  }
  set_variable = { name = wj_ty_local_page value = 0 }
  WJ_ty_local_page = yes
 }
}
WJ_ty_local_page = {
 clear_variable_list = wj_ty_local_page_roster
 save_scope_as = wj_ty_local_page_viewer
 set_variable = { name = wj_ty_local_page_index value = 0 }
 save_scope_value_as = { name = wj_ty_local_page_start value = { value = var:wj_ty_local_page multiply = 20 } }
 save_scope_value_as = { name = wj_ty_local_page_end value = { value = scope:wj_ty_local_page_start add = 20 } }
 every_in_list = {
  variable = wj_ty_local_filtered_roster
  save_scope_as = wj_ty_local_page_candidate
  scope:wj_ty_local_page_viewer = {
   if = {
    limit = { var:wj_ty_local_page_index >= scope:wj_ty_local_page_start var:wj_ty_local_page_index < scope:wj_ty_local_page_end }
    add_to_variable_list = { name = wj_ty_local_page_roster target = scope:wj_ty_local_page_candidate }
   }
   change_variable = { name = wj_ty_local_page_index add = 1 }
  }
 }
 set_variable = { name = wj_ty_local_filtered_count value = var:wj_ty_local_page_index }
 remove_variable = wj_ty_local_page_index
}
