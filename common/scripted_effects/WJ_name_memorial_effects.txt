WJ_name_before_death = {
 if = { limit = { is_alive = yes WJ_han_jin_law_trigger = yes WJ_inner_emperor_trigger = no NOT = { has_variable = wj_last_hanjin } }
  set_variable = { name = wj_last_hanjin value = yes }
  save_scope_as = wj_last_person
  clear_saved_scope = wj_capture_relation
  clear_saved_scope = wj_capture_relative
  WJ_name_capture_office = yes
  if = { limit = {
   NOT = { exists = scope:wj_capture_office }
   WJ_peer_valid = no WJ_harem_member = no
   NOR = { exists = primary_title WJ_name_office_councillor = yes has_any_court_position = yes WJ_mufu_member_valid = yes }

  }

  if = { limit = { exists = scope:wj_capture_title } set_variable = { name = wj_last_title value = scope:wj_capture_title } }

}
