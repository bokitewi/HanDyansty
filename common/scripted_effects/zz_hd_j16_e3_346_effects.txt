hd_j16_node_ranmin_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = hd_nbc_ranmin_done }
			current_date >= 350.2.1
		}
		if = {
			limit = { current_date >= 352.1.1 }
			set_global_variable = hd_nbc_ranmin_done
			set_global_variable = hd_j16_ranmin_skip
			debug_log = "HD_J16 e3_r10 SKIP window_closed"
		}
		else_if = {
			limit = { hd_j16_dead_trigger = { ID = hd_nbc_ran_min } }
			set_global_variable = hd_nbc_ranmin_done
			set_global_variable = hd_j16_ranmin_skip
			debug_log = "HD_J16 e3_r10 SKIP actor_dead"
		}
		else = {
			if = {
				limit = { exists = scope:hd_j16_rm_ran }
				clear_saved_scope = hd_j16_rm_ran
			}
			hd_j16_resolve_effect = { ID = hd_nbc_ran_min AS = hd_j16_rm_ran }
			if = {
				limit = {
					has_global_variable = hd_j16_shahu_done
					hd_j16_dead_trigger = { ID = shangdang_shi_010 }
					exists = scope:hd_j16_rm_ran
					scope:hd_j16_rm_ran = {
						is_adult = yes
						is_imprisoned = no
					}
					title:d_weijun.holder.top_liege ?= {
						dynasty ?= dynasty:shi_2
						highest_held_title_tier >= tier_kingdom
						NOT = { this = scope:hd_j16_rm_ran }
					}
				}
				set_global_variable = hd_nbc_ranmin_done
				scope:hd_j16_rm_ran = { trigger_event = hd_nbc.0130 }
				debug_log = "HD_J16 e3_r10 FIRE event_sent"
			}
			else_if = {
				limit = { NOT = { has_global_variable = hd_j16_ranmin_wlog } }
				set_global_variable = hd_j16_ranmin_wlog
				debug_log = "HD_J16 e3_r10 WAIT not_ready"
			}
		}
	}
}

# 内战开战（Z20 李城、Z40 石鉴；审查 10-27）：叛方据地多半与魏郡不接壤（石鉴据京兆），定死魏郡时 wct 回退到别的郡、关键县邺城不在目标里，结局恒判负。

		hd_j16_b4_war_go_effect = { ATT = $ATT$ DEF = $DEF$ TARGET = $PREF$ KIND = $KIND$ LOG = $LOG$ }

						holder ?= {

							}
						}
					}
