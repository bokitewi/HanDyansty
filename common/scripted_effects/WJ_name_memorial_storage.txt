# on_death still has living data. Save event scopes before inheritance clears it.
# Object/flag references and numeric/bool values use their respective native save interfaces.
WJ_name_save_memorial_field = {
 clear_saved_scope = wj_memorial_$KEY$
 var:$KEY$ ?= { save_scope_as = wj_memorial_$KEY$ }
}
WJ_name_save_memorial_value = {
 clear_saved_scope = wj_memorial_$KEY$
 if = { limit = { has_variable = $KEY$ }
  save_scope_value_as = { name = wj_memorial_$KEY$ value = var:$KEY$ }
 }
}
WJ_name_store_memorial_field = {
 if = { limit = { exists = scope:wj_memorial_$KEY$ NOT = { has_dead_character_variable = $KEY$ } }
  # ponytail: native dead variables require a duration; 10000 years covers supported campaigns.
  set_dead_character_variable = { name = $KEY$ value = scope:wj_memorial_$KEY$ years = 10000 }
 }
}
