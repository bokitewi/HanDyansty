# 只看性格：扩张分 > 守成分则扩张（两者皆 0 时扩张）。Scope：发动者
hd_strat_assign_mode_effect = {
	if = {
		limit = {
			OR = {
				hd_court_stance_moni_trigger = yes	# 2026-10-10 乱世霸权（D4）：谋逆者更爱宣战，一律扩张
				hd_strat_expand_score_value > hd_strat_hold_score_value
				AND = {
					hd_strat_expand_score_value = 0
					hd_strat_hold_score_value = 0
				}
			}
		}
		set_variable = { name = hd_strat_mode value = flag:expand }
		if = {
			limit = { NOT = { has_character_modifier = hd_strat_expander_modifier } }
			add_character_modifier = hd_strat_expander_modifier
		}
	}
	else = {
		set_variable = { name = hd_strat_mode value = flag:hold }
		remove_character_modifier = hd_strat_expander_modifier
	}
}

# story 结束时清理。Scope：发动者
hd_strat_cleanup_effect = {
	remove_variable = hd_strat_region

	remove_variable = hd_strat_mode
	remove_character_modifier = hd_strat_expander_modifier
}

# 脉冲。Scope：发动者
hd_strat_pulse_effect = {
	if = {
		limit = {
			hd_strat_pulse_ready_trigger = yes
			var:hd_strat_mode ?= flag:expand
			has_variable = hd_strat_region
		}
		save_scope_as = hd_strat_actor
		var:hd_strat_region = { save_scope_as = hd_strat_region }
		hd_strat_find_target_effect = yes
		if = {
			limit = { exists = scope:hd_strat_target }
			hd_strat_engage_target_effect = yes
		}
		else = {
			hd_strat_next_region_effect = yes
			hd_submission_queue_followup_effect = { DELAY_MIN = 10 DELAY_MAX = 40 }
		}
	}
}

# 选目标：1) 相邻且非挚友 2) 区内兜底扫描（每年最多一次）3) 相邻挚友（最后才打）
hd_strat_find_target_effect = {
	clear_saved_scope = hd_strat_target
	ordered_neighboring_top_liege_realm_owner = {
		limit = {
			hd_strat_region_target_trigger = yes
			hd_strat_friendly_target_trigger = no
		}
		order_by = hd_strat_target_priority_value
		save_scope_as = hd_strat_target
	}
	# 184 剧本流程重制 D34（2026-10-21）：去掉不接壤的区内兜底扫描（飞地与乱吃地来源之一），只打接壤势力。
	if = {
		limit = {
			always = no
			NOT = { exists = scope:hd_strat_target }
			NOT = { has_character_flag = hd_strat_scan_cooldown }
		}
		add_character_flag = { flag = hd_strat_scan_cooldown days = 365 }
		every_county_in_region = {
			region = scope:hd_strat_region
			limit = { exists = holder }
			holder.top_liege = {
				if = {
					limit = {
						NOT = { is_in_list = hd_strat_candidates }
						hd_strat_region_target_trigger = yes
						hd_strat_friendly_target_trigger = no
					}
					add_to_list = hd_strat_candidates
				}
			}
		}
		ordered_in_list = {
			list = hd_strat_candidates
			order_by = hd_strat_target_priority_value
			save_scope_as = hd_strat_target
		}
	}
	if = {
		limit = { NOT = { exists = scope:hd_strat_target } }
		ordered_neighboring_top_liege_realm_owner = {
			limit = { hd_strat_region_target_trigger = yes }
			order_by = hd_strat_target_priority_value
			save_scope_as = hd_strat_target
		}
	}
}

# 先招降：1) 仁义性格或挚友目标 -> 无论胜算都先招降 2) 其余目标只要对方按现有条件“可能接受”也先招降
# 被拒/无效后下次脉冲（offer_from 标记 120 天内）直接宣战。
hd_strat_engage_target_effect = {
	# 2026-10-25 归附谈判（boki 拍板）：AI 对象差 60 以内且付得起，就带着预开价先招降
	scope:hd_strat_actor = { save_scope_as = actor }
	scope:hd_strat_target = { save_scope_as = recipient }
	if = {
		limit = {
			scope:hd_strat_target = {
				is_ai = yes
				NOT = { var:hd_strat_offer_from ?= scope:hd_strat_actor }
				highest_held_title_tier < scope:hd_strat_actor.highest_held_title_tier
				hd_submission_recipient_legal_trigger = { LIEGE = scope:hd_strat_actor }
				NOT = { hd_submission_ai_refuses_trigger = { LIEGE = scope:hd_strat_actor } }
			}
			hd_submission_liege_trigger = yes
		}
		hd_neg_ai_prepare_offer_effect = yes
	}
	if = {
		limit = {
			scope:hd_strat_target = {
				NOT = { var:hd_strat_offer_from ?= scope:hd_strat_actor }
				highest_held_title_tier < scope:hd_strat_actor.highest_held_title_tier
				hd_submission_recipient_legal_trigger = { LIEGE = scope:hd_strat_actor }
				NOT = { hd_submission_ai_refuses_trigger = { LIEGE = scope:hd_strat_actor } }
			}
			hd_submission_liege_trigger = yes
			OR = {
				hd_strat_offers_first_trigger = yes
				scope:hd_strat_target = { hd_strat_friendly_target_trigger = yes }
				scope:hd_strat_target = { var:hd_neg_viable ?= 1 }
				# 玩家目标：AI 无从预开价（由玩家在事件里开价），同样差 60 以内就先招降
				AND = {
					scope:hd_strat_target = { is_ai = no }
					hd_neg_base_accept_value >= {
						value = hd_neg_max_gap
						multiply = -1
					}
				}
			}
		}
		# 互动自带冷却：发起方 90 天、同一对象 2 年；冷却中不重复发送（避免大量脚本报错），也不转为宣战
		# 引擎冷却还可能来自别的调度器（开战前招降）发过的同一互动，所以发送前再确认互动当前可发，
		# 否则会先写下 offer_from 标记、发送失败，下次脉冲就直接转为宣战
		if = {
			limit = {
				NOT = { has_variable = hd_strat_offer_cd }
				is_character_interaction_valid = {
					recipient = scope:hd_strat_target
					interaction = hd_warlord_offer_vassalization_interaction
				}
			}
			set_variable = { name = hd_strat_offer_cd days = 90 }
			scope:hd_strat_target = {
				set_variable = { name = hd_strat_offer_from value = scope:hd_strat_actor days = 730 }
			}
			if = {
				# 2026-10-25 归附谈判：劝玩家归附改走事件，玩家可开价
				limit = { scope:hd_strat_target = { is_ai = no } }
				hd_neg_offer_to_player_effect = yes
			}
			else = {
				run_interaction = {
					interaction = hd_warlord_offer_vassalization_interaction
					actor = scope:hd_strat_actor
					recipient = scope:hd_strat_target
					send_threshold = decline
				}
			}
		}
	}
	else = {
		# 宣战前撤掉预开价并清掉借用的作用域，不带进宣战回调
		hd_neg_ai_offer_cleanup_effect = yes
		clear_saved_scope = actor
		clear_saved_scope = recipient
		hd_strat_declare_war_effect = yes
	}
	# 没发出去就撤掉预开价
	if = {
		limit = { exists = scope:recipient }
		hd_neg_ai_offer_cleanup_effect = yes
		clear_saved_scope = actor
		clear_saved_scope = recipient
	}
}

# 按阶段选 CB：斗争=吞并（hd_dc_tunbing_cb）；否则逐鹿中原（区内、与本国相邻的公国）。统一战争 CB 已删（boki 10-12 20:28）
hd_strat_declare_war_effect = {
	clear_saved_scope = hd_strat_war_title
	# 2026-10-25：吞并占位目标与 CB 的 valid_to_start 一致——主头衔无法理县（僭号动态帝国）时用首府县
	clear_saved_scope = hd_strat_tb_title
	scope:hd_strat_target = {
		if = {
			limit = { primary_title = { hd_tb_primary_title_target_trigger = { DEF = scope:hd_strat_target } } }
			primary_title = { save_scope_as = hd_strat_tb_title }
		}
		else_if = {	# 霸权/无地主头衔照旧不走吞并（落到下面逐鹿分支），只给无法理县的普通主头衔改用首府县
			limit = { primary_title = { tier < tier_hegemony is_landless_type_title = no } }
			capital_county ?= { save_scope_as = hd_strat_tb_title }
		}
	}
	# 挟天子 CB 已删（boki 10-13）
	if = {
		limit = {
			hd_cycle_phase_is_struggle_trigger = yes
			exists = scope:hd_strat_tb_title
			can_declare_war = {
				defender = scope:hd_strat_target
				casus_belli = hd_dc_tunbing_cb	# 2026-10-08：战略 AI 改用吞并 CB
				target_titles = { scope:hd_strat_tb_title }	# 2026-10-21：吞并现带占位目标头衔
			}
		}
		start_war = {
			cb = hd_dc_tunbing_cb
			target = scope:hd_strat_target
			target_title = scope:hd_strat_tb_title
		}
	}
	else = {
		scope:hd_strat_target = {
			random_realm_county = {
				limit = {
					title_province = { geographical_region = scope:hd_strat_region }
					any_neighboring_county = { holder.top_liege ?= scope:hd_strat_actor }
				}
				duchy ?= { save_scope_as = hd_strat_war_title }
			}
		}
		if = {
			limit = {
				exists = scope:hd_strat_war_title
				can_declare_war = {
					defender = scope:hd_strat_target
					casus_belli = chinese_consolidation_cb
					target_titles = { scope:hd_strat_war_title }
				}
			}
			start_war = {
				cb = chinese_consolidation_cb
				target = scope:hd_strat_target
				target_title = scope:hd_strat_war_title
			}
		}
	}
	add_character_flag = { flag = hd_strat_war_cooldown days = 30 }
}
