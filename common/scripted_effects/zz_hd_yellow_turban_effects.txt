# Effects for the 184 Yellow Turban historical uprising.

hd_yellow_turban_spawn_event_troops_effect = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_character_flag = hd_yellow_turban_event_troops_spawned }
			# Spawn only with a real province: an army spawned at a null province
			# corrupts the army object and hard-crashes once time starts moving.
			OR = {
				exists = capital_province
				exists = location
			}
		}
		save_temporary_scope_as = hd_yellow_turban_troop_owner
		if = {
			limit = { exists = capital_province }
			capital_province = { save_temporary_scope_as = hd_yellow_turban_spawn_province }
		}
		else = {
			location = { save_temporary_scope_as = hd_yellow_turban_spawn_province }
		}
		spawn_army = {
			name = hd_yellow_turban_event_troops
			levies = $LEVIES$
			location = scope:hd_yellow_turban_spawn_province
			origin = scope:hd_yellow_turban_spawn_province
			uses_supply = yes
			inheritable = yes
			save_temporary_scope_as = hd_yellow_turban_spawned_army
		}
		if = {
			limit = { exists = scope:hd_yellow_turban_spawned_army }
			scope:hd_yellow_turban_spawned_army = { assign_commander = scope:hd_yellow_turban_troop_owner }
		}
		add_character_flag = hd_yellow_turban_event_troops_spawned
	}
}

hd_yellow_turban_setup_armed_camp_effect = {
	if = {
		limit = {
			is_alive = yes
			has_government = landless_adventurer_government
			# add_realm_law_skip_effects requires an actual realm; campless
			# adventurer courtiers fail it and would get broken event troops.
			is_ruler = yes
		}
		add_realm_law_skip_effects = $PURPOSE$
		hd_yellow_turban_spawn_event_troops_effect = { LEVIES = 5000 }
	}
}

hd_yellow_turban_setup_unarmed_camp_effect = {
	if = {
		limit = {
			is_alive = yes
			has_government = landless_adventurer_government
			is_ruler = yes
		}
		add_realm_law_skip_effects = $PURPOSE$
	}
}

hd_yellow_turban_add_attacker_effect = {
	# This effect is called while the current scope is the newly created war.
	# Save that war explicitly; root is the opening event scope, not the war.
	save_temporary_scope_as = hd_yellow_turban_attacker_war
	if = {
		limit = {
			exists = $CHARACTER$
			$CHARACTER$ = {
				is_alive = yes
				NOT = { is_participant_in_war = scope:hd_yellow_turban_attacker_war }
			}
		}
		add_attacker = $CHARACTER$
	}
}

hd_yellow_turban_rebellion_effect = {
	character:zhang_jiao ?= { save_scope_as = hd_yellow_turban_zhang_jiao }
	character:zhang_man_cheng ?= { save_scope_as = hd_yellow_turban_zhang_mancheng }
	character:bo_cai ?= { save_scope_as = hd_yellow_turban_bo_cai }
	title:e_han.holder ?= { save_scope_as = hd_yellow_turban_defender }

	# Zhang Liang and Zhang Bao remain followers and commanders in Zhang Jiao's camp.
	character:zhang_liang ?= {
		if = {
			limit = { is_alive = yes }
			set_employer = scope:hd_yellow_turban_zhang_jiao
			set_knight_status = force
		}
	}
	character:zhang_bao ?= {
		if = {
			limit = { is_alive = yes }
			set_employer = scope:hd_yellow_turban_zhang_jiao
			set_knight_status = force
		}
	}

	# Julu front: Zhang Jiao and five northern/eastern commanders.
	scope:hd_yellow_turban_zhang_jiao = {
		start_war = {
			cb = hd_yellow_turban_rebellion_war
			target = scope:hd_yellow_turban_defender
			target_title = title:e_han
		}
		random_character_war = {
			limit = {
				primary_attacker = scope:hd_yellow_turban_zhang_jiao
				primary_defender = scope:hd_yellow_turban_defender
				using_cb = hd_yellow_turban_rebellion_war
			}
			save_scope_as = hd_yellow_turban_zhang_jiao_war
		}
	}
	if = {
		limit = { exists = scope:hd_yellow_turban_zhang_jiao_war }
		scope:hd_yellow_turban_zhang_jiao_war = {
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:guan_hai }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:zhang_lei_gong }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:xu_he }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:zhou_cang }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:pei_yuan_shao }
		}
		scope:hd_yellow_turban_zhang_jiao = {
			add_character_flag = hd_yellow_turban_primary_war_started
			hd_yellow_turban_spawn_event_troops_effect = { LEVIES = 15000 }
		}
	}
	else = {
		scope:hd_yellow_turban_zhang_jiao = { add_character_flag = hd_yellow_turban_primary_war_finished }
	}

	# Nanyang front: Zhang Mancheng, Zhao Hong, Han Zhong and Sun Xia.
	scope:hd_yellow_turban_zhang_mancheng = {
		start_war = {
			cb = hd_yellow_turban_rebellion_war
			target = scope:hd_yellow_turban_defender
			target_title = title:e_han
		}
		random_character_war = {
			limit = {
				primary_attacker = scope:hd_yellow_turban_zhang_mancheng
				primary_defender = scope:hd_yellow_turban_defender
				using_cb = hd_yellow_turban_rebellion_war
			}
			save_scope_as = hd_yellow_turban_zhang_mancheng_war
		}
	}
	if = {
		limit = { exists = scope:hd_yellow_turban_zhang_mancheng_war }
		scope:hd_yellow_turban_zhang_mancheng_war = {
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:zhao_hong_2 }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:han_zhong }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:sun_xia }
		}
		scope:hd_yellow_turban_zhang_mancheng = {
			add_character_flag = hd_yellow_turban_primary_war_started
			hd_yellow_turban_spawn_event_troops_effect = { LEVIES = 8000 }
		}
	}
	else = {
		scope:hd_yellow_turban_zhang_mancheng = { add_character_flag = hd_yellow_turban_primary_war_finished }
	}

	# Yingchuan and Runan front: Bo Cai and five local commanders.
	scope:hd_yellow_turban_bo_cai = {
		start_war = {
			cb = hd_yellow_turban_rebellion_war
			target = scope:hd_yellow_turban_defender
			target_title = title:e_han
		}
		random_character_war = {
			limit = {
				primary_attacker = scope:hd_yellow_turban_bo_cai
				primary_defender = scope:hd_yellow_turban_defender
				using_cb = hd_yellow_turban_rebellion_war
			}
			save_scope_as = hd_yellow_turban_bo_cai_war
		}
	}
	if = {
		limit = { exists = scope:hd_yellow_turban_bo_cai_war }
		scope:hd_yellow_turban_bo_cai_war = {
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:peng_tuo }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:he_yi }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:huang_shao }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:liu_pi }
			hd_yellow_turban_add_attacker_effect = { CHARACTER = character:gong_dou }
		}
		scope:hd_yellow_turban_bo_cai = {
			add_character_flag = hd_yellow_turban_primary_war_started
			hd_yellow_turban_spawn_event_troops_effect = { LEVIES = 9000 }
		}
	}
	else = {
		scope:hd_yellow_turban_bo_cai = { add_character_flag = hd_yellow_turban_primary_war_finished }
	}

	# Prevent the opening event from repeating even if one scripted declaration failed.
	scope:hd_yellow_turban_zhang_jiao = { add_character_flag = hd_yellow_turban_rebellion_started }
	hd_yellow_turban_try_resolve_campaign_failure_effect = yes
}

hd_yellow_turban_mark_primary_war_finished_effect = {
	scope:attacker ?= { add_character_flag = hd_yellow_turban_primary_war_finished }
	hd_yellow_turban_try_resolve_campaign_failure_effect = yes
}

# This is also called after the three declaration attempts. A failed declaration
# counts as an ended, non-winning front instead of leaving the campaign unresolved.
hd_yellow_turban_try_resolve_campaign_failure_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_yellow_turban_outcome_resolved }
			character:zhang_jiao ?= { has_character_flag = hd_yellow_turban_primary_war_finished }
			character:zhang_man_cheng ?= { has_character_flag = hd_yellow_turban_primary_war_finished }
			character:bo_cai ?= { has_character_flag = hd_yellow_turban_primary_war_finished }
		}
		hd_yellow_turban_campaign_failed_effect = yes
	}
}

# Any one enforced victory resolves the global campaign in favor of the Yellow Turbans.
# The other two wars are deliberately left running.
hd_yellow_turban_war_victory_effect = {
	scope:attacker ?= {
		add_character_flag = hd_yellow_turban_primary_war_finished
		add_character_flag = hd_yellow_turban_primary_war_won
	}
	if = {
		limit = { NOT = { exists = global_var:hd_yellow_turban_outcome_resolved } }
		set_global_variable = hd_yellow_turban_outcome_resolved
		set_global_variable = hd_yellow_turban_han_fell
		every_player = { trigger_event = hd_yellow_turban.0002 }
	}
}

# Failure is global only after all three primary wars have ended without a victory.
hd_yellow_turban_campaign_failed_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_yellow_turban_outcome_resolved } }
		set_global_variable = hd_yellow_turban_outcome_resolved
		set_global_variable = hd_yellow_turban_han_survived
		every_player = { trigger_event = hd_yellow_turban.0003 }

		character:liu_bei ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = hd_liu_bei_aftermath_started }
					NOT = { has_character_flag = hd_liu_bei_aftermath_pending }
				}
				add_character_flag = {
					flag = hd_liu_bei_aftermath_pending
					days = 150
				}
				trigger_event = {
					id = hd_liu_bei_aftermath.0001
					months = 3
				}
			}
		}

		hd_try_start_palace_crisis_effect = yes
	}
}
