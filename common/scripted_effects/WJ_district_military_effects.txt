# Actual prisoner belongs to the current justice minister, not the entry recipient
# or district head. The same native wrapper resolves relationships/protection.
WJ_dd_send_to_justice_effect = {
 if = {
  limit = { WJ_dd_send_to_justice_trigger = yes }
  scope:actor ?= { save_scope_as = wj_dd_entry_actor }
  scope:recipient ?= { save_scope_as = wj_dd_entry_recipient }
  scope:wj_dd_commander.top_liege.cp:minister_justice = {
   save_scope_as = wj_dd_jailer
   save_scope_as = actor
   # A scoped, synchronous native crime reason for the public military conviction.
   # It is removed below; the realm-bound conviction record is not consumed.
   add_opinion = { target = scope:wj_dd_target modifier = WJ_dd_lawful_detention_reason opinion = 0 }
  }
  scope:wj_dd_target = { save_scope_as = recipient }
  imprison_character_effect = {
   TARGET = scope:wj_dd_target
   IMPRISONER = scope:wj_dd_jailer
  }
  scope:wj_dd_jailer = {
   remove_opinion = { target = scope:wj_dd_target modifier = WJ_dd_lawful_detention_reason }
  }
  scope:wj_dd_entry_actor ?= { save_scope_as = actor }
  scope:wj_dd_entry_recipient ?= { save_scope_as = recipient }
 }
}
