hd_617_game_start_effect = {
	hd_617_cleanup_effect = yes
	# 衣冠南渡是 4 世纪的事，只设标记不启动（同 531）
	set_global_variable = hd_yiguan_started
	# 僚人入蜀 343–344 年已发生（随机规则下仍放行，计划第 13 节）
	if = {
		limit = { NOT = { has_game_rule = hd_liao_rule_random } }
		set_global_variable = { name = hd_liao_migration_launched value = yes }
	}
	# 天意青睐"每月主动开战"要求讨董链已结束（184 专用变量），同 304、370、531。
	# 它也会让战前招降、请求归附全局生效，群雄用 hd_617_rebel 标记挡住（zzzzzz_hd_submission_triggers.txt）
	set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
	# 终结乱世锁：唐建国（S8）完成或跳过、或 620.1.1 解锁（tgp_dynastic_cycle_decisions.txt）
	set_global_variable = hd_617_mandate_lock
	# 天子京城：江都（广陵县）。江都之变后移洛阳
	title:h_china = {
		set_capital_county = title:c_uuii_p_7062
		set_variable = { name = hd_court_capital_county value = title:c_uuii_p_7062 }
	}
	# 传国玉玺归炀帝。设 hd_han_treasures_initialized，184 的宝物脚本不会补发
	character:imperial_sui_2 ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { exists = global_var:hd_han_treasures_initialized }
			}
			hd_han_create_imperial_seal_effect = { OWNER = this }
			set_global_variable = hd_han_treasures_initialized
		}
	}
	hd_617_dynastic_cycle_effect = yes
	# 边境战争：群雄打隋的郡守、东都、西京时只对上被打的那一支，炀帝满 1095 天才能下场（照 22）
	hd_617_border_law_effect = { RULER = character:imperial_sui_2 }
	hd_617_tributaries_effect = yes
	hd_617_rebel_flags_effect = yes
	# 魏刀儿上谷人，库里写成乌桓（计划复核建议 12）
	character:20193002 ?= {
		if = {
			limit = { is_alive = yes NOT = { culture = culture:han } }
			set_culture = culture:han
		}
	}
	hd_617_realm_display_effect = yes
	debug_log = "HD_617 STAGE GAME_START"
}

	character:$DROP$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = no
				character:$KEEP$ ?= { is_alive = yes }
			}
			if = {
				limit = { is_female = no character:$KEEP$ = { is_female = no } }
				every_child = {
					limit = { is_alive = yes }
					if = {
						limit = { NOT = { father ?= character:$KEEP$ } }
						set_father = character:$KEEP$
					}
				}
			}
			death = { death_reason = death_vanished }
			debug_log = "HD_617 DUP_REMOVED $DROP$"
		}
	}
}

# 天朝循环：群雄逐鹿开局，五德取隋的火德，进度清零（写法同 hd_531_dynastic_cycle_effect）
hd_617_dynastic_cycle_effect = {
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = title:h_china.holder
			NOT = { exists = situation:dynastic_cycle }
		}
		start_situation = {
			type = dynastic_cycle
			start_phase = situation_dynastic_cycle_phase_chaos
		}
	}
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = situation:dynastic_cycle
			NOT = { exists = global_var:saas_dynastic_cycle_initialized }
		}
		situation:dynastic_cycle = {
			title:h_china = {
				set_variable = { name = years_since_imperial_examination value = 0 }
			}
			tgp_set_wuking_element_effect = { ELEMENT = fire }
			every_situation_participant = {
				recalculate_participant_group = situation:dynastic_cycle
			}
			every_situation_sub_region = {
				hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_hd_standoff }
			}
			trigger_event = { on_action = hd_617_dc_zero_progress days = 2 }
		}
		set_global_variable = {
			name = saas_dynastic_cycle_initialized
			value = yes
		}
		debug_log = "HD_617 DYNASTIC_CYCLE CHAOS FIRE"
	}
}

# 边境战争法：$RULER$ 是独立统治者且没有这条法律时补上（照 22 的 hd_22_border_law_effect，直接给统治者本人）
hd_617_border_law_effect = {
	$RULER$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_independent_ruler = yes
				NOT = { has_realm_law = top_liege_not_redirected_to_border_wars_law }
			}
			add_realm_law_skip_effects = top_liege_not_redirected_to_border_wars_law
			debug_log = "HD_617 BORDER_LAW_ADDED"
		}
	}
}

# 北方群雄称臣突厥：天朝政体的朝贡方挂游牧宗主，用 tributary_settled（原版 00_interaction_effects.txt 的游牧宗主分支，计划复核必改 4）。
# 不调 start_tributary_interaction_effect：它结尾会加停战（531 计划复核必改 1）
hd_617_tributary_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_independent_ruler = yes
				is_tributary = no
				character:ashina_016 ?= { is_alive = yes is_independent_ruler = yes has_government = nomad_government }
			}
			start_tributary = {
				contract_group = tributary_settled
				suzerain = character:ashina_016
			}
			if = {
				limit = { is_tributary = yes }
				hidden_effect = {
					tributary_contract_set_obligation_level = { type = default_tributary_taxes level = 1 }
					tributary_contract_set_obligation_level = { type = default_tributary_levies level = 1 }
					tributary_contract_set_obligation_level = { type = default_tributary_prestige level = 1 }
				}
				debug_log = "HD_617 TRIBUTARY $WHO$"
			}
			else = { debug_log = "HD_617 TRIBUTARY_FAILED $WHO$" }
		}
	}
}
