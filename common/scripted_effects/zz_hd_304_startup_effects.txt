	set_global_variable = hd_wuhu_chanyu_liuyuan_done
	set_global_variable = hd_wuhu_chanyu_lite_done
	set_global_variable = hd_wuhu_chanyu_shile_done
	# 慕容廆开局已持昌黎
	set_global_variable = { name = hd_hi_done_muronghui value = yes }

	set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
	# 天子京城：洛阳
	title:h_china = {
		set_capital_county = title:c_luoyang
		set_variable = { name = hd_court_capital_county value = title:c_luoyang }
	}
	# 传国玉玺归惠帝
	title:h_china.holder ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { exists = global_var:hd_han_treasures_initialized }
			}
			hd_han_create_imperial_seal_effect = { OWNER = this }
			set_global_variable = hd_han_treasures_initialized
		}
	}
	hd_304_dynastic_cycle_effect = yes
	hd_nbc_culture_effect = yes
	hd_304_tributaries_effect = yes
	hd_304_realm_display_effect = yes

# 天朝循环：群雄逐鹿（斗争）开局，五德取晋的金德；进度清零（写法同 hd_nbc_dynastic_cycle_effect，阶段不同）
hd_304_dynastic_cycle_effect = {
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = title:h_china.holder
			NOT = { exists = situation:dynastic_cycle }
		}
		start_situation = {
			type = dynastic_cycle
			start_phase = situation_dynastic_cycle_phase_chaos
		}
	}
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = situation:dynastic_cycle
			NOT = { exists = global_var:saas_dynastic_cycle_initialized }
		}
		situation:dynastic_cycle = {
			title:h_china = {
				set_variable = {
					name = years_since_imperial_examination
					value = 0
				}
			}
			tgp_set_wuking_element_effect = { ELEMENT = metal }
			every_situation_participant = {
				recalculate_participant_group = situation:dynastic_cycle
			}
			every_situation_sub_region = {
				hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_hd_standoff }
			}
			trigger_event = { on_action = hd_304_dc_zero_progress days = 2 }
		}
		set_global_variable = {
			name = saas_dynastic_cycle_initialized
			value = yes
		}
		debug_log = "HD_304 DYNASTIC_CYCLE CHAOS"
	}
}

# 诸王开局独立，奉惠帝为宗主（天朝朝贡合同：只有纳贡、册封，无参战义务）。司马略、段部是封臣，不单独朝贡。
hd_304_tributary_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_independent_ruler = yes
				is_tributary = no
				NOT = { this = title:h_china.holder }
			}
			# 不调 start_tributary_interaction_effect：它结尾会给双方加 25 年停战（执行后复核必改 3）。
			# 下面与它的天朝分支等价（低档贡赋、初始地位 10），只是不加停战
			start_tributary = {
				contract_group = tributary_celestial
				suzerain = title:h_china.holder
			}
			if = {
				limit = { is_tributary = yes }
				hidden_effect = {
					tributary_contract_set_obligation_level = { type = celestial_tribute_gold level = 1 }
					tributary_contract_set_obligation_level = { type = celestial_tribute_prestige level = 1 }
				}
				add_subject_standing = 10
			}
			debug_log = "HD_304 TRIBUTARY $WHO$"
		}
	}
}

# 开局「已称王」：WJ_cr_advance_king 的静默版（不弹国号事件、不置 wj_cr_allow_naming 以免日后误开国号选择）。
# 只对独立、有实地的州级君主生效；on_game_start 时政体判定可能未到位，after_lobby 会再调一次（幂等）。
hd_304_crowned_king_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				exists = primary_title
				WJ_cr_king = yes
				primary_title = { NOT = { has_variable = wj_cr_hereditary_king } }
			}
			primary_title = { set_variable = wj_cr_hereditary_king }
			WJ_miaoshi_initialize_effect = yes
			WJ_title_style_reconcile = yes
			debug_log = "HD_304 CROWNED_KING $WHO$"
		}
	}
}

# 王号取郡（成都王取成都县，照 190 阜陵王取县），食邑取该郡首县；直接写入并登记，不走册封界面、不扣授爵花费、不发册封事件。
# 王爵不要求在天子治下（WJ_peer_personal_valid），朝贡国、独立诸侯的王爵照样保留。授爵人为 h_china 持有者（惠帝）。
# 规则只放行男性宗室的 4 级王爵，3 级公爵会被判无效，所以东嬴公司马腾改王爵（boki 10-28 拍板：泰山王，东嬴县属泰山郡）。
# 司马略（越弟、青州）史实爵位存疑，不封；其余未列出的宗室开局无爵位。

hd304_peerage_effect = {
	title:h_china.holder ?= { save_scope_as = hd304_peer_issuer }
	if = {
		limit = {
			exists = scope:hd304_peer_issuer
			scope:hd304_peer_issuer = { WJ_peer_realm_head = yes }
		}
		hd304_peer_wang_effect = { WHO = si_ma_yong NAME = d_hejian }		# 河间王
		hd304_peer_wang_effect = { WHO = si_ma_ying NAME = c_uuii_p_4282 }	# 成都王（成都县）
		hd304_peer_wang_effect = { WHO = hd_304_si_ma_yue NAME = d_donghai }	# 东海王
		hd304_peer_wang_effect = { WHO = si_ma_teng NAME = d_taishan }		# 泰山王（史实东嬴公）
		hd304_peer_wang_effect = { WHO = si_ma_xiao NAME = d_fanyang }		# 范阳王
		hd304_peer_wang_effect = { WHO = si_ma_rui NAME = d_langya }		# 琅琊王
		hd304_peer_wang_effect = { WHO = si_ma_mao NAME = d_dongping }		# 东平王
		hd304_peer_wang_effect = { WHO = si_ma_chi NAME = d_yuzhang }		# 豫章王（皇太弟）
		debug_log = "HD_304 PEERAGE DONE"
	}
	else = { debug_log = "HD_304 PEERAGE NO_ISSUER" }
	clear_saved_scope = hd304_peer_issuer
}

# 宗室王爵。NAME：王号头衔（郡或县），食邑取其首县（县则取本身）。Scope：任意；需 scope:hd304_peer_issuer
hd304_peer_wang_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_male = yes
				NOT = { has_variable = wj_peer_rank }
				NOT = { has_trait = WJ_peerage_trait }
				title:$NAME$ = { WJ_peer_name_free = yes }
			}
			title:$NAME$ = {
				if = {
					limit = { tier = tier_county }
					save_scope_as = hd304_peer_food
				}
				else = {
					title_capital_county ?= { save_scope_as = hd304_peer_food }
				}
			}
			if = {
				limit = {
					exists = scope:hd304_peer_food
					scope:hd304_peer_food = { WJ_peer_county_free = yes }
				}
				set_variable = { name = wj_peer_personal value = yes }
				set_variable = { name = wj_peer_rank value = 4 }
				set_variable = { name = wj_peer_name value = title:$NAME$ }
				set_variable = { name = wj_peer_food value = scope:hd304_peer_food }
				set_variable = { name = wj_peer_realm value = scope:hd304_peer_issuer.primary_title }
				set_variable = { name = wj_peer_last_paid value = current_year }
				WJ_peer_bind_personal = yes
				debug_log = "HD_304 PEERAGE WANG $WHO$ $NAME$"
			}
			else = { debug_log = "HD_304 PEERAGE WANG_NO_FOOD $WHO$ $NAME$" }
			clear_saved_scope = hd304_peer_food
		}
		else = { debug_log = "HD_304 PEERAGE WANG_SKIPPED $WHO$" }
	}
}
