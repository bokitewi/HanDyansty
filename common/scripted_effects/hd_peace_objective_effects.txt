hd_peace_snapshot_objective_effect = {
	var:hd_peace_treaty_war = {
		primary_attacker = { save_scope_as = hd_peace_objective_attacker }
		primary_defender = { save_scope_as = hd_peace_objective_defender }
		if = { limit = { exists = claimant } claimant = { save_scope_as = hd_peace_objective_claimant } }
		if = { limit = { hd_peace_independence_war_trigger = yes } root = { set_variable = { name = hd_peace_objective_kind value = 1 } } }
		else_if = { limit = { hd_peace_subjugation_war_trigger = yes } root = { set_variable = { name = hd_peace_objective_kind value = 2 } } }
		else_if = { limit = { OR = { using_cb = claim_cb using_cb = crusading_claim_cb } } root = { set_variable = { name = hd_peace_objective_kind value = 3 } } }
		else_if = {
			limit = { hd_peace_artifact_war_trigger = yes }
			root = { set_variable = { name = hd_peace_objective_kind value = 5 } set_variable = { name = hd_peace_objective_artifact_claims value = 1 } }
			if = { limit = { using_cb = debug_artifact_war } root = { set_variable = { name = hd_peace_objective_artifact_claims value = 0 } } }
		}
		else_if = { limit = { using_cb = mandala_raze_capital_structure_cb } root = { set_variable = { name = hd_peace_objective_kind value = 6 } } }
		else_if = { limit = { using_cb = migration_cb } root = { set_variable = { name = hd_peace_objective_kind value = 7 } } }
		else_if = { limit = { using_cb = fp1_varangian_adventurer_conquest } root = { set_variable = { name = hd_peace_objective_kind value = 8 } } }
		else_if = { limit = { using_cb = legendary_adventure } root = { set_variable = { name = hd_peace_objective_kind value = 9 } } }
		else_if = { limit = { hd_peace_landless_relocation_war_trigger = yes } root = { set_variable = { name = hd_peace_objective_kind value = 10 } } }
		else = { root = { set_variable = { name = hd_peace_objective_kind value = 4 } } }
		casus_belli = {
			every_target_title = {
				limit = {
					exists = holder
					holder = { OR = { this = scope:hd_peace_objective_defender is_vassal_or_below_of = scope:hd_peace_objective_defender } }
				}
				add_to_list = hd_peace_objective_titles
			}
		}
	}
}

hd_peace_apply_objective_effect = {
	if = {
		limit = { has_variable = hd_peace_objective }
		if = {
			limit = { var:hd_peace_objective_kind = 1 }
			scope:hd_peace_objective_attacker = {
				if = { limit = { is_tributary = yes } end_tributary = yes }
				if = {
					limit = { exists = liege }
					create_title_and_vassal_change = { type = independency save_scope_as = hd_peace_objective_change add_claim_on_loss = no }
					becomes_independent = { change = scope:hd_peace_objective_change }
					resolve_title_and_vassal_change = scope:hd_peace_objective_change
				}
			}
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 2 }
			create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd_peace_objective_change add_claim_on_loss = no }
			scope:hd_peace_objective_defender = { change_liege = { liege = scope:hd_peace_objective_attacker change = scope:hd_peace_objective_change } }
			resolve_title_and_vassal_change = scope:hd_peace_objective_change
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 3 exists = scope:hd_peace_objective_claimant }
			# Reuse only the native title/vassal transaction, not CB rewards.
			create_title_and_vassal_change = { type = conquest_claim save_scope_as = hd_peace_objective_change add_claim_on_loss = yes }
			setup_claim_cb = {
				titles = hd_peace_objective_titles
				attacker = scope:hd_peace_objective_attacker
				defender = scope:hd_peace_objective_defender
				claimant = scope:hd_peace_objective_claimant
				change = scope:hd_peace_objective_change
			}
			resolve_title_and_vassal_change = scope:hd_peace_objective_change
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 5 }
			hd_peace_apply_artifact_objective_effect = yes
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 6 }
			hd_peace_apply_raze_objective_effect = yes
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 7 }
			hd_peace_apply_migration_objective_effect = yes
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 8 }
			hd_peace_prepare_relocation_effect = yes
			hd_peace_varangian_relocation_core_effect = yes
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 9 }
			hd_peace_prepare_relocation_effect = yes
			hd_peace_legendary_relocation_core_effect = yes
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 10 }
			hd_peace_prepare_relocation_effect = yes
			clear_saved_scope = claimant
			clear_saved_scope = supporter
			clear_saved_scope = new_liege
			clear_saved_scope = old_government
			hd_peace_landless_relocation_core_effect = yes
		}
		else_if = {
			limit = { var:hd_peace_objective_kind = 4 }
			create_title_and_vassal_change = { type = conquest save_scope_as = hd_peace_objective_change add_claim_on_loss = yes }
			every_in_list = {
				variable = hd_peace_cb_reserved_counties
				limit = {
					exists = holder
					holder = { OR = { this = scope:hd_peace_objective_defender is_vassal_or_below_of = scope:hd_peace_objective_defender } }
				}
				change_title_holder = { holder = scope:hd_peace_objective_attacker change = scope:hd_peace_objective_change take_baronies = yes }
			}
			resolve_title_and_vassal_change = scope:hd_peace_objective_change
		}
	}
	remove_variable = hd_peace_objective_kind
}
