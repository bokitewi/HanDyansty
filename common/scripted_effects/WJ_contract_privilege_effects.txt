# Character scope; native obligation writes only, no title transactions or source flags.
WJ_reconcile_contract_privileges_effect = {
	WJ_zhoumu_reconcile = yes
	WJ_zw_reconcile = yes
	WJ_mr_reconcile = yes
	if = {
		limit = { is_alive = yes }
		if = {
			limit = { WJ_contract_six_offices_trigger = yes }
			if = {
				limit = { NOT = { vassal_contract_obligation_level:WJ_kaifu_office = 1 } }
				vassal_contract_set_obligation_level = { type = WJ_kaifu_office level = 1 }
			}
		}
		else_if = {
			limit = { vassal_contract_obligation_level:WJ_kaifu_office > 0 }
			vassal_contract_set_obligation_level = { type = WJ_kaifu_office level = 0 }
		}
		# Leaving a six-office post does not revoke an independently granted, still-legal manual bit.
		if = {
			limit = {
				WJ_contract_kaifu_eligible_trigger = no
				vassal_contract_obligation_level:WJ_kaifu > 0
			}
			vassal_contract_set_obligation_level = { type = WJ_kaifu level = 0 }
		}
		if = {
			limit = {
				WJ_zw_valid = no
				vassal_contract_obligation_level:WJ_huangyue > 0
			}
			WJ_zw_clear = yes
		}
		WJ_clear_invalid_maojie_effect = yes
	}
}

# A district teardown calls this after removing the office. No kaifu state is touched.
WJ_clear_invalid_maojie_effect = {
	WJ_dd_remember_jie_effect = yes
	if = {
		limit = {
			is_alive = yes
			WJ_contract_sanjie_eligible_trigger = no
			vassal_contract_obligation_level:WJ_sanjie > 0
		}
		vassal_contract_set_obligation_level = { type = WJ_sanjie level = 0 }
	}
}

# Explicit conversion cleanup: laws' on_pass can run before the new law is installed.
WJ_clear_contract_privileges_effect = {
	WJ_zhoumu_clear = yes
	WJ_zw_clear = yes
	WJ_mr_clear_rank = yes
	if = { limit = { has_variable = WJ_mr_asker } WJ_mr_failure = yes }
	remove_variable = WJ_mr_award_realm
	if = {
		limit = {
			exists = var:wj_dd_held_core
			var:wj_dd_held_core.var:wj_dd_last_holder = this
		}
		var:wj_dd_held_core = { WJ_dd_dissolve_core_effect = yes }
	}
	if = {
		limit = { vassal_contract_obligation_level:WJ_sanjie > 0 }
		vassal_contract_set_obligation_level = { type = WJ_sanjie level = 0 }
	}
	if = {
		limit = { vassal_contract_obligation_level:WJ_kaifu > 0 }
		vassal_contract_set_obligation_level = { type = WJ_kaifu level = 0 }
	}
	if = {
		limit = { vassal_contract_obligation_level:WJ_kaifu_office > 0 }
		vassal_contract_set_obligation_level = { type = WJ_kaifu_office level = 0 }
	}
}
