# 存档诊断：全局变量 hds_diag_dy_*（计数），开新局跑过 200 年后在存档 variables 段里查。
#   hds_diag_dy_attempts      丹阳开战尝试次数
#   hds_diag_dy_war_started   成功开战次数
#   hds_diag_dy_no_candidate  丹阳各县的最高领主没有一个可作对手（结盟/停战/已在交战/非独立）
#   hds_diag_dy_cdw_false     有对手但 can_declare_war 全部失败（有地时的正常路径）
#   hds_diag_dy_forced        走了强制开战（营地身份，或连续失败 3 次）
#   hds_diag_dy_start_failed  start_war 之后仍然没有战争
#   hds_diag_dy_last_actor    最近一次尝试的发起人

# 全局计数 +1。
hds_diag_bump_effect = {
	if = {
		limit = { NOT = { has_global_variable = $NAME$ } }
		set_global_variable = { name = $NAME$ value = 0 }
	}
	change_global_variable = { name = $NAME$ add = 1 }
}

# 记录当前营地头衔（营地政体时主头衔就是营地）。有地以后据此销毁营地。
hds_sun_record_camp_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = no
			government_has_flag = government_is_landless_adventurer
			exists = primary_title
			primary_title = { is_landless_type_title = yes }
		}
		set_variable = { name = hds_sun_camp_title value = primary_title }
	}
}

# 营地身份的人拿到了封地（丹阳之战胜利）：改天朝政体、销毁营地头衔。
# 2026-10-25 起丹阳战胜利结算当天即调用（hd_dc_tunbing_cb 的 wu 分支）；孙家线 60 天心跳里的调用留作兜底。
hds_sun_landed_from_camp_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			government_has_flag = government_is_landless_adventurer
		}
		change_government = celestial_government
	}
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			has_variable = hds_sun_camp_title
		}
		if = {
			limit = {
				var:hds_sun_camp_title = {
					is_landless_type_title = yes
					exists = holder
					holder = prev
				}
			}
			destroy_title = var:hds_sun_camp_title
		}
		remove_variable = hds_sun_camp_title
	}
}

# 丹阳之战发起器（孙策、孙权、孙坚共用）。ROOT = 发起人；调用方已检查剧情阶段。
# 有地：先走合法宣战（can_declare_war）；连续失败 3 次后强制开战。
# 营地（无地）：直接强制开战。强制开战仍然尊重停战/结盟/已在交战（hds_candidate_enemy_trigger）。
hds_sun_danyang_war_effect = {
	if = {
		limit = {
			is_alive = yes
			is_adult = yes
			is_imprisoned = no
			hds_can_campaign_trigger = yes
		}
		remove_character_flag = sworn_peace
		save_scope_as = hds_campaign_actor
		hd_tb_arm_effect = { KIND = wu }
		clear_saved_scope = hds_target_enemy
		clear_saved_scope = hds_dy_forced_target
		title:d_danyang = { save_scope_as = hds_target_duchy }
		set_variable = { name = hds_requested_duchy value = title:d_danyang }
		add_character_flag = { flag = hds_launch_authorized days = 1 }
		hds_diag_bump_effect = { NAME = hds_diag_dy_attempts }
		set_global_variable = { name = hds_diag_dy_last_actor value = scope:hds_campaign_actor }
		# 1) 合法宣战
		if = {
			limit = { is_landed = yes }
			title:d_danyang = {
				random_de_jure_county = {
					limit = {
						exists = holder
						holder ?= { is_alive = yes exists = top_liege }
						holder.top_liege ?= {
							hds_candidate_enemy_trigger = yes
							NOT = { has_title = title:h_china }
							scope:hds_campaign_actor ?= {
								can_declare_war = {
									defender = prev
									casus_belli = hd_dc_tunbing_cb
									target_titles = { title:d_danyang }
								}
							}
						}
					}
					holder.top_liege ?= { save_scope_as = hds_target_enemy }
				}
			}
			if = {
				limit = { exists = scope:hds_target_enemy }
				hds_launch_selected_effect = { KIND = wu }
			}
		}
		# 2) 强制开战：营地身份，或有地但已连续失败 3 次
		if = {
			limit = {
				NOT = { any_character_war = { hd_tb_is_war_trigger = { K = wu } primary_attacker = scope:hds_campaign_actor } }
				OR = {
					is_landed = no
					AND = {
						has_variable = hds_dy_fail_count
						var:hds_dy_fail_count >= 3
					}
				}
			}
			# 优先打丹阳郡头衔持有者所属的最高领主（郡头衔持有者必在其国内，满足 CB 的"守方持有目标"判定）
			title:d_danyang = {
				if = {
					limit = {
						exists = holder
						holder.top_liege ?= { hds_candidate_enemy_trigger = yes NOT = { has_title = title:h_china } }
					}
					holder.top_liege = { save_scope_as = hds_dy_forced_target }
				}
				else = {
					random_de_jure_county = {
						limit = {
							exists = holder
							holder ?= { is_alive = yes exists = top_liege }
							holder.top_liege ?= { hds_candidate_enemy_trigger = yes NOT = { has_title = title:h_china } }
						}
						holder.top_liege ?= { save_scope_as = hds_dy_forced_target }
					}
				}
			}
			if = {
				limit = { exists = scope:hds_dy_forced_target }
				hds_diag_bump_effect = { NAME = hds_diag_dy_forced }
				hd_tb_arm_effect = { KIND = wu }
				start_war = {
					cb = hd_dc_tunbing_cb
					target = scope:hds_dy_forced_target
					target_title = title:d_danyang
				}
			}
		}
		# 结果
		if = {
			limit = { any_character_war = { hd_tb_is_war_trigger = { K = wu } primary_attacker = scope:hds_campaign_actor } }
			set_variable = { name = hds_dy_fail_count value = 0 }
			remove_character_flag = hds_war_cooldown
			hds_diag_bump_effect = { NAME = hds_diag_dy_war_started }
			random_character_war = {
				limit = { hd_tb_is_war_trigger = { K = wu } primary_attacker = scope:hds_campaign_actor }
				save_scope_as = hds_real_war
			}
			set_variable = { name = hds_current_war value = scope:hds_real_war }
		}
		else = {
			if = {
				limit = { NOT = { has_variable = hds_dy_fail_count } }
				set_variable = { name = hds_dy_fail_count value = 0 }
			}
			change_variable = { name = hds_dy_fail_count add = 1 }
			if = {
				limit = {
					title:d_danyang = {
						NOT = {
							any_de_jure_county = {
								exists = holder
								holder ?= { is_alive = yes exists = top_liege }
								holder.top_liege ?= { hds_candidate_enemy_trigger = yes NOT = { has_title = title:h_china } }
							}
						}
					}
				}
				hds_diag_bump_effect = { NAME = hds_diag_dy_no_candidate }
			}
			else_if = {
				limit = { exists = scope:hds_dy_forced_target }
				hds_diag_bump_effect = { NAME = hds_diag_dy_start_failed }
			}
			else = {
				hds_diag_bump_effect = { NAME = hds_diag_dy_cdw_false }
			}
		}
		remove_character_flag = hds_launch_authorized
		remove_variable = hds_requested_duchy
		add_character_flag = sworn_peace
	}
}

# ROOT = title:h_china（hds_sq_handover_tick）。
hds_sq_handover_effect = {
	if = {
		limit = {
			has_global_variable = hds_sq_pending
			NOT = { has_global_variable = hds_sq_leader }
		}
		clear_saved_scope = hds_sq_heir
		if = {
			limit = { has_global_variable = wj_hchar_sun_quan }
			global_var:wj_hchar_sun_quan ?= {
				if = { limit = { is_alive = yes } save_scope_as = hds_sq_heir }
			}
		}
		if = {
			limit = { NOT = { exists = scope:hds_sq_heir } }
			character:sun_quan ?= {
				if = { limit = { is_alive = yes } save_scope_as = hds_sq_heir }
			}
		}
		remove_global_variable = hds_sq_pending
		if = {
			limit = { exists = scope:hds_sq_heir }
			set_global_variable = { name = hds_sq_leader value = scope:hds_sq_heir }
			create_title_and_vassal_change = { type = granted save_scope_as = hds_sq_change add_claim_on_loss = no }
			title:h_china = {
				if = {
					limit = { has_variable_list = hds_sq_inherit_titles }
					every_in_list = {
						variable = hds_sq_inherit_titles
						limit = {
							exists = holder
							NOT = { holder = scope:hds_sq_heir }
							# 孙权本人已有封地时，不接营地/庄园这类无地头衔。
							NOT = {
								AND = {
									is_landless_type_title = yes
									scope:hds_sq_heir = { is_landed = yes }
								}
							}
						}
						change_title_holder = { holder = scope:hds_sq_heir change = scope:hds_sq_change take_baronies = yes }
					}
				}
			}
			resolve_title_and_vassal_change = scope:hds_sq_change
			title:h_china = { clear_variable_list = hds_sq_inherit_titles }
			scope:hds_sq_heir = {
				house ?= {
					if = {
						limit = { NOT = { house_head ?= scope:hds_sq_heir } }
						set_house_head = scope:hds_sq_heir
					}
				}
				if = { limit = { has_global_variable = hds_sq_from_departed } add_character_flag = hds_sq_departed }
				if = { limit = { has_global_variable = hds_sq_from_wu_complete } add_character_flag = hds_sq_wu_complete }
				if = {
					limit = {
						NOT = { has_title = title:k_uuii_yangzhou }
						NOT = { has_claim_on = title:k_uuii_yangzhou }
					}
					add_pressed_claim = title:k_uuii_yangzhou
				}
				hds_qxzl_protect_effect = yes
				# 兄长部曲：孙策领过保底兵的，孙权补一支（营地/首府所在地）。
				if = {
					limit = { has_global_variable = hds_sq_troops_due }
					hds_sq_reward_effect = yes
				}
				trigger_event = { id = hds.4100 days = 1 }
			}
			hds_schedule_effect = yes
		}
		remove_global_variable = hds_sq_from_departed
		remove_global_variable = hds_sq_from_wu_complete
	}
}

# 孙权每 60 天（hds_tick_effect）。ROOT = 孙权。
hds_sq_tick_effect = {
	if = {
		limit = { hds_sq_leader_trigger = yes is_alive = yes }
		save_scope_as = hds_sq_self
		hds_finish_title_transfer_effect = yes
		hds_sun_record_camp_effect = yes
		hds_sun_landed_from_camp_effect = yes
		if = {
			limit = { has_global_variable = hds_sq_troops_due }
			hds_sq_reward_effect = yes
		}
		# 孙策生前未出走：孙权成年后脱离原主。2026-10-25 boki：同孙策，不拿县，交还封地后到历阳立营。
		if = {
			limit = {
				NOT = { has_character_flag = hds_sq_departed }
				is_adult = yes
				is_imprisoned = no
				is_ruler = yes
			}
			if = {
				# 孙坚已自立（营地或地盘）：原样继承，不交不分、不挪营（同孙策）。
				limit = { hds_sj_departed_line_trigger = yes }
				hds_release_effect = yes
				hds_sun_record_camp_effect = yes
			}
			# 2026-10-28 boki：交地立营最多试 3 次，都没成就按有地身份放行出走，记诊断。
			else_if = {
				limit = {
					has_variable = hds_sq_depart_tries
					var:hds_sq_depart_tries >= 3
				}
				add_character_flag = hds_sq_depart_landed_ok
				hds_diag_bump_effect = { NAME = hds_diag_dep_landed_fallback }
				hds_release_effect = yes
			}
			else = {
				if = {
					limit = { NOT = { has_variable = hds_sq_depart_tries } }
					set_variable = { name = hds_sq_depart_tries value = 0 }
				}
				change_variable = { name = hds_sq_depart_tries add = 1 }
				hds_sun_camp_departure_effect = { DONE_FLAG = hds_sq_camped ESTATE_FLAG = hds_sq_camp_converted }
			}
			# 2026-10-28：交地立营这条线必须真的成了冒险者才算出走；没成就下次心跳重试（同孙策）。
			if = {
				limit = {
					is_independent_ruler = yes
					OR = {
						hds_sj_departed_line_trigger = yes
						has_character_flag = hds_sq_camped
						has_character_flag = hds_sq_depart_landed_ok
					}
				}
				add_character_flag = hds_sq_departed
				add_character_flag = sworn_peace
			}
		}
		if = {
			limit = { has_character_flag = hds_sq_departed }
			if = {
				limit = { is_adult = yes is_imprisoned = no exists = liege NOT = { is_independent_ruler = yes } }
				hds_release_effect = yes
			}
			if = {
				limit = { is_landed = yes NOT = { has_variable_list = hds_pending_titles } }
				hds_claim_owned_duchies_effect = { KINGDOM = title:k_uuii_yangzhou }
			}
			if = {
				limit = {
					NOT = { has_character_flag = hds_sq_wu_complete }
					NOT = { has_variable_list = hds_pending_titles }
					hds_sun_danyang_control_trigger = { ACTOR = scope:hds_sq_self }
				}
				add_character_flag = hds_sq_wu_complete
				remove_character_flag = sworn_peace
			}
			if = {
				limit = { NOT = { has_character_flag = hds_sq_wu_complete } }
				add_character_flag = sworn_peace
				if = {
					limit = { hds_can_campaign_trigger = yes }
					hds_sun_danyang_war_effect = yes
				}
			}
			else = {
				if = {
					limit = { hds_can_campaign_trigger = yes }
					hds_try_regional_war_effect = { KINGDOM = title:k_uuii_yangzhou }
				}
			}
		}
	}
}
