# 强制孙策继承：孙坚死后无论原继承是否完整，都补发头衔并打 hds_sc_inherited。任意作用域。
hds_sun_force_inherit_effect = {
	if = {
		limit = {
			exists = character:sun_jian
			character:sun_jian ?= { is_alive = no }
			AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } global_var:wj_hchar_sun_ce ?= { is_alive = yes NOT = { has_character_flag = hds_sc_inherited } } } trigger_else = { always = no } }
		}
		# 先走原有继承流程。
		hds_sun_succession_effect = yes
		if = {
			limit = { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } global_var:wj_hchar_sun_ce ?= { NOT = { has_character_flag = hds_sc_inherited } } } trigger_else = { always = no } } }
			if = { limit = { has_global_variable = wj_hchar_sun_ce } global_var:wj_hchar_sun_ce = {
				if = {
					limit = { NOT = { has_character_flag = hds_sc_titles_inherited } }
					if = {
						limit = { exists = title:h_china.var:hds_sun_house }
						title:h_china.var:hds_sun_house = { if = { limit = { has_global_variable = wj_hchar_sun_ce } set_house_head = global_var:wj_hchar_sun_ce } }
					}
					create_title_and_vassal_change = { type = granted save_scope_as = hds_sun_force_change add_claim_on_loss = no }
					title:h_china = {
						if = {
							limit = { has_variable_list = hds_sun_death_titles }
							every_in_list = {
								variable = hds_sun_death_titles
								limit = { NOT = { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } holder = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } } }
								change_title_holder = { holder = global_var:wj_hchar_sun_ce change = scope:hds_sun_force_change take_baronies = yes }
							}
						}
						if = {
							limit = { has_variable = hds_sun_family_title }
							var:hds_sun_family_title = {
								if = {
									limit = { NOT = { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } holder = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } } }
									change_title_holder = { holder = global_var:wj_hchar_sun_ce change = scope:hds_sun_force_change }
								}
							}
						}
					}
					resolve_title_and_vassal_change = scope:hds_sun_force_change
				}
				if = {
					limit = { exists = title:h_china.var:hds_sun_inheritance_seal }
					title:h_china.var:hds_sun_inheritance_seal = {
						if = { limit = { NOT = { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } artifact_owner = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } } } if = { limit = { has_global_variable = wj_hchar_sun_ce } set_owner = global_var:wj_hchar_sun_ce } }
					}
				}
				# 无条件完成继承标记，后续离开袁术/独立链由此继续。
				add_character_flag = hds_sc_titles_inherited
				add_character_flag = hds_sc_inherited
				hds_retire_legacy_controller_effect = yes
				if = {
					limit = { NOT = { has_character_flag = hds_sc_inheritance_notice } }
					hds_queue_event_effect = { EVENT = hds.4000 LEASE = hds_lease_4000 }
				}
			} }
		}
	}
}
