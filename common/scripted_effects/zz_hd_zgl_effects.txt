# 诸葛亮事件链（boki 2026-10-31，设计见 docs/诸葛亮司马懿事件链_设计草案.md 第 2 节）。事件在 events/zz_hd_zgl_events.txt。
# 由 hd_zs_pulse_effect 每 30 天调用；需 scope:hd_zs_shu（蜀主）、scope:hd_zs_wei（天子）。电脑照史实，玩家每节点可选。
# 诸葛亮自然死亡；死后 on_death 设 global_var:hd_zgl_dead，下次心跳跑五丈原、魏延之乱、接班、李平之死。

hd_zgl_pulse_effect = {
	if = {
		limit = { exists = scope:hd_zs_shu }
		# 诸葛亮在世、在蜀
		if = {
			limit = { global_var:hd_zs_p_zhu_ge_liang ?= { hd_zs_in_shu_trigger = yes } }
			hd_zgl_check_z1_effect = yes
			hd_zgl_kaifu_maintain_effect = yes
			hd_zgl_check_liyan_effect = yes
			hd_zgl_check_weiyan_effect = yes
			hd_zgl_check_disciples_effect = yes
			hd_zgl_check_mibiao_effect = yes
			hd_zgl_check_jiang_wei_effect = yes
			hd_zgl_check_beifa_effect = yes
			hd_zgl_check_inventions_effect = yes
		}
		# 诸葛亮死后
		if = {
			limit = { exists = global_var:hd_zgl_dead }
			hd_zgl_check_wuzhangyuan_effect = yes
			hd_zgl_check_weiyan_luan_effect = yes
			hd_zgl_check_succession_effect = yes
			hd_zgl_check_liping_death_effect = yes
		}
	}
}

hd_zgl_check_z1_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_z1_done }
			hd_zgl_regent_trigger = yes
		}
		if = {
			limit = { NOT = { exists = global_var:hd_zgl_z1_seen } }
			set_global_variable = hd_zgl_z1_seen
			set_global_variable = { name = hd_zgl_z1_wait value = yes days = 30 }
		}
		else_if = {
			limit = { NOT = { exists = global_var:hd_zgl_z1_wait } }
			set_global_variable = hd_zgl_z1_done
			hd_zgl_z1_effect = yes
			global_var:hd_zs_p_zhu_ge_liang = {
				if = {
					limit = { is_ai = no }
					trigger_event = hd_zgl.0010
				}
			}
		}
	}
}

hd_zgl_z1_effect = {
	global_var:hd_zs_p_zhu_ge_liang = {
		add_character_modifier = hd_zgl_kaifu_modifier
		change_merit = 200
		# 武乡侯：封爵 1 级（食邑上党铜鞮县的武乡男爵领），爵号特判显示「武乡侯」（WJ_peerage_names.txt）
		hd_zs_family_peer_effect = { ISSUER = scope:hd_zs_shu RANK = 1 FOOD = b_uuii_p_1144 }
		if = {
			limit = { has_variable = wj_peer_rank var:wj_peer_name ?= title:b_uuii_p_1144 }
			set_variable = hd_zgl_wuxiang
		}
		hd_zs_house_unity_effect = { VALUE = 20 DESC = hd_zs_unity_rise.desc }
	}
	hd_zs_party_nudge_effect = { WHO = global_var:hd_zs_p_zhu_ge_liang LORD = scope:hd_zs_shu POWER = 10 }
	hd_zs_party_stance_effect = { WHO = global_var:hd_zs_p_zhu_ge_liang LORD = scope:hd_zs_shu STANCE = jinqu }
	# 门生蒋琬（Z4）
	if = {
		limit = { global_var:hd_zs_p_jiang_wan ?= { hd_zs_in_shu_trigger = yes is_adult = yes } }
		hd_zs_disciple_effect = { MASTER = zhu_ge_liang DISCIPLE = jiang_wan }
	}
	# 治蜀与发明的计时（Z7）；李严求开府的计时（Z2-2，非托孤开局在此起算）
	set_global_variable = { name = hd_zgl_kejiao_wait value = yes days = 365 }
	set_global_variable = { name = hd_zgl_bazhen_wait value = yes days = 1095 }
	set_global_variable = { name = hd_zgl_liannu_wait value = yes days = 1825 }
	if = {
		limit = { NOT = { exists = global_var:hd_zgl_liyan_b_timer } }
		set_global_variable = hd_zgl_liyan_b_timer
		set_global_variable = { name = hd_zgl_liyan_b_wait value = yes days = 1095 }
	}
	hd_zs_news_effect = { A = global_var:hd_zs_p_zhu_ge_liang B = scope:hd_zs_shu EVENT = hd_zgl.0900 }
	debug_log = "HD_ZGL Z1 KAIFU"
}

# 开府修正随主政而存废
hd_zgl_kaifu_maintain_effect = {
	if = {
		limit = { exists = global_var:hd_zgl_z1_done }
		global_var:hd_zs_p_zhu_ge_liang = {
			if = {
				limit = { has_character_modifier = hd_zgl_kaifu_modifier NOT = { hd_zgl_regent_trigger = yes } }
				remove_character_modifier = hd_zgl_kaifu_modifier
			}
			else_if = {
				limit = { NOT = { has_character_modifier = hd_zgl_kaifu_modifier } hd_zgl_regent_trigger = yes }
				add_character_modifier = hd_zgl_kaifu_modifier
			}
		}
	}
}

hd_zgl_check_liyan_effect = {
	# ① 同受遗诏：托孤后李严在蜀
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_liyan_a_done }
			global_var:hd_zs_p_zhu_ge_liang = { has_character_flag = hd_sg218_tuogu }
			global_var:hd_zs_p_li_yan ?= { hd_zs_in_shu_trigger = yes }
		}
		set_global_variable = hd_zgl_liyan_a_done
		global_var:hd_zs_p_li_yan = {
			add_character_modifier = { modifier = hd_zgl_zhongduhu_modifier years = 10 }
			add_opinion = { target = global_var:hd_zs_p_zhu_ge_liang modifier = hd_zs_shou_yizhao_opinion }
			reverse_add_opinion = { target = global_var:hd_zs_p_zhu_ge_liang modifier = hd_zs_shou_yizhao_opinion }
		}
		if = {
			limit = { NOT = { exists = global_var:hd_zgl_liyan_b_timer } }
			set_global_variable = hd_zgl_liyan_b_timer
			set_global_variable = { name = hd_zgl_liyan_b_wait value = yes days = 1095 }
		}
		debug_log = "HD_ZGL LIYAN SHOU_YIZHAO"
	}
	# ② 求开府割巴州
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_liyan_b_done }
			exists = global_var:hd_zgl_liyan_b_timer
			NOT = { exists = global_var:hd_zgl_liyan_b_wait }
			hd_zgl_regent_trigger = yes
			global_var:hd_zs_p_li_yan ?= { hd_zs_in_shu_trigger = yes }
		}
		set_global_variable = hd_zgl_liyan_b_done
		global_var:hd_zs_p_zhu_ge_liang = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_zgl.0020
			}
			else = { hd_zgl_liyan_refuse_effect = yes }
		}
	}
	# ③ 运粮不继：北伐打满 180 天
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_liyan_c_done }
			hd_zgl_regent_trigger = yes
			global_var:hd_zs_p_li_yan ?= { hd_zs_in_shu_trigger = yes }
			scope:hd_zs_shu = {
				any_character_war = {
					hd_zs_shu_wei_war_trigger = yes
					war_days >= 180
				}
			}
		}
		set_global_variable = hd_zgl_liyan_c_done
		global_var:hd_zs_p_zhu_ge_liang = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_zgl.0030
			}
			else = { hd_zgl_liyan_depose_effect = yes }
		}
	}
}

	global_var:hd_zs_p_li_yan ?= {
		if = {
			limit = { is_alive = yes }

				limit = { is_ai = no }
				trigger_event = hd_zgl.0031

	hd_zs_news_effect = { A = global_var:hd_zs_p_li_yan B = global_var:hd_zs_p_zhu_ge_liang EVENT = hd_zgl.0910 }
	debug_log = "HD_ZGL LIYAN DEPOSED"
}

	}
}

hd_zgl_check_weiyan_effect = {
	# ① 子午谷奇谋：第一次北伐进行中
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_ziwu_done }
			exists = global_var:hd_sg218_beifa_active
			global_var:hd_zs_p_wei_yan ?= { hd_zs_in_shu_trigger = yes }
		}
		set_global_variable = hd_zgl_ziwu_done
		if = {
			limit = { global_var:hd_zs_p_zhu_ge_liang = { is_ai = no } }
			global_var:hd_zs_p_zhu_ge_liang = { trigger_event = hd_zgl.0040 }
		}
		else_if = {
			limit = { scope:hd_zs_shu = { is_ai = no } }
			scope:hd_zs_shu = { trigger_event = hd_zgl.0040 }
		}
		else = { hd_zgl_ziwu_reject_effect = yes }
	}
	# ② 魏杨不和：每年 30%，最多 3 次
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_weiyang_wait }
			NOT = { global_var:hd_zgl_weiyang_count ?= 3 }
			hd_zgl_regent_trigger = yes
			global_var:hd_zs_p_wei_yan ?= { hd_zs_in_shu_trigger = yes }
			global_var:hd_zs_p_yang_yi_2 ?= { hd_zs_in_shu_trigger = yes }
		}
		set_global_variable = { name = hd_zgl_weiyang_wait value = yes days = 365 }
		random = {
			chance = 30
			if = {
				limit = { NOT = { exists = global_var:hd_zgl_weiyang_count } }
				set_global_variable = { name = hd_zgl_weiyang_count value = 0 }
			}
			change_global_variable = { name = hd_zgl_weiyang_count add = 1 }
			global_var:hd_zs_p_zhu_ge_liang = {
				if = {
					limit = { is_ai = no }
					trigger_event = hd_zgl.0050
				}
				else = { hd_zgl_weiyang_both_effect = yes }
			}
		}
	}
}

		}

	global_var:hd_zs_p_wei_yan ?= { add_prestige = 100 }
	debug_log = "HD_ZGL ZIWU ACCEPTED"

# ③ 魏延之乱：诸葛亮死后，魏延、杨仪都在蜀
hd_zgl_check_weiyan_luan_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_zgl_luan_done }
			global_var:hd_zs_p_wei_yan ?= { hd_zs_in_shu_trigger = yes }
			global_var:hd_zs_p_yang_yi_2 ?= { hd_zs_in_shu_trigger = yes }
		}
		set_global_variable = hd_zgl_luan_done
		set_global_variable = { name = hd_zgl_yangyi_wait value = yes days = 365 }
		global_var:hd_zs_p_wei_yan = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_zgl.0060
			}
			else = {
				hd_zs_news_effect = { A = global_var:hd_zs_p_wei_yan B = global_var:hd_zs_p_yang_yi_2 EVENT = hd_zgl.0920 }
				death = { death_reason = death_execution }
			}
		}
		debug_log = "HD_ZGL WEIYAN_LUAN"
	}
	# 杨仪怨望（史实 235）：魏延之乱一年后
	if = {
		limit = {
			exists = global_var:hd_zgl_luan_done
			NOT = { exists = global_var:hd_zgl_yangyi_wait }
			NOT = { exists = global_var:hd_zgl_yangyi_done }
		}
		set_global_variable = hd_zgl_yangyi_done
		global_var:hd_zs_p_yang_yi_2 ?= {
			if = {
				limit = { hd_zs_in_shu_trigger = yes is_ai = yes }

				add_character_modifier = { modifier = hd_zgl_yuanwang_modifier years = 10 }
				hd_zs_depose_effect = { WHO = global_var:hd_zs_p_yang_yi_2 LORD = scope:hd_zs_shu }
				debug_log = "HD_ZGL YANGYI DEPOSED"
			}
		}
	}
}
