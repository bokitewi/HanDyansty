WJ_ty_commit = {
 if = { limit = { WJ_ty_execute_valid = yes }
  save_scope_as = wj_ty_operator
  var:wj_ty_emperor = { save_scope_as = wj_ty_emperor }
  var:wj_ty_proposer = { save_scope_as = wj_ty_proposer }
  var:wj_ty_target = { save_scope_as = wj_ty_target }
  var:wj_ty_target = { save_scope_as = wj_ty_sender }
  if = { limit = { var:wj_ty_topic = 4 } var:wj_ty_emperor = { save_scope_as = wj_ty_sender } }
  save_scope_value_as = { name = wj_ty_result_topic value = var:wj_ty_topic }
  save_scope_value_as = { name = wj_ty_result_tier value = var:wj_ty_tier }
  if = { limit = { var:wj_ty_topic = 4 }
   save_scope_value_as = { name = wj_ty_result_dismiss_kind value = var:wj_ty_dismiss_kind }
   save_scope_value_as = { name = wj_ty_result_removal_mode value = var:wj_ty_removal_mode }
   save_scope_value_as = { name = wj_ty_result_revoke_peerage value = 0 }
   if = { limit = { has_variable = wj_ty_revoke_peerage } save_scope_value_as = { name = wj_ty_result_revoke_peerage value = 1 } }
  }
  if = { limit = { var:wj_ty_topic = 1 }
   save_scope_value_as = { name = wj_ty_amount value = WJ_ty_merit_amount }
   scope:wj_ty_target = { change_merit = scope:wj_ty_amount set_variable = { name = wj_ty_merit_cooldown value = 1 years = 1 } }
   if = { limit = { var:wj_ty_tier = 3 } scope:wj_ty_target = {
     remove_character_modifier = WJ_ty_merit_1
     remove_character_modifier = WJ_ty_merit_2
     add_character_modifier = { modifier = WJ_ty_merit_3 }
   } }
   else_if = { limit = { var:wj_ty_tier = 2 } scope:wj_ty_target = {
    if = { limit = { NOR = { has_character_modifier = WJ_ty_merit_3 } }
     remove_character_modifier = WJ_ty_merit_1
     add_character_modifier = { modifier = WJ_ty_merit_2 }
    }
   } }
   else_if = { limit = { var:wj_ty_tier = 1 } scope:wj_ty_target = {
    if = { limit = { NOR = { has_character_modifier = WJ_ty_merit_2 has_character_modifier = WJ_ty_merit_3 } }
     add_character_modifier = { modifier = WJ_ty_merit_1 }
    }
   } }
  }
  else_if = { limit = { var:wj_ty_topic = 2 }
   save_scope_value_as = { name = wj_ty_amount value = WJ_ty_gold_amount }
   save_scope_value_as = { name = wj_ty_treasury value = WJ_ty_treasury_payment }
   save_scope_value_as = { name = wj_ty_private value = WJ_ty_private_payment }
   scope:wj_ty_emperor = {
    remove_treasury = scope:wj_ty_treasury
    remove_short_term_gold = scope:wj_ty_private
   }
   scope:wj_ty_target = { add_gold = scope:wj_ty_amount }
  }
  else_if = { limit = { var:wj_ty_topic = 3 } WJ_ty_appointment_commit = yes }
  else_if = { limit = { var:wj_ty_topic = 4 } WJ_ty_removal_commit = yes }
  if = { limit = { OR = { var:wj_ty_topic = 1 var:wj_ty_topic = 2 } scope:wj_ty_target != scope:wj_ty_proposer }
   if = { limit = { var:wj_ty_tier = 1 }
    scope:wj_ty_target = { add_opinion = { modifier = WJ_ty_reward_small_favor target = scope:wj_ty_proposer years = 5 } }
   }
   else = { scope:wj_ty_target = { add_opinion = { modifier = WJ_ty_reward_favor target = scope:wj_ty_proposer years = 5 } } }
   if = { limit = { var:wj_ty_tier = 3 }
    scope:wj_ty_proposer = { add_hook = { type = favor_hook target = scope:wj_ty_target } }
   }
  }
  if = { limit = { OR = {
   AND = { var:wj_ty_topic = 3 scope:wj_ty_appointment_success = yes }
   AND = { var:wj_ty_topic = 4 scope:wj_ty_removal_success = yes }
   OR = { var:wj_ty_topic = 1 var:wj_ty_topic = 2 }
  } }
   WJ_ty_commit_success = yes
  }
 }
}
WJ_ty_commit_success = {

  if = { limit = { var:wj_ty_office = 1 } scope:wj_ty_target = { set_variable = { name = wj_ty_chancellor_cooldown value = 1 months = 6 } } }
  else = {
   scope:wj_ty_target = { set_variable = { name = wj_ty_appoint_cooldown value = 1 months = 6 } }
   if = { limit = { var:wj_ty_appointment_mode = 2 exists = var:wj_ty_incumbent } var:wj_ty_incumbent = { set_variable = { name = wj_ty_appoint_cooldown value = 1 months = 6 } } }
  }
 }

  if = { limit = { var:wj_ty_dismiss_kind = 1 } scope:wj_ty_target = { set_variable = { name = wj_ty_dismiss_cooldown value = 1 months = 6 } } }
  else_if = { limit = { var:wj_ty_dismiss_kind = 2 } scope:wj_ty_target = { set_variable = { name = wj_ty_local_cooldown value = 1 months = 6 } } }
  else = { scope:wj_ty_target = { set_variable = { name = wj_ty_chancellor_cooldown value = 1 months = 6 } } }

 WJ_ty_close = yes
}
