### Han Dynasty defensive league effects ###

hd_mark_league_leader_effect = {
	$LEADER$ = {
		set_variable = { name = hd_league_leader value = $LEADER$ }
		if = {
			limit = { NOT = { has_character_modifier = hd_league_leader_modifier } }
			add_character_modifier = { modifier = hd_league_leader_modifier }
		}
	}
	if = {
		limit = { exists = confederation }
		confederation = {
			every_confederation_member = {
				set_variable = { name = hd_league_leader value = $LEADER$ }
				if = {
					limit = {
						this != $LEADER$
						has_character_modifier = hd_league_leader_modifier
					}
					remove_character_modifier = hd_league_leader_modifier
				}
			}
		}
	}
}

hd_update_league_votes_effect = {
	if = {
		limit = {
			exists = confederation
			confederation = {
				OR = {
					confederation_type = confederation_type:hd_defensive_league_type
					confederation_type = confederation_type:hd_anti_regent_league_type
				}
			}
		}
		confederation = {
			save_scope_as = hd_current_league
			if = {
				limit = { NOT = { has_variable = hd_league_creation_year } }
				set_variable = { name = hd_league_creation_year value = current_year }
			}
			# The event alliance is intentionally excluded from the election loop.
			if = {
				limit = { confederation_type = confederation_type:hd_defensive_league_type }
				every_confederation_member = {
					if = {
						limit = { is_ai = yes }
						set_variable = { name = hd_best_league_candidate value = this }
						set_variable = { name = hd_best_league_score value = 0 }
						scope:hd_current_league = {
							every_confederation_member = {
								limit = { this != prev }
								if = {
									limit = { opinion = { target = prev value >= 80 } }
									prev = { change_variable = { name = hd_best_league_score add = 80 } }
								}
								if = {
									limit = { opinion = { target = prev value >= 30 } }
									prev = { change_variable = { name = hd_best_league_score add = 30 } }
								}
							}
						}
						# AI votes for the highest visible candidate; the initial default
						# remains self if no relationship provides a stronger preference.
						set_variable = { name = hd_league_vote value = var:hd_best_league_candidate }
						remove_variable = hd_best_league_candidate
						remove_variable = hd_best_league_score
					}
					if = {
						limit = { NOT = { has_variable = hd_league_vote } }
						set_variable = { name = hd_league_vote value = this }
					}
				}
			}
			every_confederation_member = {
				if = {
					limit = {
						has_variable = hd_league_leader
						var:hd_league_leader = { is_alive = no }
					}
					remove_variable = hd_league_leader
				}
			}
		}
	}
}

hd_finalize_league_election_effect = {
	if = {
		limit = {
			exists = confederation
			confederation = { confederation_type = confederation_type:hd_defensive_league_type }
		}
		confederation = {
			save_scope_as = hd_election_league
			set_variable = { name = hd_total_voters value = 0 }
			every_confederation_member = {
				change_variable = { name = hd_total_voters add = 1 }
			}
			# One pass per candidate; a candidate must receive 60% of votes.
			every_confederation_member = {
				save_temporary_scope_as = hd_candidate
				set_variable = { name = hd_candidate_votes value = 0 }
				scope:hd_election_league = {
					every_confederation_member = {
						limit = {
							has_variable = hd_league_vote
							var:hd_league_vote = scope:hd_candidate
						}
						scope:hd_candidate = { change_variable = { name = hd_candidate_votes add = 1 } }
					}
				}
				if = {
					limit = {
						var:hd_candidate_votes >= 1
						var:hd_candidate_votes >= {
							value = scope:hd_election_league.var:hd_total_voters
							multiply = 0.6
						}
					}
					scope:hd_election_league = { set_variable = { name = hd_winning_candidate value = scope:hd_candidate } }
				}
				scope:hd_candidate = { remove_variable = hd_candidate_votes }
			}
			if = {
				limit = { has_variable = hd_winning_candidate }
				hd_mark_league_leader_effect = { LEADER = var:hd_winning_candidate }
				every_confederation_member = {
					if = {
						limit = { has_character_modifier = hd_league_disorganized_modifier }
						remove_character_modifier = hd_league_disorganized_modifier
					}
					remove_variable = hd_league_vote
				}
			}
			else = {
				every_confederation_member = {
					if = { limit = { NOT = { has_character_modifier = hd_league_disorganized_modifier } } add_character_modifier = { modifier = hd_league_disorganized_modifier } }
				}
			}
			remove_variable = hd_winning_candidate
			remove_variable = hd_total_voters
		}
	}
}

hd_disband_league_effect = {
	if = {
		limit = { exists = confederation }
		confederation = {
			save_scope_as = hd_disbanding_league
			every_confederation_member = {
				save_temporary_scope_as = hd_leaving_member
				remove_variable = hd_league_leader
				remove_variable = hd_league_vote
				if = { limit = { has_character_modifier = hd_league_leader_modifier } remove_character_modifier = hd_league_leader_modifier }
			if = { limit = { has_character_modifier = hd_league_disorganized_modifier } remove_character_modifier = hd_league_disorganized_modifier }
			confederation = { remove_confederation_member = scope:hd_leaving_member }
			}
		}
	}
}

hd_call_league_members_to_all_offensive_wars_effect = {
	if = {
		limit = { exists = confederation }
		confederation = {
			save_scope_as = hd_calling_league
			every_confederation_member = {
				limit = {
					this != root
					is_alive = yes
					is_imprisoned = no
				}
				save_temporary_scope_as = hd_league_member_to_call
				root = {
					every_character_war = {
						limit = { primary_attacker = root }
						save_temporary_scope_as = hd_league_war_to_join
						if = {
							limit = {
								scope:hd_league_member_to_call = {
									NOT = { is_attacker_in_war = scope:hd_league_war_to_join }
								}
							}
							add_attacker = scope:hd_league_member_to_call
						}
					}
				}
			}
		}
	}
}

# root: the anti-regent leader
# scope:hd_anti_regent_joiner: the character accepting or being forced into the league
hd_anti_regent_add_real_member_effect = {
	save_scope_as = hd_anti_regent_joining_leader
	scope:hd_anti_regent_joiner = {
		if = {
			limit = { is_alive = yes is_landed = yes is_independent_ruler = no }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = hd_anti_regent_joiner_independence
				add_claim_on_loss = no
			}
			becomes_independent = { change = scope:hd_anti_regent_joiner_independence }
			resolve_title_and_vassal_change = scope:hd_anti_regent_joiner_independence
		}
	}

	if = {
		limit = {
			has_variable = hd_anti_regent_confederation
			exists = confederation
			confederation = {
				this = root.var:hd_anti_regent_confederation
				confederation_type = confederation_type:hd_anti_regent_league_type
			}
		}
		confederation = {
			save_scope_as = hd_anti_regent_real_confederation
		}
	}
	else_if = {
		limit = {
			hd_valid_anti_regent_member_trigger = { CHARACTER = this }
			is_confederation_member = no
			exists = scope:hd_anti_regent_joiner
			scope:hd_anti_regent_joiner = {
				hd_valid_anti_regent_member_trigger = { CHARACTER = this }
				is_confederation_member = no
			}
		}
		remove_variable = hd_anti_regent_confederation
		create_confederation = {
			name = HD_ANTI_REGENT_LEAGUE_NAME
			type = confederation_type:hd_anti_regent_league_type
		}
		if = {
			limit = {
				exists = scope:new_confederation
				scope:new_confederation = {
					confederation_type = confederation_type:hd_anti_regent_league_type
				}
				is_member_of_confederation = scope:new_confederation
			}
			set_variable = {
				name = hd_anti_regent_confederation
				value = scope:new_confederation
			}
			scope:new_confederation = { save_scope_as = hd_anti_regent_real_confederation }
		}
	}

	if = {
		limit = {
			exists = scope:hd_anti_regent_real_confederation
			scope:hd_anti_regent_real_confederation = {
				confederation_type = confederation_type:hd_anti_regent_league_type
			}
			scope:hd_anti_regent_joining_leader = {
				is_member_of_confederation = scope:hd_anti_regent_real_confederation
			}
			scope:hd_anti_regent_joiner = {
				OR = {
					hd_valid_anti_regent_member_trigger = { CHARACTER = this }
					is_member_of_confederation = scope:hd_anti_regent_real_confederation
				}
			}
		}
		if = {
			limit = {
				scope:hd_anti_regent_joiner = {
					NOT = { is_member_of_confederation = scope:hd_anti_regent_real_confederation }
				}
			}
			scope:hd_anti_regent_real_confederation = {
				add_confederation_member = scope:hd_anti_regent_joiner
			}
		}
	}

	if = {
		limit = {
			exists = scope:hd_anti_regent_real_confederation
			scope:hd_anti_regent_joining_leader = {
				is_member_of_confederation = scope:hd_anti_regent_real_confederation
			}
			scope:hd_anti_regent_joiner = {
				is_member_of_confederation = scope:hd_anti_regent_real_confederation
			}
		}
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = hd_anti_regent_members
						target = scope:hd_anti_regent_joining_leader
					}
				}
			}
			add_to_variable_list = {
				name = hd_anti_regent_members
				target = scope:hd_anti_regent_joining_leader
			}
		}
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = hd_anti_regent_members
						target = scope:hd_anti_regent_joiner
					}
				}
			}
			add_to_variable_list = {
				name = hd_anti_regent_members
				target = scope:hd_anti_regent_joiner
			}
		}
	}
}

# At the three-month deadline, force the first valid historical co-founder.
hd_anti_regent_prepare_first_war_effect = {
	if = {
		limit = { NOT = { hd_anti_regent_real_confederation_ready_trigger = yes } }
		if = {
			limit = {
				character:cao_cao ?= { hd_anti_regent_join_candidate_trigger = yes }
			}
			character:cao_cao = { save_scope_as = hd_anti_regent_joiner }
			hd_anti_regent_add_real_member_effect = yes
		}
		if = {
			limit = {
				NOT = { hd_anti_regent_real_confederation_ready_trigger = yes }
				character:qiao_mao ?= { hd_anti_regent_join_candidate_trigger = yes }
			}
			character:qiao_mao = { save_scope_as = hd_anti_regent_joiner }
			hd_anti_regent_add_real_member_effect = yes
		}
		if = {
			limit = {
				NOT = { hd_anti_regent_real_confederation_ready_trigger = yes }
				character:han_fu ?= { hd_anti_regent_join_candidate_trigger = yes }
			}
			character:han_fu = { save_scope_as = hd_anti_regent_joiner }
			hd_anti_regent_add_real_member_effect = yes
		}
		if = {
			limit = {
				NOT = { hd_anti_regent_real_confederation_ready_trigger = yes }
				character:kong_rong ?= { hd_anti_regent_join_candidate_trigger = yes }
			}
			character:kong_rong = { save_scope_as = hd_anti_regent_joiner }
			hd_anti_regent_add_real_member_effect = yes
		}
	}

	if = {
		limit = { hd_anti_regent_real_confederation_ready_trigger = yes }
		trigger_event = { id = hd_anti_regent.0004 days = 1 }
	}
	else = { hd_anti_regent_failure_effect = yes }
}

hd_anti_regent_call_members_to_all_offensive_wars_effect = {
	if = {
		limit = {
			hd_anti_regent_real_confederation_ready_trigger = yes
		}
		confederation = {
			every_confederation_member = {
				limit = {
					this != root
					is_alive = yes
					is_imprisoned = no
				}
				save_temporary_scope_as = hd_anti_regent_member_to_call
				root = {
					every_character_war = {
						limit = { primary_attacker = root }
						save_temporary_scope_as = hd_anti_regent_war_to_join
						if = {
							limit = {
								scope:hd_anti_regent_member_to_call = {
									NOT = { is_attacker_in_war = scope:hd_anti_regent_war_to_join }
								}
							}
							add_attacker = scope:hd_anti_regent_member_to_call
						}
					}
				}
			}
		}
	}
}

# Cao Cao must receive the same historical invitation whether he reaches
# Chenliu before or after Yuan Shao opens the roster.
hd_anti_regent_add_cao_effect = {
	if = {
		limit = { this = character:cao_cao is_alive = yes }
		if = {
			limit = {
				exists = global_var:hd_anti_regent_league_leader
				global_var:hd_anti_regent_league_leader = {
					is_alive = yes
					has_character_flag = hd_anti_regent_league_leader
					has_variable = hd_anti_regent_target
				}
			}
			global_var:hd_anti_regent_league_leader = {
				save_scope_as = hd_inviting_leader
				var:hd_anti_regent_target = { save_scope_as = hd_anti_regent_target_scope }
			}
			hd_anti_regent_dispatch_invitation_effect = yes
		}
		else = { add_character_flag = hd_anti_regent_cao_pending }
	}
}

hd_anti_regent_dispatch_invitation_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			exists = scope:hd_inviting_leader
			exists = scope:hd_anti_regent_target_scope
			NOT = { this = scope:hd_inviting_leader }
			NOT = { this = scope:hd_anti_regent_target_scope }
			NOT = { has_character_flag = hd_anti_regent_invitation_pending }
			NOT = { has_character_flag = hd_anti_regent_invitation_resolved }
		}
		scope:hd_inviting_leader = { change_variable = { name = hd_anti_regent_pending_invites add = 1 } }
		set_variable = { name = hd_anti_regent_inviter value = scope:hd_inviting_leader }
		add_character_flag = hd_anti_regent_invitation_pending
		remove_character_flag = hd_anti_regent_cao_pending
		trigger_event = { id = hd_anti_regent.0003 days = 2 }
	}
}

hd_anti_regent_invitation_resolved_effect = {
	if = {
		limit = {
			has_character_flag = hd_anti_regent_invitation_pending
			NOT = { has_character_flag = hd_anti_regent_invitation_resolved }
		}
		remove_character_flag = hd_anti_regent_invitation_pending
		add_character_flag = hd_anti_regent_invitation_resolved
		if = {
			limit = { has_variable = hd_anti_regent_inviter }
			var:hd_anti_regent_inviter = { save_scope_as = hd_invitation_leader }
			remove_variable = hd_anti_regent_inviter
			if = {
				limit = {
					exists = scope:hd_invitation_leader
					scope:hd_invitation_leader = {
						is_alive = yes
						has_character_flag = hd_anti_regent_league_leader
						has_variable = hd_anti_regent_pending_invites
					}
				}
				scope:hd_invitation_leader = {
					change_variable = { name = hd_anti_regent_pending_invites add = -1 }
					if = {
						limit = { var:hd_anti_regent_pending_invites <= 0 }
						remove_variable = hd_anti_regent_pending_invites
					}
				}
			}
		}
	}
}
