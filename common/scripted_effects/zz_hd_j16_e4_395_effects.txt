	if = {
		limit = {
			exists = $WAR$
			exists = scope:$KEY$
			NOT = {
				$WAR$ = {
					casus_belli = {
						any_target_title = {
							OR = {
								this = scope:$KEY$
								any_in_de_jure_hierarchy = { this = scope:$KEY$ }
							}
						}
					}
				}
			}
		}
		if = {
			limit = { exists = scope:hd_j16_e4b_wk_orig }
			clear_saved_scope = hd_j16_e4b_wk_orig
		}
		if = {
			limit = { exists = scope:hd_j16_e4b_wk_cap }
			clear_saved_scope = hd_j16_e4b_wk_cap
		}
		scope:$KEY$ = { save_scope_as = hd_j16_e4b_wk_orig }
		clear_saved_scope = $KEY$
		$DEF$ = {
			capital_county ?= { save_scope_as = hd_j16_e4b_wk_cap }
		}
		if = {
			limit = {
				exists = scope:hd_j16_e4b_wk_cap
				scope:hd_j16_e4b_wk_cap.holder ?= { target_is_same_character_or_above = $DEF$ }
				$WAR$ = {
					casus_belli = {
						any_target_title = {
							OR = {
								this = scope:hd_j16_e4b_wk_cap
								any_in_de_jure_hierarchy = { this = scope:hd_j16_e4b_wk_cap }
							}
						}
					}
				}
			}
			scope:hd_j16_e4b_wk_cap = { save_scope_as = $KEY$ }
		}
		else = {
			$WAR$ = {
				casus_belli = {
					ordered_target_title = {
						limit = {
							OR = {
								AND = {
									tier = tier_county
									holder ?= { target_is_same_character_or_above = $DEF$ }
								}
								any_in_de_jure_hierarchy = {
									tier = tier_county
									holder ?= { target_is_same_character_or_above = $DEF$ }
								}
							}
						}
						order_by = tier
						if = {
							limit = { tier = tier_county }
							save_scope_as = $KEY$
						}
						else = {
							random_in_de_jure_hierarchy = {
								limit = {
									tier = tier_county
									holder ?= { target_is_same_character_or_above = $DEF$ }
								}
								save_scope_as = $KEY$
							}
						}
					}
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:$KEY$ } }
			scope:hd_j16_e4b_wk_orig = { save_scope_as = $KEY$ }
			debug_log = "HD_J16 e4b WAR_KEY keep_original_no_target_county"
		}
		else = {
			debug_log = "HD_J16 e4b WAR_KEY moved_into_war_target"
		}
	}
}

						has_global_variable = hd_j16_yanwei_peace

					}

						limit = { exists = scope:hd_j16_e4b_pw }

					}

						if = {
							limit = { is_alive = yes }

					}

							exists = scope:hd_j16_e4b_yan

					}
				}
			}
			set_global_variable = hd_j16_xiyan_won
			debug_log = "HD_J16 e4_m55 FIRE result_won"

		else = {
			set_global_variable = hd_j16_xiyan_lost
			debug_log = "HD_J16 e4_m55 FIRE result_lost"
		}
	}
