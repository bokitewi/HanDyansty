# Applicant scope throughout. Existing peerage picker performs the mandatory final placement.
WJ_ab_begin = {
 # Shared entry: another pending formal handover cannot be overwritten.
 if = { limit = {
  is_alive = yes NOT = { has_variable = wj_ab_emperor }
  OR = {
   AND = { has_variable = wj_ab_history exists = var:wj_ab_history_throne.holder this != var:wj_ab_history_throne.holder
    var:wj_ab_history_throne.holder = { is_alive = yes NOT = { has_variable = wj_ab_successor } }
   }
   AND = { NOT = { has_variable = wj_ab_history } this != top_liege
    top_liege = { is_alive = yes NOT = { has_variable = wj_ab_successor } }
   }
  }
 }

 if = { limit = { has_variable = wj_ab_history }
  set_variable = { name = wj_ab_emperor value = var:wj_ab_history_throne.holder }
  set_variable = { name = wj_ab_throne value = var:wj_ab_history_throne }
 }
 else = {
  set_variable = { name = wj_ab_emperor value = top_liege }
  if = { limit = { top_liege = { has_title = title:h_china } }
   set_variable = { name = wj_ab_throne value = title:h_china }
  }
  else = { set_variable = { name = wj_ab_throne value = top_liege.primary_title } }
 }
 set_variable = { name = wj_ab_round value = 1 }
 set_variable = { name = wj_ab_declines value = 0 }
 set_variable = { name = wj_ab_penalty value = 0 }
 if = { limit = { NOT = { has_variable = wj_ab_ticket } } set_variable = { name = wj_ab_ticket value = 0 } }
 change_variable = { name = wj_ab_ticket add = 1 }
 save_scope_value_as = { name = wj_ab_ticket value = var:wj_ab_ticket }
 save_scope_as = wj_ab_successor
 clear_variable_list = wj_ab_advisors
 var:wj_mufu_slot_1 ?= { if = { limit = { is_alive = yes WJ_mufu_member_valid = yes var:wj_mufu_owner = root } root = { add_to_variable_list = { name = wj_ab_advisors target = prev } } } }
 var:wj_mufu_slot_2 ?= { if = { limit = { is_alive = yes WJ_mufu_member_valid = yes var:wj_mufu_owner = root } root = { add_to_variable_list = { name = wj_ab_advisors target = prev } } } }
 var:wj_mufu_slot_3 ?= { if = { limit = { is_alive = yes WJ_mufu_member_valid = yes var:wj_mufu_owner = root } root = { add_to_variable_list = { name = wj_ab_advisors target = prev } } } }
 var:wj_mufu_slot_4 ?= { if = { limit = { is_alive = yes WJ_mufu_member_valid = yes var:wj_mufu_owner = root } root = { add_to_variable_list = { name = wj_ab_advisors target = prev } } } }
 var:wj_mufu_slot_5 ?= { if = { limit = { is_alive = yes WJ_mufu_member_valid = yes var:wj_mufu_owner = root } root = { add_to_variable_list = { name = wj_ab_advisors target = prev } } } }
 random_in_list = { variable = wj_ab_advisors save_scope_as = wj_ab_advisor }
 clear_variable_list = wj_ab_advisors
 if = { limit = { NOT = { exists = scope:wj_ab_advisor } } var:wj_ab_emperor = { random_councillor = { limit = { is_alive = yes NOT = { this = root } } save_scope_as = wj_ab_advisor } } }
 if = { limit = { exists = scope:wj_ab_advisor } set_variable = { name = wj_ab_advisor value = scope:wj_ab_advisor } }
 var:wj_ab_emperor = {
  set_variable = { name = wj_ab_successor value = scope:wj_ab_successor }
  add_character_modifier = { modifier = WJ_ab_protection }
 }
 trigger_event = WJ_ab.1
 }
 else_if = { limit = { NOT = { has_variable = wj_ab_emperor } }
  remove_variable = wj_ab_history remove_variable = wj_ab_history_liege remove_variable = wj_ab_history_throne
 }
}
WJ_ab_decline = {
 change_variable = { name = wj_ab_declines add = 1 }
 if = { limit = { var:wj_ab_round < 3 }
  change_variable = { name = wj_ab_round add = 1 }
  trigger_event = { id = WJ_ab.1 days = 10 }
 }
 else = { trigger_event = WJ_ab.3 }
}
WJ_ab_accept = {
 if = { limit = { var:wj_ab_round < 3 } trigger_event = WJ_ab.2 }
 else = { WJ_ab_transfer = yes }
}
WJ_ab_cleanup = {
 var:wj_ab_emperor ?= { remove_character_modifier = WJ_ab_protection remove_variable = wj_ab_successor }
 remove_variable = wj_ab_emperor remove_variable = wj_ab_throne remove_variable = wj_ab_round remove_variable = wj_ab_declines
 remove_variable = wj_ab_penalty remove_variable = wj_ab_transferred remove_variable = wj_ab_placement
 remove_variable = wj_ab_country_stem clear_variable_list = wj_ab_titles
 remove_variable = wj_ab_advisor clear_variable_list = wj_ab_advisors
 remove_variable = wj_ab_history remove_variable = wj_ab_history_liege remove_variable = wj_ab_history_throne
}
WJ_ab_abort = {
 if = { limit = { exists = scope:wj_ab_ticket var:wj_ab_ticket = scope:wj_ab_ticket }
  if = { limit = { has_variable = wj_ab_placement } WJ_peer_end_workflow = yes }
  WJ_ab_cleanup = yes
 }
}
WJ_ab_transfer = {
 if = { limit = { has_variable = wj_ab_history } WJ_ab_history_finish = yes }
 else_if = { limit = { WJ_ab_transfer_valid = yes }
  save_scope_as = wj_ab_successor
  var:wj_ab_emperor = { save_scope_as = wj_ab_old_emperor capital_county = { save_scope_as = wj_ab_imperial_capital } }
  var:wj_ab_throne = { save_scope_as = wj_ab_throne }
  # Preserve the existing fief's name before native transfer/death callbacks reconcile title mirrors.
  if = { limit = { any_held_title = { var:wj_rf_rank = 2 } }
   every_held_title = { limit = { var:wj_rf_rank = 2 } save_scope_as = wj_ab_old_kingdom }
  }
  else = { every_held_title = { limit = { var:wj_rf_rank = 1 } save_scope_as = wj_ab_old_kingdom } }
  set_variable = { name = wj_ab_country_stem value = scope:wj_ab_old_kingdom }
  clear_variable_list = wj_ab_titles
  scope:wj_ab_old_emperor = {
   every_held_title = {
    # main makes the hegemony landless during division; the throne itself must still transfer.
    # 218 scenario: the emptied e_han (no de jure county) stays with the abdicated emperor.
    limit = { OR = { this = scope:wj_ab_throne AND = { is_noble_family_title = no OR = { is_landless_type_title = no has_variable = wj_rf_rank } tgp_is_minister_title = no NOT = { has_variable = SanGong_DongGong_title } NOT = { AND = { this = title:e_han NOT = { any_de_jure_county = { count >= 1 } } } } } } }
    scope:wj_ab_successor = { add_to_variable_list = { name = wj_ab_titles target = prev } }
   }
  }
  # Reuse main's release of the outgoing court's controller, so next-day repair cannot reinstall it.
  if = { limit = { scope:wj_ab_throne = title:h_china } hd_court_release_effect = yes }
  create_title_and_vassal_change = { type = granted save_scope_as = wj_ab_change add_claim_on_loss = no }
  every_in_list = { variable = wj_ab_titles change_title_holder_include_vassals = { holder = scope:wj_ab_successor change = scope:wj_ab_change } }
  resolve_title_and_vassal_change = scope:wj_ab_change
  clear_variable_list = wj_ab_titles
  if = { limit = { scope:wj_ab_throne.holder = this }
   WJ_cx_abolish = yes
   set_variable = wj_ab_transferred
   set_primary_title_to = scope:wj_ab_throne
   set_realm_capital = scope:wj_ab_imperial_capital
   scope:wj_ab_throne = { set_title_name_dynamic = WJ_ab_new_country_name set_definitive_form = no }
   dynasty ?= { remove_variable = hd_fixed_realm_name set_variable = { name = hd_fixed_realm_title value = root.primary_title } }
   # Restore constituent commanderies before destroying the former real-fief kingdom/empire.
   every_held_title = { limit = { var:wj_rf_rank = 1 } save_scope_as = wj_ab_dismantled WJ_rf_restore_destroyed_fief = yes root = { destroy_title = scope:wj_ab_dismantled } }
   # 218 scenario: e_hd218_wei, e_bashu and e_jiangdong are de jure empires carrying rank 2, so they are never destroyed here.
   if = { limit = { scope:wj_ab_old_kingdom = { var:wj_rf_rank = 2 NOT = { this = title:e_hd218_wei } NOT = { this = title:e_bashu } NOT = { this = title:e_jiangdong } } } destroy_title = scope:wj_ab_old_kingdom }
   # 218 Wei is destroyed by its own effect instead (also clears Yecheng), so the new emperor cannot grant it out (boki 10-25).
   if = { limit = { scope:wj_ab_old_kingdom ?= title:e_hd218_wei } hd_sg218_destroy_wei_effect = yes }
   if = { limit = { exists = global_var:hd218_wei_wang_title } debug_log = "HD218 WJ_AB_TRANSFER" }
   set_variable = wj_cr_empire_settled
   add_realm_law = crowned_emperor set_variable = crowned_emperor_var remove_variable = crowned_king_var
   WJ_ab_reward = yes
   WJ_mr_end_request = yes WJ_mr_clear_rank = yes remove_variable = WJ_mr_award_realm
   WJ_title_style_reconcile_realm = yes
   # The native title transaction transfers vassals, not the old emperor's courtiers.
   create_title_and_vassal_change = { type = swear_fealty save_scope_as = wj_ab_retirement_change add_claim_on_loss = no }
   scope:wj_ab_old_emperor = { change_liege = { liege = scope:wj_ab_successor change = scope:wj_ab_retirement_change } }
   resolve_title_and_vassal_change = scope:wj_ab_retirement_change
   WJ_peer_end_workflow = yes
   set_variable = { name = wj_peer_target value = scope:wj_ab_old_emperor }
   set_variable = { name = wj_peer_crown value = this }
   if = { limit = { NOT = { has_variable = wj_peer_ticket } } set_variable = { name = wj_peer_ticket value = 0 } }
   change_variable = { name = wj_peer_ticket add = 1 }
   save_scope_value_as = { name = wj_peer_ticket value = var:wj_peer_ticket }
   set_variable = { name = wj_peer_flow_version value = 2 }
   set_variable = wj_ab_placement
   trigger_event = WJ_ab.4
  }
  else = { WJ_ab_cleanup = yes }
 }
 else = { if = { limit = { NOT = { has_variable = wj_ab_transferred } } WJ_ab_abort = yes } }
}

# One reward shared by the general decision, petitions, aspirations and historical abdications.
WJ_ab_reward = {
 save_scope_value_as = { name = wj_ab_reward_total value = { value = 1000 add = { value = var:wj_ab_declines multiply = 1000 } subtract = var:wj_ab_penalty } }
 add_legitimacy = scope:wj_ab_reward_total
 send_interface_toast = { type = event_generic_neutral title = WJ_ab_settled left_icon = root custom_tooltip = WJ_ab_settled_tt }
}
WJ_ab_history_begin = {
 if = { limit = {
  is_alive = yes exists = $THRONE$.holder this != $THRONE$.holder
  NOT = { has_variable = wj_ab_emperor } NOT = { has_variable = wj_ab_history }
  $THRONE$.holder = { is_alive = yes NOT = { has_variable = wj_ab_successor } }
 }
  set_variable = { name = wj_ab_history_throne value = $THRONE$ }
  set_variable = { name = wj_ab_history value = flag:$KIND$ }
  set_variable = { name = wj_ab_history_liege value = top_liege }
  # A fresh event makes the receiver ROOT even when requested from the old emperor's event.
  trigger_event = WJ_ab.5
 }
}
WJ_ab_history_finish = {
 if = { limit = {
  WJ_ab_flow_valid = yes NOT = { has_variable = wj_ab_transferred }
  var:wj_ab_throne = var:wj_ab_history_throne var:wj_ab_throne.holder = var:wj_ab_emperor
  top_liege = var:wj_ab_history_liege
 }
  # Seal this ticket before native title callbacks can fire.
  set_variable = wj_ab_transferred
  switch = {
   trigger = var:wj_ab_history
   flag:tang = { hd_617_tang_effect_after_ritual = yes }
   flag:zheng = { hd_617_zheng_effect_after_ritual = yes }
   flag:caopi = { hd_sg218_shouchan_effect_after_ritual = yes }
   flag:song_nbc = { hd_nbc_song_effect_after_ritual = yes }
   flag:huanxuan = { hd_j16_cuanchu_effect_after_ritual = yes }
   flag:song_j16 = { hd_j16_song_exec_effect_after_ritual = yes }
   flag:jin = { hd263_shandai_effect_after_ritual = yes }
   flag:yuanxiu = { hd_531_abdicate_effect_after_ritual = yes }
   flag:liubei_asp = { hd_asp_lb_shouchan_effect_after_ritual = yes }
   flag:asp_hd_asp_title_chenliu = { hd_asp_shanrang_effect_after_ritual = { TITLE_NAME = hd_asp_title_chenliu } }
   flag:asp_hd_asp_title_haixi = { hd_asp_shanrang_effect_after_ritual = { TITLE_NAME = hd_asp_title_haixi } }
   flag:asp_hd_asp_title_lingling = { hd_asp_shanrang_effect_after_ritual = { TITLE_NAME = hd_asp_title_lingling } }
  }
  if = { limit = { var:wj_ab_throne.holder = this }
   WJ_cx_abolish = yes
   set_primary_title_to = var:wj_ab_throne
   set_variable = wj_cr_empire_settled
   add_realm_law = crowned_emperor set_variable = crowned_emperor_var remove_variable = crowned_king_var
   WJ_ab_reward = yes
   if = { limit = { OR = {
    var:wj_ab_history = flag:liubei_asp
    var:wj_ab_history = flag:asp_hd_asp_title_chenliu
    var:wj_ab_history = flag:asp_hd_asp_title_haixi
    var:wj_ab_history = flag:asp_hd_asp_title_lingling
   } } hd_asp_complete_effect = yes hd_asp_notify_effect = yes }
  }
  WJ_ab_cleanup = yes
 }
 else = { WJ_ab_abort = yes }
}
