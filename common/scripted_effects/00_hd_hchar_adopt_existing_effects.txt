# UUII 延迟出场整合：主包 history 已有的名人直接登记为“已出场”，
# 延迟出场不再重复生成；wj_hchar_<ID> 指向现有人物，供改用全局变量的脚本使用。
# wj_hchar_adopted_<ID> 标记沿用人物，接回父亲逻辑据此跳过，不改主包父子关系。
zz_hd_hchar_adopt_one_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = wj_hchar_seen_$ID$ }
			exists = character:$CHAR$
		}
		set_global_variable = { name = wj_hchar_$ID$ value = character:$CHAR$ }
		set_global_variable = { name = wj_hchar_seen_$ID$ value = yes }
		set_global_variable = { name = wj_hchar_adopted_$ID$ value = yes }
	}
}
