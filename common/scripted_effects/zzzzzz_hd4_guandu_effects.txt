hd4_build_north_cache_effect = {
	title:h_china = {
		if = {
			limit = {
				NOT = {
					has_variable = hd4_north_cached
				}
			}
			clear_variable_list = hd4_north_counties
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_county
					title_province = {
						geographical_region = china_north_world
					}
				}
				save_scope_as = hd4_north_county
				title:h_china = {
					add_to_variable_list = {
						name = hd4_north_counties
						target = scope:hd4_north_county
					}
				}
			}
			if = {
				limit = {
					any_in_list = {
						variable = hd4_north_counties
						always = yes
					}
				}
				set_variable = hd4_north_cached
			}
		}
	}
}
hd4_guandu_tick_effect = {
	if = {
		limit = {
			exists = global_var:hd_anti_regent_chain_resolved
			NOT = {
				title:h_china = {
					has_variable = hd4_legacy_guandu
				}
			}
			NOT = {
				title:h_china = {
					has_variable = hd4_guandu_done
				}
			}
		}
		hd4_build_north_cache_effect = yes
		if = {
			limit = {
				hd4_guandu_actors_trigger = yes
			}
			clear_saved_scope = hd4_live_guandu
			character:cao_cao = {
				random_character_war = {
					limit = {
						using_cb = hd4_guandu_cb
						primary_attacker = character:cao_cao
						primary_defender = character:yuan_shao
					}
					save_scope_as = hd4_live_guandu
				}
			}
			if = {
				limit = {
					exists = scope:hd4_live_guandu
				}
				set_global_variable = hd_guandu_world_active
			}
			else = {
				remove_global_variable = hd_guandu_world_active
				remove_global_variable = hd_guandu_world_resolved
				if = {
					limit = {
						hd4_north_only_two_trigger = yes
					}
					title:h_china = {
						if = {
							limit = {
								NOT = {
									has_variable = hd4_northern_two_announced
								}
							}
							set_variable = hd4_northern_two_announced
							character:cao_cao = {
								trigger_event = hd4_guandu.0001
							}
							character:yuan_shao = {
								trigger_event = hd4_guandu.0001
							}
						}
					}
				}
				if = {
					limit = {
						title:h_china = {
							has_variable = hd4_northern_two_announced
						}
					}
					character:cao_cao = {
						if = {
							limit = {
								has_character_flag = hd4_guandu_authorized
							}
							hd4_guandu_prepare_effect = yes
						}
						else_if = {
							limit = {
								NOT = {
									has_character_flag = hd4_guandu_invite_lease
								}
							}
							add_character_flag = {
								flag = hd4_guandu_invite_lease
								days = 180
							}
							trigger_event = hd4_guandu.0002
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						exists = character:cao_cao
						character:cao_cao = {
							is_alive = no
						}
					}
					AND = {
						exists = character:yuan_shao
						character:yuan_shao = {
							is_alive = no
						}
					}
				}
			}
			title:h_china = {
				set_variable = hd4_guandu_done
			}
			set_global_variable = hd_guandu_world_resolved
			remove_global_variable = hd_guandu_world_active
			character:cao_cao ?= {
				remove_character_flag = hd4_guandu_reservation
			}
			character:yuan_shao ?= {
				remove_character_flag = hd4_guandu_reservation
			}
		}
	}
}
hd4_guandu_prepare_effect = {
	if = {
		limit = {
			hd4_guandu_can_attempt_trigger = yes
		}
		clear_variable_list = hd4_guandu_targets
		every_realm_county = {
			every_neighboring_county = {
				limit = {
					AND = {
						exists = holder
						holder = {
							OR = {
								this = character:yuan_shao
								is_vassal_or_below_of = character:yuan_shao
							}
						}
					}
				}
				de_jure_liege = {
					if = {
						limit = {
							tier = tier_duchy
						}
						save_scope_as = hd4_border_duchy
						character:cao_cao = {
							if = {
								limit = {
									NOT = {
										is_target_in_variable_list = {
											name = hd4_guandu_targets
											target = scope:hd4_border_duchy
										}
									}
								}
								add_to_variable_list = {
									name = hd4_guandu_targets
									target = scope:hd4_border_duchy
								}
							}
						}
					}
				}
			}
		}
		set_variable = {
			name = hd4_guandu_target_count
			value = 0
		}
		every_in_list = {
			variable = hd4_guandu_targets
			character:cao_cao = {
				change_variable = {
					name = hd4_guandu_target_count
					add = 1
				}
			}
		}
		if = {
			limit = {
				var:hd4_guandu_target_count > 0
				var:hd4_guandu_target_count <= 256
			}
			add_character_flag = {
				flag = hd4_guandu_reservation
				days = 61
			}
			character:yuan_shao = {
				add_character_flag = {
					flag = hd4_guandu_reservation
					days = 61
				}
			}
			hd4_guandu_declare_exact_targets_effect = yes
		}
	}
}
hd4_guandu_victory_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = hd4_guandu_victory_applied
			}
		}
		scope:defender = {
			every_held_title = {
				limit = {
					tier >= tier_county
					is_landless_type_title = no
					is_noble_family_title = no
				}
				save_scope_as = hd4_claimed_title
				character:cao_cao = {
					if = {
						limit = {
							NOT = {
								has_title = scope:hd4_claimed_title
							}
						}
						add_pressed_claim = scope:hd4_claimed_title
					}
				}
			}
		}
		hd4_guandu_snapshot_land_effect = yes
		hd4_target_conquest_effect = yes
		title:h_china = {
			hd4_guandu_reconcile_land_effect = yes
		}
		set_variable = hd4_guandu_victory_applied
		character:yuan_shao = {
			if = {
				limit = {
					is_alive = yes
					NOT = {
						has_character_flag = hd4_yuan_grief_applied
					}
				}
				add_character_flag = hd4_yuan_partition_armed
				add_character_flag = hd4_yuan_grief_applied
				if = {
					limit = {
						has_trait = wounded_1
					}
					remove_trait = wounded_1
				}
				if = {
					limit = {
						NOT = {
							has_trait = wounded_2
						}
						NOT = {
							has_trait = wounded_3
						}
					}
					add_trait = wounded_2
				}
				add_character_modifier = {
					modifier = hd4_yuan_grief
				}
				trigger_event = hd4_yuan.0002
			}
		}
		hd4_guandu_finish_effect = yes
		character:cao_cao ?= {
			if = {
				limit = {
					is_alive = yes
				}
				trigger_event = hd4_guandu.0003
			}
		}
	}
}

hd4_guandu_finish_effect = {

	}
	set_global_variable = hd_guandu_world_resolved
	remove_global_variable = hd_guandu_world_active
	remove_global_variable = hd_guandu_world_open
	hd_clear_guandu_wuchao_effect = yes

	}

		remove_character_flag = hd4_guandu_reservation
	}
}

hd4_target_conquest_effect = {
	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = hd4_target_conquest_change
		add_claim_on_loss = no
	}
	every_in_list = {
		list = target_titles
		conquest_cb_title_transfer = {
			RELIGIOUS_WAR = no
		}
		if = {
			limit = {
				OR = {
					NOT = {
						exists = holder
					}
					AND = {
						exists = holder
						holder = {
							OR = {
								this = scope:defender
								is_vassal_or_below_of = scope:defender
							}
						}
					}
				}
				NOT = {
					is_in_list = titles_taken
				}
			}
			add_to_list = titles_taken
		}
	}
	every_in_list = {
		list = vassals_taken
		change_liege = {
			liege = scope:attacker
			change = scope:hd4_target_conquest_change
		}
	}
	every_in_list = {
		list = titles_taken
		change_title_holder = {
			holder = scope:attacker
			change = scope:hd4_target_conquest_change
			take_baronies = yes
		}
	}
	resolve_title_and_vassal_change = scope:hd4_target_conquest_change
}
hd4_guandu_snapshot_land_effect = {
	title:h_china = {
		clear_variable_list = hd4_guandu_land_receipts
	}
	scope:defender = {
		add_character_flag = hd4_guandu_enemy_member
		primary_heir ?= {
			add_character_flag = hd4_guandu_enemy_member
		}
		every_vassal_or_below = {
			add_character_flag = hd4_guandu_enemy_member
			primary_heir ?= {
				add_character_flag = hd4_guandu_enemy_member
			}
		}
	}
	every_in_list = {
		list = target_titles
		every_de_jure_county = {
			limit = {
				AND = {
					exists = holder
					holder = {
						OR = {
							this = scope:defender
							is_vassal_or_below_of = scope:defender
						}
					}
				}
			}
			hd4_guandu_register_land_ticket_effect = yes
		}
		if = {
			limit = {
				OR = {
					NOT = {
						exists = holder
					}
					AND = {
						exists = holder
						holder = {
							OR = {
								this = scope:defender
								is_vassal_or_below_of = scope:defender
							}
						}
					}
				}
			}
			hd4_guandu_register_land_ticket_effect = yes
		}
	}
	title:h_china = {
		set_variable = hd4_guandu_land_pending
	}
}
hd4_guandu_register_land_ticket_effect = {
	save_scope_as = hd4_guandu_land_title
	set_variable = hd4_guandu_land_ticket
	if = {
		limit = {
			exists = holder
		}
		set_variable = {
			name = hd4_guandu_original_holder
			value = holder
		}
	}
	title:h_china = {
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = hd4_guandu_land_receipts
						target = scope:hd4_guandu_land_title
					}
				}
			}
			add_to_variable_list = {
				name = hd4_guandu_land_receipts
				target = scope:hd4_guandu_land_title
			}
		}
	}
}
hd4_guandu_reconcile_land_effect = {
	if = {
		limit = {
			has_variable = hd4_guandu_land_pending
		}
		if = {
			limit = {
				AND = {
					exists = character:cao_cao
					character:cao_cao = {
						is_alive = yes
						is_landed = yes
					}
				}
			}
			every_in_list = {
				variable = hd4_guandu_land_receipts
				limit = {
					has_variable = hd4_guandu_land_ticket
				}
				if = {
					limit = {
						AND = {
							exists = holder
							holder = {
								OR = {
									this = character:cao_cao
									is_vassal_or_below_of = character:cao_cao
								}
							}
						}
					}
					remove_variable = hd4_guandu_land_ticket
				}
				else_if = {
					limit = {
						OR = {
							NOT = {
								exists = holder
							}
							AND = {
								exists = holder.top_liege
								holder.top_liege = {
									has_character_flag = hd4_guandu_enemy_member
								}
							}
						}
					}
					create_title_and_vassal_change = {
						type = conquest
						save_scope_as = hd4_guandu_receipt_change
						add_claim_on_loss = no
					}
					change_title_holder = {
						holder = character:cao_cao
						change = scope:hd4_guandu_receipt_change
						take_baronies = yes
					}
					resolve_title_and_vassal_change = scope:hd4_guandu_receipt_change
					if = {
						limit = {
							AND = {
								exists = holder
								holder = {
									OR = {
										this = character:cao_cao
										is_vassal_or_below_of = character:cao_cao
									}
								}
							}
						}
						remove_variable = hd4_guandu_land_ticket
					}
				}
				else = {
					remove_variable = hd4_guandu_land_ticket
					set_variable = hd4_guandu_land_external
				}
			}
			if = {
				limit = {
					NOT = {
						any_in_list = {
							variable = hd4_guandu_land_receipts
							has_variable = hd4_guandu_land_ticket
						}
					}
				}
				remove_variable = hd4_guandu_land_pending
				set_variable = hd4_guandu_land_confirmed
			}
		}
		else_if = {
			limit = {
				AND = {
					exists = character:cao_cao
					character:cao_cao = {
						is_alive = no
					}
				}
			}
			every_in_list = {
				variable = hd4_guandu_land_receipts
				remove_variable = hd4_guandu_land_ticket
			}
			remove_variable = hd4_guandu_land_pending
			set_variable = hd4_guandu_land_owner_died
		}
	}
}
