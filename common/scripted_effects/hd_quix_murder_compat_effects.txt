# CK3 1.19 has already resolved the murder scheme before the result pool is
# entered. This effect preserves the old preparation scene without rerolling.
hd_quix_execute_murder_setup_effect = {
	if = {
		limit = { scope:scheme.var:hd_quix_murder_result_success = yes }
		trigger_event = { on_action = hd_quix_murder_success_resolution }
	}
	else = {
		trigger_event = { on_action = hd_quix_murder_failure_resolution }
	}
}

# The old "do not execute" option becomes the native 1.19 scheme restart.
# 2026-10-09 C7: the result was already rolled before this setup scene. If it
# was a failure, backing out must not dodge the failure consequences. Mirror
# vanilla murder_outcome_reworked.0003: its `after` always fires 0004 (which
# applies murder_failure_effect), and a restart is only offered when the
# attempt was not discovered. Every branch shows only one neutral narrative
# line, so option b neither reveals the pre-rolled result nor promises a
# restart that a discovered failure would not get.
hd_quix_defer_murder_setup_effect = {
	custom_tooltip = hd_quix_defer_murder_setup_tt
	hidden_effect = {
		if = {
			limit = { scope:scheme.var:hd_quix_murder_result_success = yes }
			restart_murder_scheme_effect = yes
		}
		else_if = {
			limit = { exists = scope:scheme_discovered }
			scope:scheme = { end_scheme = yes }
			scope:target = { trigger_event = murder_outcome_reworked.0004 }
		}
		else = {
			restart_murder_scheme_effect = yes
			scope:target = { trigger_event = murder_outcome_reworked.0004 }
		}
	}
}
