WJ_miaoshi_initialize_effect = {
 if = { limit = { NOT = { has_variable = Dengji_sub_realm_var } }

        if = {
            limit = {
                NOT = { has_variable = Dengji_years_var }
            }
            set_variable = {#储存登基年限
                name = Dengji_years_var
                value = 0
            }
        }
        if = {
            limit = {
                NOT = { has_variable = Dengji_sub_realm_var }
            }
            set_variable = {#储存登基时领土规模
                name = Dengji_sub_realm_var
                value = sub_realm_size
            }
        }
        if = {
            limit = {
                NOT = { has_variable = SanGong_XiongZhu_var }
            }
            set_variable = {#储存达成雄主所需要的领土规模
                name = SanGong_XiongZhu_var
                value = SanGong_XiongZhu_value
            }
        }
        if = {
            limit = {
                NOT = { has_variable = SanGong_HunJun_var }
            }
            set_variable = {#储存达成昏君所需要的领土规模
                name = SanGong_HunJun_var
                value = SanGong_HunJun_value
            }
        }
        if = {#登基时处于不稳定阶段,储存中兴之主变量
            limit = {
                NOT = { has_variable = SanGong_Zhongxing_var }
                any_character_situation = { this = situation:dynastic_cycle
                    OR = { 
                        OR = { situation_current_phase = situation_dynastic_cycle_phase_chaos situation_current_phase = situation_dynastic_cycle_phase_hd_standoff }               # 混乱阶段
                        situation_current_phase = situation_dynastic_cycle_phase_instability_conquest # 不稳定-征服阶段
                        OR = { situation_current_phase = situation_dynastic_cycle_phase_instability situation_current_phase = situation_dynastic_cycle_phase_hd_collapse }          # 基础不稳定阶段
                    }
                }
            }
            set_variable = SanGong_Zhongxing_var
        }
        else_if = {#登基时处于稳定阶段,储存昏君变量
            limit = {
                NOT = { has_variable = SanGong_HunJun_var2 }
                any_character_situation = { this = situation:dynastic_cycle
                    OR = {
                        situation_current_phase = situation_dynastic_cycle_phase_stability_expansion
                        OR = { situation_current_phase = situation_dynastic_cycle_phase_stability_advancement situation_current_phase = situation_dynastic_cycle_phase_hd_recovery }
                    }
                }
            }
            set_variable = SanGong_HunJun_var2
        }
    
 }
}
