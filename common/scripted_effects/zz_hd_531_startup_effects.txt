hd_531_game_start_effect = {
	hd_531_cleanup_effect = yes
	# 终结乱世锁：韩陵之战结束或 534 年解除（tgp_dynastic_cycle_decisions.txt）
	set_global_variable = hd_531_mandate_lock
	# 衣冠南渡是 4 世纪的事：尔朱兆、天光都是北方胡人王国，不设的话开局第一个月就会触发（计划复核必改 10）。

	set_global_variable = hd_yiguan_started
	# 历史入侵：尔朱荣已死；高欢起兵、宇文泰据关中由本剧本接管。侯景之乱（545 年起）保留
	set_global_variable = { name = hd_hi_done_erzhurong value = yes }
	set_global_variable = { name = hd_hi_done_gaohuan value = yes }
	set_global_variable = { name = hd_hi_done_yuwentai value = yes }
	# 天意青睐"每月主动开战"要求讨董链已结束（184 专用变量），同 304
	set_global_variable = { name = hd_anti_regent_chain_resolved value = yes }
	# 天子京城：洛阳
	title:h_china = {
		set_capital_county = title:c_luoyang
		set_variable = { name = hd_court_capital_county value = title:c_luoyang }
	}
	# 传国玉玺归节闵帝
	title:h_china.holder ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { exists = global_var:hd_han_treasures_initialized }
			}
			hd_han_create_imperial_seal_effect = { OWNER = this }
			set_global_variable = hd_han_treasures_initialized
		}
	}
	hd_531_dynastic_cycle_effect = yes
	hd_nbc_culture_effect = yes
	hd_531_rouran_culture_effect = yes
	hd_531_tributaries_effect = yes
	debug_log = "HD_531 STAGE GAME_START"
}

# 天朝循环：群雄逐鹿（斗争）开局，五德取北魏的水德；进度清零（写法同 hd_304_dynastic_cycle_effect）
hd_531_dynastic_cycle_effect = {
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = title:h_china.holder
			NOT = { exists = situation:dynastic_cycle }
		}
		start_situation = {
			type = dynastic_cycle
			start_phase = situation_dynastic_cycle_phase_chaos
		}
	}
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = situation:dynastic_cycle
			NOT = { exists = global_var:saas_dynastic_cycle_initialized }
		}
		situation:dynastic_cycle = {
			title:h_china = {
				set_variable = {
					name = years_since_imperial_examination
					value = 0
				}
			}
			tgp_set_wuking_element_effect = { ELEMENT = water }
			every_situation_participant = {
				recalculate_participant_group = situation:dynastic_cycle
			}
			every_situation_sub_region = {
				hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_hd_standoff }
			}
			trigger_event = { on_action = hd_531_dc_zero_progress days = 2 }
		}
		set_global_variable = {
			name = saas_dynastic_cycle_initialized
			value = yes
		}
		debug_log = "HD_531 DYNASTIC_CYCLE CHAOS"
	}
}

# 尔朱兆、天光、仲远开局独立，奉节闵帝为宗主（天朝朝贡合同：只有纳贡、册封，无参战义务）。
# 不调 start_tributary_interaction_effect：它结尾会给双方加 25 年停战（计划复核必改 1），写法照 304
hd_531_tributary_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_independent_ruler = yes
				is_tributary = no
				NOT = { this = title:h_china.holder }
			}
			start_tributary = {
				contract_group = tributary_celestial
				suzerain = title:h_china.holder
			}
			if = {
				limit = { is_tributary = yes }
				hidden_effect = {
					tributary_contract_set_obligation_level = { type = celestial_tribute_gold level = 1 }
					tributary_contract_set_obligation_level = { type = celestial_tribute_prestige level = 1 }
				}
				add_subject_standing = 10
			}
			debug_log = "HD_531 TRIBUTARY $WHO$"
		}
	}
}

# 挟天子：尔朱世隆为权臣兼丞相，元恭为傀儡，京城洛阳（照 hd253_regency_effect，换人）。Scope：尔朱世隆
hd_531_regency_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			character:hd_531_yuan_gong ?= { is_alive = yes has_title = title:h_china }
			liege ?= character:hd_531_yuan_gong
		}
		character:hd_531_yuan_gong = {
			set_location = {
				location = title:c_luoyang.title_province
				stick_to_location = yes
			}
		}
		# 玩家扮演时功名补到帝国级预期，否则丞相授不上（218 实测）
		if = {
			limit = {
				is_ai = no
				merit_level < merit_level_expected_empire_value
			}
			while = {
				limit = { merit_level < merit_level_expected_empire_value }
				change_merit_level = 1
			}
		}
		hd218_grant_chengxiang_effect = yes
		WJ_install_historical_secretariat_effect = { OFFICE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang LAW = WJ_grand_chengxiang_law }
		hd_court_install_regency_effect = yes
		hd_court_set_stance_effect = { STANCE = moni }
		character:hd_531_yuan_gong = {
			if = {
				limit = { has_diarchy_type = grand_secretariat diarch ?= character:erzhu_009 }
				set_diarchy_swing = 100
			}
		}
		# 终身辅政锁：天子罢免后自动补回（hd_dc_kit_ten_regent_watch）
		title:h_china = { set_variable = { name = hd_dc_ten_qjc_regent value = character:erzhu_009 } }
		# 权臣任期：标记在就不能废帝（hd_court_depose_ceremonial_decision）。照 253 设 12 年，立孝武帝节点执行时清掉（计划复核必改 6）
		add_character_flag = { flag = hd_court_regent_tenure years = 12 }
		if = {
			limit = { has_title = title:e_WJ_minister_cheng_xiang }
			debug_log = "HD_531 REGENCY ERZHU_SHILONG OK"
		}
		else = { debug_log = "HD_531 REGENCY ERZHU_SHILONG NO_CHENGXIANG" }
	}
	else = { debug_log = "HD_531 REGENCY SKIPPED" }
}
