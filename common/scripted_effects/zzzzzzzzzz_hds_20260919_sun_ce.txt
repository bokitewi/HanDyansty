hds_sc_tick_effect = {
    if = {
        limit = { has_global_variable = wj_hchar_sun_ce AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } is_alive = yes }
        hds_retire_legacy_controller_effect = yes
        hds_finish_title_transfer_effect = yes
        # 2026-10-10：营地身份打赢丹阳后转天朝政体、销毁营地头衔。
        hds_sun_record_camp_effect = yes
        hds_sun_landed_from_camp_effect = yes
        if = {
            limit = { has_character_flag = hds_sc_inherited NOT = { has_character_flag = hds_sc_inheritance_notice } }
            hds_queue_event_effect = { EVENT = hds.4000 LEASE = hds_lease_4000 }
        }
        if = {
            limit = { hds_sc_depart_ready_trigger = yes }
            if = {
                # Sun Jian already left Yuan Shu himself (hds.5000): skip the Yuan Shu departure story.
                limit = { character:sun_jian ?= { has_character_flag = hds_sj_departed } }
                hds_sc_depart_effect = yes
            }
            else = { hds_queue_event_effect = { EVENT = hds.4001 LEASE = hds_lease_4001 } }
        }
        if = {
            limit = { has_character_flag = hds_sc_departed }
            # Repairs an old completed-Wu save once; later lost land is not reset forever.
            if = {
                limit = { NOT = { has_character_flag = hds_nr_danyang_control_migrated } }
                add_character_flag = hds_nr_danyang_control_migrated
                add_character_flag = hds_sc_danyang_target_version
                if = {
                    limit = { has_character_flag = hds_sc_wu_complete NOT = { hds_sc_wu_control_trigger = yes } }
                    remove_character_flag = hds_sc_wu_complete
                    remove_character_flag = hds_sc_wu_notice
                }
            }
            # A previously departed actor may have been re-vassalized by another integration pulse.
            # The story already committed to independence, so restore it while the living adult is free.
            if = {
                limit = { is_adult = yes is_imprisoned = no exists = liege NOT = { is_independent_ruler = yes } }
                hds_release_effect = yes
            }
            if = { limit = { NOT = { has_character_flag = hds_sc_wu_complete } } add_character_flag = sworn_peace }
            hds_sc_seal_trade_effect = yes
            hds_sc_reward_effect = yes
            if = {
                limit = { hds_sc_zhou_ready_trigger = yes }
                hds_queue_event_effect = { EVENT = hds.4002 LEASE = hds_lease_4002 }
            }
            if = {
                limit = { has_global_variable = wj_hchar_zhou_yu AND = { trigger_if = { limit = { has_global_variable = wj_hchar_zhou_yu } global_var:wj_hchar_zhou_yu = { is_alive = no } } trigger_else = { always = no } } }
                add_character_flag = hds_sc_zhou_unavailable
            }
            # Title repair only for counties already controlled; never hands over enemy territory for free.
            if = {
                limit = { is_landed = yes NOT = { has_variable_list = hds_pending_titles } }
                hds_claim_owned_duchies_effect = { KINGDOM = title:k_uuii_yangzhou }
            }
            if = {
                limit = { hds_sc_wu_control_trigger = yes NOT = { has_variable_list = hds_pending_titles } }
                add_character_flag = hds_sc_wu_complete
                remove_character_flag = sworn_peace
            }
            if = {
                limit = { NOT = { has_character_flag = hds_sc_wu_complete } hds_can_campaign_trigger = yes }
                if = {
                    limit = { has_character_flag = hds_sc_wu_authorized }
                    hds_try_wu_war_effect = yes
                }
                else = { hds_queue_event_effect = { EVENT = hds.4003 LEASE = hds_lease_4003 } }
            }
            if = {
                limit = { has_character_flag = hds_sc_wu_complete }
                if = {
                    limit = { NOT = { has_character_flag = hds_sc_wu_notice } }
                    hds_queue_event_effect = { EVENT = hds.4004 LEASE = hds_lease_4004 }
                }
                hds_sc_yangzhou_historical_pulse_effect = yes
            }
        }
    }
}
# 孙策出走前重置领地：先取得于湖，其余交原上级；无上级则分立。
# 只由一次性的出走流程调用，丹阳宣战重试不会再次剥夺战果。
hds_sc_yuhu_settlement_effect = {
	save_scope_as = hds_sc_settlement_actor
	clear_saved_scope = hds_sc_settlement_liege
	liege ?= { save_scope_as = hds_sc_settlement_liege }
	hds_sun_record_camp_effect = yes
	create_title_and_vassal_change = { type = usurped save_scope_as = hds_sc_yuhu_change add_claim_on_loss = no }
	title:c_uuii_p_2098 = {
		change_title_holder = { holder = scope:hds_sc_settlement_actor change = scope:hds_sc_yuhu_change take_baronies = yes }
	}
	resolve_title_and_vassal_change = scope:hds_sc_yuhu_change
	if = {
		limit = { has_title = title:c_uuii_p_2098 }
		set_realm_capital = title:c_uuii_p_2098
		hds_sun_landed_from_camp_effect = yes
		# 已有于湖，不再转成冒险者；仅清除原家族庄园。
		hds_estate_to_camp_effect = { DONE_FLAG = hds_sc_camp_converted }
		if = {
			limit = { exists = scope:hds_sc_settlement_liege }
			create_title_and_vassal_change = { type = granted save_scope_as = hds_sc_return_change add_claim_on_loss = no }
			every_vassal = {
				limit = { NOT = { highest_held_title_tier = tier_barony capital_county = title:c_uuii_p_2098 } }
				change_liege = { liege = scope:hds_sc_settlement_liege change = scope:hds_sc_return_change }
			}
			every_held_title = {
				limit = {
					NOT = { this = title:c_uuii_p_2098 }
					NOT = { tier = tier_barony county = title:c_uuii_p_2098 }
				}
				change_title_holder = { holder = scope:hds_sc_settlement_liege change = scope:hds_sc_return_change }
			}
			resolve_title_and_vassal_change = scope:hds_sc_return_change
		}
		else = {
			# 直辖县按原版迁徙分立流程生成当地领主，并立即独立。
			every_held_title = {
				limit = { tier = tier_county is_noble_family_title = no NOT = { this = title:c_uuii_p_2098 } }
				save_temporary_scope_as = seceding_county
				create_character = {
					location = scope:seceding_county.title_province
					template = post_migration_minor_warlord_template
					save_scope_as = hds_sc_local_ruler
				}
				create_title_and_vassal_change = { type = granted save_scope_as = hds_sc_county_change add_claim_on_loss = no }
				change_title_holder = { holder = scope:hds_sc_local_ruler change = scope:hds_sc_county_change take_baronies = yes }
				resolve_title_and_vassal_change = scope:hds_sc_county_change
				scope:hds_sc_local_ruler = { hds_release_effect = yes }
			}
			# 不属于直辖县的零散男爵领归还所在县的领主。
			create_title_and_vassal_change = { type = granted save_scope_as = hds_sc_barony_change add_claim_on_loss = no }
			every_held_title = {
				limit = { tier = tier_barony NOT = { county = title:c_uuii_p_2098 } exists = county.holder }
				change_title_holder = { holder = this.county.holder change = scope:hds_sc_barony_change }
			}
			resolve_title_and_vassal_change = scope:hds_sc_barony_change
			every_vassal = {
				limit = { NOT = { highest_held_title_tier = tier_barony capital_county = title:c_uuii_p_2098 } }
				if = {
					limit = { highest_held_title_tier = tier_barony exists = capital_county.holder }
					create_title_and_vassal_change = { type = granted save_scope_as = hds_sc_barony_liege_change add_claim_on_loss = no }
					change_liege = { liege = this.capital_county.holder change = scope:hds_sc_barony_liege_change }
					resolve_title_and_vassal_change = scope:hds_sc_barony_liege_change
				}
				else = { hds_release_effect = yes }
			}
			every_held_title = {
				limit = { tier > tier_county }
				scope:hds_sc_settlement_actor = { destroy_title = prev }
			}
		}
		hds_release_effect = yes
		set_primary_title_to = title:c_uuii_p_2098
		add_character_flag = hds_sc_foothold_done
		add_character_flag = hds_sc_camp_converted
		title:h_china = {
			remove_variable = hds_sun_family_title
			remove_variable = hds_sun_original_estate
		}
	}
}

# Shared by Sun Jian and Sun Ce: right after leaving Yuan Shu and before the Danyang invasion,
# 销毁家族庄园（家族头衔），改为冒险者营地；原廷臣随营。
# (Vanilla yearly_playable_pulse destroys the estate of an independent noble-family-only ruler anyway.)
# A landed character only loses the estate (a landed ruler cannot hold a camp).
hds_estate_to_camp_effect = {
    if = {
        limit = {
            is_alive = yes
            NOT = { has_character_flag = $DONE_FLAG$ }
            any_held_title = { is_noble_family_title = yes }
        }
        save_scope_as = hds_camp_leader
        every_courtier = {
            limit = { is_alive = yes }
            add_to_temporary_list = hds_camp_followers
        }
        every_held_title = {
            limit = { is_noble_family_title = yes }
            save_temporary_scope_as = hds_camp_family_title
            prev = { destroy_title = scope:hds_camp_family_title }
        }
        if = {
            limit = { is_landed = no is_ruler = no }
            create_landless_adventurer_title_effect = {
                REASON = flag:voluntary
                FLAVOR_CHAR = this
            }
        }
        if = {
            limit = { is_ruler = yes is_independent_ruler = no }
            create_title_and_vassal_change = { type = independency save_scope_as = hds_camp_independence_change add_claim_on_loss = no }
            becomes_independent = { change = scope:hds_camp_independence_change }
            resolve_title_and_vassal_change = scope:hds_camp_independence_change
        }
        if = {
            limit = { is_ruler = yes }
            every_in_list = {
                list = hds_camp_followers
                limit = {
                    is_alive = yes
                    is_ruler = no
                    is_imprisoned = no
                    NOT = { is_courtier_of = scope:hds_camp_leader }
                }
                set_employer = scope:hds_camp_leader
            }
        }
        add_character_flag = $DONE_FLAG$
    }
}
# Sun Ce scope. hds_sc_inherited keeps hds_sun_estate_inherited_trigger true afterwards.
# On the camp line (Sun Jian already converted) Sun Ce inherits the camp and holds no estate: no-op.
hds_sc_estate_to_camp_effect = {
    if = {
        limit = { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } has_character_flag = hds_sc_inherited }
        hds_estate_to_camp_effect = { DONE_FLAG = hds_sc_camp_converted }
    }
}
# Sun Jian scope (hds_sj_independence_effect). The estate cache is cleared so the old
# estate-transfer path stays off; hds_sun_snapshot_death_effect still lists the camp title
# (duchy tier) in hds_sun_death_titles and hds_sun_force_inherit_effect hands it to Sun Ce.
hds_sj_estate_to_camp_effect = {
    if = {
        limit = { this = character:sun_jian }
        hds_estate_to_camp_effect = { DONE_FLAG = hds_sj_camp_converted }
        if = {
            limit = { has_character_flag = hds_sj_camp_converted }
            set_global_variable = hds_sun_camp_line
            title:h_china = {
                remove_variable = hds_sun_family_title
                remove_variable = hds_sun_original_estate
            }
        }
    }
}
hds_sc_seal_trade_effect = {
    if = {
        limit = {
            is_alive = yes AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
            has_character_flag = hds_sc_departed
            NOT = { has_character_flag = hds_sc_seal_delivered }
            exists = character:yuan_shu
            character:yuan_shu ?= { is_alive = yes }
            any_character_artifact = { has_variable = han_artifact_imperial_seal }
        }
        random_character_artifact = {
            limit = { has_variable = han_artifact_imperial_seal }
            save_scope_as = hds_nr_traded_seal
            set_owner = character:yuan_shu
        }
        if = {
            limit = { scope:hds_nr_traded_seal = { artifact_owner = character:yuan_shu } }
            add_character_flag = hds_sc_seal_delivered
            set_variable = { name = hds_sc_traded_seal value = scope:hds_nr_traded_seal }
        }
    }
}
