# 发动义军起义
# $LEADER$：首领（已有角色或刚生成）；$COUNTY$：起事县；$MEN$：义军人数；$BIG$ = yes 时目标另含两个相邻沦陷县
# 结果：scope:hd_yiguan_up_war（成功时存在）
# 守方已在应付一场义军 / 举兵起义 / 起兵北还时不再开战（hd_yiguan_no_active_uprising_trigger，§5.3）
############################
hd_yiguan_spawn_uprising_effect = {
	clear_saved_scope = hd_yiguan_up_war
	clear_saved_scope = hd_yiguan_up_n1
	clear_saved_scope = hd_yiguan_up_n2
	clear_saved_scope = hd_yiguan_up_defender
	$COUNTY$ = {
		save_scope_as = hd_yiguan_up_county
		title_province = { save_scope_as = hd_yiguan_up_province }
		holder ?= {
			top_liege = { save_scope_as = hd_yiguan_up_defender }
		}
	}
	$LEADER$ = { save_scope_as = hd_yiguan_up_leader }

	if = {
		limit = {
			always = $BIG$
			exists = scope:hd_yiguan_up_defender
		}
		scope:hd_yiguan_up_county = {
			random_neighboring_county = {
				limit = {
					has_variable = hd_yiguan_occ
					holder ?= { top_liege = scope:hd_yiguan_up_defender }
				}
				save_scope_as = hd_yiguan_up_n1
			}
			random_neighboring_county = {
				limit = {
					has_variable = hd_yiguan_occ
					holder ?= { top_liege = scope:hd_yiguan_up_defender }
					NOT = { this = scope:hd_yiguan_up_n1 }
				}
				save_scope_as = hd_yiguan_up_n2
			}
		}
	}

	if = {
		limit = {
			exists = scope:hd_yiguan_up_defender
			NOT = { scope:hd_yiguan_up_leader = scope:hd_yiguan_up_defender }
			scope:hd_yiguan_up_leader = { is_alive = yes }
			scope:hd_yiguan_up_defender = { hd_yiguan_no_active_uprising_trigger = yes }
		}
		scope:hd_yiguan_up_leader = {
			if = {
				limit = { NOT = { has_government = landless_adventurer_government } }
				set_location = scope:hd_yiguan_up_province
				create_landless_adventurer_title_effect = {
					REASON = flag:historical
					FLAVOR_CHAR = scope:hd_yiguan_up_leader
				}
			}
			if = {
				limit = {
					exists = scope:hd_yiguan_up_n1
					exists = scope:hd_yiguan_up_n2
				}
				start_war = {
					cb = hd_yiguan_uprising_cb
					target = scope:hd_yiguan_up_defender
					target_title = scope:hd_yiguan_up_county
					target_title = scope:hd_yiguan_up_n1
					target_title = scope:hd_yiguan_up_n2
				}
			}
			else_if = {
				limit = { exists = scope:hd_yiguan_up_n1 }
				start_war = {
					cb = hd_yiguan_uprising_cb
					target = scope:hd_yiguan_up_defender
					target_title = scope:hd_yiguan_up_county
					target_title = scope:hd_yiguan_up_n1
				}
			}
			else = {
				start_war = {
					cb = hd_yiguan_uprising_cb
					target = scope:hd_yiguan_up_defender
					target_title = scope:hd_yiguan_up_county
				}
			}
			random_character_war = {
				limit = {
					primary_attacker = scope:hd_yiguan_up_leader
					using_cb = hd_yiguan_uprising_cb
				}
				save_scope_as = hd_yiguan_up_war
			}
		}
		if = {
			limit = { exists = scope:hd_yiguan_up_war }
			scope:hd_yiguan_up_leader = {
				spawn_army = {
					name = hd_yiguan_uprising_army
					levies = $MEN$
					location = scope:hd_yiguan_up_province
					origin = scope:hd_yiguan_up_province
					uses_supply = yes
					inheritable = no
					war = scope:hd_yiguan_up_war
					save_temporary_scope_as = hd_yiguan_up_army
				}
			}
			scope:hd_yiguan_up_army ?= { assign_commander = scope:hd_yiguan_up_leader }
			# 催化剂"沦陷区汉人起义"：只在守方为胡人宗主（非汉化）时投；流民帅作乱、北府兵变打的是南方政权，不投
			if = {
				limit = {
					scope:hd_yiguan_up_defender = {
						hd_yiguan_is_hu_trigger = yes
						NOT = { hd_yiguan_is_sinicized_trigger = yes }
					}
				}
				hd_yiguan_fire_catalyst_effect = {
					CATALYST = catalyst_hd_yiguan_uprising
					CHAR = scope:hd_yiguan_up_leader
				}
			}
		}
		else = {
			debug_log = "hd_yiguan_war: uprising start_war FAILED"
		}
	}
}

# 义军胜利：取得目标县与占领县，独立并改天朝制，销毁营地
# 守方为胡人宗主（非汉化，即沦陷区起义）时：主头衔不足公国级则补立郡号、重算参与者组；
# 响应起义而一并独立的封臣 3 天后收"归南之议"（hd_yiguan.2120，旧事件改编模块；事件自查是否与南方政权接壤），同旧版；首领本人不收
############################
hd_yiguan_uprising_victory_effect = {
	if = {
		limit = { exists = root.war }
		root.war = { save_scope_as = hd_yiguan_victory_war }
		clear_saved_scope = hd_yiguan_up_hu_def
		if = {
			limit = {
				scope:defender = {
					hd_yiguan_is_hu_trigger = yes
					NOT = { hd_yiguan_is_sinicized_trigger = yes }
				}
			}
			scope:defender = { save_scope_as = hd_yiguan_up_hu_def }
		}
		hd_uprising_settle_occupied_counties_on_victory_effect = {
			WAR = scope:hd_yiguan_victory_war
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = hd_yiguan_up_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = target_titles
			limit = {
				holder ?= {
					OR = {
						this = scope:defender
						target_is_liege_or_above = scope:defender
					}
				}
			}
			change_title_holder = {
				holder = scope:attacker
				change = scope:hd_yiguan_up_change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:hd_yiguan_up_change
		hd_uprising_grant_independence_effect = {
			WAR = scope:hd_yiguan_victory_war
		}
		scope:attacker = {
			if = {
				limit = {
					is_alive = yes
					is_landed = yes
				}
				hd_force_celestial_effect = yes
				hd_yiguan_war_leave_camp_effect = yes
				if = {
					limit = { exists = scope:hd_yiguan_up_hu_def }
					hd_yiguan_war_ensure_duchy_effect = yes
					hd_yiguan_refresh_realm_effect = yes
					hd_yiguan_recalc_group_effect = yes
					# 首领本人不发归南之议（§1 第 7 轮：胜则独立、进北方汉人组；要归附可自用"归附南朝"决议）
				}
			}
		}

		# 响应起义的封臣
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = hd_yiguan_ally_change
			add_claim_on_loss = no
		}
		scope:hd_yiguan_victory_war = {
			every_war_attacker = {
				limit = {
					NOT = { this = scope:attacker }
					is_landed = yes
					target_is_liege_or_above = scope:defender
				}
				becomes_independent = { change = scope:hd_yiguan_ally_change }
				add_to_list = hd_yiguan_freed_allies
			}
		}
		resolve_title_and_vassal_change = scope:hd_yiguan_ally_change
		every_in_list = {
			list = hd_yiguan_freed_allies
			limit = { is_alive = yes }
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = scope:hd_yiguan_victory_war
				result = victory
			}
			if = {
				limit = { exists = scope:hd_yiguan_up_hu_def }
				hd_yiguan_war_ensure_duchy_effect = yes
				hd_yiguan_refresh_realm_effect = yes
				hd_yiguan_recalc_group_effect = yes
				if = {
					limit = { hd_yiguan_active_trigger = yes }
					trigger_event = {
						id = hd_yiguan.2120
						days = 3
					}
				}
			}
		}
	}
}

# 义军失败或白和：响应起义的封臣记"附逆"
hd_yiguan_uprising_failed_allies_effect = {
	if = {
		limit = { exists = root.war }
		root.war = {
			every_war_attacker = {
				limit = {
					NOT = { this = scope:attacker }
					target_is_liege_or_above = scope:defender
				}
				add_prestige = medium_prestige_loss
				scope:defender = {
					add_opinion = {
						target = prev
						modifier = hd_yiguan_rebel_ally_opinion
					}
				}
			}
		}
	}
}

# 夺取法理头衔内守方领地。Scope：法理头衔（州或郡）。
# 须已有 scope:attacker、scope:defender、scope:change（宣战理由结算时由引擎提供，change 由调用方建立）
# 先例：mod zz_hd_253_beifa_cb.txt、原版 10_tgp_china_wars.txt:816（china_seize_duchy_cb）
hd_yiguan_war_seize_dejure_effect = {
	save_scope_as = hd_yiguan_seize_root
	every_de_jure_county = {
		limit = {
			is_landless_type_title = no
			holder ?= {
				OR = {
					this = scope:defender
					target_is_liege_or_above = scope:defender
				}
			}
		}
		change_title_holder = {
			holder = scope:attacker
			change = scope:change
			take_baronies = yes
		}
	}
	# 守方本人在此法理头衔下持有的郡（州目标时；目标本身另由下段处理）
	every_in_de_jure_hierarchy = {
		continue = { tier > tier_duchy }
		limit = {
			tier = tier_duchy
			NOT = { this = scope:hd_yiguan_seize_root }
			holder ?= scope:defender
		}
		change_title_holder = {
			holder = scope:attacker
			change = scope:change
		}
	}
	# 目标头衔本身由守方持有时一并交出
	if = {
		limit = { holder ?= scope:defender }
		change_title_holder = {
			holder = scope:attacker
			change = scope:change
		}
	}
}

# 营地善后。Scope：刚获地的前冒险者。销毁营地头衔（原版 destroy_laamp_effect，07_dlc_ep3_scripted_effects.txt:643）
# 须在改政体之后调用（原版 ep3_become_landed_transfer_effect 的顺序：先换政体、末尾再销毁营地）
hd_yiguan_war_leave_camp_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			any_held_title = { has_variable = adventurer_creation_reason }
		}
		save_scope_as = hd_yiguan_camp_leaver
		destroy_laamp_effect = { ADVENTURER = scope:hd_yiguan_camp_leaver }
	}
}

# 补立郡号。Scope：刚独立或刚获地的汉人（在宣战理由结算中调用；scope:defender 为本战守方）。
# 参与者组要求实领公爵级以上（§5.3 hd_yiguan_is_regime_base_trigger）；实领不足公爵级时依次尝试，保证能归入北方汉人组：
#  1. 都城所在法理郡无人持有：为其立郡（原版 00_decisions_effects.txt:43，type = created）
#  2. 该郡由本战守方本人持有、且不是守方的主头衔：以征服转给胜者（只转头衔不带封臣；守方封臣公爵的郡不夺）
#  3. 否则新建动态郡号，以都城县命名（原版 create_popular_revolt_title_effect，00_faction_effects.txt:757；
#     不设 set_landless_title，否则过不了政权基础判定）。日后得到其他公爵级头衔时自动销毁
hd_yiguan_war_ensure_duchy_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			hd_yiguan_war_landed_tier_value < 3	# 实领不足公爵级（官职等无地头衔不算）
		}
		save_scope_as = hd_yiguan_duchy_receiver
		clear_saved_scope = hd_yiguan_duchy_county
		clear_saved_scope = hd_yiguan_new_duchy
		capital_county ?= {
			save_scope_as = hd_yiguan_duchy_county
			duchy ?= { save_scope_as = hd_yiguan_new_duchy }
		}
		if = {
			limit = {
				exists = scope:hd_yiguan_new_duchy
				scope:hd_yiguan_new_duchy = {
					NOT = { exists = holder }
				}
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = hd_yiguan_duchy_change
				add_claim_on_loss = no
			}
			scope:hd_yiguan_new_duchy = {
				change_title_holder = {
					holder = scope:hd_yiguan_duchy_receiver
					change = scope:hd_yiguan_duchy_change
				}
			}
			resolve_title_and_vassal_change = scope:hd_yiguan_duchy_change
		}
		else_if = {
			limit = {
				exists = scope:defender
				exists = scope:hd_yiguan_new_duchy
				scope:hd_yiguan_new_duchy = {
					holder ?= scope:defender
				}
				scope:defender = {
					primary_title ?= {
						NOT = { this = scope:hd_yiguan_new_duchy }
					}
				}
			}
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = hd_yiguan_duchy_change
				add_claim_on_loss = yes
			}
			scope:hd_yiguan_new_duchy = {
				change_title_holder = {
					holder = scope:hd_yiguan_duchy_receiver
					change = scope:hd_yiguan_duchy_change
				}
			}
			resolve_title_and_vassal_change = scope:hd_yiguan_duchy_change
		}
		else_if = {
			limit = { exists = scope:hd_yiguan_duchy_county }
			create_dynamic_title = {
				tier = duchy
				name = hd_yiguan_war_dyn_duchy_name
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = hd_yiguan_duchy_change
				add_claim_on_loss = no
			}
			scope:new_title = {
				set_capital_county = scope:hd_yiguan_duchy_county
				set_color_from_title = scope:hd_yiguan_duchy_county
				set_no_automatic_claims = yes
				set_destroy_on_gain_same_tier = yes
				set_delete_on_destroy = yes
				set_can_be_named_after_dynasty = no
				change_title_holder = {
					holder = scope:hd_yiguan_duchy_receiver
					change = scope:hd_yiguan_duchy_change
				}
			}
			resolve_title_and_vassal_change = scope:hd_yiguan_duchy_change
			scope:new_title = { generate_coa = yes }
			debug_log = "hd_yiguan_war: ensure_duchy created dynamic duchy"
		}
	}
}

# AI 北伐一次。Scope：南方政权 AI。
# 在北方八州里随机取一个与本国接壤、属胡人或汉化政权（兵力不超过本国 1.25 倍、无停战、未结盟）的县，以其所在州为目标、其独立宗主为守方
hd_yiguan_war_ai_try_beifa_effect = {
	save_scope_as = hd_yiguan_aw_att
	clear_saved_scope = hd_yiguan_aw_county
	clear_saved_scope = hd_yiguan_aw_def
	clear_saved_scope = hd_yiguan_aw_title
	clear_saved_scope = hd_yiguan_aw_war
	random_county_in_region = {
		region = hd_yiguan_north_region
		limit = {
			holder ?= {
				top_liege = {
					NOT = { this = scope:hd_yiguan_aw_att }
					OR = {
						hd_yiguan_is_hu_regime_trigger = yes
						hd_yiguan_is_sinicized_regime_trigger = yes
					}
					NOT = { government_has_flag = government_is_herder }
					NOT = { is_at_war_with = scope:hd_yiguan_aw_att }
					NOT = { is_allied_to = scope:hd_yiguan_aw_att }
					NOT = {
						scope:hd_yiguan_aw_att = { has_truce = prev }
					}
					# 朝贡关系两向都不打（同原版 herders_and_tributary_constraints，00_war_and_peace_triggers.txt:1144）
					NOT = { is_tributary_of_suzerain_or_above = scope:hd_yiguan_aw_att }
					NOT = {
						scope:hd_yiguan_aw_att = { is_tributary_of_suzerain_or_above = prev }
					}
					max_military_strength <= {
						value = scope:hd_yiguan_aw_att.max_military_strength
						multiply = 1.25
					}
				}
			}
			any_neighboring_county = {
				holder ?= { target_is_same_character_or_above = scope:hd_yiguan_aw_att }
			}
		}
		save_scope_as = hd_yiguan_aw_county
	}
	if = {
		limit = { exists = scope:hd_yiguan_aw_county }
		scope:hd_yiguan_aw_county = {
			holder.top_liege = { save_scope_as = hd_yiguan_aw_def }
			kingdom ?= { save_scope_as = hd_yiguan_aw_title }
		}
	}
	if = {
		limit = {
			exists = scope:hd_yiguan_aw_def
			exists = scope:hd_yiguan_aw_title
		}
		start_war = {
			cb = hd_yiguan_beifa_cb
			target = scope:hd_yiguan_aw_def
			target_title = scope:hd_yiguan_aw_title
		}
		random_character_war = {
			limit = {
				primary_attacker = scope:hd_yiguan_aw_att
				using_cb = hd_yiguan_beifa_cb
			}
			save_scope_as = hd_yiguan_aw_war
		}
		if = {
			limit = { exists = scope:hd_yiguan_aw_war }
			add_prestige = -150
			save_scope_as = hd_yiguan_aw_beifa_done
			debug_log = "hd_yiguan_war: AI beifa declared"
			debug_log_scopes = yes
		}
		else = {
			debug_log = "hd_yiguan_war: AI beifa start_war FAILED"
		}
	}
}
