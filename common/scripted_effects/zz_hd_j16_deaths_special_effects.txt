# 佛图澄（A5-G4★B）：hd_fo 运行时生成（tools/hd_fo/gen_hd_fo_monks.py，ID fo_tu_cheng，指针 global_var:hd_fo_p_fo_tu_cheng），
# 库中没有 history 条目，不能用 character:fo_tu_cheng，所以不走 hd_j16_death_effect 的解析，直接取 hd_fo 指针。
# 日期 348.12.15（设计写 348.12；《高僧传》作永和四年十二月八日，农历换算约为 349 年 1 月【待核】）。
# 在世：按 N 类病逝（借 core 的 hd_j16_death_natural_effect：AI 当即病逝，玩家弹病笃）；死后 hd_fo_on_death 照常发圆寂通知。
# 已死、或到 DATE10（358.12.15）仍无指针（未生成、开局晚于 348 年）：只设 done。
############################################################
hd_j16_death_fotucheng_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = hd_nbc_death_fo_tu_cheng_done }
			current_date >= 348.12.15
		}
		if = {
			limit = { hd_fo_monk_alive_trigger = { ID = fo_tu_cheng } }
			if = {
				limit = { exists = scope:hd_j16_dv }
				clear_saved_scope = hd_j16_dv
			}
			global_var:hd_fo_p_fo_tu_cheng = { save_scope_as = hd_j16_dv }
			hd_j16_death_natural_effect = { ID = fo_tu_cheng }
		}
		else_if = {
			limit = {
				OR = {
					global_var:hd_fo_p_fo_tu_cheng ?= { is_alive = no }
					current_date >= 358.12.15
				}
			}
			set_global_variable = hd_nbc_death_fo_tu_cheng_done
			debug_log = "HD_J16 death_fo_tu_cheng SKIP missing"
		}
	}
}

hd_j16_death_huanwen_effect = {
	# 不另查 done：已死或已 done 时 hd_j16_resolve_effect 只存在世的人、取不到就不设阻断，hd_j16_death_effect 自己查 done
	if = {
		limit = { current_date >= 373.8.18 }
		if = {
			limit = { exists = scope:hd_j16_hw }
			clear_saved_scope = hd_j16_hw
		}
		hd_j16_resolve_effect = { ID = longkang_huan_017 AS = hd_j16_hw }
		if = {
			limit = {
				scope:hd_j16_hw ?= {
					is_ai = yes
					primary_title ?= {
						current_heir ?= { is_ai = yes }
					}
				}
			}
			title:h_china = {
				set_variable = { name = hd_j16_handoff_block value = scope:hd_j16_hw }
			}
			debug_log = "HD_J16 death_longkang_huan_017 FIRE handoff_block"
		}
		hd_j16_death_effect = { ID = longkang_huan_017 DATE = 373.8.18 DATE10 = 383.8.18 }
	}
