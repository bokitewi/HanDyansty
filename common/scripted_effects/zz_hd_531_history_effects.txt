# 被杀（有凶手）。电脑角色直接死；玩家角色不杀，只记日志（玩家改写了历史就不强加）
hd_531_kill_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ killer = character:$KILLER$ }
			debug_log = "HD_531 KILLED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_531 KILL_SKIPPED_PLAYER $WHO$"
		}
	}
}
# 被杀（无具名凶手：部下所杀、自尽）
hd_531_die_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ }
			debug_log = "HD_531 DIED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_531 DIE_SKIPPED_PLAYER $WHO$"
		}
	}
}

# $NEW$ 接收 $OLD$ 的全部头衔与封臣（不含世族、朝官头衔），$OLD$ 不死。
# 照 hd253_feifang_effect：同一个 change 里先转头衔、后改投封臣（执行后复核必改 3，新主可能是无地廷臣）
hd_531_take_realm_effect = {
	$NEW$ = { save_scope_as = hd_531_new_ruler }
	create_title_and_vassal_change = { type = granted save_scope_as = hd_531_take_change add_claim_on_loss = no }
	$OLD$ = {
		every_held_title = {
			limit = {
				is_noble_family_title = no
				tgp_is_minister_title = no
				is_landless_type_title = no
				tier >= tier_county
			}
			change_title_holder = {
				holder = scope:hd_531_new_ruler
				change = scope:hd_531_take_change
				take_baronies = yes
			}
		}
		every_vassal = {
			limit = { NOT = { this = scope:hd_531_new_ruler } }
			change_liege = { liege = scope:hd_531_new_ruler change = scope:hd_531_take_change }
		}
	}
	resolve_title_and_vassal_change = scope:hd_531_take_change
}

	}

		limit = {
			current_date >= 532.12.20
			exists = global_var:hd_531_xiaowu_done
			NOT = { exists = global_var:hd_531_death_yuan_lang_done }
		}

	character:hd_531_yuan_lang = { save_scope_as = hd_531_lang }
	scope:hd_531_lang = {
		if = {
			limit = { is_landed = yes }
			if = {
				limit = { exists = liege }
				liege = { save_scope_as = hd_531_lang_recipient }
			}
			else = {
				title:h_china.holder ?= { save_scope_as = hd_531_lang_recipient }
			}
			if = {
				limit = {
					exists = scope:hd_531_lang_recipient
					scope:hd_531_lang_recipient = { NOT = { this = scope:hd_531_lang } }
				}
				hd_531_take_realm_effect = { NEW = scope:hd_531_lang_recipient OLD = scope:hd_531_lang }
			}
		}
	}
	character:hd_hi_gaohuan = {
		if = {
			limit = { scope:hd_531_lang = { is_landed = no } }
			add_courtier = scope:hd_531_lang
		}
		hd_531_change_era_effect = { KEY = zhongxing YEAR = 1 }
		primary_title = { set_title_name = dynn_title_wei }
		add_prestige = medium_prestige_gain

# S2 攻邺（普泰元年十一月起，高欢引兵攻邺，次年正月拔之，531.12 起）。守方只有相州刺史刘诞（拍板 1 默认 A）

hd_531_check_ye_effect = {
	if = {
		limit = {
			current_date >= 531.12.1
			NOT = { exists = global_var:hd_531_ye_started }
		}
		set_global_variable = hd_531_ye_started
		if = {
			limit = {
				character:hd_hi_gaohuan ?= {
					is_alive = yes
					is_independent_ruler = yes
					is_imprisoned = no
				}
				character:hd_531_liu_dan ?= {
					is_alive = yes
					has_title = title:d_weijun
				}
			}
			character:hd_hi_gaohuan = { trigger_event = hd_531.0120 }
		}
		else = {
			set_global_variable = hd_531_ye_done
			debug_log = "HD_531 S2 YE SKIPPED"
		}
	}
	# 战后：邺城归高欢一系算高欢胜，刘诞被杀
	else_if = {
		limit = {
			exists = global_var:hd_531_ye_decided
			NOT = { exists = global_var:hd_531_ye_done }
			NOT = { character:hd_hi_gaohuan ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:ye } } }
			NOT = { title:d_weijun.holder ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:ye } } }
		}
		set_global_variable = hd_531_ye_done
		if = {
			limit = {
				character:hd_hi_gaohuan ?= { is_alive = yes }
				title:c_uuii_p_1172.holder ?= {
					OR = {
						this = character:hd_hi_gaohuan
						top_liege = character:hd_hi_gaohuan
					}
				}
			}
			set_global_variable = hd_531_ye_gaohuan_won
			hd_531_kill_effect = { WHO = hd_531_liu_dan REASON = death_execution KILLER = hd_hi_gaohuan }
			debug_log = "HD_531 S2 YE GAOHUAN_WON"
		}
		else = {
			set_global_variable = hd_531_ye_gaohuan_lost
			debug_log = "HD_531 S2 YE GAOHUAN_LOST"
		}
	}
}
