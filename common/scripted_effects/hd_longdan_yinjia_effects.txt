hd_create_longdan_yinjia_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = hd_longdan_yinjia_name
		description = hd_longdan_yinjia_description
		type = armor_lamellar
		template = hd_longdan_yinjia_template
		visuals = hd_longdan_yinjia
		wealth = scope:wealth
		quality = scope:quality
		max_durability = 100
		history = { type = created_before_history }
		modifier = artifact_armor_4_artifact_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}
	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = hd_longdan_yinjia
	}
}
