# One real office. Context is saved only to find the former court after independence.
WJ_cx_finish_appointment = {
 # Successful office/seat assignment also binds the emperor's placeholder trait.
 if = {
  limit = {
   has_title = title:e_WJ_minister_cheng_xiang
   has_council_position = WJ_minister_cheng_xiang
   is_councillor_of = title:h_china.holder
  }
  title:h_china.holder = { add_trait = WJ_lost_imperial_authority }
 }
 if = {
  limit = {
   has_title = title:e_WJ_minister_cheng_xiang
   has_council_position = WJ_minister_cheng_xiang
   is_councillor_of = title:h_china.holder
   NOT = { has_variable = wj_cx_court }
  }
  set_variable = { name = wj_cx_court value = title:h_china }
  set_variable = { name = wj_cx_name value = flag:WJ_cx_name_chengxiang }
  remove_variable = wj_cx_name_text
  remove_localized_text = wj_cx_name
  if = { limit = { game_start_date = 190.1.1 current_date = 190.1.1 this = character:dong_zhuo_2 }
   set_variable = { name = wj_cx_name value = flag:WJ_cx_name_xiangguo }
  }
  title:e_WJ_minister_cheng_xiang = {
   set_de_jure_liege_title = title:h_china
   if = { limit = { holder = { var:wj_cx_name = flag:WJ_cx_name_xiangguo } }
    set_title_name = WJ_cx_name_xiangguo
   }
   else = { set_title_name = WJ_cx_name_chengxiang }
   set_definitive_form = no
  }
  title:e_WJ_minister_cheng_xiang = {
   if = { limit = { NOT = { has_title_law = noble_family_succession_law } }
    add_title_law = WJ_chancellor_life_law
   }
  }
  title:h_china.holder = {
   add_realm_law_skip_effects = WJ_grand_chengxiang_law
   WJ_sync_grand_secretariat_effect = yes
   if = { limit = { diarch = prev } set_diarchy_swing = 60 }
  }
  WJ_zw_grant_chancellor = yes
  # Deferred until the outer historical/ordinary grant has finished all effects.
  if = { limit = { is_ai = no } trigger_event = { id = WJ_chancellor.10 delayed = yes } }
 }
}

# Ordinary death, dismissal, departure and destruction share this exit.
# An actual hereditary title transfer is handled separately by on_title_gain.
WJ_cx_abolish = {
 if = {
  limit = {
   NOT = { has_character_flag = wj_cx_cleaning }
   OR = { has_variable = wj_cx_court has_title = title:e_WJ_minister_cheng_xiang }
  }
  add_character_flag = wj_cx_cleaning
  save_scope_as = wj_cx_departing
  title:h_china.holder ?= { save_scope_as = wj_cx_emperor }
  if = { limit = { exists = var:wj_cx_court.holder }
   var:wj_cx_court.holder = { save_scope_as = wj_cx_emperor }
  }
  # Stop stale historical / private-title-heir repair before callbacks fire.
  if = { limit = { exists = title:h_china.var:hd_court_regent title:h_china.var:hd_court_regent = scope:wj_cx_departing }
   hd_court_release_effect = yes
  }
  scope:wj_cx_emperor ?= {
   if = { limit = { exists = var:hd_dong_fixed_regent var:hd_dong_fixed_regent = scope:wj_cx_departing }
    remove_variable = hd_dong_fixed_regent
   }
  }
  if = {
   limit = { exists = scope:wj_cx_emperor this != scope:wj_cx_emperor }
   create_title_and_vassal_change = { type = granted save_scope_as = wj_cx_return add_claim_on_loss = no }
   every_vassal = {
    # An existing same-court de-jure liege wins; never create a title or a cycle.
    save_scope_as = wj_cx_vassal
    scope:wj_cx_emperor = { save_scope_as = wj_cx_return_liege }
    clear_saved_scope = wj_cx_dejure
    primary_title.de_jure_liege ?= { save_scope_as = wj_cx_dejure }
    # Internal de-jure search is not a player-facing repeated effect.
    hidden_effect = {
     while = {
      count = 5
      limit = { exists = scope:wj_cx_dejure scope:wj_cx_return_liege = scope:wj_cx_emperor }
      scope:wj_cx_dejure = {
       if = {
        limit = {
         exists = holder
         holder != scope:wj_cx_departing
         holder != scope:wj_cx_vassal
         holder = {
          top_liege = scope:wj_cx_emperor
          NOT = { is_vassal_or_below_of = scope:wj_cx_vassal }
          highest_held_title_tier > scope:wj_cx_vassal.highest_held_title_tier
         }
        }
        holder = { save_scope_as = wj_cx_return_liege }
        clear_saved_scope = wj_cx_dejure
       }
       else_if = { limit = { exists = de_jure_liege } de_jure_liege = { save_scope_as = wj_cx_dejure } }
       else = { clear_saved_scope = wj_cx_dejure }
      }
     }
    }
    change_liege = { liege = scope:wj_cx_return_liege change = scope:wj_cx_return }
   }
   resolve_title_and_vassal_change = scope:wj_cx_return
  }
  if = { limit = { has_council_position = WJ_minister_cheng_xiang }
   scope:wj_cx_emperor ?= { always_fire_councillor = scope:wj_cx_departing }
  }
  if = { limit = { has_title = title:e_WJ_minister_cheng_xiang }
   title:e_WJ_minister_cheng_xiang = {
    remove_variable = wj_cx_successor
   }
   destroy_title = title:e_WJ_minister_cheng_xiang
  }
  remove_variable = wj_cx_court
  remove_character_flag = officer_cheng_xiang
  WJ_zw_clear = yes
  if = { limit = { hd_holds_any_ministry_title_trigger = no }
   remove_character_modifier = minister_scheme_slots_modifier
  }
  # A successor emperor keeps his own realm; only the old office is abolished.
  scope:wj_cx_emperor ?= {
   if = { limit = { NOT = { exists = title:e_WJ_minister_cheng_xiang.holder } }
    remove_trait = WJ_lost_imperial_authority
   }
   if = { limit = { has_realm_law = WJ_grand_chengxiang_law NOT = { exists = title:e_WJ_minister_cheng_xiang.holder } }
    if = { limit = { exists = diarch diarch = scope:wj_cx_departing } end_diarchy = yes }
    add_realm_law_skip_effects = grand_chancellor_law
    if = { limit = { exists = title:e_minister_chancellor.holder }
     WJ_sync_grand_secretariat_effect = yes
    }
   }
  }
  WJ_reconcile_contract_privileges_effect = yes
  remove_character_flag = wj_cx_cleaning
 }
}

WJ_cx_before_death = {
 WJ_name_before_death = yes
 if = {
  limit = {
   title:e_WJ_minister_cheng_xiang = {
    has_title_law = noble_family_succession_law
    exists = current_heir
    current_heir = { WJ_cx_heir_can_serve = yes }
   }
  }
  add_character_flag = { flag = wj_cx_death_handoff days = 2 }
  title:e_WJ_minister_cheng_xiang = {
   set_variable = { name = wj_cx_successor value = current_heir days = 2 }
  }
 }
 else = { WJ_cx_abolish = yes }
}
