# Shared story heartbeat. Historical state applies to all owners; automatic
# schemes and wars are restricted to AI characters.
hd_historical_actor_story_tick_effect = {
	if = {
		limit = { NOT = { hds_southern_actor_trigger = yes } }

	save_scope_as = hd_historical_actor
	if = {
		limit = { this = character:gong_sun_zan exists = global_var:hd_anti_regent_chain_resolved NOT = { has_character_flag = hd_gongsun_youzhou_claim_received } }
		if = { limit = { NOT = { has_title = title:k_uuii_youzhou } NOT = { has_claim_on = title:k_uuii_youzhou } } add_pressed_claim = title:k_uuii_youzhou }
		add_character_flag = hd_gongsun_youzhou_claim_received
		random_owned_story = { type = hd_story_gongsun_zan_life set_variable = { name = hd_gongsun_zan_life_phase value = flag:hd_story_phase_northern_conquest } }
	}
	if = {
		limit = { this = character:cao_cao exists = global_var:hd_anti_regent_chain_resolved NOT = { has_character_flag = hd_cao_yan_qing_claims_received } }
		if = { limit = { NOT = { has_title = title:k_uuii_yanzhou } NOT = { has_claim_on = title:k_uuii_yanzhou } } add_pressed_claim = title:k_uuii_yanzhou }
		if = { limit = { NOT = { has_title = title:k_uuii_qingzhou } NOT = { has_claim_on = title:k_uuii_qingzhou } } add_pressed_claim = title:k_uuii_qingzhou }
		add_character_flag = hd_cao_yan_qing_claims_received
	}
	if = {
		limit = { this = character:yuan_shu NOT = { has_character_flag = hd_yuan_shu_story_activated } exists = title:d_runan.holder title:d_runan.holder = { OR = { this = scope:hd_historical_actor is_vassal_or_below_of = scope:hd_historical_actor } } }
		add_character_flag = hd_yuan_shu_story_activated
		add_character_modifier = { modifier = hd_yuan_shu_assertive_ai }
		random_owned_story = { type = hd_story_yuan_shu_life set_variable = { name = hd_yuan_shu_life_phase value = flag:hd_story_phase_runan_ambition } }
	}
	if = {
		limit = { this = character:liu_biao has_title = title:k_uuii_jingzhou NOT = { has_character_modifier = hd_liu_biao_stable_ai } }
		add_character_modifier = { modifier = hd_liu_biao_stable_ai }
		random_owned_story = { type = hd_story_liu_biao_life set_variable = { name = hd_liu_biao_life_phase value = flag:hd_story_phase_guarding_jingzhou } }
	}
	if = {
		limit = { is_ai = yes }
		if = { limit = { this = character:yuan_shao has_character_flag = hd_yuan_jizhou_claim_received random = { chance = 50 } } hd_yuan_shao_ai_war_effect = yes }
		else_if = { limit = { this = character:gong_sun_zan exists = global_var:hd_anti_regent_chain_resolved } hd_gongsun_zan_ai_war_effect = yes }
		else_if = { limit = { this = character:cao_cao exists = global_var:hd_anti_regent_chain_resolved } hd_cao_cao_ai_war_effect = yes }
		else_if = { limit = { this = character:sun_ce has_character_flag = hd_sun_ce_story_active } hd_sun_ce_ai_war_effect = yes }
		else_if = { limit = { this = character:wang_yun exists = global_var:hd_anti_regent_chain_resolved } hd_wang_yun_ai_scheme_effect = yes }
		else_if = { limit = { this = character:yuan_shu has_character_flag = hd_yuan_shu_story_activated } hd_yuan_shu_ai_effect = yes }
	}

	}
}

hd_historical_ai_can_start_very_high_war_trigger = {
	is_landed = yes
	is_independent_ruler = yes
	is_at_war = no
	war_chest_gold >= war_chest_gold_maximum
}

hd_historical_ai_can_start_extreme_war_trigger = {
	is_landed = yes
	is_independent_ruler = yes
	is_at_war = no
	war_chest_gold >= halved_ai_war_chest_gold_maximum
}

hd_yuan_shao_ai_war_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_yuan_jizhou_pending } hd_historical_ai_can_start_very_high_war_trigger = yes }
		# Hard first pass: only legal independent land-neighbors in the northern
		# theatre. If none of their wars are valid, fall back to the old pool.
		ai_start_best_war = {
			cb = { claim_cb duchy_conquest_cb vassalization_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
			is_valid = {
				scope:target_character = {
					is_independent_ruler = yes
					capital_province = { geographical_region = china_north_world }
					NOT = { this = character:cao_cao }
					any_neighboring_top_liege_realm_owner = {
						this = character:yuan_shao
					}
				}
			}
			on_failure = {
				ai_start_best_war = {
					cb = { claim_cb duchy_conquest_cb vassalization_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
					is_valid = { scope:target_character = { capital_province = { geographical_region = china_north_world } NOT = { this = character:cao_cao } } }
				}
			}
		}
	}
}

hd_cao_cao_ai_war_effect = {
	if = {
		limit = { hd_cao_fengying_can_declare_trigger = yes }
		hd_cao_fengying_start_war_effect = yes
	}
	if = {
		limit = { hd_historical_ai_can_start_extreme_war_trigger = yes }
		# Preserve the legacy generic pulse as a repair path, but give it the
		# same land-neighbor-first behavior as Cao's dedicated regional story.
		ai_start_best_war = {
			cb = { claim_cb duchy_conquest_cb vassalization_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
			is_valid = {
				scope:target_character = {
					is_independent_ruler = yes
					capital_province = { geographical_region = china_north_world }
					trigger_if = {
						limit = {
							NOT = { exists = global_var:hd_guandu_world_open }
							NOT = { exists = global_var:hd_guandu_world_active }
						}
						NOT = { this = character:yuan_shao }
					}
					any_neighboring_top_liege_realm_owner = {
						this = character:cao_cao
					}
				}
			}
			on_failure = {
				ai_start_best_war = {
					cb = { claim_cb duchy_conquest_cb vassalization_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
					is_valid = { scope:target_character = { capital_province = { geographical_region = china_north_world } trigger_if = { limit = { NOT = { exists = global_var:hd_guandu_world_open } NOT = { exists = global_var:hd_guandu_world_active } } NOT = { this = character:yuan_shao } } } }
				}
			}
		}
	}
}
