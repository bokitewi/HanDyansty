# Shared story heartbeat. Historical state applies to all owners; automatic
# schemes and wars are restricted to AI characters.
hd_historical_actor_story_tick_effect = {
	save_scope_as = root
	if = {
		limit = { this = character:gong_sun_zan exists = global_var:hd_anti_regent_chain_resolved NOT = { has_character_flag = hd_gongsun_youzhou_claim_received } }
		if = { limit = { NOT = { has_title = title:k_uuii_youzhou } NOT = { has_claim_on = title:k_uuii_youzhou } } add_pressed_claim = title:k_uuii_youzhou }
		add_character_flag = hd_gongsun_youzhou_claim_received
		random_owned_story = { type = hd_story_gongsun_zan_life set_variable = { name = hd_gongsun_zan_life_phase value = flag:hd_story_phase_northern_conquest } }
	}
	if = {
		limit = { this = character:cao_cao exists = global_var:hd_anti_regent_chain_resolved NOT = { has_character_flag = hd_cao_yan_qing_claims_received } }
		if = { limit = { NOT = { has_title = title:k_uuii_yanzhou } NOT = { has_claim_on = title:k_uuii_yanzhou } } add_pressed_claim = title:k_uuii_yanzhou }
		if = { limit = { NOT = { has_title = title:k_uuii_qingzhou } NOT = { has_claim_on = title:k_uuii_qingzhou } } add_pressed_claim = title:k_uuii_qingzhou }
		add_character_flag = hd_cao_yan_qing_claims_received
	}
	if = {
		limit = { this = character:yuan_shu NOT = { has_character_flag = hd_yuan_shu_story_activated } exists = title:d_runan.holder title:d_runan.holder = { OR = { this = root is_vassal_or_below_of = root } } }
		add_character_flag = hd_yuan_shu_story_activated
		add_character_modifier = { modifier = hd_yuan_shu_assertive_ai }
		random_owned_story = { type = hd_story_yuan_shu_life set_variable = { name = hd_yuan_shu_life_phase value = flag:hd_story_phase_runan_ambition } }
	}
	if = {
		limit = { this = character:liu_biao has_title = title:k_uuii_jingzhou NOT = { has_character_modifier = hd_liu_biao_stable_ai } }
		add_character_modifier = { modifier = hd_liu_biao_stable_ai }
		random_owned_story = { type = hd_story_liu_biao_life set_variable = { name = hd_liu_biao_life_phase value = flag:hd_story_phase_guarding_jingzhou } }
	}
	if = {
		limit = { is_ai = yes }
		if = { limit = { this = character:yuan_shao has_character_flag = hd_yuan_jizhou_claim_received random = { chance = 50 } } hd_yuan_shao_ai_war_effect = yes }
		else_if = { limit = { this = character:gong_sun_zan exists = global_var:hd_anti_regent_chain_resolved } hd_gongsun_zan_ai_war_effect = yes }
		else_if = { limit = { this = character:cao_cao exists = global_var:hd_anti_regent_chain_resolved } hd_cao_cao_ai_war_effect = yes }
		else_if = { limit = { this = character:liu_bei has_title = title:k_uuii_jingzhou } hd_liu_bei_ai_effect = yes }
		else_if = { limit = { this = character:sun_ce has_character_flag = hd_sun_ce_story_active } hd_sun_ce_ai_war_effect = yes }
		else_if = { limit = { this = character:wang_yun exists = global_var:hd_anti_regent_chain_resolved } hd_wang_yun_ai_scheme_effect = yes }
		else_if = { limit = { this = character:yuan_shu has_character_flag = hd_yuan_shu_story_activated } hd_yuan_shu_ai_effect = yes }
	}
}

hd_historical_ai_can_start_very_high_war_trigger = {
	is_landed = yes
	is_independent_ruler = yes
	is_at_war = no
	war_chest_gold >= war_chest_gold_maximum
}

hd_historical_ai_can_start_extreme_war_trigger = {
	is_landed = yes
	is_independent_ruler = yes
	is_at_war = no
	war_chest_gold >= halved_ai_war_chest_gold_maximum
}

hd_yuan_shao_ai_war_effect = {
	if = {
		limit = { hd_historical_ai_can_start_very_high_war_trigger = yes }
		ai_start_best_war = {
			cb = { claim_cb duchy_conquest_cb vassalization_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
			is_valid = { scope:target_character = { capital_province = { geographical_region = china_north_world } NOT = { this = character:cao_cao } } }
		}
	}
}

hd_wang_yun_ai_scheme_effect = {
	if = {
		limit = { character:lv_bu ?= { is_alive = yes } NOT = { any_scheme = { type = sway scheme_target_character = character:lv_bu } } can_start_scheme = { type = sway target_character = character:lv_bu } }
		start_scheme = { type = sway target_character = character:lv_bu }
		random_scheme = { type = sway limit = { scheme_target_character = character:lv_bu } add_scheme_modifier = { type = hd_wang_yun_historical_sway_scheme } }
	}
	if = {
		limit = { character:dong_zhuo_2 ?= { is_alive = yes } NOT = { any_scheme = { type = murder scheme_target_character = character:dong_zhuo_2 } } can_start_scheme = { type = murder target_character = character:dong_zhuo_2 } }
		start_scheme = { type = murder target_character = character:dong_zhuo_2 }
		random_scheme = { type = murder limit = { scheme_target_character = character:dong_zhuo_2 } add_scheme_modifier = { type = hd_wang_yun_historical_murder_scheme } }
	}
	random_owned_story = { type = hd_story_wang_yun_life set_variable = { name = hd_wang_yun_life_phase value = flag:hd_story_phase_plotting_dong_zhuo } }
}

hd_yuan_shu_ai_effect = {
	# Political title acquisition takes precedence over ordinary hostile schemes.
	if = {
		limit = { NOT = { any_scheme = { OR = { scheme_type = murder scheme_type = fabricate_hook scheme_type = claim_throne } } } }
		random_neighboring_and_across_water_top_liege_realm_owner = {
			limit = { is_alive = yes exists = primary_heir primary_heir = root root = { can_start_scheme = { type = murder target_character = prev } } }
			save_scope_as = hd_yuan_shu_inheritance_target
		}
	}
	if = {
		limit = { exists = liege liege = { is_alive = yes } can_start_scheme = { type = claim_throne target_character = liege } NOT = { any_scheme = { type = claim_throne } } }
		start_scheme = { type = claim_throne target_character = liege }
	}
	else_if = {
		limit = { exists = scope:hd_yuan_shu_inheritance_target }
		start_scheme = { type = murder target_character = scope:hd_yuan_shu_inheritance_target }
	}
	else_if = {
		limit = { NOT = { any_scheme = { OR = { scheme_type = murder scheme_type = fabricate_hook scheme_type = claim_throne } } } }
		random_vassal = {
			limit = { is_alive = yes highest_held_title_tier >= tier_county root = { can_start_scheme = { type = fabricate_hook target_character = prev } } }
			weight = { base = 1 modifier = { add = 100 is_powerful_vassal = yes } modifier = { add = 100 joined_faction ?= { faction_power >= faction_power_halfway_threshold } } }
			root = { start_scheme = { type = fabricate_hook target_character = prev } }
		}
	}
	# Sway weak neighboring rulers so the normal offer-vassalization AI can act.
	if = {
		limit = { NOT = { any_scheme = { type = sway } } }
		random_neighboring_and_across_water_top_liege_realm_owner = {
			limit = { highest_held_title_tier < root.highest_held_title_tier OR = { capital_county = { target_is_de_jure_liege_or_above = title:k_uuii_yuzhou } capital_county = { target_is_de_jure_liege_or_above = title:k_uuii_yangzhou } } root = { can_start_scheme = { type = sway target_character = prev } } }
			weight = { base = 1 modifier = { add = 100 current_military_strength < root.current_military_strength } }
			root = { start_scheme = { type = sway target_character = prev } }
		}
	}
	if = {
		limit = { hd_historical_ai_can_start_extreme_war_trigger = yes }
		ai_start_best_war = {
			cb = { vassalization_cb tribal_subjugation_cb sovereignty_cb }
			is_valid = { scope:target_character = { capital_county = { target_is_de_jure_liege_or_above = title:k_uuii_yuzhou } } }
			on_failure = {
				ai_start_best_war = {
					cb = { vassalization_cb tribal_subjugation_cb sovereignty_cb }
					is_valid = { scope:target_character = { capital_county = { target_is_de_jure_liege_or_above = title:k_uuii_yangzhou } } }
					on_failure = {
						ai_start_best_war = {
							cb = { claim_cb duchy_conquest_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
							is_valid = { scope:target_character = { capital_county = { target_is_de_jure_liege_or_above = title:k_uuii_yuzhou } } }
							on_failure = {
								ai_start_best_war = {
									cb = { claim_cb duchy_conquest_cb individual_duchy_de_jure_cb county_conquest_cb individual_county_de_jure_cb }
									is_valid = { scope:target_character = { capital_county = { target_is_de_jure_liege_or_above = title:k_uuii_yangzhou } } }
								}
							}
						}
					}
				}
			}
		}
	}
}
