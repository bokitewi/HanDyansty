# 开局人口清除：抽 7000 名候选人，各自在 1~90 天内随机一天无声消失。
# 新开局大厅结束后自动执行一次；之后玩家可通过决议手动再执行。
hd_popclean_run_effect = {
	hd_popclean_protect_history_effect = yes
	every_living_character = {
		limit = {
			hd_popclean_candidate_trigger = yes
			NOT = { has_character_flag = hd_popclean_marked }
			hd_popclean_tier_value > 0
		}
		# 随机数必须在效果里直接取（原版 grab_gold_fuzz_number_effect 同法）；放进具名脚本值里取不到随机。
		save_temporary_scope_value_as = {
			name = hd_popclean_roll
			value = { integer_range = { min = 0 max = 999 } }
		}
		set_variable = {
			name = hd_popclean_key
			value = { value = hd_popclean_tier_value multiply = 1000 add = scope:hd_popclean_roll }
			days = 1
		}
		add_to_list = hd_popclean_pool
	}
	# 候选人数不足 7000 时以实际人数为上限（max 大于列表会报错）。
	save_temporary_scope_value_as = {
		name = hd_popclean_pool_size
		value = { every_in_list = { list = hd_popclean_pool add = 1 } }
	}
	set_global_variable = { name = hd_popclean_target value = { value = 7000 max = scope:hd_popclean_pool_size } }
	set_global_variable = { name = hd_popclean_scheduled value = 0 }
	ordered_in_list = {
		list = hd_popclean_pool
		order_by = var:hd_popclean_key
		max = global_var:hd_popclean_target
		# 标记期间不得入议会（见 hd_council_start_open_trigger），免得 AI 把待清的人任命进去。
		add_character_flag = { flag = hd_popclean_marked days = 100 }
		trigger_event = { id = hd_popclean.1 days = { 1 90 } }
		change_global_variable = { name = hd_popclean_scheduled add = 1 }
	}
	every_player = {
		send_interface_message = {
			type = event_generic_neutral
			title = hd_popclean_message_title
			desc = hd_popclean_message_desc
		}
	}
}
