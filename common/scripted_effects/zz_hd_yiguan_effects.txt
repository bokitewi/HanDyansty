# 催化剂发射（§5.2）
# 任意作用域；$CATALYST$ 催化剂键，$CHAR$ 触发人（root / scope:x 等；不要传 this、prev）
# 局势存在、当前阶段的去向列了该催化剂、本年计次 < 上限 才发
# 写法先例：原版 war_on_actions.txt:1111-1117（先判 situation_top_has_catalyst 再带 character 触发）；
#           mod hd_dc_kit_effects.txt:34-48（局势变量计次）
############################################################
hd_yiguan_fire_catalyst_effect = {
	situation:hd_yiguan_nandu ?= {
		if = {
			limit = {
				situation_top_has_catalyst = $CATALYST$
				OR = {
					NOT = { has_variable = hd_yiguan_cc_$CATALYST$ }
					var:hd_yiguan_cc_$CATALYST$ < hd_yiguan_catalyst_yearly_cap
				}
			}
			if = {
				limit = { NOT = { has_variable = hd_yiguan_cc_$CATALYST$ } }
				set_variable = { name = hd_yiguan_cc_$CATALYST$ value = 0 }
			}
			change_variable = { name = hd_yiguan_cc_$CATALYST$ add = 1 }
			if = {
				limit = { exists = $CHAR$ }	# CHAR = root 时为 exists = root（先例：原版 00_commander_phase_events.txt:47）
				trigger_situation_catalyst = {
					catalyst = $CATALYST$
					character = $CHAR$
				}
			}
			else = {
				trigger_situation_catalyst = $CATALYST$
			}
		}
	}
}

############################################################
# 拖延催化剂发射（负值；作者 2026-10-26 定：冲突期兵事延缓停滞，停滞期汉化推迟动荡）
# 任意作用域；$CATALYST$ 拖延催化剂键（只登记在一个阶段的一个去向里），$TOWARD$ 该去向的阶段键，
# $VALUE$ 扣点绝对值，$CHAR$ 触发人。该去向已积点数不足 $VALUE$ 时不发（不扣成负数，先例：原版
# dynastic_cycle_on_actions.txt:225-227 先判 phase_takeover_points >= 0 再触发扣分催化剂）。
# 先判 situation_top_has_catalyst，保证 $TOWARD$ 是当前阶段的去向，再读点数（否则读点数会报错）。
############################################################
hd_yiguan_fire_delay_catalyst_effect = {
	situation:hd_yiguan_nandu ?= {
		if = {
			limit = {
				situation_top_has_catalyst = $CATALYST$
				OR = {
					NOT = { has_variable = hd_yiguan_cc_$CATALYST$ }
					var:hd_yiguan_cc_$CATALYST$ < hd_yiguan_catalyst_yearly_cap
				}
			}
			if = {
				limit = {
					situation_top_sub_region = { phase_takeover_points:$TOWARD$ >= $VALUE$ }
				}
				if = {
					limit = { NOT = { has_variable = hd_yiguan_cc_$CATALYST$ } }
					set_variable = { name = hd_yiguan_cc_$CATALYST$ value = 0 }
				}
				change_variable = { name = hd_yiguan_cc_$CATALYST$ add = 1 }
				if = {
					limit = { exists = $CHAR$ }
					trigger_situation_catalyst = {
						catalyst = $CATALYST$
						character = $CHAR$
					}
				}
				else = {
					trigger_situation_catalyst = $CATALYST$
				}
			}
		}
	}
}

# 开局判定（§1 第 9 轮、§6 第 8 条）：每月检查（无天朝循环时每年兜底），一局一次
# 洛阳、长安都已落入胡人；北方沦陷县 ≥180；胡人强权（独立、王国级以上、北方 ≥150 县）≥2 个，或其中有帝国级

hd_yiguan_try_start_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_yiguan_started }
			hd_yiguan_enabled_trigger = yes
			NOT = { exists = situation:hd_yiguan_nandu }
			hd_yiguan_two_capitals_lost_trigger = yes	# 最便宜的放前面：两次头衔查找
			OR = {
				any_independent_ruler = {
					highest_held_title_tier >= tier_empire
					any_held_title = {	# 官职等无地头衔不算（同 zz_hd_yiguan_triggers.txt 政权基础）
						tier >= tier_empire
						is_landless_type_title = no
					}
					hd_yiguan_hu_power_trigger = yes
				}
				any_independent_ruler = {
					count >= 2
					hd_yiguan_hu_power_trigger = yes
				}
			}
			any_county_in_region = {
				region = hd_yiguan_north_region
				count >= 180
				hd_yiguan_occupied_county_trigger = yes
			}
		}
		hd_yiguan_start_effect = yes
	}
}

# 逃亡期开局（常规触发与 304 剧本兜底共用，调用方已判条件）
hd_yiguan_start_effect = {
	set_global_variable = { name = hd_yiguan_started value = 1 }
	set_global_variable = { name = hd_hev_wuhu_yiguan value = 1 }
	if = {
		limit = { NOT = { exists = situation:hd_yiguan_nandu } }

		}

		hd_yiguan_notify_effect = { NOTICE = 1 }
	}
}

# 阶段更替（局势各阶段 on_start 调用，root = 局势；原版 _situations.info:244-249）
# $PHASE$ = taowang / chongtu / tingzhi，即刚进入的阶段
# 首个阶段（开局）不通告：那时还没有 hd_yiguan_cur_phase，或已由开局初始化记成同一阶段；
# 之后进入逃亡 / 冲突 / 停滞分别通告 0002 / 0003 / 0004

hd_yiguan_phase_started_effect = {
	if = {
		limit = {
			has_variable = hd_yiguan_cur_phase
			NOT = { var:hd_yiguan_cur_phase = flag:$PHASE$ }
		}
		set_variable = { name = hd_yiguan_cur_phase value = flag:$PHASE$ }
		hd_yiguan_phase_notice_effect = yes
	}
	else = {
		set_variable = { name = hd_yiguan_cur_phase value = flag:$PHASE$ }
	}
}

# 局势作用域。按刚记下的阶段发通告
hd_yiguan_phase_notice_effect = {
	if = {
		limit = { var:hd_yiguan_cur_phase = flag:taowang }
		hd_yiguan_notify_effect = { NOTICE = 2 }
	}
	else_if = {
		limit = { var:hd_yiguan_cur_phase = flag:chongtu }
		hd_yiguan_notify_effect = { NOTICE = 3 }
	}
	else_if = {
		limit = { var:hd_yiguan_cur_phase = flag:tingzhi }
		hd_yiguan_notify_effect = { NOTICE = 4 }
	}
}

# 局势作用域。按引擎的当前阶段记下（开局初始化用）
hd_yiguan_store_phase_effect = {
	if = {
		limit = { situation_current_phase = hd_yiguan_phase_taowang }
		set_variable = { name = hd_yiguan_cur_phase value = flag:taowang }
	}
	else_if = {
		limit = { situation_current_phase = hd_yiguan_phase_chongtu }
		set_variable = { name = hd_yiguan_cur_phase value = flag:chongtu }
	}
	else_if = {
		limit = { situation_current_phase = hd_yiguan_phase_tingzhi }
		set_variable = { name = hd_yiguan_cur_phase value = flag:tingzhi }
	}
}

		hd_yiguan_unmark_county_effect = yes
		hd_yiguan_clear_fading_county_effect = yes

# 县作用域。沦陷余殃：县已不满足沦陷条件（改为非汉文化、宗主汉化或归汉人）即去掉
hd_yiguan_clear_fading_county_effect = {
	if = {
		limit = {
			has_county_modifier = hd_yiguan_occupied_fading_county_modifier
			NOT = { hd_yiguan_occupied_county_trigger = yes }
		}
		remove_county_modifier = hd_yiguan_occupied_fading_county_modifier
	}
}

# 县作用域。局势进行中：沦陷则加标记与修正，否则去掉
hd_yiguan_mark_county_effect = {
	if = {
		limit = {
			hd_yiguan_county_in_north_trigger = yes
			hd_yiguan_occupied_county_trigger = yes
		}
		if = {
			limit = { NOT = { has_variable = hd_yiguan_occ } }
			set_variable = { name = hd_yiguan_occ value = 1 }
		}
		if = {
			limit = { NOT = { has_county_modifier = hd_yiguan_occupied_county_modifier } }
			add_county_modifier = { modifier = hd_yiguan_occupied_county_modifier }
		}
	}
	else = {
		hd_yiguan_unmark_county_effect = yes
	}
}

# 县作用域。去掉沦陷变量与常驻修正
hd_yiguan_unmark_county_effect = {
	if = {

hd_yiguan_heliu_acceptance_effect = {
	culture ?= { save_scope_as = hd_yiguan_heliu_culture }
	if = {
		limit = { exists = scope:hd_yiguan_heliu_culture }
		every_county_in_region = {
			region = hd_yiguan_north_region
			limit = {
				culture ?= {
					has_cultural_pillar = heritage_chinese
					NOT = { this = scope:hd_yiguan_heliu_culture }
				}
			}
			culture = {
				if = {
					limit = { NOT = { is_in_list = hd_yiguan_heliu_cultures } }
					add_to_list = hd_yiguan_heliu_cultures
				}
			}
		}
		every_in_list = {
			list = hd_yiguan_heliu_cultures
			change_cultural_acceptance = {
				target = scope:hd_yiguan_heliu_culture
				value = 50
				desc = hd_yiguan_heliu_acceptance_desc
			}
		}
	}
}

# 局势作用域
hd_yiguan_fallback_effect = {
	clear_saved_scope = hd_yiguan_declarer
	set_variable = { name = hd_yiguan_n_han value = hd_yiguan_north_han_counties_value }
	set_variable = { name = hd_yiguan_n_sin value = hd_yiguan_north_sin_counties_value }
	set_variable = { name = hd_yiguan_n_hu value = hd_yiguan_north_hu_counties_value }
	if = {
		limit = {
			var:hd_yiguan_n_han >= var:hd_yiguan_n_sin
			var:hd_yiguan_n_han >= var:hd_yiguan_n_hu
		}
		hd_yiguan_end_effect = { OUTCOME = guangfu }
	}
	else_if = {
		limit = { var:hd_yiguan_n_sin >= var:hd_yiguan_n_hu }
		hd_yiguan_end_effect = { OUTCOME = heliu }
	}
	else = {
		hd_yiguan_end_effect = { OUTCOME = lunsang }
	}
}

hd_yiguan_aftermath_yearly_effect = {
	if = {
		limit = {
			exists = global_var:hd_yiguan_fading_window
			NOT = { exists = situation:hd_yiguan_nandu }
		}
		every_county_in_region = {
			region = hd_yiguan_north_region
			limit = { has_county_modifier = hd_yiguan_occupied_fading_county_modifier }
			hd_yiguan_clear_fading_county_effect = yes
		}
	}
	if = {
		limit = {
			exists = global_var:hd_yiguan_outcome
			exists = global_var:hd_yiguan_tuduan_window
			NOT = { exists = situation:hd_yiguan_nandu }
			has_global_variable_list = hd_yiguan_qiaojun_list
		}
		every_in_global_list = {
			variable = hd_yiguan_qiaojun_list
			limit = { has_county_modifier = hd_yiguan_qiaojun_county_modifier }
			holder ?= {
				top_liege = {
					if = {
						limit = { NOT = { is_in_list = hd_yiguan_tuduan_lords } }
						add_to_list = hd_yiguan_tuduan_lords
					}
				}
			}
		}
		every_in_list = {
			list = hd_yiguan_tuduan_lords
			limit = {
				is_alive = yes
				NOT = { has_variable = hd_yiguan_event_cd }
			}
			random = {
				chance = 25
				trigger_event = hd_yiguan.1210
			}
		}
	}
}
