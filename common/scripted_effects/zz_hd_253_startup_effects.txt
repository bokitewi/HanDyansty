# Scope：受封者。granted 授予一个头衔（已在手则跳过）
hd253_grant_title_effect = {
	save_temporary_scope_as = hd253_grantee
	if = {
		limit = { NOT = { has_title = title:$TITLE$ } }
		create_title_and_vassal_change = { type = granted save_scope_as = hd253_grant_change add_claim_on_loss = no }
		title:$TITLE$ = {
			change_title_holder = { holder = scope:hd253_grantee change = scope:hd253_grant_change }
		}
		resolve_title_and_vassal_change = scope:hd253_grant_change
	}
}

# Scope：新主君。$WHO$ 改投本人
hd253_change_liege_effect = {
	save_temporary_scope_as = hd253_new_liege
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				NOT = { liege ?= scope:hd253_new_liege }
			}
			create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd253_liege_change add_claim_on_loss = no }
			change_liege = { liege = scope:hd253_new_liege change = scope:hd253_liege_change }
			resolve_title_and_vassal_change = scope:hd253_liege_change
		}
	}
}

hd253_ruler_governors_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_landed = yes exists = capital_county }
			save_temporary_scope_as = hd253_gov_ruler
			every_held_title = {
				limit = { hd253_governor_duchy_trigger = { RULER = scope:hd253_gov_ruler } }
				add_to_temporary_list = hd253_gov_duchies_$WHO$
			}
			every_in_list = {
				list = hd253_gov_duchies_$WHO$
				hd253_make_governor_effect = { RULER = scope:hd253_gov_ruler }
			}
			debug_log = "HD253 GOVERNORS DONE $WHO$"
		}
	}
}

# Scope：郡。新郡守作 $RULER$ 的廷臣生成，再受封成封臣
hd253_make_governor_effect = {
	save_temporary_scope_as = seceding_county
	create_character = {
		template = post_migration_minor_warlord_template
		employer = $RULER$
		dynasty = generate # 天朝政体要求君主有王朝
		save_temporary_scope_as = hd253_gov_new
	}
	create_title_and_vassal_change = { type = granted save_scope_as = hd253_gov_change add_claim_on_loss = no }
	every_in_de_jure_hierarchy = {
		limit = {
			tier = tier_county
			holder ?= $RULER$
		}
		change_title_holder = { holder = scope:hd253_gov_new change = scope:hd253_gov_change take_baronies = yes }
	}
	change_title_holder = { holder = scope:hd253_gov_new change = scope:hd253_gov_change }
	resolve_title_and_vassal_change = scope:hd253_gov_change
	scope:hd253_gov_new = {
		if = {
			limit = { NOT = { liege ?= $RULER$ } }
			create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd253_gov_fix_change add_claim_on_loss = no }
			change_liege = { liege = $RULER$ change = scope:hd253_gov_fix_change }
			resolve_title_and_vassal_change = scope:hd253_gov_fix_change
		}
		if = {
			limit = { NOT = { has_government = celestial_government } }
			change_government = celestial_government
		}
	}
}
