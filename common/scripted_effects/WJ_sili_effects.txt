# Invalid incumbent only. New Sili is selected by player / native contract AI.
WJ_sili_reconcile_character_effect = {
	if = {
		limit = {
			is_alive = yes
			vassal_contract_has_flag = WJ_celestial_province_sili
			NOT = { WJ_sili_eligible_trigger = yes }
		}
		vassal_contract_set_obligation_level = { type = celestial_provinces level = 3 }
		if = {
			limit = {
				government_has_flag = government_is_celestial
				NOT = { has_realm_law = celestial_military_appointment_succession_law }
			}
			add_realm_law_skip_effects = celestial_military_appointment_succession_law
		}
	}
}

# Only after relevant title/liege/capital/government/law changes, never a pulse.
WJ_sili_reconcile_subrealm_effect = {
	WJ_sili_reconcile_character_effect = yes
	every_vassal_or_below = {
		limit = { vassal_contract_has_flag = WJ_celestial_province_sili }
		WJ_sili_reconcile_character_effect = yes
	}
}
