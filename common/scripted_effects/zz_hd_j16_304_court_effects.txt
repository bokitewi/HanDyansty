# 304 专用权臣换人（02 §1.4）。Scope：新权臣。
# 旧权臣一律取 title:h_china.var:hd_court_regent（在世与否都可），调用方不必另存 scope:hd_j16_old_regent（02 原写法，自拟改为自取）。
# 与 hd_court_takeover_effect 的根本区别：不碰 e_han、天子直辖地与天子其他封臣，不改京城（京城由调用方另调 hd_j16_set_court_capital_effect）
############################################################
hd_j16_swap_regent_effect = {
	save_scope_as = hd_j16_sw_new
	if = {
		limit = { exists = scope:hd_j16_sw_old }
		clear_saved_scope = hd_j16_sw_old
	}
	if = {
		limit = {
			exists = title:h_china.var:hd_court_regent
			NOT = { title:h_china.var:hd_court_regent = scope:hd_j16_sw_new }
		}
		title:h_china.var:hd_court_regent = { save_scope_as = hd_j16_sw_old }
	}
	# 1 旧权臣罢官（解双头、免议员、销毁丞相等无地帝国级头衔）
	if = {
		limit = { scope:hd_j16_sw_old ?= { is_alive = yes } }
		scope:hd_j16_sw_old = { hd_court_strip_office_effect = yes }
	}
	# 2 清 hd_court 记录与权臣施加的傀儡
	hd_court_release_effect = yes
	# 3 新权臣改为天子封臣
	scope:hd_j16_sw_new = { hd_j16_join_court_effect = yes }
	# 4 装权臣（不得用 takeover）
	if = {
		limit = {
			scope:hd_j16_sw_new = {
				liege ?= title:h_china.holder
				hd_court_regent_eligible_trigger = yes
			}
		}
		scope:hd_j16_sw_new = {
			hd_court_install_regency_effect = yes
			add_character_flag = { flag = hd_court_regent_tenure years = 1 }
		}
		debug_log = "HD_J16 bawang FIRE swap_regent"
	}
	else = {
		debug_log = "HD_J16 bawang WAIT swap_regent_failed"
	}
	# 5 旧权臣原为朝贡王：罢归藩国（A2-F9★A）
	scope:hd_j16_sw_old ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				has_variable = hd_j16_was_tributary
			}
			hd_j16_return_fief_effect = yes
		}
	}
}

# 开局颙保持独立朝贡国（hd_j16_304_bawang_setup_effect 已不再改封臣、不装权臣）；B1 挟驾西迁执行时调 hd_j16_yong_enter_court_effect：
#   ① 入朝：改投天子、装为权臣（原 304 开局 #1/#2/#4/#5 的写法，搬到这里）；
#   ② 实封公：照 395 封宋公（hd_j16_songgong_exec_effect）新建王国级动态头衔「河间」（wj_rf_rank 1，名干 0 号走无爵名），
#      关中四郡（京兆、扶风、始平、咸阳）逐郡 WJ_rf_transfer_duchy 并入，国都长安；开局的封爵王爵由 WJ_peer_clear 清掉（实封取代原爵，照 WJ_rf_commit）；
#   ③ 进王：照 395 进宋王（hd_j16_songwang_exec_effect）新建帝国级「河间」（wj_rf_rank 2）套住公国。跳过 WJ_rf_can_promote 的门槛。
# 指针 global_var:hd_j16_yong_fief / hd_j16_yong_wang。罢归藩国（hd_j16_return_fief_effect）时先销毁这两个头衔并补回封爵王爵，见 hd_j16_yong_rf_dissolve_effect。
# 执行失败（颙失地、不是天子封臣等）只记日志不设指针，颙保留开局的封爵王爵。日志 HD_J16 xiqian FIRE/WAIT yong_*。

hd_j16_yong_enter_court_effect = {
	save_scope_as = hd_j16_yong_ec
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			exists = title:h_china.holder
			NOT = { this = title:h_china.holder }
		}
		# 开局到西迁之间已有别的宗王经「入朝辅政」决议当上权臣：照决议同一分支换权臣（旧权臣罢官，原朝贡王罢归）
		if = {
			limit = {
				exists = title:h_china.var:hd_court_regent
				NOT = { title:h_china.var:hd_court_regent = this }
			}
			hd_j16_swap_regent_effect = yes
			debug_log = "HD_J16 xiqian FIRE yong_swap_regent"
		}
		# 颙已是权臣：不再入朝；否则改投天子、装权臣
		else_if = {
			limit = { NOT = { title:h_china.var:hd_court_regent ?= this } }
			if = {
				limit = { NOT = { liege ?= title:h_china.holder } }
				hd_j16_join_court_effect = yes
			}
			if = {
				limit = {
					liege ?= title:h_china.holder
					hd_court_regent_eligible_trigger = yes
				}
				hd_court_install_regency_effect = yes
				add_character_flag = { flag = hd_court_regent_tenure years = 2 }
				debug_log = "HD_J16 xiqian FIRE yong_regent"
			}
			else = {
				debug_log = "HD_J16 xiqian WAIT yong_regent_failed"
			}
		}
	}
	hd_j16_yong_rf_effect = yes
	# 权臣主头衔记录跟着新主头衔走（照 263 S5 / 395 封宋公），免 hd_court 权臣交接分支交错人
	if = {
		limit = { title:h_china.var:hd_court_regent ?= this }
		title:h_china = { set_variable = { name = hd_court_regent_title value = scope:hd_j16_yong_ec.primary_title } }
	}
}

# 实封公 + 进王。Scope：颙。已有指针、颙失地、不是天子封臣时不做
hd_j16_yong_rf_effect = {
	hidden_effect = { # 选项预览解析不了执行时才建的头衔（同 263 S5 / S6）
		if = {
			limit = {
				NOT = { exists = global_var:hd_j16_yong_fief }
				exists = title:h_china.holder
				is_alive = yes
				is_landed = yes
				NOT = { this = top_liege }
				top_liege = title:h_china.holder
			}
			save_scope_as = wj_rf_beneficiary
			title:h_china.holder = { save_scope_as = wj_rf_sovereign }
			title:d_jingzhao = {
				save_scope_as = wj_rf_capital_duchy
				title_capital_county = { save_scope_as = wj_rf_capital_county }
			}
			create_dynamic_title = { tier = kingdom name = hd_j16_hejian_fief_name }
			scope:new_title = {
				save_scope_as = wj_rf_new_fief
				set_variable = { name = wj_rf_rank value = 1 }
				set_definitive_form = yes
				set_variable = { name = wj_rf_name_code value = 0 }    # 「河间」不在 WJ 1–13 号词干里：0 号走 wj_rf_stem_capital 的无爵名
				set_variable = { name = wj_rf_stem_capital value = scope:wj_rf_new_fief }
				set_capital_county = scope:wj_rf_capital_county
				set_coa = title:d_jingzhao
				set_color_from_title = title:d_jingzhao
			}
			set_global_variable = { name = hd_j16_yong_fief value = scope:wj_rf_new_fief }
			create_title_and_vassal_change = { type = granted save_scope_as = wj_rf_change add_claim_on_loss = no }
			scope:wj_rf_new_fief = {
				change_title_holder = {
					holder = scope:wj_rf_beneficiary
					change = scope:wj_rf_change
					government_base = scope:wj_rf_sovereign
				}
			}
			hd_j16_yong_rf_duchy_effect = { DUCHY = d_jingzhao }
			hd_j16_yong_rf_duchy_effect = { DUCHY = d_fufeng }
			hd_j16_yong_rf_duchy_effect = { DUCHY = d_shiping }
			hd_j16_yong_rf_duchy_effect = { DUCHY = d_xianyang }
			# 国都长安县交颙（照 WJ_rf_commit；玩家持有不夺）
			scope:wj_rf_capital_county = {
				if = {
					limit = {
						NOT = { holder ?= scope:wj_rf_beneficiary }
						OR = {
							NOT = { exists = holder }
							holder ?= {
								top_liege = scope:wj_rf_sovereign
								is_ai = yes
							}
						}
					}
					change_title_holder = {
						holder = scope:wj_rf_beneficiary
						change = scope:wj_rf_change
						government_base = scope:wj_rf_sovereign
					}
				}
				set_variable = { name = wj_rf_capital_of value = scope:wj_rf_new_fief }
			}
			resolve_title_and_vassal_change = scope:wj_rf_change
			if = {
				limit = { scope:wj_rf_new_fief.holder ?= this }
				if = {
					limit = { has_variable = wj_peer_rank }
					WJ_peer_clear = yes    # 照 WJ_rf_commit：实封公取代原爵（开局的封爵王爵）
				}
				if = {
					limit = { scope:wj_rf_capital_county.holder ?= this }
					set_realm_capital = scope:wj_rf_capital_county
				}
				set_primary_title_to = scope:wj_rf_new_fief
				set_variable = { name = wj_rf_cooldown value = yes years = 3 }
				WJ_rf_refresh_holder = yes
				scope:wj_rf_new_fief = { save_scope_as = wj_rf_law_title }
				trigger_event = WJ_rf.4
				debug_log = "HD_J16 xiqian FIRE yong_fief"
				# 进王延后 1 天（on_action hd_j16_yong_wang_delayed）：公、王不在同一次执行里连建，和 263 S5/S6、395 J84/J88 的分两次执行一致，免 saved scope 串号
				trigger_event = { on_action = hd_j16_yong_wang_delayed days = 1 }
			}
			else = {
				debug_log = "HD_J16 xiqian WAIT yong_fief_title_bad"
			}
		}
		else = {
			debug_log = "HD_J16 xiqian WAIT yong_rf_not_ready"
		}
	}
}

# 关中一郡并入河间国：郡无人持有或持有者是颙本人、天子一系的 AI 才动（同 395 封宋公）
hd_j16_yong_rf_duchy_effect = {
	title:$DUCHY$ = {
		if = {
			limit = {
				OR = {
					holder ?= scope:wj_rf_beneficiary
					holder ?= {
						top_liege = scope:wj_rf_sovereign
						is_ai = yes
					}
					AND = {
						NOT = { exists = holder }
						any_de_jure_county = { holder ?= { top_liege = scope:wj_rf_sovereign } }
					}
				}
			}
			WJ_rf_transfer_duchy = yes
		}
	}
}

# 进王：新建帝国级「河间」套住河间国（同 395 进宋王）。Scope：颙；scope:wj_rf_new_fief 等由调用方存好，这里重存
hd_j16_yong_rf_wang_effect = {
	if = {
		limit = {
			exists = global_var:hd_j16_yong_fief
			has_title = global_var:hd_j16_yong_fief
			NOT = { exists = global_var:hd_j16_yong_wang }
			top_liege = {
				has_title = title:h_china
				NOT = { this = prev }
			}
		}
		save_scope_as = wj_rf_beneficiary
		top_liege = { save_scope_as = wj_rf_sovereign }
		global_var:hd_j16_yong_fief = { save_scope_as = wj_rf_new_fief }
		create_dynamic_title = { tier = empire name = hd_j16_hejian_fief_name }
		scope:new_title = {
			save_scope_as = wj_rf_new_kingdom
			set_variable = { name = wj_rf_rank value = 2 }
			set_variable = { name = wj_rf_constituent_fief value = scope:wj_rf_new_fief }
			set_definitive_form = yes
			set_variable = { name = wj_rf_name_code value = 0 }
			set_variable = { name = wj_rf_stem_capital value = scope:wj_rf_new_fief }
			set_capital_county = scope:wj_rf_new_fief.title_capital_county
			set_coa = scope:wj_rf_new_fief
			set_color_from_title = scope:wj_rf_new_fief
		}
		create_title_and_vassal_change = { type = granted save_scope_as = wj_rf_change add_claim_on_loss = no }
		scope:wj_rf_new_kingdom = { change_title_holder = { holder = scope:wj_rf_beneficiary change = scope:wj_rf_change } }
		change_liege = { liege = scope:wj_rf_sovereign change = scope:wj_rf_change }
		resolve_title_and_vassal_change = scope:wj_rf_change
		if = {
			limit = { scope:wj_rf_new_kingdom.holder ?= scope:wj_rf_beneficiary }
			set_global_variable = { name = hd_j16_yong_wang value = scope:wj_rf_new_kingdom }
			scope:wj_rf_new_fief = { set_de_jure_liege_title = scope:wj_rf_new_kingdom }
			set_primary_title_to = scope:wj_rf_new_kingdom
			WJ_rf_refresh_holder = yes
			scope:wj_rf_new_kingdom = { save_scope_as = wj_rf_law_title }
			scope:wj_rf_new_fief.title_capital_county = { save_scope_as = wj_rf_capital_county }
			trigger_event = WJ_rf.4
			if = {
				limit = { title:h_china.var:hd_court_regent ?= this }
				title:h_china = { set_variable = { name = hd_court_regent_title value = scope:wj_rf_new_kingdom } }
			}
			debug_log = "HD_J16 xiqian FIRE yong_wang"
		}
		else = {
			debug_log = "HD_J16 xiqian WAIT yong_wang_title_bad"
		}
	}
}

# 罢归藩国前销毁河间王国、河间国（照 hd263_jin_dissolve_effect：先王国后公国，法理由 WJ_rf_destroyed 钩子还原），
# 再补回开局的封爵王爵（河间王，取河间郡首县）。Scope：持有河间国的人（颙，或继承了河间国的人；非颙时只销毁不补爵）
hd_j16_yong_rf_dissolve_effect = {
	save_scope_as = hd_j16_yong_dis
	global_var:hd_j16_yong_wang ?= {
		if = {
			limit = { holder ?= scope:hd_j16_yong_dis }
			scope:hd_j16_yong_dis = { destroy_title = global_var:hd_j16_yong_wang }
		}
	}
	global_var:hd_j16_yong_fief ?= {
		if = {
			limit = { holder ?= scope:hd_j16_yong_dis }
			scope:hd_j16_yong_dis = { destroy_title = global_var:hd_j16_yong_fief }
		}
	}
	remove_global_variable = hd_j16_yong_wang
	remove_global_variable = hd_j16_yong_fief
	WJ_rf_refresh_holder = yes
	if = {
		limit = {
			this = character:si_ma_yong
			exists = title:h_china.holder
		}
		title:h_china.holder = { save_scope_as = hd304_peer_issuer }
		hd304_peer_wang_effect = { WHO = si_ma_yong NAME = d_hejian }
		clear_saved_scope = hd304_peer_issuer
	}
	debug_log = "HD_J16 bawang FIRE yong_rf_dissolved"
}
