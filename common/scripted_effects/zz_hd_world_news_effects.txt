# 大历史事件全局通报（events/zz_hd_world_news_events.txt）。
# 调用前存好 scope:hd_news_a 主事人、scope:hd_news_b 对手（无对手时同 a）、scope:hd_news_place 事发地。
# 已有专属弹窗的人不再重复收到：按情况选下面三个之一。

# 除主事人外所有玩家
hd_world_news_send_effect = {
	every_player = {
		limit = { NOT = { this = scope:hd_news_a } }
		trigger_event = $EVENT$
	}
}

# 除主事人和对手外所有玩家
hd_world_news_send_skip_both_effect = {
	every_player = {
		limit = {
			NOT = { this = scope:hd_news_a }
			NOT = { this = scope:hd_news_b }
		}
		trigger_event = $EVENT$
	}
}

# 所有玩家
hd_world_news_send_all_effect = {
	every_player = { trigger_event = $EVENT$ }
}

# 史实叛乱（REV-01～09）：叛军首领与被攻打的一方不收，守方另有专属弹窗
hd_world_news_revolt_effect = {
	scope:hd_revolt_leader = { save_scope_as = hd_news_a }
	scope:hd_revolt_defender = { save_scope_as = hd_news_b }
	scope:hd_revolt_spawn_province = { save_scope_as = hd_news_place }
	hd_world_news_send_skip_both_effect = { EVENT = hd_world_news.0019 }
}
