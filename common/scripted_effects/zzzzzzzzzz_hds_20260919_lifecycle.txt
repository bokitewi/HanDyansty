hds_best_friend_setup_effect = {
    if = {
        limit = {
            NOT = { exists = global_var:hds_sc_zhou_friendship_done }
            exists = character:sun_ce
            exists = character:zhou_yu
            character:sun_ce ?= { is_alive = yes }
            character:zhou_yu ?= { is_alive = yes }
        }
        character:sun_ce ?= {
            if = {
                limit = { is_alive = yes NOT = { has_relation_best_friend = character:zhou_yu } }
                set_relation_best_friend = character:zhou_yu
            }
            if = {
                limit = { is_alive = yes has_relation_best_friend = character:zhou_yu }
                set_global_variable = hds_sc_zhou_friendship_done
            }
        }
    }
}
hds_sun_snapshot_living_effect = {
    character:sun_jian ?= {
        if = {
            limit = { is_alive = yes exists = domicile any_held_title = { is_noble_family_title = yes } }
            random_held_title = {
                limit = { is_noble_family_title = yes }
                save_scope_as = hds_original_family
                set_destroy_if_invalid_heir = no
                set_destroy_on_succession = no
            }
            domicile = { save_scope_as = hds_original_estate }
            title:h_china = {
                if = {
                    limit = { NOT = { has_variable = hds_sun_family_title } }
                    set_variable = { name = hds_sun_family_title value = scope:hds_original_family }
                }
                if = {
                    limit = { NOT = { has_variable = hds_sun_original_estate } }
                    set_variable = { name = hds_sun_original_estate value = scope:hds_original_estate }
                }
            }
            if = {
                limit = { exists = character:sun_ce character:sun_ce ?= { is_alive = yes } }
                set_designated_heir_unsafe = character:sun_ce
            }
        }
    }
}
# Repair missing old-save cache from Sun Ce's actual family/estate, never from an adventurer camp.
hds_nr_recover_sun_cache_effect = {
    if = {
        limit = {
            exists = character:sun_jian
            character:sun_jian ?= { is_alive = no }
            exists = character:sun_ce
            character:sun_ce ?= { is_alive = yes exists = house }
            OR = {
                NOT = { exists = title:h_china.var:hds_sun_family_title }
                NOT = { exists = title:h_china.var:hds_sun_original_estate }
            }
        }
        character:sun_ce ?= {
            if = {
                limit = {
                    is_alive = yes
                    NOT = { exists = title:h_china.var:hds_sun_family_title }
                    NOT = { exists = title:h_china.var:hds_sun_original_estate }
                }
                hds_ensure_estate_effect = yes
            }
            if = {
                limit = { is_alive = yes exists = domicile any_held_title = { is_noble_family_title = yes } }
                random_held_title = {
                    limit = { is_noble_family_title = yes }
                    save_scope_as = hds_nr_recovered_family
                }
                domicile = { save_scope_as = hds_nr_recovered_estate }
                title:h_china = {
                    if = {
                        limit = { NOT = { exists = var:hds_sun_family_title } }
                        set_variable = { name = hds_sun_family_title value = scope:hds_nr_recovered_family }
                    }
                    if = {
                        limit = {
                            NOT = { exists = var:hds_sun_original_estate }
                            var:hds_sun_family_title = scope:hds_nr_recovered_family
                        }
                        set_variable = { name = hds_sun_original_estate value = scope:hds_nr_recovered_estate }
                    }
                }
            }
        }
    }
}
hds_sun_succession_effect = {
    hds_nr_recover_sun_cache_effect = yes
    if = {
        limit = {
            exists = character:sun_jian
            character:sun_jian ?= { is_alive = no }
            exists = character:sun_ce
            character:sun_ce ?= { is_alive = yes NOT = { has_character_flag = hds_sc_inherited } }
            exists = title:h_china.var:hds_sun_family_title
            exists = title:h_china.var:hds_sun_original_estate
        }
        character:sun_ce ?= {
            if = {
                limit = { is_alive = yes }
                # Monotonic title receipt: estate attachment retries must not grant titles again.
                if = {
                    limit = { NOT = { has_character_flag = hds_sc_titles_inherited } }
                    if = {
                        limit = { exists = title:h_china.var:hds_sun_house }
                        title:h_china.var:hds_sun_house = { set_house_head = character:sun_ce }
                    }
                    create_title_and_vassal_change = {
                        type = granted save_scope_as = hds_sun_inheritance_change add_claim_on_loss = no
                    }
                    title:h_china = {
                        if = {
                            limit = { has_variable_list = hds_sun_death_titles }
                            every_in_list = {
                                variable = hds_sun_death_titles
                                limit = { NOT = { holder = character:sun_ce } }
                                change_title_holder = {
                                    holder = character:sun_ce change = scope:hds_sun_inheritance_change take_baronies = yes
                                }
                            }
                        }
                        var:hds_sun_family_title = {
                            if = {
                                limit = { NOT = { holder = character:sun_ce } }
                                change_title_holder = { holder = character:sun_ce change = scope:hds_sun_inheritance_change }
                            }
                        }
                    }
                    resolve_title_and_vassal_change = scope:hds_sun_inheritance_change
                    if = {
                        limit = {
                            title:h_china.var:hds_sun_family_title = { holder = character:sun_ce }
                            title:h_china = {
                                NOT = {
                                    any_in_list = {
                                        variable = hds_sun_death_titles
                                        NOT = { holder = character:sun_ce }
                                    }
                                }
                            }
                        }
                        add_character_flag = hds_sc_titles_inherited
                    }
                }
                if = {
                    limit = {
                        exists = title:h_china.var:hds_sun_inheritance_seal
                        title:h_china.var:hds_sun_inheritance_seal = { NOT = { owner = character:sun_ce } }
                    }
                    title:h_china.var:hds_sun_inheritance_seal = { set_owner = character:sun_ce }
                }
                # The seal is optional: losing/destroying an artifact cannot block the estate story.
                if = {
                    limit = {
                        hds_sun_estate_inherited_trigger = yes
                        has_character_flag = hds_sc_titles_inherited
                    }
                    add_character_flag = hds_sc_inherited
                    title:h_china.var:hds_sun_family_title = { set_destroy_if_invalid_heir = yes }
                    hds_retire_legacy_controller_effect = yes
                    if = {
                        limit = { NOT = { has_character_flag = hds_sc_inheritance_notice } }
                        hds_queue_event_effect = { EVENT = hds.4000 LEASE = hds_lease_4000 }
                    }
                }
            }
        }
    }
}
