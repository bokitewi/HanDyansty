hd4_tao_tick_effect = {
	if = {
		limit = {
			exists = global_var:hd_anti_regent_chain_resolved
		}
		if = {
			limit = {
				NOT = {
					has_character_flag = hd4_tao_activated
				}
			}
			hd4_tao_establish_effect = yes
		}
		if = {
			limit = {
				has_character_flag = hd4_tao_activated
			}
			hd4_tao_danyang_effect = yes
			hd4_tao_zang_ba_effect = yes
			if = {
				limit = {
					NOT = {
						has_character_flag = hd4_tao_complete
					}
					NOT = {
						has_character_flag = hd4_tao_split_grace
					}
				}
				hd4_tao_create_owned_duchies_effect = yes
				if = {
					limit = {
						hd4_tao_complete_trigger = yes
					}
					hd4_tao_finish_effect = yes
				}
				else_if = {
					limit = {
						is_ai = yes
						hd4_tao_can_campaign_trigger = yes
					}
					hd4_tao_try_war_effect = yes
				}
			}
			else_if = {
				limit = {
					has_character_flag = hd4_tao_complete
				}
				add_character_flag = sworn_peace
				if = {
					limit = {
						NOT = {
							has_character_modifier = hd4_tao_guarding_xuzhou
						}
					}
					add_character_modifier = {
						modifier = hd4_tao_guarding_xuzhou
					}
				}
			}
		}
	}
}
hd4_tao_establish_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = title:k_uuii_xuzhou.title_capital_county
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd4_tao_setup_change
			add_claim_on_loss = no
		}
		if = {
			limit = {
				NOT = {
					any_held_title = {
						tier = tier_county
						is_landless_type_title = no
					}
				}
			}
			title:k_uuii_xuzhou.title_capital_county = {
				change_title_holder = {
					holder = character:tao_qian
					change = scope:hd4_tao_setup_change
					take_baronies = yes
				}
			}
		}
		title:k_uuii_xuzhou = {
			change_title_holder = {
				holder = character:tao_qian
				change = scope:hd4_tao_setup_change
			}
		}
		resolve_title_and_vassal_change = scope:hd4_tao_setup_change
		hd4_release_character_effect = yes
		if = {
			limit = {
				has_title = title:k_uuii_xuzhou
				is_landed = yes
				is_independent_ruler = yes
			}
			add_character_flag = hd4_tao_activated
			add_character_flag = {
				flag = hd4_tao_split_grace
				days = 61
			}
			add_character_flag = sworn_peace
			add_character_modifier = {
				modifier = hd4_tao_story_ai
			}
			add_character_flag = hd4_tao_qxzl_protected
			add_character_flag = hd_qxzl_seen
			remove_character_flag = hd_qxzl_pending
			if = {
				limit = {
					NOT = {
						any_owned_story = {
							story_type = hd4_story_tao
						}
					}
				}
				create_story = hd4_story_tao
			}
			trigger_event = hd4_tao.0001
		}
	}
}
hd4_tao_danyang_effect = {
	if = {
		limit = {
			is_landed = yes
			exists = capital_province
			NOT = {
				has_character_flag = hd4_tao_danyang_paid
			}
		}
		add_character_flag = hd4_tao_danyang_paid
		spawn_army = {
			name = hd4_tao_danyang_army
			men_at_arms = {
				type = hd_danyang_jingbing
				men = 3000
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
}
hd4_tao_zang_ba_effect = {
	if = {
		limit = {
			NOT = {
				has_character_flag = hd4_tao_zang_resolved
			}
			exists = global_var:hd_qxzl_scan_done
			AND = {
				exists = character:zang_ba
				character:zang_ba = {
					is_alive = yes
				}
			}
		}
		character:zang_ba = {
			if = {
				limit = {
					is_landed = yes
					is_independent_ruler = yes
					highest_held_title_tier < character:tao_qian.highest_held_title_tier
				}
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = hd4_zang_change
					add_claim_on_loss = no
				}
				change_liege = {
					liege = character:tao_qian
					change = scope:hd4_zang_change
				}
				resolve_title_and_vassal_change = scope:hd4_zang_change
			}
			if = {
				limit = {
					is_vassal_or_below_of = character:tao_qian
				}
				add_character_flag = hd4_tao_qxzl_protected
				add_character_flag = hd_qxzl_seen
				remove_character_flag = hd_qxzl_pending
				character:tao_qian = {
					add_character_flag = hd4_tao_zang_resolved
					trigger_event = hd4_tao.0002
				}
			}
		}
	}
}
hd4_tao_create_owned_duchies_effect = {
	title:k_uuii_xuzhou = {
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_duchy
				NOT = {
					AND = {
						exists = holder
						holder = {
							OR = {
								this = character:tao_qian
								is_vassal_or_below_of = character:tao_qian
							}
						}
					}
				}
				any_de_jure_county = {
					always = yes
				}
				NOT = {
					any_de_jure_county = {
						NOT = {
							AND = {
								exists = holder
								holder = {
									OR = {
										this = character:tao_qian
										is_vassal_or_below_of = character:tao_qian
									}
								}
							}
						}
					}
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = hd4_tao_empty_duchy
				add_claim_on_loss = no
			}
			change_title_holder = {
				holder = character:tao_qian
				change = scope:hd4_tao_empty_duchy
			}
			resolve_title_and_vassal_change = scope:hd4_tao_empty_duchy
		}
	}
}
hd4_tao_finish_effect = {
	add_character_flag = hd4_tao_complete
	add_character_flag = sworn_peace
	add_character_modifier = {
		modifier = hd4_tao_guarding_xuzhou
	}
	trigger_event = hd4_tao.0003
}
hd4_tao_try_war_effect = {
	remove_character_flag = sworn_peace
	clear_saved_scope = hd4_tao_target
	clear_saved_scope = hd4_tao_enemy
	title:k_uuii_xuzhou = {
		random_in_de_jure_hierarchy = {
			limit = {
				hd4_tao_target_duchy_trigger = yes
				any_de_jure_county = {
					AND = {
						exists = holder
						holder = {
							top_liege = {
								is_alive = yes
								is_landed = yes
								is_independent_ruler = yes
								NOT = {
									this = character:tao_qian
								}
								current_military_strength <= hd4_tao_strength_ceiling
								NOT = {
									is_allied_to = character:tao_qian
								}
								NOT = {
									is_at_war_with = character:tao_qian
								}
								character:tao_qian = {
									NOT = {
										has_truce = prev
									}
								}
							}
						}
					}
				}
			}
			save_scope_as = hd4_tao_target
		}
	}
	if = {
		limit = {
			exists = scope:hd4_tao_target
		}
		scope:hd4_tao_target = {
			random_de_jure_county = {
				limit = {
					AND = {
						exists = holder
						holder = {
							top_liege = {
								is_alive = yes
								is_landed = yes
								is_independent_ruler = yes
								NOT = {
									this = character:tao_qian
								}
								current_military_strength <= hd4_tao_strength_ceiling
								NOT = {
									is_allied_to = character:tao_qian
								}
								NOT = {
									is_at_war_with = character:tao_qian
								}
								character:tao_qian = {
									NOT = {
										has_truce = prev
									}
									can_declare_war = {
										defender = prev
										casus_belli = hd4_tao_xuzhou_cb
										target_titles = {
											scope:hd4_tao_target
										}
									}
								}
							}
						}
					}
				}
				holder.top_liege = {
					save_scope_as = hd4_tao_enemy
				}
			}
		}
	}
	if = {
		limit = {
			exists = scope:hd4_tao_enemy
			hd4_tao_can_campaign_trigger = yes
			can_declare_war = {
				defender = scope:hd4_tao_enemy
				casus_belli = hd4_tao_xuzhou_cb
				target_titles = {
					scope:hd4_tao_target
				}
			}
		}
		start_war = {
			cb = hd4_tao_xuzhou_cb
			target = scope:hd4_tao_enemy
			target_title = scope:hd4_tao_target
		}
	}
	add_character_flag = sworn_peace
}


hd4_tao_protect_new_vassals_effect = {
    character:tao_qian = {
        every_vassal_or_below = {
            limit = { is_alive = yes is_landed = yes highest_held_title_tier >= tier_county }
            add_character_flag = hd4_tao_qxzl_protected
            add_character_flag = hd_qxzl_seen
            remove_character_flag = hd_qxzl_pending
        }
    }
}
