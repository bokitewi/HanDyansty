# Original house scope. Evaluate the complete native signed expression before
# testing its sign, including the original house-head and reversal modifiers.
# ACTOR is the live regional context; CHARACTER retains native attribution.
# The succession event uses its actual player-heir recipient as ACTOR and its
# deceased predecessor as CHARACTER, preserving native change attribution.
hd_dc_add_unity_effect = {
	save_temporary_scope_value_as = { name = hd_dc_unity_base value = $VALUE$ }
	add_unity_value = {
		value = {
			value = scope:hd_dc_unity_base
			if = {
				limit = { scope:hd_dc_unity_base < 0 }
				multiply = $ACTOR$.hd_dc_unity_loss_multiplier_value
			}
		}
		character = $CHARACTER$
		desc = $DESC$
	}
}
