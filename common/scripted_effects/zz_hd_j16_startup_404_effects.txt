# 安置一人。COURT 在世、有地；WHO 在世、无地、不是统治者（持世族头衔等无地头衔者也算统治者，add_courtier 会失败）、不是 COURT 本人、还没有雇主。
hd_j16_404c_place_effect = {
	if = { limit = { exists = scope:hd_j16_404c_court } clear_saved_scope = hd_j16_404c_court }
	if = { limit = { exists = scope:hd_j16_404c_who } clear_saved_scope = hd_j16_404c_who }
	character:$COURT$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
			}
			save_scope_as = hd_j16_404c_court
		}
	}
	character:$WHO$ ?= {
		if = {
			limit = {
				exists = scope:hd_j16_404c_court
				is_alive = yes
				is_landed = no
				is_ruler = no
				NOT = { this = scope:hd_j16_404c_court }
				NOT = { exists = employer }
			}
			save_scope_as = hd_j16_404c_who
			scope:hd_j16_404c_court = { add_courtier = scope:hd_j16_404c_who }
			set_knight_status = forbid    # 防 AI 自动选为骑士，降低节点前阵亡风险（见头注④）
			debug_log = "HD_J16 startup_404 FIRE place_$WHO$"
		}
		else = {
			debug_log = "HD_J16 startup_404 SKIP place_$WHO$"
		}
	}
}
