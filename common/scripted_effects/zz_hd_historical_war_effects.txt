	}

		limit = { exists = global_var:hd_guandu_fired NOT = { exists = global_var:hd_guandu_world_resolved } }
		set_global_variable = hd_guandu_world_resolved
	}
	if = {
		limit = { exists = global_var:hd_chibi_fired NOT = { exists = global_var:hd_chibi_world_resolved } }
		set_global_variable = hd_chibi_world_resolved
	}
	if = {
		limit = { exists = global_var:hd_rushu_fired NOT = { exists = global_var:hd_legacy_rushu_suppressed } }
		set_global_variable = hd_legacy_rushu_suppressed
	}

		limit = { exists = global_var:hd_yiling_fired NOT = { exists = global_var:hd_lb25_world_resolved } }
		set_global_variable = hd_lb25_world_resolved
	}
}

# One idempotent entry point serves new games, monthly save migration, and
# later Liu Bei phase adapters. The owned story is the source of truth.
hd_update_liu_bei_life_story_effect = {
	if = {
		limit = {
			game_start_date = 184.1.1
			character:liu_bei = { is_alive = yes }
		}
		character:liu_bei = {
			if = {
				limit = {
					NOT = {
						any_owned_story = { type = hd_story_liu_bei_life }
					}
				}
				create_story = {
					type = hd_story_liu_bei_life
				}
			}

			random_owned_story = {
				type = hd_story_liu_bei_life
				first_valid = {
					triggered_effect = {
						trigger = {
							OR = {
								exists = global_var:hd_lb25_world_open
								exists = global_var:hd_lb25_world_resolved
							}
						}
						effect = {
							set_variable = {
								name = hd_liu_bei_life_phase
								value = flag:hd_liu_bei_phase_yiling
							}
						}
					}
					triggered_effect = {
						trigger = {
							OR = {
								exists = global_var:hd_lb24_world_open
								exists = global_var:hd_lb24_world_resolved
							}
						}
						effect = {
							set_variable = {
								name = hd_liu_bei_life_phase
								value = flag:hd_liu_bei_phase_jingzhou
							}
						}
					}
					triggered_effect = {
						trigger = { NOT = { exists = var:hd_liu_bei_life_phase } }
						effect = {
							set_variable = {
								name = hd_liu_bei_life_phase
								value = flag:hd_liu_bei_phase_rising
							}
						}
					}
				}
			}
		}

		# Story-scoped effects above can leave the execution context on the
		# story object. Re-enter Liu Bei explicitly before character flags and
		# character events are evaluated.
		character:liu_bei = {
			# The first repair build used a permanent pending flag. Convert any
			# save made with that intermediate build before applying the timed lock.
			if = {
				limit = {
					NOT = { has_character_flag = hd_liu_bei_dispatch_state_v2_migrated }
				}
				remove_character_flag = hd_peach_garden_event_pending
				add_character_flag = hd_liu_bei_dispatch_state_v2_migrated
			}

			# Mark dispatch before opening the event so concurrent entry points
			# cannot create two Peach Garden windows.
			if = {
				limit = {
					NOT = { has_character_flag = hd_peach_garden_oath_completed }
					NOT = { has_character_flag = hd_peach_garden_oath_unavailable }
					NOT = { has_character_flag = hd_peach_garden_event_pending }
					NOT = { has_character_flag = hd_peach_garden_event_opened }
				}
				if = {
					limit = { current_date > 184.12.31 }
					add_character_flag = hd_peach_garden_oath_unavailable
				}
				else_if = {
					limit = {
						current_date >= 184.1.1
						current_date <= 184.12.31
						character:guan_yu ?= {
							is_alive = yes
							is_landed = no
						}
						character:zhang_fei ?= {
							is_alive = yes
							is_landed = no
						}
					}
					add_character_flag = {
						flag = hd_peach_garden_event_pending
						days = 60
					}
					trigger_event = hd_peach_garden.0001
				}
			}

			# An old save may already contain the resolved Yellow Turban state
			# without the personal follow-up having been scheduled.
			if = {
				limit = {
					current_date <= 190.12.31
					exists = global_var:hd_yellow_turban_outcome_resolved
					exists = global_var:hd_yellow_turban_han_survived
					NOT = { has_character_flag = hd_liu_bei_aftermath_started }
					NOT = { has_character_flag = hd_liu_bei_aftermath_pending }
					NOT = { has_character_flag = hd_liu_bei_adventurer_path }
					NOT = { has_character_flag = hd_liu_bei_received_anxi }
				}
				add_character_flag = {
					flag = hd_liu_bei_aftermath_pending
					days = 150
				}
				trigger_event = {
					id = hd_liu_bei_aftermath.0001
					months = 3
				}
			}
		}
	}
}

hd_open_fengying_world_opportunity_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_fengying_world_open }
			NOT = { exists = global_var:hd_fengying_world_claimed }
			NOT = { exists = global_var:hd_fengying_world_resolved }
		}
		set_global_variable = hd_fengying_world_open
		character:liu_xie = { save_scope_as = hd_fengying_emperor }
		if = {
			limit = {
				character:cao_cao = {
					is_alive = yes
					is_landed = yes
					is_independent_ruler = yes
					NOT = { has_character_flag = hd_cao_fengying_story_owner }
					NOT = { any_owned_story = { type = hd_story_cao_fengying } }
				}
			}
			character:cao_cao = {
				create_story = { type = hd_story_cao_fengying save_scope_as = hd_cao_fengying_story }
				trigger_event = hd_fengying.0001
			}
		}
		if = {
			limit = {
				character:yuan_shao = {
					is_alive = yes
					is_landed = yes
					is_independent_ruler = yes
					NOT = { has_character_flag = hd_yuan_fengying_story_owner }
					NOT = { any_owned_story = { type = hd_story_yuan_fengying } }
				}
			}
			character:yuan_shao = {
				create_story = { type = hd_story_yuan_fengying save_scope_as = hd_yuan_fengying_story }
				trigger_event = hd_fengying.0011
			}
		}
	}
}

	if = {

		set_global_variable = hd_chibi_risk_huang_gai
	}
	if = {
		limit = { title:k_uuii_jingzhou.holder.top_liege = character:cao_cao }

		remove_global_variable = hd_chibi_risk_jingzhou_loyalty

	}
	else = {
		title:k_uuii_jingzhou.holder = { trigger_event = hd_historical_world.0210 }
	}
}
