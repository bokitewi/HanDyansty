# 北伐之威（0～100）。以第一次北伐开战为基准点（boki 要求「只看北伐战果」：218 开局的汉中之战、东三郡不算）：
#   基准点后每场新蜀魏战争 +15（逐月衰减 1）；蜀持快照魏县比基准点多出的每县 +3；蜀军占领未割让的每县 +1

hd_zs_threat_effect = {
	if = {
		limit = { global_var:hd_zs_war_score > 0 }
		change_global_variable = { name = hd_zs_war_score add = -1 }
	}
	set_global_variable = { name = hd_beifa_wei value = 0 }
	# 基准点：第一次北伐已开战（218 由 hd_sg218_beifa_war_effect、其他开局由本链设 hd_sg218_beifa_started）
	if = {
		limit = {
			NOT = { exists = global_var:hd_zs_beifa_base }
			exists = global_var:hd_sg218_beifa_started
		}
		hd_zs_count_held_effect = yes
		set_global_variable = { name = hd_zs_beifa_base value = global_var:hd_zs_held_now }
		debug_log = "HD_ZS BEIFA_BASELINE_SET"
	}
	if = {
		limit = {
			exists = global_var:hd_zs_beifa_base
			exists = scope:hd_zs_shu
			exists = scope:hd_zs_wei
			NOT = { scope:hd_zs_shu = scope:hd_zs_wei }
		}
		scope:hd_zs_shu = {
			every_character_war = {
				limit = {
					NOT = { has_variable = hd_zs_counted }
					hd_zs_shu_wei_war_trigger = yes
				}
				set_variable = hd_zs_counted
				change_global_variable = { name = hd_zs_war_score add = 15 }
				debug_log = "HD_ZS BEIFA_WAR_COUNTED"
			}
		}
		# 持县：比基准点多出的部分
		hd_zs_count_held_effect = yes
		set_global_variable = { name = hd_zs_held_gain value = global_var:hd_zs_held_now }
		change_global_variable = { name = hd_zs_held_gain subtract = global_var:hd_zs_beifa_base }
		if = {
			limit = { global_var:hd_zs_held_gain > 0 }
			change_global_variable = { name = hd_zs_held_gain multiply = 3 }
			change_global_variable = { name = hd_beifa_wei add = global_var:hd_zs_held_gain }
		}
		# 占领：蜀军控制而尚未归蜀的快照县
		every_in_global_list = {
			variable = hd_zs_front_counties
			limit = {
				NOT = { holder ?= { top_liege = scope:hd_zs_shu } }
				county_controller ?= { top_liege = scope:hd_zs_shu }
			}
			change_global_variable = { name = hd_beifa_wei add = 1 }
		}
	}
	if = {
		limit = { global_var:hd_zs_war_score > 100 }
		set_global_variable = { name = hd_zs_war_score value = 100 }
	}
	change_global_variable = { name = hd_beifa_wei add = global_var:hd_zs_war_score }
	if = {
		limit = { global_var:hd_beifa_wei > 100 }
		set_global_variable = { name = hd_beifa_wei value = 100 }
	}
}

# 蜀现持快照魏县数（挡住北伐的判定用）→ global_var:hd_zs_held_now
hd_zs_count_held_effect = {
	set_global_variable = { name = hd_zs_held_now value = 0 }
	if = {
		limit = { exists = scope:hd_zs_shu }
		every_in_global_list = {
			variable = hd_zs_front_counties
			limit = { holder ?= { top_liege = scope:hd_zs_shu } }
			change_global_variable = { name = hd_zs_held_now add = 1 }
		}
	}
}

# 收门生（照 hd218_ma_su_disciple_effect）：解除门生原有恩师，举主功名补到门生之上。Scope：任意
hd_zs_disciple_effect = {
	global_var:hd_zs_p_$DISCIPLE$ ?= {
		if = {
			limit = {
				is_alive = yes
				global_var:hd_zs_p_$MASTER$ ?= { is_alive = yes }
				NOT = { this = global_var:hd_zs_p_$MASTER$ }
				NOT = { has_relation_elder = global_var:hd_zs_p_$MASTER$ }
			}
			save_scope_as = hd_zs_disciple
			every_relation = {
				type = elder
				limit = { NOT = { this = global_var:hd_zs_p_$MASTER$ } }
				remove_relation_disciple = scope:hd_zs_disciple
			}
			global_var:hd_zs_p_$MASTER$ = {
				while = {
					count = 10
					if = {
						limit = { merit_level <= scope:hd_zs_disciple.merit_level }
						change_merit_level = 1
					}
				}
			}
			set_relation_elder = global_var:hd_zs_p_$MASTER$
			debug_log = "HD_ZS DISCIPLE $DISCIPLE$ -> $MASTER$"
		}
	}
}

# 废为庶人：$WHO$ 失去全部实地头衔（交 $LORD$），无地后作 $LORD$ 的廷臣，加修正。Scope：任意
hd_zs_depose_effect = {
	$WHO$ = {
		if = {
			limit = { is_alive = yes }
			create_title_and_vassal_change = { type = revoked save_scope_as = hd_zs_depose_change add_claim_on_loss = no }
			every_held_title = {
				limit = { is_noble_family_title = no is_landless_type_title = no }
				change_title_holder = { holder = $LORD$ change = scope:hd_zs_depose_change take_baronies = yes }
			}
			resolve_title_and_vassal_change = scope:hd_zs_depose_change
			if = {
				limit = { is_ruler = no NOT = { is_courtier_of = $LORD$ } }
				set_employer = $LORD$
			}
			add_character_modifier = { modifier = hd_zgl_shuren_modifier years = 10 }
		}
	}
}

# 世族爵位直写（照 hd190_peer_hou_effect / WJ_peer_commit 世族分支）。Scope：受封者；$ISSUER$ 为授爵者（帝国级郡国制君主）
# RANK：1 亭侯（食邑男爵领）、2 县侯（食邑县）；NAME = FOOD
hd_zs_family_peer_effect = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_variable = wj_peer_rank }
			NOT = { has_trait = WJ_peerage_trait }
			top_liege ?= $ISSUER$
			NOT = { this = top_liege }
			$ISSUER$ = { WJ_peer_realm_head = yes }
			title:$FOOD$ = { WJ_peer_name_free = yes }
		}
		WJ_peer_get_family = yes
		if = {
			limit = {
				exists = scope:wj_peer_family
				scope:wj_peer_family = { holder = prev NOT = { has_variable = wj_peer_rank } }
			}
			scope:wj_peer_family = {
				set_variable = { name = wj_peer_rank value = $RANK$ }
				set_variable = { name = wj_peer_name value = title:$FOOD$ }
				set_variable = { name = wj_peer_food value = title:$FOOD$ }
				set_variable = { name = wj_peer_realm value = $ISSUER$.primary_title }
				set_variable = { name = wj_peer_last_paid value = current_year }
				WJ_peer_bind_family = yes
			}
			debug_log = "HD_ZS PEER OK $FOOD$"
		}
		else = { debug_log = "HD_ZS PEER NO_FAMILY $FOOD$" }
	}
	else = { debug_log = "HD_ZS PEER SKIPPED $FOOD$" }
}
