# 仅由原版 fund_inspiration.1051 寻宝完成事件调用；每次只生成一件奖励。
# 每件未出世的宝物占 5/100，已出世的权重归还原版奖励。
hd_qixing_adventure_reward_effect = {
	$OWNER$ = { save_scope_as = owner }
	$CREATOR$ = { save_scope_as = adventurer }
	random_list = {
		5 = {
			trigger = { NOT = { exists = global_var:hd_qixing_jia_created } }
			scope:owner = { hd_create_qixing_jia_effect = yes }
		}
		5 = {
			trigger = { NOT = { exists = global_var:hd_qixing_dao_created } }
			scope:owner = { hd_create_qixing_dao_effect = yes }
		}
		90 = {
			modifier = {
				add = 5
				exists = global_var:hd_qixing_jia_created
			}
			modifier = {
				add = 5
				exists = global_var:hd_qixing_dao_created
			}
			inspiration_adventure_create_artifact_effect = {
				OWNER = scope:owner
				CREATOR = scope:adventurer
			}
		}
	}
}

hd_create_qixing_jia_effect = {
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = hd_qixing_jia_name
		description = hd_qixing_jia_desc
		rarity = illustrious
		type = armor_brigandine
		template = hd_qixing_unique_artifact_template
		visuals = armor_japan
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_prowess_4_modifier
		save_scope_as = newly_created_artifact
		history = {
			type = discovered
			actor = scope:adventurer
			recipient = scope:owner
			location = scope:adventurer.location
		}
	}
	scope:newly_created_artifact = {
		add_artifact_modifier = artifact_knight_effectiveness_9_modifier
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = hd_qixing_jia
	}
	# 永不清除：即使毁坏也不再生成。
	set_global_variable = hd_qixing_jia_created
	save_scope_value_as = { name = armor_artifact value = yes }
}

hd_create_qixing_dao_effect = {
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = hd_qixing_dao_name
		description = hd_qixing_dao_desc
		rarity = illustrious
		type = sword
		template = hd_qixing_unique_artifact_template
		visuals = japanese_sword
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_prowess_6_modifier
		save_scope_as = newly_created_artifact
		history = {
			type = discovered
			actor = scope:adventurer
			recipient = scope:owner
			location = scope:adventurer.location
		}
	}
	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = hd_qixing_dao
	}
	set_global_variable = hd_qixing_dao_created
	save_scope_value_as = { name = weapon_artifact value = yes }
}
