# 清掉可能随 trigger_event 带过来的保存作用域（不存在时不清，免报错）
hd_j16_dir_clear_scope_effect = {
	if = {

		clear_saved_scope = $S$
	}

hd_j16_dir_dr = {
	every_county_in_region = {
		region = $R$
		limit = { exists = duchy }
		duchy = {
			if = {
				limit = { NOT = { is_target_in_global_variable_list = { name = hd_j16_dir_qin_targets target = this } } }
				add_to_global_variable_list = { name = hd_j16_dir_qin_targets target = this }
			}
		}
	}
}

# E3-W30 导演启动【机制】（03 §1.1「启用条件」「启动」；01 §2.3 预留号）
# 由 hd_j16_e3_346_effect 调用（入口那一行由 4-B 加）。因此实际能启动的日期只在 E3 粗筛内（349.1.1 – 374.1.1，及首跳全段扫）。
# 窗口 351.1.1 – 382.6.1；条件：史实事件规则开；E3-Q10 苻健入关已 done（hd_nbc_fujian_done；370 开局由开局前标设）；
#   能解析出秦主（长安之主，苻氏，独立；玩家也算，导演对玩家只空转）；未设 hd_j16_qin_collapse / hd_j16_dir_qin_off。
# 首次满足时在 title:h_china 上排 hd_j16_dir_bootstrap（不依赖开局日期，304、346、370 一样）。
# 跨开局：304、346 心跳（351 起）；370 首跳即启动（开局直接从 S3 起，见 bootstrap）；404 前标（开局年 ≥ 404 只设 done）。
# 变量：hd_j16_dir_w30_done、_result（fire）、_skip、_wlog；面板 hd_hev_j16_dir_qin_start
############################################################
hd_j16_dir_w30_effect = {
	if = {
		limit = { NOT = { has_global_variable = hd_j16_dir_w30_done } }
		# ① 自保护：404 开局窗口已过
		if = {
			limit = { global_var:hd_j16_start_year >= 404 }
			set_global_variable = hd_j16_dir_w30_done
			debug_log = "HD_J16 e3_w30 SKIP prestart"
		}
		else_if = {
			limit = { current_date >= 351.1.1 }
			# ② 截止跳过
			if = {
				limit = { current_date >= 382.6.1 }
				set_global_variable = hd_j16_dir_w30_done
				set_global_variable = hd_j16_dir_w30_skip
				debug_log = "HD_J16 e3_w30 SKIP window_closed"
			}
			# ③ 前秦已瓦解或导演已下线
			else_if = {
				limit = {
					OR = {
						has_global_variable = hd_j16_qin_collapse
						has_global_variable = hd_j16_dir_qin_off
					}
				}
				set_global_variable = hd_j16_dir_w30_done
				set_global_variable = hd_j16_dir_w30_skip
				debug_log = "HD_J16 e3_w30 SKIP qin_collapse"
			}
			else = {
				# ④ 解析秦主（按位置）
				hd_j16_dir_clear_scope_effect = { S = hd_j16_w30_qin }
				title:c_uuii_p_867.holder.top_liege ?= {
					if = {
						limit = {
							is_independent_ruler = yes
							dynasty ?= dynasty:fu_1
						}
						save_scope_as = hd_j16_w30_qin
					}
				}
				# ⑤ 条件
				if = {
					limit = {
						hd_history_events_enabled_trigger = yes
						has_global_variable = hd_nbc_fujian_done
						exists = scope:hd_j16_w30_qin
					}
					set_global_variable = hd_j16_dir_w30_done
					set_global_variable = hd_j16_dir_w30_result
					set_global_variable = hd_hev_j16_dir_qin_start
					title:h_china = { trigger_event = { on_action = hd_j16_dir_bootstrap days = 1 } }
					debug_log = "HD_J16 e3_w30 FIRE bootstrap_scheduled"
				}
				# ⑥ 等待
				else_if = {
					limit = { NOT = { has_global_variable = hd_j16_dir_w30_wlog } }
					set_global_variable = hd_j16_dir_w30_wlog
					debug_log = "HD_J16 e3_w30 WAIT not_ready"
				}
			}
		}
	}
}

# 脉冲（on_action hd_j16_dir_pulse，Root：title:h_china；61 天一次，续期在 on_action 里）
############################################################
hd_j16_dir_pulse_effect = {
	if = {
		limit = {
			has_global_variable = hd_j16_dir_qin_active
			NOT = { has_global_variable = hd_j16_dir_qin_stopped }
		}
		# 每跳开头清掉随 trigger_event 带过来的保存作用域（照 hd184r :84-86；10-23 作用域串人修复）
		hd_j16_dir_clear_scope_effect = { S = hd_j16_dir_actor }
		hd_j16_dir_clear_scope_effect = { S = hd_j16_dir_tgt }
		hd_j16_dir_clear_scope_effect = { S = hd_j16_dir_def }
		hd_j16_dir_clear_scope_effect = { S = hd_j16_dir_newwar }
		hd_j16_dir_clear_scope_effect = { S = hd_news_a }
		hd_j16_dir_clear_scope_effect = { S = hd_news_b }
		if = {
			limit = { hd_j16_dir_stop_trigger = yes }
			hd_j16_dir_stop_effect = yes
		}
		else = {
			# 按位置解析秦主
			title:c_uuii_p_867.holder.top_liege ?= {
				if = {
					limit = {
						is_independent_ruler = yes
						dynasty ?= dynasty:fu_1
					}
					save_scope_as = hd_j16_dir_actor
				}
			}
			# 秦主换人：旧秦主在世就当场摘掉旗标（死者不动，报错清单第 28 条）
			if = {
				limit = {
					has_global_variable = hd_j16_dir_qin_actor
					global_var:hd_j16_dir_qin_actor ?= {
						is_alive = yes
						has_character_flag = hd_j16_dir_member
						trigger_if = {
							limit = { exists = scope:hd_j16_dir_actor }
							NOT = { this = scope:hd_j16_dir_actor }
						}
					}
				}
				global_var:hd_j16_dir_qin_actor = { remove_character_flag = hd_j16_dir_member }
				debug_log = "HD_J16 dir_s0 FIRE actor_changed"
			}
			if = {
				limit = { exists = scope:hd_j16_dir_actor }
				set_global_variable = { name = hd_j16_dir_qin_actor value = scope:hd_j16_dir_actor }
				# 旗标：AI 秦主每跳重给 75 天（长于 61 天的脉冲），导演停下后自然过期；玩家秦主不托管
				scope:hd_j16_dir_actor = {
					if = {
						limit = {
							is_alive = yes
							is_ai = yes
						}
						if = {
							limit = { has_character_flag = hd_j16_dir_member }
							remove_character_flag = hd_j16_dir_member
						}
						add_character_flag = { flag = hd_j16_dir_member days = 75 }
					}
					else_if = {
						limit = {
							is_alive = yes
							has_character_flag = hd_j16_dir_member
						}
						remove_character_flag = hd_j16_dir_member
					}
				}
				# 整段暂停：只续旗标（保护洛阳、东征等节点战争不被天意青睐、战略 AI 抢名额）
				if = {
					limit = { hd_j16_dir_paused_trigger = yes }
					if = {
						limit = { NOT = { has_global_variable = hd_j16_dir_pause_wlog } }
						set_global_variable = hd_j16_dir_pause_wlog
						hd_j16_dir_log_effect = { TYPE = WAIT REASON = paused_yan_war }
					}
				}
				# 阶段已走完（hd_j16_dir_qin_done）：不再进入下面这段，只续旗标，直到停止续期
				else_if = {
					limit = { NOT = { has_global_variable = hd_j16_dir_qin_done } }
					if = {
						limit = { scope:hd_j16_dir_actor = { hd_j16_dir_actor_ready_trigger = yes } }
						scope:hd_j16_dir_actor = { hd_j16_dir_tick_effect = yes }
					}
					else = {
						hd_j16_dir_log_once_effect = { TYPE = WAIT REASON = actor_not_ready WLOG = notready_wlog }
					}
				}
			}
			else_if = {
				limit = { NOT = { has_global_variable = hd_j16_dir_noactor_wlog } }
				set_global_variable = hd_j16_dir_noactor_wlog
				hd_j16_dir_log_effect = { TYPE = WAIT REASON = no_actor }
			}
		}
	}
}

# 旗标：前秦已瓦解或淝水已 done 时当场摘掉；其余（383.1.1 到期、E4-W40 下线）改给 730 天，

hd_j16_dir_stop_effect = {
	set_global_variable = hd_j16_dir_qin_stopped
	if = {
		limit = { has_global_variable = hd_j16_dir_qin_actor }
		global_var:hd_j16_dir_qin_actor ?= {
			if = {
				limit = {
					is_alive = yes
					has_character_flag = hd_j16_dir_member
				}
				remove_character_flag = hd_j16_dir_member
				if = {
					limit = {
						is_ai = yes
						NOT = { has_global_variable = hd_j16_qin_collapse }
						NOT = { has_global_variable = hd_nbc_feishui_done }
					}
					add_character_flag = { flag = hd_j16_dir_member days = 730 }
				}
			}
		}
	}
	clear_global_variable_list = hd_j16_dir_qin_targets
	clear_global_variable_list = hd_j16_dir_qin_cand
	debug_log = "HD_J16 dir_s0 FIRE stopped"
}

hd_j16_dir_tick_effect = {
	hd_j16_dir_build_effect = yes
	# 一、阶段转换（作废、闸门下跳过、节点完成、控制率/空转）
	hd_j16_dir_transition_effect = yes
	if = {
		limit = { has_global_variable = hd_j16_dir_rebuild }
		remove_global_variable = hd_j16_dir_rebuild
		hd_j16_dir_build_effect = yes
	}
	# 二、找目标、开战（被门挡住的阶段不计空转）
	if = {
		limit = { NOT = { has_global_variable = hd_j16_dir_qin_done } }
		if = {
			limit = { has_global_variable = hd_j16_dir_gated }
			hd_j16_dir_log_once_effect = { TYPE = WAIT REASON = gated WLOG = gated_wlog }
		}
		else = {
			hd_j16_dir_collect_effect = yes
			if = {
				limit = { NOT = { any_in_global_list = { variable = hd_j16_dir_qin_cand always = yes } } }
				# 空转 +1；本阶段的导演战争还在打时不计（免得战中因「无新目标」提前判完成）【自拟】
				if = {
					limit = { NOT = { hd_j16_dir_stage_war_ongoing_trigger = yes } }
					change_global_variable = { name = hd_j16_dir_qin_idle add = 1 }
					hd_j16_dir_log_once_effect = { TYPE = WAIT REASON = idle WLOG = idle_wlog }
				}
			}
			else = {
				set_global_variable = { name = hd_j16_dir_qin_idle value = 0 }
				if = {
					limit = { current_date >= 382.6.1 }
					hd_j16_dir_log_once_effect = { TYPE = WAIT REASON = no_new_war_after_382_6 WLOG = late_wlog }

# 用法：hd_j16_border_start_war_effect = { ATTACKER = <角色> DEFENDER = <角色> TARGET = <县或郡头衔> KIND = <场次> }
# - 只打一县：原定县须由守方或其下属持有、且与攻方最高领主的领土接壤；不接壤时改取同郡内满足条件的一县；仍无则不宣战。
#   不走 hd_wct：连通筛选在连不到时会改成「守方所有接壤郡」，与「胜只取一县」不合。
# - 成功判据同 hd_nbc_start_war_effect：exists = scope:hd_nbc_war_$KIND$；战争上设 hd_nbc_war_kind = flag:$KIND$，全局设 hd_nbc_$KIND$_active，
#   另存全局 hd_j16_border_$KIND$_target（实际所争的县，节点判结局用）。
# - CB 结算不分发 KIND（00 §6）：胜负由节点按该县归属自判，_active 由节点结局时自清。
############################################################
hd_j16_border_start_war_effect = {
	# 成功判据是 scope:hd_nbc_war_$KIND$，先清掉随心跳带来的同名旧作用域，免得宣战失败也被当成成功
	hd_j16_dir_clear_scope_effect = { S = hd_nbc_war_$KIND$ }
	hd_j16_dir_clear_scope_effect = { S = hd_j16_bd_me }
	hd_j16_dir_clear_scope_effect = { S = hd_j16_bd_att }
	hd_j16_dir_clear_scope_effect = { S = hd_j16_bd_def }
	hd_j16_dir_clear_scope_effect = { S = hd_j16_bd_tgt }
	$ATTACKER$ = {
		save_scope_as = hd_j16_bd_me
		top_liege = { save_scope_as = hd_j16_bd_att }
	}
	$DEFENDER$ = { save_scope_as = hd_j16_bd_def }
	# TARGET 可传县或郡：县先试本县，不行再取同郡内一县；郡直接取郡内一县
	$TARGET$ = {
		if = {
			limit = {
				tier = tier_county
				hd_j16_border_county_ok_trigger = yes
			}
			save_scope_as = hd_j16_bd_tgt
		}
		else_if = {
			limit = { tier = tier_county }
			duchy ?= {
				random_de_jure_county = {
					limit = { hd_j16_border_county_ok_trigger = yes }
					save_scope_as = hd_j16_bd_tgt
				}
			}
		}
		else_if = {
			limit = { tier = tier_duchy }
			random_de_jure_county = {
				limit = { hd_j16_border_county_ok_trigger = yes }
				save_scope_as = hd_j16_bd_tgt
			}
		}
	}
	if = {
		limit = { exists = scope:hd_j16_bd_tgt }
		scope:hd_j16_bd_me = {
			if = {
				limit = { is_allied_to = scope:hd_j16_bd_def }
				break_alliance_due_to_attack_effect = { TARGET = scope:hd_j16_bd_def }
			}
			start_war = {
				cb = hd_j16_border_cb
				target = scope:hd_j16_bd_def
				target_title = scope:hd_j16_bd_tgt
			}
			random_character_war = {
				limit = {
					using_cb = hd_j16_border_cb
					primary_attacker = scope:hd_j16_bd_me
					primary_defender = scope:hd_j16_bd_def
					NOT = { has_variable = hd_nbc_war_kind }
				}
				set_variable = { name = hd_nbc_war_kind value = flag:$KIND$ }
				save_scope_as = hd_nbc_war_$KIND$
			}
		}
		if = {
			limit = { exists = scope:hd_nbc_war_$KIND$ }
			set_global_variable = hd_nbc_$KIND$_active
			set_global_variable = { name = hd_j16_border_$KIND$_target value = scope:hd_j16_bd_tgt }
			debug_log = "HD_J16 border FIRE war_started $KIND$"
		}
		else = {
			debug_log = "HD_J16 border WAIT war_start_failed $KIND$"
		}
	}
	else = {
		debug_log = "HD_J16 border WAIT not_adjacent $KIND$"
	}
}
