# The native dinner death branch keeps integer weight 1000 out of 100000.
# Only the ordinary drunken-stupor branch pays for the increased trait chance.
hd_dc_prepare_dinner_trait_pool_effect = {
	hd_dc_freeze_existing_trait_pool_region_effect = yes
	save_scope_value_as = { name = hd_dc_dinner_trait_add value = 23976 }
	save_scope_value_as = { name = hd_dc_dinner_stupor_add value = 74925 }
	if = {
		limit = { scope:hd_dc_existing_pool_active = 1 can_be_drunkard = yes }
		save_scope_value_as = { name = hd_dc_hazard_base value = 24 }
		save_scope_value_as = { name = hd_dc_hazard_multiplier value = scope:hd_dc_existing_trait_multiplier }
		save_scope_value_as = { name = hd_dc_dinner_trait_weight value = { value = hd_dc_hazard_probability_value multiply = 1000 round = yes } }
		save_scope_value_as = { name = hd_dc_dinner_trait_add value = { value = scope:hd_dc_dinner_trait_weight subtract = 24 } }
		save_scope_value_as = { name = hd_dc_dinner_stupor_add value = { value = 98925 subtract = scope:hd_dc_dinner_trait_weight } }
	}
	else_if = {
		limit = { scope:hd_dc_existing_pool_active = 1 }
		# An ineligible trait's removed share returns to ordinary stupor;
		# otherwise disabling that branch would increase the death probability.
		save_scope_value_as = { name = hd_dc_dinner_stupor_add value = 98925 }
	}
}

# B is the sum of currently legal native negative-outcome weights. All bad
# branches keep their original relative weights; all other branches share one
# complementary factor. No second random draw or AI option weight is added.
hd_dc_scale_existing_trait_pool_effect = {
	if = {
		limit = { scope:hd_dc_existing_bad_weight > 0 scope:hd_dc_existing_pool_active = 1 }
		save_scope_value_as = { name = hd_dc_hazard_base value = { value = scope:hd_dc_existing_bad_weight multiply = 100 divide = { value = $OTHER$ add = scope:hd_dc_existing_bad_weight } } }
		save_scope_value_as = { name = hd_dc_hazard_multiplier value = scope:hd_dc_existing_trait_multiplier }
		save_scope_value_as = { name = hd_dc_existing_trait_chance value = hd_dc_hazard_probability_value }
		save_scope_value_as = { name = hd_dc_existing_other_factor value = { value = 100 subtract = scope:hd_dc_existing_trait_chance multiply = scope:hd_dc_existing_bad_weight multiply = 1000 divide = scope:hd_dc_existing_trait_chance divide = $OTHER$ } }
	}
}

hd_dc_prepare_education_trait_pool_effect = {
	hd_dc_freeze_existing_trait_pool_region_effect = yes
	save_scope_value_as = { name = hd_dc_existing_other_factor value = 1000 }
	save_scope_value_as = { name = hd_dc_existing_bad_weight value = 0 }
	if = { limit = { hd_dc_existing_pool_drunkard_trigger = yes } save_scope_value_as = { name = hd_dc_existing_bad_weight value = { value = scope:hd_dc_existing_bad_weight add = 10 } } }
	if = { limit = { hd_dc_existing_pool_rakish_trigger = yes } save_scope_value_as = { name = hd_dc_existing_bad_weight value = { value = scope:hd_dc_existing_bad_weight add = 10 } } }
	if = { limit = { hd_dc_existing_profligate_candidate_trigger = yes } save_scope_value_as = { name = hd_dc_existing_bad_weight value = { value = scope:hd_dc_existing_bad_weight add = 10 } } }
	hd_dc_scale_existing_trait_pool_effect = { OTHER = 50 }
}

# Region/F were frozen before move_to_pool. Recheck native trait eligibility
# after the move, at the exact point where the native random_list checks it.
hd_dc_prepare_inspiration_trait_pool_effect = {
	save_scope_value_as = { name = hd_dc_existing_other_factor value = 1000 }
	save_scope_value_as = { name = hd_dc_existing_bad_weight value = 0 }
	if = { limit = { hd_dc_existing_depressed_candidate_trigger = yes } save_scope_value_as = { name = hd_dc_existing_bad_weight value = { value = scope:hd_dc_existing_bad_weight add = 25 } } }
	if = { limit = { can_be_drunkard = yes } save_scope_value_as = { name = hd_dc_existing_bad_weight value = { value = scope:hd_dc_existing_bad_weight add = 25 } } }
	if = { limit = { can_be_hashishiyah = yes } save_scope_value_as = { name = hd_dc_existing_bad_weight value = { value = scope:hd_dc_existing_bad_weight add = 25 } } }
	hd_dc_scale_existing_trait_pool_effect = { OTHER = 50 }
}
hd_dc_freeze_existing_trait_pool_region_effect = {
	save_scope_value_as = { name = hd_dc_existing_pool_active value = 0 }
	save_scope_value_as = { name = hd_dc_existing_trait_multiplier value = 1 }
	if = {
		limit = { is_alive = yes is_adult = yes exists = situation:dynastic_cycle any_character_situation = { this = situation:dynastic_cycle } }
		save_scope_value_as = { name = hd_dc_existing_pool_active value = 1 }
		save_scope_value_as = { name = hd_dc_existing_trait_multiplier value = hd_dc_social_hazard_multiplier_value }
	}
}
