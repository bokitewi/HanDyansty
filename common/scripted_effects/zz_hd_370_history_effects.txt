# 被杀（有凶手）。电脑角色直接死；玩家角色不杀，只记日志
hd_370_kill_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ killer = $KILLER$ }
			debug_log = "HD_370 KILLED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_370 KILL_SKIPPED_PLAYER $WHO$"
		}
	}
}
# 被杀（无具名凶手）
hd_370_die_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ }
			debug_log = "HD_370 DIED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_370 DIE_SKIPPED_PLAYER $WHO$"
		}
	}
}

# $NEW$ 接收 $OLD$ 的全部头衔与封臣（不含世族、朝官头衔），$OLD$ 不死。
# 照 hd_531_take_realm_effect：同一个 change 里先转头衔、后改投封臣（新主可能是无地廷臣）
hd_370_take_realm_effect = {
	$NEW$ = { save_scope_as = hd_370_new_ruler }
	create_title_and_vassal_change = { type = granted save_scope_as = hd_370_take_change add_claim_on_loss = no }
	$OLD$ = {
		every_held_title = {
			limit = {
				is_noble_family_title = no
				tgp_is_minister_title = no
				is_landless_type_title = no
				tier >= tier_county
			}
			change_title_holder = {
				holder = scope:hd_370_new_ruler
				change = scope:hd_370_take_change
				take_baronies = yes
			}
		}
		every_vassal = {
			limit = { NOT = { this = scope:hd_370_new_ruler } }
			change_liege = { liege = scope:hd_370_new_ruler change = scope:hd_370_take_change }
		}
	}
	resolve_title_and_vassal_change = scope:hd_370_take_change
}

# 结束 $WHO$ 身上场次为 $KIND$ 的战争（白和）
hd_370_end_kind_war_effect = {
	character:$WHO$ ?= {
		every_character_war = {
			limit = { var:hd_nbc_war_kind ?= flag:$KIND$ }
			end_war = white_peace
		}
	}
}

hd_370_yuanzhen_death_effect = {
	if = {
		limit = {
			current_date >= 370.4.10
			NOT = { exists = global_var:hd_370_yuanzhen_death_done }
		}
		if = {
			limit = { character:chenjun_yuan_zhen ?= { is_alive = yes is_ai = yes } }
			set_global_variable = hd_370_yuanzhen_death_done
			if = {
				limit = {
					character:chenjun_yuan_zhen = { is_landed = yes }
					character:chenjun_yuan_jin ?= { is_alive = yes is_imprisoned = no }
				}
				hd_370_take_realm_effect = { NEW = character:chenjun_yuan_jin OLD = character:chenjun_yuan_zhen }
				character:chenjun_yuan_jin = {
					hd_nbc_become_independent_effect = yes
					if = {
						limit = { has_title = title:c_uuii_p_1961 }
						set_realm_capital = title:c_uuii_p_1961
					}
				}
				debug_log = "HD_370 YUANZHEN SHOUCHUN_TO_YUANJIN"
			}
			character:chenjun_yuan_zhen = { death = { death_reason = death_natural_causes } }
			debug_log = "HD_370 HIST_DEATH chenjun_yuan_zhen"
		}
		else_if = {
			limit = { NOT = { character:chenjun_yuan_zhen ?= { is_alive = yes } } }
			set_global_variable = hd_370_yuanzhen_death_done
		}
		else = {
			hd_nbc_hist_death_effect = { WHO = chenjun_yuan_zhen DATE = 370.4.10 }
		}
	}
}

hd_370_outcome_deaths_effect = {
	# 拓跋什翼犍：建元十二年十二月秦灭代，什翼犍死于内乱。只在代已失云中、或正与秦交战时
	if = {
		limit = {
			current_date >= 376.12.12
			NOT = { exists = global_var:hd_370_death_tuoba_005_done }
		}
		set_global_variable = hd_370_death_tuoba_005_done
		if = {
			limit = {
				character:tuoba_005 ?= {
					is_alive = yes
					OR = {
						NOT = { title:c_uuii_p_1333.holder ?= { OR = { this = character:tuoba_005 top_liege = character:tuoba_005 } } }
						AND = {
							exists = character:lueyang_fu_020
							is_at_war_with = character:lueyang_fu_020
						}
					}
				}
			}
			hd_370_die_effect = { WHO = tuoba_005 REASON = death_murder }
		}
		else = { debug_log = "HD_370 DEATH_SKIP tuoba_005" }
	}
	# 慕容暐：太元九年十二月谋刺苻坚事泄被杀。只在他已入苻坚宫廷（燕亡出降）时
	if = {
		limit = {
			current_date >= 385.1.10
			NOT = { exists = global_var:hd_370_death_murong_wei_done }
		}
		set_global_variable = hd_370_death_murong_wei_done
		if = {
			limit = {
				character:changli_murong_077 ?= {
					is_alive = yes
					is_landed = no
					employer ?= character:lueyang_fu_020
				}
				character:lueyang_fu_020 ?= { is_alive = yes }
			}
			hd_370_kill_effect = { WHO = changli_murong_077 REASON = death_execution KILLER = character:lueyang_fu_020 }
		}
		else = { debug_log = "HD_370 DEATH_SKIP changli_murong_077" }
	}
	# 苻坚：淝水败后为姚苌所执，缢于新平佛寺（太元十年八月辛丑）。只在他已失长安时
	if = {
		limit = {
			current_date >= 385.10.16
			NOT = { exists = global_var:hd_370_death_fu_jian_done }
		}
		set_global_variable = hd_370_death_fu_jian_done
		if = {
			limit = {
				character:lueyang_fu_020 ?= {
					is_alive = yes
					NOT = { title:c_uuii_p_867.holder ?= { OR = { this = character:lueyang_fu_020 top_liege = character:lueyang_fu_020 } } }
				}
			}
			if = {
				limit = { character:shaodang_yao_006 ?= { is_alive = yes } }
				hd_370_kill_effect = { WHO = lueyang_fu_020 REASON = death_murder_known KILLER = character:shaodang_yao_006 }
			}
			else = { hd_370_die_effect = { WHO = lueyang_fu_020 REASON = death_murder } }
		}
		else = { debug_log = "HD_370 DEATH_SKIP lueyang_fu_020" }
	}
}

# S1 洛阳之战（开局即在打）。慕容筑出降分支：370.3.14（二月朔）起、370.7.9 止（计划复核必改 1 甲）
############################################################
hd_370_check_luoyang_effect = {
	# 慕容筑以金墉降秦
	if = {
		limit = {
			current_date >= 370.3.14
			exists = global_var:hd_370_luoyang_started
			NOT = { exists = global_var:hd_370_luoyang_done }
			NOT = { exists = global_var:hd_370_zhu_checked }
		}
		if = {
			limit = { current_date >= 370.7.9 }
			set_global_variable = hd_370_zhu_checked
			debug_log = "HD_370 S1 ZHU_SURRENDER SKIPPED_DEADLINE"
		}
		else_if = {
			limit = {
				character:hd_370_murong_zhu ?= {
					is_alive = yes
					is_imprisoned = no
					has_title = title:d_henan
					top_liege = character:changli_murong_077
				}
				character:lueyang_fu_020 ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } }
			}
			set_global_variable = hd_370_zhu_checked
			character:hd_370_murong_zhu = {
				if = {
					limit = { is_ai = yes }
					hd_370_zhu_surrender_effect = yes
				}
				else = { trigger_event = hd_370.0110 }
			}
		}
		else_if = {
			limit = { NOT = { character:hd_370_murong_zhu ?= { is_alive = yes } } }
			set_global_variable = hd_370_zhu_checked
			debug_log = "HD_370 S1 ZHU_SURRENDER SKIPPED_DEAD"
		}
	}
	# 战后：金墉（c_uuii_p_786）归秦一系即秦胜
	if = {
		limit = {
			exists = global_var:hd_370_luoyang_started
			NOT = { exists = global_var:hd_370_luoyang_done }
			NOT = { character:lueyang_fu_020 ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
			NOT = { character:changli_murong_077 ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
		}
		set_global_variable = hd_370_luoyang_done
		if = {
			limit = {
				title:c_uuii_p_786.holder ?= {
					OR = {
						this = character:lueyang_fu_020
						top_liege = character:lueyang_fu_020
					}
				}
			}
			set_global_variable = hd_370_luoyang_qin_won
			debug_log = "HD_370 S1 LUOYANG QIN_WON"
		}
		else = {
			set_global_variable = hd_370_luoyang_qin_lost
			debug_log = "HD_370 S1 LUOYANG QIN_LOST"
		}
	}
}

# 慕容筑出降：先白和结束洛阳之战，再把河南郡交给王猛（王猛不在就交苻坚），慕容筑入苻坚宫廷。Scope 不限

			character:wang_meng_3 ?= {

				top_liege = character:lueyang_fu_020
			}
		}
		character:wang_meng_3 = { save_scope_as = hd_370_luoyang_recipient }
	}
	else = {
		character:lueyang_fu_020 = { save_scope_as = hd_370_luoyang_recipient }
	}
	hd_370_take_realm_effect = { NEW = scope:hd_370_luoyang_recipient OLD = character:hd_370_murong_zhu }
	character:lueyang_fu_020 ?= {
		if = {
			limit = { character:hd_370_murong_zhu ?= { is_alive = yes is_landed = no } }
			add_courtier = character:hd_370_murong_zhu
		}
	}
	set_global_variable = hd_370_zhu_surrendered
	debug_log = "HD_370 S1 ZHU_SURRENDERED"
}

# 苻坚出兵：洛阳之战还没打完就先白和（计划复核必改 1 乙作兜底）；目标魏郡（邺）。
# 开战成功才给事件兵：王猛所部 25,000 在蒲阪（"步骑六万"），慕容评 15,000 在潞城（"中外精兵三十万"按比例折算）。
# 失败就清掉 started，下次心跳重试，截止日兜底
hd_370_dongzheng_war_effect = {
	if = {
		limit = {
			exists = global_var:hd_370_luoyang_started
			NOT = { exists = global_var:hd_370_luoyang_done }
		}
		hd_370_end_kind_war_effect = { WHO = lueyang_fu_020 KIND = luoyang }
		debug_log = "HD_370 S2 LUOYANG_ENDED_FOR_DONGZHENG"
	}
	# 洛阳之战白和后 hd_nbc_war_cb 给苻坚加了对燕的停战；史实是燕先背约，不应扣威望（执行后复核必改 2）
	character:lueyang_fu_020 = { add_character_flag = { flag = allowed_war_declaration_sans_truce_penalties days = 1 } }
	hd_nbc_start_war_effect = { ATTACKER = character:lueyang_fu_020 DEFENDER = character:changli_murong_077 TARGET = title:d_weijun KIND = dongzheng }
	if = {
		limit = { exists = scope:hd_nbc_war_dongzheng }
		set_global_variable = hd_370_dongzheng_decided
		character:lueyang_fu_020 = {
			hd_nbc_spawn_army_effect = { SIZE = 25000 WHERE = title:c_uuii_p_980 NAME = hd_370_wangmeng_army_name }
		}
		character:changli_murong_077 ?= {
			hd_nbc_spawn_army_effect = { SIZE = 15000 WHERE = title:c_uuii_p_1163 NAME = hd_370_murongping_army_name }
		}
	}
	else = {
		remove_global_variable = hd_370_dongzheng_started
		# 天意青睐的苻坚可能已经以别的理由与燕开战，开不出这一场；分开记，便于从 debug.log 判断燕亡没发生的原因
		if = {
			limit = { character:lueyang_fu_020 ?= { is_at_war_with = character:changli_murong_077 } }
			debug_log = "HD_370 S2 DONGZHENG WAR_START_FAILED ALREADY_AT_WAR_WITH_YAN RETRY"
		}
		else = { debug_log = "HD_370 S2 DONGZHENG WAR_START_FAILED RETRY" }
	}
}
