# Region scope. This is the native organizational prerequisite snapshot for
# PAR-BOOT, not a fabricated party, a launched crisis or an AI action package.
# Native movement_power is the value displayed/maintained by the installed
# movement system; this observer does not replace its own annual/on-join rules.
hd_dc_select_party_pair_effect = {
	clear_saved_scope = hd_dc_par_selected_first_group
	clear_saved_scope = hd_dc_par_selected_second_group
	if = {
		limit = {
			OR = {
				AND = { var:hd_dc_selected_type = flag:par var:hd_dc_annual_epoch = current_year NOT = { var:hd_dc_par_pair_draw_year = current_year } }
				hd_dc_debug_manual_entry_trigger = { TYPE = par }
			}
			NOT = { has_variable = hd_dc_par_cooldown }
			NOT = { has_variable = hd_dc_acute_instance_id }
			NOT = { has_variable = hd_dc_pending_instance }
		}
		# Spend the attempt before rechecking the live organizations.
		set_variable = { name = hd_dc_par_pair_draw_year value = current_year }
		if = {
			limit = { hd_dc_par_snapshot_valid_trigger = yes }
			clear_saved_scope = hd_dc_par_pro
			clear_saved_scope = hd_dc_par_adv
			clear_saved_scope = hd_dc_par_exp
			clear_saved_scope = hd_dc_par_con
			every_participant_group = {
				limit = { hd_dc_par_group_qualified_trigger = yes }
				switch = {
					trigger = participant_group_type
					pro_hegemon_movement = { save_scope_as = hd_dc_par_pro }
					advancement_movement = { save_scope_as = hd_dc_par_adv }
					expansion_movement = { save_scope_as = hd_dc_par_exp }
					conservative_movement = { save_scope_as = hd_dc_par_con }
				}
			}
			# Four native types have exactly six unordered pairs. Each legal
			# pair has one ticket, independent of its members or party power.
			random_list = {
				100 = {
					trigger = { hd_dc_par_named_pair_trigger = { FIRST = scope:hd_dc_par_pro SECOND = scope:hd_dc_par_adv } }
					hd_dc_save_selected_party_pair_effect = { FIRST = scope:hd_dc_par_pro SECOND = scope:hd_dc_par_adv }
				}
				100 = {
					trigger = { hd_dc_par_named_pair_trigger = { FIRST = scope:hd_dc_par_pro SECOND = scope:hd_dc_par_exp } }
					hd_dc_save_selected_party_pair_effect = { FIRST = scope:hd_dc_par_pro SECOND = scope:hd_dc_par_exp }
				}
				100 = {
					trigger = { hd_dc_par_named_pair_trigger = { FIRST = scope:hd_dc_par_pro SECOND = scope:hd_dc_par_con } }
					hd_dc_save_selected_party_pair_effect = { FIRST = scope:hd_dc_par_pro SECOND = scope:hd_dc_par_con }
				}
				100 = {
					trigger = { hd_dc_par_named_pair_trigger = { FIRST = scope:hd_dc_par_adv SECOND = scope:hd_dc_par_exp } }
					hd_dc_save_selected_party_pair_effect = { FIRST = scope:hd_dc_par_adv SECOND = scope:hd_dc_par_exp }
				}
				100 = {
					trigger = { hd_dc_par_named_pair_trigger = { FIRST = scope:hd_dc_par_adv SECOND = scope:hd_dc_par_con } }
					hd_dc_save_selected_party_pair_effect = { FIRST = scope:hd_dc_par_adv SECOND = scope:hd_dc_par_con }
				}
				100 = {
					trigger = { hd_dc_par_named_pair_trigger = { FIRST = scope:hd_dc_par_exp SECOND = scope:hd_dc_par_con } }
					hd_dc_save_selected_party_pair_effect = { FIRST = scope:hd_dc_par_exp SECOND = scope:hd_dc_par_con }
				}
			}
		}
	}
}

hd_dc_save_selected_party_pair_effect = {
	$FIRST$ = { save_scope_as = hd_dc_par_selected_first_group }
	$SECOND$ = { save_scope_as = hd_dc_par_selected_second_group }
}

hd_dc_snapshot_party_pool_effect = {
	if = {
		limit = { NOT = { var:hd_dc_par_snapshot_year = current_year } }
		set_variable = { name = hd_dc_par_snapshot_year value = current_year }
		set_variable = { name = hd_dc_par_total_power value = 0 }
		set_variable = { name = hd_dc_par_power_readable value = yes }
		set_variable = { name = hd_dc_par_native_group_count value = 0 }
		set_variable = { name = hd_dc_par_qualified_group_count value = 0 }
		set_variable = { name = hd_dc_par_ordered_hostile_pair_count value = 0 }
		remove_variable = hd_dc_par_snapshot_ruler
		remove_variable = hd_dc_par_sample_first_type
		remove_variable = hd_dc_par_sample_second_type
		remove_variable = hd_dc_par_sample_first_leader
		remove_variable = hd_dc_par_sample_second_leader
		set_variable = { name = hd_dc_par_snapshot_status value = flag:no_hegemon }
		if = {
			limit = { exists = title:h_china.holder title:h_china.holder = { is_alive = yes } }
			set_variable = { name = hd_dc_par_snapshot_ruler value = title:h_china.holder }
			every_participant_group = {
				limit = { hd_dc_par_native_movement_trigger = yes }
				save_scope_as = hd_dc_par_group
				set_variable = { name = hd_dc_par_snapshot_year value = current_year }
				set_variable = { name = hd_dc_par_office_members value = 0 }
				remove_variable = hd_dc_par_leader_snapshot
				remove_variable = hd_dc_par_power_snapshot
				clear_variable_list = hd_dc_par_member_snapshot
				switch = {
					trigger = participant_group_type
					pro_hegemon_movement = { set_variable = { name = hd_dc_par_group_type value = flag:pro_hegemon_movement } }
					advancement_movement = { set_variable = { name = hd_dc_par_group_type value = flag:advancement_movement } }
					expansion_movement = { set_variable = { name = hd_dc_par_group_type value = flag:expansion_movement } }
					conservative_movement = { set_variable = { name = hd_dc_par_group_type value = flag:conservative_movement } }
				}
				situation:dynastic_cycle = { change_variable = { name = hd_dc_par_native_group_count add = 1 } }
				every_situation_group_participant = {
					limit = { hd_dc_par_office_member_trigger = yes }
					save_scope_as = hd_dc_par_member
					scope:hd_dc_par_group = {
						change_variable = { name = hd_dc_par_office_members add = 1 }
						add_to_variable_list = { name = hd_dc_par_member_snapshot target = scope:hd_dc_par_member }
					}
				}
				if = { limit = { exists = var:movement_leader } set_variable = { name = hd_dc_par_leader_snapshot value = var:movement_leader } }
				if = {
					limit = { has_variable = movement_power var:movement_power >= 0 }
					set_variable = { name = hd_dc_par_power_snapshot value = var:movement_power }
					situation:dynastic_cycle = { change_variable = { name = hd_dc_par_total_power add = scope:hd_dc_par_group.var:hd_dc_par_power_snapshot } }
				}
				else = {
					# A missing/negative native value has no proven comparable
					# share. Preserve the evidence gap instead of clipping it.
					situation:dynastic_cycle = { set_variable = { name = hd_dc_par_power_readable value = no } }
				}
			}
			if = {
				limit = { var:hd_dc_par_native_group_count < 2 }
				set_variable = { name = hd_dc_par_snapshot_status value = flag:insufficient_native_groups }
			}
			else_if = {
				limit = { OR = { var:hd_dc_par_power_readable = no var:hd_dc_par_total_power <= 0 } }
				set_variable = { name = hd_dc_par_snapshot_status value = flag:unreadable_comparable_power }
			}
			else = {
				every_participant_group = {
					limit = { hd_dc_par_group_qualified_trigger = yes }
					save_scope_as = hd_dc_par_first_group
					situation:dynastic_cycle = {
						change_variable = { name = hd_dc_par_qualified_group_count add = 1 }
						every_participant_group = {
							limit = { hd_dc_par_hostile_pair_trigger = yes }
							save_scope_as = hd_dc_par_second_group
							situation:dynastic_cycle = {
								change_variable = { name = hd_dc_par_ordered_hostile_pair_count add = 1 }
								if = {
									limit = { NOT = { has_variable = hd_dc_par_sample_first_type } }
									# A diagnostic first pair, not a weighted BOOT choice.
									set_variable = { name = hd_dc_par_sample_first_type value = scope:hd_dc_par_first_group.var:hd_dc_par_group_type }
									set_variable = { name = hd_dc_par_sample_second_type value = scope:hd_dc_par_second_group.var:hd_dc_par_group_type }
									set_variable = { name = hd_dc_par_sample_first_leader value = scope:hd_dc_par_first_group.var:hd_dc_par_leader_snapshot }
									set_variable = { name = hd_dc_par_sample_second_leader value = scope:hd_dc_par_second_group.var:hd_dc_par_leader_snapshot }
								}
							}
						}
					}
				}
				if = {
					limit = { var:hd_dc_par_ordered_hostile_pair_count > 0 }
					set_variable = { name = hd_dc_par_snapshot_status value = flag:organizational_pair_found }
				}
				else = { set_variable = { name = hd_dc_par_snapshot_status value = flag:no_qualified_hostile_pair } }
			}
		}
	}
}
