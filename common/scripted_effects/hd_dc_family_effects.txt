# NEG-17: the existing spouse dispute roll, after native personality terms.
# A lover/soulmate outside marriage does not become kin by this adapter.
hd_dc_prepare_spouse_dispute_roll_effect = {
	save_scope_value_as = { name = hd_dc_family_chance value = $BASE$ }
	if = {
		limit = {
			exists = situation:dynastic_cycle
			any_character_situation = { this = situation:dynastic_cycle }
			is_alive = yes
			exists = $CHARACTER$
			this != $CHARACTER$
			$CHARACTER$ = { is_alive = yes }
			is_spouse_of = $CHARACTER$
		}
		save_scope_value_as = { name = hd_dc_hazard_base value = scope:hd_dc_family_chance }
		save_scope_value_as = { name = hd_dc_hazard_multiplier value = hd_dc_social_hazard_multiplier_value }
		save_scope_value_as = { name = hd_dc_family_chance value = hd_dc_hazard_probability_value }
	}
}
