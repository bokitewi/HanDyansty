# ROOT = living Sun Jian, only called from an eligible battle phase outcome.
hds_nr_battle_death_effect = {
    if = {
        limit = { is_alive = yes this = character:sun_jian exists = scope:combat_side }
        save_scope_as = hds_nr_casualty
        clear_saved_scope = hds_nr_killer
        save_died_in_battle_variables_effect = {
            ENEMY = scope:combat_side.enemy_side.side_primary_participant
            LEADER = scope:combat_side.side_primary_participant
            LOCATION = scope:combat_side.combat.location
        }
        scope:combat_side.enemy_side = {
            random_side_knight = {
                limit = {
                    is_alive = yes
                    this.prowess >= scope:hds_nr_casualty.prowess_for_combat_event_opponent
                }
                save_scope_as = hds_nr_killer
                add_prestige = knight_prestige_gain_on_kill_inverse
                knight_increase_prowess_chance_effect = yes
            }
        }
        if = {
            limit = { exists = scope:hds_nr_killer scope:hds_nr_killer ?= { is_alive = yes } }
            scope:combat_side = {
                battle_event = {
                    key = "$ROLE$_killed_by_enemy"
                    left_portrait = scope:hds_nr_casualty
                    right_portrait = scope:hds_nr_killer
                    type = death
                }
            }
            death = { death_reason = death_battle killer = scope:hds_nr_killer }
        }
        else = {
            scope:combat_side = {
                battle_event = {
                    key = "$ROLE$_killed_no_enemy"
                    left_portrait = scope:hds_nr_casualty
                    type = death
                }
            }
            death = { death_reason = death_battle }
        }
    }
}
