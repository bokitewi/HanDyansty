# NEG-17/18: prepare one probability for an existing hostile relation branch.
# BASE already includes every native chance modifier. This does not choose an
# opponent, change an AI option weight, or replace the native relation effect.
# Actual kin/spouses qualify for family disputes; non-kin require old grievance.
# A relative who is also a rival still receives F once in this shared roll.
hd_dc_prepare_rivalry_roll_effect = {
	save_scope_value_as = { name = hd_dc_rivalry_chance value = $BASE$ }
	if = {
		limit = {
			exists = situation:dynastic_cycle
			any_character_situation = { this = situation:dynastic_cycle }
			is_alive = yes
			exists = $CHARACTER$
			this != $CHARACTER$
			$CHARACTER$ = { is_alive = yes }
			OR = {
				is_close_or_extended_family_of = $CHARACTER$
				is_spouse_of = $CHARACTER$
				opinion = { target = $CHARACTER$ value < 0 }
				has_relation_potential_rival = $CHARACTER$
				has_relation_grudge = $CHARACTER$
				has_relation_rival = $CHARACTER$
				has_relation_nemesis = $CHARACTER$
			}
		}
		save_scope_value_as = { name = hd_dc_hazard_base value = scope:hd_dc_rivalry_chance }
		save_scope_value_as = { name = hd_dc_hazard_multiplier value = hd_dc_social_hazard_multiplier_value }
		save_scope_value_as = { name = hd_dc_rivalry_chance value = hd_dc_hazard_probability_value }
	}
}
