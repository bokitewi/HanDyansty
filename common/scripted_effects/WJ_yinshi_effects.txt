# ---------------------------------------------------------------------------
# 进入隐居：辞官 → 特质 → 拒任期 → 禁止任骑士 → 开启故事
# ---------------------------------------------------------------------------
WJ_yinshi_enter_effect = {
 if = {
  limit = { NOT = { has_trait = WJ_yangwang_trait } }
  WJ_yinshi_resign_effect = yes
  add_trait = WJ_yangwang_trait
  if = {
   limit = { appointment_timeout_days = 0 }
   set_appointment_timeout = { years = 150 desc = appointment_timeout_desc_refusing }
   add_character_flag = wj_yangwang_timeout_owned
  }
  else = {
   # 已有外部拒任期时不覆盖，到期后由 WJ_yangwang_timeout.0001 接管。
   trigger_event = { id = WJ_yangwang_timeout.0001 days = WJ_yangwang_timeout_handoff_days }
  }
  set_knight_status = forbid
  if = {
   limit = { NOT = { any_owned_story = { story_type = story_WJ_yinshi } } }
   create_story = story_WJ_yinshi
  }
 }
}

# 辞官：离开幕府（作为幕僚离开、作为府主解散），离开议会/部长，卸下统兵，
# 男爵级以上非家族头衔连同附庸交还直属上级。
# 宫廷职位由 base_court_position_validity_trigger 等的隐士条件自动失效。
WJ_yinshi_resign_effect = {
 save_scope_as = wj_yinshi_resigner
 WJ_clear_han_jin_mufu_state_effect = yes
 every_liege_or_above = {
  limit = { scope:wj_yinshi_resigner = { is_councillor_of = prev } }
  fire_councillor = scope:wj_yinshi_resigner
 }
 commanding_army ?= { remove_commander = yes }
 if = {
  limit = {
   exists = liege
   WJ_yinshi_holds_office_title_trigger = yes
  }
  liege = { save_scope_as = wj_yinshi_title_receiver }
  create_title_and_vassal_change = {
   type = returned
   save_scope_as = wj_yinshi_change
   add_claim_on_loss = no
  }
  every_held_title = {
   limit = { is_noble_family_title = no }
   change_title_holder = {
    holder = scope:wj_yinshi_title_receiver
    change = scope:wj_yinshi_change
    take_baronies = yes
   }
  }
  every_vassal = {
   change_liege = {
    liege = scope:wj_yinshi_title_receiver
    change = scope:wj_yinshi_change
   }
  }
  resolve_title_and_vassal_change = scope:wj_yinshi_change
 }
}

    WJ_yinshi_inquiry_open_trigger = yes
    WJ_yinshi_local_open_trigger = yes
    AND = { exists = scope:wj_yinshi_friend WJ_yinshi_friend_open_trigger = yes }
   }
  }
  random_list = {
   25 = {
    trigger = { WJ_yinshi_learning_open_trigger = yes }
    WJ_yinshi_pick_learning_effect = yes
   }
   25 = {
    trigger = { WJ_yinshi_inquiry_open_trigger = yes }
    WJ_yinshi_pick_inquiry_effect = yes
   }
   25 = {
    trigger = { WJ_yinshi_local_open_trigger = yes }
    WJ_yinshi_pick_local_effect = yes
   }
   25 = {
    trigger = { exists = scope:wj_yinshi_friend WJ_yinshi_friend_open_trigger = yes }
    WJ_yinshi_pick_friend_effect = yes
   }
  }
 }
}

# 朋友人选：朋友关系 → 近亲 → 直属上级宫廷中好感 ≥20 者。不做全图搜索。
WJ_yinshi_find_friend_effect = {
 random_relation = {
  type = friend
  limit = { WJ_yinshi_companion_ok_trigger = yes }
  save_scope_as = wj_yinshi_friend
 }
 if = {
  limit = { NOT = { exists = scope:wj_yinshi_friend } }
  random_close_family_member = {
   limit = { WJ_yinshi_companion_ok_trigger = yes }
   save_scope_as = wj_yinshi_friend
  }
 }
 if = {
  limit = { NOT = { exists = scope:wj_yinshi_friend } exists = liege }
  liege = {
   random_courtier = {
    limit = {
     WJ_yinshi_companion_ok_trigger = yes
     opinion = { target = scope:wj_yinshi_subject value >= 20 }
    }
    save_scope_as = wj_yinshi_friend
   }
  }
 }
}

# ---------------------------------------------------------------------------
# 入幕邀请（WJ_yinshi.5001）。scope:actor = 府主，scope:recipient = 隐士。
# ---------------------------------------------------------------------------
# 故事发起：基础影响力在接受时一次结清，无额外许诺。
WJ_yinshi_mufu_story_accept_effect = {
 WJ_mufu_check_chain = yes
 if = {
  limit = { scope:wj_chain_safe = 1 WJ_mufu_request_valid = yes WJ_mufu_selected_empty = yes }
  WJ_mufu_attach = yes
  if = {
   limit = { has_variable = wj_mufu_owner var:wj_mufu_owner = scope:actor }
   scope:actor = { change_influence = { value = WJ_mufu_base_influence_cost multiply = -1 } }
  }
 }
}
# 玩家府主通过"辟召"交互发起：沿用交互原有的接受结算（含许诺条件）。
WJ_yinshi_mufu_interaction_accept_effect = {
 if = {
  limit = { WJ_mufu_request_valid = yes WJ_mufu_selected_empty = yes WJ_mufu_terms_valid = yes }
  WJ_mufu_check_chain = yes
  if = {
   limit = { scope:wj_chain_safe = 1 }
   scope:actor = { change_influence = { value = WJ_mufu_base_influence_cost multiply = -1 } }
   WJ_mufu_settle_terms = yes
   WJ_mufu_attach = yes
   if = {
    limit = { scope:actor = { is_ai = no } scope:recipient = { has_variable = wj_mufu_owner var:wj_mufu_owner ?= scope:actor } }
    scope:actor = { trigger_event = WJ_bizhao_reply.0001 }
   }
  }
 }
}
# AI 府主通过 WJ_ai_bizhao_01 发起：隐士接受后直接入幕，不结算许诺。

  WJ_mufu_check_chain = yes

   limit = { scope:wj_chain_safe = 1 }
   WJ_mufu_attach = yes
   if = {
    limit = { scope:recipient = { exists = var:wj_mufu_owner var:wj_mufu_owner = scope:actor } }
    scope:actor = { set_variable = { name = wj_ai_bizhao_month value = 1 months = 1 } }
   }
  }
 }
}
