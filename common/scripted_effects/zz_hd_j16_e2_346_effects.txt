			NOT = { has_global_variable = hd_nbc_fashu_done }

			set_global_variable = hd_j16_fashu_skip

				limit = { exists = scope:hd_j16_fs_huan }
				clear_saved_scope = hd_j16_fs_huan
			}

			}

hd_j16_fashu_check_effect = {
	if = {
		limit = {
			has_global_variable = hd_nbc_fashu_active
			NOT = { has_global_variable = hd_nbc_fashu_done }
			NOT = { has_global_variable = hd_nbc_fashu_won }
			NOT = { has_global_variable = hd_nbc_fashu_lost }
			NOT = { has_global_variable = hd_nbc_fashu_white }
			NOT = { has_global_variable = hd_nbc_fashu_invalid }
		}
		if = {
			limit = { exists = scope:hd_j16_fsc_huan }
			clear_saved_scope = hd_j16_fsc_huan
		}
		if = {
			limit = { exists = scope:hd_j16_fsc_li }
			clear_saved_scope = hd_j16_fsc_li
		}
		hd_j16_resolve_effect = { ID = longkang_huan_017 AS = hd_j16_fsc_huan }
		hd_j16_resolve_effect = { ID = bacong_li_026 AS = hd_j16_fsc_li }
		if = {
			limit = {
				NOT = { global_var:hd_j16_fashu_att ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
				NOT = { global_var:hd_j16_fashu_def ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
				NOT = { scope:hd_j16_fsc_huan ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
				NOT = { scope:hd_j16_fsc_li ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
				NOT = { title:k_uuii_yizhou.holder ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
				NOT = { title:k_uuii_jingzhou.holder ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
				NOT = { title:e_bashu.holder ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:fashu } } }
			}
			remove_global_variable = hd_nbc_fashu_active
			if = {
				limit = {
					exists = title:h_china.holder
					title:c_uuii_p_4282.holder ?= {
						OR = {
							this = title:h_china.holder
							top_liege = title:h_china.holder
						}
					}
				}
				set_global_variable = hd_nbc_fashu_won
				debug_log = "HD_J16 e2_j91 FIRE fallback_won"
			}
			else = {
				set_global_variable = hd_nbc_fashu_lost
				debug_log = "HD_J16 e2_j91 FIRE fallback_lost"
			}
		}
	}
	if = {
		limit = {
			NOT = { has_global_variable = hd_nbc_fashu_done }
			OR = {
				has_global_variable = hd_nbc_fashu_won
				has_global_variable = hd_nbc_fashu_lost
				has_global_variable = hd_nbc_fashu_white
				has_global_variable = hd_nbc_fashu_invalid
			}
		}
		set_global_variable = hd_nbc_fashu_done
		if = {
			limit = { has_global_variable = hd_nbc_fashu_won }
			if = {
				limit = { exists = scope:hd_j16_fsc_huan }
				clear_saved_scope = hd_j16_fsc_huan
			}
			hd_j16_resolve_effect = { ID = longkang_huan_017 AS = hd_j16_fsc_huan }
			if = {
				limit = { scope:hd_j16_fsc_huan ?= { is_alive = yes } }
				scope:hd_j16_fsc_huan = { trigger_event = hd_nbc.0110 }
				debug_log = "HD_J16 e2_j91 FIRE reward_sent"
			}
			else = {
				debug_log = "HD_J16 e2_j91 SKIP no_actor"
			}
		}
		else = {
			debug_log = "HD_J16 e2_j91 SKIP not_won"
		}
	}
}

hd_j16_node_shihu_di_effect = {
	if = {
		limit = { NOT = { has_global_variable = hd_j16_shihu_di_done } }
		if = {
			limit = { global_var:hd_j16_start_year >= 370 }
			set_global_variable = hd_j16_shihu_di_done
			debug_log = "HD_J16 e2_z95 SKIP prestart"
		}
		else_if = {
			limit = { current_date >= 349.1.1 }
			if = {
				limit = { current_date >= 349.5.1 }
				hd_j16_304_skip_effect = { KEY = shihu_di LOG = e2_z95 WHY = window_closed }
			}
			else_if = {
				limit = { hd_j16_dead_trigger = { ID = shangdang_shi_010 } }
				hd_j16_304_skip_effect = { KEY = shihu_di LOG = e2_z95 WHY = actor_dead }
			}
			else = {
				hd_j16_b4_res_zhao_effect = { AS = hd_j16_sd_zhao }
				hd_j16_304_res_effect = { ID = shangdang_shi_010 AS = hd_j16_sd_hu }
				if = {
					limit = {
						exists = scope:hd_j16_sd_zhao
						scope:hd_j16_sd_hu ?= { this = scope:hd_j16_sd_zhao }
						scope:hd_j16_sd_zhao.primary_title ?= { NOT = { var:hd_nbc_style ?= flag:huangdi } }
					}
					set_global_variable = hd_j16_shihu_di_done
					if = {
						limit = { scope:hd_j16_sd_hu = { is_ai = yes } }
						scope:hd_j16_sd_hu = { hd_j16_shihu_di_effect = yes }
					}
					else = {
						scope:hd_j16_sd_hu = { trigger_event = hd_j16.1004 }
						debug_log = "HD_J16 e2_z95 WAIT player_asked"
					}
				}
				else = {
					hd_j16_304_wait_effect = { KEY = shihu_di LOG = e2_z95 WHY = not_ready }
				}
			}
		}
	}
}
