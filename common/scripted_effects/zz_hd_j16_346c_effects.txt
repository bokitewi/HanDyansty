# 投靠 $LORD$：本人是统治者（含只持世族头衔者）时改宗主；否则入其宫廷。与 hd_nbc_swear_fealty_effect 的区别：不要求 is_landed（刘玄只持世族头衔）。
# 被收编者最高头衔须低于新主（报错清单），不满足时不动并记日志。Scope：人物
hd_j16_346c_join_lord_effect = {
	save_scope_as = hd_j16_c4_jl_who
	if = {
		limit = {
			is_ruler = yes
			NOT = { this = $LORD$ }
			NOT = { liege ?= $LORD$ }
			highest_held_title_tier < $LORD$.highest_held_title_tier
		}
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd_j16_c4_jl_change add_claim_on_loss = no }
		change_liege = { liege = $LORD$ change = scope:hd_j16_c4_jl_change }
		resolve_title_and_vassal_change = scope:hd_j16_c4_jl_change
	}
	else_if = {
		limit = {
			is_ruler = no
			NOT = { employer ?= $LORD$ }
		}
		$LORD$ = { add_courtier = scope:hd_j16_c4_jl_who }
	}
	else_if = {
		limit = {
			is_ruler = yes
			NOT = { this = $LORD$ }
			NOT = { liege ?= $LORD$ }
		}
		debug_log = "HD_J16 346c WAIT join_lord_tier_fail"
	}
}

# 内战（玩家被夺位、或玩家选「讨之」时用；A1：不直接夺玩家头衔）：$REBEL$ 脱离 $TARGET$ 独立，得 $TARGET$ 主头衔的宣称并以宣称开战；
# $REBEL$ 无地时只给宣称、离开宫廷。Scope：任意
hd_j16_346c_civil_war_effect = {
	hd_j16_dir_clear_scope_effect = { S = hd_j16_c4_cw_reb }
	hd_j16_dir_clear_scope_effect = { S = hd_j16_c4_cw_tgt }
	hd_j16_dir_clear_scope_effect = { S = hd_j16_c4_cw_title }
	$REBEL$ = { save_scope_as = hd_j16_c4_cw_reb }
	$TARGET$ = { save_scope_as = hd_j16_c4_cw_tgt }
	scope:hd_j16_c4_cw_tgt.primary_title ?= { save_scope_as = hd_j16_c4_cw_title }
	if = {
		limit = {
			exists = scope:hd_j16_c4_cw_title
			scope:hd_j16_c4_cw_reb ?= {
				is_alive = yes
				is_landed = yes
				NOT = { this = scope:hd_j16_c4_cw_tgt }
			}
		}
		scope:hd_j16_c4_cw_reb = {
			hd_nbc_become_independent_effect = yes
			if = {
				limit = { NOT = { has_claim_on = scope:hd_j16_c4_cw_title } }
				add_pressed_claim = scope:hd_j16_c4_cw_title
			}
			if = {
				limit = {
					is_independent_ruler = yes
					NOT = { is_at_war_with = scope:hd_j16_c4_cw_tgt }
				}
				start_war = {
					cb = claim_cb
					target = scope:hd_j16_c4_cw_tgt
					claimant = scope:hd_j16_c4_cw_reb
					target_title = scope:hd_j16_c4_cw_title
				}
			}
		}
		debug_log = "HD_J16 346c FIRE civil_war"
	}
	else_if = {
		limit = {
			exists = scope:hd_j16_c4_cw_title
			scope:hd_j16_c4_cw_reb ?= {
				is_alive = yes
				is_ruler = no
				NOT = { this = scope:hd_j16_c4_cw_tgt }
			}
		}
		scope:hd_j16_c4_cw_reb = {
			if = {
				limit = { NOT = { has_claim_on = scope:hd_j16_c4_cw_title } }
				add_pressed_claim = scope:hd_j16_c4_cw_title
			}
			move_to_pool = yes
		}
		debug_log = "HD_J16 346c FIRE civil_flee"
	}
}
