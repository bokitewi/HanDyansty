# 安置一人。WHO 在世、无地、不是统治者（持世族头衔等无地头衔者也算统治者，add_courtier 会失败，报错清单 set_employer 一条）；
# COURT 在世、有地、不是本人；WHO 还没有雇主，或 CROSS = yes 时 WHO 已成年、其雇主不在 COURT 的最高领主治下。
hd_j16_370c_place_effect = {
	if = { limit = { exists = scope:hd_j16_370c_court } clear_saved_scope = hd_j16_370c_court }
	if = { limit = { exists = scope:hd_j16_370c_top } clear_saved_scope = hd_j16_370c_top }
	if = { limit = { exists = scope:hd_j16_370c_who } clear_saved_scope = hd_j16_370c_who }
	character:$COURT$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
			}
			save_scope_as = hd_j16_370c_court
			top_liege = { save_scope_as = hd_j16_370c_top }
		}
	}
	character:$WHO$ ?= {
		if = {
			limit = {
				exists = scope:hd_j16_370c_court
				exists = scope:hd_j16_370c_top
				is_alive = yes
				is_landed = no
				is_ruler = no
				NOT = { this = scope:hd_j16_370c_court }
				NOT = { employer ?= scope:hd_j16_370c_court }
				OR = {
					NOT = { exists = employer }
					AND = {
						always = $CROSS$
						is_adult = yes
						NOT = { employer ?= { top_liege = scope:hd_j16_370c_top } }
					}
				}
			}
			save_scope_as = hd_j16_370c_who
			scope:hd_j16_370c_court = { add_courtier = scope:hd_j16_370c_who }
			if = {
				limit = {
					always = $KNIGHT$
					is_adult = yes
					employer ?= scope:hd_j16_370c_court
				}
				set_knight_status = force
			}
			debug_log = "HD_J16 startup_370 FIRE place_$WHO$"
		}
		else = {
			debug_log = "HD_J16 startup_370 SKIP place_$WHO$"
		}
	}
}

# 改投一人：WHO 在世、无地、不是统治者，雇主是 FROM 或没有雇主时改入 COURT 宫廷（COURT 在世、有地）。
hd_j16_370c_move_effect = {
	if = { limit = { exists = scope:hd_j16_370c_court } clear_saved_scope = hd_j16_370c_court }
	if = { limit = { exists = scope:hd_j16_370c_who } clear_saved_scope = hd_j16_370c_who }
	character:$COURT$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
			}
			save_scope_as = hd_j16_370c_court
		}
	}
	character:$WHO$ ?= {
		if = {
			limit = {
				exists = scope:hd_j16_370c_court
				is_alive = yes
				is_landed = no
				is_ruler = no
				NOT = { this = scope:hd_j16_370c_court }
				NOT = { employer ?= scope:hd_j16_370c_court }
				OR = {
					NOT = { exists = employer }
					employer ?= character:$FROM$
				}
			}
			save_scope_as = hd_j16_370c_who
			scope:hd_j16_370c_court = { add_courtier = scope:hd_j16_370c_who }
			debug_log = "HD_J16 startup_370 FIRE move_$WHO$"
		}
		else = {
			debug_log = "HD_J16 startup_370 SKIP move_$WHO$"
		}
	}
}
