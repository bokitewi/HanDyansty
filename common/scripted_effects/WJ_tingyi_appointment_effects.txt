# Topic 3 backend. Every public effect runs on the UI operator, never the nominee.
# Parent hooks: clear on close; snapshot after office/target selection; valid at confirmation;
# commit after WJ_ty_execute_valid. Success scope is wj_ty_appointment_success (yes/no).
# Target scope, after native dismissal. Destroy only the reviewed ministry title
# so the delayed handoff cannot refill this deliberately vacant seat. Unlike
# destroy_and_fire_minister_effect, this does not copy the target's treasury.
# The caller owns political approval; this leaf never retires land or private titles.
WJ_ty_appointment_finish_ministry = {
 save_scope_as = wj_ty_departing_minister
 save_scope_value_as = { name = wj_ty_office_released value = no }
 if = {
  limit = {
   exists = scope:wj_ty_emperor
   NOT = { has_council_position = $POSITION$ }
   NOT = { is_councillor_of = scope:wj_ty_emperor }
  }
  if = { limit = { has_title = title:$TITLE$ } scope:wj_ty_emperor = { destroy_title = title:$TITLE$ } }
  if = {
   limit = { NOT = { has_title = title:$TITLE$ } }
   
   
   
   if = { limit = { title:$TITLE$ = title:e_minister_grand_marshal } remove_character_flag = officer_da_jiang_jun }
   if = { limit = { hd_holds_any_ministry_title_trigger = no } remove_character_modifier = minister_scheme_slots_modifier }
   WJ_reconcile_contract_privileges_effect = yes
   scope:wj_ty_emperor = { WJ_sync_grand_secretariat_effect = yes }
   save_scope_value_as = { name = wj_ty_office_released value = yes }
  }
 }
}
# Target scope. Native fire retains on_fired/on_lose and dependent-post invalidation.
# Never remove an office title when the engine refused to release the actual seat.
WJ_ty_appointment_release_ministry = {
 save_scope_as = wj_ty_departing_minister
 save_scope_value_as = { name = wj_ty_office_released value = no }
 if = {
  limit = { WJ_ty_appointment_dismiss_council = { TITLE = $TITLE$ POSITION = $POSITION$ } }
  clear_saved_scope = new_councillor
  clear_saved_scope = swapped_position
  # Native tooltip evaluation does not fire the councillor. The subsequent
  # finish guard therefore cannot expose the title loss until after execution.
  # Describe that exact reviewed title here; show_as_tooltip never destroys it.
  show_as_tooltip = {
   if = { limit = { has_title = title:$TITLE$ }
    scope:wj_ty_emperor = { destroy_title = title:$TITLE$ }
   }
  }
  if = { limit = { scope:wj_ty_departing_minister = { WJ_ty_imperial_release_context = yes } }
   # Same native final primitive used by destroy_and_fire_minister_effect,
   # without copying treasury or deleting any unrelated title.
   scope:wj_ty_emperor = { always_fire_councillor = scope:wj_ty_departing_minister }
  }
  else = { scope:wj_ty_emperor = { fire_councillor = scope:wj_ty_departing_minister } }
  WJ_ty_appointment_finish_ministry = { TITLE = $TITLE$ POSITION = $POSITION$ }
 }
}

     trigger_if = { limit = { var:wj_ty_office = 8 } scope:wj_ty_target = { NOT = { has_court_position = favored_minister_court_position } } }
     trigger_else = { scope:wj_ty_target = { NOT = { has_court_position = chief_eunuch_court_position } } }

  if = { limit = { scope:wj_ty_office_released = yes } save_scope_value_as = { name = wj_ty_commit_ok value = yes } }

}

# Native routes: petition_liege_0101 recruits ordinary courtiers before assigning;
# fill_the_ministry calls got_minister_position_effect for rulers. The shared
# ministry effect owns appointment/title retirement/vassal handling, not this UI.
WJ_ty_appointment_assign_council = {
 if = { limit = { scope:wj_ty_target = { is_ruler = yes } }
  scope:wj_ty_emperor = { save_scope_as = councillor_liege }
  scope:wj_ty_target = { got_minister_position_effect = { MINISTER_TITLE = $TITLE$ MINISTER_POSITION = $POSITION$ } }
 }
 else_if = { limit = { scope:wj_ty_target = { NOT = { is_courtier_of = scope:wj_ty_emperor } } }
  scope:wj_ty_emperor = { add_courtier = scope:wj_ty_target }
 }
 # on_title_gain/shared ministry may already have assigned the seat. Its internal
 # has_council_position check compares only the position key, so a local incumbent
 # of the same kind still needs assignment to the actual imperial council.
 if = { limit = {
  scope:wj_ty_target = {
   NOT = { AND = { has_council_position = $POSITION$ is_councillor_of = scope:wj_ty_emperor } }
   OR = { is_ruler = no has_title = title:$TITLE$ }
   WJ_ty_appointment_ready = { TITLE = $TITLE$ POSITION = $POSITION$ }
  }
 }
  scope:wj_ty_emperor = { assign_councillor_type = { type = $POSITION$ target = scope:wj_ty_target remove_existing_councillor = yes fire_on_actions = yes } }
 }
 if = { limit = { WJ_ty_appointment_destination_valid = { TITLE = $TITLE$ POSITION = $POSITION$ } }
  save_scope_value_as = { name = wj_ty_appointment_success value = yes }
 }
}

# Quiet native seat writes + one exact two-title transaction. No normal ministry
# transfer/retirement is invoked for either participant. The parent owns approval.
WJ_ty_appointment_swap_apply = {
 if = { limit = { WJ_ty_execute_valid = yes WJ_ty_appointment_valid = yes var:wj_ty_appointment_mode = 2 }
  # Transient markers also guard project callbacks if an engine version runs
  # position hooks despite quiet assignment. One-day expiry is a failure fallback.
  scope:wj_ty_target = { set_variable = { name = wj_ty_swapping_council value = 1 days = 1 } }
  var:wj_ty_incumbent = { set_variable = { name = wj_ty_swapping_council value = 1 days = 1 } }
  var:wj_ty_appointment_source_task = { save_scope_as = swapped_position }
  # fire_on_actions=no isolates the intermediate seat state from leave/hire events.
  scope:wj_ty_emperor = {
   always_fire_councillor_skip_effects = scope:wj_ty_target
   always_fire_councillor_skip_effects = scope:wj_ty_operator.var:wj_ty_incumbent
  }
  # Native valid_character must see the actual new title before seating its holder.
  create_title_and_vassal_change = { type = appointment save_scope_as = wj_ty_pair_change add_claim_on_loss = no }
  var:wj_ty_appointment_source_title = { change_title_holder = { holder = scope:wj_ty_operator.var:wj_ty_incumbent change = scope:wj_ty_pair_change } }
  var:wj_ty_appointment_destination_title = { change_title_holder = { holder = scope:wj_ty_target change = scope:wj_ty_pair_change } }
  resolve_title_and_vassal_change = scope:wj_ty_pair_change
  if = { limit = {
   var:wj_ty_appointment_source_title.holder = var:wj_ty_incumbent
   var:wj_ty_appointment_destination_title.holder = var:wj_ty_target
  }
   scope:wj_ty_emperor = {
    assign_council_task = { council_task = scope:wj_ty_operator.var:wj_ty_appointment_destination_task target = scope:wj_ty_target fire_on_actions = no }
    assign_council_task = { council_task = scope:wj_ty_operator.var:wj_ty_appointment_source_task target = scope:wj_ty_operator.var:wj_ty_incumbent fire_on_actions = no }
   }
  }
  if = { limit = { WJ_ty_swap_result_reached = yes }
   clear_saved_scope = swapped_position
   scope:wj_ty_emperor = { save_scope_as = councillor_liege }
   scope:wj_ty_target = { WJ_ty_swap_refresh_member = yes }
   var:wj_ty_incumbent = { WJ_ty_swap_refresh_member = yes }
  }
  if = { limit = { WJ_ty_swap_result_reached = yes } save_scope_value_as = { name = wj_ty_appointment_success value = yes } }
  else = {
   # Restore the reviewed pair on native refusal; never publish a partial swap.
   if = { limit = { NOT = { AND = {
    var:wj_ty_appointment_source_title.holder = var:wj_ty_target
    var:wj_ty_appointment_destination_title.holder = var:wj_ty_incumbent
   } } }
    create_title_and_vassal_change = { type = appointment save_scope_as = wj_ty_pair_restore add_claim_on_loss = no }
    var:wj_ty_appointment_source_title = { change_title_holder = { holder = scope:wj_ty_target change = scope:wj_ty_pair_restore } }
    var:wj_ty_appointment_destination_title = { change_title_holder = { holder = scope:wj_ty_operator.var:wj_ty_incumbent change = scope:wj_ty_pair_restore } }
    resolve_title_and_vassal_change = scope:wj_ty_pair_restore
   }
   if = { limit = { scope:wj_ty_target = { is_councillor_of = scope:wj_ty_emperor } } scope:wj_ty_emperor = { always_fire_councillor_skip_effects = scope:wj_ty_target } }
   if = { limit = { var:wj_ty_incumbent = { is_councillor_of = scope:wj_ty_emperor } } scope:wj_ty_emperor = { always_fire_councillor_skip_effects = scope:wj_ty_operator.var:wj_ty_incumbent } }
   scope:wj_ty_emperor = {
    assign_council_task = { council_task = scope:wj_ty_operator.var:wj_ty_appointment_source_task target = scope:wj_ty_target fire_on_actions = no }
    assign_council_task = { council_task = scope:wj_ty_operator.var:wj_ty_appointment_destination_task target = scope:wj_ty_operator.var:wj_ty_incumbent fire_on_actions = no }
   }
   scope:wj_ty_emperor = { save_scope_as = councillor_liege }
   clear_saved_scope = swapped_position
   scope:wj_ty_target = { WJ_ty_swap_refresh_member = yes }
   var:wj_ty_incumbent = { WJ_ty_swap_refresh_member = yes }
  }
  scope:wj_ty_target = { remove_variable = wj_ty_swapping_council }
  var:wj_ty_incumbent = { remove_variable = wj_ty_swapping_council }
  clear_saved_scope = swapped_position
  clear_saved_scope = wj_ty_pair_change
  clear_saved_scope = wj_ty_pair_restore
 }
}
WJ_ty_appointment_commit = {
 remove_variable = wj_ty_appointment_failed
 save_scope_value_as = { name = wj_ty_appointment_success value = no }
 if = { limit = { WJ_ty_execute_valid = yes WJ_ty_appointment_valid = yes }
  save_scope_as = wj_ty_operator
  var:wj_ty_emperor = { save_scope_as = wj_ty_emperor }
  var:wj_ty_target = { save_scope_as = wj_ty_target }
  clear_saved_scope = swapped_position
  clear_saved_scope = wj_ty_old_task
  scope:wj_ty_target.council_task ?= { save_scope_as = wj_ty_old_task }
  if = { limit = { var:wj_ty_appointment_mode = 2 } WJ_ty_appointment_swap_apply = yes }
  else_if = { limit = { var:wj_ty_office = 1 }
   # The special script owns grant + native transfer + regency. Do not resign the
   # old seat or clear the reviewed military holder before that script succeeds.
   scope:wj_ty_emperor = { save_scope_as = councillor_liege }
   scope:wj_ty_target = { got_minister_position_effect = { MINISTER_TITLE = e_WJ_minister_cheng_xiang MINISTER_POSITION = WJ_minister_cheng_xiang } }
   if = { limit = { WJ_ty_appointment_destination_valid = { TITLE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang } }
    if = { limit = { exists = var:wj_ty_appointment_zhongwai NOT = { var:wj_ty_appointment_zhongwai = scope:wj_ty_target } }
     var:wj_ty_appointment_zhongwai = { WJ_zw_clear = yes }
    }
    save_scope_value_as = { name = wj_ty_appointment_success value = yes }
   }
  }
  else_if = { limit = { var:wj_ty_office = 2 } WJ_ty_appointment_assign_council = { POSITION = councillor_chancellor TITLE = e_minister_chancellor } }
  else_if = { limit = { var:wj_ty_office = 3 } WJ_ty_appointment_assign_council = { POSITION = councillor_spymaster TITLE = e_minister_censor } }
  else_if = { limit = { var:wj_ty_office = 4 } WJ_ty_appointment_assign_council = { POSITION = councillor_marshal TITLE = e_minister_of_war } }
  
  else_if = { limit = { var:wj_ty_office = 6 } WJ_ty_appointment_assign_council = { POSITION = minister_grand_marshal TITLE = e_minister_grand_marshal } }
  else_if = { limit = { var:wj_ty_office = 7 } WJ_ty_appointment_assign_court = { POSITION = WJ_dasima_court_position } }
  else_if = { limit = { var:wj_ty_office = 8 } WJ_ty_appointment_assign_court = { POSITION = favored_minister_court_position } }
  else_if = { limit = { var:wj_ty_office = 9 } WJ_ty_appointment_assign_court = { POSITION = chief_eunuch_court_position } }
  else_if = { limit = { var:wj_ty_office = 10 } WJ_ty_appointment_assign_council = { POSITION = minister_personnel TITLE = e_minister_of_personnel } }
  
  
  else_if = { limit = { var:wj_ty_office = 13 } WJ_ty_appointment_assign_council = { POSITION = councillor_steward TITLE = e_minister_of_revenue } }
  else_if = { limit = { var:wj_ty_office = 14 } WJ_ty_appointment_assign_council = { POSITION = minister_justice TITLE = e_minister_of_justice } }
  else_if = { limit = { var:wj_ty_office = 15 } WJ_ty_appointment_assign_council = { POSITION = councillor_court_chaplain TITLE = e_minister_of_rites } }
  else_if = { limit = { var:wj_ty_office = 16 } WJ_ty_appointment_assign_council = { POSITION = minister_works TITLE = e_minister_of_works } }
  # Even if an engine operation failed, never replay a partially consumed draft.
  WJ_ty_invalidate = yes
  remove_variable = wj_ty_appointment_snapshot
 }
 if = { limit = { scope:wj_ty_appointment_success = no }
  set_variable = { name = wj_ty_appointment_failed value = 1 }
 }
}
