# Immediate Yellow Turban county transfer and uprising settlement effects.

hd_uprising_transfer_occupied_county_effect = {
	root = { save_temporary_scope_as = hd_uprising_county_recipient }
	scope:county = { save_temporary_scope_as = hd_uprising_occupied_county }

	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = hd_uprising_county_change
		add_claim_on_loss = no
	}

	scope:hd_uprising_occupied_county = {
		change_title_holder = {
			holder = scope:hd_uprising_county_recipient
			change = scope:hd_uprising_county_change
			take_baronies = yes
		}
	}
	resolve_title_and_vassal_change = scope:hd_uprising_county_change

	# End wars which have lost their last territorial county. Yellow Turban
	# attackers remain valid while landless because their camps are war actors.
	if = {
		limit = {
			hd_uprising_attacker_cannot_continue_trigger = {
				WAR = scope:war
				ATTACKER = scope:war.primary_attacker
			}
		}
		scope:war = { end_war = defender }
	}
	else_if = {
		limit = {
			hd_uprising_defender_cannot_continue_trigger = {
				DEFENDER = scope:war.primary_defender
			}
		}
		scope:war = { end_war = attacker }
	}
}

hd_uprising_settle_occupied_counties_on_victory_effect = {
	# Freeze every eligible county before changing any title. This prevents
	# overlapping defender realms from causing duplicate transfers or changing
	# the collection while it is being iterated.
	$WAR$ = {
		save_scope_as = hd_uprising_settlement_war
		every_war_defender = {
			save_temporary_scope_as = hd_uprising_current_defender
			every_sub_realm_county = {
				limit = {
					tier = tier_county
					is_landless_type_title = no
					NOT = { is_in_list = hd_uprising_victory_counties }
					holder ?= {
						OR = {
							this = scope:hd_uprising_current_defender
							is_vassal_or_below_of = scope:hd_uprising_current_defender
						}
					}
					county_controller ?= {
						is_alive = yes
						is_attacker_in_war = scope:hd_uprising_settlement_war
					}
				}
				add_to_temporary_list = hd_uprising_victory_counties
			}
		}
	}

	if = {
		limit = {
			any_in_list = {
				list = hd_uprising_victory_counties
				count > 0
			}
		}

		create_title_and_vassal_change = {
			type = conquest_populist
			save_scope_as = hd_uprising_victory_change
			add_claim_on_loss = no
		}

		every_in_list = {
			list = hd_uprising_victory_counties
			limit = {
				tier = tier_county
				is_landless_type_title = no
				county_controller ?= {
					is_alive = yes
					is_attacker_in_war = scope:hd_uprising_settlement_war
				}
			}
			county_controller ?= {
				save_temporary_scope_as = hd_uprising_victory_recipient
			}
			change_title_holder = {
				holder = scope:hd_uprising_victory_recipient
				change = scope:hd_uprising_victory_change
				take_baronies = yes
			}
		}

		resolve_title_and_vassal_change = scope:hd_uprising_victory_change
	}
}

hd_uprising_grant_independence_effect = {
	create_title_and_vassal_change = {
		type = independency
		save_scope_as = hd_uprising_independence_change
		add_claim_on_loss = no
	}

	scope:attacker = {
		change_liege_or_become_independent = {
			CHANGE = scope:hd_uprising_independence_change
			VASSAL = this
		}
		add_truce_both_ways = {
			character = scope:defender
			days = 1825
			war = $WAR$
			result = victory
		}
	}

	resolve_title_and_vassal_change = scope:hd_uprising_independence_change
}

hd_uprising_annex_attacker_realm_effect = {
	# All Yellow Turban attackers begin landless. On defeat, return every county
	# currently held by any attacker in this war to the Han primary defender.
	root.war = {
		every_war_attacker = {
			every_held_title = {
				limit = {
					tier = tier_county
					is_landless_type_title = no
				}
				add_to_list = hd_uprising_attacker_counties
			}
		}
	}

	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = hd_uprising_annex_change
		add_claim_on_loss = no
	}

	every_in_list = {
		list = hd_uprising_attacker_counties
		change_title_holder = {
			holder = scope:defender
			change = scope:hd_uprising_annex_change
			take_baronies = yes
		}
	}

	resolve_title_and_vassal_change = scope:hd_uprising_annex_change
}
