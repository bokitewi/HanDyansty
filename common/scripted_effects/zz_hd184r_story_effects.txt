# 重试计数。Scope：title:h_china。KEY 为计数变量名，ON_ACTION 为重试的 on_action。
hd184r_retry_effect = {
	if = { limit = { NOT = { has_variable = $KEY$ } } set_variable = { name = $KEY$ value = 0 } }
	change_variable = { name = $KEY$ add = 1 }
	if = {
		limit = { var:$KEY$ < 8 }
		trigger_event = { on_action = $ON_ACTION$ days = 90 }
	}
	else = { debug_log = "HD184R P5 RETRY_EXPIRED" }
}

hd184r_zhufu_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd184r_zhufu_done }
			character:zhu_fu ?= { is_alive = yes is_ai = yes }
			character:shi_xie ?= { is_alive = yes is_landed = yes }
		}
		set_global_variable = hd184r_zhufu_done
		character:shi_xie = { save_scope_as = hd184r_shi }
		character:zhu_fu = { save_scope_as = hd184r_zhu }
		create_title_and_vassal_change = { type = granted save_scope_as = hd184r_zhufu_change add_claim_on_loss = no }
		scope:hd184r_zhu = {
			every_held_title = {
				limit = { tier >= tier_county is_landless_type_title = no is_noble_family_title = no }
				add_to_temporary_list = hd184r_zhufu_titles
			}
		}
		every_in_list = {
			list = hd184r_zhufu_titles
			change_title_holder_include_vassals = { holder = scope:hd184r_shi change = scope:hd184r_zhufu_change }
		}
		# 交州王国头衔：无主则授予；由 AI 持有（非玩家）则移交头衔本身
		title:k_uuii_jiaozhou = {
			if = {
				limit = {
					OR = {
						NOT = { exists = holder }
						holder ?= { is_ai = yes NOT = { this = scope:hd184r_shi } NOT = { this = scope:hd184r_zhu } }
					}
				}
				change_title_holder = { holder = scope:hd184r_shi change = scope:hd184r_zhufu_change }
			}
		}
		resolve_title_and_vassal_change = scope:hd184r_zhufu_change
		scope:hd184r_zhu = {
			if = { limit = { is_alive = yes } death = { death_reason = death_murder } }
		}
		scope:hd184r_shi = {
			if = {
				limit = { is_independent_ruler = no is_ai = yes }
				hd_force_independent_celestial_effect = yes
			}
		}
		# 兄弟分任三郡太守（士燮本人持有该郡头衔时）
		hd184r_shi_brother_effect = { BRO = shi_yi DUCHY = d_hepu }
		hd184r_shi_brother_effect = { BRO = shi_huang_you DUCHY = d_jiuzhen }
		hd184r_shi_brother_effect = { BRO = shi_wu DUCHY = d_nanhai }
		scope:hd184r_shi = { save_scope_as = hd_news_a }
		scope:hd184r_zhu = { save_scope_as = hd_news_b }
		hd_world_news_send_all_effect = { EVENT = hd184r.0205 }
		debug_log = "HD184R P5 ZHUFU_KILLED_SHIXIE_JIAOZHOU"
	}
	else_if = {
		limit = { NOT = { exists = global_var:hd184r_zhufu_done } }
		title:h_china = { hd184r_retry_effect = { KEY = hd184r_zhufu_tries ON_ACTION = hd184r_sched_zhufu } }
	}
}

hd184r_shi_brother_effect = {
	if = {
		limit = {
			character:$BRO$ ?= { is_alive = yes is_adult = yes is_ai = yes is_landed = no }
			title:$DUCHY$.holder ?= scope:hd184r_shi
		}
		create_title_and_vassal_change = { type = granted save_scope_as = hd184r_shi_bro_change add_claim_on_loss = no }
		title:$DUCHY$ = {
			change_title_holder = { holder = character:$BRO$ change = scope:hd184r_shi_bro_change }
			every_de_jure_county = {
				limit = { holder ?= scope:hd184r_shi NOT = { this = scope:hd184r_shi.capital_county } }
				change_title_holder = { holder = character:$BRO$ change = scope:hd184r_shi_bro_change }
			}
		}
		character:$BRO$ = { change_liege = { liege = scope:hd184r_shi change = scope:hd184r_shi_bro_change } }
		resolve_title_and_vassal_change = scope:hd184r_shi_bro_change
		character:$BRO$ = { hd_force_celestial_effect = yes }
	}
}

				limit = { is_ai = yes hd184r_director_war_slot_trigger = yes }
				save_scope_as = hd184r_actor
				clear_saved_scope = hd184r_target_duchy
				clear_saved_scope = hd184r_defender
				title:k_uuii_xuzhou = {
					random_in_de_jure_hierarchy = {
						continue = { tier > tier_duchy }
						limit = {
							tier = tier_duchy
							any_de_jure_county = {
								hd184r_county_valid_front_trigger = yes
								holder.top_liege ?= character:tao_qian
							}
						}
						save_scope_as = hd184r_target_duchy
					}
				}
				if = {
					limit = { exists = scope:hd184r_target_duchy }
					hd_tb_arm_effect = { KIND = historical }

					debug_log = "HD184R P5 CAO_FA_XU_WAR"
				}
				else = { debug_log = "HD184R P5 CAO_FA_XU_NO_FRONT" }
			}
		}
	}
	else_if = {
		limit = { NOT = { exists = global_var:hd184r_caosong_done } }
		title:h_china = { hd184r_retry_effect = { KEY = hd184r_caosong_tries ON_ACTION = hd184r_sched_caosong } }

# WHO 来投 LORD。WHO 须在世、成年、AI、无地，且不在 LORD 处。
hd184r_advisor_join_effect = {
	if = {
		limit = {
			character:$WHO$ ?= {
				is_alive = yes is_adult = yes is_ai = yes is_landed = no is_imprisoned = no
				NOT = { employer ?= { is_ai = no } } # D4：不从玩家宫廷挖人
			}
			exists = scope:hd184r_lord
			scope:hd184r_lord = { is_alive = yes is_landed = yes }
			NOT = { character:$WHO$ = { is_courtier_of = scope:hd184r_lord } }
		}
		character:$WHO$ = {
			set_employer = scope:hd184r_lord
			if = {
				limit = { is_courtier_of = scope:hd184r_lord }
				add_opinion = { target = scope:hd184r_lord modifier = grateful_opinion opinion = 30 }
				save_scope_as = hd184r_advisor
				scope:hd184r_lord = {
					if = { limit = { is_ai = no } trigger_event = hd184r.0206 }
				}
				debug_log = "HD184R P5 ADVISOR_JOINED"
				debug_log_scopes = yes
			}
		}
	}
}
