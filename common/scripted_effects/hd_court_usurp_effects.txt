# 乱世霸权·篡位与僭号（第 5 期，2026-10-10，L5/L6）
# 设计见 docs/乱世霸权_天子与权臣_设计草案.md 第 9 节"第 5 期"。

# 篡位共通：权臣取霸权（复用天命交接），旧天子另有安置，正统压到 $LEVEL$ 级，
# 天下郡国制君主 10 年内可用"讨篡"CB，尊汉者对其好感下降。
# $TITLE_TIER$ / $TITLE_NAME$：旧天子所得头衔的层级与名字。
#   duchy   = 无地封号（禅代"山阳"）
#   kingdom = 原版礼仪君主（2026-10-24 历史志向任务卡 D43：取代原「虚位天子」修正）
# Scope：权臣
hd_court_usurp_common_effect = {
	save_scope_as = hd_court_usurper
	title:h_china.holder = { save_scope_as = hd_court_deposed_emperor }
	hd_court_release_effect = yes
	scope:hd_court_deposed_emperor = {
		if = {
			limit = { has_active_diarchy = yes }
			end_diarchy = yes
		}
	}
	hd_transfer_mandate_to_claimant_effect = yes
	scope:hd_court_deposed_emperor ?= {
		if = {
			limit = { is_alive = yes }
			hd_court_deposed_title_$TITLE_TIER$_effect = { TITLE_NAME = $TITLE_NAME$ }
		}
	}
	# 篡位者：正统极低
	if = {
		limit = { legitimacy > legitimacy_level_$LEVEL$ }
		add_legitimacy = {
			value = legitimacy_level_$LEVEL$
			subtract = legitimacy
		}
	}
	remove_variable = hd_court_stance
	remove_character_modifier = hd_court_stance_zunhan_modifier
	remove_character_modifier = hd_court_stance_moni_modifier
	title:h_china = {
		set_variable = { name = hd_court_usurper value = scope:hd_court_usurper years = 10 }
	}
	every_ruler = {
		limit = {
			hd_court_stance_zunhan_trigger = yes
			NOT = { this = scope:hd_court_usurper }
		}
		add_opinion = {
			target = scope:hd_court_usurper
			modifier = hd_court_usurper_opinion
		}
	}
	hd_dc_kit_usurp_effect = yes
	every_player = {
		limit = { tgp_does_this_player_care_about_the_dynastic_cycle = yes }
		trigger_event = hd_court.0020
	}
}

# 僭号称帝（袁术式）：立一个动态帝国，与汉帝平分正统；自此谋逆，尊汉的邻国得讨伐之名。Scope：称帝者
hd_court_proclaim_emperor_effect = {
	# Keep this legacy decision's legitimacy split; the new foundation hook must not also pay.
	set_variable = wj_cr_empire_settled
	save_scope_as = hd_court_pseudo_founder
	create_dynamic_title = {
		tier = empire
		name = hd_court_pseudo_empire_name
	}
	scope:new_title = {
		set_coa = scope:hd_court_pseudo_founder.house
		set_variable = hd_court_pseudo_empire	# 2026-10-25：吞并胜利时据此销毁（hd_tb_pseudo_empire_trigger）
		save_scope_as = hd_court_pseudo_empire
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = hd_court_pseudo_change
		add_claim_on_loss = no
	}
	scope:hd_court_pseudo_empire = {
		change_title_holder = {
			holder = scope:hd_court_pseudo_founder
			change = scope:hd_court_pseudo_change
		}
	}
	resolve_title_and_vassal_change = scope:hd_court_pseudo_change
	set_primary_title_to = scope:hd_court_pseudo_empire
	# 平分正统：天子失其半，称帝者得之
	title:h_china.holder ?= {
		save_scope_as = hd_court_split_emperor
		if = {
			limit = { legitimacy > 0 }
			save_scope_value_as = {
				name = hd_court_split_legitimacy
				value = {
					value = legitimacy
					multiply = 0.5
				}
			}
			add_legitimacy = {
				value = scope:hd_court_split_legitimacy
				multiply = -1
			}
		}
	}
	if = {
		limit = { exists = scope:hd_court_split_legitimacy }
		add_legitimacy = scope:hd_court_split_legitimacy
	}
	hd_court_set_stance_effect = { STANCE = moni }
	add_character_modifier = hd_court_pseudo_emperor_modifier
	every_ruler = {
		limit = {
			hd_court_stance_zunhan_trigger = yes
			NOT = { this = scope:hd_court_pseudo_founder }
		}
		add_opinion = {
			target = scope:hd_court_pseudo_founder
			modifier = hd_court_pseudo_emperor_opinion
		}
	}
	every_neighboring_top_liege_realm_owner = {
		limit = {
			hd_court_stance_zunhan_trigger = yes
			hd_court_warlord_trigger = yes
		}
		set_variable = { name = hd_court_fengzhao_target value = scope:hd_court_pseudo_founder years = 10 }
	}
	hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_proclaim_emperor ACTOR = scope:hd_court_pseudo_founder }
	every_player = {
		limit = { tgp_does_this_player_care_about_the_dynastic_cycle = yes }
		trigger_event = hd_court.0021
	}
}

# ===== 旧天子安置（hd_court_usurp_common_effect 按层级调用）=====

# 公爵级：无地封号并称臣（218、南北朝禅代的"山阳"）。Scope：旧天子；scope:hd_court_usurper = 新天子
hd_court_deposed_title_duchy_effect = {
	create_dynamic_title = {
		tier = duchy
		name = $TITLE_NAME$
	}
	scope:new_title = {
		set_landless_title = yes
		set_definitive_form = yes
		save_scope_as = hd_court_deposed_title
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_court_deposed_title_change
		add_claim_on_loss = no
	}
	scope:hd_court_deposed_title = {
		change_title_holder = {
			holder = scope:hd_court_deposed_emperor
			change = scope:hd_court_deposed_title_change
		}
	}
	resolve_title_and_vassal_change = scope:hd_court_deposed_title_change
	# 无地无世族的旧天子在天命交接中不会自动臣服，此处补上
	if = {
		limit = { NOT = { liege ?= scope:hd_court_usurper } }
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = hd_court_deposed_fealty_change
			add_claim_on_loss = no
		}
		change_liege = {
			liege = scope:hd_court_usurper
			change = scope:hd_court_deposed_fealty_change
		}
		resolve_title_and_vassal_change = scope:hd_court_deposed_fealty_change
	}
}

# ===== 原版礼仪君主（2026-10-24，历史志向任务卡 D43~D50，设计见 docs/历史志向任务卡_设计草案.md §3.1）=====
# 原版写法（日本 e_japan / 朝鲜 10_dlc_tgp_korea_scripted_effects）：
#   权臣主头衔（此处为 h_china）var:administrative_ui_special_title = 礼仪头衔
#   礼仪头衔 var:ceremonial_title = 权臣主头衔；礼仪头衔由权臣的封臣持有，单人继承
# 注意：原版规定霸权下的礼仪头衔要销毁（tgp_destroy_ceremonial_liege_title_effect 的"Hegemony"条件），
# 本 mod 的 tgp_destroy_ceremonial_throne_effect 是空操作才保住它，同步原版时不得恢复。

# 建立礼仪君主。Scope：旧天子（礼仪君主）；scope:hd_court_ceremonial_regent = 持有霸权的权臣
# $TITLE_NAME$：头衔名（一律"皇帝"，hd_court_ceremonial_title_name）。头衔按原版为普通虚衔（D57）；
# 兜底（D62）：礼仪君主既无县（含男爵领）、也无世族头衔时才设为无地型，否则一县不剩的人保不住普通头衔
hd_court_establish_ceremonial_effect = {
	save_scope_as = hd_court_ceremonial_monarch
	create_dynamic_title = {
		tier = kingdom
		name = $TITLE_NAME$
	}
	scope:new_title = {
		set_definitive_form = yes
		set_variable = {
			name = ceremonial_title
			value = title:h_china
		}
		save_scope_as = hd_court_ceremonial_title
	}
	if = {
		limit = {
			scope:hd_court_ceremonial_monarch = {
				NOR = {
					any_held_title = {
						tier <= tier_county
						is_landless_type_title = no
					}
					any_held_title = { is_noble_family_title = yes }
				}
			}
		}
		scope:hd_court_ceremonial_title = { set_landless_title = yes }
	}
	if = {
		limit = { exists = scope:hd_court_ceremonial_monarch.house }
		scope:hd_court_ceremonial_title = { set_coa = scope:hd_court_ceremonial_monarch.house }
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_court_ceremonial_change
		add_claim_on_loss = no
	}
	scope:hd_court_ceremonial_title = {
		change_title_holder = {
			holder = scope:hd_court_ceremonial_monarch
			change = scope:hd_court_ceremonial_change
		}
	}
	resolve_title_and_vassal_change = scope:hd_court_ceremonial_change
	scope:hd_court_ceremonial_title = { add_title_law = single_heir_succession_law }
	title:h_china = {
		set_variable = {
			name = administrative_ui_special_title
			value = scope:hd_court_ceremonial_title
		}
	}
	# 礼仪君主须是权臣的封臣
	if = {
		limit = { NOT = { liege ?= scope:hd_court_ceremonial_regent } }
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = hd_court_ceremonial_fealty
			add_claim_on_loss = no
		}
		change_liege = {
			liege = scope:hd_court_ceremonial_regent
			change = scope:hd_court_ceremonial_fealty
		}
		resolve_title_and_vassal_change = scope:hd_court_ceremonial_fealty
	}
	# 天命交接会给旧天子加"前帝"；做礼仪君主时去掉（头衔被销毁时再加回）
	if = {
		limit = { has_trait = former_emperor }
		remove_trait = former_emperor
	}
}

# D47：「还政天子」派系成功——礼仪君主取回天命，礼仪头衔随天命交接销毁（D50），原权臣称臣并失去权臣身份。
# Scope = root：礼仪君主（须经事件触发，天命交接以 root 为宣称者）
hd_court_restore_ceremonial_effect = {
	hd_court_release_effect = yes
	hd_transfer_mandate_to_claimant_effect = yes
	if = {
		limit = { has_legitimacy = yes }
		add_legitimacy = major_legitimacy_gain
	}
}

# D49/D50：销毁礼仪头衔，解开两头变量（照原版 tgp_destroy_ceremonial_liege_title_effect 的收尾）。
# 礼仪君主若不是 root、也不持霸权，得"前帝"特质；名下别无头衔时照原版清空国库。
# Scope：人物（销毁动作在此作用域执行）。读 title:h_china.var:administrative_ui_special_title
hd_court_destroy_ceremonial_effect = {
	title:h_china.var:administrative_ui_special_title ?= { save_scope_as = hd_court_old_ceremonial_title }
	if = {
		limit = { exists = scope:hd_court_old_ceremonial_title }
		scope:hd_court_old_ceremonial_title.holder ?= {
			remove_character_flag = ceremonial_liege_flag
			if = {
				limit = {
					is_alive = yes
					NOT = { this = root }
					NOT = { has_title = title:h_china }
				}
				add_trait = former_emperor
			}
			if = {
				limit = {
					has_treasury = yes
					NOT = {
						any_held_title = {
							NOT = { this = scope:hd_court_old_ceremonial_title }
						}
					}
				}
				empty_treasury_when_abandoning_landed_life_effect = yes
			}
		}
		title:h_china.holder ?= { remove_character_flag = ceremonial_regent_flag }
		scope:hd_court_old_ceremonial_title = { remove_variable = ceremonial_title }
		title:h_china = { remove_variable = administrative_ui_special_title }
		destroy_title = scope:hd_court_old_ceremonial_title
	}
}
