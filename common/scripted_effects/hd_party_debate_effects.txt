# 天朝循环大修·朋党：立场辩论 effect（Scope 约定：host = 发起人；scope:activity = 辩论活动）

# 活动开始：标记朝廷"辩论中"，发起人进入 10 年冷却，记录目标立场
hd_party_debate_start_effect = {
	scope:host = {
		set_variable = { name = hd_party_debate_cd days = hd_party_debate_cd_days }
		remove_variable = hd_party_ai_debate_wish
		var:hd_party_my_court ?= { save_scope_as = hd_party_court }
	}
	scope:activity = {
		set_variable = { name = hd_debate_progress value = 0 }
		if = { limit = { has_activity_option = { category = hd_debate_target option = hd_target_zunwang } } set_variable = { name = hd_debate_target value = flag:zunwang } }
		if = { limit = { has_activity_option = { category = hd_debate_target option = hd_target_kaituo } } set_variable = { name = hd_debate_target value = flag:kaituo } }
		if = { limit = { has_activity_option = { category = hd_debate_target option = hd_target_jinqu } } set_variable = { name = hd_debate_target value = flag:jinqu } }
		if = { limit = { has_activity_option = { category = hd_debate_target option = hd_target_shoujiu } } set_variable = { name = hd_debate_target value = flag:shoujiu } }
		if = { limit = { has_activity_option = { category = hd_debate_target option = hd_target_zhongli } } set_variable = { name = hd_debate_target value = flag:zhongli } }
	}
	scope:hd_party_court ?= {
		if = { limit = { scope:host.var:hd_party_my_group ?= flag:huanguan } set_variable = hd_party_huanguan_debating }
		if = { limit = { scope:host.var:hd_party_my_group ?= flag:waiqi } set_variable = hd_party_waiqi_debating }
		if = { limit = { scope:host.var:hd_party_my_group ?= flag:wuxun } set_variable = hd_party_wuxun_debating }
		if = { limit = { scope:host.var:hd_party_my_group ?= flag:shizu } set_variable = hd_party_shizu_debating }
		if = { limit = { scope:host.var:hd_party_my_group ?= flag:hanmen } set_variable = hd_party_hanmen_debating }
	}
}

# 收尾：解除朝廷"辩论中"标记（Scope：发起人）
hd_party_debate_cleanup_effect = {
	var:hd_party_my_court ?= {
		remove_variable = hd_party_huanguan_debating
		remove_variable = hd_party_waiqi_debating
		remove_variable = hd_party_wuxun_debating
		remove_variable = hd_party_shizu_debating
		remove_variable = hd_party_hanmen_debating
	}
}

# 把嘉宾分入 拥护/中立/反对。首次按意图记录阵营（var:hd_debate_side，120 天过期），之后按记录（事件可改变阵营）
hd_party_debate_sort_guests_effect = {
	scope:activity = {
		every_attending_character = {
			if = {
				limit = { NOT = { has_variable = hd_debate_side } }
				if = {
					limit = {
						OR = {
							this = scope:host
							has_activity_intent = hd_debate_support_intent
						}
					}
					set_variable = { name = hd_debate_side value = flag:support days = 120 }
				}
				else_if = {
					limit = { has_activity_intent = hd_debate_oppose_intent }
					set_variable = { name = hd_debate_side value = flag:oppose days = 120 }
				}
				else = {
					set_variable = { name = hd_debate_side value = flag:neutral days = 120 }
				}
			}
		}
		hd_party_debate_apply_sides_effect = yes
	}
}

# Scope：辩论活动。按 var:hd_debate_side 重新分组
hd_party_debate_apply_sides_effect = {
	every_attending_character = {
		scope:activity = {
			remove_from_guest_subset = { name = hd_debate_support target = prev }
			remove_from_guest_subset = { name = hd_debate_neutral target = prev }
			remove_from_guest_subset = { name = hd_debate_oppose target = prev }
		}
		if = {
			limit = { var:hd_debate_side ?= flag:support }
			scope:activity = { add_to_guest_subset = { name = hd_debate_support target = prev } }
		}
		else_if = {
			limit = { var:hd_debate_side ?= flag:oppose }
			scope:activity = { add_to_guest_subset = { name = hd_debate_oppose target = prev } }
		}
		else = {
			scope:activity = { add_to_guest_subset = { name = hd_debate_neutral target = prev } }
		}
	}
}

# Scope：嘉宾。改换阵营为 $SIDE$ 并立即刷新分组
hd_party_debate_set_side_effect = {
	set_variable = { name = hd_debate_side value = flag:$SIDE$ days = 120 }
	scope:activity = { hd_party_debate_apply_sides_effect = yes }
}

# 改变辩论进度 $VALUE$（Scope：任意，需 scope:activity）
hd_party_debate_progress_effect = {
	scope:activity = {
		hd_party_add_var_effect = { NAME = hd_debate_progress VALUE = $VALUE$ }
		clamp_variable = { name = hd_debate_progress min = -100 max = 100 }
	}
}

# 玩家嘉宾（拥护/反对方）每阶段的小事件（Scope：嘉宾）

# 结局（Scope：发起人）
hd_party_debate_resolve_effect = {
	save_scope_as = hd_party_debate_host
	var:hd_party_my_court ?= { save_scope_as = hd_party_court }
	scope:activity = {
		save_scope_value_as = { name = hd_party_debate_final value = { value = var:hd_debate_progress add = hd_party_debate_power_mod_value } }
	}
	if = {
		limit = {
			exists = scope:hd_party_court
			scope:hd_party_debate_final >= hd_party_debate_success_threshold
		}
		scope:activity = { set_variable = { name = hd_debate_result value = flag:success } }
		hd_party_debate_success_effect = yes
		trigger_event = hd_party_debate.0090
	}
	else = {
		scope:activity = { set_variable = { name = hd_debate_result value = flag:fail } }
		hd_party_debate_fail_effect = yes
		trigger_event = hd_party_debate.0091
	}
	hd_party_debate_cleanup_effect = yes
}
