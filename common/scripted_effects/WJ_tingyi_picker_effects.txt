# Display projection only. Root = actual GUI player, never the appointing emperor.
# Parent hooks: after generating wj_ty_candidates and setting wj_ty_picker, call
# WJ_ty_picker_init. On target selection, cancel, and WJ_ty_close call WJ_ty_picker_clear.
# Neither hook changes the base list, target, confirmed draft, or appointment state.
WJ_ty_picker_clear = {
 WJ_ty_local_picker_clear = yes
 clear_variable_list = wj_ty_picker_candidates
 clear_variable_list = wj_ty_picker_filter_options
 remove_variable = wj_ty_picker_sort
 remove_variable = wj_ty_picker_sort_ascending
 remove_variable = wj_ty_picker_filter_faith
 remove_variable = wj_ty_picker_filter_culture
 remove_variable = wj_ty_picker_filters_open
}

WJ_ty_picker_init = {
 WJ_ty_picker_clear = yes
 set_variable = { name = wj_ty_picker_sort value = 10 }
 set_variable = { name = wj_ty_picker_filter_faith value = 0 }
 set_variable = { name = wj_ty_picker_filter_culture value = 0 }
 add_to_variable_list = { name = wj_ty_picker_filter_options target = 0 }
 add_to_variable_list = { name = wj_ty_picker_filter_options target = 1 }
 add_to_variable_list = { name = wj_ty_picker_filter_options target = 2 }
 WJ_ty_picker_sort_candidates_effect = yes
}

# Rebuild only on init or a sort/filter click; never from IsShown/IsValid or GUI ticks.
# Native ordered list + the existing dasima key/filter rules, scoped to the GUI player.
WJ_ty_picker_sort_candidates_effect = {
 clear_variable_list = wj_ty_picker_candidates
 save_scope_as = wj_ty_picker_operator
 if = {
  limit = { WJ_ty_session = yes has_variable = wj_ty_picker has_variable = wj_ty_picker_sort }
  ordered_in_list = {
   variable = wj_ty_candidates
   # Native ordered iterators default to one result. Count the whole source list.
   max = { value = 0 every_in_list = { variable = wj_ty_candidates add = 1 } }
   check_range_bounds = no
   limit = {
 trigger_if = {
  limit = { scope:wj_ty_picker_operator = { has_variable = wj_ty_picker_filter_faith var:wj_ty_picker_filter_faith = 1 } }
  faith = scope:wj_ty_picker_operator.faith
 }
 trigger_if = {
  limit = { scope:wj_ty_picker_operator = { has_variable = wj_ty_picker_filter_faith var:wj_ty_picker_filter_faith = 2 } }
  NOT = { faith = scope:wj_ty_picker_operator.faith }
 }
 trigger_if = {
  limit = { scope:wj_ty_picker_operator = { has_variable = wj_ty_picker_filter_culture var:wj_ty_picker_filter_culture = 1 } }
  culture = scope:wj_ty_picker_operator.culture
 }
 trigger_if = {
  limit = { scope:wj_ty_picker_operator = { has_variable = wj_ty_picker_filter_culture var:wj_ty_picker_filter_culture = 2 } }
  NOT = { culture = scope:wj_ty_picker_operator.culture }
 }
}
   order_by = {
 value = diplomacy
 add = martial
 add = stewardship
 add = intrigue
 add = learning
 if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 0 } value = age }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 1 } value = diplomacy }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 2 } value = martial }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 3 } value = stewardship }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 4 } value = intrigue }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 5 } value = learning }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 6 } value = highest_held_title_tier }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 7 } value = current_military_strength }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 8 } value = "opinion(scope:wj_ty_picker_operator)" }
 else_if = { limit = { scope:wj_ty_picker_operator.var:wj_ty_picker_sort = 9 } value = prowess }
 if = { limit = { scope:wj_ty_picker_operator = { has_variable = wj_ty_picker_sort_ascending } } multiply = -1 }
}
   scope:wj_ty_picker_operator = { add_to_variable_list = { name = wj_ty_picker_candidates target = prev } }
  }
 }
}
