# Shared machinery and the nine 184-196 historical regional revolt war effects (phase 1).
#
# Execution order per plan: leader -> defender -> war -> war scope -> war-bound event troops.
# The fired flag is set only after the war scope actually exists, so a failed declaration
# is retried on the next monthly pulse until the window closes.
#
# All revolt leaders start as landless adventurers; on victory the CB converts the leader
# to the mod's kuzhu_warlord_government and carves out the designated duchies as an
# independent warlord realm.

### Save the spawn county's culture/faith for leader creation. Falls back to han/jingxue
# so a county without valid culture or faith can never produce a cultureless leader
# (cultureless leaders triggered the runtime obedience/legitimacy error spam).
hd_historical_revolt_save_culture_faith_effect = {
	if = {
		limit = { exists = scope:hd_revolt_spawn_province.county.culture }
		scope:hd_revolt_spawn_province = {
			county = {
				culture = { save_temporary_scope_as = hd_revolt_culture }
			}
		}
	}
	else = {
		culture:han = { save_temporary_scope_as = hd_revolt_culture }
	}
	if = {
		limit = { exists = scope:hd_revolt_spawn_province.county.faith }
		scope:hd_revolt_spawn_province = {
			county = {
				faith = { save_temporary_scope_as = hd_revolt_faith }
			}
		}
	}
	else = {
		faith:jingxue = { save_temporary_scope_as = hd_revolt_faith }
	}
}

### Make sure the revolt leader operates from a landless adventurer camp at the spawn point.
hd_historical_revolt_ensure_adventurer_effect = {
	if = {
		limit = {
			NOT = { has_government = landless_adventurer_government }
		}
		set_location = $LOCATION$
		create_landless_adventurer_title_effect = {
			REASON = flag:historical
			FLAVOR_CHAR = scope:hd_revolt_leader
		}
	}
}

### Reuse a living historical leader; otherwise create one lowborn stand-in before firing.
# This fallback is only available before the uprising's permanent fired flag is set.
hd_historical_revolt_save_or_create_named_leader_effect = {
	if = {
		limit = { $CHARACTER$ = { is_alive = yes } }
		$CHARACTER$ = { save_scope_as = hd_revolt_leader }
	}
	else = {
		scope:hd_revolt_spawn_province = {
			create_character = {
				location = scope:hd_revolt_spawn_province
				template = new_commander_character
				name = $NAME$
				age = $AGE$
				gender_female_chance = 0
				random_traits = no
				dynasty = none
				culture = $CULTURE$
				faith = $FAITH$
				trait = brave
				diplomacy = { $DIPLOMACY$ $DIPLOMACY$ }
				martial = { $MARTIAL$ $MARTIAL$ }
				stewardship = { $STEWARDSHIP$ $STEWARDSHIP$ }
				intrigue = { $INTRIGUE$ $INTRIGUE$ }
				learning = { $LEARNING$ $LEARNING$ }
				prowess = { $PROWESS$ $PROWESS$ }
				save_scope_as = hd_revolt_leader
			}
		}
	}
}

### Pick the defender among the direct holders of the candidate duchies.
# ROOT must be the revolt leader. The revolt always declares on the character actually
# holding the candidate titles (e.g. the Jiaozhi duke for the Zhutian general), never on
# their top liege - the CB's target_top_liege_if_outside_realm = no keeps the holder as
# the war defender. The direct holder of the most candidates becomes scope:hd_revolt_defender;
# ties resolve to the first candidate in LIST1 (design order).
# LIST1/LIST2 are per-revolt temporary list names so two revolts firing in the same
# monthly pulse can never pollute each other's candidate pools.
hd_historical_revolt_pick_defender_effect = {

	# Collect the distinct living direct holders of the candidate duchies.
	every_in_list = {
		list = $LIST1$
		if = {
			limit = {
				exists = holder
				holder = { is_alive = yes }
			}
			holder = {
				if = {
					limit = { NOT = { is_in_list = $LIST2$ } }
					add_to_list = $LIST2$
				}
			}
		}
	}

	every_in_list = {
		list = $LIST2$
		set_variable = {
			name = hd_revolt_hold_count
			value = 0
		}
	}
	every_in_list = {
		list = $LIST1$
		if = {
			limit = {
				exists = holder
				holder = { is_in_list = $LIST2$ }
			}
			holder = {
				change_variable = {
					name = hd_revolt_hold_count
					add = 1
				}
			}
		}
	}

	ordered_in_list = {
		list = $LIST2$
		order_by = var:hd_revolt_hold_count
		max = 1
		check_range_bounds = no
		save_scope_as = hd_revolt_defender
	}
	every_in_list = {
		list = $LIST2$
		remove_variable = hd_revolt_hold_count
	}

	# Save the first formal target held by the chosen defender. Per-revolt launch
	# branches add every other defender-held title formally.
	every_in_list = {
		list = $LIST1$
		limit = {
			exists = holder
			holder = scope:hd_revolt_defender
		}
		if = {
			limit = {
				NOT = { exists = scope:hd_revolt_primary_target }
			}
			save_scope_as = hd_revolt_primary_target
		}
	}
}

### CB victory: take the designated titles and every county occupied by a war attacker.
hd_historical_revolt_victory_effect = {
	$WAR$ = {
		save_scope_as = hd_hist_revolt_settlement_war
	}

	# Freeze the target counties before any title changes. Occupied target
	# counties are granted by the shared effect; the holder check below then
	# skips them instead of reassigning them to the primary attacker.
	every_in_list = {
		list = target_titles
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_county
				is_landless_type_title = no
				holder ?= { target_is_liege_or_above = scope:defender }
				NOT = { is_in_list = hd_hist_revolt_target_counties }
			}
			add_to_temporary_list = hd_hist_revolt_target_counties
		}
	}
	if = {
		limit = {
			scope:hd_hist_revolt_settlement_war ?= {
				has_variable = hd_revolt_wangguo_joined
			}
		}
		title:d_fufeng = {
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_county
					is_landless_type_title = no
					holder ?= { target_is_liege_or_above = scope:defender }
					NOT = { is_in_list = hd_hist_revolt_target_counties }
				}
				add_to_temporary_list = hd_hist_revolt_target_counties
			}
		}
	}

	# The shared effect freezes actual county controllers and grants every
	# occupied county directly, including counties outside the formal targets.
	hd_uprising_settle_occupied_counties_on_victory_effect = {
		WAR = $WAR$
	}

	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = hd_hist_revolt_conquest_change
		add_claim_on_loss = no
	}

	# Any target county not already granted to an actual occupier goes directly
	# to the primary attacker, including all of its baronies.
	every_in_list = {
		list = hd_hist_revolt_target_counties
		limit = {
			holder ?= { target_is_liege_or_above = scope:defender }
		}
		change_title_holder = {
			holder = scope:attacker
			change = scope:hd_hist_revolt_conquest_change
			take_baronies = yes
		}
		remove_from_list = hd_hist_revolt_target_counties
	}

	# Transfer the formal target titles directly. The vanilla conquest helper
	# preserves eligible holders as vassals, so it is intentionally not used.
	every_in_list = {
		list = target_titles
		limit = {
			holder ?= { target_is_liege_or_above = scope:defender }
		}
		change_title_holder = {
			holder = scope:attacker
			change = scope:hd_hist_revolt_conquest_change
			take_baronies = yes
		}
	}
	if = {
		limit = {
			scope:hd_hist_revolt_settlement_war ?= {
				has_variable = hd_revolt_wangguo_joined
			}
		}
		title:d_fufeng = {
			if = {
				limit = {
					holder ?= { target_is_liege_or_above = scope:defender }
				}
				change_title_holder = {
					holder = scope:attacker
					change = scope:hd_hist_revolt_conquest_change
					take_baronies = yes
				}
			}
		}
	}

	resolve_title_and_vassal_change = scope:hd_hist_revolt_conquest_change

	# Leave the defeated realm and set a five-year truce.
	hd_uprising_grant_independence_effect = {
		WAR = root.war
	}

	# The new carve-out realm always uses the mod's warlord government.
	scope:attacker = {
		every_held_title = {
			limit = {
				is_noble_family_title = yes
			}
			prev = { destroy_title = prev }
		}
		change_government = kuzhu_warlord_government
		kuzhu_warlord_trait_xp_initialize_effect = yes
		kuzhu_warlord_succession_law_setup_effect = yes
	}
	if = {
		limit = { scope:attacker = { has_character_flag = hd_revolt_leader_beigong } }
		hd_historical_revolt_beigong_vassalize_effect = yes
	}
}

### After Beigong Boyu's victory, force the three historical Liangzhou leaders
# into his landed warlord realm and notify each recipient with a single forced event.
hd_historical_revolt_beigong_vassalize_effect = {
	scope:attacker = { save_scope_as = hd_beigong_victor }

		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				NOT = { is_vassal_of = scope:hd_beigong_victor }
			}
			trigger_event = hd_historical_revolts.0010
		}
	}
	character:han_sui ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
				NOT = { is_vassal_of = scope:hd_beigong_victor }
			}
			trigger_event = hd_historical_revolts.0010
		}
	}
	character:ma_teng ?= {

}

hd_historical_revolt_beigong_force_vassal_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			exists = scope:hd_beigong_victor
			scope:hd_beigong_victor = { is_alive = yes }
			NOT = { is_vassal_of = scope:hd_beigong_victor }
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_beigong_vassal_change
			add_claim_on_loss = no
		}
		change_liege = {
			liege = scope:hd_beigong_victor
			change = scope:hd_beigong_vassal_change
		}
		resolve_title_and_vassal_change = scope:hd_beigong_vassal_change
	}
}

### Start one war with one, two, or three formal target titles.
# ROOT is the revolt leader. These helpers are deliberately separate because start_war
# accepts repeated target_title fields, not a post-declaration title list.
hd_historical_revolt_start_one_target_war_effect = {
	start_war = {
		cb = hd_historical_revolt_cb
		target = scope:hd_revolt_defender
		target_title = $TARGET1$
	}
	random_character_war = {
		limit = {
			primary_attacker = scope:hd_revolt_leader
			using_cb = hd_historical_revolt_cb
		}
		save_scope_as = hd_revolt_war
	}
}

hd_historical_revolt_start_two_target_war_effect = {
	start_war = {
		cb = hd_historical_revolt_cb
		target = scope:hd_revolt_defender
		target_title = $TARGET1$
		target_title = $TARGET2$
	}
	random_character_war = {
		limit = {
			primary_attacker = scope:hd_revolt_leader
			using_cb = hd_historical_revolt_cb
		}
		save_scope_as = hd_revolt_war
	}
}

# Jiangxia Man uprising was ~180; this date is a phase-1 gameplay shift) - anonymous chief
############################

hd_historical_revolt_jiangxia_effect = {
	set_global_variable = hd_revolt_jiangxia_committed
	province:1867 = { save_scope_as = hd_revolt_spawn_province }
	hd_historical_revolt_save_culture_faith_effect = yes
	scope:hd_revolt_spawn_province = {
		create_character = {
			location = scope:hd_revolt_spawn_province
			template = new_commander_character
			name = hd_jiangxia_manshuai
			age = 35
			gender_female_chance = 0
			random_traits = no
			dynasty = none
			culture = scope:hd_revolt_culture
			faith = scope:hd_revolt_faith
			trait = brave
			trait = wrathful
			martial = { 14 14 }
			diplomacy = { 6 6 }
			stewardship = { 6 6 }
			intrigue = { 9 9 }
			learning = { 4 4 }
			prowess = { 13 13 }
			save_scope_as = hd_revolt_leader
		}
	}
	scope:hd_revolt_leader = {
		hd_historical_revolt_ensure_adventurer_effect = { LOCATION = scope:hd_revolt_spawn_province }
		title:d_jiangxia = { add_to_list = hd_revolt_candidates_jiangxia }
		hd_historical_revolt_pick_defender_effect = {
			LIST1 = hd_revolt_candidates_jiangxia
			LIST2 = hd_revolt_lieges_jiangxia
		}
	}
	if = {
		limit = {
			exists = scope:hd_revolt_defender
			exists = scope:hd_revolt_primary_target
			NOT = { scope:hd_revolt_leader = scope:hd_revolt_defender }
		}
		scope:hd_revolt_leader = {
			hd_historical_revolt_start_one_target_war_effect = { TARGET1 = scope:hd_revolt_primary_target }
		}
		if = {
			limit = { exists = scope:hd_revolt_war }
			scope:hd_revolt_leader = {
				spawn_army = {
					# TOTAL = 12000
					name = hd_revolt_army_jiangxia
					levies = 8000
					men_at_arms = { type = light_footmen men = 2500 }
					men_at_arms = { type = bowmen men = 1500 }
					location = scope:hd_revolt_spawn_province
					origin = scope:hd_revolt_spawn_province
					uses_supply = yes
					inheritable = no
					war = scope:hd_revolt_war
					save_temporary_scope_as = hd_revolt_spawned_army
				}
			}
			scope:hd_revolt_spawned_army = { assign_commander = scope:hd_revolt_leader }
			scope:hd_revolt_leader = { add_character_flag = hd_revolt_leader_jiangxia }
			set_global_variable = hd_revolt_jiangxia_fired
			scope:hd_revolt_defender = {
				if = {
					limit = { is_ai = no }
					trigger_event = hd_historical_revolts.0004
				}
			}
		}
		else = {
		}
	}
	else = {
	}
}

# REV-05 187.4 Wang Guo of Liangzhou - merges into the ongoing Beigong Boyu war
# when both face the same defender, otherwise starts its own war for Fufeng
############################

hd_historical_revolt_wangguo_effect = {
	set_global_variable = hd_revolt_wangguo_committed
	province:1520 = { save_scope_as = hd_revolt_spawn_province }
	hd_historical_revolt_save_culture_faith_effect = yes
	hd_historical_revolt_save_or_create_named_leader_effect = {
		CHARACTER = character:wang_guo NAME = uuii_book_wang_guo_name AGE = 30
		CULTURE = scope:hd_revolt_culture FAITH = scope:hd_revolt_faith
		DIPLOMACY = 7 MARTIAL = 7 STEWARDSHIP = 7 INTRIGUE = 7 LEARNING = 6 PROWESS = 6
	}
	scope:hd_revolt_leader = {
		hd_historical_revolt_ensure_adventurer_effect = { LOCATION = scope:hd_revolt_spawn_province }
		title:d_fufeng = { add_to_list = hd_revolt_candidates_wangguo }
		hd_historical_revolt_pick_defender_effect = {
			LIST1 = hd_revolt_candidates_wangguo
			LIST2 = hd_revolt_lieges_wangguo
		}
	}
	if = {
		limit = {
			exists = scope:hd_revolt_defender
			exists = scope:hd_revolt_primary_target
			NOT = { scope:hd_revolt_leader = scope:hd_revolt_defender }
		}
		# Merge check: the Beigong Boyu revolt war (possibly inherited) is still running
		# against the very same defender.
		scope:hd_revolt_defender = {
			random_character_war = {
				limit = {
					using_cb = hd_historical_revolt_cb
					any_war_attacker = {
						has_character_flag = hd_revolt_leader_beigong
					}
				}
				save_scope_as = hd_revolt_beigong_war
			}
		}
		if = {
			limit = { exists = scope:hd_revolt_beigong_war }
			scope:hd_revolt_beigong_war = {
				add_attacker = scope:hd_revolt_leader
				set_variable = hd_revolt_wangguo_joined
			}
			scope:hd_revolt_leader = {
				spawn_army = {
					# TOTAL = 16000
					name = hd_revolt_army_wangguo
					levies = 4000
					men_at_arms = { type = light_footmen men = 3000 }
					men_at_arms = { type = light_horsemen men = 5000 }
					men_at_arms = { type = horse_archers men = 4000 }
					location = scope:hd_revolt_spawn_province
					origin = scope:hd_revolt_spawn_province
					uses_supply = yes
					inheritable = no
					war = scope:hd_revolt_beigong_war
					save_temporary_scope_as = hd_revolt_spawned_army
				}
			}
			scope:hd_revolt_spawned_army = { assign_commander = scope:hd_revolt_leader }
			scope:hd_revolt_leader = {
				add_character_flag = hd_revolt_leader_wangguo
			}
			set_global_variable = hd_revolt_wangguo_fired
			scope:hd_revolt_defender = {
				if = {
					limit = { is_ai = no }
					trigger_event = hd_historical_revolts.0005
				}
			}
		}
		else = {
			scope:hd_revolt_leader = {
				hd_historical_revolt_start_one_target_war_effect = { TARGET1 = scope:hd_revolt_primary_target }
			}
			if = {
				limit = { exists = scope:hd_revolt_war }
				scope:hd_revolt_leader = {
					spawn_army = {
						# TOTAL = 16000
						name = hd_revolt_army_wangguo
						levies = 4000
						men_at_arms = { type = light_footmen men = 3000 }
						men_at_arms = { type = light_horsemen men = 5000 }
						men_at_arms = { type = horse_archers men = 4000 }
						location = scope:hd_revolt_spawn_province
						origin = scope:hd_revolt_spawn_province
						uses_supply = yes
						inheritable = no
						war = scope:hd_revolt_war
						save_temporary_scope_as = hd_revolt_spawned_army
					}
				}
				scope:hd_revolt_spawned_army = { assign_commander = scope:hd_revolt_leader }
				scope:hd_revolt_leader = { add_character_flag = hd_revolt_leader_wangguo }
				set_global_variable = hd_revolt_wangguo_fired
				scope:hd_revolt_defender = {
