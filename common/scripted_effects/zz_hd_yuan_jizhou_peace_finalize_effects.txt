# Finalize Han Fu's handover when the historical Gongsun Zan vs Han Fu war
# is invalidated by Jizhou changing hands. This is deliberately idempotent.
hd_yuan_jizhou_finalize_handover_peace_effect = {
	if = {
		limit = {
			exists = global_var:hd_yuan_jizhou_handover_started
			exists = title:k_uuii_jizhou.holder
			title:k_uuii_jizhou.holder = character:yuan_shao
			character:yuan_shao = { is_alive = yes }
		}
		# The historical Gongsun Zan-Han Fu war is over permanently once Yuan
		# actually receives Jizhou. The watchdog must never recreate it against Yuan.
		set_global_variable = hd4_jizhou_finished
		if = {
			limit = { NOT = { exists = global_var:hd_yuan_jizhou_handover_complete } }
			set_global_variable = hd_yuan_jizhou_handover_complete
			set_global_variable = hd_yuan_jizhou_realm_transferred
		}
		character:yuan_shao = {
			if = {
				limit = {
					exists = character:gong_sun_zan
					character:gong_sun_zan = { is_alive = yes }
					NOT = { has_relation_rival = character:gong_sun_zan }
				}
				set_relation_rival = {
					target = character:gong_sun_zan
					reason = rival_historical
				}
			}
			add_character_flag = hd_yuan_jizhou_claim_received
			if = {
				limit = { NOT = { has_character_flag = hd4_jizhou_handover_notified } }
				add_character_flag = hd4_jizhou_handover_notified
				trigger_event = hd_yuan_jizhou.0011
			}
		}
	}
}
