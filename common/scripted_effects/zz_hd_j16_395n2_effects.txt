# 换主（同一政权内，不杀人）：$NEW$ 在 $OLD$ 一系内。$NEW$ 若是有地封臣先独立，再把 $OLD$ 的县至帝国级头衔（世族、朝官、无地型头衔除外）与封臣尽归 $NEW$；
# $OLD$ 留作廷臣（照 p5-A1 先例 hd_370_take_realm_effect，另加先独立）。Scope：任意
hd_j16_n2_take_realm_effect = {
	if = {
		limit = {
			$NEW$ = { is_alive = yes }
			$OLD$ = {
				is_alive = yes
				is_landed = yes
			}
			NOT = { $NEW$ = { this = $OLD$ } }
		}
		if = {
			limit = { exists = scope:hd_j16_n2_tr_new }
			clear_saved_scope = hd_j16_n2_tr_new
		}
		$NEW$ = { save_scope_as = hd_j16_n2_tr_new }
		scope:hd_j16_n2_tr_new = { hd_nbc_become_independent_effect = yes }
		create_title_and_vassal_change = { type = granted save_scope_as = hd_j16_n2_tr_change add_claim_on_loss = no }
		$OLD$ = {
			every_held_title = {
				limit = {
					is_noble_family_title = no
					tgp_is_minister_title = no
					is_landless_type_title = no
					tier >= tier_county
				}
				change_title_holder = {
					holder = scope:hd_j16_n2_tr_new
					change = scope:hd_j16_n2_tr_change
					take_baronies = yes
				}
			}
			every_vassal = {
				limit = { NOT = { this = scope:hd_j16_n2_tr_new } }
				change_liege = { liege = scope:hd_j16_n2_tr_new change = scope:hd_j16_n2_tr_change }
			}
		}
		resolve_title_and_vassal_change = scope:hd_j16_n2_tr_change
		debug_log = "HD_J16 $TAG$ FIRE handover"
	}
	else = { debug_log = "HD_J16 $TAG$ SKIP handover_invalid" }
}

# 换主执行体（节点 AI 分支与玩家「逊位」选项共用）。Scope：现任君主（AI 分支在 scope:<现任> = { } 内调用，事件选项里即事件 root；
#   不用 root：心跳里的 root 是 on_action 的根，不是现任）。
# 解析史实继任者 $ID$：已是现任 → 只设 $RESULT$；在世、未被囚、在现任一系内 → 设 $RESULT$ 并换主；否则记 SKIP（玩家晚回复、继任者已死等）
hd_j16_n2_handover_effect = {
	save_scope_as = hd_j16_n2_ho_old
	if = {
		limit = { exists = scope:hd_j16_n2_ho_new }
		clear_saved_scope = hd_j16_n2_ho_new
	}
	hd_j16_resolve_effect = { ID = $ID$ AS = hd_j16_n2_ho_new }
	if = {
		limit = { scope:hd_j16_n2_ho_new ?= { this = scope:hd_j16_n2_ho_old } }
		set_global_variable = $RESULT$
		debug_log = "HD_J16 $TAG$ FIRE already_ruler"
	}
	else_if = {
		limit = {
			scope:hd_j16_n2_ho_old = {
				is_alive = yes
				is_landed = yes
			}
			scope:hd_j16_n2_ho_new ?= {
				is_imprisoned = no
				top_liege ?= scope:hd_j16_n2_ho_old
			}
		}
		set_global_variable = $RESULT$
		hd_j16_n2_take_realm_effect = { NEW = scope:hd_j16_n2_ho_new OLD = scope:hd_j16_n2_ho_old TAG = $TAG$ }
	}
	else = { debug_log = "HD_J16 $TAG$ SKIP player_late_reply" }
}

# 继统一步（X30、X34、X42 次步、X70、X74 共用）：上游 $UP$ 已设、前任 $PREV$ 已死；按位置取政权之主（$GETTER$ 存为 scope:$SC$）；
# 继任者 $NEXT$ 在世，在其一系或已是其主 → 设 $DONE$：现任是 AI 或已是继任者 → 换主执行体；现任是玩家 → 弹 $EVENT$（a 逊位 / b 拒之）。
# 条件未齐：等待，WAIT 只记一次（$DONE$_wlog）。Scope：任意
hd_j16_n2_succ_step_effect = {
	if = {
		limit = {
			has_global_variable = $UP$
			hd_j16_dead_trigger = { ID = $PREV$ }
		}
		$GETTER$ = { AS = $SC$ }
		if = {
			limit = { exists = scope:hd_j16_n2_ss_next }
			clear_saved_scope = hd_j16_n2_ss_next
		}
		hd_j16_resolve_effect = { ID = $NEXT$ AS = hd_j16_n2_ss_next }
		if = {
			limit = {
				exists = scope:$SC$
				scope:hd_j16_n2_ss_next ?= {
					is_imprisoned = no
					OR = {
						this = scope:$SC$
						top_liege ?= scope:$SC$
					}
				}
			}
			set_global_variable = $DONE$
			scope:$SC$ = {
				if = {
					limit = {
						OR = {
							is_ai = yes
							this = scope:hd_j16_n2_ss_next
						}
					}
					hd_j16_n2_handover_effect = { ID = $NEXT$ RESULT = $RESULT$ TAG = $TAG$ }
				}
				else = {
					trigger_event = $EVENT$
					debug_log = "HD_J16 $TAG$ WAIT player_asked"
				}
			}
		}
		else_if = {
			limit = { NOT = { has_global_variable = $DONE$_wlog } }
			set_global_variable = $DONE$_wlog
			debug_log = "HD_J16 $TAG$ WAIT not_ready"
		}
	}
	else_if = {
		limit = { NOT = { has_global_variable = $DONE$_wlog } }
		set_global_variable = $DONE$_wlog
		debug_log = "HD_J16 $TAG$ WAIT not_ready"
	}
}

			limit = { exists = scope:hd_j16_x90_key }

		}

		}
		if = {
			limit = { exists = scope:hd_nbc_war_xiliang }
			set_global_variable = hd_j16_xiliang_started
			hd_j16_n2_war_key_effect = { WAR = scope:hd_nbc_war_xiliang DEF = scope:hd_j16_x90_def KEY = hd_j16_x90_key }
			set_global_variable = { name = hd_j16_xiliang_end_att value = scope:hd_j16_x90_att }
			set_global_variable = { name = hd_j16_xiliang_end_def value = scope:hd_j16_x90_def }
			set_global_variable = { name = hd_j16_xiliang_end_key value = scope:hd_j16_x90_key }
			hd_j16_n2_front_army_effect = { DUCHY = scope:hd_j16_x90_key.duchy SIDE = scope:hd_j16_x90_att WHO = scope:hd_j16_x90_att SIZE = 10000 NAME = hd_j16_n2_army_xiliang_name }
			scope:hd_j16_x90_def = {
				if = {
					limit = { is_ai = no }
					trigger_event = hd_j16.5796
				}
			}
			debug_log = "HD_J16 e5_x90 FIRE war_started"
		}
		else_if = {
			limit = { NOT = { has_global_variable = hd_j16_xiliang_end_wsf_wlog } }
			set_global_variable = hd_j16_xiliang_end_wsf_wlog

		}
	}
}

			NOT = { global_var:hd_j16_xiliang_end_def ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:xiliang } } }
		}
		set_global_variable = hd_j16_xiliang_end_done

			limit = { has_global_variable = hd_nbc_xiliang_active }
			remove_global_variable = hd_nbc_xiliang_active
		}
		if = {
			limit = {
				global_var:hd_j16_xiliang_end_key.holder ?= {
					OR = {
						this = global_var:hd_j16_xiliang_end_att
						top_liege ?= global_var:hd_j16_xiliang_end_att
						top_liege ?= { hd_j16_n2_beiliang_dyn_trigger = yes }
					}
				}
			}
			set_global_variable = hd_j16_xiliang_end_won
			debug_log = "HD_J16 e5_x90 FIRE result_won"
		}
		else = {
			set_global_variable = hd_j16_xiliang_end_lost
			debug_log = "HD_J16 e5_x90 FIRE result_lost"
		}
	}
