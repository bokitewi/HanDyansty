# 184 剧本：刘备入主徐州之后。吕布来投 -> 吕布夺徐州 -> 刘备投刘表。
# 入口只有刘备的月度心跳 hd_lb_tick_effect（已过 hd_lb_actor_trigger）；条件不满足就等下个月，
# 只有完成或玩家拒绝才结束节点。每个节点一把带期限的全局待决锁，双方都是玩家时按先后串行。

# Scope：刘备。
hd_lbxz_tick_effect = {
	if = {
		limit = { this = character:liu_bei }
		# 屏蔽期间结束已有的原版冒险者 AI 故事（闸门只挡新建）。
		character:lv_bu ?= {
			if = {
				limit = { hd_lbxz_block_adventurer_ai_trigger = yes }
				every_owned_story = {
					limit = { story_type = story_adventurer_ai }
					end_story = yes
				}
			}
		}
		if = {
			limit = { hd_lbxz_block_adventurer_ai_trigger = yes }
			every_owned_story = {
				limit = { story_type = story_adventurer_ai }
				end_story = yes
			}
		}
		# 得地后原版 on_action 可能重设政体，短期内补回天朝制。
		character:lv_bu ?= {
			if = {
				limit = { has_character_flag = hd_lbxz_gov_check }
				hd_force_celestial_effect = yes
			}
		}
		if = {
			limit = { has_character_flag = hd_lbxz_gov_check }
			hd_force_celestial_effect = yes
		}
		# 丹阳兵首领夺徐州后同样补天朝制
		global_var:hd_lbxz_seizer ?= {
			if = {
				limit = {
					is_alive = yes
					has_character_flag = hd_lbxz_gov_check
					NOT = { this = character:lv_bu }
				}
				hd_force_celestial_effect = yes
			}
		}
		hd_lbxz_recall_retinue_effect = yes
		# 夺城收尾失败（刘备仍有残留头衔等）时按月重试清扫与收尾。
		if = {
			limit = {
				exists = global_var:hd_lbxz_seize_failed
				NOT = { exists = global_var:hd_lbxz_seized }
			}
			hd_lbxz_sweep_remainder_effect = yes
			hd_lbxz_finish_seize_effect = yes
			hd_lb_sync_story_effect = yes
		}
		# 没有冒险者 DLC 时刘备失地无法建营，夺城节点直接跳过。
		if = {
			limit = {
				exists = global_var:hd_lbxz_shelter_done
				NOT = { exists = global_var:hd_lbxz_seize_resolved }
				NOT = { has_dlc_feature = roads_to_power }
			}
			character:lv_bu ?= { remove_character_flag = hd_lbxz_sheltered }
			set_global_variable = hd_lbxz_seize_resolved
			debug_log = "HD_LBXZ SEIZE_SKIPPED_NO_RTP"
		}
		if = {
			limit = {
				NOT = { exists = global_var:hd_lbxz_shelter_pending }
				hd_lbxz_shelter_ready_trigger = yes
			}
			hd_lbxz_open_shelter_effect = yes
		}
		if = {
			limit = {
				NOT = { exists = global_var:hd_lbxz_seize_pending }
				hd_lbxz_seize_ready_trigger = yes
			}
			hd_lbxz_open_seize_effect = yes
		}
		# 2026-10-24 boki：吕布不来夺徐州时的兜底
		hd_lbxz_danyang_effect = yes
		if = {
			limit = {
				NOT = { exists = global_var:hd_lbxz_jing_pending }
				hd_lbxz_jing_ready_trigger = yes
			}
			hd_lbxz_open_jing_effect = yes
		}
		# 第二段：依附刘表后三顾茅庐、隆中对、入蜀。
		hd_lbxz_zgl_tick_effect = yes
		hd_lbxz_shu_tick_effect = yes
	}
}

# 两个角色互结同盟，双向附原版 is_alliance_valid 认可的议定同盟理由。Scope：任意。
hd_lbxz_alliance_effect = {
	$A$ = {
		if = {
			limit = { NOT = { is_at_war_with = $B$ } }
			add_opinion = { modifier = event_negotiated_alliance_opinion target = $B$ }
			$B$ = { add_opinion = { modifier = event_negotiated_alliance_opinion target = $A$ } }
			if = {
				limit = { NOT = { is_allied_to = $B$ } }
				create_alliance = {
					target = $B$
					allied_through_owner = $A$
					allied_through_target = $B$
				}
			}
		}
	}
}

# Scope：刚得地的前冒险者。原版 on_action 通常已销毁营地头衔，这里只补漏。
hd_lbxz_destroy_camp_title_effect = {
	save_scope_as = hd_lbxz_camp_owner
	every_held_title = {
		limit = { has_variable = adventurer_creation_reason }
		save_scope_as = hd_lbxz_camp_title
		scope:hd_lbxz_camp_owner = { destroy_title = scope:hd_lbxz_camp_title }
	}
}

########## 节点 1：吕布来投 ##########

hd_lbxz_open_shelter_effect = {
	set_global_variable = { name = hd_lbxz_shelter_pending value = yes days = 60 }
	if = {
		limit = { character:lv_bu = { is_ai = no } }
		character:lv_bu = { trigger_event = hd_lbxz.0100 }
	}
	else = { hd_lbxz_lubu_requests_shelter_effect = yes }
}

# 吕布已决意来投，由刘备决定收留与否（两人都是玩家时在此串行）。
hd_lbxz_lubu_requests_shelter_effect = {
	if = {
		limit = { character:liu_bei = { is_ai = no } }
		set_global_variable = { name = hd_lbxz_shelter_pending value = yes days = 60 }
		character:liu_bei = { trigger_event = hd_lbxz.0101 }
	}
	else = { hd_lbxz_shelter_effect = yes }
}

hd_lbxz_shelter_declined_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_lbxz_shelter_resolved } }
		set_global_variable = hd_lbxz_shelter_resolved
		set_global_variable = hd_lbxz_shelter_declined
		debug_log = "HD_LBXZ SHELTER_DECLINED"
	}
	remove_global_variable = hd_lbxz_shelter_pending
}

# 迁营与结盟都回读成功才算完成，否则下月重试。
hd_lbxz_shelter_effect = {
	if = {
		limit = { hd_lbxz_shelter_ready_trigger = yes }
		character:lv_bu = {
			if = { limit = { is_imprisoned = yes } release_from_prison = yes }
		}
		clear_saved_scope = hd_lbxz_camp_destination
		# 史实刘备使吕布屯小沛（沛县）；沛县不在刘备治下时，改在刘备治下徐州境内一个非都城的县，最后才用都城。
		if = {
			limit = {
				exists = title:c_uuii_p_4349.holder
				title:c_uuii_p_4349.holder = {
					OR = {
						this = character:liu_bei
						is_vassal_or_below_of = character:liu_bei
					}
				}
			}
			title:c_uuii_p_4349.title_province = { save_scope_as = hd_lbxz_camp_destination }
		}
		else = {
			character:liu_bei = {
				random_sub_realm_county = {
					limit = {
						target_is_de_jure_liege_or_above = title:k_uuii_xuzhou
						NOT = { this = character:liu_bei.capital_county }
					}
					title_province = { save_scope_as = hd_lbxz_camp_destination }
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_lbxz_camp_destination } }
			character:liu_bei.capital_province ?= { save_scope_as = hd_lbxz_camp_destination }
		}
		if = {
			limit = { exists = scope:hd_lbxz_camp_destination }
			character:lv_bu = {
				if = {
					limit = { NOT = { has_variable = hd_lbxz_origin } }
					domicile.domicile_location ?= {
						save_scope_as = hd_lbxz_origin_province
						character:lv_bu = { set_variable = { name = hd_lbxz_origin value = scope:hd_lbxz_origin_province } }
					}
				}
				domicile = { move_domicile = scope:hd_lbxz_camp_destination }
			}
			hd_lbxz_alliance_effect = { A = character:liu_bei B = character:lv_bu }
		}
		if = {
			limit = {
				exists = scope:hd_lbxz_camp_destination
				character:lv_bu = {
					domicile.domicile_location ?= scope:hd_lbxz_camp_destination
					is_allied_to = character:liu_bei
				}
			}
			character:lv_bu = {
				add_character_flag = hd_lbxz_sheltered
				add_character_flag = { flag = hd_lbxz_shelter_wait days = 365 }
			}
			set_global_variable = hd_lbxz_shelter_resolved
			set_global_variable = hd_lbxz_shelter_done
			debug_log = "HD_LBXZ SHELTER_DONE"
		}
		else = { debug_log = "HD_LBXZ SHELTER_RETRY" }
	}
	remove_global_variable = hd_lbxz_shelter_pending
}

########## 节点 2：吕布夺徐州 ##########

hd_lbxz_open_seize_effect = {
	set_global_variable = { name = hd_lbxz_seize_pending value = yes days = 60 }
	if = {
		limit = { character:lv_bu = { is_ai = no } }
		character:lv_bu = { trigger_event = hd_lbxz.0200 }
	}
	else = { hd_lbxz_lubu_moves_effect = yes }
}

# 吕布已决意袭取下邳。刘备是玩家时一律由他选择（两人都是玩家时在此串行）。
hd_lbxz_lubu_moves_effect = {
	if = {
		limit = { character:liu_bei = { is_ai = no } }
		set_global_variable = { name = hd_lbxz_seize_pending value = yes days = 60 }
		character:liu_bei = { trigger_event = hd_lbxz.0201 }
	}
	else = { hd_lbxz_seize_effect = yes }
}

# 玩家吕布守约不动。
hd_lbxz_seize_declined_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_lbxz_seize_resolved } }
		set_global_variable = hd_lbxz_seize_resolved
		set_global_variable = hd_lbxz_seize_declined
		character:lv_bu ?= { remove_character_flag = hd_lbxz_sheltered }
		debug_log = "HD_LBXZ SEIZE_DECLINED_BY_LUBU"
	}
	remove_global_variable = hd_lbxz_seize_pending
}

# 玩家刘备先发制人，逐走吕布：断盟，吕布营地迁回来投前的位置。
hd_lbxz_expel_lubu_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_lbxz_seize_resolved } }
		character:liu_bei = {
			if = {
				limit = { is_allied_to = character:lv_bu }
				break_alliance = character:lv_bu
			}
		}
		character:lv_bu ?= {
			remove_character_flag = hd_lbxz_sheltered
			if = {
				limit = {
					has_variable = hd_lbxz_origin
					exists = domicile
					is_travelling = no
				}
				domicile = { move_domicile = character:lv_bu.var:hd_lbxz_origin }
			}
		}
		set_global_variable = hd_lbxz_seize_resolved
		set_global_variable = hd_lbxz_lubu_expelled
		debug_log = "HD_LBXZ LUBU_EXPELLED_BY_LIU"
	}
	remove_global_variable = hd_lbxz_seize_pending
}

# Scope：任意。仍在刘备治下有地、没有世族头衔的电脑亲随，夺城时先剥地再随刘备出走。
hd_lbxz_mark_follower_effect = {
	$FOLLOWER$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_ai = yes
				is_landed = yes
				is_vassal_or_below_of = character:liu_bei
				NOT = { any_held_title = { is_noble_family_title = yes } }
			}
			add_to_list = hd_lbxz_stripped_followers
			add_to_list = hd_lbxz_retinue
			add_character_flag = { flag = hd_lbxz_stripped days = 730 }
		}
	}
}

# 一个事务内整包移交，再校验；无论成败都结束节点，不重复执行半截事务。
hd_lbxz_seize_effect = {
	if = {
		limit = { hd_lbxz_seize_ready_trigger = yes }
		character:lv_bu = {
			if = { limit = { is_imprisoned = yes } release_from_prison = yes }
		}
		set_global_variable = { name = hd_lbxz_seizer value = character:lv_bu }
		hd_lbxz_transfer_xuzhou_effect = yes
		character:lv_bu = {
			if = {
				limit = { is_landed = yes has_title = title:c_uuii_p_284 }
				set_realm_capital = title:c_uuii_p_284
			}
			remove_character_flag = hd_lbxz_sheltered
		}

		# 先清扫刘备与被剥地亲随残留的男爵领等普通头衔，再收尾；收尾失败时由心跳按月重试。
		hd_lbxz_sweep_remainder_effect = yes
		hd_lbxz_finish_seize_effect = yes
		set_global_variable = hd_lbxz_seize_resolved
		hd_lb_sync_story_effect = yes
	}
	remove_global_variable = hd_lbxz_seize_pending
}

# 夺徐州者存在 global_var:hd_lbxz_seizer（吕布或丹阳兵首领），取出存为 scope:hd_lbxz_seizer_s；没有该变量时认吕布。
hd_lbxz_save_seizer_effect = {
	if = {
		limit = { exists = global_var:hd_lbxz_seizer }
		global_var:hd_lbxz_seizer = { save_scope_as = hd_lbxz_seizer_s }
	}
	else = { character:lv_bu ?= { save_scope_as = hd_lbxz_seizer_s } }
}

# 夺徐州的整包移交（吕布、丹阳兵之乱共用）：刘备全部县级以上普通头衔、电脑封臣、被剥地的电脑亲随一并交给夺徐州者。
# 调用前须设好 global_var:hd_lbxz_seizer。迁都与收尾由调用方处理。
hd_lbxz_transfer_xuzhou_effect = {
	hd_lbxz_save_seizer_effect = yes
	if = {
		limit = { exists = scope:hd_lbxz_seizer_s }
		clear_saved_scope = hd_lbxz_old_liege
		character:liu_bei = {
			if = {
				limit = { is_allied_to = scope:hd_lbxz_seizer_s }
				break_alliance = scope:hd_lbxz_seizer_s
			}
			# 夺徐州者取代刘备的政治位置：刘备原为帝国级主君的封臣时，夺徐州者改投该主君。
			liege ?= {
				if = {
					limit = {
						is_alive = yes
						NOT = { this = character:liu_bei }
						highest_held_title_tier > tier_kingdom
					}
					save_scope_as = hd_lbxz_old_liege
				}
			}
			every_held_title = {
				limit = {
					tier >= tier_county # baronies move with their county
					is_noble_family_title = no
					is_landless_type_title = no
				}
				add_to_list = hd_lbxz_ceded_titles
			}
			every_vassal = {
				limit = { is_alive = yes NOT = { this = scope:hd_lbxz_seizer_s } }
				if = {
					limit = { is_ai = yes }
					add_to_list = hd_lbxz_ceded_vassals
				}
				else = { add_to_list = hd_lbxz_player_vassals }
			}
			# 夺徐州者本人（丹阳兵首领原是刘备廷臣）与随他作乱的人不随刘备出走。
			every_courtier = {
				limit = {
					is_alive = yes
					NOT = { this = scope:hd_lbxz_seizer_s }
					NOT = { has_character_flag = hd_lbxz_follows_seizer }
				}
				add_to_list = hd_lbxz_retinue
			}
		}
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:guan_yu }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:zhang_fei }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:zhao_yun }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:mi_zhu }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:mi_fang }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:sun_gan }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:jian_yong }
		hd_lbxz_mark_follower_effect = { FOLLOWER = character:chen_dao }
		# 回收名单存成刘备身上的变量列表，回收不成功按月重试。
		character:liu_bei = {
			clear_variable_list = hd_lbxz_retinue_var
			set_variable = { name = hd_lbxz_recall_tries value = 0 }
		}
		every_in_list = {
			list = hd_lbxz_retinue
			save_scope_as = hd_lbxz_retinue_member
			character:liu_bei = { add_to_variable_list = { name = hd_lbxz_retinue_var target = scope:hd_lbxz_retinue_member } }
		}
		# 玩家直属封臣、被剥地亲随的玩家封臣，都不被强制改投吕布：转移前先让其独立。
		every_in_list = {
			list = hd_lbxz_stripped_followers
			every_vassal = {
				limit = { is_ai = no }
				add_to_list = hd_lbxz_player_vassals
			}
		}
		if = {
			limit = { any_in_list = { list = hd_lbxz_player_vassals always = yes } }
			create_title_and_vassal_change = { type = independency save_scope_as = hd_lbxz_player_release add_claim_on_loss = no }
			every_in_list = {
				list = hd_lbxz_player_vassals
				becomes_independent = { change = scope:hd_lbxz_player_release }
			}
			resolve_title_and_vassal_change = scope:hd_lbxz_player_release
		}
		# 原版建营会让非家人宗教顾问消失、其余议员革职：先革掉全部议员。
		character:liu_bei = {
			every_councillor = {
				limit = { NOT = { has_council_position = councillor_spouse } }
				save_scope_as = hd_lbxz_fired
				character:liu_bei = { fire_councillor = scope:hd_lbxz_fired }
			}
		}
		# 照搬吕布长安败走：失地前先让无地、不在旅行的非家人廷臣脱离，建营后再收回；家人原版本来就留着。
		every_in_list = {
			list = hd_lbxz_retinue
			limit = {
				is_alive = yes
				is_ruler = no
				is_landed = no
				is_travelling = no
				NOT = { is_in_list = hd_lbxz_stripped_followers }
				NOR = {
					is_close_or_extended_family_of = character:liu_bei
					is_consort_of = character:liu_bei
				}
			}
			move_to_pool = yes
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = hd_lbxz_seize_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = hd_lbxz_stripped_followers
			every_held_title = {
				limit = {
					tier >= tier_county
					is_noble_family_title = no
					is_landless_type_title = no
				}
				change_title_holder_include_vassals = { holder = scope:hd_lbxz_seizer_s change = scope:hd_lbxz_seize_change }
			}
			every_vassal = {
				limit = { is_ai = yes }
				change_liege = { liege = scope:hd_lbxz_seizer_s change = scope:hd_lbxz_seize_change }
			}
		}
		every_in_list = {
			list = hd_lbxz_ceded_titles
			change_title_holder_include_vassals = { holder = scope:hd_lbxz_seizer_s change = scope:hd_lbxz_seize_change }
		}
		every_in_list = {
			list = hd_lbxz_ceded_vassals
			limit = {
				is_alive = yes
				NOT = { is_in_list = hd_lbxz_stripped_followers }
			}
			change_liege = { liege = scope:hd_lbxz_seizer_s change = scope:hd_lbxz_seize_change }
		}
		if = {
			limit = { exists = scope:hd_lbxz_old_liege }
			scope:hd_lbxz_seizer_s = { change_liege = { liege = scope:hd_lbxz_old_liege change = scope:hd_lbxz_seize_change } }
		}
		resolve_title_and_vassal_change = scope:hd_lbxz_seize_change

		scope:hd_lbxz_seizer_s = {
			if = {
				limit = { is_landed = yes }
				hd_force_celestial_effect = yes
				hd_lbxz_destroy_camp_title_effect = yes
				add_character_flag = { flag = hd_lbxz_gov_check days = 60 }
			}
		}

# 一个事务内交接 scope:hd_lbxz_jing_duchy：郡头衔、原郡守直辖的郡内各县（刘表都城除外）归刘备，
# 郡内其余县主中原郡守的直属封臣改作刘备封臣。Scope：任意。
hd_lbxz_jing_take_duchy_effect = {
	scope:hd_lbxz_jing_duchy.holder = { save_scope_as = hd_lbxz_jing_old }
	scope:hd_lbxz_jing_duchy = {
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_county
				holder ?= scope:hd_lbxz_jing_old
				NOT = { character:liu_biao.capital_county ?= this }
			}
			add_to_list = hd_lbxz_jing_counties
		}
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_county
				exists = holder
				NOT = { holder = scope:hd_lbxz_jing_old }
				NOT = { holder = character:liu_bei }
				holder = { liege ?= scope:hd_lbxz_jing_old }
			}
			holder = {
				if = {
					limit = { NOT = { is_in_list = hd_lbxz_jing_vassals } }
					add_to_list = hd_lbxz_jing_vassals
				}
			}
		}
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_lbxz_jing_change
		add_claim_on_loss = no
	}
	every_in_list = {
		list = hd_lbxz_jing_counties
		change_title_holder = {
			holder = character:liu_bei
			change = scope:hd_lbxz_jing_change
			take_baronies = yes
		}
	}
	scope:hd_lbxz_jing_duchy = {
		change_title_holder = {
			holder = character:liu_bei
			change = scope:hd_lbxz_jing_change
		}
	}
	every_in_list = {
		list = hd_lbxz_jing_vassals
		change_liege = { liege = character:liu_bei change = scope:hd_lbxz_jing_change }
	}
	resolve_title_and_vassal_change = scope:hd_lbxz_jing_change
	character:liu_bei = {
		if = {
			limit = { exists = scope:hd_lbxz_jing_duchy.title_capital_county scope:hd_lbxz_jing_duchy.title_capital_county = { holder ?= character:liu_bei } }
			set_realm_capital = scope:hd_lbxz_jing_duchy.title_capital_county
		}
		else = {
			random_held_title = {
				limit = { tier = tier_county target_is_de_jure_liege_or_above = scope:hd_lbxz_jing_duchy }
				save_scope_as = hd_lbxz_jing_seat
			}
			if = {
				limit = { exists = scope:hd_lbxz_jing_seat }
				set_realm_capital = scope:hd_lbxz_jing_seat
			}
		}
	}
	debug_log = "HD_LBXZ JING_DUCHY_TAKEN"
}

				if = {
					limit = { has_relation_rival = scope:hd_lbxz_zgl_s }
					remove_relation_rival = scope:hd_lbxz_zgl_s
				}
				if = {
					limit = { NOT = { has_relation_best_friend = scope:hd_lbxz_zgl_s } }
					set_relation_best_friend = { reason = best_friend_counsel_and_support copy_reason = friend target = scope:hd_lbxz_zgl_s }
				}
			}
			if = {
				limit = { character:liu_bei = { has_relation_best_friend = scope:hd_lbxz_zgl_s } }
				debug_log = "HD_LBXZ LONGZHONG_BEST_FRIEND"
			}
			else = { debug_log = "HD_LBXZ LONGZHONG_BEST_FRIEND_FAILED" }
			# 若已入 WJ 隐士，走唯一出山出口。
			scope:hd_lbxz_zgl_s = {
				if = { limit = { has_trait = WJ_yangwang_trait } WJ_yangwang_exit_effect = yes }
			}
			hd_lbxz_zgl_close_effect = yes
			set_global_variable = hd_lbxz_longzhong_done
			debug_log = "HD_LBXZ LONGZHONG_DONE"
			character:liu_bei = {
				if = { limit = { is_ai = no } trigger_event = hd_lbxz.0410 }
			}
		}
		else_if = {
			limit = { scope:hd_lbxz_zgl_s = { is_ai = no } }
			# 玩家选了出山仍没能入宫：不再反复弹事件，按跳过处理。
			hd_lbxz_zgl_close_effect = yes
			set_global_variable = hd_lbxz_zgl_skipped
			debug_log = "HD_LBXZ LONGZHONG_PLAYER_JOIN_FAILED"
		}
		else = { debug_log = "HD_LBXZ LONGZHONG_JOIN_FAILED_RETRY" }
	}
	remove_global_variable = hd_lbxz_zgl_pending
}

# 节点 6、7 的每月入口。Scope：刘备。
hd_lbxz_shu_tick_effect = {
	# 战后判定：入蜀之战已不在刘备的战争中（胜、负、白和、作废都算结束）。
	if = {
		limit = {
			exists = global_var:hd_lbxz_shu_war_started
			NOT = { any_character_war = { var:hd_world_war_kind ?= flag:lb_yizhou } }
		}
		remove_global_variable = hd_lbxz_shu_war_started
		hd_lbxz_shu_landed_fix_effect = yes
		if = {
			limit = {
				title:c_uuii_p_4282.holder ?= {
					OR = {
						this = character:liu_bei
						target_is_liege_or_above = character:liu_bei
					}
				}
			}
			debug_log = "HD_LBXZ SHU_CHENGDU_TAKEN"
			set_global_variable = hd_hev_184_lbxz_chengdu
			hd_lbxz_shu_after_win_effect = yes
		}
		else_if = {
			limit = { global_var:hd_lbxz_shu_tries >= 2 }
			set_global_variable = hd_lbxz_shu_done
			debug_log = "HD_LBXZ SHU_GAVE_UP"
		}
		else = {
			add_character_flag = { flag = hd_lbxz_shu_cooldown years = 3 }
			debug_log = "HD_LBXZ SHU_WAR_LOST"
		}
	}
	if = {
		limit = {
			NOT = { exists = global_var:hd_lbxz_shu_pending }
			hd_lbxz_shu_ready_trigger = yes
		}
		hd_lbxz_open_shu_effect = yes
	}
}

hd_lbxz_open_shu_effect = {
	set_global_variable = { name = hd_lbxz_shu_pending value = yes days = 60 }
	# 玩家刘备在 0500 允过一次后，开战失败的重试不再弹事件。
	if = {
		limit = {
			character:liu_bei = { is_ai = no }
			NOT = { exists = global_var:hd_lbxz_shu_liu_agreed }
		}
		character:liu_bei = { trigger_event = hd_lbxz.0500 }
	}
	else = { hd_lbxz_shu_start_effect = yes }
}

hd_lbxz_shu_declined_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_lbxz_shu_done } }
		set_global_variable = hd_lbxz_shu_done
		debug_log = "HD_LBXZ SHU_DECLINED"
	}
	remove_global_variable = hd_lbxz_shu_pending
}

# 刘备若仍是封臣先独立，与原领主互给停战；宣战回读成功后补兵。目标在此重新取。
hd_lbxz_shu_start_effect = {
	if = {
		limit = { hd_lbxz_shu_ready_trigger = yes }
		title:c_uuii_p_4282.holder.top_liege = { save_scope_as = hd_lbxz_shu_target }
		character:liu_bei = {
			# 冒险者本就独立，自然跳过这一步。
			if = {
				limit = { is_independent_ruler = no }
				liege = { save_scope_as = hd_lbxz_shu_old_liege }
				create_title_and_vassal_change = { type = independency save_scope_as = hd_lbxz_shu_indep_change add_claim_on_loss = no }
				becomes_independent = { change = scope:hd_lbxz_shu_indep_change }
				resolve_title_and_vassal_change = scope:hd_lbxz_shu_indep_change
				if = {
					limit = { is_independent_ruler = yes }
					debug_log = "HD_LBXZ SHU_INDEPENDENT"
					# 原领主就是益州之主或在其阵营内时不给停战，免得宣战即毁约。
					if = {
						limit = {
							NOT = { is_at_war_with = scope:hd_lbxz_shu_old_liege }
							scope:hd_lbxz_shu_old_liege = {
								NOT = { this = scope:hd_lbxz_shu_target }
								NOT = { top_liege = scope:hd_lbxz_shu_target }
							}
						}
						add_truce_both_ways = {
							character = scope:hd_lbxz_shu_old_liege
							years = 5
							name = hd_lbxz_shu_truce
						}
					}
					scope:hd_lbxz_shu_old_liege = {
						if = { limit = { is_ai = no } trigger_event = hd_lbxz.0501 }
					}
				}
			}
			if = {
				limit = { is_independent_ruler = yes }
				hd_tb_arm_effect = { KIND = historical }
				start_war = {
					cb = hd_dc_tunbing_cb
					target = scope:hd_lbxz_shu_target
					target_title = title:d_shujun
				}
				random_character_war = {
					limit = {
						hd_tb_is_war_trigger = { K = historical }
						primary_attacker = character:liu_bei
						primary_defender = scope:hd_lbxz_shu_target
					}
					set_variable = { name = hd_world_war_kind value = flag:lb_yizhou }
					save_scope_as = hd_lbxz_shu_war
				}
			}
			if = {
				limit = { any_character_war = { var:hd_world_war_kind ?= flag:lb_yizhou } }
				if = {
					limit = { NOT = { exists = global_var:hd_lbxz_shu_tries } }
					set_global_variable = { name = hd_lbxz_shu_tries value = 0 }
				}
				change_global_variable = { name = hd_lbxz_shu_tries add = 1 }
				set_global_variable = hd_lbxz_shu_war_started
				set_global_variable = { name = hd_lbxz_shu_lord value = scope:hd_lbxz_shu_target }
				scope:hd_lbxz_shu_target.primary_title ?= {
					set_global_variable = { name = hd_lbxz_shu_lord_title value = this }
				}
				# 兵力取益州之主最大兵力的一半，6000 至 30000；有地在都城，无地在营地所在县。
				if = {
					limit = { exists = capital_county }
					capital_county = { save_scope_as = hd_lbxz_shu_army_county }
				}
				else = {
					domicile.domicile_location.county ?= { save_scope_as = hd_lbxz_shu_army_county }
				}
				if = {
					limit = {
						exists = scope:hd_lbxz_shu_army_county
						exists = scope:hd_lbxz_shu_war
					}
					spawn_army = {
						levies = {
							value = scope:hd_lbxz_shu_target.max_military_strength
							multiply = 0.5
							min = 6000
							max = 30000
						}
						location = scope:hd_lbxz_shu_army_county.title_province
						origin = scope:hd_lbxz_shu_army_county.title_province
						inheritable = no
						uses_supply = yes
						war = scope:hd_lbxz_shu_war
						name = hd_lbxz_shu_army_name
					}
				}
				debug_log = "HD_LBXZ SHU_WAR_STARTED"
			}
			else = { debug_log = "HD_LBXZ SHU_WAR_START_FAILED_RETRY" }
		}
	}
	remove_global_variable = hd_lbxz_shu_pending
}

# 拿下成都后：益州之主出降。存的人已死或已不合条件时，改认其原主头衔的现任持有人。玩家益州之主可以拒绝。
hd_lbxz_shu_after_win_effect = {
	if = {
		limit = {
			NOT = { global_var:hd_lbxz_shu_lord ?= { hd_lbxz_shu_lord_valid_trigger = yes } }
			exists = global_var:hd_lbxz_shu_lord_title
			global_var:hd_lbxz_shu_lord_title.holder ?= { hd_lbxz_shu_lord_valid_trigger = yes }
		}
		set_global_variable = { name = hd_lbxz_shu_lord value = global_var:hd_lbxz_shu_lord_title.holder }
		debug_log = "HD_LBXZ SHU_LORD_REPLACED_BY_HEIR"
	}
	if = {
		limit = { global_var:hd_lbxz_shu_lord ?= { hd_lbxz_shu_lord_valid_trigger = yes } }
		if = {
			limit = { global_var:hd_lbxz_shu_lord = { is_ai = no } }
			set_global_variable = { name = hd_lbxz_shu_pending value = yes days = 60 }
			global_var:hd_lbxz_shu_lord = { trigger_event = hd_lbxz.0510 }
		}
		else = { hd_lbxz_shu_surrender_effect = yes }
	}
	else = {
		debug_log = "HD_LBXZ SHU_SURRENDER_SKIPPED"
		hd_lbxz_shu_finish_effect = yes
	}
}

# 节点 7：益州之主交出全部县级以上普通头衔与电脑直属封臣，玩家直属封臣先独立；转完不再是统治者才入刘备宫廷。
hd_lbxz_shu_surrender_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_lbxz_shu_done }
			character:liu_bei = { is_alive = yes is_ruler = yes }
			global_var:hd_lbxz_shu_lord ?= { hd_lbxz_shu_lord_valid_trigger = yes }
		}
		global_var:hd_lbxz_shu_lord = { save_scope_as = hd_lbxz_shu_lord_s }
		scope:hd_lbxz_shu_lord_s = {
			if = { limit = { is_imprisoned = yes } release_from_prison = yes }
			every_vassal = {
				limit = { is_alive = yes }
				if = {
					limit = { is_ai = yes }
					add_to_list = hd_lbxz_shu_ai_vassals
				}
				else = { add_to_list = hd_lbxz_shu_player_vassals }
			}
		}
		if = {
			limit = { any_in_list = { list = hd_lbxz_shu_player_vassals always = yes } }
			create_title_and_vassal_change = { type = independency save_scope_as = hd_lbxz_shu_player_release add_claim_on_loss = no }
			every_in_list = {
				list = hd_lbxz_shu_player_vassals
				becomes_independent = { change = scope:hd_lbxz_shu_player_release }
			}
			resolve_title_and_vassal_change = scope:hd_lbxz_shu_player_release
		}
		# 第一次事务：县级以上普通头衔（男爵领随县走）与电脑直属封臣。
		create_title_and_vassal_change = { type = conquest save_scope_as = hd_lbxz_shu_change add_claim_on_loss = no }
		every_in_list = {
			list = hd_lbxz_shu_ai_vassals
			change_liege = { liege = character:liu_bei change = scope:hd_lbxz_shu_change }
		}
		scope:hd_lbxz_shu_lord_s = {
			every_held_title = {
				limit = {
					tier >= tier_county
					is_noble_family_title = no
					is_landless_type_title = no
				}
				change_title_holder = {
					holder = character:liu_bei
					change = scope:hd_lbxz_shu_change
					take_baronies = yes
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd_lbxz_shu_change
		# 第二次事务：清扫残留的普通男爵领。
		if = {
			limit = {
				scope:hd_lbxz_shu_lord_s = {
					any_held_title = { is_noble_family_title = no is_landless_type_title = no }
				}
			}
			create_title_and_vassal_change = { type = conquest save_scope_as = hd_lbxz_shu_sweep_change add_claim_on_loss = no }
			scope:hd_lbxz_shu_lord_s = {
				every_held_title = {
					limit = { is_noble_family_title = no is_landless_type_title = no }
					change_title_holder = { holder = character:liu_bei change = scope:hd_lbxz_shu_sweep_change }
				}
			}
			resolve_title_and_vassal_change = scope:hd_lbxz_shu_sweep_change
		}
		if = {
			limit = { scope:hd_lbxz_shu_lord_s = { is_alive = yes is_ruler = no } }
			character:liu_bei = { add_courtier = scope:hd_lbxz_shu_lord_s }
			debug_log = "HD_LBXZ SHU_SURRENDERED"
		}
		else = { debug_log = "HD_LBXZ SHU_SURRENDER_LORD_STILL_RULER" }
	}
	hd_lbxz_shu_finish_effect = yes
	remove_global_variable = hd_lbxz_shu_pending
}

hd_lbxz_shu_surrender_refused_effect = {
	debug_log = "HD_LBXZ SHU_SURRENDER_REFUSED"
	hd_lbxz_shu_finish_effect = yes
	remove_global_variable = hd_lbxz_shu_pending
}

# 收尾：补政体，刘备本人持有成都才迁都，检查封臣层级，结束入蜀。
hd_lbxz_shu_finish_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd_lbxz_shu_done } }
		character:liu_bei = {
			hd_lbxz_shu_landed_fix_effect = yes
			if = {
				limit = { is_alive = yes has_title = title:c_uuii_p_4282 }
				set_realm_capital = title:c_uuii_p_4282
			}
			else_if = {
				limit = { title:c_uuii_p_4282.holder ?= { target_is_liege_or_above = character:liu_bei } }
				debug_log = "HD_LBXZ SHU_CAPITAL_HELD_BY_VASSAL"
			}
			if = {
				limit = {
					is_alive = yes
					any_vassal = { highest_held_title_tier >= character:liu_bei.highest_held_title_tier }
				}
				debug_log = "HD_LBXZ SHU_VASSAL_TIER_WARN"
			}
		}
		set_global_variable = hd_lbxz_shu_done
		debug_log = "HD_LBXZ SHU_DONE"
	}
}
