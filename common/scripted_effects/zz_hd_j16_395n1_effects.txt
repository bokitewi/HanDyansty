# 授郡（D16 / A1 §7.1 小幅转移）：$DUCHY$ 郡头衔与郡内 $FROM$ 直辖的县授给 $TO$（granted，不加宣称）。只在 $FROM$ 是 AI、$TO$ 在世且不是 $FROM$ 时执行；
# 郡头衔由 $FROM$ 持有或无人持有才转。授后 $TO$ 的直接上级不是 $LIEGE$ 时改投 $LIEGE$（$LIEGE$ 与 $TO$ 同一人时不动）。$TAG$：日志节点号。
# $TYPE$：头衔变更类型——主君授出用 granted；封臣交还主君（$TO$ 即 $LIEGE$）用 revoked。Scope：任意
hd_j16_n1_grant_duchy_effect = {
	if = {
		limit = {
			$FROM$ = {
				is_alive = yes
				is_ai = yes
			}
			$TO$ = { is_alive = yes }
			NOT = { $FROM$ = { this = $TO$ } }
		}
		create_title_and_vassal_change = { type = $TYPE$ save_scope_as = hd_j16_n1_gd_change add_claim_on_loss = no }
		$DUCHY$ = {
			every_de_jure_county = {
				limit = { holder ?= $FROM$ }
				change_title_holder = {
					holder = $TO$
					change = scope:hd_j16_n1_gd_change
					take_baronies = yes
				}
			}
		}
		if = {
			limit = {
				OR = {
					$DUCHY$.holder ?= $FROM$
					NOT = { exists = $DUCHY$.holder }
				}
			}
			$DUCHY$ = {
				change_title_holder = {
					holder = $TO$
					change = scope:hd_j16_n1_gd_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd_j16_n1_gd_change
		$TO$ = {
			if = {
				limit = {
					is_landed = yes
					NOT = { this = $LIEGE$ }
					NOT = { liege ?= $LIEGE$ }
				}
				hd_nbc_swear_fealty_effect = { LIEGE = $LIEGE$ }
			}
			if = {
				limit = { has_title = $DUCHY$ }
				debug_log = "HD_J16 $TAG$ FIRE grant_duchy"
			}
			else = { debug_log = "HD_J16 $TAG$ WAIT grant_duchy_partial" }
		}
	}
	else = { debug_log = "HD_J16 $TAG$ SKIP grant_duchy_not_ai" }
}

# 称藩：$VASSAL$（在世、有地、独立）改投 $LIEGE$，须 $LIEGE$ 的最高头衔等级高于 $VASSAL$（清单：封臣化要求被收编者头衔低于新主君）。Scope：任意
hd_j16_n1_vassalize_effect = {

				is_alive = yes
				is_landed = yes

			NOT = { $VASSAL$ = { this = $LIEGE$ } }

				AND = {
					$LIEGE$ = { highest_held_title_tier >= tier_empire }
					$VASSAL$ = { highest_held_title_tier <= tier_kingdom }
				}

					$LIEGE$ = { highest_held_title_tier >= tier_kingdom }
					$VASSAL$ = { highest_held_title_tier <= tier_duchy }

					$LIEGE$ = { highest_held_title_tier >= tier_duchy }

				}
			}
		}

# 同一政权内换主（不杀人；照 hd_nbc_seize_realm_effect 写法、删去末尾的 death）：$NEW$ 有地且非独立时先独立；$OLD$ 的县至帝国级领地头衔
#   （不含世族、朝官头衔）逐个以 usurped 转给 $NEW$，再令 $OLD$ 的原封臣改投 $NEW$。
#   转头衔前先把 $OLD$ 的封臣存入 list（转完后 $OLD$ 已无地，every_vassal 未必取得到）；转完头衔再对 list 改投，$NEW$ 原为无地廷臣时此时已有地、可受投。
#   list 只收「仍独立或仍以 $OLD$ 为主君、尚未归 $NEW$」者，同一效果链内多次调用时前一次的 list 成员不会被误转。Scope：任意
hd_j16_n1_take_realm_effect = {
	hd_j16_n1_clear_effect = { S = hd_j16_n1_tr_new }
	hd_j16_n1_clear_effect = { S = hd_j16_n1_tr_old }
	$NEW$ = {
		save_scope_as = hd_j16_n1_tr_new
		if = {
			limit = {
				is_landed = yes
				is_independent_ruler = no
			}
			hd_nbc_become_independent_effect = yes
		}
	}
	$OLD$ = {
		save_scope_as = hd_j16_n1_tr_old
		every_vassal = {
			limit = { NOT = { this = scope:hd_j16_n1_tr_new } }
			add_to_list = hd_j16_n1_tr_vassals
		}
		every_held_title = {
			limit = {
				is_noble_family_title = no
				tgp_is_minister_title = no
				tier >= tier_county
				tier <= tier_empire
			}
			save_scope_as = hd_j16_n1_tr_title
			create_title_and_vassal_change = { type = usurped save_scope_as = hd_j16_n1_tr_change add_claim_on_loss = no }
			scope:hd_j16_n1_tr_title = {
				change_title_holder = {
					holder = scope:hd_j16_n1_tr_new
					change = scope:hd_j16_n1_tr_change
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:hd_j16_n1_tr_change
		}
	}
	if = {
		limit = { scope:hd_j16_n1_tr_new = { is_alive = yes is_landed = yes } }
		every_in_list = {
			list = hd_j16_n1_tr_vassals
			limit = {
				is_alive = yes
				is_landed = yes
				NOT = { this = scope:hd_j16_n1_tr_new }
				NOT = { liege ?= scope:hd_j16_n1_tr_new }
				OR = {
					is_independent_ruler = yes
					liege ?= scope:hd_j16_n1_tr_old
				}
			}
			hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_j16_n1_tr_new }
		}
	}
}

# 换主一步（同一政权内）：现任 $CUR$ 就是 $NEW$ → 记 seated；$CUR$ 不是燕系之主或拓跋氏 → 等待；$NEW$ 不在 $CUR$ 一系 → 等待；$CUR$ 是 AI → 转头衔、记 seated；$CUR$ 是玩家 → 弹 hd_j16.5590「逊位 / 拒之」（60 天内不重弹）。
# 结果变量 hd_j16_$TAG$_seated（换主完成）、hd_j16_$TAG$_refused（玩家拒之，由 5590 b 设）。$LOG$：日志节点号。Scope：任意
hd_j16_n1_seat_effect = {
	if = {
		limit = { $CUR$ = { this = $NEW$ } }
		set_global_variable = hd_j16_$TAG$_seated
		debug_log = "HD_J16 $LOG$ FIRE already_ruler"
	}
	# 现任须是燕系之主或拓跋氏（防止龙城落入北魏、高句丽等外人时，把外国整个转给史实之人；hd_j16_n1_seat_cur_ok_trigger）
	else_if = {
		limit = { NOT = { $CUR$ = { hd_j16_n1_seat_cur_ok_trigger = yes } } }
		if = {
			limit = { NOT = { has_global_variable = hd_j16_$TAG$_cwlog } }
			set_global_variable = hd_j16_$TAG$_cwlog
			debug_log = "HD_J16 $LOG$ WAIT seat_cur_foreign"
		}
	}
	# 史实之人须在现任一系之内（防止按位置解析到外人政权时误转）
	else_if = {
		limit = { NOT = { $NEW$ = { hd_j16_n1_in_realm_trigger = { R = $CUR$ } } } }
		if = {
			limit = { NOT = { has_global_variable = hd_j16_$TAG$_swlog } }
			set_global_variable = hd_j16_$TAG$_swlog
			debug_log = "HD_J16 $LOG$ WAIT seat_not_in_realm"
		}
	}
	else_if = {
		limit = { $CUR$ = { is_ai = yes } }
		hd_j16_n1_take_realm_effect = { NEW = $NEW$ OLD = $CUR$ }
		set_global_variable = hd_j16_$TAG$_seated
		debug_log = "HD_J16 $LOG$ FIRE seated"
	}
	else_if = {
		limit = {
			NOT = { has_global_variable = hd_j16_$TAG$_abd_asked }
			NOT = { has_global_variable = hd_j16_$TAG$_refused }
		}
		set_global_variable = { name = hd_j16_$TAG$_abd_asked value = yes days = 60 }
		$CUR$ = {
			set_variable = { name = hd_j16_n1_abd_tag value = flag:$TAG$ }
			set_variable = { name = hd_j16_n1_abd_new value = $NEW$ }
			trigger_event = hd_j16.5590
		}
		debug_log = "HD_J16 $LOG$ WAIT player_asked"
	}
}

	if = {
		limit = {
			NOT = { global_var:hd_j16_chaibi_att ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:chaibi } } }
			NOT = { global_var:hd_j16_chaibi_def ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:chaibi } } }
		}
		set_global_variable = hd_j16_chaibi_done
		if = {
			limit = { has_global_variable = hd_nbc_chaibi_active }
			remove_global_variable = hd_nbc_chaibi_active
		}
		hd_j16_n1_get_wei_ruler_effect = { AS = hd_j16_e5t50_wei }
		if = {
			limit = {
				global_var:hd_j16_border_chaibi_target.holder ?= {
					OR = {
						this = global_var:hd_j16_chaibi_att
						top_liege ?= global_var:hd_j16_chaibi_att
						# 攻方战中身故、由继承人接战：按位置重新解析出的本方之主（不认同王朝的第三方）

							exists = scope:hd_j16_e5t50_wei
							OR = {
								this = scope:hd_j16_e5t50_wei
								top_liege ?= scope:hd_j16_e5t50_wei
							}
						}
					}
				}
			}
			set_global_variable = hd_j16_chaibi_wei_won
			hd_j16_n1_clear_effect = { S = hd_j16_e5t50_di }
			hd_j16_resolve_effect = { ID = tianshui_di_001 AS = hd_j16_e5t50_di }

		}
		else = {
			set_global_variable = hd_j16_chaibi_lost
			debug_log = "HD_J16 e5_t50 FIRE result_lost"
		}
	}
}
