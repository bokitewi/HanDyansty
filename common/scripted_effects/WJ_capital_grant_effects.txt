# AI interaction and player roster use the same guarded native title transaction.
WJ_capital_grant_effect = {
 if = {
  limit = { WJ_capital_grant_selected_valid = yes }
  if = {
   limit = { scope:WJ_capital_grant_selected_title = { NOT = { exists = holder } } }
   scope:actor = { add_gold = -50 add_prestige = -50 }
  }
  create_title_and_vassal_change = { type = granted save_scope_as = WJ_capital_title_change add_claim_on_loss = no }
  scope:recipient = { change_liege = { liege = scope:actor change = scope:WJ_capital_title_change } }
  scope:WJ_capital_grant_selected_title = { change_title_holder = { holder = scope:recipient change = scope:WJ_capital_title_change } }
  resolve_title_and_vassal_change = scope:WJ_capital_title_change
  if = {
   limit = { scope:WJ_capital_grant_selected_title = scope:actor.capital_province.kingdom }
   scope:recipient = { WJ_sili_grant_capital_holder_effect = yes }
  }
  scope:actor = {
   WJ_baiguan_refresh_display_effect = yes
   send_interface_toast = {
    type = event_toast_effect_good title = WJ_capital_grant_success
    desc = WJ_capital_grant_success_desc left_icon = scope:recipient
   }
  }
 }
 else = {
  scope:actor = { send_interface_toast = { type = event_toast_effect_bad title = WJ_capital_grant_invalid } }
 }
}

# Called only when the player opens the list; reuse the roster's local pools.
WJ_capital_collect_candidate = {
 save_scope_as = recipient
 if = {
  limit = { WJ_capital_grant_selected_valid = yes }
  scope:actor = { add_to_variable_list = { name = wj_capital_candidates target = scope:recipient } }
 }
}
