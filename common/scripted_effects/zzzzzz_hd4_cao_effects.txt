hd4_cao_tick_effect = {
	if = {
		limit = {
			NOT = {
				has_character_flag = hd_cao_early_story_completed
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = hd4_cao_stage
				}
				OR = {
					has_character_flag = hd4_cao_early_requested
					has_character_flag = hd_cao_early_story_pending
					has_character_flag = hd_cao_early_story_started
					has_variable = hd_cao_early_story_phase
				}
			}
			set_variable = {
				name = hd4_cao_stage
				value = 1
			}
			if = {
				limit = {
					has_character_flag = hd_cao_callous_after_escape
				}
				set_variable = {
					name = hd4_cao_stage
					value = 7
				}
			}
			else_if = {
				limit = {
					has_variable = hd_cao_early_story_phase
					var:hd_cao_early_story_phase = flag:hd_cao_phase_assassination
				}
				set_variable = {
					name = hd4_cao_stage
					value = 4
				}
			}
			else_if = {
				limit = {
					has_variable = hd_cao_early_story_phase
					var:hd_cao_early_story_phase = flag:hd_cao_phase_sword
				}
				set_variable = {
					name = hd4_cao_stage
					value = 3
				}
			}
			else_if = {
				limit = {
					has_variable = hd_cao_early_story_phase
					var:hd_cao_early_story_phase = flag:hd_cao_phase_prepare
				}
				set_variable = {
					name = hd4_cao_stage
					value = 2
				}
			}
		}
		if = {
			limit = {
				has_variable = hd4_cao_stage
				NOT = {
					has_character_flag = hd4_cao_stage_wait
				}
			}
			hd4_cao_dispatch_stage_effect = yes
		}
	}
	if = {
		limit = {
			has_character_flag = hd_cao_early_story_completed
		}
		hd4_cao_companions_effect = yes
	}
	if = {
		limit = {
			has_character_flag = hd_cao_expansion_menu_pending
			NOT = {
				has_character_flag = hd4_cao_menu_lease
			}
		}
		add_character_flag = {
			flag = hd4_cao_menu_lease
			days = 180
		}
		trigger_event = hd_cao_expansion.0001
	}
}
hd4_cao_settle_laiwu_effect = {
	if = {
		limit = {
			this = character:cao_cao
			is_alive = yes
			is_imprisoned = no
			NOT = {
				has_character_flag = hd4_cao_laiwu_settled
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd4_cao_laiwu_change
			add_claim_on_loss = no
		}
		if = {
			limit = {
				NOT = {
					has_title = title:c_uuii_p_20
				}
			}
			title:c_uuii_p_20 = {
				change_title_holder = {
					holder = character:cao_cao
					change = scope:hd4_cao_laiwu_change
					take_baronies = yes
				}
			}
		}
		if = {
			limit = {
				NOT = {
					has_title = title:d_taishan_qingzhou
				}
			}
			title:d_taishan_qingzhou = {
				change_title_holder_include_vassals = {
					holder = character:cao_cao
					change = scope:hd4_cao_laiwu_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd4_cao_laiwu_change
		hd_force_independent_celestial_effect = yes
		if = {
			limit = {
				has_title = title:c_uuii_p_20
				has_title = title:d_taishan_qingzhou
				is_independent_ruler = yes
			}
			set_realm_capital = title:c_uuii_p_20
			set_location = {
				location = title:c_uuii_p_20.title_province
				stick_to_location = yes
			}
			domicile ?= {
				move_domicile = title:c_uuii_p_20.title_province
			}
			if = {
				limit = {
					capital_county = title:c_uuii_p_20
				}
				add_character_flag = hd4_cao_laiwu_settled
				add_character_flag = hd_cao_chenliu_migration_completed
			}
		}
	}
}

			}
		}

			random_character_artifact = {
				limit = {
					has_variable = hd_qixing_baodao
				}
				save_scope_as = hd4_cao_sword
			}
		}
		if = {
			limit = {
				NOT = {
					exists = scope:hd4_cao_sword
				}
			}
			random_character_artifact = {
				limit = {
					has_variable = hd_qixing_baodao
				}
				save_scope_as = hd4_cao_sword
			}
		}
		if = {
			limit = {
				NOT = {
					exists = scope:hd4_cao_sword
				}
				NOT = {
					has_character_flag = hd_cao_qixing_baodao_obtained
				}
			}
			add_character_flag = hd_cao_qixing_baodao_obtained
			create_artifact = {
				name = hd_qixing_baodao_name
				description = hd_qixing_baodao_description

				template = hd_legendary_weapon_template
				visuals = hd_qixing_baodao
				modifier = hd_qixing_baodao_weapon_modifier
				decaying = no
				save_scope_as = hd4_cao_sword
			}
		}
		if = {
			limit = {
				exists = scope:hd4_cao_sword
			}
			scope:hd4_cao_sword = {
				set_variable = hd_qixing_baodao
				set_owner = character:cao_cao
				equip_artifact_to_owner_replace = yes
			}
		}
		add_character_flag = hd4_cao_sword_processed
		add_character_flag = hd_cao_qixing_baodao_obtained
	}
}
