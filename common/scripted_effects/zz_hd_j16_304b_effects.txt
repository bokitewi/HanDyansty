# R-继统一步。调用方已判「本步未了结、已到起日、未过窗口」。$RULER$/$HEIR$ 为人物 ID；$KEY$ 本步语义名；$LOG$ 节点号；$HEV$ yes/no（是否点面板）。
# 了结：君主已死（SKIP actor_dead）、君主是玩家（SKIP player_ruler）、继承人已死（SKIP heir_dead）、已是储君或立储成功（FIRE，设 _result）。
# 其余（君主无地、继承人未出场、不合法）等待，下一跳再查。变量 hd_j16_$KEY$_done/_result/_skip/_wlog
hd_j16_304b_rjt_effect = {
	if = {
		limit = { hd_j16_dead_trigger = { ID = $RULER$ } }
		hd_j16_304_skip_effect = { KEY = $KEY$ LOG = $LOG$ WHY = actor_dead }
	}
	else = {
		hd_j16_304_res_effect = { ID = $RULER$ AS = hd_j16_rjt_r }
		hd_j16_304_res_effect = { ID = $HEIR$ AS = hd_j16_rjt_h }
		if = {
			limit = { scope:hd_j16_rjt_r ?= { is_ai = no } }
			hd_j16_304_skip_effect = { KEY = $KEY$ LOG = $LOG$ WHY = player_ruler }
		}
		else_if = {
			limit = { hd_j16_dead_trigger = { ID = $HEIR$ } }
			hd_j16_304_skip_effect = { KEY = $KEY$ LOG = $LOG$ WHY = heir_dead }
		}
		else_if = {
			limit = {
				exists = scope:hd_j16_rjt_h
				scope:hd_j16_rjt_r ?= {
					is_landed = yes
					designated_heir ?= scope:hd_j16_rjt_h
				}
			}
			set_global_variable = hd_j16_$KEY$_done
			set_global_variable = hd_j16_$KEY$_result
			if = {
				limit = { always = $HEV$ }
				set_global_variable = hd_hev_j16_$KEY$
			}
			debug_log = "HD_J16 $LOG$ FIRE $KEY$_already"
		}
		else_if = {
			limit = {
				exists = scope:hd_j16_rjt_h
				scope:hd_j16_rjt_r ?= {
					is_landed = yes
					NOT = { this = scope:hd_j16_rjt_h }
					OR = {
						dynasty ?= scope:hd_j16_rjt_h.dynasty
						scope:hd_j16_rjt_h.father ?= this
					}
					is_valid_designated_heir = scope:hd_j16_rjt_h
				}
			}
			scope:hd_j16_rjt_r = { set_designated_heir = scope:hd_j16_rjt_h }
			set_global_variable = hd_j16_$KEY$_done
			set_global_variable = hd_j16_$KEY$_result
			if = {
				limit = { always = $HEV$ }
				set_global_variable = hd_hev_j16_$KEY$
			}
			debug_log = "HD_J16 $LOG$ FIRE $KEY$_heir_set"
		}
		else_if = {
			# 人都到位（有地、同王朝或亲子）而引擎不认可为合法储君：单独记一次，便于实测区分（11 对非直系储君的风险，见 p3-B2 报告）
			limit = {
				exists = scope:hd_j16_rjt_h
				scope:hd_j16_rjt_r ?= {
					is_landed = yes
					NOT = { this = scope:hd_j16_rjt_h }
					OR = {
						dynasty ?= scope:hd_j16_rjt_h.dynasty
						scope:hd_j16_rjt_h.father ?= this
					}
					NOT = { is_valid_designated_heir = scope:hd_j16_rjt_h }
				}
			}
			hd_j16_304_wait_effect = { KEY = $KEY$_ih LOG = $LOG$ WHY = invalid_heir_$KEY$ }
		}
		else = {
			hd_j16_304_wait_effect = { KEY = $KEY$ LOG = $LOG$ WHY = not_ready }
		}
	}
}

# 夺位（不杀）。Root：夺位者。$OLD$（人物 scope）的领地头衔（县到帝国，不含世族、朝官头衔）尽归 root，主头衔随之；$OLD$ 留作无地之人。
# 写法照 hd_nbc_seize_realm_effect（zz_hd_nbc_history_effects.txt），去掉末尾的 death——旧主的死由卒年表执行（李班、李期、纥那各有卒年行；涉复辰待补 K 行）。
# 顺序改为：先转头衔 → 再设主头衔 → 最后收编旧主剩下的封臣，且只收最高头衔低于 root 的（报错清单：封臣化要求被收编者最高头衔低于新主君）。
hd_j16_304b_seize_effect = {
	save_scope_as = hd_j16_sz_new
	if = {
		limit = { exists = scope:hd_j16_sz_old_pt }
		clear_saved_scope = hd_j16_sz_old_pt
	}
	$OLD$.primary_title ?= { save_scope_as = hd_j16_sz_old_pt }
	hd_nbc_become_independent_effect = yes
	$OLD$ = {
		every_held_title = {
			limit = {
				is_noble_family_title = no
				tgp_is_minister_title = no
				tier >= tier_county
				tier <= tier_empire
			}
			save_scope_as = hd_j16_sz_title
			create_title_and_vassal_change = { type = usurped save_scope_as = hd_j16_sz_change add_claim_on_loss = no }
			scope:hd_j16_sz_title = {
				change_title_holder = {
					holder = scope:hd_j16_sz_new
					change = scope:hd_j16_sz_change
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:hd_j16_sz_change
		}
	}
	if = {
		limit = {
			exists = scope:hd_j16_sz_old_pt
			has_title = scope:hd_j16_sz_old_pt
		}
		set_primary_title_to = scope:hd_j16_sz_old_pt
	}
	$OLD$ = {
		every_vassal = {
			limit = {
				NOT = { this = scope:hd_j16_sz_new }
				highest_held_title_tier < scope:hd_j16_sz_new.highest_held_title_tier
			}
			hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_j16_sz_new }
		}
	}
}

	}

	if = {
		limit = { NOT = { has_global_variable = hd_j16_zongshe_done } }

			limit = { global_var:hd_j16_start_year >= 346 }
			set_global_variable = hd_j16_zongshe_done
			debug_log = "HD_J16 e1_t10 SKIP prestart"
		}
		else_if = {
			limit = { current_date >= 305.8.1 }
			if = {
				limit = { current_date >= 309.1.1 }
				if = {
					limit = { has_global_variable = hd_j16_zongshe_result }
					set_global_variable = hd_j16_zongshe_done
					debug_log = "HD_J16 e1_t10 FIRE closed"
				}
				else = {
					hd_j16_304_skip_effect = { KEY = zongshe LOG = e1_t10 WHY = window_closed }
				}
			}
			else_if = {
				limit = { hd_j16_dead_trigger = { ID = tuoba_027 } }
				if = {
					limit = { has_global_variable = hd_j16_zongshe_result }
					set_global_variable = hd_j16_zongshe_done

				}

				}
			}
			else = {
				hd_j16_304_res_effect = { ID = tuoba_027 AS = hd_j16_zsh_yilu }
				if = {
					limit = {
						hd_j16_dead_trigger = { ID = tuoba_026 }
						NOT = { hd_j16_alive_trigger = { ID = tuoba_luguan } }
						scope:hd_j16_zsh_yilu ?= {
							is_landed = yes
							is_imprisoned = no
						}
					}
					# 还有他族人持三郡之一
					if = {
						limit = {
							OR = {
								hd_j16_304b_zs_target_trigger = { T = d_dingxiang }
								hd_j16_304b_zs_target_trigger = { T = d_yunzhong }
								hd_j16_304b_zs_target_trigger = { T = d_tuoba }
							}
						}
						if = {
							limit = {
								OR = {
									scope:hd_j16_zsh_yilu = { is_ai = yes }
									has_global_variable = hd_j16_zongshe_go
								}
							}
							hd_j16_zongshe_step_effect = yes
						}
						else_if = {
							limit = { NOT = { has_global_variable = hd_j16_zongshe_asked } }
							set_global_variable = hd_j16_zongshe_asked
							scope:hd_j16_zsh_yilu = { trigger_event = hd_j16.0550 }
							debug_log = "HD_J16 e1_t10 WAIT player_asked"
						}
					}

						limit = { scope:hd_j16_zsh_yilu = { hd_j16_304b_tuoba_duchy_trigger = yes } }
						set_global_variable = hd_j16_zongshe_done
						if = {
							limit = { NOT = { has_global_variable = hd_j16_zongshe_result } }
							set_global_variable = hd_j16_zongshe_result
							set_global_variable = hd_hev_j16_zongshe
						}
						debug_log = "HD_J16 e1_t10 FIRE united"
					}
					else = {
						hd_j16_304_skip_effect = { KEY = zongshe LOG = e1_t10 WHY = no_target }
					}
				}
				else = {
					hd_j16_304_wait_effect = { KEY = zongshe LOG = e1_t10 WHY = not_ready }
				}
			}
		}
	}
