# $ATTACKER$ 以 hd253_history_war_cb 对 $DEFENDER$ 宣战，主目标 $TARGET$。战争存为 scope:hd253_war_$KIND$
hd253_start_history_war_effect = {
	$ATTACKER$ = {
		start_war = {
			cb = hd253_history_war_cb
			target = $DEFENDER$
			target_title = $TARGET$
		}
		random_character_war = {
			limit = {
				using_cb = hd253_history_war_cb
				primary_attacker = $ATTACKER$
				primary_defender = $DEFENDER$
				NOT = { has_variable = hd253_war_kind }
			}
			set_variable = { name = hd253_war_kind value = flag:$KIND$ }
			save_scope_as = hd253_war_$KIND$
		}
	}
	if = {
		limit = { exists = scope:hd253_war_$KIND$ }
		set_global_variable = hd253_$KIND$_active
		debug_log = "HD253 WAR_CONFIRMED $KIND$"
	}
	else = { debug_log = "HD253 WAR_START_FAILED $KIND$" }
}

# 战争结局。Root：战争
hd253_war_result_effect = {
	hd253_war_result_kind_effect = { KIND = xincheng RESULT = $RESULT$ }
	hd253_war_result_kind_effect = { KIND = fashu RESULT = $RESULT$ }
}
hd253_war_result_kind_effect = {
	if = {
		limit = { var:hd253_war_kind ?= flag:$KIND$ }
		set_global_variable = hd253_$KIND$_$RESULT$
		if = {
			limit = { has_global_variable = hd253_$KIND$_active }
			remove_global_variable = hd253_$KIND$_active
		}
		debug_log = "HD253 WAR_RESULT $KIND$ $RESULT$"
	}
}

# 218 局接力（2026-10-18）：各项过了最晚日期就永久跳过，记一次 HD253 RELAY_LATE。$KEY$ 同 hd253_$KEY$_done
############################################################
hd253_relay_late_effect = {
	if = {
		limit = {
			hd253_relay_trigger = yes
			current_date >= $DATE$
			NOT = { has_global_variable = hd253_$KEY$_done }
			NOT = { has_global_variable = hd253_$KEY$_late }
		}
		set_global_variable = hd253_$KEY$_late
		debug_log = "HD253 RELAY_LATE $KEY$"
	}
}

		limit = {
			NOT = { has_global_variable = hd253_zhuge_ke_done }

			NOT = { has_global_variable = hd253_xincheng_active }

			global_var:hd_sg_sun_jun ?= { is_alive = yes is_imprisoned = no }

			limit = {
				hd253_relay_trigger = yes
				character:zhu_ge_ke = { any_vassal = { is_ai = no } }

			debug_log = "HD253 RELAY SKIP ZHUGE_KE PLAYER_VASSAL"
		}

				tier >= tier_county # 218 局接力：只转县级以上
			}
		}

			limit = { hd253_relay_trigger = yes }
			change_title_holder = { holder = scope:hd253_wu_emperor change = scope:hd253_ke_change take_baronies = yes }
		}
		else = { change_title_holder = { holder = scope:hd253_wu_emperor change = scope:hd253_ke_change } }
	}
	every_vassal = {
		limit = {
			OR = {
				NOT = { hd253_relay_trigger = yes }
				is_ai = yes # 218 局接力：玩家封臣不改投
			}
		}
		change_liege = { liege = scope:hd253_wu_emperor change = scope:hd253_ke_change }
	}
	if = {
		limit = {
			hd253_relay_trigger = yes
			any_vassal = { is_ai = no }
		}
		debug_log = "HD253 RELAY ZHUGE_KE PLAYER_VASSAL_KEPT"
	}
	resolve_title_and_vassal_change = scope:hd253_ke_change
	global_var:hd_sg_sun_jun ?= { save_scope_as = hd253_sun_jun }
	death = { death_reason = death_murder_known killer = scope:hd253_sun_jun }

		if = {
			limit = {
				has_diarchy_type = grand_secretariat
				global_var:hd_sg_sun_jun ?= { is_alive = yes is_diarch_valid_trigger = yes }
			}
			set_diarch = global_var:hd_sg_sun_jun
		}
		if = {
			limit = { diarch ?= global_var:hd_sg_sun_jun }
			debug_log = "HD253 ZHUGE_KE KILLED SUN_JUN_DIARCH OK"
		}
		else = { debug_log = "HD253 ZHUGE_KE KILLED SUN_JUN_DIARCH BAD" }
	}
	debug_log = "HD253 ZHUGE_KE KILLED"
}

# 4 司马师废曹芳立曹髦（史实 254 年九月）：司马师是权臣、曹芳在位。事件给司马师：电脑行废立，玩家可罢

	hd253_relay_late_effect = { KEY = feifang DATE = 258.1.1 }
	if = {
		limit = {
			NOT = { has_global_variable = hd253_feifang_done }
			NOT = { has_global_variable = hd253_feifang_late }
			current_date >= 254.10.1
			global_var:hd_sg_cao_fang ?= { is_alive = yes has_title = title:h_china }
			global_var:hd_sg_cao_mao ?= {
				is_alive = yes
				# 10-18 加固 E：有封地也可，只要仍在魏境内（废立时先收回再继位）
				OR = {
					is_landed = no
					top_liege ?= global_var:hd_sg_cao_fang
				}
			}
			character:si_ma_shi ?= {
				is_alive = yes
				title:h_china.var:hd_court_regent ?= this
			}
			# 218 局接力：司马师仍是天子封臣；曹髦不从玩家或外国宫中拉来

				NOT = { hd253_relay_trigger = yes }

					global_var:hd_sg_cao_mao ?= { hd253_relay_pullable_trigger = yes }
				}
			}
		}
		set_global_variable = hd253_feifang_done
		if = {
			limit = {
				character:si_ma_shi = { is_ai = yes }
				global_var:hd_sg_cao_fang ?= { is_ai = no }
			}
			global_var:hd_sg_cao_fang ?= { trigger_event = hd253.0305 }
		}
		else = {
			character:si_ma_shi = { trigger_event = hd253.0300 }
		}
	}
}

# 废立：曹芳的头衔与封臣尽归曹髦，曹芳为齐王（无地，作新帝廷臣）；司马师重新挟新帝。Root：司马师
hd253_feifang_effect = {
	global_var:hd_sg_cao_fang ?= { save_scope_as = hd253_old_emp }
	global_var:hd_sg_cao_mao ?= { save_scope_as = hd253_new_emp }
	hd253_new_emp_prepare_effect = yes
	scope:hd253_old_emp = {
		if = {
			limit = { has_active_diarchy = yes }
			end_diarchy = yes
		}
	}
	create_title_and_vassal_change = { type = usurped save_scope_as = hd253_feifang_change add_claim_on_loss = no }
	scope:hd253_old_emp = {
		every_held_title = {
			# 只转县级以上；无地类头衔中只转天下霸权 h_china（它本身是 landless 头衔）
			limit = {
				tier >= tier_county
				is_noble_family_title = no
				OR = {
					is_landless_type_title = no
					this = title:h_china
				}
			}
			change_title_holder = { holder = scope:hd253_new_emp change = scope:hd253_feifang_change take_baronies = yes }
		}
		every_vassal = {
			limit = { NOT = { this = scope:hd253_new_emp } }
			change_liege = { liege = scope:hd253_new_emp change = scope:hd253_feifang_change }
		}
	}
	resolve_title_and_vassal_change = scope:hd253_feifang_change
	scope:hd253_new_emp = {
		if = {
			limit = { has_title = title:h_china }
			set_primary_title_to = title:h_china
			if = {
				limit = { NOT = { has_government = celestial_government } }
				change_government = celestial_government
			}
		}
	}
	scope:hd253_old_emp = {
		if = {
			limit = { is_landed = no NOT = { is_courtier_of = scope:hd253_new_emp } }
			set_employer = scope:hd253_new_emp
		}
	}
	# 司马师重新挟新帝（权臣变量在头衔上，傀儡与首辅二元统治要重设）
	hd_court_install_regency_effect = yes
	scope:hd253_new_emp = {
		if = {
			limit = { has_diarchy_type = grand_secretariat diarch ?= character:si_ma_shi }
			set_diarchy_swing = 100
		}
	}
	scope:hd253_new_emp = {
		if = {
			limit = { title:c_luoyang.holder ?= this }
			set_realm_capital = title:c_luoyang
		}
	}
	if = {
		limit = {
			scope:hd253_new_emp = {
				has_title = title:h_china
				diarch ?= character:si_ma_shi
			}
		}
		debug_log = "HD253 FEIFANG OK"
	}
	else_if = {
		limit = { scope:hd253_new_emp = { has_title = title:h_china } }
		debug_log = "HD253 FEIFANG OK_BUT_NO_DIARCH"
	}
	else = { debug_log = "HD253 FEIFANG BAD" }
}

# 10-18 加固 D：起兵没开出战争时记一次失败，下次心跳再问；失败满 3 次放弃。$KEY$ = guanqiu / zhuge_dan
hd253_rebel_fail_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd253_$KEY$_fails } }
		set_global_variable = { name = hd253_$KEY$_fails value = 0 }
	}
	change_global_variable = { name = hd253_$KEY$_fails add = 1 }
	if = {
		limit = { has_global_variable = hd253_$KEY$_pending }
		remove_global_variable = hd253_$KEY$_pending
	}
	if = {
		limit = { global_var:hd253_$KEY$_fails >= 3 }
		set_global_variable = hd253_$KEY$_done
		debug_log = "HD253 REBEL_GAVE_UP $KEY$"
	}
	else = { debug_log = "HD253 REBEL_RETRY $KEY$" }
}

# 换天子：$OLD$ 的头衔与封臣尽归 $NEW$，$OLD$ 留在新帝宫中；当前 scope（权臣，即 scope:hd253_regent）重新挟新帝（写法同 hd253_feifang_effect）

		}
		every_vassal = {
			limit = { NOT = { this = scope:hd253_new_emp } }
			change_liege = { liege = scope:hd253_new_emp change = scope:hd253_swap_change }
		}
	}
	resolve_title_and_vassal_change = scope:hd253_swap_change
	scope:hd253_new_emp = {

				set_realm_capital = title:c_luoyang
			}
		}
	}
	scope:hd253_old_emp = {
		if = {
			limit = { is_alive = yes is_landed = no NOT = { is_courtier_of = scope:hd253_new_emp } }
			set_employer = scope:hd253_new_emp
		}
	}
	hd_court_install_regency_effect = yes
	scope:hd253_new_emp = {
		if = {
			limit = { has_diarchy_type = grand_secretariat diarch ?= scope:hd253_regent }
			set_diarchy_swing = 100
		}
	}

		limit = { scope:hd253_new_emp = { has_title = title:h_china } }
		debug_log = "HD253 SWAP_EMPEROR OK"
	}

}
