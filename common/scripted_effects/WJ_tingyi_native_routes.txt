# UI operator is ROOT. Saved native scopes are supplied by the actual window.
# WJ_ty_* effects only prepare drafts; the native automation override is below.
WJ_ty_native_court_appoint = {
 # Called on the actual interaction actor. Discard any inherited council route.
 clear_saved_scope = wj_ty_native_task
 $OWNER$ = { save_scope_as = wj_ty_native_owner }
 $TYPE$ = { save_scope_as = wj_ty_native_court_type }
 $CANDIDATE$ = { save_scope_as = wj_ty_native_candidate }
 WJ_ty_native_appoint_effect = yes
}
WJ_ty_native_select_office = {
 if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:WJ_minister_cheng_xiang } set_variable = { name = wj_ty_office value = 1 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:councillor_chancellor } set_variable = { name = wj_ty_office value = 2 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:councillor_spymaster } set_variable = { name = wj_ty_office value = 3 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:councillor_marshal } set_variable = { name = wj_ty_office value = 4 } }
 
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:minister_grand_marshal } set_variable = { name = wj_ty_office value = 6 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:minister_personnel } set_variable = { name = wj_ty_office value = 10 } }
 
 
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:councillor_steward } set_variable = { name = wj_ty_office value = 13 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:minister_justice } set_variable = { name = wj_ty_office value = 14 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:councillor_court_chaplain } set_variable = { name = wj_ty_office value = 15 } }
 else_if = { limit = { exists = scope:wj_ty_native_task scope:wj_ty_native_task = scope:wj_ty_native_owner.council_task:minister_works } set_variable = { name = wj_ty_office value = 16 } }
 else_if = { limit = { exists = scope:wj_ty_native_court_type scope:wj_ty_native_court_type = court_position_type:WJ_dasima_court_position } set_variable = { name = wj_ty_office value = 7 } }
 else_if = { limit = { exists = scope:wj_ty_native_court_type scope:wj_ty_native_court_type = court_position_type:favored_minister_court_position } set_variable = { name = wj_ty_office value = 8 } }
 else_if = { limit = { exists = scope:wj_ty_native_court_type scope:wj_ty_native_court_type = court_position_type:chief_eunuch_court_position } set_variable = { name = wj_ty_office value = 9 } }
}
WJ_ty_native_appoint_effect = {
 if = { limit = { WJ_ty_native_appoint_valid = yes }
  WJ_ty_open = yes
  remove_variable = wj_ty_dismiss_kind
  WJ_ty_native_select_office = yes
  set_variable = { name = wj_ty_topic value = 3 }
  remove_variable = wj_ty_target
  save_scope_as = wj_ty_operator
  if = { limit = { exists = scope:wj_ty_native_candidate scope:wj_ty_native_candidate = { WJ_ty_appointment_candidate = { OPERATOR = scope:wj_ty_operator } } }
   set_variable = { name = wj_ty_target value = scope:wj_ty_native_candidate }
  }
  WJ_ty_appointment_snapshot = yes
 }
}
WJ_ty_native_dismiss_effect = {
if = { limit = { WJ_ty_native_dismiss_valid = yes }
  WJ_ty_open = yes
  WJ_ty_native_select_office = yes
  save_scope_as = wj_ty_operator
  remove_variable = wj_ty_target
  clear_variable_list = wj_ty_selected_court_positions
  set_variable = { name = wj_ty_topic value = 4 }
  set_variable = { name = wj_ty_dismiss_kind value = 1 }
  set_variable = { name = wj_ty_role value = 2 }
  set_variable = { name = wj_ty_action value = 5 }
  if = { limit = { exists = scope:wj_ty_native_task.councillor }
   set_variable = { name = wj_ty_target value = scope:wj_ty_native_task.councillor }
   set_variable = { name = wj_ty_removal_mode value = 1 }
  }
  else = {
   set_variable = { name = wj_ty_removal_mode value = 2 }
   if = { limit = { exists = scope:wj_ty_native_incumbent }
    set_variable = { name = wj_ty_target value = scope:wj_ty_native_incumbent }
   }
   else_if = { limit = { scope:wj_ty_native_court_type = court_position_type:WJ_dasima_court_position }
    scope:wj_ty_native_owner = { random_court_position_holder = { type = WJ_dasima_court_position scope:wj_ty_operator = { set_variable = { name = wj_ty_target value = prev } } } }
   }
   else_if = { limit = { scope:wj_ty_native_court_type = court_position_type:favored_minister_court_position }
    scope:wj_ty_native_owner = { random_court_position_holder = { type = favored_minister_court_position scope:wj_ty_operator = { set_variable = { name = wj_ty_target value = prev } } } }
   }
   else_if = { limit = { scope:wj_ty_native_court_type = court_position_type:chief_eunuch_court_position }
    scope:wj_ty_native_owner = { random_court_position_holder = { type = chief_eunuch_court_position scope:wj_ty_operator = { set_variable = { name = wj_ty_target value = prev } } } }
   }
   add_to_variable_list = { name = wj_ty_selected_court_positions target = scope:wj_ty_native_court_type }
  }
  WJ_ty_removal_snapshot = yes
 }
}

# Native player automation only. Does not wrap historical/event appointments.
court_position_vacated_event_effect = {
	scope:liege = {
		if = {
			limit = {
				is_ai = no
				# Player automation must not bypass deliberation for these Han offices.
				NOT = { WJ_ty_native_court_slot = { OWNER = this TYPE = court_position_type:$COURT_POSITION_TYPE$ } }
				NOT = { has_player_court_position_automation_none = court_position_type:$COURT_POSITION_TYPE$ }
			}

			court_position_type:$COURT_POSITION_TYPE$ = {
				save_scope_as = court_position_type
			}
			ordered_court_position_candidate = {
				court_position_type = scope:court_position_type
				order_by = aptitude_score:$COURT_POSITION_TYPE$

				# Previous holder might still be in the court/alive - lets not include them
				limit = { NOT = { this = scope:employee } }
				max = 2
				check_range_bounds = no

				if = {
					limit = { NOT = { exists = scope:candidate_1 } }
					save_scope_as = candidate_1
				}
				else_if = {
					limit = { NOT = { exists = scope:candidate_2 } }
					save_scope_as = candidate_2
				}
			}

			if = {
				limit = {
					OR = {
						has_player_court_position_automation_assign_best = scope:court_position_type
						has_player_court_position_automation_assign_best_or_event = scope:court_position_type
					}
					exists = scope:candidate_1
				}

				appoint_court_position = {
					recipient = scope:candidate_1
					court_position = scope:court_position_type
				}
			}
			else_if = {
				limit = {
					has_player_court_position_automation_assign_best = scope:court_position_type
				}
				# Do nothing, the player does not want an event
			}
			else_if = {
				limit = { is_landless_adventurer = yes }
				trigger_event = court_position_management_event.0002
			}
			else = {
				trigger_event = court_position_management_event.0001
			}
		}
	}
}

# Native chooser final confirmations are redirected here. Selection never appoints.
WJ_ty_native_court_draft_effect = {
 clear_saved_scope = wj_ty_native_task
 if = { limit = { WJ_ty_native_appoint_valid = yes exists = scope:candidate
  scope:candidate = { WJ_ty_native_court_candidate_valid = yes }
  trigger_if = { limit = { exists = scope:wj_ty_native_incumbent } scope:wj_ty_native_incumbent = { WJ_ty_court_position_employed = { TYPE = scope:wj_ty_native_court_type OWNER = scope:wj_ty_native_owner } } }
 }
  if = { limit = { NOT = { WJ_ty_session = yes } } WJ_ty_open = yes }
  WJ_ty_appointment_clear = yes WJ_ty_removal_clear = yes
  remove_variable = wj_ty_target
  WJ_ty_native_select_office = yes
  set_variable = { name = wj_ty_topic value = 3 }
  set_variable = { name = wj_ty_appointment_mode value = 1 }
  save_scope_as = wj_ty_operator
  if = { limit = { scope:candidate = { WJ_ty_appointment_candidate = { OPERATOR = scope:wj_ty_operator } } }
   set_variable = { name = wj_ty_target value = scope:candidate }
   WJ_ty_appointment_snapshot = yes
  }
 }
