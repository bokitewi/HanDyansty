# Name confirmation can enter once; renaming never changes identity or clocks.
WJ_yangwang_confirm_name_effect = {
 if = {
  limit = { NOT = { has_trait = WJ_yangwang_trait } WJ_yangwang_eligible_trigger = yes }
  add_trait = WJ_yangwang_trait
  if = {
   limit = { appointment_timeout_days = 0 }
   set_appointment_timeout = { years = 150 desc = appointment_timeout_desc_refusing }
   add_character_flag = wj_yangwang_timeout_owned
  }
  else = {
   # Preserve the existing source; the callback follows any extended expiry.
   trigger_event = { id = WJ_yangwang_timeout.0001 days = WJ_yangwang_timeout_handoff_days }
  }
 }
}
WJ_yangwang_exit_effect = {
 remove_trait = WJ_yangwang_trait
 if = {
  limit = { has_character_flag = wj_yangwang_timeout_owned }
  if = { limit = { appointment_timeout_days > 0 } clear_appointment_timeout = yes }
  remove_character_flag = wj_yangwang_timeout_owned
 }
}
# Parent: replace maocai's unconditional timeout-clear with a NOT has_trait guard.
# This hook is for the end of grant-maocai, and repairs this module's own ban only.
WJ_yangwang_after_maocai_effect = {
 if = {
  limit = { has_trait = WJ_yangwang_trait appointment_timeout_days = 0 }
  set_appointment_timeout = { years = 150 desc = appointment_timeout_desc_refusing }
  add_character_flag = wj_yangwang_timeout_owned
 }
}

# Periodic invitation carries no optional bribes. Base cost settles once on valid acceptance.
WJ_yangwang_mufu_accept_effect = {
 # Same real parent-chain check as public attach, before any charge.
 WJ_mufu_check_chain = yes
 if = {
  limit = { scope:wj_chain_safe = 1 WJ_mufu_request_valid = yes WJ_mufu_selected_empty = yes }
  WJ_mufu_attach = yes
  if = {
   limit = { has_variable = wj_mufu_owner var:wj_mufu_owner = scope:actor }
   scope:actor = { change_influence = { value = WJ_mufu_base_influence_cost multiply = -1 } }
  }
 }
}
WJ_yangwang_mufu_refuse_effect = {
 add_to_variable_list = { name = wj_mufu_rejected_owners target = scope:actor years = 3 }
}
