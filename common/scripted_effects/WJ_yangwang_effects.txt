# 雅号确认后进入隐居；改雅号时（已是隐士）不重复进入。
WJ_yangwang_confirm_name_effect = {
 if = {
  limit = {
   NOT = { has_trait = WJ_yangwang_trait }
   WJ_yinshi_can_enter_shown_trigger = yes
   WJ_yinshi_can_enter_valid_trigger = yes
  }
  WJ_yinshi_enter_effect = yes
 }
}

# 唯一出山出口：出山决议、接受邀请、接受授官、入幕（WJ_mufu_enter）都走这里。
WJ_yangwang_exit_effect = {
 if = {
  limit = { has_trait = WJ_yangwang_trait }
  remove_trait = WJ_yangwang_trait
  if = {
   limit = { has_character_flag = wj_yangwang_timeout_owned }
   if = { limit = { appointment_timeout_days > 0 } clear_appointment_timeout = yes }
   remove_character_flag = wj_yangwang_timeout_owned
  }
  set_knight_status = default
  set_variable = { name = wj_yinshi_cooldown_end value = WJ_yinshi_cooldown_end_value years = 15 }
  every_owned_story = {
   limit = { story_type = story_WJ_yinshi }
   end_story = yes
  }
 }
}
