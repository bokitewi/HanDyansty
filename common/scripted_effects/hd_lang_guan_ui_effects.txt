hd_lang_guan_add_candidate_effect = {
	save_scope_as = hd_lang_guan_candidate

	if = {
		limit = {
			has_character_flag = flag_lang_guan
		}
		root = {
			add_to_variable_list = {
				name = hd_lang_guan_characters
				target = scope:hd_lang_guan_candidate
			}
		}
	}
}

hd_lang_guan_rebuild_characters_effect = {
	clear_variable_list = hd_lang_guan_characters

	hd_lang_guan_add_candidate_effect = yes
	every_courtier_or_guest = {
		hd_lang_guan_add_candidate_effect = yes
	}
	every_vassal_or_below = {
		hd_lang_guan_add_candidate_effect = yes
	}
}
