# 253 局接力 263 后半段的效果（三国剧本事件衔接，2026-10-18）。由 common/on_action/zz_hd_263_relay.txt 与 zz_hd_263_history_effects.txt 调用。

# 启动接力：关掉 253 自带的出降（hd253_check_fashu_result_effect 只在没有 hd253_fashu_surrender_done 时出降），点起 263 心跳
hd263_relay_activate_effect = {
	set_global_variable = hd263_relay_on
	set_global_variable = { name = hd263_s1_pulses value = 0 }
	if = {
		limit = { NOT = { exists = global_var:hd253_fashu_surrender_done } }
		set_global_variable = hd253_fashu_surrender_done
	}
	title:h_china = { trigger_event = { on_action = hd263_pulse days = 1 } }
	debug_log = "HD263 RELAY ON"
	# 局面记录（首测定位用，不作闸门）
	if = {
		limit = { character:si_ma_zhao ?= { is_alive = yes title:h_china.var:hd_court_regent ?= this } }
		debug_log = "HD263 RELAY STATE ZHAO_REGENT YES"
	}
	else = { debug_log = "HD263 RELAY STATE ZHAO_REGENT NO" }
	if = {
		limit = { title:h_china.holder ?= { global_var:hd_sg_cao_huan ?= this } }
		debug_log = "HD263 RELAY STATE CAO_HUAN_EMPEROR YES"
	}
	else = { debug_log = "HD263 RELAY STATE CAO_HUAN_EMPEROR NO" }
	if = {
		limit = {
			title:e_bashu.holder ?= { is_alive = yes is_independent_ruler = yes }
			hd253_relay_shu_ok_trigger = yes # 同伐蜀的闸门
		}
		debug_log = "HD263 RELAY STATE SHU_INDEPENDENT YES"
	}
	else = { debug_log = "HD263 RELAY STATE SHU_INDEPENDENT NO" }
}

# $WHO$ 在魏（宫中或封臣）、无实地时，把 $DUCHY$ 连同原持有者在郡内直持的县授给他
hd263_relay_try_duchy_effect = {
	global_var:hd_sg_$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_imprisoned = no
				NOT = { hd263_relay_has_real_land_trigger = yes }
				OR = {
					AND = {
						is_ruler = no
						employer ?= { is_ai = yes OR = { this = title:h_china.holder top_liege ?= title:h_china.holder } } # 东家是玩家则不授
					}
					AND = { is_ruler = yes top_liege ?= title:h_china.holder }
				}
				title:$DUCHY$ = { hd263_relay_grantable_duchy_trigger = { WHO = $WHO$ } }
			}
			save_scope_as = hd263_relay_gen
			title:$DUCHY$.holder = { save_scope_as = hd263_relay_old }
			create_title_and_vassal_change = { type = granted save_scope_as = hd263_relay_change add_claim_on_loss = no }
			title:$DUCHY$ = {
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_county holder ?= scope:hd263_relay_old }
					change_title_holder = { holder = scope:hd263_relay_gen change = scope:hd263_relay_change take_baronies = yes }
				}
				change_title_holder = { holder = scope:hd263_relay_gen change = scope:hd263_relay_change }
			}
			resolve_title_and_vassal_change = scope:hd263_relay_change
			debug_log = "HD263 RELAY GRANT $WHO$ $DUCHY$"
		}
	}
}

# 授地后：电脑 $WHO$ 有实地、在魏境内但不直属天子，且现主君是电脑时，改投天子；最后回读
hd263_relay_fix_liege_effect = {
	global_var:hd_sg_$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_ai = yes
				hd263_relay_has_real_land_trigger = yes
				is_independent_ruler = no
				NOT = { liege ?= title:h_china.holder }
				top_liege ?= title:h_china.holder
				liege ?= { is_ai = yes }
			}
			create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd263_relay_fealty add_claim_on_loss = no }
			change_liege = { liege = title:h_china.holder change = scope:hd263_relay_fealty }
			resolve_title_and_vassal_change = scope:hd263_relay_fealty
		}
		if = {
			limit = {
				is_alive = yes
				hd263_relay_has_real_land_trigger = yes
				liege ?= title:h_china.holder
			}
			debug_log = "HD263 RELAY LAND $WHO$ OK"
		}
		else = { debug_log = "HD263 RELAY LAND $WHO$ BAD" }
	}
}

# S2 出降前：$WHO$ 在蜀汉治下（scope:hd263_shu 本人、其治下统治者，或雇主是其中之一）就记 hd263_relay_shu（218 局里出场生成器造的人未必在蜀）
hd263_relay_mark_shu_effect = {
	global_var:hd_sg_$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				OR = {
					this = scope:hd263_shu
					top_liege ?= scope:hd263_shu
					employer ?= {
						OR = {
							this = scope:hd263_shu
							top_liege ?= scope:hd263_shu
						}
					}
				}
			}
			add_character_flag = { flag = hd263_relay_shu days = 1 }
		}
	}
}

# S4 兵变前：$WHO$ 在魏境内，并且（CORE = yes 的钟会、姜维、邓艾）或者在钟会帐下（主君或雇主是 scope:hd263_zh），就记 hd263_relay_s4_kill
hd263_relay_mark_s4_effect = {
	global_var:hd_sg_$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				hd253_relay_in_wei_trigger = yes
				OR = {
					always = $CORE$
					liege ?= scope:hd263_zh
					employer ?= scope:hd263_zh
				}
			}
			add_character_flag = { flag = hd263_relay_s4_kill days = 1 }
		}
	}
}
