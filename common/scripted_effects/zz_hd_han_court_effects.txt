# One after-lobby pass, not periodic grants. Reuse native house-head/cadet handling.
hd_184_ensure_noble_families_effect = {
	if = {
		limit = { NOT = { has_game_rule = never_noble_family_title } }
		every_ruler = {
			limit = {
				is_alive = yes
				government_has_flag = government_is_celestial
				government_allows = noble_families
				highest_held_title_tier >= tier_county
				WJ_name_actual_emperor_trigger = no
				exists = house.house_head
				exists = liege
				liege = { government_allows = noble_families }
				NOR = {
					any_held_title = { is_noble_family_title = yes }
					house = { any_house_member = { any_held_title = { is_noble_family_title = yes } } }
				}
			}
			create_noble_family_effect = { GOVERNMENT_GIVER = this }
			domicile ?= { set_up_domicile_estate_effect = yes }
		}
	}
}
