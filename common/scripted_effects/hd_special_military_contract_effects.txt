# Contract changes call this immediately; title/government changes and pulses repair lifecycle state.
# Reuse vanilla refusal. Preserve an existing long-term refusal; a new contract supersedes
# a short appointment timeout so that expiry cannot create a gap before the next pulse.
hd_special_military_contract_reconcile_effect = {
	if = {
		limit = { hd_special_military_contract_active_trigger = yes }
		if = {
			limit = { appointment_timeout_days < 36500 }
			set_appointment_timeout = { years = 150 desc = appointment_timeout_desc_refusing }
			add_character_flag = hd_special_contract_timeout_owned
		}
		if = {
			limit = { hd_military_colony_contract_trigger = yes }
			if = {
				limit = { NOT = { has_realm_law = hd_military_colony_succession_law } }
				add_realm_law_skip_effects = hd_military_colony_succession_law
			}
		}
		else_if = {
			limit = {
				vassal_contract_has_flag = hd_tuntian
				NOT = { has_realm_law = celestial_military_appointment_succession_law }
			}
			add_realm_law_skip_effects = celestial_military_appointment_succession_law
		}
	}
	else = {
		if = {
			limit = { has_character_flag = hd_special_contract_timeout_owned }
			if = {
				# Do not clear a later short timeout or a refusal now owned by another system.
				limit = {
					appointment_timeout_days >= 36500
					NOR = {
						has_character_flag = wj_appointment_timeout_owned
						has_character_flag = wj_yangwang_timeout_owned
					}
				}
				clear_appointment_timeout = yes
			}
			remove_character_flag = hd_special_contract_timeout_owned
		}
		if = {
			limit = {
				has_realm_law = hd_military_colony_succession_law
				is_landed = yes
				is_independent_ruler = no
				government_has_flag = government_is_celestial
				NOT = { hd_warlord_trigger = yes }
			}
			if = {
				limit = { vassal_contract_has_flag = celestial_military_appointment }
				add_realm_law_skip_effects = celestial_military_appointment_succession_law
			}
			else = { add_realm_law_skip_effects = celestial_appointment_succession_law }
		}
	}
}
