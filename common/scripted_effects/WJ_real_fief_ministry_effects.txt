# One office on one inherited fief; no periodic/world scan and no new office identity.
WJ_rf_capture_ministry = {
 clear_saved_scope = wj_rf_inherited_office
 clear_saved_scope = wj_rf_inherited_fief
 if = { limit = { primary_title = { tgp_is_minister_title = yes } }
  primary_title = { save_scope_as = wj_rf_inherited_office }
 }
 else = { random_held_title = { limit = { tgp_is_minister_title = yes } save_scope_as = wj_rf_inherited_office } }
 if = { limit = { exists = scope:wj_rf_inherited_office top_liege = { WJ_rf_emperor = yes } }
  if = { limit = { any_held_title = { has_variable = wj_rf_rank var:wj_rf_rank = 2 } }
   random_held_title = { limit = { has_variable = wj_rf_rank var:wj_rf_rank = 2 } save_scope_as = wj_rf_inherited_fief }
  }
  else = { random_held_title = { limit = { has_variable = wj_rf_rank var:wj_rf_rank = 1 } save_scope_as = wj_rf_inherited_fief } }
  scope:wj_rf_inherited_fief = {
   set_variable = { name = wj_rf_inherited_office value = scope:wj_rf_inherited_office }
   set_variable = { name = wj_rf_inherited_court value = root.top_liege.primary_title }
  }
 }
}
WJ_rf_queue_ministry = {
 if = { limit = { exists = scope:title.var:wj_rf_inherited_office exists = scope:title.var:wj_rf_inherited_court }
  scope:title = {
   save_scope_as = wj_rf_inherited_fief
   var:wj_rf_inherited_office = { save_scope_as = wj_rf_inherited_office }
   var:wj_rf_inherited_court = { save_scope_as = wj_rf_inherited_court }
   remove_variable = wj_rf_inherited_office
   remove_variable = wj_rf_inherited_court
  }
  trigger_event = { id = WJ_rf_ministry.1 days = 30 }
 }
}
