# 京城县存入 scope:hd_court_capital。Scope：任意
hd_court_save_capital_effect = {
	if = {
		limit = { title:h_china.var:hd_court_capital_county ?= { tier = tier_county } }
		title:h_china.var:hd_court_capital_county = { save_scope_as = hd_court_capital }
	}
	else_if = {
		limit = { title:h_china.holder.capital_county ?= { tier = tier_county } }
		title:h_china.holder.capital_county = { save_scope_as = hd_court_capital }
	}
	else = {
		title:h_china.title_capital_county = { save_scope_as = hd_court_capital }
	}
}

}

# 罢去权臣的摄政与朝官头衔。Scope：权臣
# （原版 tgp_is_minister_title 已被 UUII 桩化，故按头衔形态判定朝官头衔。）
hd_court_strip_office_effect = {
	save_scope_as = hd_court_stripped
	title:h_china.holder ?= {
		if = {
			limit = {
				has_active_diarchy = yes
				diarch ?= scope:hd_court_stripped
			}
			end_diarchy = yes
		}
		if = {
			limit = { scope:hd_court_stripped = { is_councillor_of = prev } }
			always_fire_councillor = scope:hd_court_stripped
		}
	}
	every_held_title = {
		limit = {
			tier = tier_empire
			is_landless_type_title = yes
			is_noble_family_title = no
		}
		scope:hd_court_stripped = { destroy_title = prev }
	}
}

# 解除挟持：清除权臣记录与权臣施加的傀儡（别人施加的傀儡，如"实封"的世俗傀儡，不动）。Scope：任意
hd_court_release_effect = {
	title:h_china.holder ?= {
		if = {
			limit = { puppeteer ?= title:h_china.var:hd_court_regent }
			clear_puppet_status = yes
		}
	}
	title:h_china = {
		remove_variable = hd_court_regent
		remove_variable = hd_court_regent_title
	}
}

# 立权臣（不动领地）：记录、丞相 + grand_secretariat 摄政（D22）、
# AI 天子成为原版傀儡（D2），玩家天子不加傀儡、由秉政者判定锁权（D5）。
# Scope：权臣（须已是天子的封臣）
hd_court_install_regency_effect = {
	save_scope_as = hd_court_regent_scope
	title:h_china.holder = { save_scope_as = hd_court_puppet_target }
	title:h_china = {
		set_variable = { name = hd_court_regent value = scope:hd_court_regent_scope }
		set_variable = { name = hd_court_regent_title value = scope:hd_court_regent_scope.primary_title }
	}
	if = {
		limit = {
			NAND = {
				has_title = title:e_WJ_minister_cheng_xiang
				scope:hd_court_puppet_target = {
					has_diarchy_type = grand_secretariat
					diarch ?= scope:hd_court_regent_scope
				}
			}
		}
		# 郡国制开关为 no：先 granted 授丞相，免 appointment 辞头衔（2026-10-16 起对所有开局生效，184 董卓/后续权臣曾因此丢地）
		hd218_grant_chengxiang_effect = yes
		WJ_install_historical_secretariat_effect = { OFFICE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang LAW = WJ_grand_chengxiang_law }
	}
	if = {
		limit = {
			scope:hd_court_puppet_target = {
				is_ai = yes
				NOT = { puppeteer ?= scope:hd_court_regent_scope }
			}
		}
		scope:hd_court_puppet_target = {
			if = {
				limit = { exists = puppeteer }
				clear_puppet_status = yes
			}
		}
		# 2026-10-12 合并：用原版世俗统治者傀儡；同一操纵者只能有一个，天子优先
		puppet:external_ruler_puppet ?= {
			clear_puppet_status = yes
			remove_puppet_modifiers_effect = yes
		}
		set_puppet = {
			target = scope:hd_court_puppet_target
			type = external_ruler_puppet
		}
	}
}

# 挟天子成立（含夺天子）。Scope：新权臣（须满足 hd_court_regent_eligible_trigger）
# D23：已有旧权臣时，京城县与汉帝国头衔归新权臣，旧权臣罢官后带其余领地独立。
# D1b：天子直辖地、汉帝国头衔、非朝官封臣转归权臣；天子 0 县，霸权无地、定都京城。
# 写法沿用曹操奉迎（zz_hd_cao_fengying_effects.txt）：先确立帝国级君臣关系，再转直辖地与封臣。
hd_court_takeover_effect = {
	save_scope_as = hd_court_new_regent
	title:h_china.holder = { save_scope_as = hd_court_emperor }
	hd_court_save_capital_effect = yes

	# --- D23 旧权臣 ---
	if = {
		limit = {
			hd_court_has_regent_trigger = yes
			NOT = { title:h_china.var:hd_court_regent = scope:hd_court_new_regent }
		}
		title:h_china.var:hd_court_regent = { save_scope_as = hd_court_old_regent }
	}
	hd_court_release_effect = yes
	if = {
		limit = { exists = scope:hd_court_old_regent }
		scope:hd_court_old_regent = { hd_court_strip_office_effect = yes }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = hd_court_old_change
			add_claim_on_loss = yes
		}
		if = {
			limit = {
				scope:hd_court_capital.holder ?= {
					OR = {
						this = scope:hd_court_old_regent
						is_vassal_or_below_of = scope:hd_court_old_regent
					}
				}
			}
			scope:hd_court_capital = {
				change_title_holder = {
					holder = scope:hd_court_new_regent
					change = scope:hd_court_old_change
					take_baronies = yes
				}
			}
		}
		if = {
			limit = { title:e_han.holder ?= scope:hd_court_old_regent }
			title:e_han = {
				change_title_holder = {
					holder = scope:hd_court_new_regent
					change = scope:hd_court_old_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd_court_old_change
		if = {
			limit = { scope:hd_court_old_regent = { liege ?= scope:hd_court_emperor } }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = hd_court_old_indep_change
				add_claim_on_loss = no
			}
			scope:hd_court_old_regent = {
				becomes_independent = { change = scope:hd_court_old_indep_change }
			}
			resolve_title_and_vassal_change = scope:hd_court_old_indep_change
		}
	}

	# --- D1b 快照：天子的非朝官封臣与直辖实地头衔 ---
	scope:hd_court_emperor = {
		every_vassal = {
			limit = {
				NOT = { this = scope:hd_court_new_regent }
				hd_court_is_court_official_trigger = no
			}
			add_to_list = hd_court_take_vassals
		}
		every_held_title = {
			limit = {
				tier >= tier_barony
				tier < tier_hegemony
				is_landless_type_title = no
				is_noble_family_title = no
				NOT = { this = title:e_han }
			}
			add_to_list = hd_court_take_titles
		}
	}

	# --- 第一笔：汉帝国头衔与君臣关系 ---
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_court_empire_change
		add_claim_on_loss = no
	}
	if = {
		limit = {
			title:e_han.holder ?= scope:hd_court_emperor
			# 218：e_han 被抽空法理时不交给新权臣，留在天子手里（184 里 e_han 有司隶诸县，恒真）
			title:e_han = { any_de_jure_county = { count >= 1 } }
		}
		title:e_han = {
			change_title_holder = {
				holder = scope:hd_court_new_regent
				change = scope:hd_court_empire_change
			}
		}
	}
	if = {
		limit = { scope:hd_court_new_regent = { NOT = { liege ?= scope:hd_court_emperor } } }
		scope:hd_court_new_regent = {
			change_liege = {
				liege = scope:hd_court_emperor
				change = scope:hd_court_empire_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:hd_court_empire_change

	# 霸权无地并定都京城（须在天子交出最后一县之前）
	title:h_china = {
		set_landless_title = yes
		set_capital_county = scope:hd_court_capital
		set_variable = { name = hd_court_capital_county value = scope:hd_court_capital }
	}

	# --- 第二笔：直辖地与封臣 ---
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_court_domain_change
		add_claim_on_loss = no
	}
	every_in_list = {
		list = hd_court_take_titles
		limit = { holder = scope:hd_court_emperor }
		change_title_holder = {
			holder = scope:hd_court_new_regent
			change = scope:hd_court_domain_change
			take_baronies = yes
		}
	}
	every_in_list = {
		list = hd_court_take_vassals
		limit = {
			is_alive = yes
			is_ruler = yes
			highest_held_title_tier < tier_empire
			# 封臣层级须低于新权臣，否则留在天子名下
			highest_held_title_tier < scope:hd_court_new_regent.highest_held_title_tier
			liege ?= scope:hd_court_emperor
		}
		change_liege = {
			liege = scope:hd_court_new_regent
			change = scope:hd_court_domain_change
		}
	}
	resolve_title_and_vassal_change = scope:hd_court_domain_change

	scope:hd_court_new_regent = {
		if = {
			limit = { has_title = title:e_han }
			set_primary_title_to = title:e_han
		}
		hd_court_install_regency_effect = yes
		add_character_flag = { flag = hd_court_regent_tenure years = 3 }
	}

	# 天子移驻京城（同曹操奉迎）
	scope:hd_court_emperor = {
		domicile ?= { move_domicile = scope:hd_court_capital.title_province }
		set_location = {
			location = scope:hd_court_capital.title_province
			stick_to_location = yes
		}
	}

	# 通知关注天朝循环的玩家
	every_player = {
		limit = { tgp_does_this_player_care_about_the_dynastic_cycle = yes }
		trigger_event = hd_court.0001
	}
}

# D18/D19 维护：权臣死亡由继承人接续；新天子继续被挟持；失效则解除。
# 斗争、对峙期每年一次，以及权臣死亡、天子更替后由 hd_court.0003 调用。Scope：任意
hd_court_repair_effect = {
	if = {
		limit = {
			exists = title:h_china.holder
			exists = title:h_china.var:hd_court_regent
		}
		title:h_china.var:hd_court_regent = { save_scope_as = hd_court_cur_regent }
		if = {
			# 权臣仍有效：确保摄政与傀儡还在（D19 新天子即位后重建）
			limit = { hd_court_has_regent_trigger = yes }
			if = {
				limit = {
					OR = {
						title:h_china.holder = {
							NOT = {
								has_diarchy_type = grand_secretariat
								diarch ?= scope:hd_court_cur_regent
							}
						}
						title:h_china.holder = {
							is_ai = yes
							NOT = { puppeteer ?= scope:hd_court_cur_regent }
						}
					}
				}
				scope:hd_court_cur_regent = { hd_court_install_regency_effect = yes }
			}
		}
		else_if = {
			# D18：权臣已死，由其主头衔的新持有者接续（剧本摄政在奉迎剧本未了结前交给剧本处理）
			limit = {
				scope:hd_court_cur_regent = { is_alive = no }
				title:h_china.var:hd_court_regent_title.holder ?= {
					hd_court_regent_eligible_trigger = yes
					liege ?= title:h_china.holder
				}
				NAND = {
					title:h_china.holder.var:hd_dong_fixed_regent ?= scope:hd_court_cur_regent
					NOT = { exists = global_var:hd_fengying_world_resolved }
				}
				# 第 4 期：死于宫廷政变的权臣不由继承人接续
				NOT = { exists = title:h_china.var:hd_court_coup_done }
				# 两晋十六国（j16，3-A，02 §1.1）：八王时段内权臣死只 release，下一任由 B 节点或「入朝辅政」决定
				NOT = { exists = global_var:hd_j16_bawang_active }
				# 两晋十六国（j16，3-A，04 §3.2 第 4 步）：桓温卒的阻断钩子（zz_hd_j16_deaths_special_effects.txt）设了本变量＝已死权臣时不交接
				NOT = { title:h_china.var:hd_j16_handoff_block ?= scope:hd_court_cur_regent }
			}
			title:h_china.var:hd_court_regent_title.holder = {
				save_scope_as = hd_court_heir_regent
			}
			hd_court_release_effect = yes
			scope:hd_court_heir_regent = { hd_court_install_regency_effect = yes }
		}
		else_if = {
			# 两晋十六国（j16，3-A）：交接被阻断——天子获释，阻断变量用过即清（04 §3.2 第 5 步 E4-J40 的删除因此成为冗余兜底）
			limit = { title:h_china.var:hd_j16_handoff_block ?= scope:hd_court_cur_regent }
			hd_court_release_effect = yes
			title:h_china = { remove_variable = hd_j16_handoff_block }
			debug_log = "HD_J16 e4_j30 FIRE handoff_blocked"
		}
		else = {
			hd_court_release_effect = yes
		}
	}
}

# D24 攻下京城选择挟天子（10-29）：先以攻方胜利结束战争，再立为权臣。Root：占领者；scope:hd_court_occ_war
# 胜利结算后复核：天子仍在、不是占领者的封臣、占领者仍独立，才接管（否则会成环或无天子可挟）
hd_court_occ_seize_effect = {
	scope:hd_court_occ_war ?= { end_war = attacker }
	if = {
		limit = {
			exists = title:h_china.holder
			NOT = { title:h_china.holder = root }
			title:h_china.holder = { NOT = { is_vassal_or_below_of = root } }
			is_independent_ruler = yes
			hd_court_regent_eligible_trigger = yes
		}
		# 两晋十六国（j16，3-A，02 §1.4）：八王时段内攻下京城只换权臣（占领者入朝），不转天子直辖地
		if = {
			limit = { exists = global_var:hd_j16_bawang_active }
			hd_j16_swap_regent_effect = yes
			debug_log = "HD_J16 bawang FIRE d24_swap_regent"
		}
		else = {
			hd_court_takeover_effect = yes
		}
	}
	else = {
		debug_log = "HD_COURT d24 seize SKIPPED after victory"
	}
}
