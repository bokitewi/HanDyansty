# 2026-11-02 部落县交本地首领：县治是部落地产、持有人却是天朝制/封建制的县，开局生成一个本地首领接手，政体改部落制，作原持有人的封臣。
# 持有人的治所县不动；游牧、牧民政体持有的不动。调度见 common/on_action/zzzz_hd_tribal_county_chiefs.txt。
# 人物照 253 边地首领（hd253_frontier_chief_effect）：原版 post_migration_minor_warlord_template，文化/信仰/礼制取该县。

hd_tcc_all_counties_effect = {
	every_county = {
		limit = {
			title_province = { has_holding_type = tribal_holding }
			exists = holder
			holder = {
				is_landed = yes
				OR = {
					has_government = celestial_government
					has_government = feudal_government
				}
			}
			NOT = { this = holder.capital_county }
		}
		hd_tcc_make_chief_effect = yes
	}
	debug_log = "HD_TCC done"   # 每生成一个首领记一行 HD_TCC CHIEF，数 game.log 即可
}

# Scope：县
hd_tcc_make_chief_effect = {
	save_temporary_scope_as = seceding_county   # 模板里取 scope:seceding_county 的文化/信仰/礼制
	holder = { save_temporary_scope_as = hd_tcc_old }
	create_character = {
		location = title_province
		template = post_migration_minor_warlord_template
		employer = scope:hd_tcc_old
		dynasty = generate
		save_temporary_scope_as = hd_tcc_new
	}
	create_title_and_vassal_change = { type = granted save_scope_as = hd_tcc_change add_claim_on_loss = no }
	change_title_holder = { holder = scope:hd_tcc_new change = scope:hd_tcc_change take_baronies = yes }
	resolve_title_and_vassal_change = scope:hd_tcc_change
	scope:hd_tcc_new = {
		if = {
			limit = { NOT = { liege ?= scope:hd_tcc_old } }
			create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd_tcc_fix_change add_claim_on_loss = no }
			change_liege = { liege = scope:hd_tcc_old change = scope:hd_tcc_fix_change }
			resolve_title_and_vassal_change = scope:hd_tcc_fix_change
		}
		if = {
			limit = { NOT = { has_government = tribal_government } }
			change_government = tribal_government
		}
	}
	debug_log = "HD_TCC CHIEF"
}
