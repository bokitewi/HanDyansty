# 任意作用域。全局年度计数 +1（跨年自动清零，配 hd_yiguan_gt_counter_below_trigger）
hd_yiguan_gt_bump_counter_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = hd_yiguan_gt_$NAME$_year }
				NOT = { has_global_variable = hd_yiguan_gt_$NAME$_count }
				NOT = { global_var:hd_yiguan_gt_$NAME$_year = current_year }
			}
		}
		set_global_variable = { name = hd_yiguan_gt_$NAME$_year value = current_year }
		set_global_variable = { name = hd_yiguan_gt_$NAME$_count value = 0 }
	}
	change_global_variable = { name = hd_yiguan_gt_$NAME$_count add = 1 }
}

# 角色。退出世族：退还庄园建筑、国库入私囊、销毁世族头衔（先例 mod dm_government_reform_effects.txt:213-252）
hd_yiguan_gt_leave_noble_family_effect = {
	if = {
		limit = { any_held_title = { is_noble_family_title = yes } }
		refund_noble_family_domicile_buildings_effect = yes
		if = {
			limit = { has_treasury = yes }
			add_gold = treasury
			remove_treasury = { value = treasury }
		}
		every_held_title = {
			limit = { is_noble_family_title = yes }
			# 爵位（WJ peerage）挂在世族头衔上：先走爵位系统的清理。on_title_destroyed 钩子（WJ_peerage_on_actions.txt:27）本也会清，
			# 这里显式先清一次作保险；清后头衔已无 wj_peer_rank，钩子不会重复执行
			if = {
				limit = { has_variable = wj_peer_rank }
				WJ_peer_clear_family = yes
			}
			holder = { destroy_title = prev }
		}
	}
}

# 角色。记为北人，家族记为侨姓（已是侨姓者不改籍贯）。须在一切头衔、世族变动之后调用（新建世族可能分出新家族）
# 籍贯取 scope:hd_yiguan_gt_origin（县）；与 E 的 hd_yiguan_ev_mark_qiao_effect 用同一组变量
hd_yiguan_gt_mark_qiao_effect = {
	set_variable = hd_yiguan_northerner
	house ?= {
		if = {
			limit = { NOT = { has_variable = hd_yiguan_qiao_house } }
			set_variable = { name = hd_yiguan_qiao_house value = yes }
		}
		if = {
			limit = {
				NOT = { has_variable = hd_yiguan_qiao_origin }
				exists = scope:hd_yiguan_gt_origin
			}
			set_variable = { name = hd_yiguan_qiao_origin value = scope:hd_yiguan_gt_origin }
		}
	}
}

# 决议效果。Scope：迁徙者；须已存 scope:hd_yiguan_gt_dest
hd_yiguan_gt_nanqian_request_effect = {
	save_scope_as = hd_yiguan_gt_migrant
	set_variable = { name = hd_yiguan_gt_nanqian_pending value = scope:hd_yiguan_gt_dest days = 90 }
	set_variable = { name = hd_yiguan_gt_nanqian_paid value = hd_yiguan_gt_nanqian_cost_value days = 120 }
	if = {
		limit = { is_ai = yes }
		hd_yiguan_gt_bump_counter_effect = { NAME = nanqian }
	}
	scope:hd_yiguan_gt_dest = { trigger_event = hd_yiguan.4001 }
	debug_log = "hd_yiguan_gentry: nanqian requested"
}

# 角色（迁徙者）。退还已付威望并清除待复标记
hd_yiguan_gt_nanqian_refund_effect = {
	if = {
		limit = { has_variable = hd_yiguan_gt_nanqian_paid }
		add_prestige = var:hd_yiguan_gt_nanqian_paid
		remove_variable = hd_yiguan_gt_nanqian_paid
	}
	remove_variable = hd_yiguan_gt_nanqian_pending
}

# 举族南迁执行。Scope：迁徙者（root）；须已存 scope:hd_yiguan_gt_dest
# 次序：存旧主、胡人宗主与籍贯 → 收家属名单 → （玩家且投天朝制而无世族：先立世族）→ 封臣交胡人宗主 → 实地交胡人宗主
#       → 分支（天朝制保留世族改投 / 其余销毁世族入宫廷）→ 家属随行 → 打侨姓标记 → 好感、催化剂、侨吴之争
hd_yiguan_gt_nanqian_execute_effect = {
	save_scope_as = hd_yiguan_gt_migrant
	clear_saved_scope = hd_yiguan_gt_old_liege
	clear_saved_scope = hd_yiguan_gt_hu_lord
	clear_saved_scope = hd_yiguan_gt_old_top
	clear_saved_scope = hd_yiguan_gt_origin

	# 1. 旧主、胡人宗主与籍贯（籍贯：有地取都城县，无地取宅邸所在县，再不行取所在县）
	# 旧主 = 直属主君（只用来收家属名单与通知）；胡人宗主（§4 第 11 轮"交还全部封地给胡人宗主"，收地并记好感 −30）：
	# a. 无地者（只持世族头衔、有宅邸）：取宅邸所在县持有者的最高宗主，须是胡人且未汉化。
	#    本 mod 的北方无地世族多半仍挂晋帝（汉人，持 h_china），胡人占了北方也不会自动换宗主，
	#    只看宗主会把晋帝当成"胡人旧主"（判定与沦陷区汉人组 hd_yiguan_is_occupied_han_trigger 的无地一支一致；
	#    取法先例原版 10_dlc_tgp_dynastic_cycle_scripted_effects.txt:585-592）
	# b. 取不到（或有地者）：直属主君是胡人（未汉化）即取他，否则取最高宗主
	#    （4002 执行前刚复核过沦陷区判定：有地者最高宗主必为胡人；无地者走到这里说明宅邸县不在胡人治下，则最高宗主为胡人）
	liege ?= { save_scope_as = hd_yiguan_gt_old_liege }
	if = {
		limit = {
			is_landed = no
			domicile ?= {
				domicile_location ?= {
					county ?= {
						holder ?= {
							top_liege = {
								hd_yiguan_is_hu_trigger = yes
								NOT = { hd_yiguan_is_sinicized_trigger = yes }
							}
						}
					}
				}
			}
		}
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = hd_yiguan_gt_hu_lord }
	}
	else_if = {
		limit = {
			scope:hd_yiguan_gt_old_liege ?= {
				hd_yiguan_is_hu_trigger = yes
				NOT = { hd_yiguan_is_sinicized_trigger = yes }
			}
		}
		scope:hd_yiguan_gt_old_liege = { save_scope_as = hd_yiguan_gt_hu_lord }
	}
	else_if = {
		limit = {
			exists = top_liege
			NOT = { top_liege = root }
		}
		top_liege = { save_scope_as = hd_yiguan_gt_hu_lord }
	}
	else_if = {
		limit = { exists = scope:hd_yiguan_gt_old_liege }
		scope:hd_yiguan_gt_old_liege = { save_scope_as = hd_yiguan_gt_hu_lord }
	}
	# 最高宗主与胡人宗主不是同一人时另记一次好感：只在最高宗主本身是胡人（未汉化）且不是目的地时
	# （无地世族的最高宗主多半是晋帝，常常就是目的地，不能记"弃地南奔"）
	if = {
		limit = {
			exists = top_liege
			exists = scope:hd_yiguan_gt_hu_lord
			NOT = { top_liege = root }
			NOT = { top_liege = scope:hd_yiguan_gt_hu_lord }
			NOT = { top_liege = scope:hd_yiguan_gt_dest }
			top_liege = {
				hd_yiguan_is_hu_trigger = yes
				NOT = { hd_yiguan_is_sinicized_trigger = yes }
			}
		}
		top_liege = { save_scope_as = hd_yiguan_gt_old_top }
	}
	if = {
		limit = { exists = capital_county }
		capital_county = { save_scope_as = hd_yiguan_gt_origin }
	}
	else = {
		domicile ?= {
			domicile_location ?= {
				county ?= { save_scope_as = hd_yiguan_gt_origin }
			}
		}
	}
	if = {
		limit = { NOT = { exists = scope:hd_yiguan_gt_origin } }
		location ?= {
			county ?= { save_scope_as = hd_yiguan_gt_origin }
		}
	}

	# 2. 家属名单：配偶、子女、未婚兄弟姐妹（须在交出头衔之前收，交出后廷臣去向未验证）
	# 筛选仿原版 ep3_frankokratia_events.txt:7853-7871 / mpo_migration_events.txt:1825-1844
	every_spouse = {
		limit = {
			is_ruler = no
			is_imprisoned = no
			OR = {
				is_courtier_of = root
				AND = {
					exists = scope:hd_yiguan_gt_old_liege
					is_courtier_of = scope:hd_yiguan_gt_old_liege
				}
			}
		}
		add_to_list = hd_yiguan_gt_family
	}
	every_child = {
		limit = {
			is_ruler = no
			is_imprisoned = no
			is_hostage = no
			is_foreign_court_guest = no
			NOT = { is_in_list = hd_yiguan_gt_family }
			OR = {
				is_courtier_of = root
				AND = {
					exists = scope:hd_yiguan_gt_old_liege
					is_courtier_of = scope:hd_yiguan_gt_old_liege
				}
			}
		}
		add_to_list = hd_yiguan_gt_family
	}
	every_sibling = {
		limit = {
			is_ruler = no
			is_married = no
			is_imprisoned = no
			is_hostage = no
			NOT = { is_in_list = hd_yiguan_gt_family }
			OR = {
				is_courtier_of = root
				AND = {
					exists = scope:hd_yiguan_gt_old_liege
					is_courtier_of = scope:hd_yiguan_gt_old_liege
				}
			}
		}
		add_to_list = hd_yiguan_gt_family
	}

	# 3. 玩家投天朝制而无世族：趁仍持实地先立世族，免得交地后一度无头衔（§6 第 17 条）
	# create_noble_family_effect（mod 07_dlc_ep3_scripted_effects.txt:3379）：本人仍是统治者时不会自动宣誓，下面第 6 步再改投
	if = {
		limit = {
			is_ai = no
			NOT = { any_held_title = { is_noble_family_title = yes } }
			scope:hd_yiguan_gt_dest = { government_has_flag = government_is_celestial }
		}
		create_noble_family_effect = { GOVERNMENT_GIVER = scope:hd_yiguan_gt_dest }
	}

	# 4. 封臣一律交胡人宗主（先例 mpo 09_dlc_mpo_scripted_effects.txt:700-713）
	if = {
		limit = {
			exists = scope:hd_yiguan_gt_hu_lord
			any_vassal = { always = yes }
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_yiguan_gt_vassal_change
			add_claim_on_loss = no
		}
		every_vassal = {
			change_liege = {
				liege = scope:hd_yiguan_gt_hu_lord
				change = scope:hd_yiguan_gt_vassal_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_yiguan_gt_vassal_change
	}

	# 5. 实地全部交还胡人宗主（先例 mod yearly_on_actions.txt:1091-1111：stepped_down + 列表 + change_title_holder_include_vassals）
	if = {
		limit = {
			exists = scope:hd_yiguan_gt_hu_lord
			any_held_title = {
				is_landless_type_title = no
				is_noble_family_title = no
			}
		}
		create_title_and_vassal_change = {
			type = stepped_down
			save_scope_as = hd_yiguan_gt_step_change
			add_claim_on_loss = no
		}
		every_held_title = {
			limit = {
				is_landless_type_title = no
				is_noble_family_title = no
			}
			add_to_list = hd_yiguan_gt_land_titles
		}
		every_in_list = {
			list = hd_yiguan_gt_land_titles
			change_title_holder_include_vassals = {
				holder = scope:hd_yiguan_gt_hu_lord
				change = scope:hd_yiguan_gt_step_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_yiguan_gt_step_change
	}

	# 6. 分支
	if = {
		limit = {
			scope:hd_yiguan_gt_dest = { government_has_flag = government_is_celestial }
			any_held_title = { is_noble_family_title = yes }
			# 改投须主君高一级（此时已交出实地，比的是世族头衔级别；行政制世族为公国级）；目的地已是现任主君则不必改投
			OR = {
				liege ?= scope:hd_yiguan_gt_dest
				scope:hd_yiguan_gt_dest = { highest_held_title_tier > scope:hd_yiguan_gt_migrant.highest_held_title_tier }	# 写法同原版 coronation.txt:2317
			}
		}
		# 6a. 天朝制：以无地世族封臣身份改投目的地并迁宅邸（先例 mod 10_dlc_tgp_scripted_effects.txt:3578-3590）
		# 不满足级别条件的 AI 落入 6b；玩家已由 hd_yiguan_gt_nanqian_dest_trigger 挡住
		# 目的地已是现任主君（如宗主仍是晋帝的北方无地世族投晋帝）：跳过改投，只迁宅邸、收家属
		if = {
			limit = {
				NOT = { liege ?= scope:hd_yiguan_gt_dest }
			}
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = hd_yiguan_gt_fealty_change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = scope:hd_yiguan_gt_dest
				change = scope:hd_yiguan_gt_fealty_change
			}
			resolve_title_and_vassal_change = scope:hd_yiguan_gt_fealty_change
		}
		domicile ?= { move_domicile = scope:hd_yiguan_gt_dest.capital_province }	# ?= 写法见 mpo 09_dlc_mpo_scripted_effects.txt:748
		every_in_list = {
			list = hd_yiguan_gt_family
			limit = {
				is_alive = yes
				is_ruler = no
				NOT = { is_courtier_of = root }
			}
			root = { add_courtier = prev }
		}
		if = {
			limit = { NOT = { liege ?= scope:hd_yiguan_gt_dest } }
			debug_log = "hd_yiguan_gentry: nanqian swear_fealty to dest FAILED"
		}
	}
	else = {
		# 6b. 非天朝制，或 AI 无世族：退出世族，举家入目的地宫廷（写法同原版 00_courtier_and_guest_interactions.txt:751-756）
		hd_yiguan_gt_leave_noble_family_effect = yes
		if = {
			limit = {
				is_ruler = no
				NOT = { is_courtier_of = scope:hd_yiguan_gt_dest }	# 目的地本是现任主君时可能已在其宫廷
			}
			scope:hd_yiguan_gt_dest = { add_courtier = root }
			hidden_effect = { return_to_court = yes }
		}
		else_if = {
			limit = { is_ruler = yes }
			debug_log = "hd_yiguan_gentry: nanqian migrant still a ruler, add_courtier skipped"
		}
		every_in_list = {
			list = hd_yiguan_gt_family
			limit = {
				is_alive = yes
				is_ruler = no
				NOT = { is_courtier_of = scope:hd_yiguan_gt_dest }
			}
			scope:hd_yiguan_gt_dest = { add_courtier = prev }
			hidden_effect = { return_to_court = yes }
		}
	}

	# 7. 侨姓标记（最后打，落在最终的家族上）
	hd_yiguan_gt_mark_qiao_effect = yes
	every_in_list = {
		list = hd_yiguan_gt_family
		limit = { is_alive = yes }
		set_variable = hd_yiguan_northerner
	}

	# 8. 胡人宗主好感 −30（收地的胡人宗主与最高宗主各记一次）；玩家胡人旧主（收地者，或另为胡人的直属主君）收到通知
	# 4005 是胡人君主视角的通知；直属主君是汉人（如无地世族的宗主晋帝，且常常就是目的地）时不发
	scope:hd_yiguan_gt_hu_lord ?= {
		if = {
			limit = { is_alive = yes }
			add_opinion = {
				target = root
				modifier = hd_yiguan_fled_opinion
			}
			if = {
				limit = { is_ai = no }
				trigger_event = hd_yiguan.4005
			}
		}
	}
	scope:hd_yiguan_gt_old_liege ?= {
		if = {
			limit = {
				is_alive = yes
				is_ai = no
				NOT = { scope:hd_yiguan_gt_hu_lord ?= this }
				NOT = { this = scope:hd_yiguan_gt_dest }
				hd_yiguan_is_hu_trigger = yes
				NOT = { hd_yiguan_is_sinicized_trigger = yes }
			}
			trigger_event = hd_yiguan.4005
		}
	}
	scope:hd_yiguan_gt_old_top ?= {
		if = {
			limit = { is_alive = yes }
			add_opinion = {
				target = root
				modifier = hd_yiguan_fled_opinion
			}
		}
	}

	# 9. 催化剂"士族南迁"；目的地为南方政权时 30% 几率 1–3 年后起侨吴之争（事件 1203 归旧事件改编模块，只能发给南方政权）
	hd_yiguan_fire_catalyst_effect = { CATALYST = catalyst_hd_yiguan_gentry_flee CHAR = root }
	scope:hd_yiguan_gt_dest = {
		if = {
			limit = { hd_yiguan_in_group_trigger = { GROUP = hd_yiguan_grp_south } }
			random = {
				chance = 30
				trigger_event = {
					id = hd_yiguan.1203
					days = { 365 1095 }
				}
			}
		}
	}

	# 10. 收尾
	remove_variable = hd_yiguan_gt_nanqian_pending
	remove_variable = hd_yiguan_gt_nanqian_paid
	hd_yiguan_recalc_group_effect = yes
	debug_log = "hd_yiguan_gentry: nanqian done"
}

					hd_yiguan_gt_bh_def_ok_trigger = yes
					NOT = { is_in_list = hd_yiguan_gt_bh_defs_$KEY$ }
				}
			}
		}
		holder.top_liege = { add_to_list = hd_yiguan_gt_bh_defs_$KEY$ }	# 写法同 zz_hd_yiguan_event_effects.txt:hd_yiguan_join_south_effect

		list = hd_yiguan_gt_bh_defs_$KEY$

			}
		}

	}

# 任意作用域；须有 scope:hd_yiguan_gt_bh_def。北还战争目标（运行时法理王国）：
# 在 $REGION$ 州内守方体系所持的县里，按各县当前法理王国（县作用域 kingdom 链接）分组，取守方持县最多的那个王国
# 起兵北还 CB 为 target_title_tier = kingdom、target_de_jure_regions_above = yes，失效与夺地都按运行时法理结算，
# 所以目标须是县现今所挂的王国（南北朝剧本里可能是洛州、司州等），不能直接用 k_uuii_* 原州
# 结果：scope:hd_yiguan_gt_bh_kingdom，可能不存在（守方已不在该州持县）
hd_yiguan_gt_bh_pick_kingdom_effect = {
	clear_saved_scope = hd_yiguan_gt_bh_kingdom
	every_county_in_region = {
		region = $REGION$
		limit = {
			holder ?= { target_is_same_character_or_above = scope:hd_yiguan_gt_bh_def }	# 写法同 zz_hd_yiguan_cb.txt 北还 should_invalidate
			exists = kingdom
		}
		kingdom = {	# 县的运行时法理王国（读法见原版 tgp_east_asia_decisions.txt:673）
			if = {
				limit = { NOT = { is_in_list = hd_yiguan_gt_bh_kingdoms } }
				add_to_list = hd_yiguan_gt_bh_kingdoms
			}
		}
	}
	ordered_in_list = {	# 从候选王国中按法理县择优，写法同原版 00_faction_effects.txt:123-140
		list = hd_yiguan_gt_bh_kingdoms
		order_by = {	# 该王国法理下、位于 $REGION$ 州内、归守方体系的县数（数值内 every_de_jure_county 先例原版 03_dlc_fp2_script_values.txt:691）
			value = 0
			every_de_jure_county = {
				limit = {
					title_province = { geographical_region = $REGION$ }
					holder ?= { target_is_same_character_or_above = scope:hd_yiguan_gt_bh_def }
				}
				add = 1
			}
		}
		save_scope_as = hd_yiguan_gt_bh_kingdom
	}
}

# 角色（流亡者）。折算快照：须在退出世族（退还庄园、国库入私囊）之前
hd_yiguan_gt_bh_snapshot_effect = {
	remove_variable = hd_yiguan_gt_bh_buildings
	remove_variable = hd_yiguan_gt_bh_influence
	# 本家庄园建筑数：本人有宅邸取本人的，否则取家主的
	clear_saved_scope = hd_yiguan_gt_bh_bld
	if = {
		limit = { has_domicile = yes }
		save_scope_value_as = { name = hd_yiguan_gt_bh_bld value = domicile.num_domicile_buildings }	# 写法见原版 10_dlc_tgp_scripted_effects.txt:3383
	}
	else = {
		house ?= {
			house_head ?= {
				if = {
					limit = { has_domicile = yes }
					save_scope_value_as = { name = hd_yiguan_gt_bh_bld value = domicile.num_domicile_buildings }
				}
			}
		}
	}
	if = {
		limit = { exists = scope:hd_yiguan_gt_bh_bld }
		set_variable = { name = hd_yiguan_gt_bh_buildings value = scope:hd_yiguan_gt_bh_bld }
	}
	# 影响力（读法见原版 on_action/death.txt:1414-1425）
	if = {
		limit = {
			is_ruler = yes
			government_has_flag = government_has_influence
		}
		set_variable = { name = hd_yiguan_gt_bh_influence value = influence }
	}
	set_variable = { name = hd_yiguan_gt_bh_levy value = hd_yiguan_gt_beihuan_levy_base_value }
	set_variable = { name = hd_yiguan_gt_bh_companions value = hd_yiguan_gt_beihuan_companion_value }
}

		}
		add_to_list = hd_yiguan_gt_bh_cands
	}
	every_relation = {	# 门生（先例 mod hd_party_decisions.txt：ordered_relation type = disciple）
		type = disciple

			NOT = { is_in_list = hd_yiguan_gt_bh_cands }

		}
		add_to_list = hd_yiguan_gt_bh_cands
	}
	hd_yiguan_gt_bh_pick_companion_effect = yes
	if = {
		limit = { var:hd_yiguan_gt_bh_companions >= 2 }
		hd_yiguan_gt_bh_pick_companion_effect = yes
	}
	if = {
		limit = { var:hd_yiguan_gt_bh_companions >= 3 }
		hd_yiguan_gt_bh_pick_companion_effect = yes
	}
}
