hd_embattle_migrate_once_effect = {
    if = {
        limit = { NOT = { has_global_variable = hd_embattle_perf_v1_migrated } }
        # Set BEFORE scanning to make every caller idempotent.
        set_global_variable = { name = hd_embattle_perf_v1_migrated value = yes }
        clear_global_variable_list = embattle_clumsy_tactic_group
        every_living_character = {
            limit = {
                OR = {
                    has_character_flag = embattle_temp_trait_seeded
                    has_character_modifier = embattle_side_commander_advantage_modifier
                    has_variable_list = embattle_clumsy_tactic_group
                    has_variable_list = potential_knight
                    has_variable_list = potential_knight_temp
                    has_character_modifier = embattle_volly_tactic_modifier
                    has_character_modifier = embattle_charge_tactic_modifier

                    has_character_modifier = embattle_advance_tactic_modifier
                    has_character_modifier = embattle_harass_tactic_modifier
                    has_character_modifier = embattle_potshots_tactic_modifier
                    has_character_modifier = embattle_reckless_charge_tactic_modifier
                    has_character_modifier = embattle_confused_orders_tactic_modifier
                    has_character_modifier = embattle_timid_advance_tactic_modifier
                    has_character_modifier = embattle_disorganised_harass_tactic_modifier

                }
            }
            hd_embattle_clear_bonus_effect = yes
            hd_embattle_remove_legacy_tactics_effect = yes
        }
        # Old saves loaded during combat must regain valid player bonuses without waiting
        # for their stationary armies to move. This one-off player-only pass is not a pulse.
        every_player = {
            every_army = {
                limit = { exists = involved_combat_side }
                hd_embattle_maintain_army_effect = yes
            }
        }
    }
