# 将公国领地 county 文化、及头衔持有者（中华传承）与其宫廷，改写为目标汉文化分支

zgrc_apply_duchy_culture_effect = {
	title:$DUCHY$ = {
		every_de_jure_county = {
			set_county_culture = culture:$CULTURE$
			# 伯爵领级别头衔持有者及其廷臣
			holder = {
				if = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = culture:$CULTURE$
					every_courtier = {
						limit = { culture = { has_cultural_pillar = heritage_chinese } }
						set_culture = culture:$CULTURE$
					}
				}
			}
		}
		holder = {
			if = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = culture:$CULTURE$
				every_courtier = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = culture:$CULTURE$
				}
			}
		}
	}
}

# ============================================================
# 力大飞砖 · 汉文化统治者随当地风俗（帝国部堂随中华持有者）
# every_ruler 中固定原版汉文化（han）者：
#   帝国级别及以上 → 随中华霸权持有者（皇帝）文化
#   其余 → 所在位置的汉文化分支
# ============================================================
zgrc_apply_ruler_local_culture_effect = {
	every_ruler = {
		limit = {
			culture = culture:han
		}
		if = {
			limit = {
				highest_held_title_tier >= tier_empire
			}
			set_culture = title:h_china.holder.culture
			every_courtier = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = title:h_china.holder.culture
			}
			every_vassal = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = title:h_china.holder.culture
			}
		}
		else = {
			if = {
				limit = {
					location.county.culture = { has_cultural_pillar = heritage_chinese }
				}
				set_culture = location.county.culture
				every_courtier = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = location.county.culture
				}
				every_vassal = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = location.county.culture
				}
			}
			else = {
				set_culture = title:h_china.holder.culture
				every_courtier = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = title:h_china.holder.culture
				}
				every_vassal = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = title:h_china.holder.culture
				}
			}
		}
	}
}

# ============================================================
# 革新同步 —— 同步 effect
# 触发：yearly_culture_pulse + 全局计数为 0（见 on_action）
# 逻辑：随机选一个已知革新 → 同步给首领最高领主治下全部华夏礼乐文化
# ============================================================
zgrc_innovation_sync_effect = {
	# 随机选一个已知革新
	random_known_innovation = {
		save_scope_as = chosen_innovation
	}
	# 未选中（该文化无已知革新）→ 跳过同步
	if = {
		limit = { exists = scope:chosen_innovation }
		# 同步给首领最高领主治下全部华夏礼乐文化
		culture_head = {
			top_liege = {
				every_sub_realm_county = {
					culture = {
						if = {
							limit = {
								has_cultural_tradition = tradition_zgrc_hua_xia_li_yue
								NOT = { has_innovation = scope:chosen_innovation }
							}
							add_innovation = scope:chosen_innovation
							# 消息通知：该文化有领袖 → 通知其文化领袖
							if = {
								limit = { exists = culture_head }
								culture_head = {
									send_interface_toast = {
										type = event_toast_effect_good
										title = zgrc_innovation_sync_toast_title
										left_icon = this
										custom_tooltip = zgrc_innovation_sync_toast_desc
									}
								}
							}
						}
					}
				}
			}
		}
	}
}

# ============================================================
# 革新初始化 —— 与原版 han 对齐
# 依据：原版 tgp_silk_road_innovation_setup_effect 的 CHINA 段（culture:han）
# 触发：on_game_start_after_lobby（经 zgrc_game_start_init 调用）
# 效果：全部华夏礼乐文化开局解锁与原版 han 一致的中华革新（867+ 9 项；1066+ 追加 5 项）
# ============================================================
zgrc_innovation_init_effect = {
	every_culture_global = {
		limit = {
			has_cultural_tradition = tradition_zgrc_hua_xia_li_yue
		}
		# —— 全部开局（867 / 1066 / 1178）——
		add_innovation = innovation_block_printing
		add_innovation = innovation_coking
		add_innovation = innovation_composite_crossbow
		add_innovation = innovation_cupellation
		add_innovation = innovation_dragon_kiln
		add_innovation = innovation_lacquered_armor
		add_innovation = innovation_pharmacopoeia
		add_innovation = innovation_sericulture
		add_innovation = innovation_waterworks
		# —— 原版兵法承接（867 唐即得）——
		add_innovation = innovation_repeating_crossbow
		add_innovation = innovation_zgrc_han_bingfa
		# —— 1066 起追加 ——
		if = {
			limit = { game_start_date >= 1066.1.1 }
			add_innovation = innovation_bulkheads
			add_innovation = innovation_champa_rice
			add_innovation = innovation_compass
			add_innovation = innovation_double_entry_bookkeeping
			add_innovation = innovation_fire_medicine
			# —— 宋制（1066 起）——
			add_innovation = innovation_zgrc_han_wubei
		}
	}
}

##############################
# 厚生慎死·极端爆发 —— 总入口（on_war_started，War 作用域）
# 触发侧固定：防御方（scope:defender 治下封臣表态）
# 外层：防御方独立帝国级
# 大类A：防御方华夏礼乐 → 场景1-5（faction 优先：农民/民粹 → 外敌 → 藩王 → 诸侯）
# 大类B：攻击方华夏礼乐（防御方非华夏，由 else_if 顺序保证）→ 场景6 汉人聚义
# 事件传参 var zgrc_extreme_war / zgrc_extreme_liege / zgrc_extreme_attacker
##############################
zgrc_tianxia_xiangying_effect = {
	if = {
		limit = {
			scope:defender = {
				is_independent_ruler = yes
				highest_held_title_tier >= tier_empire
			}
		}
		# —— 大类A：汉帝国被攻（防御方华夏礼乐）——
		if = {
			limit = {
				scope:defender = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } }
			}
			# —— 场景1：农民起义 ——
			if = {
				limit = {
					scope:war = { primary_attacker = { is_leading_faction_type = peasant_faction } }
					scope:attacker.current_military_strength >= {
						value = {
							value = scope:defender.current_military_strength
							multiply = zgrc_extreme_ratio_peasant
						}
					}
				}
				zgrc_extreme_ask_vassals_effect = { EVENT = zgrc_common.1001 }
			}
			# —— 场景2：民粹起义（苗、越土司作乱）——
			else_if = {
				limit = {
					scope:war = {
						primary_attacker = {
							is_leading_faction_type = populist_faction
							NOT = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } }
						}
						attacker.current_military_strength >= {
							value = {
								value = defender.current_military_strength
								multiply = zgrc_extreme_ratio_populist
							}
						}
					}
				}
				zgrc_extreme_ask_vassals_effect = { EVENT = zgrc_common.1002 }
			}
			# —— 场景3：外敌入侵（蛮夷）—— 概率触发臣子 ——
			else_if = {
				limit = {
					scope:war = {
						primary_attacker = { NOT = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } } }
						casus_belli = {
							any_in_list = {
								list = target_titles
								tier >= tier_kingdom
								OR = {
									title_province ?= { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } }
									any_de_jure_county = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } }
								}
							}
						}
					}
				}
				zgrc_extreme_ask_vassals_chance_effect = {
					EVENT = zgrc_common.1003
					CHANCE = zgrc_extreme_foreign_chance_scaled
				}
			}
			# —— 场景4：藩王造反（同族，双方皆汉人）——
			else_if = {
				limit = {
					scope:attacker.house ?= scope:defender.house
					scope:attacker.current_military_strength >= {
						value = {
							value = scope:defender.current_military_strength
							multiply = zgrc_extreme_ratio_prince
						}
					}
				}
				zgrc_extreme_ask_vassals_effect = { EVENT = zgrc_common.1004 }
			}
			# —— 场景5：诸侯造反（攻击方华夏礼乐）——
			else_if = {
				limit = {
					scope:attacker = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } }
					scope:attacker.current_military_strength >= {
						value = {
							value = scope:defender.current_military_strength
							multiply = zgrc_extreme_ratio_warlord
						}
					}
				}
				zgrc_extreme_ask_vassals_effect = { EVENT = zgrc_common.1005 }
			}
		}
		# —— 大类B：汉人聚义攻异族（红巾军 / 汉人世侯聚义）——
		else_if = {
			limit = {
				scope:war = {
					primary_attacker = {
						culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
						is_leading_faction_type = populist_faction
					}
					defender = { NOT = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } } }
					attacker.current_military_strength >= {
						value = {
							value = defender.current_military_strength
							multiply = zgrc_extreme_ratio_uprising
						}
					}
				}
			}
			zgrc_extreme_ask_vassals_effect = { EVENT = zgrc_common.1006 }
		}
	}
}

# 触发臣子表态（无条件）
zgrc_extreme_ask_vassals_effect = {
	scope:defender = {
		every_vassal = {
			limit = {
				culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si }
				NOT = { scope:war = { any_war_attacker = { this = root } } }
				NOT = { scope:war = { any_war_defender = { this = root } } }
			}
			set_variable = { name = zgrc_extreme_war      value = scope:war }
			set_variable = { name = zgrc_extreme_liege    value = scope:defender }
			set_variable = { name = zgrc_extreme_attacker value = scope:attacker }
			trigger_event = { id = $EVENT$ }
		}
	}
}

zgrc_extreme_ask_vassals_chance_effect = {

			limit = {
				culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si }
				NOT = { scope:war = { any_war_attacker = { this = root } } }

		}
	}
}

zgrc_jiduan_baofa_effect = {
	# —— 事件1：民粹 —— 乡勇来投（直接来投）——
	if = {
		limit = {
			scope:war = { primary_attacker = { is_leading_faction_type = populist_faction } }
			scope:war = { primary_attacker = { culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si } } }
			scope:war = { primary_defender = { NOT = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } } } }
		}
		scope:war = {
			primary_attacker = {
				spawn_army = {
					name = zgrc_yijun_army_name
					men_at_arms = {
						type = zgrc_hanjia_yiyong
						stacks = {
							value = 10   # 王国及以下：约1000人（10队）
							if = {
								limit = { scope:war = { primary_defender = { highest_held_title_tier >= tier_empire } } }
								multiply = 2   # 帝国：约2000人（20队）
							}
							if = {
								limit = { scope:war = { primary_defender = { highest_held_title_tier >= tier_hegemony } } }
								multiply = 2   # 霸主：约4000人（40队）
							}
						}
					}
					war = scope:war
					location = root.location
					origin = root.location
					inheritable = no
				}
			}
		}
	}
	# —— 事件2：农民 —— 乡勇聚义（直接来投）——
	else_if = {
		limit = {
			scope:war = { primary_attacker = { is_leading_faction_type = peasant_faction } }
			scope:war = { primary_attacker = { culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si } } }
		}
		scope:war = {
			primary_attacker = {
				spawn_army = {
					name = zgrc_yijun_army_name
					men_at_arms = {
						type = zgrc_hanjia_yiyong
						stacks = {
							value = 10   # 王国及以下：约1000人（10队）
							if = {
								limit = { scope:war = { primary_defender = { highest_held_title_tier >= tier_empire } } }
								multiply = 2   # 帝国：约2000人（20队）
							}
							if = {
								limit = { scope:war = { primary_defender = { highest_held_title_tier >= tier_hegemony } } }
								multiply = 2   # 霸主：约4000人（40队）
							}
						}
					}
					war = scope:war
					location = root.location
					origin = root.location
					inheritable = no
				}
			}
		}
	}
	# —— 事件3：普通战争 —— 义军来投（收复故土，唯一事件 2001）——
	# 攻击方及其最高领主须同为华夏礼乐（避免异族宗主麾下的汉军假「收复」之名触发）
	else_if = {
		limit = {
			scope:war = {
				primary_attacker = {
					culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
					top_liege = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } }
				}
				primary_defender = { NOT = { culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue } } }
			}
			scope:war.casus_belli = {
				any_in_list = {
					list = target_titles
					OR = {
						title_province ?= { culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si } }   # county 级目标（?= 保护非 county）
						any_de_jure_county = { culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si } }  # duchy/kingdom 的 de_jure counties
					}
				}
			}
		}
		scope:war = {
			primary_attacker = {
				trigger_event = { id = zgrc_common.2001 }
			}
		}
	}
}
