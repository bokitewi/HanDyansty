# All hd_embattle_*_army_effect / *_appointment_effect: army scope.
# Normalization never scans knights. A failed search does not unlock itself.
hd_embattle_normalize_army_effect = {
    if = {
        limit = {
            has_variable = hd_embattle_complete
            hd_embattle_both_valid_trigger = no
            NOT = { has_variable = hd_embattle_retry_blocked }
        }
        remove_variable = hd_embattle_auto_done
    }
    if = {
        limit = { hd_embattle_left_valid_trigger = no }
        remove_variable = embattle_left_flank_commander
    }
    if = {
        limit = { hd_embattle_right_valid_trigger = no }
        remove_variable = embattle_right_flank_commander
    }
    if = {
        limit = { hd_embattle_both_valid_trigger = yes }
        set_variable = { name = hd_embattle_complete value = yes }
        remove_variable = hd_embattle_retry_blocked
    }
    else = { remove_variable = hd_embattle_complete }
}

# Exactly one legal-candidate collection per permitted attempt, for players AND AI.
hd_embattle_attempt_appointment_effect = {
    save_scope_as = hd_embattle_army
    # Lock BEFORE collection. Empty lists, no candidates, and partial success stay locked.
    set_variable = { name = hd_embattle_auto_done value = yes }
    clear_variable_list = hd_embattle_candidates
    clear_variable_list = hd_embattle_shortlist
    army_owner = {
        every_knight = {
            limit = { hd_embattle_candidate_trigger = yes }
            scope:hd_embattle_army = {
                add_to_variable_list = { name = hd_embattle_candidates target = prev }
            }
        }
    }
    random_list = {
        1 = { set_variable = { name = hd_embattle_sort_mode value = 0 } }
        1 = { set_variable = { name = hd_embattle_sort_mode value = 1 } }
        1 = { set_variable = { name = hd_embattle_sort_mode value = 2 } }
        1 = { set_variable = { name = hd_embattle_sort_mode value = 3 } }
    }
    # Preserve the four equal-weight baseline ranking modes and max=3 shortlist.
    ordered_in_list = {
        variable = hd_embattle_candidates
        order_by = {
            if = {
                limit = { scope:hd_embattle_army.var:hd_embattle_sort_mode = 0 }
                value = age
            }
            else_if = {
                limit = { scope:hd_embattle_army.var:hd_embattle_sort_mode = 1 }
                value = martial
            }
            else_if = {
                limit = { scope:hd_embattle_army.var:hd_embattle_sort_mode = 2 }
                value = highest_held_title_tier
            }
            else = { value = "opinion(scope:hd_embattle_army.army_commander)" }
        }
        max = 3
        check_range_bounds = no
        scope:hd_embattle_army = {
            add_to_variable_list = { name = hd_embattle_shortlist target = prev }
        }
    }
    if = {
        limit = {
            hd_embattle_right_valid_trigger = no
            NOT = { has_variable = hd_embattle_manual_empty_right }
        }
        random_in_list = {
            variable = hd_embattle_shortlist
            limit = { hd_embattle_candidate_trigger = yes }
            scope:hd_embattle_army = {
                set_variable = { name = embattle_right_flank_commander value = prev }
                remove_list_variable = { name = hd_embattle_shortlist target = prev }
            }
        }
    }
    if = {
        limit = {
            hd_embattle_left_valid_trigger = no
            NOT = { has_variable = hd_embattle_manual_empty_left }
        }
        random_in_list = {
            variable = hd_embattle_shortlist
            limit = { hd_embattle_candidate_trigger = yes }
            scope:hd_embattle_army = {
                set_variable = { name = embattle_left_flank_commander value = prev }
                remove_list_variable = { name = hd_embattle_shortlist target = prev }
            }
        }
    }
    clear_variable_list = hd_embattle_candidates
    clear_variable_list = hd_embattle_shortlist
    remove_variable = hd_embattle_sort_mode
    if = {
        limit = { hd_embattle_auto_vacancy_trigger = yes }
        set_variable = { name = hd_embattle_retry_blocked value = yes }
    }
    else = { remove_variable = hd_embattle_retry_blocked }
    if = {
        limit = { hd_embattle_both_valid_trigger = yes }
        set_variable = { name = hd_embattle_complete value = yes }
    }
    else = { remove_variable = hd_embattle_complete }
}

# Single-army entry. No province scan, army-count query, or recurring AI scheduler.
hd_embattle_maintain_army_effect = {
    save_scope_as = hd_embattle_army
    if = {
        limit = { exists = army_owner }
        # A genuine owner change is a new appointment context, not a timed retry.
        if = {
            limit = {
                exists = var:hd_embattle_seen_owner
                NOT = { var:hd_embattle_seen_owner = army_owner }
            }
            remove_variable = hd_embattle_auto_done
            remove_variable = hd_embattle_retry_blocked
            remove_variable = hd_embattle_complete
            remove_variable = hd_embattle_manual_empty_left
            remove_variable = hd_embattle_manual_empty_right
        }
        if = {
            limit = {
                OR = {
                    NOT = { has_variable = hd_embattle_seen_owner }
                    trigger_if = {
                        limit = { exists = var:hd_embattle_seen_owner }
                        NOT = { var:hd_embattle_seen_owner = army_owner }
                    }
                }
            }
            set_variable = { name = hd_embattle_seen_owner value = army_owner }
        }
        hd_embattle_normalize_army_effect = yes
        if = {
            limit = {
                NOT = { has_variable = hd_embattle_auto_done }
                NOT = { has_variable = hd_embattle_retry_blocked }
                exists = army_commander
                army_commander = { is_alive = yes }
            }
            if = {
                limit = { hd_embattle_auto_vacancy_trigger = yes }
                hd_embattle_attempt_appointment_effect = yes
            }
            else = { set_variable = { name = hd_embattle_auto_done value = yes } }
        }
    }
    hd_embattle_refresh_army_bonus_effect = yes
}

# Called only by a player action. A retry preserves legal deputies and opens vacancies.
hd_embattle_retry_army_effect = {
    remove_variable = hd_embattle_auto_done
    remove_variable = hd_embattle_retry_blocked
    remove_variable = hd_embattle_manual_empty_left
    remove_variable = hd_embattle_manual_empty_right
    hd_embattle_maintain_army_effect = yes
}

# Clean a former commander only if THIS army still owns their recorded bonus.
hd_embattle_refresh_army_bonus_effect = {
    save_scope_as = hd_embattle_army
    if = {
        limit = { exists = var:hd_embattle_last_buff_commander }
        var:hd_embattle_last_buff_commander = {
            if = {
                limit = {
                    has_variable = hd_embattle_buff_army
                    var:hd_embattle_buff_army = scope:hd_embattle_army
                    OR = {
                        NOT = { exists = scope:hd_embattle_army.army_commander }
                        AND = {
                            exists = scope:hd_embattle_army.army_commander
                            NOT = { this = scope:hd_embattle_army.army_commander }
                        }
                    }
                }
                hd_embattle_clear_bonus_effect = yes
            }
        }
    }
    if = {
        limit = { exists = army_commander }
        army_commander = {
            if = {
                limit = { hd_embattle_bonus_eligible_trigger = yes }
                hd_embattle_sync_bonus_effect = yes
            }
            else_if = {
                limit = {
                    has_character_flag = hd_embattle_buff_active
                    has_variable = hd_embattle_buff_army
                    var:hd_embattle_buff_army = scope:hd_embattle_army
                }
                hd_embattle_clear_bonus_effect = yes
            }
        }
    }
}


# Each call computes the desired SOURCE tier. Existing copied XP is never added twice.
hd_embattle_sync_one_trait_effect = {
    hd_embattle_sync_trait_copy_effect = { TRAIT = $TRAIT$ COPY = embattle_$TRAIT$ }
}
hd_embattle_sync_trait_copy_effect = {
    if = {
        limit = {
            NOT = { has_trait = $TRAIT$ }
            OR = {
                scope:hd_embattle_army = {
                    hd_embattle_left_valid_trigger = yes
                    var:embattle_left_flank_commander = { has_trait = $TRAIT$ }
                }
                scope:hd_embattle_army = {
                    hd_embattle_right_valid_trigger = yes
                    var:embattle_right_flank_commander = { has_trait = $TRAIT$ }
                }
            }
        }
        save_scope_value_as = { name = hd_embattle_target_xp value = 0 }
        if = {
            limit = {
                OR = {
                    scope:hd_embattle_army = {
                        hd_embattle_left_valid_trigger = yes
                        var:embattle_left_flank_commander = {
                            has_trait = $TRAIT$
                            has_trait_xp = { trait = $TRAIT$ value >= 100 }
                        }
                    }
                    scope:hd_embattle_army = {
                        hd_embattle_right_valid_trigger = yes
                        var:embattle_right_flank_commander = {
                            has_trait = $TRAIT$
                            has_trait_xp = { trait = $TRAIT$ value >= 100 }
                        }
                    }
                }
            }
            save_scope_value_as = { name = hd_embattle_target_xp value = 100 }
        }
        else_if = {
            limit = {
                OR = {
                    scope:hd_embattle_army = {
                        hd_embattle_left_valid_trigger = yes
                        var:embattle_left_flank_commander = {
                            has_trait = $TRAIT$
                            has_trait_xp = { trait = $TRAIT$ value >= 66 }
                        }
                    }
                    scope:hd_embattle_army = {
                        hd_embattle_right_valid_trigger = yes
                        var:embattle_right_flank_commander = {
                            has_trait = $TRAIT$
                            has_trait_xp = { trait = $TRAIT$ value >= 66 }
                        }
                    }
                }
            }
            save_scope_value_as = { name = hd_embattle_target_xp value = 66 }
        }
        else_if = {
            limit = {
                OR = {
                    scope:hd_embattle_army = {
                        hd_embattle_left_valid_trigger = yes
                        var:embattle_left_flank_commander = {
                            has_trait = $TRAIT$
                            has_trait_xp = { trait = $TRAIT$ value >= 33 }
                        }
                    }
                    scope:hd_embattle_army = {
                        hd_embattle_right_valid_trigger = yes
                        var:embattle_right_flank_commander = {
                            has_trait = $TRAIT$
                            has_trait_xp = { trait = $TRAIT$ value >= 33 }
                        }
                    }
                }
            }
            save_scope_value_as = { name = hd_embattle_target_xp value = 33 }
        }
        if = {
            limit = { NOT = { has_trait = $COPY$ } }
            add_trait = $COPY$
        }
        if = {
            limit = {
                NOT = { has_trait_xp = { trait = $COPY$ value = scope:hd_embattle_target_xp } }
            }
            # Signed delta also handles decreases and retained XP after trait removal.
            add_trait_xp = {
                trait = $COPY$
                value = {
                    value = scope:hd_embattle_target_xp
                    subtract = "has_trait_xp($COPY$)"
                }
            }
        }
    }
    else_if = {
        limit = { has_trait = $COPY$ }
        remove_trait = $COPY$
    }
}

# Character scope. Do not clear the pending token here: a previously queued watcher
# must either service the next battle or consume the token and stop, never duplicate.
hd_embattle_clear_bonus_effect = {
    remove_character_modifier = embattle_side_commander_advantage_modifier
    hd_embattle_clear_shared_traits_effect = yes
    remove_character_flag = hd_embattle_buff_active
    remove_variable = hd_embattle_buff_army
    remove_variable = hd_embattle_buff_combat
}

hd_embattle_sync_bonus_effect = {
    if = {
        limit = { hd_embattle_bonus_eligible_trigger = yes }
        save_scope_as = hd_embattle_beneficiary
        commanding_army = { save_scope_as = hd_embattle_army }
        commanding_army.involved_combat_side.combat = { save_scope_as = hd_embattle_current_combat }
        set_variable = { name = hd_embattle_buff_army value = scope:hd_embattle_army }
        set_variable = { name = hd_embattle_buff_combat value = scope:hd_embattle_current_combat }
        scope:hd_embattle_army = {
            set_variable = { name = hd_embattle_last_buff_commander value = scope:hd_embattle_beneficiary }
        }
        # Registry includes replaced commanders, not just the eventual side_commander.
        scope:hd_embattle_current_combat = {
            if = {
                limit = {
                    NOT = {
                        is_target_in_variable_list = {
                            name = hd_embattle_buffed_commanders
                            target = scope:hd_embattle_beneficiary
                        }
                    }
                }
                add_to_variable_list = {
                    name = hd_embattle_buffed_commanders
                    target = scope:hd_embattle_beneficiary
                }
            }
        }
        if = {
            limit = { NOT = { has_character_modifier = embattle_side_commander_advantage_modifier } }
            add_character_modifier = embattle_side_commander_advantage_modifier
        }
        if = {
            limit = { has_game_rule = embattle_side_commander_traits_sharing_enable }
            hd_embattle_sync_one_trait_effect = { TRAIT = holy_warrior }
            hd_embattle_sync_one_trait_effect = { TRAIT = aggressive_attacker }
            hd_embattle_sync_one_trait_effect = { TRAIT = unyielding_defender }
            hd_embattle_sync_one_trait_effect = { TRAIT = flexible_leader }
            hd_embattle_sync_one_trait_effect = { TRAIT = reckless }
            hd_embattle_sync_one_trait_effect = { TRAIT = cautious_leader }
            hd_embattle_sync_one_trait_effect = { TRAIT = forest_fighter }
            hd_embattle_sync_one_trait_effect = { TRAIT = open_terrain_expert }
            hd_embattle_sync_one_trait_effect = { TRAIT = rough_terrain_expert }
            hd_embattle_sync_one_trait_effect = { TRAIT = desert_warrior }
            hd_embattle_sync_one_trait_effect = { TRAIT = jungle_stalker }
            hd_embattle_sync_one_trait_effect = { TRAIT = winter_soldier }
            add_character_flag = embattle_temp_trait_seeded
        }
        else = { hd_embattle_clear_shared_traits_effect = yes }
        add_character_flag = hd_embattle_buff_active
        if = {
            limit = { NOT = { has_character_flag = hd_embattle_watch_pending } }
            add_character_flag = hd_embattle_watch_pending
            trigger_event = { id = embattle_events.0100 days = 7 }
        }
    }
}

# Legacy public function names. No tactic function is dispatched by this patch.
embattle_seed_commander_trait_effect = {
    if = {
        limit = {
            exists = scope:commander
            exists = scope:army
            has_game_rule = embattle_side_commander_traits_sharing_enable
            scope:commander = { hd_embattle_bonus_eligible_trigger = yes }
        }
        scope:army = { save_scope_as = hd_embattle_army }
        scope:commander = { hd_embattle_sync_one_trait_effect = { TRAIT = $TRAIT$ } }
    }
}
embattle_seed_commander_trait_effect_base = {
    # Legacy parameter contract retained: TRAIT is the source, SEED_TRAIT is the copy.
    if = {
        limit = {
            exists = scope:army
            has_game_rule = embattle_side_commander_traits_sharing_enable
            hd_embattle_bonus_eligible_trigger = yes
        }
        scope:army = { save_scope_as = hd_embattle_army }
        hd_embattle_sync_trait_copy_effect = { TRAIT = $TRAIT$ COPY = $SEED_TRAIT$ }
    }
}
