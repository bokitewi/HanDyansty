# Preserve the complete native three-outcome denominator before applying F once.
# Macro arguments are scalar scope values; arithmetic stays in value blocks.
hd_dc_prepare_tournament_sabotage_relation_effect = {
	save_scope_value_as = { name = hd_dc_relation_positive_1 value = { value = 10 add = intrigue min = 5 } }
	save_scope_value_as = { name = hd_dc_relation_positive_2 value = 20 }
	save_scope_value_as = { name = hd_dc_relation_negative value = { value = 5 add = scope:sabotage_target.intrigue min = 5 } }
	hd_dc_prepare_negative_relation_completion_effect = { CHARACTER = scope:sabotage_target }
}

	save_scope_value_as = {
		name = hd_dc_relation_positive_1
		value = { value = "trait_compatibility(scope:new_guest)" multiply = 3 min = -30 max = 50 add = 35 }
	}
	save_scope_value_as = {
		name = hd_dc_relation_positive_2
		value = { value = "trait_compatibility(scope:new_guest)" multiply = 2 min = -35 max = 35 add = 50 }
	}
	save_scope_value_as = {
		name = hd_dc_relation_negative
		value = {
			value = "trait_compatibility(scope:new_guest)"
			multiply = -3
			min = -9
			max = 60
			add = 15
			if = { limit = { OR = { has_trait = vengeful has_trait = callous has_trait = sadistic has_trait = wrathful } } add = 15 }
		}
	}
	hd_dc_prepare_negative_relation_completion_effect = { CHARACTER = scope:new_guest }
}

# Shared by the two real native pools, retaining the relative positive weights.
hd_dc_prepare_negative_relation_completion_effect = {
	save_scope_value_as = {
		name = hd_dc_relation_native_probability
		value = {
			value = scope:hd_dc_relation_negative
			multiply = 100
			divide = { value = scope:hd_dc_relation_positive_1 add = scope:hd_dc_relation_positive_2 add = scope:hd_dc_relation_negative }
		}
	}
	hd_dc_prepare_rivalry_roll_effect = { BASE = scope:hd_dc_relation_native_probability CHARACTER = $CHARACTER$ }
}

# yearly.4001: leave the death branch's mass untouched. Apply F only to the
# conditional nonfatal rivalry/affair choice; unavailable affair means p=100.
hd_dc_prepare_yearly_4001_relation_effect = {
	save_scope_value_as = { name = hd_dc_relation_affair_weight value = 0 }
	if = {
		limit = {
			scope:neighboring_ruler = {
				any_close_family_member = {
					is_ai = yes
					is_adult = yes
					AND = {
						is_attracted_to_gender_of = scope:vassal
						save_temporary_scope_as = attraction_check
						scope:vassal = { is_attracted_to_gender_of = scope:attraction_check }
					}
					NOR = { has_relation_lover = scope:vassal has_trait = chaste has_trait = celibate }
				}
			}
		}
		save_scope_value_as = { name = hd_dc_relation_affair_weight value = 10 }
	}
	save_scope_value_as = {
		name = hd_dc_relation_nonfatal_probability
		value = { value = 1000 divide = { value = 10 add = scope:hd_dc_relation_affair_weight } }
	}
	scope:vassal = {
		hd_dc_prepare_rivalry_roll_effect = { BASE = scope:hd_dc_relation_nonfatal_probability CHARACTER = scope:neighboring_ruler }
	}
}
