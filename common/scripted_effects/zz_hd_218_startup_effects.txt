hd218_wei_wang_effect = {
	save_scope_as = hd218_wei_wang_holder
	title:e_hd218_wei = {
		save_scope_as = hd218_wei_wang_title_scope
		set_variable = { name = wj_rf_rank value = 2 }
		set_variable = { name = wj_rf_name_code value = 0 } # 显示走 WJ_rf_title_county_name，取魏郡名"魏"
		set_variable = { name = wj_rf_stem_capital value = title:d_weijun }
	}
	# 魏国开局无人持有，授给曹操（已在手则跳过）
	if = {
		limit = { NOT = { scope:hd218_wei_wang_title_scope.holder ?= scope:hd218_wei_wang_holder } }
		create_title_and_vassal_change = { type = granted save_scope_as = hd218_wei_wang_change add_claim_on_loss = no }
		scope:hd218_wei_wang_title_scope = {
			change_title_holder = { holder = scope:hd218_wei_wang_holder change = scope:hd218_wei_wang_change }
		}
		resolve_title_and_vassal_change = scope:hd218_wei_wang_change
	}

	if = {
		limit = { scope:hd218_wei_wang_title_scope.holder ?= scope:hd218_wei_wang_holder }
		set_primary_title_to = scope:hd218_wei_wang_title_scope
		set_realm_capital = title:c_uuii_p_1172 # 邺城
		WJ_rf_refresh_holder = yes
		# 继承法：照 WJ_rf.4 的写法，魏国和邺城设世家继承（法理头衔不设无继承人销毁）
		scope:hd218_wei_wang_title_scope = {
			clear_title_laws = yes
			add_title_law = noble_family_succession_law
		}
		title:c_uuii_p_1172 = {
			# 世家继承法的 can_title_have 要求县带 wj_rf_capital_of（照 WJ_rf_commit）
			set_variable = { name = wj_rf_capital_of value = scope:hd218_wei_wang_title_scope }
			clear_title_laws = yes
			add_title_law = noble_family_succession_law
		}
		# 曹丕为魏王继承人
		if = {
			limit = { character:cao_pi ?= { is_alive = yes } }
			set_designated_heir = character:cao_pi
		}
		set_global_variable = { name = hd218_wei_wang_title value = scope:hd218_wei_wang_title_scope }
		debug_log = "HD218 WEI_WANG CREATED"
	}
	else = {
		debug_log = "HD218 WEI_WANG FAILED"
	}
}

hd218_independent_duchy_effect = {
	if = {
		limit = { title:$DUCHY$.holder ?= character:cao_cao }
		title:$DUCHY$ = { save_temporary_scope_as = seceding_county }
		create_character = {
			location = title:$DUCHY$.title_capital_county.title_province
			template = post_migration_minor_warlord_template
			dynasty = generate # 天朝政体要求君主有王朝
			save_temporary_scope_as = hd218_ind_new
		}
		create_title_and_vassal_change = { type = usurped save_scope_as = hd218_ind_change add_claim_on_loss = no }
		title:$DUCHY$ = {
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_county
					holder ?= character:cao_cao
				}
				change_title_holder = { holder = scope:hd218_ind_new change = scope:hd218_ind_change take_baronies = yes }
			}
			change_title_holder = { holder = scope:hd218_ind_new change = scope:hd218_ind_change }
		}
		resolve_title_and_vassal_change = scope:hd218_ind_change
		create_title_and_vassal_change = { type = independency save_scope_as = hd218_ind_free_change add_claim_on_loss = no }
		scope:hd218_ind_new = { becomes_independent = { change = scope:hd218_ind_free_change } }
		resolve_title_and_vassal_change = scope:hd218_ind_free_change
		scope:hd218_ind_new = {
			if = {
				limit = { NOT = { has_government = celestial_government } }
				change_government = celestial_government
			}
			if = {
				limit = { is_independent_ruler = yes has_title = title:$DUCHY$ }
				debug_log = "HD218 IND_DUCHY OK $DUCHY$"
			}
			else = { debug_log = "HD218 IND_DUCHY FAILED $DUCHY$" }
		}
	}
	else = { debug_log = "HD218 IND_DUCHY SKIPPED $DUCHY$" }
}

hd218_ruler_governors_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_landed = yes exists = capital_county }
			save_temporary_scope_as = hd218_gov_ruler
			set_global_variable = { name = hd218_gov_count_$WHO$ value = 0 }
			every_held_title = {
				limit = { hd218_governor_duchy_trigger = { RULER = scope:hd218_gov_ruler } }
				add_to_temporary_list = hd218_gov_duchies_$WHO$
			}
			every_in_list = {
				list = hd218_gov_duchies_$WHO$
				hd218_make_governor_effect = { RULER = scope:hd218_gov_ruler WHO = $WHO$ }
			}
			debug_log = "HD218 GOVERNORS DONE $WHO$"
		}
	}
}

# Scope：郡。$RULER$ 为君主；新郡守作其廷臣生成，再受封成封臣。
hd218_make_governor_effect = {
	save_temporary_scope_as = seceding_county
	save_temporary_scope_as = hd218_gov_duchy
	create_character = {
		location = scope:seceding_county.title_capital_county.title_province
		template = post_migration_minor_warlord_template
		employer = $RULER$
		dynasty = generate # 天朝政体要求君主有王朝
		save_temporary_scope_as = hd218_gov_new
	}
	create_title_and_vassal_change = { type = granted save_scope_as = hd218_gov_change add_claim_on_loss = no }
	every_in_de_jure_hierarchy = {
		limit = {
			tier = tier_county
			holder ?= $RULER$
		}
		change_title_holder = { holder = scope:hd218_gov_new change = scope:hd218_gov_change take_baronies = yes }
	}
	change_title_holder = { holder = scope:hd218_gov_new change = scope:hd218_gov_change }
	resolve_title_and_vassal_change = scope:hd218_gov_change
	scope:hd218_gov_new = {
		hd218_governor_fix_liege_effect = { LIEGE = $RULER$ }
		add_to_global_variable_list = { name = hd218_governors target = this }
	}
	change_global_variable = { name = hd218_gov_count_$WHO$ add = 1 }
}

# Scope：新郡守。受封后领主不对就补一笔改投，并补天朝政体
hd218_governor_fix_liege_effect = {
	if = {
		limit = { NOT = { liege ?= $LIEGE$ } }
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd218_gov_fix_change add_claim_on_loss = no }
		change_liege = { liege = $LIEGE$ change = scope:hd218_gov_fix_change }
		resolve_title_and_vassal_change = scope:hd218_gov_fix_change
		debug_log = "HD218 GOVERNOR LIEGE_FIXED"
	}
	if = {
		limit = { NOT = { has_government = celestial_government } }
		change_government = celestial_government
	}
}

# 无人持有的郡（复核意见"乙"）：郡头衔交给首府县的现任县主（独立、只持县），不生成新人；
# 首府在本郡、只持县、独立的其他县主改投他。条件不合就跳过（铁弗预期跳过）。
hd218_empty_duchy_governor_effect = {
	if = {
		limit = {
			NOT = { exists = title:$DUCHY$.holder }
			title:$DUCHY$.title_capital_county.holder ?= {
				is_independent_ruler = yes
				highest_held_title_tier = tier_county
			}
		}
		title:$DUCHY$.title_capital_county.holder = { save_temporary_scope_as = hd218_empty_duke }
		create_title_and_vassal_change = { type = created save_scope_as = hd218_empty_change add_claim_on_loss = no }
		title:$DUCHY$ = { change_title_holder = { holder = scope:hd218_empty_duke change = scope:hd218_empty_change } }
		resolve_title_and_vassal_change = scope:hd218_empty_change
		title:$DUCHY$ = {
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_county
					holder ?= {
						NOT = { this = scope:hd218_empty_duke }
						hd218_empty_duchy_chief_trigger = { DUCHY = $DUCHY$ }
					}
				}
				holder = { add_to_temporary_list = hd218_empty_chiefs_$DUCHY$ }
			}
		}
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd218_empty_vassal_change add_claim_on_loss = no }
		every_in_list = {
			list = hd218_empty_chiefs_$DUCHY$
			change_liege = { liege = scope:hd218_empty_duke change = scope:hd218_empty_vassal_change }
		}
		resolve_title_and_vassal_change = scope:hd218_empty_vassal_change
		debug_log = "HD218 EMPTY_DUCHY OK $DUCHY$"
	}
	else = { debug_log = "HD218 EMPTY_DUCHY SKIPPED $DUCHY$" }
}

hd218_employer_fix_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = no
				is_ruler = no # 持世族头衔的诸葛亮、司马懿是封臣，不拉进宫廷
				character:$BOSS$ ?= { is_alive = yes is_landed = yes }
				NOT = { employer ?= character:$BOSS$ }
			}
			debug_log = "HD218 EMPLOYER_FIX $WHO$ -> $BOSS$"
			set_employer = character:$BOSS$
		}
		else = {
			debug_log = "HD218 EMPLOYER_OK $WHO$"
		}
	}
}

hd218_ruler_setup_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_landed = yes }
			# 短暂统治时长 -100%，十年（原版 SHORT_REIGN_YEARS = 10）后自然无短暂统治
			add_character_modifier = {
				modifier = hd_sg218_established_rule_modifier
				years = 10
			}
			# 正统 6 级：legitimacy_level 5，门槛 legitimacy_level_6（按持有者层级换算），照 saas_game_start.txt 的写法
			# 天朝（行政）封臣没有正统（原版 use_general_legitimacy 要求行政制须为最高领主），曹操作献帝封臣时跳过
			if = {
				limit = {
					use_general_legitimacy = yes
					legitimacy < legitimacy_level_6
				}
				add_legitimacy = {
					value = legitimacy_level_6
					subtract = legitimacy
					add = 1
				}
			}
		}
	}
}
