hd_j16_node_luoyang_effect = {
	if = {
		limit = { NOT = { has_global_variable = hd_370_luoyang_done } }
		if = {
			limit = { has_global_variable = hd_370_luoyang_started }
			hd_j16_luoyang_result_effect = yes
		}
		else_if = {
			limit = { current_date >= 370.7.9 }
			set_global_variable = hd_370_luoyang_done
			set_global_variable = hd_j16_luoyang_skip
			debug_log = "HD_J16 e3_q60 SKIP window_closed"
		}
		else_if = {
			limit = { current_date >= 369.12.1 }
			if = {
				limit = { exists = scope:hd_j16_ly_qin }
				clear_saved_scope = hd_j16_ly_qin
			}
			if = {
				limit = { exists = scope:hd_j16_ly_yan }
				clear_saved_scope = hd_j16_ly_yan
			}
			title:c_uuii_p_867.holder.top_liege ?= {
				if = {
					limit = {
						is_independent_ruler = yes
						dynasty ?= dynasty:fu_1
					}
					save_scope_as = hd_j16_ly_qin
				}
			}
			title:d_henan.holder.top_liege ?= {
				if = {
					limit = {
						is_independent_ruler = yes
						dynasty ?= dynasty:murong
					}
					save_scope_as = hd_j16_ly_yan
				}
			}
			if = {
				limit = {
					exists = scope:hd_j16_ly_qin
					exists = scope:hd_j16_ly_yan
					NOT = { scope:hd_j16_ly_qin = scope:hd_j16_ly_yan }
					title:c_uuii_p_786.holder ?= {
						OR = {
							this = scope:hd_j16_ly_yan
							top_liege = scope:hd_j16_ly_yan
						}
					}
				}
				hd_j16_luoyang_war_effect = yes
			}
			else_if = {
				limit = { NOT = { has_global_variable = hd_j16_luoyang_wlog } }
				set_global_variable = hd_j16_luoyang_wlog
				debug_log = "HD_J16 e3_q60 WAIT not_ready"
			}
		}
	}
	# 结局判定后清掉永不自清的 hd_nbc_luoyang_active（01 §5）
	if = {
		limit = {
			has_global_variable = hd_370_luoyang_done
			has_global_variable = hd_nbc_luoyang_active
		}
		remove_global_variable = hd_nbc_luoyang_active
	}
}

# 结局：双方都已没有这一场次的战争，再看金墉归谁
hd_j16_luoyang_result_effect = {
	# 旧 370 档：战争是旧开局效果打响的，没存秦燕之主，按旧当事人 ID 补存
	if = {
		limit = { NOT = { has_global_variable = hd_j16_luoyang_qin } }
		if = {
			limit = { exists = scope:hd_j16_ly_qin }
			clear_saved_scope = hd_j16_ly_qin
		}
		if = {
			limit = { exists = scope:hd_j16_ly_yan }
			clear_saved_scope = hd_j16_ly_yan
		}
		hd_j16_resolve_effect = { ID = lueyang_fu_020 AS = hd_j16_ly_qin }
		hd_j16_resolve_effect = { ID = changli_murong_077 AS = hd_j16_ly_yan }
		if = {
			limit = {
				exists = scope:hd_j16_ly_qin
				exists = scope:hd_j16_ly_yan
			}
			set_global_variable = { name = hd_j16_luoyang_qin value = scope:hd_j16_ly_qin }
			set_global_variable = { name = hd_j16_luoyang_yan value = scope:hd_j16_ly_yan }
			debug_log = "HD_J16 e3_q60 WAIT oldsave_backfill"
		}
		# 苻坚或慕容暐已死、补存不成：按位置兜底（与开战时的解析方式一致）。长安、河南郡、金墉三处的 top_liege 都已没有
		# 这一场次的战争，就按金墉归属判定：金墉的 top_liege 是苻氏即秦胜。不能只设 done 不给结果，下游 E3-Q70 等读 _qin_won/_qin_lost。
		# 战争还在打（被继承人接手）就等下一跳。1-B 删过渡桥后靠这里收尾
		else = {
			if = {
				limit = {
					NOT = { title:c_uuii_p_867.holder.top_liege ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
					NOT = { title:d_henan.holder.top_liege ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
					NOT = { title:c_uuii_p_786.holder.top_liege ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
				}
				set_global_variable = hd_370_luoyang_done
				if = {
					limit = { title:c_uuii_p_786.holder.top_liege ?= { dynasty ?= dynasty:fu_1 } }
					set_global_variable = hd_370_luoyang_qin_won
					debug_log = "HD_J16 e3_q60 FIRE result_qin_won oldsave_actor_gone"
				}
				else = {
					set_global_variable = hd_370_luoyang_qin_lost
					debug_log = "HD_J16 e3_q60 FIRE result_qin_lost oldsave_actor_gone"
				}
			}
		}
	}
	if = {
		limit = {
			has_global_variable = hd_j16_luoyang_qin
			has_global_variable = hd_j16_luoyang_yan
			NOT = { global_var:hd_j16_luoyang_qin ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
			NOT = { global_var:hd_j16_luoyang_yan ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
			# 存下的秦燕之主战中死去、战争由继承人接着打时，只看存下的人会提前判定：再按位置查一遍
			NOT = { title:c_uuii_p_867.holder.top_liege ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
			NOT = { title:c_uuii_p_786.holder.top_liege ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } } }
		}
		set_global_variable = hd_370_luoyang_done
		if = {
			limit = {
				title:c_uuii_p_786.holder ?= {
					OR = {
						this = global_var:hd_j16_luoyang_qin
						top_liege = global_var:hd_j16_luoyang_qin
						top_liege ?= { dynasty ?= dynasty:fu_1 }    # 秦主战中死去、继承人接手的情形
					}
				}
			}
			set_global_variable = hd_370_luoyang_qin_won
			debug_log = "HD_J16 e3_q60 FIRE result_qin_won"
		}
		else = {
			set_global_variable = hd_370_luoyang_qin_lost
			debug_log = "HD_J16 e3_q60 FIRE result_qin_lost"
		}
	}
}

# 慕容筑出降（原 S1 出降分支；计划复核必改 1 甲）。洛阳之战本身与结局见上方 E3-Q60（hd_j16_luoyang_*），这里不重复。

#   秦主正在打洛阳之战。AI 慕容筑直接出降；玩家收 hd_370.0110：a 出降 / b 死守。
# 变量（旧名）：hd_370_zhu_checked（了结）、hd_370_zhu_surrendered（fire 结果）；新增 hd_j16_zhu_skip、_wlog
############################################################
hd_j16_node_zhu_surrender_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = hd_370_zhu_checked }
			has_global_variable = hd_370_luoyang_started
			NOT = { has_global_variable = hd_370_luoyang_done }
			current_date >= 370.3.14
		}
		if = {
			limit = { current_date >= 370.7.9 }
			set_global_variable = hd_370_zhu_checked
			set_global_variable = hd_j16_zhu_skip
			debug_log = "HD_J16 e3_q60 SKIP zhu_window_closed"
		}
		else_if = {
			limit = { hd_j16_dead_trigger = { ID = hd_370_murong_zhu } }
			set_global_variable = hd_370_zhu_checked
			set_global_variable = hd_j16_zhu_skip
			debug_log = "HD_J16 e3_q60 SKIP zhu_actor_dead"
		}
		else = {
			if = {
				limit = { exists = scope:hd_j16_zs_zhu }
				clear_saved_scope = hd_j16_zs_zhu
			}
			hd_j16_resolve_effect = { ID = hd_370_murong_zhu AS = hd_j16_zs_zhu }
			hd_j16_zhu_qin_effect = yes
			if = {
				limit = {
					exists = scope:hd_j16_zs_zhu
					scope:hd_j16_zs_zhu = {
						is_imprisoned = no
						has_title = title:d_henan
						is_independent_ruler = no
						top_liege ?= { dynasty ?= dynasty:murong }
					}
					scope:hd_j16_zs_qin ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:luoyang } }
				}
				set_global_variable = hd_370_zhu_checked
				scope:hd_j16_zs_zhu = {
					if = {
						limit = { is_ai = yes }
						hd_j16_zhu_surrender_effect = yes
					}
					else = {
						trigger_event = hd_370.0110
						debug_log = "HD_J16 e3_q60 WAIT zhu_player_asked"
					}
				}
			}
			else_if = {
				limit = { NOT = { has_global_variable = hd_j16_zhu_wlog } }
				set_global_variable = hd_j16_zhu_wlog
				debug_log = "HD_J16 e3_q60 WAIT zhu_not_ready"
			}
		}
	}
}

# 慕容筑出降（照旧 hd_370_zhu_surrender_effect，人物改用解析）：秦主身上的洛阳之战白和；河南郡交王猛（王猛是秦臣、有地），

		limit = { NOT = { exists = scope:hd_j16_zs_zhu } }
		hd_j16_resolve_effect = { ID = hd_370_murong_zhu AS = hd_j16_zs_zhu }
	}
	hd_j16_zhu_qin_effect = yes
	if = {
		limit = {
			exists = scope:hd_j16_zs_zhu
			exists = scope:hd_j16_zs_qin
			scope:hd_j16_zs_qin = { is_landed = yes }
		}
		scope:hd_j16_zs_qin = {
			every_character_war = {
				limit = { var:hd_nbc_war_kind ?= flag:luoyang }
				end_war = white_peace
			}
		}
		if = {
			limit = { exists = scope:hd_j16_zs_wm }
			clear_saved_scope = hd_j16_zs_wm
		}
		hd_j16_resolve_effect = { ID = wang_meng_3 AS = hd_j16_zs_wm }
		if = {
			limit = {
				scope:hd_j16_zs_wm ?= {
					is_landed = yes
					top_liege = scope:hd_j16_zs_qin
				}
			}
			hd_370_take_realm_effect = { NEW = scope:hd_j16_zs_wm OLD = scope:hd_j16_zs_zhu }
		}
		else = {
			hd_370_take_realm_effect = { NEW = scope:hd_j16_zs_qin OLD = scope:hd_j16_zs_zhu }
		}
		scope:hd_j16_zs_qin = {
			if = {
				limit = {
					scope:hd_j16_zs_zhu = {
						is_alive = yes
						is_landed = no
					}
				}
				add_courtier = scope:hd_j16_zs_zhu
			}
		}
		set_global_variable = hd_370_zhu_surrendered
		debug_log = "HD_J16 e3_q60 FIRE zhu_surrendered"
	}
}

# 结局（照旧 S4 战后分支，桓温改用解析）
hd_j16_shouchun_result_effect = {
	hd_j16_shouchun_huan_effect = yes
	if = {
		limit = {
			NOT = { global_var:hd_370_shouchun_defender ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:shouchun } } }
			NOT = { title:h_china.holder ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:shouchun } } }
			NOT = { scope:hd_j16_sc_huan ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:shouchun } } }
		}
		set_global_variable = hd_370_shouchun_done
		if = {
			limit = {
				title:c_uuii_p_1961.holder ?= {
					OR = {
						this = title:h_china.holder
						top_liege = title:h_china.holder
					}
				}
			}
			set_global_variable = hd_370_shouchun_jin_won
			global_var:hd_370_shouchun_defender ?= {
				if = {
					limit = {
						is_alive = yes
						is_ai = yes
						is_landed = no
					}
					if = {
						limit = { exists = scope:hd_j16_sc_huan }
						death = { death_reason = death_execution killer = scope:hd_j16_sc_huan }
					}
					else = { death = { death_reason = death_execution } }
					debug_log = "HD_J16 e3_j80 FIRE yuan_executed"
				}
			}
			debug_log = "HD_J16 e3_j80 FIRE result_jin_won"
		}
		else = {
			set_global_variable = hd_370_shouchun_jin_lost
			debug_log = "HD_J16 e3_j80 FIRE result_jin_lost"
		}
	}
}
