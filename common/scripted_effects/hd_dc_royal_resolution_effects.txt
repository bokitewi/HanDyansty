# Main instance. E1/E4/E5 adapters and the complete political epilogues remain
# separate work; only completed authored claimant-war facts are recognized here.
hd_dc_observe_royal_war_resolution_effect = {
	if = {
		limit = { hd_dc_royal_supported_war_resolution_trigger = yes }
		if = {
			limit = { hd_dc_royal_war_enthronement_trigger = yes }
			set_variable = { name = hd_dc_political_result value = flag:roy_e2 }
		}
		else_if = {
			limit = { hd_dc_royal_war_defended_trigger = yes }
			set_variable = { name = hd_dc_political_result value = flag:roy_e3 }
		}
		if = {
			limit = { has_variable = hd_dc_political_result }
			set_variable = { name = hd_dc_political_result_date value = current_date }
			set_variable = { name = hd_dc_political_result_war_epoch value = var:hd_dc_war_epoch }
			set_variable = { name = hd_dc_political_ruler value = title:h_china.holder }
			hd_dc_settle_royal_enthronement_support_effect = yes
			set_variable = { name = hd_dc_stage value = flag:aftermath }
			hd_dc_release_own_acute_lock_effect = yes
			save_scope_as = hd_dc_resolved_royal_instance
			title:h_china.holder = { trigger_event = { id = hd_dc_roy_end.0001 days = 1 } }
			every_in_list = {
				variable = hd_dc_actor_stories
				limit = { exists = var:hd_dc_actor var:hd_dc_actor = { is_alive = yes this != title:h_china.holder } }
				var:hd_dc_actor = { trigger_event = { id = hd_dc_roy_end.0002 days = 1 } }
			}
		}
	}
}

# Main instance. ROY-E2's +20 belongs to living participants in the actual
# victorious claimant war, once per person across this instance's war receipts.
# This records gratitude, not a new financial debt or fulfilled protection.
hd_dc_settle_royal_enthronement_support_effect = {
	if = {
		limit = {
			var:hd_dc_political_result = flag:roy_e2
			exists = var:hd_dc_political_ruler
			var:hd_dc_political_ruler = { is_alive = yes }
			NOT = { has_variable = hd_dc_enthronement_relations_committed }
		}
		save_scope_as = hd_dc_enthronement_instance
		every_in_list = {
			variable = hd_dc_war_records
			limit = {
				var:hd_dc_core_war_result = 1
				var:hd_dc_war_cb = flag:claimant_faction_war
				var:hd_dc_war_target = title:h_china
				var:hd_dc_war_beneficiary = scope:hd_dc_enthronement_instance.var:hd_dc_political_ruler
				var:hd_dc_final_war_roster_recorded = yes
			}
			every_in_list = {
				variable = hd_dc_final_war_attackers
				limit = {
					is_alive = yes
					this != scope:hd_dc_enthronement_instance.var:hd_dc_political_ruler
					save_temporary_scope_as = hd_dc_enthronement_supporter
					NOT = {
						scope:hd_dc_enthronement_instance = {
							any_in_list = { variable = hd_dc_enthronement_supporters this = scope:hd_dc_enthronement_supporter }
						}
					}
				}
				add_opinion = { target = scope:hd_dc_enthronement_instance.var:hd_dc_political_ruler modifier = hd_dc_enthronement_support_opinion }
				save_scope_as = hd_dc_enthronement_supporter
				scope:hd_dc_enthronement_instance = { add_to_variable_list = { name = hd_dc_enthronement_supporters target = scope:hd_dc_enthronement_supporter } }
			}
		}
		set_variable = { name = hd_dc_enthronement_relations_committed value = yes }
	}
}

# ROY-E3: acknowledge a named royal's already completed undertaking, not mere
# kinship or absence from one faction. Unknown rosters cannot prove neutrality.
# Native legal cause is checked against this person, not inherited from family.
hd_dc_settle_royal_loyal_kin_effect = {
	if = {
		limit = {
			var:hd_dc_political_result = flag:roy_e3
			var:hd_dc_political_ruler = title:h_china.holder
			exists = var:hd_dc_political_ruler
			var:hd_dc_political_ruler = { is_alive = yes }
			hd_dc_all_core_wars_resolved_trigger = yes
			NOT = { any_in_list = { variable = hd_dc_war_records NOT = { var:hd_dc_final_war_roster_recorded = yes } } }
		}
		save_scope_as = hd_dc_defended_instance
		every_in_list = {
			variable = hd_dc_actor_stories
			limit = {
				var:hd_dc_ritual_pledge_result = flag:accepted
				NOT = { has_variable = hd_dc_pending_action }
				NOT = { has_variable = hd_dc_pending_proposal }
				exists = var:hd_dc_actor
				var:hd_dc_actor = {
					is_alive = yes
					this != scope:hd_dc_defended_instance.var:hd_dc_political_ruler
					save_temporary_scope_as = hd_dc_faithful_royal
				}
				NOT = {
					scope:hd_dc_defended_instance.var:hd_dc_political_ruler = { has_imprisonment_reason = scope:hd_dc_faithful_royal }
				}
				NOT = {
					scope:hd_dc_defended_instance = {
						any_in_list = { variable = hd_dc_faithful_royals this = scope:hd_dc_faithful_royal }
					}
				}
				NOT = {
					scope:hd_dc_defended_instance = {
						any_in_list = {
							variable = hd_dc_war_records
							any_in_list = { variable = hd_dc_final_war_attackers this = scope:hd_dc_faithful_royal }
						}
					}
				}
			}
			var:hd_dc_actor = {
				add_opinion = { target = scope:hd_dc_defended_instance.var:hd_dc_political_ruler modifier = hd_dc_faithful_royal_opinion }
				save_scope_as = hd_dc_faithful_royal
			}
			scope:hd_dc_defended_instance = { add_to_variable_list = { name = hd_dc_faithful_royals target = scope:hd_dc_faithful_royal } }
		}
	}
}
