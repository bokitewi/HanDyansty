# E1-H10 大陵之战（D9：由 304 开局播种改为心跳节点；01 §2.3、§5；02 §1.6 B0）
# 窗口 304.11.20 – 305.6.1。刘渊独立在世；司马腾在世且持太原郡；两人未交战。
# AI 刘渊直接出兵；玩家刘渊收 hd_j16.0300：a 出兵 / b 暂缓（90 天后再问）。
# 开战成功才给 1.5 万事件兵；宣战失败不设 done，下一跳重试。结局按太原郡治归属自判（不进 hd_nbc_war_result_effect）。
# 变量：hd_j16_daling_started（已开战，fire 结果）、_won/_lost、_skip、_done；hd_j16_daling_att/_def（攻守方）；
#       hd_j16_daling_go（玩家已允出兵）、_wait（玩家暂缓 90 天）、_asked（已问、等回复）、_wlog（等待日志只记一次）
############################################################
hd_j16_node_daling_effect = {
	if = {
		limit = { NOT = { has_global_variable = hd_j16_daling_done } }
		if = {
			limit = { has_global_variable = hd_j16_daling_started }
			hd_j16_daling_result_effect = yes
		}
		# 旧 304 档：大陵之战是旧开局效果打响的（只有 hd_nbc_daling_active）。补存攻守方，按已开战处理，不再开第二场
		else_if = {
			limit = { has_global_variable = hd_nbc_daling_active }
			hd_j16_daling_resolve_effect = yes
			if = {
				limit = {
					exists = scope:hd_j16_dl_liu
					exists = scope:hd_j16_dl_teng
				}
				set_global_variable = hd_j16_daling_started
				set_global_variable = { name = hd_j16_daling_att value = scope:hd_j16_dl_liu }
				set_global_variable = { name = hd_j16_daling_def value = scope:hd_j16_dl_teng }
				debug_log = "HD_J16 e1_h10 WAIT oldsave_backfill"
			}
			else = {
				# 一方已不在：旧战争不论胜负，节点收尾
				set_global_variable = hd_j16_daling_done
				set_global_variable = hd_j16_daling_skip
				remove_global_variable = hd_nbc_daling_active
				debug_log = "HD_J16 e1_h10 SKIP oldsave_actor_gone"
			}
		}
		else_if = {
			limit = { current_date >= 305.6.1 }
			set_global_variable = hd_j16_daling_done
			set_global_variable = hd_j16_daling_skip
			debug_log = "HD_J16 e1_h10 SKIP window_closed"
		}
		else_if = {
			limit = { current_date >= 304.11.20 }
			hd_j16_daling_resolve_effect = yes
			if = {
				limit = {
					OR = {
						hd_j16_dead_trigger = { ID = xiongnu_liu_020 }
						hd_j16_dead_trigger = { ID = si_ma_teng }
					}
				}
				set_global_variable = hd_j16_daling_done
				set_global_variable = hd_j16_daling_skip
				debug_log = "HD_J16 e1_h10 SKIP actor_dead"
			}
			else_if = {
				limit = {
					scope:hd_j16_dl_liu ?= { is_independent_ruler = yes }
					exists = scope:hd_j16_dl_teng
					title:d_taiyuan.holder ?= scope:hd_j16_dl_teng
					NOT = { scope:hd_j16_dl_liu = { is_at_war_with = scope:hd_j16_dl_teng } }
				}
				if = {
					limit = {
						OR = {
							scope:hd_j16_dl_liu = { is_ai = yes }
							has_global_variable = hd_j16_daling_go
						}
					}
					hd_j16_daling_war_effect = yes
				}
				else_if = {
					limit = {
						NOT = { has_global_variable = hd_j16_daling_wait }
						NOT = { has_global_variable = hd_j16_daling_asked }
					}
					# 待答标记：hd_j16.0300 的两个选项清掉，没回复前不重复弹；365 天只是兜底
					set_global_variable = { name = hd_j16_daling_asked value = yes days = 365 }
					scope:hd_j16_dl_liu = { trigger_event = hd_j16.0300 }
					debug_log = "HD_J16 e1_h10 WAIT player_asked"
				}
			}
			else_if = {
				limit = { NOT = { has_global_variable = hd_j16_daling_wlog } }
				set_global_variable = hd_j16_daling_wlog
				debug_log = "HD_J16 e1_h10 WAIT not_ready"
			}
		}
	}
}

# 解析当事人：刘渊、司马腾（按 ID；本节点当事人必须是这两人）。先清旧 scope
hd_j16_daling_resolve_effect = {
	if = {
		limit = { exists = scope:hd_j16_dl_liu }
		clear_saved_scope = hd_j16_dl_liu
	}
	if = {
		limit = { exists = scope:hd_j16_dl_teng }
		clear_saved_scope = hd_j16_dl_teng
	}
	hd_j16_resolve_effect = { ID = xiongnu_liu_020 AS = hd_j16_dl_liu }
	hd_j16_resolve_effect = { ID = si_ma_teng AS = hd_j16_dl_teng }
}

# 结局：双方都已没有这一场次的战争，再看太原郡治归谁（攻方一系 = 刘渊胜）。顺带清掉永不自清的 hd_nbc_daling_active（01 §5）
hd_j16_daling_result_effect = {
	if = {
		limit = {
			NOT = { global_var:hd_j16_daling_att ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:daling } } }
			NOT = { global_var:hd_j16_daling_def ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:daling } } }
		}
		set_global_variable = hd_j16_daling_done
		if = {
			limit = {
				title:d_taiyuan.title_capital_county.holder ?= {
					OR = {
						this = global_var:hd_j16_daling_att
						top_liege = global_var:hd_j16_daling_att
					}
				}
			}
			set_global_variable = hd_j16_daling_won
			debug_log = "HD_J16 e1_h10 FIRE result_won"
		}
		else = {
			set_global_variable = hd_j16_daling_lost
			debug_log = "HD_J16 e1_h10 FIRE result_lost"
		}
		if = {
			limit = { has_global_variable = hd_nbc_daling_active }
			remove_global_variable = hd_nbc_daling_active
		}
	}
}
