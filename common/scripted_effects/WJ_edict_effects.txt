 scope:actor.top_liege = { save_scope_as = wj_edict_emperor }
 if = { limit = { exists = scope:secondary_actor scope:actor = { WJ_edict_operator_for = { CROWN = scope:secondary_actor } } }
  scope:secondary_actor = { save_scope_as = wj_edict_emperor }
 }
}
WJ_peer_pay_resource = {
 save_scope_value_as = { name = wj_peer_due value = $COST$ }
 save_scope_value_as = { name = wj_peer_crown_payment value = { value = $CROWN$.$RESOURCE$ min = 0 max = scope:wj_peer_due } }
 $CROWN$ = { $EFFECT$ = { value = scope:wj_peer_crown_payment multiply = -1 } }
 $PAYER$ = { $EFFECT$ = { value = scope:wj_peer_due subtract = scope:wj_peer_crown_payment multiply = -1 } }
}
