# Scope：title:h_china。诛董进入割据后 45 天起尝试；落位（hd_hw_run_effect）完成后才接手，最多等一年。
hd184r_director_bootstrap_effect = {
	if = {
		limit = {
			game_start_date = 190.1.1
			exists = global_var:hd184r_warlord_era
			NOT = { exists = global_var:hd184r_director_active }
		}
		if = {
			limit = {
				OR = {
					exists = global_var:hd_hw_done
					exists = global_var:hd184r_director_wait_expired
				}
			}
			set_global_variable = hd184r_director_active
			hd184r_director_mark_roster_effect = yes
			trigger_event = { on_action = hd184r_director_pulse days = 1 }
			debug_log = "HD184R P4 DIRECTOR_STARTED"
		}
		else = {
			# 每 15 天再查一次，约一年（24 次）后不再等落位
			if = { limit = { NOT = { exists = global_var:hd184r_director_wait_count } } set_global_variable = { name = hd184r_director_wait_count value = 0 } }
			change_global_variable = { name = hd184r_director_wait_count add = 1 }
			if = {
				limit = { global_var:hd184r_director_wait_count >= 24 }
				set_global_variable = hd184r_director_wait_expired
			}
			trigger_event = { on_action = hd184r_director_bootstrap days = 15 }
			debug_log = "HD184R P4 DIRECTOR_WAIT_PLACEMENT"
		}
	}
}

# Scope：名单人物。
hd184r_director_tick_effect = {
	if = {
		limit = {
			has_character_flag = hd184r_director_member
			NOT = { has_character_flag = hd184r_done }
			NOT = { has_character_flag = hd190_cao_relay_active }	# 2026-10-26：曹操陈留—兖州接力期间暂停（不检查阶段、不计空转）
			hd184r_director_actor_ready_trigger = yes
		}
		save_scope_as = hd184r_actor
		hd184r_director_build_stage_effect = yes
		# 一、阶段完成检查（受剧情门槛挡住的阶段不检查、不计空转）
		if = {
			limit = { NOT = { has_character_flag = hd184r_stage_gated } }
			hd184r_director_check_stage_effect = yes
		}
		# 二、找目标、开战（检查后若刚进入新阶段，按新阶段重建目标）
		if = {
			limit = { has_character_flag = hd184r_stage_rebuild }
			remove_character_flag = hd184r_stage_rebuild
			hd184r_director_build_stage_effect = yes
		}
		if = {
			limit = { NOT = { has_character_flag = hd184r_done } }
			hd184r_director_collect_candidates_effect = yes
			if = {
				limit = { NOT = { has_variable_list = hd184r_scand } }
				# 目标州内没有可打的接壤目标：空转 +1（门槛挡住时不计）
				if = {
					limit = { NOT = { has_character_flag = hd184r_stage_gated } }
					change_variable = { name = hd184r_idle add = 1 }
				}
				debug_log = "HD184R P4 NO_STAGE_TARGET"
				debug_log_scopes = yes
			}
			else = { set_variable = { name = hd184r_idle value = 0 } }
			hd184r_director_declare_one_effect = yes
			hd184r_director_declare_one_effect = yes
		}
		clear_variable_list = hd184r_targets
		clear_variable_list = hd184r_front
		clear_variable_list = hd184r_scand
		clear_variable_list = hd184r_ocand
	}
}

# Scope：名单人物（scope:hd184r_actor）。实控 70% 或空转满 4 次即完成本阶段。
hd184r_director_check_stage_effect = {
	if = {
		limit = { NOT = { has_character_flag = hd184r_open_mode } NOT = { has_character_flag = hd184r_done } }
		set_variable = { name = hd184r_tot value = 0 }
		set_variable = { name = hd184r_own value = 0 }
		every_in_list = {
			variable = hd184r_targets
			every_de_jure_county = {
				scope:hd184r_actor = { change_variable = { name = hd184r_tot add = 1 } }
				if = {
					limit = {
						holder ?= {
							OR = {
								this = scope:hd184r_actor
								is_vassal_or_below_of = scope:hd184r_actor
							}
						}
					}
					scope:hd184r_actor = { change_variable = { name = hd184r_own add = 1 } }
				}
			}
		}
		set_variable = { name = hd184r_need value = var:hd184r_tot }
		change_variable = { name = hd184r_need multiply = 0.7 }
		if = {
			limit = {
				OR = {
					var:hd184r_tot <= 0
					var:hd184r_own >= var:hd184r_need
					var:hd184r_idle >= 4
				}
			}
			if = {
				limit = { var:hd184r_idle >= 4 }
				debug_log = "HD184R P4 STAGE_DONE_IDLE"
			}
			else = { debug_log = "HD184R P4 STAGE_DONE_CONTROL" }
			debug_log_scopes = yes
			change_variable = { name = hd184r_stage add = 1 }
			set_variable = { name = hd184r_idle value = 0 }
			add_character_flag = hd184r_stage_rebuild
		}
		remove_variable = hd184r_tot
		remove_variable = hd184r_own
		remove_variable = hd184r_need
	}
}

# Scope：名单人物（scope:hd184r_actor）。前线郡：与本人领地接壤、却不在本人领内的县所在的郡。
# 变量列表 hd184r_scand：前线郡里属于目标州且有合法守方的；hd184r_ocand：其余在汉霸权法理内且有合法守方的（只进取型用）。
hd184r_director_collect_candidates_effect = {
	clear_variable_list = hd184r_front
	clear_variable_list = hd184r_scand
	clear_variable_list = hd184r_ocand
	# 用 sub_realm：挟天子的权臣是天子的封臣，只算他自己的辖区
	every_sub_realm_county = {
		every_neighboring_county = {
			limit = {
				exists = holder
				NOT = {
					holder = {
						OR = {
							this = scope:hd184r_actor
							is_vassal_or_below_of = scope:hd184r_actor
						}
					}
				}
			}
			duchy ?= {
				save_temporary_scope_as = hd184r_front_duchy
				scope:hd184r_actor = {
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = hd184r_front target = scope:hd184r_front_duchy } } }
						add_to_variable_list = { name = hd184r_front target = scope:hd184r_front_duchy }
					}
				}
			}
		}
	}
	every_in_list = {
		variable = hd184r_front
		limit = { hd184r_duchy_has_valid_defender_trigger = yes }
		save_temporary_scope_as = hd184r_cand_duchy
		if = {
			limit = {
				scope:hd184r_actor = {
					NOT = { has_character_flag = hd184r_stage_gated }
					is_target_in_variable_list = { name = hd184r_targets target = scope:hd184r_cand_duchy }
				}
			}
			scope:hd184r_actor = { add_to_variable_list = { name = hd184r_scand target = scope:hd184r_cand_duchy } }
		}
		else_if = {
			limit = {
				scope:hd184r_actor = {
					has_character_flag = hd184r_type_jinqu
					# 被门槛挡住的目标州郡不从「州外」口子漏进来（官渡前的冀州、赤壁前的荆南）
					NOT = { is_target_in_variable_list = { name = hd184r_targets target = scope:hd184r_cand_duchy } }
				}
				target_is_de_jure_liege_or_above = title:h_china

		}
	}

# Scope：名单人物（scope:hd184r_actor）。有进攻名额时，先目标州、后州外，挑一个郡开战。
hd184r_director_declare_one_effect = {
	if = {
		limit = { hd184r_director_war_slot_trigger = yes }
		clear_saved_scope = hd184r_target_duchy
		clear_saved_scope = hd184r_defender
		random_in_list = {
			variable = hd184r_scand
			limit = { hd184r_duchy_has_valid_defender_trigger = yes }
			save_scope_as = hd184r_target_duchy
		}
		if = {
			limit = { NOT = { exists = scope:hd184r_target_duchy } }
			random_in_list = {
				variable = hd184r_ocand
				limit = { hd184r_duchy_has_valid_defender_trigger = yes }
				save_scope_as = hd184r_target_duchy
			}
		}
		if = {
			limit = { exists = scope:hd184r_target_duchy }
			scope:hd184r_target_duchy = {
				random_de_jure_county = {
					limit = { hd184r_county_valid_front_trigger = yes }
					holder = { top_liege = { save_scope_as = hd184r_defender } }
				}
			}
		}
		# 2026-10-23 宣战前复核：保存作用域会随脉冲的 trigger_event 带到下一季、下一个人，
		# 曾让别人的目标（辽东、武都等）漏给本人。只认本人本季候选里、仍合法的郡与守方。
		if = {
			limit = {
				exists = scope:hd184r_target_duchy
				exists = scope:hd184r_defender
				OR = {
					is_target_in_variable_list = { name = hd184r_scand target = scope:hd184r_target_duchy }
					is_target_in_variable_list = { name = hd184r_ocand target = scope:hd184r_target_duchy }
				}
				scope:hd184r_target_duchy = {
					any_de_jure_county = {
						hd184r_county_valid_front_trigger = yes
						holder.top_liege ?= scope:hd184r_defender
					}
				}
			}
			# 宣战前临时解除和平锁（同天意青睐引擎），宣战后恢复
			if = {
				limit = { has_character_flag = sworn_peace }
				remove_character_flag = sworn_peace
				add_character_flag = { flag = hd184r_restore_sworn_peace days = 1 }
			}
			hd_tb_arm_effect = { KIND = historical }
			start_war = {
				cb = hd_dc_tunbing_cb
				target = scope:hd184r_defender
				target_title = scope:hd184r_target_duchy
			}
			if = {
				limit = { has_character_flag = hd184r_restore_sworn_peace }
				remove_character_flag = hd184r_restore_sworn_peace
				add_character_flag = sworn_peace
			}
			if = {
				limit = {
					any_character_war = {
						primary_attacker = scope:hd184r_actor
						primary_defender = scope:hd184r_defender
					}
				}
				debug_log = "HD184R P4 WAR_DECLARED"
				debug_log_scopes = yes
			}
			else = {
				debug_log = "HD184R P4 WAR_START_FAILED"
				debug_log_scopes = yes
			}
		}
		else_if = {
			limit = { exists = scope:hd184r_target_duchy }
			debug_log = "HD184R P4 STALE_TARGET_SKIPPED"
		}
	}
}
