# Scope：王朝。开局静默预填，已有就不动
hd_hanfl_dewu_prefill_effect = {
	if = {
		limit = { NOT = { has_variable = hd_hanfl_dewu } }
		set_variable = { name = hd_hanfl_dewu value = flag:$DEWU$ }
		add_dynasty_modifier = { modifier = hd_hanfl_dewu_$DEWU$_modifier }
		# 天子王朝的德运另记一份，供下一位天子（受禅时可能没有前持有人）取前朝德运
		if = {
			limit = {
				exists = title:h_china.holder.dynasty
				title:h_china.holder.dynasty = this
			}
			set_global_variable = { name = hd_hanfl_last_tianzi_dewu value = flag:$DEWU$ }
		}
	}
}

# Scope：角色（hd_hanfl.1301 选项）。$DEWU$ = 所选；$PREV$ = 按五行相生，承它就是正（火生土、土生金、金生水、水生木、木生火）
hd_hanfl_dewu_choose_effect = {
	hd_hanfl_dewu_set_effect = { DEWU = $DEWU$ }
	if = {
		limit = { NOT = { exists = scope:hd_hanfl_dewu_prev } }
		add_legitimacy = minor_legitimacy_gain
	}
	else_if = {
		limit = { scope:hd_hanfl_dewu_prev = flag:$PREV$ }
		custom_tooltip = hd_hanfl_dewu_right_tt
		add_legitimacy = medium_legitimacy_gain
	}
	else = {
		custom_tooltip = hd_hanfl_dewu_wrong_tt
		add_prestige = minor_prestige_loss
	}
	debug_log = "HD_HANFL DEWU $DEWU$"
}

# root = title:h_china。每 90 天一次，过了本剧本最后截止日或关了开关就不再排
hd_hanfl_timeline_pulse_effect = {
	if = {
		limit = { hd_history_events_enabled_trigger = yes }
		if = {
			limit = { game_start_date = 22.11.13 }
			hd_hanfl_tl_22_effect = yes
		}
		else_if = {
			limit = { game_start_date = 89.1.30 }
			hd_hanfl_tl_89_effect = yes
		}
		else_if = {
			limit = { game_start_date = 190.1.1 }
			hd_hanfl_tl_190_effect = yes
		}
		else_if = {
			limit = { game_start_date = 218.1.1 }
			hd_hanfl_tl_218_effect = yes
		}
		if = {
			limit = {
				OR = {
					AND = { game_start_date = 22.11.13 current_date < 30.1.1 }
					AND = { game_start_date = 89.1.30 current_date < 140.1.1 }
					AND = { game_start_date = 190.1.1 current_date < 215.1.1 }
					AND = { game_start_date = 218.1.1 current_date < 230.1.1 }
				}
			}
			title:h_china = { trigger_event = { on_action = hd_hanfl_timeline_pulse days = 90 } }
		}
		else = {
			debug_log = "HD_HANFL TL PULSE END"
		}
	}
	else = {
		debug_log = "HD_HANFL TL PULSE STOP RULE_OFF"
	}
}
