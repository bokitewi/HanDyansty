# NEG-14, first verified source: court.6060.c's actual drinking conduct.
# Additional native conduct sources retain their own effects and eligibility.
# Only the legal adult substance-trait opportunity receives our multiplier;
# all other cases keep the native chance, including outside-region gameplay.
hd_dc_prepare_substance_trait_roll_effect = {
	save_scope_value_as = { name = hd_dc_substance_chance value = $BASE$ }
	save_scope_value_as = { name = hd_dc_substance_trait value = flag:$TRAIT$ }
	if = {
		limit = {
			is_alive = yes
			exists = situation:dynastic_cycle
			any_character_situation = { this = situation:dynastic_cycle }
			OR = {
				AND = { scope:hd_dc_substance_trait = flag:drunkard can_be_drunkard = yes }
				AND = { scope:hd_dc_substance_trait = flag:hashishiyah can_be_hashishiyah = yes }
			}
		}
		save_scope_value_as = { name = hd_dc_hazard_base value = scope:hd_dc_substance_chance }
		save_scope_value_as = { name = hd_dc_hazard_multiplier value = hd_dc_social_hazard_multiplier_value }
		save_scope_value_as = { name = hd_dc_substance_chance value = hd_dc_hazard_probability_value }
	}
}

# can_be_drunkard is the installed native adult/trait/faith eligibility contract.
# This is a chance of the existing nonhereditary trait, never an AI option weight.
hd_dc_roll_drunkard_effect = {
	if = {
		limit = { can_be_drunkard = yes }
		if = {
			limit = { exists = situation:dynastic_cycle any_character_situation = { this = situation:dynastic_cycle } }
			save_scope_as = hd_dc_trait_actor
			save_scope_value_as = { name = hd_dc_hazard_base value = $BASE$ }
			save_scope_value_as = { name = hd_dc_hazard_multiplier value = hd_dc_social_hazard_multiplier_value }
			save_scope_value_as = { name = hd_dc_trait_chance value = hd_dc_hazard_probability_value }
			hidden_effect = {
				hd_dc_create_record_effect = { KIND = trait_roll }
				scope:hd_dc_record = {
					save_scope_as = hd_dc_trait_receipt
					set_variable = { name = hd_dc_actor value = scope:hd_dc_trait_actor }
					set_variable = { name = hd_dc_trait value = flag:drunkard }
					set_variable = { name = hd_dc_trait_source value = flag:$SOURCE$ }
					set_variable = { name = hd_dc_trait_base_chance value = scope:hd_dc_hazard_base }
					set_variable = { name = hd_dc_trait_multiplier value = scope:hd_dc_hazard_multiplier }
					set_variable = { name = hd_dc_trait_actual_chance value = scope:hd_dc_trait_chance }
				}
			}
			random = {
				chance = scope:hd_dc_trait_chance
				add_trait = drunkard
			}
			hidden_effect = {
				scope:hd_dc_trait_receipt = {
					if = {
						limit = { scope:hd_dc_trait_actor = { has_trait = drunkard } }
						set_variable = { name = hd_dc_trait_result value = flag:granted }
					}
					else = { set_variable = { name = hd_dc_trait_result value = flag:not_granted } }
					set_variable = { name = hd_dc_trait_checked_date value = current_date }
				}
			}
		}
		else = {
			random = { chance = $BASE$ add_trait = drunkard }
		}
	}
}
