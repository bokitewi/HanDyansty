hd_create_wolong_cao_cao_armor_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = hd_wolong_cao_cao_armor_name
		description = hd_wolong_cao_cao_armor_description
		type = armor_lamellar
		template = hd_wolong_armor_template
		visuals = hd_wolong_cao_cao_armor
		wealth = scope:wealth
		quality = scope:quality
		max_durability = 100
		history = { type = created_before_history }
		modifier = artifact_prowess_1_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}
	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = hd_wolong_cao_cao_armor
		equip_artifact_to_owner_replace = yes
	}
}
