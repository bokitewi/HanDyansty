hd_smy_check_s1_effect = {
	# 四友：开局（入场）后 60 天，曹丕在世且为曹操继承人或已为君主
	if = {
		limit = {
			NOT = { exists = global_var:hd_smy_s1_done }
			NOT = { exists = global_var:hd_smy_s1_wait }
			global_var:hd_zs_p_cao_pi ?= {
				is_alive = yes
				OR = {
					is_landed = yes
					AND = {
						exists = character:cao_cao
						character:cao_cao = { is_alive = yes }
						is_primary_heir_of = character:cao_cao
					}
				}
			}
		}
		set_global_variable = hd_smy_s1_done
		scope:hd_smy_y = {
			if = {
				limit = { NOT = { has_relation_friend = global_var:hd_zs_p_cao_pi } }
				set_relation_friend = { target = global_var:hd_zs_p_cao_pi reason = friend_introduced_heir }
			}
			change_merit = 100
			if = {
				limit = { is_ai = no }
				trigger_event = hd_smy.0010
			}
		}
		global_var:hd_zs_p_yingchuan_chen_004 ?= {
			if = {
				limit = { is_alive = yes hd_zs_serves_cao_trigger = yes NOT = { has_relation_friend = global_var:hd_zs_p_cao_pi } }
				set_relation_friend = { target = global_var:hd_zs_p_cao_pi reason = friend_introduced_heir }
			}
		}
		debug_log = "HD_SMY S1 SIYOU"
	}
	# 嗣位：曹操已死、曹丕有地
	if = {
		limit = {
			NOT = { exists = global_var:hd_smy_s1b_done }
			character:cao_cao ?= { is_alive = no }
			global_var:hd_zs_p_cao_pi ?= { is_alive = yes is_landed = yes }
		}
		set_global_variable = hd_smy_s1b_done
		scope:hd_smy_y = {
			change_merit = 150
			if = {
				limit = { is_ai = no }
				trigger_event = hd_smy.0020
			}
		}
		global_var:hd_zs_p_cao_pi = {
			if = {
				limit = {
					is_ai = yes
					scope:hd_smy_y = { top_liege = global_var:hd_zs_p_cao_pi.top_liege NOT = { is_councillor_of = global_var:hd_zs_p_cao_pi } }
				}
				scope:hd_smy_y = { add_character_flag = hd_cp_merit_exempt }
				assign_councillor_type = { type = councillor_steward target = scope:hd_smy_y }
			}
		}
		debug_log = "HD_SMY S1 SIWEI"
	}
	# 受禅：曹丕为天子，授尚书令；抚军计时 3 年
	if = {
		limit = {
			NOT = { exists = global_var:hd_smy_s1c_done }
			global_var:hd_zs_p_cao_pi ?= { is_alive = yes has_title = title:h_china }
			scope:hd_smy_y = { hd_zs_in_wei_trigger = yes }
		}
		set_global_variable = hd_smy_s1c_done
		set_global_variable = { name = hd_smy_fujun_wait value = yes days = 1095 }
		global_var:hd_zs_p_cao_pi = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_smy.0030
			}
			else = { hd_smy_shangshu_effect = yes }
		}
	}
}

# 授尚书令（先 granted 交给本人，再走 WJ 拜官；照 hd218_grant_chengxiang_effect，免 appointment 辞地）。需 scope:hd_zs_wei、scope:hd_smy_y
hd_smy_shangshu_effect = {
	scope:hd_smy_y = {
		if = {
			limit = { NOT = { has_title = title:e_minister_of_personnel } }
			add_character_flag = hd_cp_merit_exempt
			create_title_and_vassal_change = { type = granted save_scope_as = hd_smy_ss_change add_claim_on_loss = no }
			title:e_minister_of_personnel = {
				change_title_holder = { holder = scope:hd_smy_y change = scope:hd_smy_ss_change }
			}
			resolve_title_and_vassal_change = scope:hd_smy_ss_change
			scope:hd_zs_wei = { save_scope_as = councillor_liege }
			got_minister_position_effect = { MINISTER_TITLE = e_minister_of_personnel MINISTER_POSITION = minister_personnel }
			clear_saved_scope = councillor_liege
		}
		if = {
			limit = { has_title = title:e_minister_of_personnel }
			debug_log = "HD_SMY S1 SHANGSHU OK"
		}
		else = { debug_log = "HD_SMY S1 SHANGSHU FAIL" }
	}
}

# 授河内郡（温县在内），作天子直属封臣；不夺玩家的县。Scope：司马懿；需 scope:hd_zs_wei
hd_smy_grant_henei_effect = {
	save_scope_as = hd_smy_y
	if = {
		limit = { NOT = { has_title = title:d_henei } }
		create_title_and_vassal_change = { type = granted save_scope_as = hd_smy_henei_change add_claim_on_loss = no }
		title:d_henei = {
			every_de_jure_county = {
				limit = {
					holder ?= { top_liege = scope:hd_zs_wei is_ai = yes NOT = { this = scope:hd_smy_y } }
				}
				change_title_holder = { holder = scope:hd_smy_y change = scope:hd_smy_henei_change take_baronies = yes }
			}
			if = {
				limit = {
					OR = {
						NOT = { exists = holder }
						holder ?= { is_ai = yes top_liege = scope:hd_zs_wei }
					}
				}
				change_title_holder = { holder = scope:hd_smy_y change = scope:hd_smy_henei_change }
			}
		}
		resolve_title_and_vassal_change = scope:hd_smy_henei_change
	}
	if = {
		limit = { is_landed = yes NOT = { liege ?= scope:hd_zs_wei } NOT = { this = scope:hd_zs_wei } }
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd_smy_fealty add_claim_on_loss = no }
		change_liege = { liege = scope:hd_zs_wei change = scope:hd_smy_fealty }
		resolve_title_and_vassal_change = scope:hd_smy_fealty
	}
	if = {
		limit = { has_title = title:c_uuii_p_1077 }
		set_realm_capital = title:c_uuii_p_1077
	}
	if = {
		limit = { is_landed = yes }
		debug_log = "HD_SMY HENEI OK"
	}
	else = { debug_log = "HD_SMY HENEI FAIL" }
}

# 坚壁拒守、巾帼之辱、挡住北伐
hd_smy_check_s4_sub_effect = {
	# 坚壁拒守：司马懿参战的蜀魏战争打满 120 天（每场一次）
	if = {
		limit = {
			exists = scope:hd_zs_shu
			scope:hd_zs_shu = {
				any_character_war = {
					hd_zs_shu_wei_war_trigger = yes
					is_participant = scope:hd_smy_y
					war_days >= 120
					NOT = { has_variable = hd_smy_jianbi_asked }
				}
			}
		}
		scope:hd_zs_shu = {
			random_character_war = {
				limit = {
					hd_zs_shu_wei_war_trigger = yes
					is_participant = scope:hd_smy_y
					war_days >= 120
					NOT = { has_variable = hd_smy_jianbi_asked }
				}
				set_variable = hd_smy_jianbi_asked
			}
		}
		set_global_variable = hd_hev_smy_06
		scope:hd_smy_y = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_smy.0061
			}
			else = { hd_smy_jianbi_effect = yes }
		}
	}
	# 巾帼之辱：坚守 30 天后，诸葛亮在世并主政（一次）
	if = {
		limit = {
			NOT = { exists = global_var:hd_smy_jinguo_done }
			exists = global_var:hd_smy_jianbi_chosen
			NOT = { exists = global_var:hd_smy_jinguo_wait }
			hd_zgl_regent_trigger = yes
		}
		set_global_variable = hd_smy_jinguo_done
		scope:hd_smy_y = {
			if = {
				limit = { is_ai = no }
				trigger_event = hd_smy.0062
			}
			else = { hd_smy_jinguo_accept_effect = yes }
		}
	}
	# 挡住北伐：北伐开始时记下蜀持魏县数；北伐全部结束时未增加，都督司马懿得赏
	if = {
		limit = { hd_zs_beifa_active_trigger = yes }
		if = {
			limit = { NOT = { exists = global_var:hd_smy_beifa_watch } }
			set_global_variable = hd_smy_beifa_watch
			hd_zs_count_held_effect = yes
			set_global_variable = { name = hd_smy_held_at_start value = global_var:hd_zs_held_now }
		}
	}
	else_if = {
		limit = { exists = global_var:hd_smy_beifa_watch }
		remove_global_variable = hd_smy_beifa_watch
		hd_zs_count_held_effect = yes
		if = {
			limit = {
				global_var:hd_zs_held_now <= global_var:hd_smy_held_at_start
				scope:hd_smy_y = { has_character_modifier = hd_smy_dudu_modifier }
			}
			scope:hd_smy_y = {
				change_merit = hd_smy_repel_merit_value
				change_influence = hd_smy_repel_influence_value
				add_prestige = 200
			}
			hd_zs_party_nudge_effect = { WHO = scope:hd_smy_y LORD = scope:hd_zs_wei POWER = 10 }
			debug_log = "HD_SMY S4 BEIFA_REPELLED"
		}
	}
}
