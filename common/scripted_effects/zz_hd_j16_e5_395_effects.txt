# 封臣出征（照 hd_nbc_vassal_war_effect；06 §2.1「先按 A1 §4.1 改成解析写法」：人物改传 scope，不再写 character:$WHO$）。
# $WHO$ 先自宣；不成且非独立时由其最高领主宣战，$WHO$ 有地且不在战中则加入攻方。成功判据 exists = scope:hd_nbc_war_<KIND>；实际主攻方存为 scope:hd_j16_vw_att
hd_j16_395s_vassal_war_effect = {
	if = {
		limit = { exists = scope:hd_j16_vw_who }
		clear_saved_scope = hd_j16_vw_who
	}
	if = {
		limit = { exists = scope:hd_j16_vw_def }
		clear_saved_scope = hd_j16_vw_def
	}
	if = {
		limit = { exists = scope:hd_j16_vw_top }
		clear_saved_scope = hd_j16_vw_top
	}
	if = {
		limit = { exists = scope:hd_j16_vw_att }
		clear_saved_scope = hd_j16_vw_att
	}
	$WHO$ = { save_scope_as = hd_j16_vw_who }
	$DEF$ = { save_scope_as = hd_j16_vw_def }
	hd_tb_hist_start_war_effect = { ATTACKER = scope:hd_j16_vw_who DEFENDER = scope:hd_j16_vw_def TARGET = $TARGET$ KIND = $KIND$ }	# 灭国场次走吞并 CB（10-28）
	if = {
		limit = { exists = scope:hd_nbc_war_$KIND$ }
		scope:hd_j16_vw_who = { save_scope_as = hd_j16_vw_att }
	}
	else_if = {
		limit = {
			scope:hd_j16_vw_who = { is_independent_ruler = no }
			NOT = { scope:hd_j16_vw_who.top_liege = scope:hd_j16_vw_def }
		}
		scope:hd_j16_vw_who.top_liege = { save_scope_as = hd_j16_vw_top }
		hd_tb_hist_start_war_effect = { ATTACKER = scope:hd_j16_vw_top DEFENDER = scope:hd_j16_vw_def TARGET = $TARGET$ KIND = $KIND$ }
		if = {
			limit = { exists = scope:hd_nbc_war_$KIND$ }
			scope:hd_j16_vw_top = { save_scope_as = hd_j16_vw_att }
			scope:hd_nbc_war_$KIND$ = {
				if = {
					limit = {
						scope:hd_j16_vw_who = {
							is_landed = yes
							is_at_war = no
						}
					}
					add_attacker = scope:hd_j16_vw_who
				}
			}
		}
	}
}

					hd_j16_370b_grantable_trigger = { DUCHY = title:d_shujun TO = scope:hd_j16_ps_zhu }

		}
		else = {

			debug_log = "HD_J16 e5_j76 FIRE result_lost"

			limit = { has_global_variable = hd_nbc_pingshu_active }

		}
	}
