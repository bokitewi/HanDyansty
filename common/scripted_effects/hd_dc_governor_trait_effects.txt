# Recheck the original offered candidate when the player finally answers.
# Removing an invalid offered flag cannot select a replacement or award a trait.
hd_dc_count_governor_trait_candidate_effect = {
	if = {
		limit = { has_character_flag = ep3_governor_yearly_3100_$TRAIT$ }
		if = {
			limit = { hd_dc_governor_trait_candidate_trigger = { TRAIT = $TRAIT$ } }
			save_scope_value_as = { name = hd_dc_governor_bad_weight value = { value = scope:hd_dc_governor_bad_weight add = $WEIGHT$ } }
		}
		else = { remove_character_flag = ep3_governor_yearly_3100_$TRAIT$ }
	}
}

# One native outcome random_list, not an AI option weight. Its negative group
# has dynamic weight B and its stress-only branch has weight 45. Keep every
# negative branch and its relative share, and solve the complementary weight
# for p'=hazard(100*B/(45+B),F). The native duplicate hashish slot counts twice.
# All outcome weights use 1000 units per native point, preserving small deltas.
hd_dc_prepare_governor_trait_roll_effect = {
	save_scope_value_as = { name = hd_dc_governor_relief_factor value = 1000 }
	if = {
		limit = { is_alive = yes is_adult = yes exists = situation:dynastic_cycle any_character_situation = { this = situation:dynastic_cycle } }
		save_scope_value_as = { name = hd_dc_governor_bad_weight value = 0 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = depressed WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = reclusive WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = irritable WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = drunkard WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = hashishiyah WEIGHT = 10 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = comfort_eater WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = contrite WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = improvident WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = rakish WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = profligate WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = flagellant WEIGHT = 5 }
		hd_dc_count_governor_trait_candidate_effect = { TRAIT = inappetetic WEIGHT = 5 }
		if = {
			limit = { scope:hd_dc_governor_bad_weight > 0 }
			save_scope_value_as = { name = hd_dc_hazard_base value = { value = scope:hd_dc_governor_bad_weight multiply = 100 divide = { value = 45 add = scope:hd_dc_governor_bad_weight } } }
			save_scope_value_as = { name = hd_dc_hazard_multiplier value = hd_dc_social_hazard_multiplier_value }
			save_scope_value_as = { name = hd_dc_governor_trait_chance value = hd_dc_hazard_probability_value }
			save_scope_value_as = { name = hd_dc_governor_relief_factor value = { value = 100 subtract = scope:hd_dc_governor_trait_chance multiply = scope:hd_dc_governor_bad_weight multiply = 1000 divide = scope:hd_dc_governor_trait_chance divide = 45 } }
		}
	}
}
