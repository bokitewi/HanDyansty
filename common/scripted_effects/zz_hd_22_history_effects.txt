# 被杀。电脑角色直接死；玩家角色不杀，只记日志
hd_22_die_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ }
			debug_log = "HD_22 DIED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_22 DIE_SKIPPED_PLAYER $WHO$"
		}
	}
}
hd_22_kill_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ killer = $KILLER$ }
			debug_log = "HD_22 KILLED $WHO$"
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD_22 KILL_SKIPPED_PLAYER $WHO$"
		}
	}
}

# $NEW$ 接收 $OLD$ 的全部实地头衔（含 h_china）与封臣（照 hd_370_take_realm_effect）
hd_22_take_realm_effect = {
	$NEW$ = { save_scope_as = hd_22_new_ruler }
	create_title_and_vassal_change = { type = granted save_scope_as = hd_22_take_change add_claim_on_loss = no }
	$OLD$ = {
		every_held_title = {
			limit = {
				is_noble_family_title = no
				tgp_is_minister_title = no
				is_landless_type_title = no
				tier >= tier_county
			}
			change_title_holder = {
				holder = scope:hd_22_new_ruler
				change = scope:hd_22_take_change
				take_baronies = yes
			}
		}
		every_vassal = {
			limit = { NOT = { this = scope:hd_22_new_ruler } }
			change_liege = { liege = scope:hd_22_new_ruler change = scope:hd_22_take_change }
		}
	}
	resolve_title_and_vassal_change = scope:hd_22_take_change
}

# 把郡头衔 $DUCHY$ 与其持有者手里的本郡各县交给 $TO$（scope）；$KEEP$ 是不交出的县
hd_22_give_duchy_effect = {
	if = {
		limit = { exists = $DUCHY$.holder }
		$DUCHY$.holder = { save_scope_as = hd_22_old_duke }
		create_title_and_vassal_change = { type = granted save_scope_as = hd_22_duchy_change add_claim_on_loss = no }
		$DUCHY$ = {
			change_title_holder = { holder = $TO$ change = scope:hd_22_duchy_change }
			every_de_jure_county = {
				limit = {
					holder ?= scope:hd_22_old_duke
					NOT = { this = $KEEP$ }
				}
				change_title_holder = { holder = $TO$ change = scope:hd_22_duchy_change take_baronies = yes }
			}
		}
		resolve_title_and_vassal_change = scope:hd_22_duchy_change
	}
}

# 边境战争：$DEFENDER$ 的最高宗主没有边境战争法时补上（照 218 汉中之战）

	}
}

hd_22_outcome_deaths_effect = {
	# 刘歆、王涉、董忠：地皇四年七月谋劫王莽事泄（刘歆、王涉自杀，董忠被杀）。前提：王莽仍持 h_china
	if = {
		limit = {
			current_date >= 23.8.1
			NOT = { exists = global_var:hd_22_conspiracy_done }
		}
		set_global_variable = hd_22_conspiracy_done
		if = {
			limit = { title:h_china.holder ?= character:wang_mang }
			hd_22_die_effect = { WHO = liu_xin_2 REASON = death_suicide }
			hd_22_die_effect = { WHO = wang_she_2 REASON = death_suicide }
			hd_22_kill_effect = { WHO = dong_zhong REASON = death_execution KILLER = character:wang_mang }
		}
		else = { debug_log = "HD_22 CONSPIRACY SKIPPED_NO_WANGMANG" }
	}
	# 刘縯：昆阳之后为更始所杀。前提：更始已立、刘縯是刘玄的封臣、昆阳汉军胜；电脑刘縯才死

		limit = {
			NOT = { exists = global_var:hd_22_bosheng_done }
			current_date >= 23.7.7
			OR = {
				exists = global_var:hd_22_kunyang_done
				current_date >= 24.1.1
			}
		}
		set_global_variable = hd_22_bosheng_done
		if = {
			limit = {
				exists = global_var:hd_22_kunyang_han_won
				character:WJ_history_gengshi ?= { is_alive = yes has_title = title:e_hd_western_han }
				character:liu_yan ?= { is_alive = yes top_liege = character:WJ_history_gengshi NOT = { this = character:WJ_history_gengshi } }
			}
			hd_22_kill_effect = { WHO = liu_yan REASON = death_execution KILLER = character:WJ_history_gengshi }
		}
		else = { debug_log = "HD_22 BOSHENG_DEATH SKIPPED" }
	}
}

			NOT = { exists = global_var:hd_22_s1_transferring }

	}

		}
