hd_character_starting_resource_floors_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { has_character_flag = hd_starting_resource_floors_initialized }
			}
			if = {
				limit = { prestige_level < $FAME$ }
				add_prestige_level = { value = $FAME$ subtract = prestige_level }
			}
			if = {
				limit = { prestige < $PRESTIGE$ }
				add_prestige_no_experience = { value = $PRESTIGE$ subtract = prestige }
			}
			if = {
				limit = { government_has_flag = government_has_influence }
				if = {
					limit = { influence < $INFLUENCE$ }
					change_influence_no_experience = { value = $INFLUENCE$ subtract = influence }
				}
			}
			else = { debug_log = "HD START RESOURCES INFLUENCE_UNSUPPORTED $WHO$" }
			add_character_flag = hd_starting_resource_floors_initialized
		}
	}
}
