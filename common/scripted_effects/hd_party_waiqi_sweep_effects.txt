# 天朝循环·朋党：外戚扫描（2026-10-08 用户指定；手工维护，由生成的 hd_party_refresh_waiqi_list_effect 末尾调用）
# 外戚 = 朝廷君主的配偶、母亲、侧室、子女配偶所属宗族（dynasty）的全部成员，
#        以及母亲、正妻、嫔妃的兄弟姐妹（2026-10-10，含同父异母/同母异父，如何苗）。
# 刷新时机随后族列表：开局、每年、婚姻发生时。
#
# 1. 入组：后族成员只要 成年、在本国（最高领主是本朝廷君主）或离宫在野且原属本朝、非天子、非独立统治者、非宦官、非他党现任领袖，
#    还不是局势参与者的加为手动参与者，并一律改为外戚（覆盖原有的组别锁定 movement_member），随即重新归组。
#    同时记下所属朝廷 var:hd_party_waiqi_court：离宫在野（游荡或做客）后仍算本朝外戚（2026-10-10）。
# 2. 失势：本朝廷的外戚（在本国或离宫在野）已不属后族的，清除组别锁定与所属朝廷并重新归组；
#    已无资格留在局势中的手动参与者直接移出。外戚党现任领袖不动。
# 候选人判定见 hd_party_identity_triggers.txt 的 hd_party_waiqi_sweep_candidate_trigger。

# Scope：朝廷君主
hd_party_waiqi_sweep_effect = {
	save_scope_as = hd_party_wq_court

	# 1. 后族成员入外戚
	every_in_list = {
		variable = hd_party_waiqi_dynasties
		every_dynasty_member = {
			limit = { hd_party_waiqi_sweep_candidate_trigger = yes }
			hd_party_waiqi_join_effect = yes
		}
	}
	every_in_list = {
		variable = hd_party_waiqi_kin
		limit = { hd_party_waiqi_sweep_candidate_trigger = yes }
		hd_party_waiqi_join_effect = yes
	}

	# 2. 失势外戚解除（先收集再处理，避免一边遍历组一边改组成员）
	situation:dynastic_cycle ?= {
		situation_participant_group:pro_hegemon_movement ?= {
			every_situation_group_participant = {
				limit = {
					var:movement_member ?= flag:pro_hegemon
					OR = {
						top_liege ?= scope:hd_party_wq_court
						var:hd_party_waiqi_court ?= scope:hd_party_wq_court
					}
					hd_party_is_waiqi_trigger = no
					NOT = { scope:hd_party_wq_court.var:hd_party_waiqi_leader ?= this }
					NOT = { top_participant_group:dynastic_cycle.var:movement_leader ?= this }
				}
				add_to_list = hd_party_wq_lapsed
			}
		}
	}
	every_in_list = {
		list = hd_party_wq_lapsed
		remove_variable = movement_member
		remove_variable = hd_party_waiqi_court
		if = {
			limit = {
				has_variable = hd_party_manual
				NOT = { hd_party_manual_participant_valid_trigger = yes }
			}
			remove_variable = hd_party_manual
			situation:dynastic_cycle ?= { remove_manual_participant = prev }
		}
		else = {
			recalculate_participant_group = situation:dynastic_cycle
		}
	}
}

# Scope：后族成员；scope:hd_party_wq_court = 朝廷君主。记下所属朝廷，加入局势并归外戚
hd_party_waiqi_join_effect = {
	set_variable = { name = hd_party_waiqi_court value = scope:hd_party_wq_court }
	if = {
		limit = { NOT = { any_character_situation = { situation_type = dynastic_cycle } } }
		set_variable = hd_party_manual
		situation:dynastic_cycle ?= { add_manual_participant = prev }
	}
	if = {
		limit = {
			OR = {
				NOT = { var:movement_member ?= flag:pro_hegemon }
				NOT = { top_participant_group:dynastic_cycle ?= { participant_group_type = pro_hegemon_movement } }
			}
		}
		set_variable = { name = movement_member value = flag:pro_hegemon }
		recalculate_participant_group = situation:dynastic_cycle
	}
}
