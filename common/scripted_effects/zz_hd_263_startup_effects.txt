# 副本：在世子女先改挂到保留的那一份，再静默去世
hd263_dup_effect = {
	character:$DROP$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = no
				character:$KEEP$ ?= { is_alive = yes }
			}
			if = {
				limit = { is_female = no }
				every_child = {
					limit = { is_alive = yes NOT = { father ?= character:$KEEP$ } }
					set_father = character:$KEEP$
				}
			}
			else = {
				every_child = {
					limit = { is_alive = yes NOT = { mother ?= character:$KEEP$ } }
					set_mother = character:$KEEP$
				}
			}
			death = { death_reason = death_vanished }
			debug_log = "HD263 DUP_REMOVED $DROP$"
		}
	}
}

hd263_frontier_chief_effect = {
	if = {
		limit = {
			title:$DUCHY$.holder ?= { OR = { this = character:cao_huan this = character:sun_xiu } }
		}
		title:$DUCHY$.holder = { save_temporary_scope_as = hd263_placeholder }
		title:$DUCHY$ = { save_temporary_scope_as = seceding_county }
		create_character = {
			location = title:$DUCHY$.title_capital_county.title_province
			template = post_migration_minor_warlord_template
			dynasty = generate
			save_temporary_scope_as = hd263_chief_new
		}
		create_title_and_vassal_change = { type = usurped save_scope_as = hd263_chief_change add_claim_on_loss = no }
		title:$DUCHY$ = {
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_county holder ?= scope:hd263_placeholder NOT = { this = title:c_uuii_p_3777 } }   # 险渎归魏，郡守生成后交辽东
				change_title_holder = { holder = scope:hd263_chief_new change = scope:hd263_chief_change take_baronies = yes }
			}
			change_title_holder = { holder = scope:hd263_chief_new change = scope:hd263_chief_change }
		}
		resolve_title_and_vassal_change = scope:hd263_chief_change
		create_title_and_vassal_change = { type = independency save_scope_as = hd263_chief_free_change add_claim_on_loss = no }
		scope:hd263_chief_new = { becomes_independent = { change = scope:hd263_chief_free_change } }
		resolve_title_and_vassal_change = scope:hd263_chief_free_change
		scope:hd263_chief_new = {
			if = {
				limit = { NOT = { has_government = $GOV$ } }
				change_government = $GOV$
			}
			if = {
				limit = { is_independent_ruler = yes has_title = title:$DUCHY$ }
				debug_log = "HD263 FRONTIER_CHIEF OK $DUCHY$"
			}
			else = { debug_log = "HD263 FRONTIER_CHIEF BAD $DUCHY$" }
		}
	}
	else = { debug_log = "HD263 FRONTIER_CHIEF SKIPPED $DUCHY$" }
}

# 挟天子：司马昭为权臣兼丞相，曹奂为傀儡，京城洛阳（照 hd253_regency_effect 换人；辅政锁先设，免得授丞相时封臣被收走）。Scope：司马昭
hd263_regency_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			character:cao_huan ?= { is_alive = yes has_title = title:h_china }
			liege ?= character:cao_huan
		}
		character:cao_huan = {
			set_location = {
				location = title:c_luoyang.title_province
				stick_to_location = yes
			}
		}
		# 玩家扮演时功名补到帝国级预期，否则丞相授不上（218 实测）
		if = {
			limit = {
				is_ai = no
				merit_level < merit_level_expected_empire_value
			}
			while = {
				limit = { merit_level < merit_level_expected_empire_value }
				change_merit_level = 1
			}
		}
		title:h_china = { set_variable = { name = hd_dc_ten_qjc_regent value = character:si_ma_zhao } }
		hd218_grant_chengxiang_effect = yes
		WJ_install_historical_secretariat_effect = { OFFICE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang LAW = WJ_grand_chengxiang_law }
		hd_court_install_regency_effect = yes
		hd_court_set_stance_effect = { STANCE = moni }
		character:cao_huan = {
			if = {
				limit = { has_diarchy_type = grand_secretariat diarch ?= character:si_ma_zhao }
				set_diarchy_swing = 100
			}
		}
		# 权臣任期到 266.3.1（S8 禅代之后），本 mod 的废帝决议不会抢先
		add_character_flag = { flag = hd_court_regent_tenure days = 893 }
		if = {
			limit = { has_title = title:e_WJ_minister_cheng_xiang }
			debug_log = "HD263 REGENCY SI_MA_ZHAO OK"
		}
		else = { debug_log = "HD263 REGENCY SI_MA_ZHAO NO_CHENGXIANG" }
	}
	else = { debug_log = "HD263 REGENCY SKIPPED" }
}
