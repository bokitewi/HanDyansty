# 天朝循环·时代特色 Step 13：紧张期、新朝征服期的危机抽签（AUHCO 移植；K1 每 4 年，K2 权重）
# 危机本体复用崩溃期效果（hd_collapse_*），因此 hd_collapse_emperor_still_valid_trigger 已放宽到这两期

# Scope：dynastic_cycle 局势（紧张 / 新朝征服期年度）
hd_dc_kit_tension_yearly_effect = {
	change_variable = { name = hd_dc_kit_tension_crisis_years add = 1 }
	if = {
		limit = { var:hd_dc_kit_tension_crisis_years >= hd_dc_tension_crisis_interval_years }
		set_variable = { name = hd_dc_kit_tension_crisis_years value = 0 }
		title:h_china.holder ?= {
			trigger_event = { id = hd_collapse.0005 days = { 1 60 } }
		}
	}
}

# Scope：天子
hd_dc_kit_select_tension_crisis_effect = {
	random_list = {
		100 = {	# = hd_dc_crisis_w_disaster	# 天灾
			hd_collapse_spawn_disaster_effect = yes
		}
		100 = {	# = hd_dc_crisis_w_epidemic	# 瘟疫
			hd_dc_kit_spawn_epidemic_effect = yes
		}
		100 = {	# = hd_dc_crisis_w_authority	# 权威危机
			trigger_event = hd_collapse.4001
		}
		100 = {	# = hd_dc_crisis_w_peasant	# 民变
			trigger = { hd_collapse_peasant_target_valid_trigger = yes }
			hd_collapse_start_peasant_uprising_effect = yes
		}
		33 = {	# = hd_dc_crisis_w_independence	# 集体独立
			trigger = { hd_collapse_independence_target_valid_trigger = yes }
			trigger_event = hd_collapse.1001
		}
		33 = {	# = hd_dc_crisis_w_warlord	# 军阀称帝
			trigger = { hd_collapse_warlord_target_valid_trigger = yes }
			hd_collapse_start_warlord_revolt_effect = yes
		}
	}
}

# Scope：天子。在天子领内随机一县爆发大疫
hd_dc_kit_spawn_epidemic_effect = {
	random_realm_county = {
		limit = { target_is_de_jure_liege_or_above = title:h_china }
		random_list = {
			1 = { create_epidemic_outbreak = { type = smallpox intensity = major } }
			1 = { create_epidemic_outbreak = { type = typhus intensity = major } }
			1 = { create_epidemic_outbreak = { type = measles intensity = major } }
			1 = { create_epidemic_outbreak = { type = consumption intensity = major } }
		}
	}
}

# Scope：dynastic_cycle 局势（月度）
hd_dc_kit_monthly_effect = {
	if = {
		limit = { hd_dc_phase_is_luanshi_trigger = yes }
		every_independent_ruler = {
			limit = {
				is_ai = yes
				highest_held_title_tier >= tier_duchy
				hd_dc_in_china_trigger = yes
			}
			random = {
				chance = hd_dc_war_no_treasury_chance
				trigger_event = { id = hd_dc_kit.0001 days = { 1 30 } }
			}
		}
	}
}
