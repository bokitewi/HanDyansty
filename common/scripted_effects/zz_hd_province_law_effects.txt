# 2026-10-31 擅兴律/疆界之制：开局按皇权预设两组法律（docs/郡国制_擅兴律与疆界之制_设计.md §4）

# Scope：角色。已有同档法律时再加会报错，先判。
hd_province_law_set_effect = {
	if = {
		limit = { NOT = { has_realm_law = $WAR$ } }
		add_realm_law_skip_effects = $WAR$
	}
	if = {
		limit = { NOT = { has_realm_law = $BORDER$ } }
		add_realm_law_skip_effects = $BORDER$
	}
}

# Scope：郡国制独立君主（皇权已抽定）。套用各组允许的最严档，并同步到能持有这两组法律的各级封臣。
hd_province_law_startup_preset_effect = {
	if = {
		limit = { has_realm_law = celestial_bureaucracy_3 }
		hd_province_law_set_effect = { WAR = province_war_law_3 BORDER = province_border_law_2 }
		every_vassal_or_below = {
			limit = { hd_province_law_valid_trigger = yes }
			hd_province_law_set_effect = { WAR = province_war_law_3 BORDER = province_border_law_2 }
		}
	}
	else_if = {
		limit = { has_realm_law = celestial_bureaucracy_2 }
		hd_province_law_set_effect = { WAR = province_war_law_2 BORDER = province_border_law_1 }
		every_vassal_or_below = {
			limit = { hd_province_law_valid_trigger = yes }
			hd_province_law_set_effect = { WAR = province_war_law_2 BORDER = province_border_law_1 }
		}
	}
	else_if = {
		limit = { has_realm_law = celestial_bureaucracy_1 }
		hd_province_law_set_effect = { WAR = province_war_law_1 BORDER = province_border_law_0 }
		every_vassal_or_below = {
			limit = { hd_province_law_valid_trigger = yes }
			hd_province_law_set_effect = { WAR = province_war_law_1 BORDER = province_border_law_0 }
		}
	}
	else = {
		hd_province_law_set_effect = { WAR = province_war_law_0 BORDER = province_border_law_0 }
		every_vassal_or_below = {
			limit = { hd_province_law_valid_trigger = yes }
			hd_province_law_set_effect = { WAR = province_war_law_0 BORDER = province_border_law_0 }
		}
	}
}
