# The realm's actual primary title owns the annual lock, including failed
# draws. A succession cannot give that same realm a second annual opportunity.
hd_dc_check_extra_faction_year_effect = {
	if = {
		limit = {
			exists = situation:dynastic_cycle
			is_alive = yes is_landed = yes exists = primary_title
			any_character_situation = { this = situation:dynastic_cycle }
			primary_title = { OR = { NOT = { has_variable = hd_dc_extra_faction_year } NOT = { var:hd_dc_extra_faction_year = current_year } } }
		}
		save_scope_as = hd_dc_extra_faction_target
		primary_title = {
			save_scope_as = hd_dc_extra_realm
			set_variable = { name = hd_dc_extra_faction_year value = current_year }
			set_variable = { name = hd_dc_extra_faction_chance value = 0 }
			set_variable = { name = hd_dc_extra_faction_result value = flag:hd_dc_extra_faction_low_w }
			remove_variable = hd_dc_extra_faction_type
			remove_variable = hd_dc_extra_faction_actor
			remove_variable = hd_dc_extra_faction_county
			remove_variable = hd_dc_extra_faction_claimant
			remove_variable = hd_dc_extra_faction_claimed_title
		}
		if = {
			limit = { situation:dynastic_cycle.var:hd_dc_pressure_raw >= 400 }
			save_scope_value_as = { name = hd_dc_hazard_base value = 10 }
			save_scope_value_as = { name = hd_dc_hazard_multiplier value = hd_dc_faction_hazard_multiplier_value }
			scope:hd_dc_extra_realm = {
				set_variable = { name = hd_dc_extra_faction_chance value = hd_dc_hazard_probability_value }
				set_variable = { name = hd_dc_extra_faction_result value = flag:hd_dc_extra_faction_missed }
			}
			random = {
				chance = scope:hd_dc_extra_realm.var:hd_dc_extra_faction_chance
				hd_dc_select_extra_faction_type_effect = yes
			}
		}
	}
}

# Creation is followed by a real faction readback. No ultimatum, discontent
# boost, forced membership or war result is part of this extra opportunity.
hd_dc_record_extra_faction_effect = {
	scope:hd_dc_extra_faction_target = {
		if = {
			limit = { any_targeting_faction = { faction_is_type = $TYPE$ } }
			scope:hd_dc_extra_realm = {
				set_variable = { name = hd_dc_extra_faction_type value = flag:$TYPE$ }
				set_variable = { name = hd_dc_extra_faction_result value = flag:hd_dc_extra_faction_created }
			}
		}
	}
}
hd_dc_create_extra_character_faction_effect = {
	scope:hd_dc_extra_realm = { set_variable = { name = hd_dc_extra_faction_result value = flag:hd_dc_extra_faction_creation_failed } }
	situation:dynastic_cycle = {
		random_situation_participant = {
			limit = { hd_dc_extra_faction_creator_trigger = { TYPE = $TYPE$ } }
			save_scope_as = hd_dc_extra_creator
			create_faction = { type = $TYPE$ target = scope:hd_dc_extra_faction_target }
			hd_dc_record_extra_faction_effect = { TYPE = $TYPE$ }
			if = {
				limit = { scope:hd_dc_extra_realm.var:hd_dc_extra_faction_result = flag:hd_dc_extra_faction_created }
				scope:hd_dc_extra_realm = { set_variable = { name = hd_dc_extra_faction_actor value = scope:hd_dc_extra_creator } }
			}
		}
	}
}
hd_dc_create_extra_county_faction_effect = {
	scope:hd_dc_extra_realm = { set_variable = { name = hd_dc_extra_faction_result value = flag:hd_dc_extra_faction_creation_failed } }
	situation:dynastic_cycle = {
		random_in_list = {
			variable = hd_dc_counties
			limit = { hd_dc_extra_faction_county_trigger = { TYPE = $TYPE$ } }
			save_scope_as = hd_dc_extra_county
			title_create_faction = { type = $TYPE$ target = scope:hd_dc_extra_faction_target }
			hd_dc_record_extra_faction_effect = { TYPE = $TYPE$ }
			if = {
				limit = { scope:hd_dc_extra_realm.var:hd_dc_extra_faction_result = flag:hd_dc_extra_faction_created }
				scope:hd_dc_extra_realm = { set_variable = { name = hd_dc_extra_faction_county value = scope:hd_dc_extra_county } }
			}
		}
	}
}
