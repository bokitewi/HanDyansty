# Root：天子。丹阳郡头衔及郡内仍归天子一系的县转给司马睿，睿改作天子封臣
hd_304_jianye_effect = {
	character:si_ma_rui = { save_scope_as = hd_304_rui }
	title:d_danyang = {
		every_de_jure_county = {
			limit = {
				holder ?= {
					NOT = { this = scope:hd_304_rui }
					OR = {
						this = root
						top_liege = root
					}
				}
			}
			save_temporary_scope_as = hd_304_jy_county
			hd_nbc_give_county_effect = { COUNTY = scope:hd_304_jy_county TO = scope:hd_304_rui }
		}
	}
	if = {
		limit = {
			NOT = { title:d_danyang.holder ?= scope:hd_304_rui }
		}
		create_title_and_vassal_change = { type = granted save_scope_as = hd_304_jy_change add_claim_on_loss = no }
		title:d_danyang = { change_title_holder = { holder = scope:hd_304_rui change = scope:hd_304_jy_change } }
		resolve_title_and_vassal_change = scope:hd_304_jy_change
	}
	# 睿离开琅邪国，琅琊郡交还徐州之主（执行后复核建议 4），以免改投天子后在越境内留下飞地
	title:k_uuii_xuzhou.holder ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { this = scope:hd_304_rui }
				NOT = { this = root }
			}
			save_scope_as = hd_304_xuzhou_lord
			title:d_langya = {
				every_de_jure_county = {
					limit = { holder ?= scope:hd_304_rui }
					save_temporary_scope_as = hd_304_ly_county
					hd_nbc_give_county_effect = { COUNTY = scope:hd_304_ly_county TO = scope:hd_304_xuzhou_lord }
				}
			}
			if = {
				limit = { title:d_langya.holder ?= scope:hd_304_rui }
				create_title_and_vassal_change = { type = granted save_scope_as = hd_304_ly_change add_claim_on_loss = no }
				title:d_langya = { change_title_holder = { holder = scope:hd_304_xuzhou_lord change = scope:hd_304_ly_change } }
				resolve_title_and_vassal_change = scope:hd_304_ly_change
			}
		}
	}
	scope:hd_304_rui = {
		hd_nbc_swear_fealty_effect = { LIEGE = root }
		if = {
			limit = { title:c_uuii_p_4389.holder ?= this }
			set_realm_capital = title:c_uuii_p_4389
		}
	}
	set_global_variable = hd_hev_304_jianye
	debug_log = "HD_304 S1 JIANYE DONE"
}

# Root：石勒。scope:hd_304_han：汉国君主
hd_304_shile_join_effect = {
	if = {
		limit = {
			any_held_title = {
				tier = tier_county
				is_landless_type_title = no
			}
		}
		# 已有地：只改投汉国
		hd_nbc_become_independent_effect = yes
		hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_304_han }
	}
	else = {
		# 无地（仍是冒险者）：从汉国君主直领县里挑一县，优先上党、平阳一带；不挑京城，除非只剩京城
		scope:hd_304_han = {
			random_held_title = {
				limit = {
					tier = tier_county
					is_landless_type_title = no
					NOT = { this = scope:hd_304_han.capital_county }
				}
				weight = {
					base = 1
					modifier = {
						add = 100
						OR = {
							duchy = title:d_shangdang
							duchy = title:d_shangdang_sili
							duchy = title:d_jianxing
							duchy = title:d_pingyang
							duchy = title:d_wuxiang
						}
					}
				}
				save_scope_as = hd_304_shile_county
			}
			if = {
				limit = { NOT = { exists = scope:hd_304_shile_county } }
				capital_county ?= { save_scope_as = hd_304_shile_county }
			}
		}
		if = {
			limit = { exists = scope:hd_304_shile_county }
			create_title_and_vassal_change = { type = granted save_scope_as = hd_304_shile_change add_claim_on_loss = no }
			scope:hd_304_shile_county = {
				change_title_holder = {
					holder = root
					change = scope:hd_304_shile_change
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:hd_304_shile_change
			if = {
				limit = { has_title = title:d_hd_304_shile_camp }
				destroy_title = title:d_hd_304_shile_camp
			}
			if = {
				limit = { NOT = { has_government = celestial_government } }
				change_government = celestial_government
			}
			hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_304_han }
			hd_nbc_spawn_army_effect = { SIZE = 8000 WHERE = scope:hd_304_shile_county NAME = hd_304_shile_army_name }
		}
	}
	set_global_variable = hd_hev_304_shile
	debug_log = "HD_304 S2 SHILE JOINED"
}

			limit = {
				exists = scope:hd_304_han

					is_imprisoned = no
					OR = {
						any_neighboring_top_liege_realm_owner = { this = title:h_china.holder }

							OR = {
								duchy = title:d_hedong

								duchy = title:d_henei
								duchy = title:d_hongnong
								duchy = title:d_rongyang
								duchy = title:d_ji
							}
						}
					}
				}
				title:h_china.holder ?= {
					is_alive = yes
					NOT = { this = scope:hd_304_han }
				}
				title:c_luoyang.holder ?= {
					OR = {
						this = title:h_china.holder
						top_liege = title:h_china.holder
					}
				}
			}
			scope:hd_304_han = { trigger_event = hd_304.0140 }
		}
		else = {
			set_global_variable = hd_304_yongjia_done
			debug_log = "HD_304 S4 YONGJIA SKIPPED"
		}
	}

		limit = {
			exists = global_var:hd_304_yongjia_started
			exists = global_var:hd_304_yongjia_decided	# 汉主已在事件里决定出兵
			NOT = { exists = global_var:hd_304_yongjia_done }

			NOT = { title:k_uuii_bingzhou.holder ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:yongjia } } }
		}
		set_global_variable = hd_304_yongjia_done
		hd_304_save_han_ruler_effect = yes

			limit = {
				exists = scope:hd_304_han
				title:c_luoyang.holder ?= {
					OR = {
						this = scope:hd_304_han
						top_liege = scope:hd_304_han
					}
				}
				title:h_china.holder ?= {
					is_alive = yes
					is_ai = yes
					is_imprisoned = no
					NOT = { this = scope:hd_304_han }
				}
			}
			title:h_china.holder = { save_scope_as = hd_304_captive }
			scope:hd_304_han = {
				imprison = {
					target = scope:hd_304_captive
					type = house_arrest
				}
			}
			debug_log = "HD_304 S4 YONGJIA EMPEROR CAPTURED"
		}

					OR = {

						top_liege = scope:hd_304_han
					}
				}

				hd_yiguan_enabled_trigger = yes
			}
			hd_yiguan_start_effect = yes
			debug_log = "HD_304 S4 YONGJIA YIGUAN STARTED"
		}
		debug_log = "HD_304 S4 YONGJIA ENDED"
	}
}
