# 运行时做，不改 history/titles；d_weijun_sili（朝歌）不在本项。写法照 hd_nbc_give_county_effect / hd_nbc_swear_fealty_effect。

hd_j16_304c_ying_lose_ye_effect = {
	if = {
		limit = { exists = scope:hd_j16_yl_ying }
		clear_saved_scope = hd_j16_yl_ying
	}
	if = {
		limit = { exists = scope:hd_j16_yl_tz }
		clear_saved_scope = hd_j16_yl_tz
	}
	character:si_ma_ying ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = yes
			}
			save_scope_as = hd_j16_yl_ying
		}
	}
	title:h_china.holder ?= {
		if = {
			limit = { is_alive = yes }
			save_scope_as = hd_j16_yl_tz
		}
	}
	# 收尾 pX-A（Q3；01 §7.2-3「不在没有选项的情况下夺走玩家的头衔」）：颖是玩家时整段不做，保留书签上的领地（含邺）
	if = {
		limit = { scope:hd_j16_yl_ying ?= { is_ai = no } }
		debug_log = "HD_J16 startup_304 SKIP player_ruler"
	}
	else_if = {
		limit = {
			exists = scope:hd_j16_yl_ying
			exists = scope:hd_j16_yl_tz
			NOT = { scope:hd_j16_yl_ying = scope:hd_j16_yl_tz }
			OR = {
				scope:hd_j16_yl_ying = { has_title = title:d_weijun }
				title:c_uuii_p_1172.holder ?= {
					OR = {
						this = scope:hd_j16_yl_ying
						target_is_liege_or_above = scope:hd_j16_yl_ying
					}
				}
			}
		}
		# ① 保底县：颖在魏郡以外已无任何县时，先取高邑（须在颖一方、非天子本人所持）
		if = {
			limit = {
				scope:hd_j16_yl_ying = {
					NOT = {
						any_held_title = {
							tier = tier_county
							NOT = { target_is_de_jure_liege_or_above = title:d_weijun }
						}
					}
				}
			}
			if = {
				limit = {
					title:c_uuii_p_4258.holder ?= {
						is_ai = yes	# 收尾 pX-A（Q3 同理）：高邑持有者是玩家时不取其县（整段随之不做，见 SKIP ying_lose_ye_no_keep_county）
						NOT = { this = scope:hd_j16_yl_tz }
						target_is_liege_or_above = scope:hd_j16_yl_ying
					}
				}
				hd_nbc_give_county_effect = { COUNTY = title:c_uuii_p_4258 TO = scope:hd_j16_yl_ying }
				scope:hd_j16_yl_ying = { set_realm_capital = title:c_uuii_p_4258 }
				debug_log = "HD_J16 startup_304 FIRE ying_keep_gaoyi"
			}
		}
		if = {
			limit = {
				scope:hd_j16_yl_ying = {
					any_held_title = {
						tier = tier_county
						NOT = { target_is_de_jure_liege_or_above = title:d_weijun }
					}
				}
			}
			# ② 首府若仍在魏郡（颖在魏郡外本已有县、未经 ① 的情形），先迁到魏郡以外的县
			if = {
				limit = {
					scope:hd_j16_yl_ying.capital_county ?= { target_is_de_jure_liege_or_above = title:d_weijun }
				}
				scope:hd_j16_yl_ying = {
					random_held_title = {
						limit = {
							tier = tier_county
							NOT = { target_is_de_jure_liege_or_above = title:d_weijun }
						}
						save_scope_as = hd_j16_yl_cap
					}
					if = {
						limit = { exists = scope:hd_j16_yl_cap }
						set_realm_capital = scope:hd_j16_yl_cap
					}
				}
			}
			# ③ 郡头衔：颖本人持有则转天子；颖下属持有则其人改投天子
			if = {
				limit = { title:d_weijun.holder ?= scope:hd_j16_yl_ying }
				create_title_and_vassal_change = { type = usurped save_scope_as = hd_j16_yl_change add_claim_on_loss = no }
				title:d_weijun = {
					change_title_holder = {
						holder = scope:hd_j16_yl_tz
						change = scope:hd_j16_yl_change
					}
				}
				resolve_title_and_vassal_change = scope:hd_j16_yl_change
			}
			else_if = {
				limit = {
					title:d_weijun.holder ?= {
						NOT = { this = scope:hd_j16_yl_tz }
						target_is_liege_or_above = scope:hd_j16_yl_ying
					}
				}
				title:d_weijun.holder = { hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_j16_yl_tz } }
			}
			# ④ 魏郡法理辖县逐县处理
			title:d_weijun = {
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_county }
					save_scope_as = hd_j16_yl_c
					if = {
						limit = { holder ?= scope:hd_j16_yl_ying }
						hd_nbc_give_county_effect = { COUNTY = scope:hd_j16_yl_c TO = scope:hd_j16_yl_tz }
					}
					else_if = {
						limit = {
							holder ?= {
								NOT = { this = scope:hd_j16_yl_tz }
								target_is_liege_or_above = scope:hd_j16_yl_ying
							}
						}
						holder = { hd_nbc_swear_fealty_effect = { LIEGE = scope:hd_j16_yl_tz } }
					}
				}
			}
			debug_log = "HD_J16 startup_304 FIRE ying_lose_ye"
		}
		else = {
			debug_log = "HD_J16 startup_304 SKIP ying_lose_ye_no_keep_county"
		}
	}
	else = {
		debug_log = "HD_J16 startup_304 SKIP ying_lose_ye_not_ready"
	}
}
