# Direct catalyst bridge. Keep the engine call in the original caller scope so
# native actor attribution and non-chaos phase progression remain unchanged.
hd_dc_catalyst_effect = {
	situation:dynastic_cycle ?= { hd_dc_map_catalyst_effect = { CATALYST = $CATALYST$ } }
	trigger_situation_catalyst = $CATALYST$
}
hd_dc_catalyst_with_actor_effect = {
	situation:dynastic_cycle ?= { hd_dc_map_catalyst_effect = { CATALYST = $catalyst$ } }
	trigger_situation_catalyst = { catalyst = $catalyst$ character = $character$ }
}

# One row from the installed phase tables. A/E use 4000 native points, all other
# columns use 2000. H selects one policy, never the sum of both recovery paths.
# Without a court retain its last policy; a new, leaderless region starts A.
hd_dc_catalyst_row_effect = {
	if = {
		limit = { situation_current_phase = situation_dynastic_cycle_phase_stability_advancement }
		set_variable = { name = hd_dc_catalyst_w value = { value = $A$ divide = 2 } }
	}
	else_if = {
		limit = { situation_current_phase = situation_dynastic_cycle_phase_stability_expansion }
		set_variable = { name = hd_dc_catalyst_w value = { value = $E$ divide = 2 } }
	}
	else_if = {
		limit = { situation_current_phase = situation_dynastic_cycle_phase_instability_conquest }
		set_variable = { name = hd_dc_catalyst_w value = $C$ }
	}
	else_if = {
		limit = {
			OR = {
				situation_current_phase = situation_dynastic_cycle_phase_instability
				situation_current_phase = situation_dynastic_cycle_phase_chaos
			}
		}
		set_variable = { name = hd_dc_catalyst_w value = $I$ }
	}
	if = {
		limit = { situation_current_phase = situation_dynastic_cycle_phase_instability_conquest }
		if = {
			limit = { var:hd_dc_h_policy = flag:advancement }
			set_variable = { name = hd_dc_catalyst_h value = $CHA$ }
		}
		else = { set_variable = { name = hd_dc_catalyst_h value = $CHE$ } }
	}
	else = {
		if = {
			limit = { var:hd_dc_h_policy = flag:advancement }
			set_variable = { name = hd_dc_catalyst_h value = $HA$ }
		}
		else = { set_variable = { name = hd_dc_catalyst_h value = $HE$ } }
	}
}
