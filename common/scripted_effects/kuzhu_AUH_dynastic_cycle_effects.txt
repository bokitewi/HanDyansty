#危世效果
kuzhu_entering_instability_effect = {
	#苦竹：不是开局
	if = {
		limit = {
			current_date_is_start_date_trigger = no
		}
		#苦竹：封臣独立条件	
		every_vassal = {
			limit = {
				is_landed = yes
				OR = {
					#非法理封臣
					primary_title = {
						NOT = { hegemony = title:h_china }
					}
					#非天朝制封臣
					NOT = { government_has_flag = government_is_celestial }				
				} 
			}
			#封臣独立
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = change
				add_claim_on_loss = yes
			}
			becomes_independent = {
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
			break_subject_contract_and_establish_tributary_effect = { SUZERAIN = title:h_china.holder TRIBUTARY = this }
		}
		#苦竹：封臣变军阀条件		
		every_vassal = {
			limit = {
				is_ai = yes
				is_landed = yes
				NOT = { has_character_flag = flag_kuzhu_new_governor }
				#忠诚相关
				OR = { 
					has_trait = ambitious
					has_trait = disloyal
					has_trait = arrogant
					has_trait = arbitrary
					has_trait = greedy
					has_trait = deceitful
				}
				OR = {
					#不满的守旧派封臣
					AND = {
						opinion = {
							target = liege
							value <= -20
						}
						top_participant_group:dynastic_cycle ?= { participant_group_type = conservative_movement }
					}					
					#不满的野心封臣
					AND = {
						opinion = {
							target = liege
							value <= -40
						}
						has_trait = ambitious
					}
					#非常不满的封臣
					opinion = {
						target = liege
						value <= -60
					}
				}
			}
			kuzhu_convert_to_warlord_from_admin_effect = yes
			add_character_flag = {
				flag = flag_kuzhu_new_warlord
				years = 10
			}
		}		
	}
}

#乱世效果(2026-08重写:天子保位+霸权不灭版本)
#第一层:绝对禁止销毁霸权——不执行任何destroy_title,天子保留h_china与自己的领地,君权降2级、正统性降2级
#第二层:非天朝制直属封臣(仅every_vassal,不碰every_vassal_or_below)全部立刻独立,其封臣随整个realm一起离开
#第三层:剩余天朝制封臣按辖区叛乱度一次性判定
#     帝国级: <=20保持封臣, 21-39转为朝贡者, >=40独立
#     每降低一级头衔,两个阈值都上移5点;只处理伯爵及以上,霸权级不处理
#     忠顺角色只通过辖区叛乱度的-60因素影响,不再获得硬性豁免
#判定范围含皇帝封臣链上任意层级(如王国封臣手下的公爵),仅在进入群雄割据时判定这一次
kuzhu_chaos_shattering_effect = {
	hidden_effect = {
		title:h_china = {
			holder = { save_temporary_scope_as = old_emperor_temp }
			# Reset imperial examination activity
			set_variable = {
				name = years_since_imperial_examination
				value = 0
			}
		}

		############ 第一层:天子保住霸权与领地 ############
		scope:old_emperor_temp = {
			#君权降2级(天朝官僚制0-3级;非天朝政体时回落到王权法0-4级),已经很低则降到最低
			if = {
				limit = { has_realm_law = celestial_bureaucracy_3 }
				add_realm_law_skip_effects = celestial_bureaucracy_1
			}
			else_if = {
				limit = { has_realm_law = celestial_bureaucracy_2 }
				add_realm_law_skip_effects = celestial_bureaucracy_0
			}
			else_if = {
				limit = { has_realm_law = celestial_bureaucracy_1 }
				add_realm_law_skip_effects = celestial_bureaucracy_0
			}
			else_if = {
				limit = { has_realm_law = crown_authority_4 }
				add_realm_law_skip_effects = crown_authority_2
			}
			else_if = {
				limit = { has_realm_law = crown_authority_3 }
				add_realm_law_skip_effects = crown_authority_1
			}
			else_if = {
				limit = { has_realm_law = crown_authority_2 }
				add_realm_law_skip_effects = crown_authority_0
			}
			else_if = {
				limit = { has_realm_law = crown_authority_1 }
				add_realm_law_skip_effects = crown_authority_0
			}
			#正统性直降2级
			add_legitimacy = hd_legitimacy_drop_two_levels_loss_value
		}

		############ 第二层:非天朝制直属封臣全部立刻独立 ############
		#只检查直属封臣;独立后其麾下封臣随整个realm离开,不产生越级脱离
		scope:old_emperor_temp = {
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = chaos_change
				add_claim_on_loss = yes
			}
			every_vassal = {
				limit = {
					is_landed = yes
					NOT = { government_has_flag = government_is_celestial }
				}
				becomes_independent = {
					change = scope:chaos_change
				}
			}
			resolve_title_and_vassal_change = scope:chaos_change
		}

		############ 第三层:辖区叛乱度一次性判定(按最高持有头衔分层) ############
		#遍历剩余封臣链上的伯爵至帝国级封臣;霸权级不参与本层判定
		scope:old_emperor_temp = {
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = hd_chaos_rebellion_change
				add_claim_on_loss = yes
			}
			every_vassal_or_below = {
				limit = {
					is_landed = yes
					highest_held_title_tier >= tier_county
					highest_held_title_tier <= tier_empire
				}
				set_variable = {
					name = hd_chaos_rebellion_temp
					value = kuzhu_governor_political_power
				}
				# 帝国级: <=20保持封臣, 21-39朝贡, >=40独立
				if = {
					limit = {
						highest_held_title_tier = tier_empire
						var:hd_chaos_rebellion_temp >= 40
					}
					becomes_independent = {
						change = scope:hd_chaos_rebellion_change
					}
				}
				else_if = {
					limit = {
						highest_held_title_tier = tier_empire
						var:hd_chaos_rebellion_temp > 20
					}
					break_subject_contract_and_establish_tributary_effect = {
						SUZERAIN = scope:old_emperor_temp
						TRIBUTARY = this
					}
				}
				# 王国级: <=25保持封臣, 26-44朝贡, >=45独立
				else_if = {
					limit = {
						highest_held_title_tier = tier_kingdom
						var:hd_chaos_rebellion_temp >= 45
					}
					becomes_independent = { change = scope:hd_chaos_rebellion_change }
				}
				else_if = {
					limit = {
						highest_held_title_tier = tier_kingdom
						var:hd_chaos_rebellion_temp > 25
					}
					break_subject_contract_and_establish_tributary_effect = {
						SUZERAIN = scope:old_emperor_temp
						TRIBUTARY = this
					}
				}
				# 公爵级: <=30保持封臣, 31-49朝贡, >=50独立
				else_if = {
					limit = {
						highest_held_title_tier = tier_duchy
						var:hd_chaos_rebellion_temp >= 50
					}
					becomes_independent = { change = scope:hd_chaos_rebellion_change }
				}
				else_if = {
					limit = {
						highest_held_title_tier = tier_duchy
						var:hd_chaos_rebellion_temp > 30
					}
					break_subject_contract_and_establish_tributary_effect = {
						SUZERAIN = scope:old_emperor_temp
						TRIBUTARY = this
					}
				}
				# 伯爵级: <=35保持封臣, 36-54朝贡, >=55独立
				else_if = {
					limit = {
						highest_held_title_tier = tier_county
						var:hd_chaos_rebellion_temp >= 55
					}
					becomes_independent = { change = scope:hd_chaos_rebellion_change }
				}
				else_if = {
					limit = {
						highest_held_title_tier = tier_county
						var:hd_chaos_rebellion_temp > 35
					}
					break_subject_contract_and_establish_tributary_effect = {
						SUZERAIN = scope:old_emperor_temp
						TRIBUTARY = this
					}
				}
				remove_variable = hd_chaos_rebellion_temp
			}
			resolve_title_and_vassal_change = scope:hd_chaos_rebellion_change
		}

		#独立/转朝贡的封臣立即重算派系归属(留任者凭movement_member变量保持原派系)
		situation:dynastic_cycle = {
			every_situation_participant = {
				recalculate_participant_group = situation:dynastic_cycle
			}
		}

		############ 给有实力的诸侯发夺天命story,驱动AI争夺天命 ############
		situation:dynastic_cycle = {
			every_situation_participant = {
				if = {
					limit = {
						NOR = {
							has_trait = conqueror
							any_owned_story = {
								story_type = story_conqueror
							}
							any_owned_story = {
								story_type = story_greatest_of_khans
							}
							any_owned_story = {
								story_type = story_mongol_invasion
							}
							any_owned_story = {
								story_type = story_take_mandate_of_heaven
							}
						}
						is_landed = yes
						highest_held_title_tier >= tier_duchy
						is_tributary = no
						is_adult = yes
						is_incapable = no
						this != scope:old_emperor_temp
						trigger_if = {
							limit = { is_ai = yes }
							ai_has_cautious_personality = no
							ai_boldness >= 0
							ai_greed >= 0
						}
					}
					create_story = story_take_mandate_of_heaven
				}
			}
		}
	}
}
