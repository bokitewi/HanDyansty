# 按指针写国号样式（写法同 hd_nbc_style_effect，只是不按 character:<id> 找人）
hd263_style_ptr_effect = {
	global_var:hd_sg_$WHO$ ?= {
		if = {
			limit = { is_alive = yes exists = primary_title }
			primary_title = { set_variable = { name = hd_nbc_style value = flag:$STYLE$ } }
		}
	}
}

# 218 年开局到 253.3.1：接上 253 心跳，并让 263 的接力看守也在本局启动。253 开局的授官、都督区、开局两战、开局资源都不做
hd253_relay_activate_effect = {
	if = {
		limit = { NOT = { exists = global_var:hd253_relay_on } }
		set_global_variable = hd253_relay_on
		hd_sg_ptr_refresh_effect = yes
		title:h_china = { trigger_event = { on_action = hd253_pulse days = 1 } }
		if = {
			limit = { NOT = { exists = global_var:hd263_relay_watch_started } }
			set_global_variable = hd263_relay_watch_started
			title:h_china = { trigger_event = { on_action = hd263_relay_watch days = 30 } }
		}
		debug_log = "HD253 RELAY ON"
		# 局面记录（首测定位用，不作闸门）
		if = {
			limit = { title:h_china.var:hd_court_regent ?= { this = character:si_ma_shi } }
			debug_log = "HD253 RELAY STATE SHI_REGENT YES"
		}
		else = { debug_log = "HD253 RELAY STATE SHI_REGENT NO" }
		if = {
			limit = { global_var:hd_sg_cao_fang ?= { is_alive = yes has_title = title:h_china } }
			debug_log = "HD253 RELAY STATE CAO_FANG_EMPEROR YES"
		}
		else = { debug_log = "HD253 RELAY STATE CAO_FANG_EMPEROR NO" }
		if = {
			limit = {
				title:e_bashu.holder ?= { is_alive = yes is_independent_ruler = yes }
				hd253_relay_shu_ok_trigger = yes
			}
			debug_log = "HD253 RELAY STATE SHU_INDEPENDENT YES"
		}
		else = { debug_log = "HD253 RELAY STATE SHU_INDEPENDENT NO" }
		if = {
			limit = { title:e_jiangdong.holder ?= { is_alive = yes is_independent_ruler = yes } }
			debug_log = "HD253 RELAY STATE WU_INDEPENDENT YES"
		}
		else = { debug_log = "HD253 RELAY STATE WU_INDEPENDENT NO" }
	}
}
