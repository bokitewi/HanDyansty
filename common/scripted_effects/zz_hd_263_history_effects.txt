# 找到进行中的伐蜀之战，存为 scope:hd263_war
hd263_find_war_effect = {
	title:h_china.holder ?= {
		random_character_war = {
			limit = {
				using_cb = hd253_history_war_cb
				var:hd253_war_kind ?= flag:fashu
			}
			save_scope_as = hd263_war
		}
	}
}

# 世界通报（两人）
hd263_news_effect = {
	$A$ = { save_scope_as = hd_news_a }
	$B$ = { save_scope_as = hd_news_b }
	hd_world_news_send_effect = { EVENT = $EVENT$ }
}

# 起兵：以独立战争攻其主君，另给 $LEVIES$ 事件兵（照 hd253_rebel_effect，兵数做成参数）。Root：起兵者
hd263_rebel_effect = {
	save_scope_as = hd263_rebel
	liege = { save_scope_as = hd263_rebel_target }
	start_war = {
		cb = independence_war
		target = scope:hd263_rebel_target
	}
	random_character_war = {
		limit = {
			primary_attacker = scope:hd263_rebel
			using_cb = independence_war
		}
		save_scope_as = hd263_rebel_war
	}
	if = {
		limit = { exists = scope:hd263_rebel_war }
		spawn_army = {
			levies = $LEVIES$
			location = capital_province
			origin = capital_province
			inheritable = yes
			uses_supply = yes
			war = scope:hd263_rebel_war
			name = hd263_rebel_army_name
		}
		debug_log = "HD263 REBEL_WAR OK"
	}
	else = { debug_log = "HD263 REBEL_WAR FAILED" }
}

# S1 伐蜀（after_lobby 起；失败或白和、战败、失效后 264.1.16 前可再宣一次）
############################################################
hd263_s1_try_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd263_s1_cancel }
			NOT = { exists = global_var:hd263_s1_asked }
			NOT = { exists = global_var:hd263_s2_done }
			NOT = { exists = global_var:hd253_fashu_active }
			NOT = { exists = global_var:hd253_fashu_won }
			current_date < 264.1.16
			OR = {
				NOT = { exists = global_var:hd263_s1_tries }
				global_var:hd263_s1_tries < 2
			}
			exists = title:h_china.holder
			title:e_bashu.holder ?= {
				is_alive = yes
				is_independent_ruler = yes
				NOT = { is_at_war_with = title:h_china.holder }
			}
		}
		set_global_variable = hd263_s1_asked
		if = {
			limit = {
				character:si_ma_zhao ?= {
					is_alive = yes
					is_ai = no
					title:h_china.var:hd_court_regent ?= this
				}
			}
			character:si_ma_zhao = { trigger_event = hd263.0100 }
		}
		else = { hd263_s1_ask_emperor_effect = yes }
	}
}

		save_scope_as = hd263_att

			limit = { exists = global_var:hd263_s1_tries }
			change_global_variable = { name = hd263_s1_tries add = 1 }
		}
		else = { set_global_variable = { name = hd263_s1_tries value = 1 } }

		hd263_s1_general_effect = { WHO = zhu_ge_xu }

	}
	else = {
		remove_global_variable = hd263_s1_asked   # 下个月重试
		debug_log = "HD263 S1 WAR_START_FAILED"
	}

# 加入攻方并在首府给事件兵：钟会二万五千、邓艾一万、诸葛绪一万；加入失败则兵给天子。Scope：将领
hd263_s1_join_effect = {
	save_scope_as = hd263_gen
	hd263_find_war_effect = yes
	if = {
		limit = {
			exists = scope:hd263_war
			is_alive = yes
			is_landed = yes
			is_independent_ruler = no
			NOT = { is_attacker_in_war = scope:hd263_war }
		}
		scope:hd263_war = { add_attacker = scope:hd263_gen }
	}
	if = {
		limit = { exists = scope:hd263_war is_attacker_in_war = scope:hd263_war }
		hd263_s1_spawn_effect = { WHO = scope:hd263_gen }
		debug_log = "HD263 S1 JOIN_OK"
	}
	else_if = {
		limit = { exists = scope:hd263_war }
		hd263_s1_spawn_effect = { WHO = title:h_china.holder }
		debug_log = "HD263 S1 JOIN_FAILED_LEVIES_TO_EMPEROR"
	}
}

# 战争计数与结局：白和、战败、失效后允许重宣（264.1.16 前），过期则 S1–S4 作罢
hd263_s1_watch_effect = {
	if = {
		limit = { hd263_s1_war_active_trigger = yes }
		change_global_variable = { name = hd263_s1_pulses add = 1 }
	}
	hd263_s1_end_kind_effect = { RESULT = white }
	hd263_s1_end_kind_effect = { RESULT = lost }
	hd263_s1_end_kind_effect = { RESULT = invalid }
	if = {
		limit = {
			NOT = { exists = global_var:hd263_s1_cancel }
			NOT = { exists = global_var:hd263_s2_done }
			NOT = { exists = global_var:hd253_fashu_won }
			NOT = { hd263_s1_war_active_trigger = yes }
			current_date >= 264.1.16
		}
		set_global_variable = hd263_s1_cancel
		debug_log = "HD263 S1 GIVEN_UP"
	}
}
hd263_s1_end_kind_effect = {
	if = {
		limit = { exists = global_var:hd253_fashu_$RESULT$ }
		remove_global_variable = hd253_fashu_$RESULT$
		if = {
			limit = { exists = global_var:hd263_s1_asked }
			remove_global_variable = hd263_s1_asked
		}
		debug_log = "HD263 S1 ENDED_$RESULT$"
	}
}

# S2 刘禅出降：S1 打满 60 天后攻方战分 ≥ 50 或占涪、绵竹、雒、成都任一县；或 S1 已胜；264.1.16 兜底
############################################################
hd263_s2_check_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd263_s2_done }
			NOT = { exists = global_var:hd263_s1_cancel }
			OR = {
				exists = global_var:hd253_fashu_won
				AND = {
					hd263_s1_war_active_trigger = yes
					OR = {
						current_date >= 264.1.16
						AND = {
							global_var:hd263_s1_pulses >= 2
							OR = {
								title:c_uuii_p_3629.title_province = { is_occupied = yes }
								title:c_uuii_p_3295.title_province = { is_occupied = yes }
								title:c_uuii_p_3282.title_province = { is_occupied = yes }
								title:c_uuii_p_4282.title_province = { is_occupied = yes }
								title:h_china.holder = {
									any_character_war = {
										using_cb = hd253_history_war_cb
										var:hd253_war_kind ?= flag:fashu
										attacker_war_score >= 50
									}
								}
							}
						}
					}
				}
			}
		}
		set_global_variable = hd263_s2_done
		if = {
			limit = { title:e_bashu.holder ?= { is_alive = yes is_ai = no } }
			title:e_bashu.holder = { trigger_event = hd263.0200 }
		}
		else = { hd263_s2_surrender_effect = yes }
	}
}

# 出降：①以攻方胜结束 S1（on_victory 把益州交天子）②汉帝头衔归天子、封臣改投（姜维降钟会）③益州与蜀郡、蜀地郡守交邓艾 ④绵竹诸将与刘谌之死（只限电脑）
hd263_s2_surrender_effect = {
	hd263_find_war_effect = yes
	if = {
		limit = { exists = scope:hd263_war }
		scope:hd263_war = { end_war = attacker }
	}
	title:e_bashu.holder ?= { save_scope_as = hd263_shu }
	title:h_china.holder ?= { save_scope_as = hd263_wei }
	if = {
		limit = { exists = scope:hd263_shu exists = scope:hd263_wei }
		# ② 汉帝的头衔与封臣
		create_title_and_vassal_change = { type = conquest save_scope_as = hd263_s2_change add_claim_on_loss = no }
		scope:hd263_shu = {
			every_held_title = {
				limit = { is_noble_family_title = no is_landless_type_title = no }
				change_title_holder = { holder = scope:hd263_wei change = scope:hd263_s2_change take_baronies = yes }
			}
			every_vassal = {
				change_liege = { liege = scope:hd263_wei change = scope:hd263_s2_change }
			}
		}
		resolve_title_and_vassal_change = scope:hd263_s2_change
		# 刘禅仍持本族世族头衔时仍是统治者，不能 set_employer：改为向魏帝称臣（安乐公，同 253 的出降）
		scope:hd263_shu = {
			if = {
				limit = { is_alive = yes is_ruler = no NOT = { is_courtier_of = scope:hd263_wei } }
				set_employer = scope:hd263_wei
			}
			else_if = {
				limit = { is_alive = yes is_ruler = yes NOT = { liege ?= scope:hd263_wei } }
				create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd263_s2_fealty add_claim_on_loss = no }
				change_liege = { liege = scope:hd263_wei change = scope:hd263_s2_fealty }
				resolve_title_and_vassal_change = scope:hd263_s2_fealty
			}
		}
		# 姜维（上一步已随众改投天子）：电脑再降钟会；玩家先问（hd263.0210：降钟会／拒命）
		character:jiang_wei ?= {
			if = {
				limit = { is_alive = yes is_landed = yes is_ai = no }
				trigger_event = hd263.0210
			}
			else_if = {
				limit = { is_alive = yes is_landed = yes }
				hd263_jiang_wei_submit_effect = yes
			}
		}
		# ③ 益州交邓艾
		hd263_s2_dengai_yizhou_effect = yes
		# ④ 绵竹之战与刘谌（只限电脑）
		hd263_die_effect = { WHO = zhu_ge_zhan REASON = death_battle }
		hd263_die_effect = { WHO = zhu_ge_shang REASON = death_battle }
		hd263_die_effect = { WHO = huang_chong REASON = death_battle }
		hd263_die_effect = { WHO = li_qiu REASON = death_battle }
		hd263_die_effect = { WHO = zhang_zun_2 REASON = death_battle }
		hd263_die_effect = { WHO = fu_qian REASON = death_battle }
		hd263_die_effect = { WHO = liu_chen REASON = death_suicide }
		hd263_news_effect = { A = scope:hd263_shu B = scope:hd263_wei EVENT = hd263.0910 }
		if = {
			limit = {
				scope:hd263_shu = {
					is_landed = no
					OR = { is_courtier_of = scope:hd263_wei liege ?= scope:hd263_wei }
				}
				title:k_uuii_yizhou.holder ?= { NOT = { this = scope:hd263_shu } }
			}
			debug_log = "HD263 S2 TRANSFER_OK"
		}
		else = { debug_log = "HD263 S2 TRANSFER_BAD" }
	}
	else = { debug_log = "HD263 S2 SKIPPED_NO_RULER" }
}

# 只杀电脑角色；玩家角色不杀，只记日志
hd263_die_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes is_ai = yes }
			death = { death_reason = $REASON$ }
		}
		else_if = {
			limit = { is_alive = yes }
			debug_log = "HD263 PLAYER_SPARED $WHO$"
		}
	}
}

# 姜维降钟会（史实降于涪）；钟会不在或不是天子封臣则降天子。Scope：任意

	# 汉已亡，姜维的都督区（开局建在梁州上）先解散
	title:k_liaozhou = {
		if = {
			limit = { has_variable = wj_dd_core }
			WJ_dd_dissolve_core_effect = yes
			debug_log = "HD263 S2 DISTRICT_DISSOLVED"
		}
	}
	character:jiang_wei ?= {
		if = {
			limit = { is_alive = yes is_landed = yes }
			if = {
				limit = {
					character:yingchuan_zhong_024 ?= {
						is_alive = yes
						is_landed = yes
						liege ?= title:h_china.holder
					}
				}
				# 封臣须低于主君：梁州（王国）先交钟会，再让姜维、张翼、廖化改投钟会（执行后复核必改 1）
				character:yingchuan_zhong_024 = { save_scope_as = hd263_zh }

					if = {
						limit = { holder ?= character:jiang_wei }
						change_title_holder = { holder = scope:hd263_zh change = scope:hd263_jw_change }
					}
				}
				character:zhang_yi_4 ?= {
					if = {
						limit = { is_alive = yes is_landed = yes liege ?= character:jiang_wei }
						change_liege = { liege = scope:hd263_zh change = scope:hd263_jw_change }
					}
				}
				character:liao_hua ?= {
					if = {
						limit = { is_alive = yes is_landed = yes liege ?= character:jiang_wei }
						change_liege = { liege = scope:hd263_zh change = scope:hd263_jw_change }
					}
				}
				change_liege = { liege = scope:hd263_zh change = scope:hd263_jw_change }
				resolve_title_and_vassal_change = scope:hd263_jw_change
				if = {
					limit = { has_title = title:d_hanzhong }
					set_primary_title_to = title:d_hanzhong
				}
				scope:hd263_zh = {
					if = {
						limit = { has_title = title:k_liaozhou NOT = { has_title = title:k_uuii_yizhou } }

					}
				}
				if = {
					limit = {
						liege ?= scope:hd263_zh
						highest_held_title_tier <= tier_duchy
						scope:hd263_zh = { highest_held_title_tier >= tier_kingdom }
					}
					debug_log = "HD263 S2 TIER_OK"
				}
				else = { debug_log = "HD263 S2 TIER_BAD" }
				debug_log = "HD263 S2 JIANG_WEI -> ZHONG_HUI"
			}
			else_if = {
				limit = { NOT = { liege ?= title:h_china.holder } }
				create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd263_jw_change add_claim_on_loss = no }
				change_liege = { liege = title:h_china.holder change = scope:hd263_jw_change }
				resolve_title_and_vassal_change = scope:hd263_jw_change
				debug_log = "HD263 S2 JIANG_WEI -> EMPEROR"
			}
		}
	}
}

		}
