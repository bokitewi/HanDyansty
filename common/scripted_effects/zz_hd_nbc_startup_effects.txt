# 天朝循环：按你定的"中期"从对峙阶段开局，五德取晋的金德（写法同 hd218_dynastic_cycle_effect）
hd_nbc_dynastic_cycle_effect = {
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = title:h_china.holder
			NOT = { exists = situation:dynastic_cycle }
		}
		start_situation = {
			type = dynastic_cycle
			start_phase = situation_dynastic_cycle_phase_hd_standoff
		}
	}
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = situation:dynastic_cycle
			NOT = { exists = global_var:saas_dynastic_cycle_initialized }
		}
		situation:dynastic_cycle = {
			title:h_china = {
				set_variable = {
					name = years_since_imperial_examination
					value = 0
				}
			}
			title:h_china.holder ?= {
				if = {
					limit = { legitimacy < legitimacy_level_6 }
					add_legitimacy = {
						value = legitimacy_level_6
						subtract = legitimacy
					}
				}
			}
			tgp_set_wuking_element_effect = { ELEMENT = metal }
			every_situation_participant = {
				recalculate_participant_group = situation:dynastic_cycle
			}
			every_situation_sub_region = {
				hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_chaos }
			}
			trigger_event = { on_action = hd218_dc_zero_progress days = 2 }
		}
		set_global_variable = {
			name = saas_dynastic_cycle_initialized
			value = yes
		}
		debug_log = "HD_NBC DYNASTIC_CYCLE STANDOFF"
	}
}

# 国号与称谓。
#   国号：set_title_name 改主头衔名；FIXED = yes 时把国号记到王朝上（hd_fixed_realm_name），继承人沿用。
#   只给专属一个政权的王朝记固定国号：两燕（同属慕容）、前凉张氏、西凉李氏是通用宗族，只改头衔名。
#   称谓：主头衔设 hd_nbc_style（tianwang/huangdi/wang/gong），由 WJ_native_province_pilot 自定义本地化显示；
#   h_china 不设，晋帝、楚帝永远显示天子。
hd_nbc_realm_name_effect = {
	character:$WHO$ ?= {
		if = {
			limit = { is_alive = yes exists = primary_title }
			primary_title = { set_title_name = $NAME$ }
			if = {
				limit = { always = $FIXED$ }
				dynasty ?= {
					remove_variable = hd_fixed_realm_title
					set_variable = { name = hd_fixed_realm_name value = flag:$NAME$ }
				}
			}
		}
	}
}

# 头衔与席位对应见 common/council_positions（got_minister_position_effect 的参数）。
hd_nbc_minister_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { has_title = title:$TITLE$ }
				NOT = { this = scope:councillor_liege }
			}
			if = {
				limit = {
					is_landed = no
					NOT = { employer ?= scope:councillor_liege }
				}
				scope:councillor_liege = { add_courtier = character:$WHO$ }
			}
			got_minister_position_effect = { MINISTER_TITLE = $TITLE$ MINISTER_POSITION = $POS$ }
		}
	}
}

# Character：原文化是 xianbei / xiongnu 的鲜卑各部、柔然、高车王族改用新文化。开局时对全部在世者执行，hbirth 出场时也执行
hd_nbc_assign_culture_effect = {
	if = {
		limit = {
			exists = culture
			OR = {
				culture = culture:xianbei
				culture = culture:xiongnu
			}
		}
		if = {
			limit = {
				culture = culture:xianbei
				dynasty ?= {
					OR = {
						# 拓跋与代北"帝室十姓"、勋臣诸姓
						this = dynasty:tuoba
						this = dynasty:zhangsun
						this = dynasty:daxi_dyn
						this = dynasty:yizhan_dyn
						this = dynasty:helou_dyn
						this = dynasty:chiluo_dyn
						this = dynasty:monalou_dyn
						this = dynasty:bulugu_dyn
						this = dynasty:helan
						this = dynasty:yuchi
						this = dynasty:chigan_dyn
						this = dynasty:yilou_dyn
					}
				}
			}
			set_culture = culture:hd_tuoba
		}
		else_if = {
			limit = { culture = culture:xianbei dynasty ?= dynasty:murong }
			set_culture = culture:hd_murong
		}
		else_if = {
			limit = { culture = culture:xianbei house ?= house:liaoxi_duan }
			set_culture = culture:hd_duan
		}
		else_if = {
			limit = { culture = culture:xianbei dynasty ?= dynasty:yuwen }
			set_culture = culture:hd_yuwen
		}
		else_if = {
			limit = { culture = culture:xianbei dynasty ?= dynasty:qifu }
			set_culture = culture:hd_qifu
		}
		else_if = {
			limit = { culture = culture:xianbei dynasty ?= dynasty:tufa }
			set_culture = culture:hd_hexi_xianbei
		}
		else_if = {
			limit = {
				culture = culture:xianbei
				dynasty ?= {
					OR = {
						this = dynasty:Rouran_yujiulv_dynasty
						this = dynasty:Rouran_ru_dynasty
					}
				}
			}
			set_culture = culture:hd_rouran
		}
		else_if = {
			limit = { culture = culture:xiongnu dynasty ?= dynasty:hulv_dyn }
			set_culture = culture:hd_gaoche
		}
	}
}
