# Scope：角色。罢官：终止其对天子的辅政、免去议事会席位、销毁所持天朝大臣头衔（含 WJ 内阁）
hd_dc_kit_ten_strip_office_effect = {
	save_scope_as = hd_dc_ten_stripped
	title:h_china.holder ?= {
		if = {
			limit = {
				has_active_diarchy = yes
				diarch ?= scope:hd_dc_ten_stripped
			}
			end_diarchy = yes
		}
		if = {
			limit = { scope:hd_dc_ten_stripped = { is_councillor_of = prev } }
			always_fire_councillor = scope:hd_dc_ten_stripped
		}
	}
	every_held_title = {
		limit = { tgp_is_minister_title = yes }
		scope:hd_dc_ten_stripped = { destroy_title = prev }
	}
	if = {
		limit = { tgp_has_minister_title = no }
		remove_character_modifier = minister_scheme_slots_modifier
	}
}

# 天子将 $TARGET$ 下狱（弹劾 / 清君侧 / 天子夺权）。
# 这类下狱属"铲除权臣"，不应同时计为天子"诛戮大臣"/"打压朋党"（核心 on_imprison 钩子会按天子发射这两项）：
# 下狱前若天子身上没有这两项的诱因冷却旗标，临时加 2 天旗标使核心钩子跳过（旗标名沿用 hd_dc_kit_fire_effect 的约定）
hd_dc_kit_ten_emperor_imprison_effect = {
	title:h_china.holder ?= {
		if = {
			limit = { NOT = { has_character_flag = hd_dc_cd_catalyst_hd_purge_vip } }
			add_character_flag = { flag = hd_dc_cd_catalyst_hd_purge_vip days = 2 }
		}
		if = {
			limit = { NOT = { has_character_flag = hd_dc_cd_catalyst_hd_suppress_party } }
			add_character_flag = { flag = hd_dc_cd_catalyst_hd_suppress_party days = 2 }
		}
		imprison = {
			target = $TARGET$
			type = dungeon
		}
	}
}

# Scope：天子。挑一名"权臣"存为 scope:hd_dc_ten_pm_pick：势力最强的在任朋党党魁 → 丞相 → 辅政
hd_dc_kit_ten_pick_power_minister_effect = {
	var:hd_party_huanguan_leader ?= { add_to_list = hd_dc_ten_pm_cands }
	var:hd_party_waiqi_leader ?= { add_to_list = hd_dc_ten_pm_cands }
	var:hd_party_wuxun_leader ?= { add_to_list = hd_dc_ten_pm_cands }
	var:hd_party_shizu_leader ?= { add_to_list = hd_dc_ten_pm_cands }
	var:hd_party_hanmen_leader ?= { add_to_list = hd_dc_ten_pm_cands }
	ordered_in_list = {
		list = hd_dc_ten_pm_cands
		limit = {
			is_imprisoned = no
			hd_dc_kit_ten_is_power_minister_trigger = yes
		}
		order_by = hd_party_debate_sort_value
		save_scope_as = hd_dc_ten_pm_pick
	}
	if = {
		limit = { NOT = { exists = scope:hd_dc_ten_pm_pick } }
		title:e_minister_chancellor.holder ?= {
			if = {
				limit = {
					is_imprisoned = no
					hd_dc_kit_ten_is_power_minister_trigger = yes
				}
				save_scope_as = hd_dc_ten_pm_pick
			}
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:hd_dc_ten_pm_pick }
			has_active_diarchy = yes
		}
		diarch ?= {
			if = {
				limit = { is_imprisoned = no }
				save_scope_as = hd_dc_ten_pm_pick
			}
		}
	}
}

hd_dc_kit_ten_yearly_effect = {
	title:h_china.holder ?= {
		save_scope_as = hd_dc_ten_emperor
		# 乱象：仅紧张期；离开紧张期即清除
		if = {
			limit = {
				hd_dc_phase_is_trigger = { PHASE = instability }
				hd_party_court_initialized_trigger = yes
			}
			hd_dc_kit_ten_chaos_update_effect = yes
		}
		else = {
			hd_dc_kit_ten_clear_chaos_effect = yes
		}
		# 中兴改革进度（天子身故则随之中辍）
		if = {
			limit = { has_variable = hd_dc_ten_restoration_years }
			change_variable = { name = hd_dc_ten_restoration_years add = 1 }
			if = {
				limit = { var:hd_dc_ten_restoration_years >= hd_dc_ten_restoration_duration_years }
				trigger_event = hd_dc_ten.0005
			}
			else_if = {
				limit = { var:hd_dc_ten_restoration_years = 8 }
				trigger_event = hd_dc_ten.0004
			}
			else_if = {
				limit = { var:hd_dc_ten_restoration_years = 5 }
				trigger_event = hd_dc_ten.0003
			}
			else_if = {
				limit = { var:hd_dc_ten_restoration_years = 2 }
				trigger_event = hd_dc_ten.0002
			}
		}
		# 清君侧所立终身辅政：被解除则补回
		hd_dc_kit_ten_regent_repair_check_effect = yes
	}
	# 党锢禁锢：十年内不得复任
	hd_dc_kit_ten_office_ban_enforce_effect = yes
}

hd_dc_kit_ten_chaos_update_effect = {
	if = {
		limit = { hd_dc_kit_ten_dominant_party_trigger = { G = waiqi } }
		hd_dc_kit_ten_set_chaos_effect = { STATE = consort }
	}
	else_if = {
		limit = { hd_dc_kit_ten_dominant_party_trigger = { G = huanguan } }
		hd_dc_kit_ten_set_chaos_effect = { STATE = eunuch }
	}
	else_if = {
		limit = { hd_dc_kit_ten_dominant_party_trigger = { G = wuxun } }
		hd_dc_kit_ten_set_chaos_effect = { STATE = warlord }
	}
	else = {
		hd_dc_kit_ten_set_chaos_effect = { STATE = partisan }
	}
	hd_dc_kit_ten_chaos_apply_yearly_effect = yes
	random = {
		chance = 40	# = hd_dc_ten_chaos_event_chance
		hd_dc_kit_ten_chaos_event_effect = yes
	}
}

# 切换到 $STATE$（已是则不动）；玩家收到通知
hd_dc_kit_ten_set_chaos_effect = {
	if = {
		limit = { NOT = { has_character_modifier = hd_dc_kit_ten_chaos_$STATE$_modifier } }
		hd_dc_kit_ten_clear_chaos_effect = yes
		add_character_modifier = hd_dc_kit_ten_chaos_$STATE$_modifier
		if = {
			limit = { is_ai = no }
			trigger_event = hd_dc_ten.0041
		}
	}
}

hd_dc_kit_ten_clear_chaos_effect = {
	remove_character_modifier = hd_dc_kit_ten_chaos_consort_modifier
	remove_character_modifier = hd_dc_kit_ten_chaos_eunuch_modifier
	remove_character_modifier = hd_dc_kit_ten_chaos_warlord_modifier
	remove_character_modifier = hd_dc_kit_ten_chaos_partisan_modifier
	remove_variable = hd_dc_ten_boost_waiqi
	remove_variable = hd_dc_ten_boost_huanguan
	remove_variable = hd_dc_ten_boost_wuxun
}

# 本年乱象附带效果（"势力 +40%"变量 400 天，跨过朋党年度结算的先后）
hd_dc_kit_ten_chaos_apply_yearly_effect = {
	save_scope_as = hd_dc_ten_emperor
	if = {
		limit = { hd_dc_kit_ten_chaos_is_trigger = { STATE = consort } }
		set_variable = { name = hd_dc_ten_boost_waiqi value = yes days = 400 }
		var:hd_party_waiqi_leader ?= {
			add_character_modifier = {
				modifier = hd_dc_kit_ten_consort_leader_modifier
				months = 13
			}
		}
	}
	else_if = {
		limit = { hd_dc_kit_ten_chaos_is_trigger = { STATE = eunuch } }
		set_variable = { name = hd_dc_ten_boost_huanguan value = yes days = 400 }
		# 大臣对天子好感 −10
		hd_dc_kit_ten_collect_officials_effect = yes
		every_in_list = {
			list = hd_dc_ten_officials
			limit = {
				is_alive = yes
				NOT = { var:hd_party_my_group ?= flag:huanguan }
			}
			add_opinion = {
				target = scope:hd_dc_ten_emperor
				modifier = hd_dc_kit_ten_eunuch_meddling_opinion
				months = 13
			}
		}
		# 贪墨：宦官党前三名之一侵吞内帑（原版"侵吞国库"诱因）
		random = {
			chance = 30	# = hd_dc_ten_embezzle_chance
			random_in_list = {
				variable = hd_party_huanguan_top
				limit = {
					is_alive = yes
					is_imprisoned = no
					NOT = { this = scope:hd_dc_ten_emperor }
				}
				save_scope_as = hd_dc_ten_embezzler
			}
			if = {
				limit = {
					exists = scope:hd_dc_ten_embezzler
					gold > 0
				}
				save_scope_value_as = {
					name = hd_dc_ten_embezzled
					value = hd_dc_kit_ten_embezzle_value
				}
				remove_short_term_gold = scope:hd_dc_ten_embezzled
				scope:hd_dc_ten_embezzler = { add_gold = scope:hd_dc_ten_embezzled }
				hd_dc_kit_fire_effect = { CATALYST = catalyst_governor_embezzlement ACTOR = scope:hd_dc_ten_embezzler }
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_bad
						title = hd_dc_ten_embezzle_toast
						left_icon = scope:hd_dc_ten_embezzler
						custom_tooltip = hd_dc_ten_embezzle_toast_tt
					}
				}
			}
		}
	}
	else_if = {
		limit = { hd_dc_kit_ten_chaos_is_trigger = { STATE = warlord } }
		set_variable = { name = hd_dc_ten_boost_wuxun value = yes days = 400 }
		every_vassal = {
			limit = { is_landed = yes }
			add_character_modifier = {
				modifier = hd_dc_kit_ten_warlord_vassal_modifier
				months = 13
			}
		}
	}
	else_if = {
		limit = { hd_dc_kit_ten_chaos_is_trigger = { STATE = partisan } }
		hd_dc_kit_ten_collect_officials_effect = yes
		every_in_list = {
			list = hd_dc_ten_officials
			limit = { is_alive = yes }
			add_character_modifier = {
				modifier = hd_dc_kit_ten_partisan_official_modifier
				months = 13
			}
		}
	}
}

# 一网打尽：囚禁党魁与前三名，势力归零并禁用十年，全党好感 −50；按该党原势力占比的概率由该党最强封臣起兵清君侧

		name = hd_dc_ten_purged_share

			}
		}
	}
	var:hd_party_$G$_leader ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { this = scope:hd_dc_ten_emperor }
			}
			add_to_list = hd_dc_ten_purge_list
		}
	}
	every_in_list = {
		variable = hd_party_$G$_top
		limit = {
			is_alive = yes
			NOT = { this = scope:hd_dc_ten_emperor }
			NOT = { is_in_list = hd_dc_ten_purge_list }
		}
		add_to_list = hd_dc_ten_purge_list
	}
	every_in_list = {
		list = hd_dc_ten_purge_list
		hd_dc_kit_ten_strip_office_effect = yes
		scope:hd_dc_ten_emperor = {
			imprison = {
				target = prev
				type = dungeon
			}
		}
	}
	every_in_list = {
		variable = hd_party_$G$_members
		limit = {
			is_alive = yes
			NOT = { this = scope:hd_dc_ten_emperor }
		}
		add_opinion = {
			target = scope:hd_dc_ten_emperor
			modifier = hd_dc_kit_ten_purge_opinion
