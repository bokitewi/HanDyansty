# Shared palace-crisis entry point.  It is deliberately idempotent because it
# can be called by Yellow Turban resolution and by Liu Hong's death on-action.
hd_force_start_palace_crisis_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_yellow_turban_han_fell }
			NOT = { exists = global_var:hd_palace_crisis_queued }
			NOT = { exists = global_var:hd_palace_crisis_started }
			NOT = { exists = global_var:hd_palace_crisis_completed }
		}
		set_global_variable = hd_palace_crisis_force_entry
		hd_try_start_palace_crisis_effect = yes
		remove_global_variable = hd_palace_crisis_force_entry
	}
}

hd_try_start_palace_crisis_effect = {
	if = {
		limit = {
			OR = {
				hd_palace_crisis_conditions_met = yes
				AND = {
					exists = global_var:hd_palace_crisis_force_entry
					NOT = { exists = global_var:hd_yellow_turban_han_fell }
					NOT = { exists = global_var:hd_palace_crisis_queued }
					NOT = { exists = global_var:hd_palace_crisis_started }
					NOT = { exists = global_var:hd_palace_crisis_completed }
				}
			}
		}
		set_global_variable = hd_palace_crisis_queued

		# Prefer the player-controlled historical principals.  If none is human,
		# He Jin owns the historical AI-facing controller event.
		if = {
			limit = { character:he_jin ?= { is_alive = yes is_ai = no } }
			character:he_jin = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:yuan_shao ?= { is_alive = yes is_ai = no } }
			character:yuan_shao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes is_ai = no } }
			character:cao_cao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:he_jin ?= { is_alive = yes } }
			character:he_jin = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:yuan_shao ?= { is_alive = yes } }
			character:yuan_shao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes } }
			character:cao_cao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else = {
			hd_palace_crisis_cleanup_effect = yes
		}
	}
}

# The global pointer is only a pre-creation ownership token. Stale delayed
# events cannot consume a replacement dispatch after this token changes.
hd_anti_regent_clear_pending_dispatch_effect = {
	if = {
		limit = { exists = global_var:hd_anti_regent_pending_leader }
		global_var:hd_anti_regent_pending_leader = {
			if = {
				limit = { is_alive = yes }
				if = {
					limit = { this = character:yuan_shao }
					hd_yuan_abort_bohai_migration_effect = yes
				}
				remove_variable = hd_palace_deposition_regent
			}
		}
	}
	remove_global_variable = hd_anti_regent_pending_leader
}

hd_luoyang_noble_purge_effect = {
	# Every estate in Luoyang is an explicit target. The regent and his court
	# have already moved to Chang'an, so every noble family still here is purged.
	title:b_uuii_p_787.title_province = {
		every_province_domicile = {
			limit = { is_domicile_type = estate }
			owner ?= {
				save_scope_as = hd_luoyang_estate_owner
				house ?= {
						every_house_member = {
							limit = {
								is_alive = yes
								NOT = { this = scope:hd_luoyang_estate_owner }
							}
							death = { death_reason = death_execution killer = root }
						}
					}
					every_courtier_or_guest = {
						limit = { is_alive = yes }
						death = { death_reason = death_execution killer = root }
					}
					if = {
						limit = { is_alive = yes }
						death = { death_reason = death_execution killer = root }
					}
				}
		}
	}
}

hd_anti_regent_cleanup_effect = {
	hd_anti_regent_clear_pending_dispatch_effect = yes
	if = {
		limit = { has_variable = hd_anti_regent_confederation }
		var:hd_anti_regent_confederation = {
			if = {
				limit = { confederation_type = confederation_type:hd_anti_regent_league_type }
				disband_confederation = yes
			}
		}
		remove_variable = hd_anti_regent_confederation
	}
	if = {
		limit = { has_variable = hd_anti_regent_target }
		var:hd_anti_regent_target = {
			if = {
				limit = { is_alive = yes }
				remove_character_flag = hd_current_han_regent
				remove_character_flag = hd_anti_regent_invitations_complete
				remove_variable = hd_anti_regent_leader
			}
		}
	}
	if = {
		limit = { has_variable_list = hd_anti_regent_members }
		every_in_list = {
			variable = hd_anti_regent_members
			remove_variable = hd_anti_regent_contribution
			remove_character_flag = hd_anti_regent_spoils_round
			remove_character_flag = hd_anti_regent_league_member
		}
		clear_variable_list = hd_anti_regent_members
	}
	if = {
		limit = { is_alive = yes }
		remove_character_flag = hd_anti_regent_league_leader
		remove_character_flag = hd_anti_regent_league_member
		remove_character_flag = hd_yuan_bohai_arrived
		remove_character_flag = hd_anti_regent_first_war_started
		remove_character_flag = hd_anti_regent_first_war_fallback_ready
		remove_character_flag = hd_anti_regent_second_war_started
		remove_character_flag = hd_anti_regent_invitations_dispatched
		remove_character_flag = hd_anti_regent_invitations_pending
		remove_character_flag = hd_anti_regent_invitations_complete
		remove_character_flag = hd_anti_regent_creation_pending
		remove_character_flag = hd_anti_regent_creation_delay_scheduled
		remove_character_flag = hd_anti_regent_creation_started
		remove_character_flag = hd_anti_regent_cao_pending
		remove_variable = hd_anti_regent_target
		remove_variable = hd_anti_regent_pending_invites
		remove_variable = hd_anti_regent_contribution
	}
	character:cao_cao ?= {
		if = { limit = { is_alive = yes } remove_character_flag = hd_anti_regent_cao_pending }
	}
	remove_global_variable = hd_anti_regent_league_leader
}
