# At the fixed 189 deadline, unresolved Yellow Turban wars no longer block the
# historical palace transition. A Yellow Turban victory remains authoritative.
hd_force_start_palace_crisis_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:hd_yellow_turban_han_fell }
			NOT = { exists = global_var:hd_yellow_turban_outcome_resolved }
		}
		hd_yellow_turban_campaign_failed_effect = yes
	}
	if = {
		limit = { NOT = { exists = global_var:hd_yellow_turban_han_fell } }
		character:liu_hong_3 ?= { add_character_flag = hd_palace_crisis_liu_hong_death_signal }
		hd_try_start_palace_crisis_effect = yes
	}
}

# Shared palace-crisis entry point. It is deliberately idempotent because it
# can be called by Yellow Turban resolution and by Liu Hong's death on-action.
hd_try_start_palace_crisis_effect = {
	if = {
		limit = { hd_palace_crisis_conditions_met = yes }
		set_global_variable = hd_palace_crisis_queued

		# Prefer the player-controlled historical principals.  If none is human,
		# He Jin owns the historical AI-facing controller event.
		if = {
			limit = { character:he_jin ?= { is_alive = yes is_ai = no } }
			character:he_jin = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:yuan_shao ?= { is_alive = yes is_ai = no } }
			character:yuan_shao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes is_ai = no } }
			character:cao_cao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:he_jin ?= { is_alive = yes } }
			character:he_jin = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:yuan_shao ?= { is_alive = yes } }
			character:yuan_shao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes } }
			character:cao_cao = {
				add_character_flag = hd_palace_crisis_controller
				trigger_event = { id = hd_palace_crisis.0001 days = 35 }
			}
		}
		else = {
			hd_palace_crisis_cleanup_effect = yes
		}
	}
}

# A queued controller owns exactly one delivery attempt. Consume that ownership
# before redispatching so a later duplicate callback from the old event cannot
# cancel the replacement. The second failure closes the chain cleanly.
hd_palace_recover_queued_effect = {
	if = {
		limit = {
			has_character_flag = hd_palace_crisis_controller
			exists = global_var:hd_palace_crisis_queued
			NOT = { exists = global_var:hd_palace_crisis_started }
		}
		if = {
			limit = { is_alive = yes }
			remove_character_flag = hd_palace_crisis_controller
		}
		remove_global_variable = hd_palace_crisis_queued
		if = {
			limit = { NOT = { exists = global_var:hd_palace_crisis_queue_retry_used } }
			set_global_variable = hd_palace_crisis_queue_retry_used
			hd_try_start_palace_crisis_effect = yes
		}
		else = { hd_palace_crisis_cleanup_effect = yes }
	}
}

# The controller flag is also the ownership token after the opening event. A
# failed delayed node and on_death may both call this effect; only the first wins.
hd_palace_recover_started_controller_effect = {
	if = {
		limit = {
			has_character_flag = hd_palace_crisis_controller
			exists = global_var:hd_palace_crisis_started
		}
		if = {
			limit = { is_alive = yes }
			remove_character_flag = hd_palace_crisis_controller
		}
		hd_palace_continue_after_controller_death_effect = yes
	}
}

# If the player-controlled Yuan Shao or Cao Cao dies after the palace chain
# has already opened, hand the next node back to the historical sequence
# instead of leaving the global chain marked as started with no event owner.
hd_palace_continue_after_controller_death_effect = {
	if = {
		limit = { exists = global_var:hd_palace_crisis_started }
		if = {
			limit = { character:he_jin ?= { is_alive = yes } }
			character:he_jin = {
				trigger_event = { id = hd_palace_crisis.0004 days = 35 }
			}
		}
		else = {
			set_global_variable = hd_palace_he_jin_killed
			if = {
				limit = { character:yuan_shao ?= { is_alive = yes } }
				character:yuan_shao = {
					trigger_event = { id = hd_palace_crisis.0005 days = 35 }
				}
			}
			else_if = {
				limit = { character:cao_cao ?= { is_alive = yes } }
				character:cao_cao = {
					trigger_event = { id = hd_palace_crisis.0005 days = 35 }
				}
			}
			else = {
				hd_palace_purge_effect = yes
				if = {
					limit = {
						title:k_uuii_liangzhou = { has_variable = hd_palace_liangzhou_invited }
						title:k_uuii_liangzhou.holder ?= { is_alive = yes has_character_flag = hd_palace_liangzhou_marching }
					}
					title:k_uuii_liangzhou.holder = { hd_palace_try_finalize_liangzhou_effect = yes }
				}
				else = { hd_palace_crisis_cleanup_effect = yes }
			}
		}
	}
}

hd_liangzhou_start_march_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			NOT = { has_character_flag = hd_palace_liangzhou_marching }
		}
		add_character_flag = hd_palace_liangzhou_marching
		set_variable = { name = hd_palace_liangzhou_marcher value = this }
		title:k_uuii_liangzhou = {
			set_variable = { name = hd_palace_liangzhou_marcher value = root }
		}

		# Every landed vassal below the Liangzhou title receives a real decision.
		# The eventual title independence is handled only after the marcher's arrival.
		every_vassal_or_below = {
			limit = {
				is_alive = yes
				is_landed = yes
				NOT = { this = root }
			}
			set_variable = { name = hd_palace_liangzhou_inviter value = root }
			trigger_event = { id = hd_palace_crisis.0010 days = 2 }
		}

		# Administrative estates cannot travel with a travel plan. Move both the
		# ruler and domicile directly to Luoyang, then run the same finalization.
		set_location = {
			location = title:b_uuii_p_787.title_province
			stick_to_location = yes
		}
		domicile ?= { move_domicile = title:b_uuii_p_787.title_province }
		add_character_flag = hd_palace_liangzhou_arrived
		hd_palace_try_finalize_liangzhou_effect = yes
	}
}

hd_palace_try_finalize_liangzhou_effect = {
	save_scope_as = hd_palace_liangzhou_finalize_ruler
	if = {
		limit = {
			scope:hd_palace_liangzhou_finalize_ruler = {
				has_character_flag = hd_palace_liangzhou_marching
				has_character_flag = hd_palace_liangzhou_arrived
			}
			exists = global_var:hd_palace_purge_completed
			exists = title:k_uuii_liangzhou.holder
			title:k_uuii_liangzhou = { var:hd_palace_liangzhou_marcher = scope:hd_palace_liangzhou_finalize_ruler }
			scope:hd_palace_liangzhou_finalize_ruler = {
				NOT = { has_character_flag = hd_palace_liangzhou_settled }
			}
		}
		scope:hd_palace_liangzhou_finalize_ruler = { add_character_flag = hd_palace_liangzhou_settled }

		# Only Sili's de jure dukes become direct vassals of the new governor.
		# Counts and barons keep their existing hierarchy under those dukes.
		title:k_uuii_sili = {
			every_in_de_jure_hierarchy = {
				continue = { tier > tier_duchy }
				limit = {
					tier = tier_duchy
					exists = holder
					holder = {
						is_alive = yes
						is_landed = yes
						NOT = { this = scope:hd_palace_liangzhou_finalize_ruler }
					}
				}
				holder = { add_to_list = hd_palace_sili_duchy_holders }
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_palace_sili_change
			add_claim_on_loss = no
		}
		title:k_uuii_sili = {
			change_title_holder_include_vassals = {
				holder = scope:hd_palace_liangzhou_finalize_ruler
				change = scope:hd_palace_sili_change
			}
		}
		every_in_list = {
			list = hd_palace_sili_duchy_holders
			limit = { NOT = { liege = scope:hd_palace_liangzhou_finalize_ruler } }
			change_liege = {
				liege = scope:hd_palace_liangzhou_finalize_ruler
				change = scope:hd_palace_sili_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_palace_sili_change
		scope:hd_palace_liangzhou_finalize_ruler = {
			random_owned_story = {
				type = hd_story_dong_zhuo_life
				set_variable = { name = hd_dong_zhuo_life_phase value = flag:hd_dong_zhuo_life_sili }
			}
		}

		# The Liangzhou ruler is the regent; the diarchy itself belongs to the
		# current Han emperor, not to the incoming marcher.
		scope:hd_palace_liangzhou_finalize_ruler = { hd_force_dong_regency_effect = yes }
		scope:hd_palace_liangzhou_finalize_ruler = {
			remove_character_flag = hd_palace_liangzhou_marching
			trigger_event = { id = hd_palace_crisis.0015 days = 365 }
		}
	}
}

hd_luoyang_noble_purge_effect = {
	# Every estate in Luoyang is an explicit target. The regent and his court
	# have already moved to Chang'an, so every noble family still here is purged.
	title:b_uuii_p_787.title_province = {
		every_province_domicile = {
			limit = { is_domicile_type = estate }
			owner ?= {
				save_scope_as = hd_luoyang_estate_owner
				house ?= {
						every_house_member = {
							limit = {
								is_alive = yes
								NOT = { this = scope:hd_luoyang_estate_owner }
							}
							death = { death_reason = death_execution killer = root }
						}
					}
					every_courtier_or_guest = {
						limit = { is_alive = yes }
						death = { death_reason = death_execution killer = root }
					}
					if = {
						limit = { is_alive = yes }
						death = { death_reason = death_execution killer = root }
					}
				}
		}
	}
}

hd_anti_regent_first_war_victory_effect = {
	save_scope_as = hd_anti_regent_first_war_recipient
	if = {
		limit = {
			scope:hd_anti_regent_first_war_recipient = {
				is_alive = yes
			}
			OR = {
				title:d_henan.holder != scope:hd_anti_regent_first_war_recipient
				title:d_luoyang.holder != scope:hd_anti_regent_first_war_recipient
			}
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = hd_anti_regent_first_war_change
			add_claim_on_loss = no
		}
		if = {
			limit = { title:d_henan.holder != scope:hd_anti_regent_first_war_recipient }
			title:d_henan = {
				change_title_holder_include_vassals = {
					holder = scope:hd_anti_regent_first_war_recipient
					change = scope:hd_anti_regent_first_war_change
				}
			}
		}
		if = {
			limit = { title:d_luoyang.holder != scope:hd_anti_regent_first_war_recipient }
			title:d_luoyang = {
				change_title_holder_include_vassals = {
					holder = scope:hd_anti_regent_first_war_recipient
					change = scope:hd_anti_regent_first_war_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd_anti_regent_first_war_change
	}
}

hd_anti_regent_cleanup_effect = {
	hd_anti_regent_clear_pending_dispatch_effect = yes
	every_confederation = {
		limit = { confederation_type = confederation_type:hd_anti_regent_league_type }
		every_confederation_member = {
			add_to_list = hd_anti_regent_confederates_to_remove
		}
	}
	every_in_list = {
		list = hd_anti_regent_confederates_to_remove
		if = {
			limit = {
				exists = confederation
				confederation = { confederation_type = confederation_type:hd_anti_regent_league_type }
			}
			confederation = { remove_confederation_member = root }
		}
	}
	if = {
		limit = { has_variable = hd_anti_regent_target }
		var:hd_anti_regent_target = {
			if = {
				limit = { is_alive = yes }
				remove_character_flag = hd_current_han_regent
				remove_variable = hd_anti_regent_leader
			}
		}
	}
	if = {
		limit = { has_variable_list = hd_anti_regent_members }
		every_in_list = {
			variable = hd_anti_regent_members
			remove_variable = hd_anti_regent_contribution
			remove_character_flag = hd_anti_regent_spoils_round
			remove_character_flag = hd_anti_regent_league_member
		}
		clear_variable_list = hd_anti_regent_members
	}
	if = {
		limit = { has_variable_list = hd_anti_regent_invitees }
		every_in_list = {
			variable = hd_anti_regent_invitees
			remove_character_flag = hd_anti_regent_invitation_pending
			remove_character_flag = hd_anti_regent_join_pending
			remove_variable = hd_anti_regent_inviter
		}
		clear_variable_list = hd_anti_regent_invitees
	}
	if = {
		limit = { is_alive = yes }
		remove_character_flag = hd_anti_regent_league_leader
		remove_character_flag = hd_anti_regent_league_member
		remove_character_flag = hd_anti_regent_dispatch_ready
		remove_character_flag = hd_yuan_bohai_arrived
		remove_character_flag = hd_anti_regent_first_war_started
		remove_character_flag = hd_anti_regent_second_war_started
		remove_character_flag = hd_anti_regent_invitations_dispatched
		remove_character_flag = hd_anti_regent_invitations_pending
		remove_character_flag = hd_anti_regent_creation_pending
		remove_character_flag = hd_anti_regent_creation_delay_scheduled
		remove_character_flag = hd_anti_regent_creation_started
		remove_character_flag = hd_anti_regent_cao_pending
		remove_character_flag = hd_anti_regent_takeover_pending
		remove_variable = hd_anti_regent_target
		remove_variable = hd_anti_regent_contribution
	}
	character:cao_cao ?= {
		if = { limit = { is_alive = yes } remove_character_flag = hd_anti_regent_cao_pending }
	}
	remove_global_variable = hd_anti_regent_league_leader
}
