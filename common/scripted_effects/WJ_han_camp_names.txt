# Only the just-created native camp, never inherited/existing/manual company names.
WJ_finalize_new_han_camp_name_effect = {
 if = {
  limit = {
   exists = scope:new_title exists = scope:new_landless_adventurer
   this = scope:new_landless_adventurer
   WJ_han_name_style_trigger = yes
   NOT = { has_character_flag = new_laamp_title_name_entered_text }
   scope:new_title = { holder = scope:new_landless_adventurer }
  }
  if = {
   limit = { exists = scope:new_landless_adventurer_location.county }
   scope:new_landless_adventurer_location.county = { save_scope_as = wj_camp_county }
  }
  if = {
   limit = { exists = scope:wj_camp_county.duchy }
   scope:wj_camp_county.duchy = { save_scope_as = wj_camp_commandery }
  }
  # Native set_title_name_dynamic resolves the name once. Later purpose changes
  # change the category only; they must not overwrite the player's company name.
  random_list = {
   10 = { trigger = { exists = scope:new_landless_adventurer_house } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_house } }
   10 = { trigger = { exists = scope:new_landless_adventurer_house } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_house_followers } }
   10 = { trigger = { exists = scope:new_landless_adventurer_house } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_house_retinue } }
   10 = { trigger = { exists = scope:wj_camp_county } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_county } }
   10 = { trigger = { exists = scope:wj_camp_county } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_county_followers } }
   10 = { trigger = { exists = scope:wj_camp_county } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_county_companions } }
   10 = { trigger = { exists = scope:wj_camp_commandery } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_commandery } }
   10 = { trigger = { exists = scope:wj_camp_commandery } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_commandery_followers } }
   10 = { trigger = { exists = scope:wj_camp_commandery } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_commandery_companions } }
   10 = { trigger = { exists = scope:wj_camp_commandery exists = scope:new_landless_adventurer_house } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_commandery_house } }
   10 = { trigger = { exists = scope:wj_camp_county exists = scope:new_landless_adventurer_house } scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_county_house } }
   10 = { scope:new_title = { set_title_name_dynamic = WJ_han_camp_name_personal } }
  }
 }
}
