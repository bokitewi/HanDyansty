# 五胡内附 / 僚人入蜀 公共效果（见 docs/五胡内附_僚人入蜀_设计草案.md）。
# 族群统一以 flag 表示：flag:xiongnu / xianbei / jieren / diren / qiang / wuhuan。
# 约定：族群总是先存成 scope:hd_wuhu_tribe（save_scope_value_as），再传入触发器/效果。

############################
# 族群数据
############################

### 由当前 scope（角色或伯爵领）的文化推出族群，存为 scope:hd_wuhu_tribe
hd_wuhu_save_tribe_from_culture_effect = {
	if = {
		limit = { culture = culture:xiongnu }
		save_scope_value_as = { name = hd_wuhu_tribe value = flag:xiongnu }
	}
	else_if = {
		limit = { culture = culture:xianbei }
		save_scope_value_as = { name = hd_wuhu_tribe value = flag:xianbei }
	}
	else_if = {
		limit = { culture = culture:jieren }
		save_scope_value_as = { name = hd_wuhu_tribe value = flag:jieren }
	}
	else_if = {
		limit = { culture = culture:diren }
		save_scope_value_as = { name = hd_wuhu_tribe value = flag:diren }
	}
	else_if = {
		limit = { culture = culture:wuhuan }
		save_scope_value_as = { name = hd_wuhu_tribe value = flag:wuhuan }
	}
	else_if = {
		limit = { hd_wuhu_is_qiang_culture_trigger = yes }
		save_scope_value_as = { name = hd_wuhu_tribe value = flag:qiang }
	}
}

### 族群的文化与信仰，存为 scope:hd_wuhu_culture / scope:hd_wuhu_rite。
# $COUNTY$：安置的伯爵领。羌按安置地取最近的羌种文化：本地 → 邻接 → 羌安置区内任意羌种伯爵领 → 种羌。
hd_wuhu_save_tribe_culture_rite_effect = {
	if = {
		limit = { scope:hd_wuhu_tribe = flag:xiongnu }
		culture:xiongnu = { save_scope_as = hd_wuhu_culture }
		rite:tengri_pagan = { save_scope_as = hd_wuhu_rite }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:xianbei }
		culture:xianbei = { save_scope_as = hd_wuhu_culture }
		rite:xianbei_polytheism = { save_scope_as = hd_wuhu_rite }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:jieren }
		culture:jieren = { save_scope_as = hd_wuhu_culture }
		rite:mazdayasna = { save_scope_as = hd_wuhu_rite }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:diren }
		culture:diren = { save_scope_as = hd_wuhu_culture }
		rite:xirongfangling = { save_scope_as = hd_wuhu_rite }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:wuhuan }
		culture:wuhuan = { save_scope_as = hd_wuhu_culture }
		rite:tengri_pagan = { save_scope_as = hd_wuhu_rite }
	}
	else = {
		rite:xirongfangling = { save_scope_as = hd_wuhu_rite }
		$COUNTY$ = {
			if = {
				limit = { hd_wuhu_is_qiang_culture_trigger = yes }
				culture = { save_scope_as = hd_wuhu_culture }
			}
			else_if = {
				limit = {
					any_neighboring_county = { hd_wuhu_is_qiang_culture_trigger = yes }
				}
				random_neighboring_county = {
					limit = { hd_wuhu_is_qiang_culture_trigger = yes }
					culture = { save_scope_as = hd_wuhu_culture }
				}
			}
			else_if = {
				limit = {
					any_county_in_region = {
						region = hd_wuhu_qiang_settle_region
						hd_wuhu_is_qiang_culture_trigger = yes
					}
				}
				random_county_in_region = {
					region = hd_wuhu_qiang_settle_region
					limit = { hd_wuhu_is_qiang_culture_trigger = yes }
					culture = { save_scope_as = hd_wuhu_culture }
				}
			}
			else = {
				culture:zhongqiang = { save_scope_as = hd_wuhu_culture }
			}
		}
	}
}

# 胡首生成与安置
############################

### 在 $LOCATION$（province）生成一名胡首，使用 scope:hd_wuhu_culture / scope:hd_wuhu_rite，存为 scope:hd_wuhu_chief
hd_wuhu_create_chief_effect = {
	create_character = {
		template = new_commander_character
		age = { 25 40 }
		gender_female_chance = 0
		dynasty = generate
		culture = scope:hd_wuhu_culture
		rite = scope:hd_wuhu_rite
		location = $LOCATION$
		save_scope_as = hd_wuhu_chief
	}
}

### Character（胡首）：按族群设政体。匈奴、鲜卑、乌桓 = 游牧制（补建游牧头衔/住所）；氐、羌、羯 = 部族制
hd_wuhu_apply_chief_government_effect = {
	if = {
		limit = { hd_wuhu_tribe_is_nomadic_trigger = { TRIBE = scope:hd_wuhu_tribe } }
		change_government = nomad_government
		if = {
			limit = {
				is_landed = yes
				NOT = { any_held_title = { is_nomad_title = yes } }
			}
			save_scope_as = hd_wuhu_nomad_holder
			create_nomad_title = {
				name = nomad_title_name
				holder = scope:hd_wuhu_nomad_holder
				government = nomad_government
			}
		}
	}
	else = {
		change_government = tribal_government
	}
}

### Character（胡首）：写入内附标记（值 = 族群 flag）
hd_wuhu_mark_settled_effect = {
	set_variable = {
		name = hd_wuhu_settled
		value = scope:hd_wuhu_tribe
	}
}

### County：文化与信仰改为 scope:hd_wuhu_culture / scope:hd_wuhu_rite
hd_wuhu_flip_county_culture_rite_effect = {
	set_county_culture = scope:hd_wuhu_culture
	set_county_rite = scope:hd_wuhu_rite
}

### 史实胡首若在别处有地，本人放弃原有领地（交给主继承人，没有则交给一名新生成的同族首领）
hd_wuhu_chief_abandon_lands_effect = {
	if = {
		limit = { is_landed = yes }
		save_scope_as = hd_wuhu_abandoning_chief
		if = {
			limit = {
				exists = primary_heir
				primary_heir = { is_alive = yes }
			}
			primary_heir = { save_scope_as = hd_wuhu_abandon_receiver }
		}
		else = {
			create_character = {
				template = new_commander_character
				age = { 25 40 }
				gender_female_chance = 0
				dynasty = generate
				culture = scope:hd_wuhu_abandoning_chief.culture
				rite = scope:hd_wuhu_abandoning_chief.rite
				location = scope:hd_wuhu_abandoning_chief.capital_province
				save_scope_as = hd_wuhu_abandon_receiver
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_wuhu_abandon_change
			add_claim_on_loss = no
		}
		every_held_title = {
			limit = { is_landless_type_title = no }
			change_title_holder = {
				holder = scope:hd_wuhu_abandon_receiver
				change = scope:hd_wuhu_abandon_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_wuhu_abandon_change
	}
}

hd_wuhu_tribe_conversion_add_effect = {
	if = {
		limit = { scope:hd_wuhu_tribe = flag:xiongnu }
		change_global_variable = { name = hd_wuhu_conv_xiongnu add = 1 }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:xianbei }
		change_global_variable = { name = hd_wuhu_conv_xianbei add = 1 }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:jieren }
		change_global_variable = { name = hd_wuhu_conv_jieren add = 1 }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:diren }
		change_global_variable = { name = hd_wuhu_conv_diren add = 1 }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:wuhuan }
		change_global_variable = { name = hd_wuhu_conv_wuhuan add = 1 }
	}
	else_if = {
		limit = { scope:hd_wuhu_tribe = flag:qiang }
		change_global_variable = { name = hd_wuhu_conv_qiang add = 1 }
	}
}

### Character（内附胡首）：每年 10% 改一块安置区内、与本族接壤、属于其宗主领国的伯爵领。每族累计 ≤ 20。
hd_wuhu_yearly_conversion_effect = {
	save_scope_value_as = { name = hd_wuhu_tribe value = var:hd_wuhu_settled }
	if = {
		limit = { is_independent_ruler = yes }
		save_scope_as = hd_wuhu_conv_liege
	}
	else = {
		liege = { save_scope_as = hd_wuhu_conv_liege }
	}
	if = {
		limit = {
			hd_wuhu_tribe_conversion_count < hd_wuhu_convert_cap
			scope:hd_wuhu_conv_liege = {
				any_realm_county = {
					hd_wuhu_convertible_county_trigger = { TRIBE = scope:hd_wuhu_tribe LIEGE = scope:hd_wuhu_conv_liege }
				}
			}
		}
		random = {
			chance = hd_wuhu_convert_chance
			scope:hd_wuhu_conv_liege = {
				random_realm_county = {
					limit = {
						hd_wuhu_convertible_county_trigger = { TRIBE = scope:hd_wuhu_tribe LIEGE = scope:hd_wuhu_conv_liege }
					}
					weight = {
						base = 100
						modifier = {
							factor = 0.1
							culture = { has_cultural_pillar = heritage_chinese }
						}
					}
					save_scope_as = hd_wuhu_conv_county
					hd_wuhu_save_tribe_culture_rite_effect = { COUNTY = scope:hd_wuhu_conv_county }
					set_county_culture = scope:hd_wuhu_culture
					random = {
						chance = 50
						set_county_rite = scope:hd_wuhu_rite
					}
					add_county_modifier = {
						modifier = hd_foreign_settlement_county_modifier
						years = 20
					}
					hd_wuhu_tribe_conversion_add_effect = yes
				}
			}
		}
	}
}
