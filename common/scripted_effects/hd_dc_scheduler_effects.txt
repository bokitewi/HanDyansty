# Situation scope. Fixed yearly opportunity for the two authored crisis types.
hd_dc_try_crisis_events_effect = {
	if = {
		limit = {
			exists = title:h_china.holder
			OR = {
				OR = { situation_current_phase = situation_dynastic_cycle_phase_instability situation_current_phase = situation_dynastic_cycle_phase_hd_collapse }
				situation_current_phase = situation_dynastic_cycle_phase_instability_conquest
			}
			NOT = { var:hd_dc_crisis_year = current_year }
		}
		set_variable = { name = hd_dc_crisis_year value = current_year }
		if = {
			limit = { NOR = { has_variable = hd_dc_acute_instance_id has_variable = hd_dc_pending_instance has_variable = hd_dc_pending_party_case } }
			random = {
				chance = 10
				hd_dc_initialize_scope_effect = yes
				hd_dc_snapshot_party_pool_effect = yes
				remove_variable = hd_dc_selected_type
				random_list = {
					100 = {
						trigger = { NOT = { has_variable = hd_dc_roy_cooldown } any_situation_participant = { hd_dc_royal_candidate_trigger = yes } }
						set_variable = { name = hd_dc_selected_type value = flag:roy }
					}
					100 = {
						trigger = {
							NOT = { has_variable = hd_dc_par_cooldown }
							hd_dc_par_snapshot_valid_trigger = yes
							var:hd_dc_par_ordered_hostile_pair_count > 0
							any_in_list = { variable = hd_dc_counties hd_dc_party_case_county_trigger = yes }
						}
						set_variable = { name = hd_dc_selected_type value = flag:par }
					}
				}
				if = {
					limit = { var:hd_dc_selected_type = flag:roy }
					hd_dc_boot_royal_effect = yes
					if = {
						limit = { exists = scope:hd_dc_instance scope:hd_dc_instance.var:hd_dc_stage = flag:acute }
						scope:hd_dc_instance = { hd_dc_dispatch_roy01_effect = yes }
					}
				}
				else_if = {
					limit = { var:hd_dc_selected_type = flag:par }
					hd_dc_select_party_pair_effect = yes
					hd_dc_prepare_party_policy_case_effect = yes
				}
				remove_variable = hd_dc_selected_type
			}
		}
	}
}
