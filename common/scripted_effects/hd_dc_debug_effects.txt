# Character entry. W/H arguments are display-point deltas, never raw writes.
hd_dc_debug_change_points_effect = {
	if = {
		limit = { debug_only = yes is_ai = no exists = situation:dynastic_cycle }
		situation:dynastic_cycle = {
			hd_dc_change_points_effect = {
				W = { value = $W$ multiply = hd_dc_pressure_max divide = 100 }
				H = { value = $H$ multiply = hd_dc_prosperity_max divide = 100 }
				SOURCE = debug
			}
			hd_dc_apply_pressure_sources_effect = yes
		}
	}
}

# An explicit, once-per-player scene, not part of natural event generation.
# The rival relationship is created today; no prior crime or signature is faked.
hd_dc_debug_prepare_ord19_effect = {
	if = {
		limit = {
			debug_only = yes is_ai = no is_alive = yes is_adult = yes is_landed = yes is_imprisoned = no
			exists = capital_province
			any_character_situation = { this = situation:dynastic_cycle }
			NOT = { has_variable = hd_dc_debug_ord19_prepared }
			NOT = { has_variable = hd_dc_pending_ordinary }
		}
		save_scope_as = hd_dc_debug_ruler
		create_character = {
			age = 45 gender_female_chance = 0 dynasty = none
			culture = root.culture faith = root.faith location = root.capital_province
			save_scope_as = hd_dc_debug_parent
		}
		create_character = {
			age = 20 gender_female_chance = 0 dynasty = none
			culture = root.culture faith = root.faith location = root.capital_province
			save_scope_as = hd_dc_debug_visitor
		}
		create_character = {
			age = 40 gender_female_chance = 0 dynasty = none
			culture = root.culture faith = root.faith location = root.capital_province
			save_scope_as = hd_dc_debug_guarantor
		}
		add_courtier = scope:hd_dc_debug_parent
		add_courtier = scope:hd_dc_debug_visitor
		add_courtier = scope:hd_dc_debug_guarantor
		scope:hd_dc_debug_visitor = { set_father = scope:hd_dc_debug_parent }
		set_relation_rival = { target = scope:hd_dc_debug_parent reason = hd_dc_debug_rival_reason }
		add_opinion = { target = scope:hd_dc_debug_guarantor modifier = hd_dc_debug_trust_opinion }
		scope:hd_dc_debug_visitor = { add_opinion = { target = scope:hd_dc_debug_guarantor modifier = hd_dc_debug_trust_opinion } }
		scope:hd_dc_debug_guarantor = {
			add_opinion = { target = scope:hd_dc_debug_ruler modifier = hd_dc_debug_trust_opinion }
			add_opinion = { target = scope:hd_dc_debug_visitor modifier = hd_dc_debug_trust_opinion }
		}
		add_gold = 300
		set_variable = { name = hd_dc_debug_ord19_prepared value = yes }
		set_variable = { name = hd_dc_debug_ord19_visitor value = scope:hd_dc_debug_visitor }
		set_variable = { name = hd_dc_debug_ord19_guarantor value = scope:hd_dc_debug_guarantor }
	}
}

# Targets use the displayed 0..100 scale. Reuse the sole incremental writer.
hd_dc_debug_start_roy_effect = {
	if = {
		limit = { hd_dc_debug_roy_entry_trigger = yes }
		save_scope_as = hd_dc_debug_requester
		save_scope_value_as = { name = hd_dc_debug_manual value = flag:roy }
		situation:dynastic_cycle = {
			hd_dc_boot_royal_effect = yes
			scope:hd_dc_instance ?= { hd_dc_dispatch_roy01_effect = yes }
		}
		clear_saved_scope = hd_dc_debug_manual
	}
}

hd_dc_debug_start_par_effect = {
	if = {
		limit = { hd_dc_debug_par_entry_trigger = yes }
		save_scope_as = hd_dc_debug_requester
		save_scope_value_as = { name = hd_dc_debug_manual value = flag:par }
		situation:dynastic_cycle = {
			hd_dc_select_party_pair_effect = yes
			hd_dc_prepare_party_policy_case_effect = yes
		}
		clear_saved_scope = hd_dc_debug_manual
	}
}

hd_dc_debug_request_support_effect = {
	if = {
		limit = { debug_only = yes is_ai = no }
		save_scope_as = recipient
		clear_saved_scope = hd_dc_debug_claimant
		situation:dynastic_cycle.var:hd_dc_latest_roy_instance ?= {
			every_in_list = {
				variable = hd_dc_actor_stories
				limit = { NOT = { exists = scope:hd_dc_debug_claimant } var:hd_dc_actor = { save_temporary_scope_as = actor hd_dc_royal_offer_pair_trigger = yes } }
				var:hd_dc_actor = { save_scope_as = hd_dc_debug_claimant }
			}
		}
		scope:hd_dc_debug_claimant ?= { save_scope_as = actor hd_dc_send_royal_support_offer_effect = yes }
	}
}

				SOURCE = debug
			}
			hd_dc_apply_pressure_sources_effect = yes
		}
	}
}
