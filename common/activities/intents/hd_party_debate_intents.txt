	scripted_animation = { animation = debating }

hd_debate_support_intent = {
	icon = intent_diplomacy
	scripted_animation = { animation = admiration }
	auto_complete = yes
	ai_will_do = {
		value = 0
		if = {
			limit = { scope:activity ?= { has_activity_option = { category = hd_debate_target option = hd_target_zunwang } } }
			add = hd_party_stance_weight_zunwang_value
		}
		if = {
			limit = { scope:activity ?= { has_activity_option = { category = hd_debate_target option = hd_target_kaituo } } }
			add = hd_party_stance_weight_kaituo_value
		}
		if = {
			limit = { scope:activity ?= { has_activity_option = { category = hd_debate_target option = hd_target_jinqu } } }
			add = hd_party_stance_weight_jinqu_value
		}
		if = {
			limit = { scope:activity ?= { has_activity_option = { category = hd_debate_target option = hd_target_shoujiu } } }
			add = hd_party_stance_weight_shoujiu_value
		}
		if = {
			limit = { scope:activity ?= { has_activity_option = { category = hd_debate_target option = hd_target_zhongli } } }
			add = hd_party_stance_weight_zhongli_value
		}
		if = {
			limit = { var:hd_party_my_group ?= scope:host.var:hd_party_my_group }
			add = 10
		}
	}
}
