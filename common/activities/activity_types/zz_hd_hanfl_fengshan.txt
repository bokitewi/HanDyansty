# 汉代风味内容 A6：封禅大典（2026-10-25）。计划：/mnt/project-files/汉代风味内容/汉代风味内容_计划.md §4.7
# 单地点奉高（c_uuii_p_6934，泰山脚下），三阶段：斋戒 3201 → 登封 3202 → 禅梁父 3203（结算、玉牒、王朝 30 年冷却）。
# 开始时正统等级低于 3 级的，结算改为"德不配位"（活动变量 hd_hanfl_fengshan_unworthy）。
# 图标在 HanJIn_Clothes：gfx/interface/icons/activities/activity_hd_hanfl_fengshan(.dds/_header.dds)、gfx/interface/icons/activity_phases/hd_hanfl_fengshan_phase_*.dds

activity_hd_hanfl_fengshan = {
	is_shown = {
		hd_hanfl_is_tianzi_trigger = yes
	}

	can_start_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
		custom_tooltip = {
			text = hd_hanfl_fengshan_need_taishan_tt
			exists = title:c_uuii_p_6934.holder
			title:c_uuii_p_6934.holder.top_liege = root
		}
		custom_tooltip = {
			text = hd_hanfl_fengshan_recent_tt
			NOT = { dynasty ?= { has_variable = hd_hanfl_fengshan_recent } }
		}
	}

	can_always_plan = no

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			hd_hanfl_is_tianzi_trigger = yes
		}
		exists = title:c_uuii_p_6934.holder
		title:c_uuii_p_6934.holder.top_liege = scope:host
	}

	on_invalidated = {
		scope:activity = {
			activity_type = { save_scope_as = activity_type }
			activity_location = { save_scope_as = location }
		}
		if = {
			limit = { has_attending_activity_guests = no }
			scope:host = { trigger_event = activity_system.0100 }
		}
		if = {
			limit = { scope:host = { is_incapable = yes } }
			scope:host = { trigger_event = activity_system.0330 }
			every_attending_character = {
				limit = { this != scope:host }
				trigger_event = activity_system.0331
			}
		}
	}

	on_host_death = {
		every_attending_character = {
			limit = { this != scope:host }
			trigger_event = activity_system.0320
		}
	}

	###################
	# PARAMETERS
	###################

	is_single_location = yes

	province_filter = landed_title
	ai_province_filter = landed_title
	province_filter_target = "c_uuii_p_6934"

	is_location_valid = {
		this = title:c_uuii_p_6934.title_province
	}

	province_description = {
		desc = activity_hd_hanfl_fengshan_province_desc
	}

	cost = {
		gold = {
			if = {
				limit = { has_treasury = no }
				add = major_gold_value
			}
		}
		treasury = {
			if = {
				limit = { has_treasury = yes }
				add = major_treasury_value
			}
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				legitimacy_level >= 4
				is_at_war = no
			}
			add = 5
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 120
	}

	###################
	# GUEST HANDLING
	###################

	max_guests = 50

	guest_invite_rules = {
		defaults = {
			1 = activity_invite_rule_close_family
			1 = activity_invite_rule_vassals
			2 = activity_invite_rule_courtiers
			2 = activity_invite_rule_tributaries
		}
	}

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		in_diplomatic_range = scope:host
	}

	host_intents = {
		intents = { legitimacy_intent reduce_stress_intent }
		default = legitimacy_intent
	}

	guest_intents = {
		intents = { reduce_stress_intent befriend_attendee_intent }
		default = reduce_stress_intent
	}

	guest_join_chance = {
		base = 20
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
	}

	open_invite = no
	max_guest_arrival_delay_time = { months = 6 }

	###################
	# PHASES
	###################

	phases = {
		# 斋戒
		hd_hanfl_fengshan_phase_zhai = {
			order = 1
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3201
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}

		# 登封
		hd_hanfl_fengshan_phase_feng = {
			order = 2
			is_predefined = yes

			# 进入阶段后须推进一次才会转为进行中（同清议）
			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3202
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}

		# 禅梁父（结算）
		hd_hanfl_fengshan_phase_shan = {
			order = 3
			is_predefined = yes

			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3203
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}
	}

	on_start = {
		if = {
			limit = { scope:host = { legitimacy_level < 3 } }
			scope:activity = { set_variable = hd_hanfl_fengshan_unworthy }
		}
	}

	###################
	# WINDOW CHARACTERS
	###################

	window_characters = {
		host = {
			camera = camera_body

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { personality_honorable personality_bold }
				}
			}
		}

		guest = {
			camera = camera_body_left

			effect = {
				every_attending_character = {
					limit = {
						location = scope:activity.activity_location
						this != scope:host
					}
					add_to_list = characters
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { pondering throne_room_conversation_1 }
				}
			}
		}
	}

	###################
	# GRAPHICS
	###################

	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_coronation.dds"

	background = {
		texture = "gfx/interface/illustrations/event_scenes/tgp_holysite_asia.dds"
		environment = "environment_tgp_holysite_asia"
		ambience = "event:/DLC/EP4/SFX/Events/Event_Backgrounds/tgp_holysite_asia"
	}
}
