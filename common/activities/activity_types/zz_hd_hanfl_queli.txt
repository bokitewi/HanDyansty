# 汉代风味内容 A4：幸阙里（2026-10-25）。计划：/mnt/project-files/汉代风味内容/汉代风味内容_计划.md §4.5
# 单地点曲阜（c_qufu），三阶段都由东道主事件驱动（hd_hanfl.3001~3003），写法照 WJ_qingyi_activity.txt。
# 图标在 HanJIn_Clothes：gfx/interface/icons/activities/activity_hd_hanfl_queli(.dds/_header.dds)、gfx/interface/icons/activity_phases/hd_hanfl_queli_phase_*.dds

activity_hd_hanfl_queli = {
	is_shown = {
		government_has_flag = government_is_celestial
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_hd_hanfl_queli }
		is_available_adult = yes
		custom_tooltip = {
			text = hd_hanfl_queli_need_qufu_tt
			exists = title:c_qufu.holder
			title:c_qufu.holder.top_liege = root
		}
	}

	can_always_plan = no

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
		}
		exists = title:c_qufu.holder
		title:c_qufu.holder.top_liege = scope:host
	}

	on_invalidated = {
		scope:activity = {
			activity_type = { save_scope_as = activity_type }
			activity_location = { save_scope_as = location }
		}
		if = {
			limit = { has_attending_activity_guests = no }
			scope:host = { trigger_event = activity_system.0100 }
		}
		if = {
			limit = { scope:host = { is_incapable = yes } }
			scope:host = { trigger_event = activity_system.0330 }
			every_attending_character = {
				limit = { this != scope:host }
				trigger_event = activity_system.0331
			}
		}
	}

	on_host_death = {
		every_attending_character = {
			limit = { this != scope:host }
			trigger_event = activity_system.0320
		}
	}

	###################
	# PARAMETERS
	###################

	is_single_location = yes

	province_filter = landed_title
	ai_province_filter = landed_title
	province_filter_target = "c_qufu"

	is_location_valid = {
		this = title:c_qufu.title_province
	}

	province_description = {
		desc = activity_hd_hanfl_queli_province_desc
	}

	cooldown = { years = 10 }

	cost = {
		gold = {
			if = {
				limit = { has_treasury = no }
				add = medium_gold_value
			}
		}
		treasury = {
			if = {
				limit = { has_treasury = yes }
				add = medium_treasury_value
			}
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { legitimacy_level < 3 }
			add = 10
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	###################
	# GUEST HANDLING
	###################

	max_guests = 30

	guest_invite_rules = {
		defaults = {
			1 = activity_invite_rule_close_family
			1 = activity_invite_rule_vassals
			2 = activity_invite_rule_courtiers
		}
	}

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		in_diplomatic_range = scope:host
	}

	host_intents = {
		intents = { legitimacy_intent reduce_stress_intent }
		default = legitimacy_intent
	}

	guest_intents = {
		intents = { reduce_stress_intent befriend_attendee_intent }
		default = reduce_stress_intent
	}

	guest_join_chance = {
		base = 10
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
	}

	special_guests = {
		hd_hanfl_kongshi = {
			is_required = no
			can_pick = {
				this != scope:host
				is_adult = yes
				house ?= house:qufu_kong
			}
			ai_will_do = { value = 100 }
		}
	}

	open_invite = no
	max_guest_arrival_delay_time = { months = 4 }

	###################
	# PHASES
	###################

	phases = {
		# 谒祠
		hd_hanfl_queli_phase_ye = {
			order = 1
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3001
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}

		# 大会孔氏
		hd_hanfl_queli_phase_hui = {
			order = 2
			is_predefined = yes

			# 进入阶段后须推进一次才会转为进行中（同清议）
			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3002
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}

		# 请讲经（结算）
		hd_hanfl_queli_phase_jiang = {
			order = 3
			is_predefined = yes

			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3003
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}
	}

	###################
	# WINDOW CHARACTERS
	###################

	window_characters = {
		host = {
			camera = camera_body

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { personality_honorable pondering }
				}
			}
		}

		hd_hanfl_kongshi = {
			camera = camera_body_right

			effect = {
				if = {
					limit = {
						exists = scope:activity.special_guest:hd_hanfl_kongshi
						scope:activity.special_guest:hd_hanfl_kongshi.location = scope:activity.activity_location
					}
					special_guest:hd_hanfl_kongshi = { add_to_list = characters }
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { storyteller personality_honorable }
				}
			}
		}

		guest = {
			camera = camera_body_left

			effect = {
				every_attending_character = {
					limit = {
						location = scope:activity.activity_location
						this != scope:host
						NOT = { scope:activity.special_guest:hd_hanfl_kongshi ?= this }
					}
					add_to_list = characters
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { pondering throne_room_conversation_1 }
				}
			}
		}
	}

	###################
	# GRAPHICS
	###################

	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_pilgrimage.dds"

	background = {
		texture = "gfx/interface/illustrations/event_scenes/tgp_temple_asia.dds"
		environment = "environment_tgp_temple_asia"
		ambience = "event:/DLC/EP4/SFX/Events/Event_Backgrounds/tgp_temple_asia"
	}
}
