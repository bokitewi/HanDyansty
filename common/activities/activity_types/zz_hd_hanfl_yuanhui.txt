# 汉代风味内容 A6：元会（正旦大朝会）（2026-10-25）。计划：/mnt/project-files/汉代风味内容/汉代风味内容_计划.md §4.8
# 只给天子，只能在正月发起，在首都举行。三阶段：朝贺 3301 → 上计 3302 → 九宾散乐 3303（结算）。
# 图标在 HanJIn_Clothes：gfx/interface/icons/activities/activity_hd_hanfl_yuanhui(.dds/_header.dds)、gfx/interface/icons/activity_phases/hd_hanfl_yuanhui_phase_*.dds

activity_hd_hanfl_yuanhui = {
	is_shown = {
		hd_hanfl_is_tianzi_trigger = yes
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_hd_hanfl_yuanhui }
		is_available_adult = yes
		exists = capital_province
		custom_tooltip = {
			text = hd_hanfl_yuanhui_need_january_tt
			current_month = 1
		}
	}

	can_always_plan = no

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			hd_hanfl_is_tianzi_trigger = yes
		}
	}

	on_invalidated = {
		scope:activity = {
			activity_type = { save_scope_as = activity_type }
			activity_location = { save_scope_as = location }
		}
		if = {
			limit = { has_attending_activity_guests = no }
			scope:host = { trigger_event = activity_system.0100 }
		}
		if = {
			limit = { scope:host = { is_incapable = yes } }
			scope:host = { trigger_event = activity_system.0330 }
			every_attending_character = {
				limit = { this != scope:host }
				trigger_event = activity_system.0331
			}
		}
	}

	on_host_death = {
		every_attending_character = {
			limit = { this != scope:host }
			trigger_event = activity_system.0320
		}
	}

	###################
	# PARAMETERS
	###################

	is_single_location = yes

	province_filter = capital
	ai_province_filter = capital

	is_location_valid = {
		this = scope:host.capital_province
	}

	province_description = {
		desc = activity_hd_hanfl_yuanhui_province_desc
	}

	cooldown = { years = 1 }

	cost = {
		gold = {
			if = {
				limit = { has_treasury = no }
				add = medium_gold_value
			}
		}
		treasury = {
			if = {
				limit = { has_treasury = yes }
				add = medium_treasury_value
			}
		}
	}

	# AI 低频：每月查一次（只有正月能发起，按年查可能永远碰不上正月），每个正月一成机会，约十年一次
	ai_will_do = {
		value = 10
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 1
	}

	###################
	# GUEST HANDLING
	###################

	max_guests = 50

	guest_invite_rules = {
		defaults = {
			1 = activity_invite_rule_vassals
			1 = activity_invite_rule_tributaries
			2 = activity_invite_rule_courtiers
		}
	}

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		in_diplomatic_range = scope:host
	}

	host_intents = {
		intents = { legitimacy_intent impose_obedience_intent reduce_stress_intent }
		default = legitimacy_intent
	}

	guest_intents = {
		intents = { reduce_stress_intent befriend_attendee_intent }
		default = reduce_stress_intent
	}

	guest_join_chance = {
		base = 30
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
	}

	open_invite = no
	max_guest_arrival_delay_time = { months = 3 }

	###################
	# PHASES
	###################

	phases = {
		# 朝贺
		hd_hanfl_yuanhui_phase_he = {
			order = 1
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3301
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}

		# 上计
		hd_hanfl_yuanhui_phase_ji = {
			order = 2
			is_predefined = yes

			# 进入阶段后须推进一次才会转为进行中（同清议）
			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3302
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}

		# 九宾散乐（结算）
		hd_hanfl_yuanhui_phase_yue = {
			order = 3
			is_predefined = yes

			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = hd_hanfl.3303
				}
			}

			on_phase_active_activity = {
				progress_activity_phase_after = { weeks = 1 }
			}
		}
	}

	###################
	# WINDOW CHARACTERS
	###################

	window_characters = {
		host = {
			camera = camera_body

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { toast_goblet personality_bold }
				}
			}
		}

		guest = {
			camera = camera_body_left

			effect = {
				every_attending_character = {
					limit = {
						location = scope:activity.activity_location
						this != scope:host
					}
					add_to_list = characters
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { drink_goblet throne_room_conversation_1 }
				}
			}
		}
	}

	###################
	# GRAPHICS
	###################

	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_feast.dds"

	background = {
		texture = "gfx/interface/illustrations/event_scenes/tgp_asia_throne_room.dds"
		environment = "environment_tgp_frontend_celestial_main"
		ambience = "event:/DLC/EP4/SFX/Events/Event_Backgrounds/tgp_study_asia"
	}
}
