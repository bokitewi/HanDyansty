hds_nr_sun_jian_commander_mortality = {
    type = commander
    is_valid = {
        exists = root
        exists = character:sun_jian
        this = character:sun_jian
        is_alive = yes
        has_character_modifier = hds_nr_sun_jian_mortal_risk
        exists = scope:combat_side
        exists = scope:combat_side.enemy_side

        OR = {
            has_trait_rank = { trait = wounded rank >= 1 }
            AND = { martial <= low_skill_rating prowess <= low_skill_rating }
        }
        NOR = {
            AND = { is_ai = no has_game_rule = very_easy_difficulty }
            ai_should_get_extreme_conqueror_bonuses = yes
        }
    }
    chance = {
        base = 1

        modifier = {
            factor = {
                value = 30
                subtract = prowess
                
                divide = 30
                min = 0.1
            }
        }
        modifier = {
            scope:combat_side.side_strength > scope:combat_side.enemy_side.side_strength
            factor = {
                value = scope:combat_side.enemy_side.side_strength
                divide = scope:combat_side.side_strength
                multiply = 1.4
                max = 1
            }
        }
        modifier = {
            scope:combat_side.side_strength > scope:combat_side.enemy_side.vastly_outnumbered_combat_side_threshold
            factor = 0
        }
        modifier = { faith = { has_doctrine_parameter = death_is_glory } factor = 1.1 }
        modifier = { has_trait = brave factor = 2 }
        modifier = { has_trait = craven factor = 0.5 }
        modifier = { has_trait_rank = { trait = wounded rank = 3 } factor = 4 }
    }
    effect = { hds_nr_battle_death_effect = { ROLE = commander } }
}
