# Native court position; compatible with council jobs and other part-time court posts.
WJ_dasima_court_position = {
 sort_order = 5001
 max_available_positions = 1
 minimum_rank = empire
 skill = martial
 court_position_asset = {
  animation = marshal
  background = "gfx/interface/illustrations/event_scenes/tgp_study_asia.dds"
 }
 aptitude_level_breakpoints = { 20 40 60 80 }
 aptitude = { value = martial multiply = 2 add = general_aptitude_bonus }
 is_shown = { WJ_inner_emperor_trigger = yes }
 valid_position = { WJ_inner_emperor_trigger = yes }
 is_shown_character = { scope:employee = { court_position_employee_shown_trigger = yes } }
 valid_character = {
  scope:liege = { save_temporary_scope_as = wj_inner_emperor }
  scope:employee = { WJ_dasima_candidate_trigger = yes }
 }
 revoke_cost = {}
 salary = { }
 received_salary = { }
 modifier = {
  monthly_influence_mult = 0.1
  monthly_merit = 1
 }
 on_court_position_received = {
  scope:employee = { if = { limit = { NOT = { has_variable = wj_dasima_emperor } } add_opinion = { target = scope:liege modifier = WJ_dasima_granted_opinion } } }
  scope:liege = { set_variable = { name = wj_dasima_holder value = scope:employee } }
  scope:employee = {
   set_variable = { name = wj_dasima_emperor value = scope:liege }
   set_variable = { name = wj_dasima_realm value = scope:liege.primary_title }
  }
 }
 on_court_position_revoked = {
  scope:employee = { add_opinion = { target = scope:liege modifier = WJ_dasima_removed_opinion } WJ_dasima_clear_effect = yes }
 }
 on_court_position_invalidated = { scope:employee = { WJ_dasima_clear_effect = yes } }
 on_court_position_vacated = { scope:employee = { WJ_dasima_clear_effect = yes } }
 ai_position_score = { value = 0 }
}
