hd_party_convert_identity_decision = {

	title = hd_party_convert_identity_decision
	desc = hd_party_convert_identity_decision_desc
	selection_tooltip = hd_party_convert_identity_decision_tooltip
	picture = {
		reference = "gfx/interface/illustrations/decisions/tgp_scholars.dds"
	}
	sort_order = 79

	is_shown = {
		exists = var:movement_member
		is_adult = yes
		any_character_situation = { situation_type = dynastic_cycle }
		top_participant_group:dynastic_cycle ?= {
			NOR = {
				participant_group_type = hegemon_ruler
				participant_group_type = other_rulers
				participant_group_type = hd_siyi_movement
			}
		}
	}
	is_valid_showing_failures_only = {
		is_imprisoned = no
	}
	is_valid = {
		custom_tooltip = {
			text = hd_party_convert_cd_tt
			NOT = { has_variable = hd_party_convert_cd }
		}
		prestige >= hd_party_convert_prestige_cost
	}

	widget = {
		gui = "decision_view_widget_generic_multichoice_with_effects"
		controller = decision_option_list_controller
		decision_to_second_step_button = "HD_PARTY_CONVERT_CONFIRM"
		show_from_start = no

		item = {
			value = hd_conv_huanguan
			current_description = hd_party_convert_huanguan_desc
			localization = hd_party_convert_huanguan_name
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_pro.dds"
			flat = yes
			is_valid = {
				NOT = { var:movement_member ?= flag:undecided }
				hd_party_is_huanguan_trigger = yes
			}
			ai_chance = { value = 0 }
		}
		item = {
			value = hd_conv_waiqi
			current_description = hd_party_convert_waiqi_desc
			localization = hd_party_convert_waiqi_name
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_unaligned.dds"
			flat = yes
			is_valid = {
				NOT = { var:movement_member ?= flag:pro_hegemon }
				hd_party_is_waiqi_trigger = yes
			}
			ai_chance = {
				value = 0
				if = {
					limit = { var:hd_party_my_court.var:hd_party_favored ?= flag:waiqi }
					add = 10
				}
			}
		}
		item = {
			value = hd_conv_wuxun
			current_description = hd_party_convert_wuxun_desc
			localization = hd_party_convert_wuxun_name
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_expand.dds"
			flat = yes
			is_valid = {
				NOT = { var:movement_member ?= flag:expansion }
				hd_party_is_wuxun_trigger = yes
			}
			ai_chance = {
				value = 0
				if = {
					limit = { var:hd_party_my_court.var:hd_party_favored ?= flag:wuxun }
					add = 10
				}
			}
		}
		item = {
			value = hd_conv_shizu
			current_description = hd_party_convert_shizu_desc
			localization = hd_party_convert_shizu_name
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_anti.dds"
			flat = yes
			is_valid = {
				NOT = { var:movement_member ?= flag:conservative }
				hd_party_is_shizu_trigger = yes
			}
			ai_chance = {
				value = 0
				if = {
					limit = { var:hd_party_my_court.var:hd_party_favored ?= flag:shizu }
					add = 10
				}
			}
		}
		item = {
			value = hd_conv_hanmen
			current_description = hd_party_convert_hanmen_desc
			localization = hd_party_convert_hanmen_name
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_advance.dds"
			flat = yes
			is_valid = {
				NOT = { var:movement_member ?= flag:advancement }
				hd_party_is_shizu_trigger = no
			}
			ai_chance = {
				value = 0
				if = {
					limit = { var:hd_party_my_court.var:hd_party_favored ?= flag:hanmen }
					add = 10
				}
			}
		}
	}

	effect = {
		add_prestige = {
			value = hd_party_convert_prestige_cost
			multiply = -1
		}
		if = {
			limit = { legitimacy > 0 }
			add_legitimacy = {
				value = hd_party_convert_legitimacy_cost
				multiply = -1
			}
		}
		if = {
			limit = { government_has_flag = government_has_influence }
			change_influence = major_influence_loss
		}
		switch = {
			trigger = yes
			scope:hd_conv_huanguan = { hd_party_convert_identity_effect = { FLAG = undecided } }
			scope:hd_conv_waiqi = { hd_party_convert_identity_effect = { FLAG = pro_hegemon } }
			scope:hd_conv_wuxun = { hd_party_convert_identity_effect = { FLAG = expansion } }
			scope:hd_conv_shizu = { hd_party_convert_identity_effect = { FLAG = conservative } }
			scope:hd_conv_hanmen = { hd_party_convert_identity_effect = { FLAG = advancement } }
		}
	}

	ai_check_interval = 120
	ai_potential = {
		is_ai = yes
		exists = var:hd_party_my_court
		var:hd_party_my_court = { exists = var:hd_party_favored }
		prestige >= 1500
	}
	ai_will_do = {
		base = 0
		# 只在目标身份是被青睐的朋党、且自己所属朋党不被青睐时考虑
		modifier = {
			add = 25
			hd_party_member_favored_trigger = no
		}
	}
}

# 投身乱世（草案 9.2）：斗争期外戚/武勋/寒门/世族的无地成年人

hd_party_join_chaos_decision = {
	decision_group_type = dynastic_cycle
	title = hd_party_join_chaos_decision
	desc = hd_party_join_chaos_decision_desc
	selection_tooltip = hd_party_join_chaos_decision_tooltip
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_war.dds"
	}
	sort_order = 77

	is_shown = {
		hd_party_can_join_chaos_trigger = yes
	}
	is_valid = {
		OR = {
			martial >= 10
			prowess >= 12
		}
		OR = {
			has_trait = ambitious
			has_trait = brave
			has_trait = cynical
		}
		gold >= 100
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	cost = {
		gold = 100
	}

	effect = {
		save_scope_as = hd_party_rebel
		location ?= { save_scope_as = hd_party_rebel_location }
		# 带 2 名同身份的无地亲属或门生作为随从
		ordered_close_family_member = {
			limit = {
				is_landed = no
				is_adult = yes
				is_available = yes
				var:movement_member ?= scope:hd_party_rebel.var:movement_member
			}
			order_by = prowess
			max = 2
			check_range_bounds = no
			add_to_list = hd_party_rebel_followers
		}
		if = {
			limit = {
				NOT = { any_in_list = { list = hd_party_rebel_followers count >= 2 } }
			}
			ordered_relation = {
				type = disciple
				limit = {
					is_landed = no
					is_adult = yes
					is_available = yes
				}
				order_by = prowess
				max = 2
				check_range_bounds = no
				add_to_list = hd_party_rebel_followers
			}
		}
		create_landless_adventurer_title_effect = {
			REASON = flag:runaway_allowed
			FLAVOR_CHAR = scope:hd_party_rebel
		}
		every_in_list = {
			list = hd_party_rebel_followers
			scope:hd_party_rebel = { add_courtier = prev }
		}
		if = {
			limit = { exists = scope:hd_party_rebel_location }
			spawn_army = {
				name = hd_party_join_chaos_army
				levies = hd_party_join_chaos_troops_value
				location = scope:hd_party_rebel_location
				inheritable = yes
			}
		}
		if = {
			limit = {
				is_ai = yes
				NOT = { owns_story_of_type = story_adventurer_ai }
			}
			create_story = story_adventurer_ai
		}
	}

	ai_check_interval = 60
	ai_potential = {
		is_ai = yes
		hd_party_can_join_chaos_trigger = yes
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 10
			has_trait = ambitious
		}
		modifier = {
			add = 5
			has_trait = brave
		}
	}
}
