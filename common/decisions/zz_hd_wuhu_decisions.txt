# 五胡内附：史实大单于起兵（玩家重大决定；AI 由年度脉冲处理）。见 docs/五胡内附_僚人入蜀_设计草案.md 4.6。

hd_wuhu_chanyu_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major
	desc = hd_wuhu_chanyu_decision_desc
	selection_tooltip = hd_wuhu_chanyu_decision_tooltip
	confirm_text = hd_wuhu_chanyu_decision_confirm

	ai_check_interval = 0

	is_shown = {
		hd_wuhu_enabled_trigger = yes
		OR = {
			AND = {
				character:liu_yuan_3 ?= this
				NOT = { exists = global_var:hd_wuhu_chanyu_liuyuan_done }
			}
			AND = {
				character:shi_lei ?= this
				NOT = { exists = global_var:hd_wuhu_chanyu_shile_done }
			}
			AND = {
				OR = {
					character:li_te ?= this
					AND = {
						character:bacong_li_xiong ?= this
						NOT = { character:li_te ?= { is_alive = yes } }
					}
				}
				NOT = { exists = global_var:hd_wuhu_chanyu_lite_done }
			}
			AND = {
				character:fu_hong ?= this
				NOT = { exists = global_var:hd_wuhu_chanyu_fuhong_done }
			}
			AND = {
				character:shaodang_yao_yi_zhong ?= this
				NOT = { exists = global_var:hd_wuhu_chanyu_yaoyizhong_done }
			}
		}
	}

	is_valid = {
		custom_description = {
			text = hd_wuhu_chanyu_decision_year_ok
			trigger_if = {
				limit = { character:liu_yuan_3 ?= this }
				hd_wuhu_year_from_trigger = { START = 299 }
			}
			trigger_else_if = {
				limit = { character:shi_lei ?= this }
				hd_wuhu_year_from_trigger = { START = 305 }
			}
			trigger_else_if = {
				limit = {
					OR = {
						character:li_te ?= this
						character:bacong_li_xiong ?= this
					}
				}
				hd_wuhu_year_from_trigger = { START = 296 }
			}
			trigger_else = {
				hd_wuhu_year_from_trigger = { START = 344 }
			}
		}
		custom_description = {
			text = hd_wuhu_chanyu_decision_phase_ok
			hd_wuhu_dongluan_phase_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		prestige = 1000
	}

	effect = {
		custom_tooltip = hd_wuhu_chanyu_decision_effect_tt
		if = {
			limit = { character:liu_yuan_3 ?= this }
			hd_wuhu_chanyu_launch_liuyuan_effect = yes
		}
		else_if = {
			limit = { character:shi_lei ?= this }
			hd_wuhu_chanyu_launch_shile_effect = yes
		}
		else_if = {
			limit = {
				OR = {
					character:li_te ?= this
					character:bacong_li_xiong ?= this
				}
			}
			hd_wuhu_chanyu_launch_lite_effect = yes
		}
		else_if = {
			limit = { character:fu_hong ?= this }
			hd_wuhu_chanyu_launch_fuhong_effect = yes
		}
		else_if = {
			limit = { character:shaodang_yao_yi_zhong ?= this }
			hd_wuhu_chanyu_launch_yaoyizhong_effect = yes
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
