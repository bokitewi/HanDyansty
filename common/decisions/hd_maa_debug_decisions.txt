# 调试：直接建立幽州突骑军制并把兵源条件补齐（控制力/文化接受度）
hd_debug_prepare_youzhou_tuqi_decision = {
	title = hd_debug_prepare_youzhou_tuqi_decision
	desc = hd_debug_prepare_youzhou_tuqi_decision_desc
	selection_tooltip = hd_debug_prepare_youzhou_tuqi_decision_tooltip
	decision_group_type = major
	sort_order = 2990

	is_shown = {
		has_character_flag = hd_maa_debug_testing
		NOT = { has_character_flag = hd_established_youzhou_tuqi }
	}

	effect = {
		add_character_flag = hd_established_youzhou_tuqi
		add_character_flag = hd_source_youzhou_tuqi
		every_sub_realm_county = {
			limit = {
				title_province = { geographical_region = hd_region_youzhou_tuqi }
				county_control < 50
			}
			change_county_control = { value = 60 }
		}
		culture = {
			change_cultural_acceptance = {
				target = cul:youzhou
				value = 60
				desc = hd_maa_debug_acceptance_desc
			}
		}
		custom_tooltip = hd_debug_prepare_youzhou_tuqi_decision_effect
	}

	ai_check_interval = 0
}
