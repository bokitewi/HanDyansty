# 汉化改制（北方版）：衣冠南渡期间的北方胡人王国级以上独立统治者（非天子）
############################################################
hd_yiguan_sinicization_decision = {
	decision_group_type = major
	picture = {
		reference = "gfx/interface/illustrations/decisions/tgp_scholars.dds"
	}
	sort_order = 59

	is_shown = {
		hd_yiguan_active_trigger = yes
		NOT = { has_title = title:h_china }
		hd_yiguan_north_power_trigger = yes
		NOT = { has_variable = hd_dc_con_sinicization_done }
		NOT = { hd_yiguan_is_sinicized_trigger = yes }
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}
	is_valid = {
		custom_tooltip = {
			text = hd_dc_con_sinicization_active_tt
			NOT = { has_variable = hd_dc_con_sinicization_stage }
		}
		custom_tooltip = {
			text = hd_dc_con_baosu_blocks_tt
			NOT = { has_character_flag = hd_dc_con_baosu }
		}
		custom_tooltip = {
			text = hd_dc_con_sinicization_abandoned_tt
			NOT = { has_character_flag = hd_dc_con_sinicization_abandoned }
		}
	}

	effect = {
		custom_tooltip = hd_dc_con_sinicization_decision_effect_tt
		show_as_tooltip = {
			every_vassal = {
				limit = { culture = root.culture }
				custom = hd_dc_con_native_vassals_tt
				add_opinion = {
					target = root
					modifier = hd_dc_con_sinicization_native_opinion
				}
			}
			every_vassal = {
				limit = { hd_dc_is_chinese_heritage_trigger = yes }
				custom = hd_dc_con_han_vassals_tt
				add_opinion = {
					target = root
					modifier = hd_dc_con_sinicization_han_opinion
				}
			}
		}
		hidden_effect = {
			if = {
				limit = { NOT = { has_variable = hd_dc_con_native_culture } }
				culture = { save_scope_as = hd_dc_con_start_culture }
				set_variable = {
					name = hd_dc_con_native_culture
					value = scope:hd_dc_con_start_culture
				}
			}
			set_variable = hd_yiguan_north_reform
			set_variable = {
				name = hd_dc_con_sinicization_stage
				value = 0
			}
			trigger_event = {
				id = hd_yiguan.3001
				days = { 730 1095 }
			}
		}
	}

	ai_check_interval = 60
	ai_will_do = {
		base = 10
		modifier = {
			add = 20
			learning >= 15
		}
		modifier = {
			add = 10
			has_trait = ambitious
		}
		modifier = {
			add = hd_yiguan_zhuoyong_ai_value
			has_variable = hd_yiguan_zhuoyong_count
		}
		modifier = {
			add = -20
			OR = {
				has_trait = stubborn
				has_trait = paranoid
			}
		}
	}
}

		}
	}

# 还都旧京：光复后 20 年内，持有洛阳或长安的霸权持有者
############################################################
hd_yiguan_return_capital_decision = {
	decision_group_type = major
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	sort_order = 57

	is_shown = {
		exists = global_var:hd_yiguan_restore_window
		NOT = { exists = global_var:hd_yiguan_capital_returned }
		has_title = title:h_china
	}
	is_valid = {
		custom_tooltip = {
			text = hd_yiguan_return_capital_hold_tt
			OR = {
				has_title = title:c_luoyang
				has_title = title:c_uuii_p_867
			}
		}
		custom_tooltip = {
			text = hd_yiguan_return_capital_already_tt
			NOR = {
				capital_county = title:c_luoyang
				capital_county = title:c_uuii_p_867
			}
		}
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}
	cost = {
		gold = major_gold_value
	}

	effect = {
		set_global_variable = hd_yiguan_capital_returned
		if = {
			limit = { has_title = title:c_luoyang }
			set_realm_capital = title:c_luoyang
		}
		else = {
			set_realm_capital = title:c_uuii_p_867
		}
		add_prestige = major_prestige_gain
		add_legitimacy = major_legitimacy_gain
	}

	ai_check_interval = 60
	ai_will_do = {
		base = 60
	}
}
