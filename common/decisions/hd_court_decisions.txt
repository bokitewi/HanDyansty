hd_court_proclaim_emperor_decision = {
	picture = { reference = "gfx/interface/illustrations/event_story/tgp_dynastic_cycle_stability.dds" }
	desc = hd_court_proclaim_emperor_decision_desc
	selection_tooltip = hd_court_proclaim_emperor_decision_tooltip
	confirm_text = hd_court_proclaim_emperor_decision_confirm
	decision_group_type = dynastic_cycle

	is_shown = {
		hd_court_privileges_active_trigger = yes
		hd_court_warlord_trigger = yes
		highest_held_title_tier = tier_kingdom
		NOT = { has_character_modifier = hd_court_pseudo_emperor_modifier }
	}

	is_valid_showing_failures_only = {
		is_available_at_peace_adult = yes
		custom_description = {
			text = hd_court_proclaim_size_tt
			sub_realm_size >= 30
		}
		custom_description = {
			text = hd_court_proclaim_army_tt
			trigger_if = {
				limit = { hd_court_has_regent_trigger = yes }
				current_military_strength >= title:h_china.var:hd_court_regent.current_military_strength
			}
			trigger_else = {
				current_military_strength >= title:h_china.holder.current_military_strength
			}
		}
		custom_description = {
			text = hd_court_proclaim_popular_tt
			any_realm_county = {
				percent >= 0.6
				county_opinion >= 10
			}
		}
	}

	cost = { prestige = 1000 }

	effect = {
		custom_tooltip = hd_court_proclaim_emperor_decision_effect_tt
		hd_court_proclaim_emperor_effect = yes
	}

	ai_check_interval = 60
	ai_potential = {
		hd_court_warlord_trigger = yes
		highest_held_title_tier = tier_kingdom
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 15
			hd_court_stance_moni_trigger = yes
		}
		modifier = {
			add = 15
			has_trait = ambitious
		}
		modifier = {
			factor = 0
			hd_court_stance_zunhan_trigger = yes
		}
	}
}
