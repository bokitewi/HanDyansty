		NOT = { is_at_war_with = title:h_china.holder }

			text = hd_j16_ruchao_reach_tt
			OR = {
				title:h_china.var:hd_court_capital_county.holder ?= {
					OR = {
						this = root
						target_is_liege_or_above = root
					}
				}
				title:h_china.holder.location.county.holder ?= {
					OR = {
						this = root
						target_is_liege_or_above = root
					}

				AND = {
					hd_court_has_regent_trigger = no

					title:h_china.var:hd_court_capital_county.duchy ?= {
						any_de_jure_county = {
							OR = {
								holder ?= {
									OR = {
										this = root
										target_is_liege_or_above = root
									}
								}
								any_neighboring_county = {
									holder ?= {
										OR = {
											this = root
											target_is_liege_or_above = root
										}
									}
								}
							}
						}
					}
				}
			}
		}
		custom_description = {
			text = hd_j16_ruchao_army_tt
			trigger_if = {
				limit = { hd_court_has_regent_trigger = yes }
				max_military_strength >= title:h_china.var:hd_court_regent.max_military_strength
			}
		}
		custom_description = {
			text = hd_j16_ruchao_cd_tt
			NOT = { has_character_flag = hd_j16_chuzhen_cd }
		}
	}

	cost = { prestige = 300 }

		hidden_effect = {
			save_scope_as = hd_j16_rc_new
			if = {
				limit = { exists = title:h_china.var:hd_court_regent }
				hd_j16_swap_regent_effect = yes
			}

				hd_j16_join_court_effect = yes

				}
			}

				limit = {
					house ?= house:henei_sima
					NOT = { this = scope:hd_j16_rc_new }
					NOT = { has_title = title:h_china }
				}
				add_opinion = {
					target = scope:hd_j16_rc_new
					modifier = hd_j16_zhuanshan_opinion
				}
			}
			title:h_china.holder ?= {

					add_opinion = {
						target = scope:hd_j16_rc_new
						modifier = hd_j16_zhuanshan_opinion

			add = 60
			this = character:hd_304_si_ma_yue
			has_global_variable = hd_j16_huanluo_done

			add = 30

	picture = { reference = "gfx/interface/illustrations/decisions/tgp_retire.dds" }

	selection_tooltip = hd_j16_chuanxi_decision_tooltip

		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		NOT = { hd_court_is_regent_trigger = yes }
		# 审查修正：只看「迎驾之战已开」（B2 跳过或玩家越观望后仍可用；started 不清，全局只能开一次）
		NOT = { has_global_variable = hd_j16_yingjia_started }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		hd_court_has_regent_trigger = yes
		NOT = { is_at_war_with = title:h_china.var:hd_court_regent }
		custom_description = {
			text = hd_j16_chuanxi_cd_tt
			NOT = { has_character_flag = hd_j16_chuanxi_cd }
		}
	}

	cost = { prestige = 300 }

	effect = {
		custom_tooltip = hd_j16_chuanxi_decision_tooltip
		hidden_effect = {
			add_character_flag = { flag = hd_j16_chuanxi_cd years = 5 }
			# 目标与 B2 同一选法（3-B1 返工：京城县所在郡（干净且郡内有县属天子一方）优先，否则权臣首府所在郡，无权臣取颙首府所在郡；不看接壤，得胜不割地）：起兵者是本人
			hd_j16_304_res_effect = { ID = si_ma_yong AS = hd_j16_yj_yong }
			if = {
				limit = { exists = scope:hd_j16_yj_yue }
				clear_saved_scope = hd_j16_yj_yue
			}
			save_scope_as = hd_j16_yj_yue
			hd_j16_yingjia_pick_target_effect = yes
			if = {
				limit = { exists = scope:hd_j16_yj_tgt }
				hd_j16_yingjia_war_effect = { SIZE = 8000 WANGJUN = no }
			}
			if = {
				limit = { exists = scope:hd_nbc_war_yingjia }
				# 邀诸王：与权臣不睦（好感 < 0）的 AI 独立宗王加入攻方；朝贡天子者先结束朝贡（记 hd_j16_was_tributary），照越、王浚的写法
				every_ruler = {
					limit = {
						is_ai = yes
						is_independent_ruler = yes
						is_at_war = no
						house ?= house:henei_sima
						NOT = { this = root }
						NOT = { has_title = title:h_china }
						opinion = {
							target = title:h_china.var:hd_court_regent
							value < 0
						}
					}
					if = {
						limit = { is_tributary = yes }
						if = {
							limit = {
								exists = title:h_china.holder
								is_tributary_of_suzerain_or_above = title:h_china.holder
							}
							set_variable = { name = hd_j16_was_tributary value = yes }
						}
						end_tributary = yes
					}
					save_scope_as = hd_j16_cx_ally
					scope:hd_nbc_war_yingjia = { add_attacker = scope:hd_j16_cx_ally }
				}
			}
			else = {
				# 无目标或宣战失败：退还威望、撤冷却（审查修正）
				add_prestige = 300
				remove_character_flag = hd_j16_chuanxi_cd
				debug_log = "HD_J16 e1_b10 WAIT decision_chuanxi_refunded"
			}
			debug_log = "HD_J16 e1_b10 FIRE decision_chuanxi"
		}
	}

	ai_check_interval = 0
	ai_will_do = { base = 0 }
