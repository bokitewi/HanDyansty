# 郡国制·军阀契约：「拥兵自重」决议（2026-10-24 用户指定）
# - 郡国制有地封臣，当前须为武官（契约带 celestial_military_appointment），不限时期
# - 被「收回权力」后 5 年内不能再自请（hd_warlord_revoked_cooldown）
# - 后果：契约改为军阀（不缴赋税、不出征召兵、欢呼继承），主君好感 -40（十年衰减）
# - AI 少用：每 120 个月评估一次，看与主君的实力对比、好感与性格；忠顺于主君或兵力不足主君四分之一则不发起

hd_warlord_declare_decision = {
	decision_group_type = admin
	picture = {
		reference = "gfx/interface/illustrations/decisions/tgp_dominance.dds"
	}
	sort_order = 60

	is_shown = {
		hd_warlord_declare_shown_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		NOT = { is_at_war_with = liege }
	}

	is_valid = {
		custom_tooltip = {
			text = hd_warlord_declare_military_tt
			vassal_contract_has_flag = celestial_military_appointment
		}
		custom_tooltip = {
			text = hd_warlord_declare_cooldown_tt
			NOT = { has_character_flag = hd_warlord_revoked_cooldown }
		}
	}

	effect = {
		hd_warlord_establish_effect = yes
		liege = {
			add_opinion = {
				target = root
				modifier = hd_warlord_declared_opinion
			}
			hidden_effect = {
				send_interface_message = {
					type = event_generic_bad
					title = hd_warlord_declared_msg
					left_icon = root
				}
			}
		}
	}

	ai_check_interval = 120
	ai_will_do = {
		base = 0
		# 实力对比（本人 / 主君）
		modifier = {
			add = 25
			max_military_strength >= liege.max_military_strength
		}
		modifier = {
			add = 10
			max_military_strength < liege.max_military_strength
			max_military_strength >= {
				value = liege.max_military_strength
				multiply = 0.5
			}
		}
		# 对主君越不满越想拥兵自重
		modifier = {
			add = 10
			opinion = {
				target = liege
				value <= 0
			}
		}
		modifier = {
			add = 15
			opinion = {
				target = liege
				value <= -25
			}
		}
		modifier = {
			add = 10
			has_trait = ambitious
		}
		modifier = {
			add = 5
			has_trait = arrogant
		}
		modifier = {
			add = 5
			has_trait = greedy
		}
		modifier = {
			add = 5
			has_trait = wrathful
		}
		modifier = {
			add = -10
			has_trait = content
		}
		modifier = {
			add = -5
			has_trait = humble
		}
		modifier = {
			add = -5
			has_trait = craven
		}
		# （置于末尾，避免被 add 抵消）
		modifier = {
			factor = 0
			has_trait = loyal
		}
		modifier = {
			factor = 0
			max_military_strength < {
				value = liege.max_military_strength
				multiply = 0.25
			}
		}
		modifier = {
			factor = 0
			is_obedient_to = liege
		}
	}
}
