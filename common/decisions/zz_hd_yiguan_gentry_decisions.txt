hd_yiguan_qizu_nanqian_decision = {
	decision_group_type = major
	picture = {

	sort_order = 58

	}

		prestige = hd_yiguan_gt_nanqian_cost_value

	# 选目的地（先例 原版 pam_antipope_decisions.txt:167-220）
	widget = {
		gui = "decision_view_widget_select_character"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT"

		setup_items = {
			situation:hd_yiguan_nandu ?= {
				every_situation_participant = {
					limit = { hd_yiguan_gt_nanqian_dest_trigger = { MIGRANT = scope:actor } }
					add_to_list = item_list
				}
			}
		}
		default_item = {
			situation:hd_yiguan_nandu ?= {
				ordered_situation_participant = {
					limit = { hd_yiguan_gt_nanqian_dest_trigger = { MIGRANT = scope:actor } }
					order_by = hd_yiguan_gt_nanqian_dest_score_value
					save_scope_as = selected_item
				}
			}
		}
		is_item_valid = {
			hd_yiguan_gt_nanqian_dest_trigger = { MIGRANT = scope:actor }
		}
		ai_select_item = {
			situation:hd_yiguan_nandu ?= {
				ordered_situation_participant = {
					limit = { hd_yiguan_gt_nanqian_dest_trigger = { MIGRANT = scope:actor } }
					order_by = hd_yiguan_gt_nanqian_dest_score_value
					save_scope_as = selected_item
				}
			}
		}
	}

	effect = {
		custom_tooltip = hd_yiguan_qizu_nanqian_decision_effect_tt
		if = {
			limit = { exists = scope:selected_item }
			scope:selected_item = { save_scope_as = hd_yiguan_gt_dest }
			hidden_effect = { hd_yiguan_gt_nanqian_request_effect = yes }
		}
	}

	ai_check_interval = 60
	ai_potential = {
		is_ai = yes
		is_adult = yes
	}
	ai_will_do = {
		base = 0
		modifier = {	# 无地世族牵挂少
			add = 15
			is_landed = no
		}
		modifier = {
			add = 10
			hd_yiguan_phase_taowang_trigger = yes
		}
		modifier = {	# 为胡主所忌
			add = 15
			liege ?= {
				opinion = {
					target = root
					value <= -20
				}
			}
		}
		modifier = {	# 自己也恨其主
			add = 10
			exists = liege
			opinion = {
				target = liege
				value <= -20
			}
		}
		modifier = {
			add = 5
			OR = {
				has_trait = paranoid
				has_trait = zealous
			}
		}
		modifier = {
			add = -15
			OR = {
				has_trait = content
				has_trait = greedy
			}
		}
		modifier = {	# 有地者不轻弃祖业
			add = -10
			is_landed = yes
		}
		modifier = {	# 郡守以上更不肯走
			add = -15
			any_held_title = {
				tier >= tier_duchy
				is_landless_type_title = no
			}
		}
		modifier = {
			factor = 0
			is_at_war = yes
		}
	}
}
