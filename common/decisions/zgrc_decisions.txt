# 浸染汉风（臣子）：非天子的文化领袖，事汉主 / 邻近天朝而慕化
zgrc_decision_jin_ran_han_feng_subject = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/zgrc_hanfeng.dds"
	}
	decision_group_type = zgrc

	ai_check_interval_by_tier = {
		barony = 0
		county = 60
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 0
	}

	selection_tooltip = zgrc_decision_jin_ran_han_feng_subject_tooltip
	desc = zgrc_decision_jin_ran_han_feng_subject_desc

	is_shown = {
		# 大前提：本人是文化领袖
		exists = culture.culture_head
		this = culture.culture_head
		# 非天子（天子走亲裁版）
		NOT = { has_title = title:h_china }
		exists = title:h_china.holder
		zgrc_culture_unassimilated_trigger = yes
		# 本 mod：能发起推行汉化者走推行汉化（发起即浸染），此处不再显示
		NOT = { zgrc_hd_reform_path_trigger = yes }
		# 两条路：① 独立且邻接华夏霸权；② 汉人统治者之臣
		OR = {
			# 路径①：独立统治者，邻接华夏霸权（汉人天子）
			AND = {
				is_independent_ruler = yes
				any_neighboring_and_across_water_top_liege_realm_owner = {
					has_title = title:h_china
					culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
				}
			}
			# 路径②：汉人统治者之臣
			AND = {
				NOT = { is_independent_ruler = yes }
				exists = top_liege
				top_liege = {
					culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_physically_able_adult = yes
		is_at_war = no
	}

	is_valid = {
		zgrc_culture_unassimilated_trigger = yes
		custom_tooltip = {
			text = zgrc_yong_xia_bian_yi_need_culture_head
			exists = culture.culture_head
			this = culture.culture_head
		}
		# 天子须康健且未开战
		title:h_china.holder = {
			is_adult = yes
			is_imprisoned = no
			is_at_war = no
		}
	}

	cost = {
		prestige = {
			value = zgrc_jin_ran_han_feng_cost
		}
	}

	cooldown = {
		years = zgrc_jin_ran_han_feng_cd
	}

	effect = {
		zgrc_jin_ran_han_feng_apply_effect = yes
	}

	ai_potential = {
		NOT = { has_title = title:h_china }
		zgrc_culture_unassimilated_trigger = yes
		NOT = { zgrc_hd_reform_path_trigger = yes }
		exists = culture.culture_head
		this = culture.culture_head
		OR = {
			# 路径①：独立统治者，邻接华夏霸权（汉人天子）
			AND = {
				is_independent_ruler = yes
				any_neighboring_and_across_water_top_liege_realm_owner = {
					has_title = title:h_china
					culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
				}
			}
			# 路径②：汉人统治者之臣
			AND = {
				NOT = { is_independent_ruler = yes }
				exists = top_liege
				top_liege = {
					culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
				}
			}
		}
	}

	ai_will_do = {
		base = -50
		modifier = { add = 20 knows_language = language_chinese }
		modifier = { add = 10 has_trait = shrewd }
		modifier = { add = 15 religion = religion:confucianism_religion }
		modifier = { add = 10 religion = religion:taoism_religion }
		modifier = { add = 15 has_trait = confucian_education }
		modifier = { add = 10 has_trait = calm }
	}
