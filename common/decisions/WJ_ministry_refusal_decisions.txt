WJ_ministry_refuse_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/tgp_retire.dds" }
	decision_group_type = admin
	sort_order = 76
	desc = WJ_ministry_refuse_decision_desc
	selection_tooltip = WJ_ministry_refuse_decision_tooltip
	confirm_text = WJ_ministry_refuse_decision_confirm
	is_shown = {
		is_alive = yes
		is_adult = yes
		government_has_flag = government_is_celestial
		this != top_liege
		top_liege = { tgp_has_access_to_ministry_trigger = yes }
		NOT = { has_character_flag = wj_refuses_ministry_appointments }
	}
	is_valid_showing_failures_only = { is_incapable = no }
	effect = {
		custom_tooltip = WJ_ministry_refuse_decision_effect_tt
		hidden_effect = {
			add_character_flag = wj_refuses_ministry_appointments
			add_character_flag = { flag = wj_ministry_refusal_ai_cooldown years = 10 }
		}
	}
	ai_check_interval = 120
	ai_potential = {
		is_ai = yes
		WJ_ai_ministry_local_authority_trigger = yes
		NOT = { has_character_flag = wj_ministry_refusal_ai_cooldown }
	}
	ai_will_do = {
		base = 100
		modifier = { factor = 0 WJ_ministry_refusal_desire_value < 20 }
	}
}
