	}

			text = hd_j16_lxuan_holds_yizhou_tt

		}
		trigger_if = {
			limit = { is_independent_ruler = no }
			custom_description = {
				text = hd_j16_lxuan_indep_war_tt
				NOT = { has_character_flag = hd_j16_lxuan_indep_war }
			}
			NOT = { is_at_war_with = liege }
		}
	}

			limit = { is_independent_ruler = no }
			custom_tooltip = hd_j16_lxuan_juhanzhi_indep_effect_tt
			hidden_effect = {
				save_scope_as = hd_j16_c4_jh_lx
				liege = { save_scope_as = hd_j16_c4_jh_lord }
				add_character_flag = { flag = hd_j16_lxuan_indep_war years = 5 }
				start_war = {
					cb = independence_war
					target = scope:hd_j16_c4_jh_lord
				}

			}
		}
		else = {
			custom_tooltip = hd_j16_lxuan_juhanzhi_name_effect_tt
			hidden_effect = {
				add_character_flag = hd_j16_lxuan_juhanzhi_used
				hd_nbc_found_realm_effect = { NAME = dynn_title_han STYLE = wang FIXED = no }
				primary_title = { set_variable = hd_j16_lxuan_han }
				debug_log = "HD_J16 lxuan FIRE juhanzhi_named"
			}
		}
	}

	picture = { reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds" }
	desc = hd_j16_lxuan_qibing_decision_desc
	selection_tooltip = hd_j16_lxuan_qibing_decision_tooltip
	confirm_text = hd_j16_lxuan_qibing_decision_confirm
	decision_group_type = major

	is_shown = {
		hd_j16_346c_lxuan_card_trigger = yes
		is_landed = no
		NOT = { has_character_flag = hd_j16_lxuan_qibing_used }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	is_valid = {
		custom_description = {
			text = hd_j16_lxuan_in_yizhou_tt
			hd_j16_346c_in_yizhou_trigger = yes
		}
		gold >= 100
		prestige >= 300
		custom_description = {
			text = hd_j16_lxuan_qibing_county_tt
			any_county_in_region = {
				region = uuii_han_yizhou_region
				hd_j16_346c_qibing_county_trigger = yes
			}
		}
	}

	cost = {
		gold = 100
		prestige = 300
	}

	effect = {
		custom_tooltip = hd_j16_lxuan_qibing_effect_tt
		hidden_effect = {
			add_character_flag = hd_j16_lxuan_qibing_used
			save_scope_as = hd_j16_c4_qb_lx
			hd_j16_dir_clear_scope_effect = { S = hd_j16_c4_qb_c }
			if = {
				limit = { location.county ?= { hd_j16_346c_qibing_county_trigger = yes } }
				location.county = { save_scope_as = hd_j16_c4_qb_c }
			}
			else_if = {
				limit = {
					title:d_shujun = {
						any_de_jure_county = { hd_j16_346c_qibing_county_trigger = yes }
					}
				}
				title:d_shujun = {
					random_de_jure_county = {
						limit = { hd_j16_346c_qibing_county_trigger = yes }
						save_scope_as = hd_j16_c4_qb_c
					}
				}
			}
			else = {
				random_county_in_region = {
					region = uuii_han_yizhou_region
					limit = { hd_j16_346c_qibing_county_trigger = yes }
					save_scope_as = hd_j16_c4_qb_c
				}
			}
			if = {
				limit = { exists = scope:hd_j16_c4_qb_c }
				hd_nbc_give_county_effect = { COUNTY = scope:hd_j16_c4_qb_c TO = scope:hd_j16_c4_qb_lx }
				hd_nbc_become_independent_effect = yes
				scope:hd_j16_c4_qb_c.duchy ?= {
					if = {
						limit = {
							NOT = { holder ?= scope:hd_j16_c4_qb_lx }
							NOT = { scope:hd_j16_c4_qb_lx = { has_claim_on = prev } }
						}
						save_scope_as = hd_j16_c4_qb_d
						scope:hd_j16_c4_qb_lx = { add_pressed_claim = scope:hd_j16_c4_qb_d }
					}
				}
				if = {
					limit = { scope:hd_j16_c4_qb_c.holder ?= scope:hd_j16_c4_qb_lx }
					hd_nbc_spawn_army_effect = { SIZE = 3000 WHERE = scope:hd_j16_c4_qb_c NAME = hd_j16_346c_army_lxuan }
				}
				trigger_event = hd_j16.1401
				debug_log = "HD_J16 lxuan FIRE qibing"
			}
			else = {
				debug_log = "HD_J16 lxuan WAIT qibing_no_county"
			}
		}
	}

	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
