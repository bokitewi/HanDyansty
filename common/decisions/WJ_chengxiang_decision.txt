# 创建丞相：特殊决议只负责首次授衔、入阁并建立原版固化摄政。
wj_create_cheng_xiang_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/tgp_kowtow.dds"
	}
	decision_group_type = major
	title = wj_create_cheng_xiang_decision
	desc = wj_create_cheng_xiang_decision_desc
	selection_tooltip = wj_create_cheng_xiang_decision_tooltip
	confirm_text = wj_create_cheng_xiang_decision_confirm
	sort_order = 70

	is_shown = {
		government_has_flag = government_is_celestial
		this != top_liege
		has_tgp_dlc_trigger = yes
		exists = title:h_china.holder
		top_liege = title:h_china.holder
		NOT = { has_title = title:e_WJ_minister_cheng_xiang }
	}

	is_valid = {
		WJ_chengxiang_decision_qualification = yes
	}

	effect = {
		if = {
			limit = { WJ_chengxiang_decision_qualification = yes }
			title:h_china.holder = { save_scope_as = councillor_liege }
   clear_saved_scope = swapped_position
   got_minister_position_effect = { MINISTER_TITLE = e_WJ_minister_cheng_xiang MINISTER_POSITION = WJ_minister_cheng_xiang }

			if = {
				limit = { has_title = title:e_WJ_minister_cheng_xiang has_council_position = WJ_minister_cheng_xiang }
				save_scope_as = wj_cx_announced
    title:h_china.holder = {
     if = { limit = { exists = diarch diarch = scope:wj_cx_announced } set_diarchy_swing = 60 }
     if = { limit = { is_ai = no } trigger_event = WJ_chancellor.1 }
     every_vassal_or_below = { limit = { is_ai = no } trigger_event = WJ_chancellor.1 }
     every_courtier = { limit = { is_ai = no } trigger_event = WJ_chancellor.1 }
    }
			}
		}
	}

	ai_check_interval = 12
	ai_potential = { is_diarch_of_target = title:h_china.holder }
	ai_will_do = { base = 100 }
}
