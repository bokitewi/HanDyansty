# Installed celestial_siphon_treasury_interaction.ai_will_do: every additive
# term precedes the three native factors. _character_interactions.info defines
# this as the final sending probability after target selection, clamped 0..100.
hd_dc_siphon_base_probability_value = {
	value = -50
	subtract = "opinion(scope:recipient)"
	add = { value = ai_boldness multiply = 0.25 }
	add = ai_greed
	add = { value = ai_honor multiply = -0.5 }
	if = { limit = { debt_level = 1 } add = 10 }
	if = { limit = { debt_level = 2 } add = 20 }
	if = { limit = { debt_level = 3 } add = 40 }
	if = { limit = { debt_level = 4 } add = 60 }
	if = { limit = { debt_level = 5 } add = 80 }
	if = { limit = { debt_level >= 6 } add = 100 }
	if = { limit = { has_trait = greedy } add = 30 }
	if = { limit = { has_trait = deceitful } add = 20 }
	if = { limit = { has_trait = arbitrary } add = 10 }
	if = {
		limit = {
			scope:actor = {
				OR = { is_vassal_or_below_of = scope:recipient is_liege_or_above_of = scope:recipient }
			}
		}
		add = -20
	}
	if = {
		limit = {
			scope:recipient = { has_title = title:h_china }
			situation:dynastic_cycle ?= {
				any_participant_group = {
					participant_group_type = conservative_movement
					NOT = { participant_group_has_character = scope:actor }
				}
			}
		}
		add = -100
	}
	if = {
		limit = { should_prioritise_hostile_action_against_due_to_personal_relations_trigger = { ACTOR = scope:actor RECIPIENT = scope:recipient } }
		multiply = 1.5
	}
	if = {
		limit = { is_in_same_movement_as = { TARGET = scope:recipient } }
		multiply = {
			value = 0.25
			if = { limit = { is_in_pro_hegemon_or_hegemon_movement = yes } multiply = 0.5 }
		}
	}
	if = {
		limit = { should_avoid_hostile_action_against_due_to_personal_relations_trigger = { ACTOR = scope:actor RECIPIENT = scope:recipient } }
		multiply = 0
	}
}

hd_dc_siphon_initiation_probability_value = {
	if = {
		limit = { hd_dc_scheme_hazard_inputs_trigger = { BASE = hd_dc_siphon_base_probability_value ACTOR = scope:actor } }
	}
	value = hd_dc_hazard_probability_value
}
