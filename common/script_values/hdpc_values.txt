# HDPC 1.0.0 | CK3 1.19 baseline | 软目标，不是强制人数上限。
hdpc_target_low_value = 8000
hdpc_start_threshold_value = 10000
hdpc_target_high_value = 11000
hdpc_interval_years_value = 10
hdpc_base_move_budget_value = 250
hdpc_max_move_budget_value = 2000
hdpc_min_age_value = 30
hdpc_talent_threshold_value = 16
hdpc_normal_court_share_value = 0.25
hdpc_high_court_share_value = 0.35

# Global scope. This is a MOVE budget, never a number of deaths.
# In CK3, max is an UPPER clamp and min is a LOWER clamp.
hdpc_move_budget_value = {
    value = global_var:hdpc_living
    subtract = hdpc_start_threshold_value
    add = hdpc_base_move_budget_value
    max = hdpc_max_move_budget_value
    min = 0
}

# Ruler scope. Baron-only courts are not traversed by this package.
hdpc_court_reserve_value = {
    value = 12
    if = {
        limit = { highest_held_title_tier >= tier_duchy }
        add = 4
    }
    if = {
        limit = { highest_held_title_tier >= tier_kingdom }
        add = 8
    }
    if = {
        limit = { highest_held_title_tier >= tier_empire }
        add = 8
    }
    if = {
        limit = { highest_held_title_tier >= tier_hegemony }
        add = 8
    }
}

# No rounding API is required: the executor requires remaining budget >= 1.
hdpc_court_move_budget_value = {
    value = local_var:hdpc_court_size
    if = {
        limit = { global_var:hdpc_pressure = 2 }
        multiply = hdpc_high_court_share_value
    }
    else = { multiply = hdpc_normal_court_share_value }
    max = {
        value = local_var:hdpc_court_size
        subtract = local_var:hdpc_court_reserve
    }
    min = 0
}
