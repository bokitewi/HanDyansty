# Scope：发动者。目标军力上限 = 自身当前军力 x2（沿用曹/袁引擎）
hd_strat_strength_ceiling_value = {
	value = current_military_strength
	multiply = 2
}

# Scope：候选目标（需 scope:hd_strat_actor / scope:hd_strat_region）。越大越优先
hd_strat_target_priority_value = {
	value = 0
	if = {
		limit = {
			scope:hd_strat_actor = {
				OR = {
					has_relation_nemesis = prev
					has_relation_rival = prev
				}
			}
		}
		add = 100
	}
	if = {
		limit = { scope:hd_strat_actor = { opinion = { target = prev value < 50 } } }
		add = 50
	}
	if = {
		limit = { capital_province ?= { geographical_region = scope:hd_strat_region } }
		add = 30
	}
	# 军力越弱越优先
	subtract = {
		value = current_military_strength
		divide = 1000
		max = 29
	}
}

# Scope：角色。文档"开拓"权重：勇敢30 野心勃勃30 冷酷15 暴怒10 狂热10

}
