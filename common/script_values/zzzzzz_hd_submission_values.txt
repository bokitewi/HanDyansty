# Interaction scope: attacker CURRENT strength / recipient MAXIMUM strength.
# The denominator is clamped to 1; one lost battle does not manufacture a 100x ratio.
hd_submission_power_ratio = {
    value = scope:actor.current_military_strength
    divide = { value = scope:recipient.max_military_strength min = 1 }
}

# Interaction scope. Same-faith = zero. Preserve pluralism/hostility distinctions.
hd_submission_faith_value = {
    value = 0
    if = {
        limit = { scope:recipient.faith != scope:actor.faith }
        add = -40
        if = {
            limit = { scope:recipient.faith = { faith_hostility_level = { target = scope:actor.faith value >= faith_hostile_level } } }
            add = -40
        }
        if = {
            limit = { scope:recipient.faith = { faith_hostility_level = { target = scope:actor.faith value >= faith_evil_level } } }
            add = -40
        }
        if = {
            limit = { scope:recipient.faith = { has_doctrine = doctrine_pluralism_pluralistic } }
            multiply = 0.5
            if = {
                limit = { scope:actor.faith = { has_doctrine = doctrine_pluralism_pluralistic } }
                multiply = 0.5
            }
        }
    }
}

hd_submission_legitimacy_value = {
    value = 0
    if = { limit = { scope:actor = { has_legitimacy_flag = extra_increased_vassalization_acceptance } } add = 75 }
    else_if = { limit = { scope:actor = { has_legitimacy_flag = very_increased_vassalization_acceptance } } add = 50 }
    else_if = { limit = { scope:actor = { has_legitimacy_flag = increased_vassalization_acceptance } } add = 25 }
    else_if = { limit = { scope:actor = { has_legitimacy_flag = massively_reduced_vassalization_acceptance } } add = -75 }
    else_if = { limit = { scope:actor = { has_legitimacy_flag = very_reduced_vassalization_acceptance } } add = -50 }
    else_if = { limit = { scope:actor = { has_legitimacy_flag = reduced_vassalization_acceptance } } add = -25 }
}

hd_submission_momentum_acceptance_value = {
    value = 0
    if = {
        limit = { scope:actor = { has_variable = hd_submission_momentum } }
        add = scope:actor.var:hd_submission_momentum
        multiply = 5
        if = { limit = { scope:actor.var:hd_submission_momentum >= 5 } add = 5 }
    }
}

# Character scope; 0/1/2/3/4/5 momentum -> 70/65/60/55/50/45 percent.
hd_submission_recovery_fraction = {
    value = 0.70
    if = {
        limit = { has_variable = hd_submission_momentum }
        subtract = { value = var:hd_submission_momentum multiply = 0.05 }
    }
    min = 0.45
    max = 0.70
}

hd_submission_minimum_army_value = {
    value = max_military_strength
    multiply = hd_submission_recovery_fraction
}
