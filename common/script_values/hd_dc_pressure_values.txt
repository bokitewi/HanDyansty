# Native progress is not a second W/H authority. Only the persistent W selects
# a band. Exact 20/40/60/80 percent boundaries enter the higher band.
hd_dc_pressure_band_value = {
	value = 1
	if = { limit = { var:hd_dc_pressure_raw >= 1600 } add = 4 }
	else_if = { limit = { var:hd_dc_pressure_raw >= 1200 } add = 3 }
	else_if = { limit = { var:hd_dc_pressure_raw >= 800 } add = 2 }
	else_if = { limit = { var:hd_dc_pressure_raw >= 400 } add = 1 }
}

# County scope. These are cumulative targets, not a monthly percentage levy.
hd_dc_control_loss_target_value = {
	value = situation:dynastic_cycle.var:hd_dc_pressure_band
	subtract = 1
	multiply = 0.05
	multiply = var:hd_dc_control_baseline
}
hd_dc_development_loss_target_value = {
	value = situation:dynastic_cycle.var:hd_dc_pressure_band
	subtract = 1
	multiply = 0.02
	multiply = var:hd_dc_development_baseline
	floor = yes
}
hd_dc_control_loss_due_value = {
	value = hd_dc_control_loss_target_value
	subtract = var:hd_dc_control_processed
	min = 0
	max = county_control
}
hd_dc_development_loss_due_value = {
	value = hd_dc_development_loss_target_value
	subtract = var:hd_dc_development_processed
	min = 0
	max = development_level
}
