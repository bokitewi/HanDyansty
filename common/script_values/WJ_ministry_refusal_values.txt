# Decision taker scope/root. Positive values favor keeping local authority.
# No offered title exists here; court distance is only a general proxy.
WJ_ministry_refusal_desire_value = {
	value = 0
	if = {
		limit = { WJ_ministry_major_local_authority_trigger = yes }
		add = 25
	}
	else_if = {
		limit = { WJ_ai_ministry_local_authority_trigger = yes }
		add = 10
	}
	if = { limit = { WJ_dd_head_trigger = yes } add = 20 }
	if = {
		limit = { exists = capital_province exists = top_liege.capital_province }
		if = {
			limit = {
				capital_province = {
					squared_distance = { target = root.top_liege.capital_province value > 130000 }
				}
			}
			add = 15
		}
		else = { add = -10 }
	}
	if = {
		limit = { has_trait = ambitious }
		if = { limit = { WJ_ministry_major_local_authority_trigger = yes } add = 10 }
		else = { add = -15 }
	}
	if = { limit = { has_trait = content } add = 10 }
	if = { limit = { has_trait = loyal } add = -20 }
	if = { limit = { has_trait = disloyal } add = 15 }
	if = { limit = { has_trait = paranoid } add = 10 }
	if = { limit = { has_trait = trusting } add = -10 }
	if = { limit = { opinion = { target = top_liege value >= 60 } } add = -15 }
	else_if = { limit = { opinion = { target = top_liege value <= -40 } } add = 20 }
	if = { limit = { has_relation_best_friend = top_liege } add = -25 }
	else_if = { limit = { has_relation_friend = top_liege } add = -15 }
	if = { limit = { has_relation_nemesis = top_liege } add = 30 }
	else_if = { limit = { has_relation_rival = top_liege } add = 20 }
	if = { limit = { is_close_family_of = top_liege } add = -10 }
}
