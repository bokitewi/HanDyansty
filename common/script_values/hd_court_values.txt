	value = 50
	if = { limit = { has_trait = honest } add = 25 }
	if = { limit = { has_trait = just } add = 25 }
	if = { limit = { has_trait = loyal } add = 40 }
	if = { limit = { has_trait = content } add = 20 }
	if = { limit = { has_trait = humble } add = 20 }
	if = { limit = { has_trait = compassionate } add = 10 }
	if = { limit = { hd_cycle_non_han_trigger = yes } multiply = 0.3 }
	if = {
		limit = {
			exists = scope:hd_court_emperor_for_stance
			opinion = { target = scope:hd_court_emperor_for_stance value >= 30 }
		}
		add = 30
	}

		limit = {
			exists = scope:hd_court_emperor_for_stance
			dynasty ?= scope:hd_court_emperor_for_stance.dynasty
		}
		add = 100
	}
	min = 1
}
