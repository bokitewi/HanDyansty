# 天朝循环大修·朋党：覆盖两个原版 script value（同名后加载者生效），只在末尾追加朋党钩子。
# 原版位置：07_appointment_values.txt / 00_ep1_script_values.txt。游戏更新后需对比原版重新合并。

appointment_score_final_factors = {
	add = appointment_expand_power_base_bonus
	if = {
		limit = {
			tgp_realm_has_ceremonial_liege_trigger = yes
		}
		# Petition
		if = {
			limit = {
				exists = var:petition_liege_regent_support
				top_liege.primary_title.var:administrative_ui_special_title.holder ?= var:petition_liege_regent_support
				scope:title ?= top_liege.primary_title # Only top liege title
			}
			multiply = {
				value = 1.1
				desc = petition_liege_regent_support_desc
			}
		}
		# PETITION
		else_if = {
			limit = {
				exists = house.var:petition_liege_appointment_support
				top_liege.primary_title.var:administrative_ui_special_title.holder ?= house.var:petition_liege_appointment_support
				exists = scope:title
				scope:title ?= top_liege.primary_title # Only governorships
			}
			multiply = {
				value = 1.15
				desc = ceremonial_liege_appointment_support_desc
			}
		}
	}

	# 天朝循环大修·朋党：被青睐朋党的成员任命分数 +15%；政策查禁 −50 / 限制 −20 / 促进 +20
	if = {
		limit = { hd_party_member_favored_trigger = yes }
		multiply = {
			value = 1.15
			desc = HD_PARTY_APPOINT_FAVORED
		}
	}
	if = {
		limit = { hd_party_member_policy_is_trigger = { POLICY = ban } }
		add = {
			value = -50
			desc = HD_PARTY_APPOINT_BAN
		}
	}
	else_if = {
		limit = { hd_party_member_policy_is_trigger = { POLICY = restrict } }
		add = {
			value = -20
			desc = HD_PARTY_APPOINT_RESTRICT
		}
	}
	else_if = {
		limit = { hd_party_member_policy_is_trigger = { POLICY = promote } }
		add = {
			value = 20
			desc = HD_PARTY_APPOINT_PROMOTE
		}
	}
}

court_position_candidate_score_base_value = {
	scope:liege = {
		# Not very likely to hire rivals
		if = {
			limit = {
				has_relation_rival = scope:employee
			}
			add = -1000
		}
		if = {
			limit = {
				NOT = { exists = scope:firing_court_position }
				scope:employee = { has_any_court_position = yes }
			}
			add = -150
		}
	}
	# 天朝循环大修·朋党：宫廷职位候选分（被青睐 +20；政策 ±）
	scope:employee = {
		if = {
			limit = { hd_party_member_favored_trigger = yes }
			add = 20
		}
		if = {
			limit = { hd_party_member_policy_is_trigger = { POLICY = ban } }
			add = -50
		}
		else_if = {
			limit = { hd_party_member_policy_is_trigger = { POLICY = restrict } }
			add = -20
		}
		else_if = {
			limit = { hd_party_member_policy_is_trigger = { POLICY = promote } }
			add = 20
		}
	}
}

# 天朝循环大修·朋党：门生势力加成只计同一政权（同一最高领主）的门生。原版位置：00_interaction_values.txt
every_disciple_movement_power_value = {
	value = 0
	every_relation = {
		type = disciple
		limit = { top_liege ?= prev.top_liege }
		if = {
			limit = { has_variable = movement_power }
			add = var:movement_power
		}
		else = {
			add = movement_power_character_value
		}
	}
}
