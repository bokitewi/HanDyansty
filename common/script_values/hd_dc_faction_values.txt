# P is independent of scheme F and has no extra chaos multiplier. The native
# faction target determines whether this faction belongs to the affected area.
hd_dc_faction_hazard_multiplier_value = {
	value = 1
	if = {
		limit = { exists = situation:dynastic_cycle any_character_situation = { this = situation:dynastic_cycle } }
		switch = {
			trigger = situation:dynastic_cycle.var:hd_dc_pressure_band
			2 = { value = 1.1 }
			3 = { value = 1.25 }
			4 = { value = 1.5 }
			5 = { value = 1.8 }
		}
	}
}

# Exact source sums in ai_demand_chance; no intermediate clamp, no alteration
# to membership, discontent, power thresholds, creation scores or war results.
hd_dc_independence_demand_base_value = {
	value = 0
	add = { value = faction_power multiply = 0.5 }
	if = { limit = { faction_power > 110 } add = { value = faction_power multiply = 1 } }
	if = { limit = { faction_target = { is_at_war = yes } } add = 100 }
	if = {
		limit = { faction_target = { involved_activity ?= { has_activity_type = activity_coronation activity_host = prev } } }
		add = -1000
	}
}

hd_dc_ceremonial_liberty_demand_base_value = {
	value = 0
	add = { value = faction_power multiply = 0.375 }
	if = { limit = { faction_power > 110 } add = { value = faction_power multiply = 0.75 } }
	if = {
		limit = { faction_target = { any_war_enemy = { NOT = { target_is_liege_or_above = root.faction_target } } } }
		add = -75
	}
}

hd_dc_liberty_demand_base_value = {
	value = hd_dc_ceremonial_liberty_demand_base_value
	if = {
		limit = { faction_target = { involved_activity ?= { has_activity_type = activity_coronation activity_host = prev } } }
		add = -1000
	}
}
