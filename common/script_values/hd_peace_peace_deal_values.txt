# Peace Deals Expanded — values used by the war overview negotiation panel.
# ROOT is always the character composing the proposal. The active war and the
# opposing primary war leader are stored as variables by hd_peace_open_treaty_effect.

hd_peace_actor_warscore_value = {
	value = 0
	if = {
		limit = { exists = var:hd_peace_treaty_war }
		var:hd_peace_treaty_war = {
			if = {
				limit = { primary_attacker = prev }
				add = attacker_war_score
			}
			else = {
				add = defender_war_score
			}
		}
	}
}

# A payer may commit all current liquid gold plus one complete year of positive
# income. This intentionally permits debt up to that future-income allowance.
hd_peace_max_offer_gold_value = {
	value = gold
	add = {
		value = monthly_character_income
		multiply = 12
		min = 0
	}
	min = 0
	floor = yes
}

hd_peace_max_demand_gold_value = {
	value = 0
	if = {
		limit = { exists = var:hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			add = gold
			add = {
				value = monthly_character_income
				multiply = 12
				min = 0
			}
		}
	}
	min = 0
	floor = yes
}

# Herd can only be exchanged between nomadic rulers. Unlike gold, negotiated
# herd cannot create debt: the payer's current herd is the hard cap.
hd_peace_max_offer_herd_value = {
	value = 0
	if = {
		limit = { hd_peace_nomadic_herd_exchange_available_trigger = yes }
		add = domicile.herd
	}
	min = 0
	floor = yes
}

hd_peace_max_demand_herd_value = {
	value = 0
	if = {
		limit = { hd_peace_nomadic_herd_exchange_available_trigger = yes }
		var:hd_peace_treaty_opponent = { add = domicile.herd }
	}
	min = 0
	floor = yes
}

hd_peace_default_offer_herd_value = {
	value = hd_peace_max_offer_herd_value
	multiply = 0.25
	min = 1
	floor = yes
}

hd_peace_default_demand_herd_value = {
	value = hd_peace_max_demand_herd_value
	multiply = 0.25
	min = 1
	floor = yes
}
hd_peace_offer_county_count_value = {
	value = 0
	if = {
		limit = { has_variable_list = hd_peace_offer_counties }
		every_in_list = {
			variable = hd_peace_offer_counties
			add = 1
		}
	}
}

hd_peace_demand_county_count_value = {
	value = 0
	if = {
		limit = { has_variable_list = hd_peace_demand_counties }
		every_in_list = {
			variable = hd_peace_demand_counties
			add = 1
		}
	}
}

# The fixed yearly payment is 20% of the payer's annual income. It is stored at
# signature time, so later income changes do not silently rewrite the treaty.
hd_peace_actor_annual_tribute_value = {
	value = monthly_character_income
	multiply = 2.4
	min = 1
	floor = yes
}

hd_peace_opponent_annual_tribute_value = {
	value = 0
	if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = { add = hd_peace_actor_annual_tribute_value }
	}
}

# Severity of concessions offered by the player. Concessions make acceptance
# easier, hence they are added back in hd_peace_deal_acceptance_value.
hd_peace_offer_cost_value = {
	value = 0
	if = {
		limit = { has_variable = hd_peace_offer_gold }
		add = {
			value = var:hd_peace_offer_gold
			divide = {
				value = hd_peace_max_offer_gold_value
				min = 1
			}
			multiply = 45
		}
	}
	if = {
		limit = { has_variable = hd_peace_offer_herd }
		add = {
			value = var:hd_peace_offer_herd
			divide = {
				value = hd_peace_max_offer_herd_value
				min = 1
			}
			multiply = 45
		}
	}
	if = { limit = { has_variable_list = hd_peace_offer_counties } every_in_list = { variable = hd_peace_offer_counties add = hd_peace_county_price_value } }
	if = { limit = { has_variable_list = hd_peace_offer_released_vassals } every_in_list = { variable = hd_peace_offer_released_vassals add = hd_peace_release_vassal_price_value } }
	if = { limit = { has_variable = hd_peace_offer_tribute } add = 30 }
	if = { limit = { has_variable = hd_peace_offer_suzerainty } add = 75 }
	if = { limit = { has_variable = hd_peace_offer_claims } add = 18 }
	if = { limit = { has_variable = hd_peace_offer_hostage } add = 18 }
	if = { limit = { has_variable = hd_peace_offer_break_alliances } add = 22 }
	if = { limit = { has_variable = hd_peace_offer_humiliation } add = 18 }
	if = { limit = { has_variable = hd_peace_offer_prisoner_release } add = 12 }
	if = { limit = { has_variable = hd_peace_offer_conversion } add = 40 }
	if = { limit = { has_variable = hd_peace_offer_disarmament } add = 28 }
}

# Severity of demands imposed on the opponent.
hd_peace_demand_cost_value = {
	value = 0
	if = { limit = { has_variable = hd_peace_objective } add = hd_peace_objective_price_value }
	if = {
		limit = { has_variable = hd_peace_demand_gold }
		add = {
			value = var:hd_peace_demand_gold
			divide = {
				value = hd_peace_max_demand_gold_value
				min = 1
			}
			multiply = 45
		}
	}
	if = {
		limit = { has_variable = hd_peace_demand_herd }
		add = {
			value = var:hd_peace_demand_herd
			divide = {
				value = hd_peace_max_demand_herd_value
				min = 1
			}
			multiply = 45
		}
	}
	if = { limit = { has_variable_list = hd_peace_demand_counties } every_in_list = { variable = hd_peace_demand_counties add = hd_peace_demand_county_price_value } }
	if = { limit = { has_variable_list = hd_peace_demand_titles } every_in_list = { variable = hd_peace_demand_titles add = hd_peace_demand_title_price_value } }
	if = { limit = { has_variable_list = hd_peace_demand_released_vassals } every_in_list = { variable = hd_peace_demand_released_vassals add = hd_peace_release_vassal_price_value } }
	if = { limit = { has_variable = hd_peace_demand_tribute } add = 30 }
	if = { limit = { has_variable = hd_peace_demand_suzerainty } add = 75 }
	if = { limit = { has_variable = hd_peace_demand_claims } add = 18 }
	if = { limit = { has_variable = hd_peace_demand_hostage } add = 18 }
	if = { limit = { has_variable = hd_peace_demand_break_alliances } add = 22 }
	if = { limit = { has_variable = hd_peace_demand_humiliation } add = 18 }
	if = { limit = { has_variable = hd_peace_demand_prisoner_release } add = 12 }
	if = { limit = { has_variable = hd_peace_demand_conversion } add = 40 }
	if = { limit = { has_variable = hd_peace_demand_disarmament } add = 28 }
}

hd_peace_common_term_value = {
	value = 0
	if = { limit = { has_variable = hd_peace_long_truce } add = 5 }
	if = { limit = { has_variable = hd_peace_extended_truce } add = 8 }
	if = { limit = { has_variable = hd_peace_forced_alliance } add = 5 }
}

# Positive values favour the proposer; negative values favour the opponent.
# The three-point dead zone keeps merely symbolic payments neutral.
hd_peace_treaty_net_value = {
 value = 0
 if = { limit = { has_variable = hd_peace_locked_net } add = var:hd_peace_locked_net }
 else = { add = hd_peace_demand_cost_value subtract = hd_peace_offer_cost_value round = yes }
}
# Positive means the opposing AI accepts. War score is real leverage, while
# offered and demanded clauses are evaluated symmetrically.
hd_peace_deal_acceptance_value = {
	value = hd_peace_actor_warscore_value
	add = hd_peace_offer_cost_value
	subtract = hd_peace_demand_cost_value
	add = hd_peace_common_term_value
	add = hd_peace_war_outcome_adjustment_value
	subtract = hd_peace_war_resistance_value
	round = yes
}
