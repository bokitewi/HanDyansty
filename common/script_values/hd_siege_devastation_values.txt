	value = 0
	location = {
		every_army_in_location = {
			if = {
				limit = {
					is_army_in_siege = yes

# 围城完成回调：root 是实际占领者；当前作用域为被占领省份。
hd_siege_destroy_building_soldier_value = {
	value = 0
	every_army_in_location = {
		limit = {
			army_owner = {
				OR = {
					this = root
					scope:war = {
						is_attacker = root
						is_attacker = prev
					}
					scope:war = {
						is_defender = root
						is_defender = prev
					}
				}
			}
		}
		add = army_size
	}
	divide = 100000
	add = 1
}
