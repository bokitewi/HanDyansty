# Only war-settlement callers with the real attacker/defender scopes use this.
# Keep the original fixed duration; do not import standard-truce perk rules.
hd_dc_fixed_five_year_war_truce_days = {
	value = 1825
	if = {
		limit = {
			OR = {
				scope:attacker = { hd_dc_in_chaos_region_trigger = yes }
				scope:defender = { hd_dc_in_chaos_region_trigger = yes }
			}
		}
		multiply = 0.5
	}
}

hd_dc_fixed_ten_year_war_truce_days = {
	value = hd_dc_fixed_five_year_war_truce_days
	multiply = 2
}

# Adventure victories create protection for the ruler's relinquished territory.
hd_dc_fixed_forty_year_war_truce_days = {
	value = hd_dc_fixed_five_year_war_truce_days
	multiply = 8
}

# The caller freezes the existing file-local 5/15 year value before evaluation.
# The real tributary war supplies its combatants; negotiated tribute is excluded.
hd_dc_tribute_war_truce_years = {
	value = scope:hd_dc_truce_base_years
	if = {
		limit = {
			OR = {
				scope:tributary_war.primary_attacker = { hd_dc_in_chaos_region_trigger = yes }
				scope:tributary_war.primary_defender = { hd_dc_in_chaos_region_trigger = yes }
			}
		}
		multiply = 0.5
	}
}

# A petition can settle several different wars; use this iteration's real war.
hd_dc_petition_war_truce_years = {
	value = 5
	if = {
		limit = {
			OR = {
				scope:this_war.primary_attacker = { hd_dc_in_chaos_region_trigger = yes }
				scope:this_war.primary_defender = { hd_dc_in_chaos_region_trigger = yes }
			}
		}
		multiply = 0.5
	}
}
