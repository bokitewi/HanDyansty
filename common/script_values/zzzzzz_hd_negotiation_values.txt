# ===== 劝其归附 ai_accept 的条件项（interaction scope：actor = 受附者，recipient = 归附者）=====
# 同时认窗口选项（玩家预开价）与归附者身上的条件变量（AI 预开价、on_send 抄录），二者择一不叠加
hd_neg_gold_accept_value = {
	value = 0
	if = {
		limit = { exists = scope:hd_neg_o_gold3 scope:hd_neg_o_gold3 = yes }
		add = hd_neg_gold_accept_3
	}
	else_if = {
		limit = { exists = scope:hd_neg_o_gold2 scope:hd_neg_o_gold2 = yes }
		add = hd_neg_gold_accept_2
	}
	else_if = {
		limit = { exists = scope:hd_neg_o_gold1 scope:hd_neg_o_gold1 = yes }
		add = hd_neg_gold_accept_1
	}
	else_if = {
		limit = {
			scope:recipient = {
				var:hd_neg_liege ?= scope:actor
				has_variable = hd_neg_gold_tier
			}
		}
		if = {
			limit = { scope:recipient.var:hd_neg_gold_tier >= 3 }
			add = hd_neg_gold_accept_3
		}
		else_if = {
			limit = { scope:recipient.var:hd_neg_gold_tier = 2 }
			add = hd_neg_gold_accept_2
		}
		else = { add = hd_neg_gold_accept_1 }
	}
}
