# Potential outbreak province scope, after the disease's existing local rules.
# Only the fixed regional county set receives D; no extra outbreak is spawned.
hd_dc_epidemic_hazard_multiplier_value = {
	value = 1
	if = {
		limit = {
			exists = situation:dynastic_cycle
			exists = county
			county = { save_temporary_scope_as = hd_dc_epidemic_county }
			situation:dynastic_cycle = {
				is_target_in_variable_list = { name = hd_dc_counties target = scope:hd_dc_epidemic_county }
			}
		}
		switch = {
			trigger = situation:dynastic_cycle.var:hd_dc_pressure_band
			2 = { value = 1.05 }
			3 = { value = 1.10 }
			4 = { value = 1.15 }
			5 = { value = 1.20 }
		}
	}
}
