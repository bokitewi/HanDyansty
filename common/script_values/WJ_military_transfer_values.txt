# Score components use the issuing character as ROOT, A alone supplies acceptance.
WJ_mt_trait_score = {
 value = 0
 if = { limit = { exists = var:wj_mt_a }
  var:wj_mt_a = {
   if = { limit = { has_trait = loyal } add = 30 }
   if = { limit = { has_trait = disloyal } add = -25 }
   if = { limit = { has_trait = ambitious } add = -20 }
   if = { limit = { has_trait = content } add = 15 }
   if = { limit = { has_trait = humble } add = 8 }
   if = { limit = { has_trait = arrogant } add = -8 }
   if = { limit = { has_trait = generous } add = 8 }
   if = { limit = { has_trait = greedy } add = -10 }
   if = { limit = { has_trait = calm } add = 5 }
   if = { limit = { has_trait = wrathful } add = -5 }
   if = { limit = { has_trait = stubborn } add = -8 }
   if = { limit = { has_trait = patient } add = 5 }
  }
 }
}
WJ_mt_opinion_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a } var:wj_mt_a = { value = "opinion(scope:wj_mt_operator)" } divide = 5 }
 min = -20 max = 20
}
WJ_mt_trust_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a }
  var:wj_mt_a = {
   if = { limit = { has_relation_best_friend = scope:wj_mt_operator } add = 15 }
   else_if = { limit = { has_relation_friend = scope:wj_mt_operator } add = 10 }
   if = { limit = { exists = dynasty exists = scope:wj_mt_operator.dynasty dynasty = scope:wj_mt_operator.dynasty } add = 8 }
   if = { limit = { scope:wj_mt_operator = { has_relation_disciple = prev } } add = 10 }
   if = { limit = { WJ_mufu_member_valid = yes var:wj_mufu_owner = scope:wj_mt_operator } add = 10 }
  }
 }
 max = 20
}
WJ_mt_hostility_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a }
  var:wj_mt_a = {
   if = { limit = { has_relation_nemesis = scope:wj_mt_operator } add = -20 }
   else_if = { limit = { has_relation_rival = scope:wj_mt_operator } add = -15 }
  }
 }
}
WJ_mt_authority_score = {
 value = 0
 if = { limit = { WJ_zw_valid = yes } add = 15 }
 else_if = { limit = { vassal_contract_has_flag = WJ_shichijie } add = 15 }
 else_if = { limit = { vassal_contract_has_flag = WJ_chijie } add = 10 }
 else_if = { limit = { vassal_contract_has_flag = WJ_jiajie } add = 5 }
 # The emperor cannot commit insubordination against a subordinate.
 if = { limit = { exists = var:wj_mt_a NOT = { var:wj_mt_a = top_liege } } add = 5
  if = { limit = { WJ_zw_huangyue_valid = yes } add = 10 }
 }
}
WJ_mt_war_score = {
 value = 0
 if = { limit = { exists = var:wj_mt_a var:wj_mt_a = { is_at_war = yes } } add = -10 }
}
WJ_mt_count_score = {
 value = 0 every_in_list = { variable = wj_mt_pending add = 1 }
 subtract = 1 min = 0 max = 6 multiply = -4
}
WJ_mt_fraction_value = {
 value = 0
 if = { limit = { var:wj_mt_source_troops > 0 }
  value = var:wj_mt_pending_troops divide = var:wj_mt_source_troops multiply = 100
 }
}
WJ_mt_fraction_score = {
 value = 0
 if = { limit = { WJ_mt_fraction_value > 75 } value = -20 }
 else_if = { limit = { WJ_mt_fraction_value > 50 } value = -10 }
 else_if = { limit = { WJ_mt_fraction_value > 25 } value = -5 }
 else = { value = 0 }
}
WJ_mt_acceptance = {
 value = 50
 add = WJ_mt_trait_score add = WJ_mt_opinion_score add = WJ_mt_trust_score
 add = WJ_mt_hostility_score add = WJ_mt_authority_score add = WJ_mt_war_score
 add = WJ_mt_count_score add = WJ_mt_fraction_score
 min = 5 max = 95
 if = { limit = { exists = var:wj_mt_a var:wj_mt_a = this } value = 100 }
}
# 20 gold per full size step; paid to A even when A is the issuing player.
WJ_mt_cost = {
 value = 0
 if = { limit = { var:wj_mt_mode = 2 } every_in_list = { variable = wj_mt_pending add = { value = maa_size multiply = 20 } } }
}
