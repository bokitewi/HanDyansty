# Score components use the issuing character as ROOT, A alone supplies acceptance.
WJ_mt_trait_score = {
 value = 0
 if = { limit = { exists = var:wj_mt_a } var:wj_mt_a = {
  if = { limit = { has_trait = loyal } add = 50 }
  if = { limit = { has_trait = disloyal } add = -40 }
  if = { limit = { has_trait = ambitious } add = -30 }
  if = { limit = { has_trait = content } add = 25 }
  if = { limit = { has_trait = humble } add = 15 }
  if = { limit = { has_trait = arrogant } add = -15 }
  if = { limit = { has_trait = generous } add = 15 }
  if = { limit = { has_trait = greedy } add = -15 }
  if = { limit = { has_trait = calm } add = 10 }
  if = { limit = { has_trait = wrathful } add = -10 }
  if = { limit = { has_trait = stubborn } add = -15 }
  if = { limit = { has_trait = patient } add = 10 }
  if = { limit = { has_trait = diligent } add = 15 }
  if = { limit = { has_trait = honest } add = 15 }
  if = { limit = { has_trait = just } add = 15 }
  if = { limit = { has_trait = craven } add = 10 }
  if = { limit = { has_trait = forgiving } add = 10 }
  if = { limit = { has_trait = paranoid } add = -15 }
  if = { limit = { has_trait = vengeful } add = -15 }
  if = { limit = { has_trait = lazy } add = -10 }
  if = { limit = { has_trait = arbitrary } add = -10 }
  if = { limit = { has_trait = impatient } add = -10 }
 } }
}
WJ_mt_opinion_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a } var:wj_mt_a = { value = "opinion(scope:wj_mt_operator)" } multiply = 0.3 }
 min = -30 max = 30
}
WJ_mt_trust_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a }
  var:wj_mt_a = {
   if = { limit = { has_relation_best_friend = scope:wj_mt_operator } add = 30 }
   else_if = { limit = { has_relation_friend = scope:wj_mt_operator } add = 20 }
   if = { limit = { exists = dynasty exists = scope:wj_mt_operator.dynasty dynasty = scope:wj_mt_operator.dynasty } add = 15 }
   if = { limit = { scope:wj_mt_operator = { has_relation_disciple = prev } } add = 20 }
   if = { limit = { WJ_mufu_member_valid = yes var:wj_mufu_owner = scope:wj_mt_operator } add = 20 }
  }
 }
}
WJ_mt_hostility_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a }
  var:wj_mt_a = {
   if = { limit = { has_relation_nemesis = scope:wj_mt_operator } add = -35 }
   else_if = { limit = { has_relation_rival = scope:wj_mt_operator } add = -20 }
  }
 }
}
# Exactly one mandate level; office command is counted independently below.
WJ_mt_command_score = {
 value = 0
 if = { limit = { WJ_inner_emperor_trigger = yes } value = 50 }
 else_if = { limit = { WJ_zw_huangyue_valid = yes } value = 40 }
 else_if = { limit = { vassal_contract_has_flag = WJ_shichijie } value = 30 }
 else_if = { limit = { vassal_contract_has_flag = WJ_chijie } value = 20 }
 else_if = { limit = { vassal_contract_has_flag = WJ_jiajie } value = 10 }
}
WJ_mt_zhongwai_score = { value = 0 if = { limit = { WJ_zw_valid = yes } value = 25 } }
WJ_mt_dasima_score = { value = 0 if = { limit = { WJ_mt_dasima = yes } value = 20 } }
WJ_mt_crime_score = { value = 0 if = { limit = { exists = var:wj_mt_a NOT = { var:wj_mt_a = top_liege } } value = 5 } }

WJ_mt_authority_score = { value = WJ_mt_command_score add = WJ_mt_zhongwai_score add = WJ_mt_dasima_score add = WJ_mt_crime_score }
WJ_mt_martial_score = { value = martial multiply = 0.5 max = 20 }
WJ_mt_reinforcement_score = { value = 0 if = { limit = { exists = var:wj_mt_b var:wj_mt_b = { is_at_war = yes } } value = 15 } }
WJ_mt_war_score = { value = 0 if = { limit = { exists = var:wj_mt_a var:wj_mt_a = { is_at_war = yes } } value = -20 } }
WJ_mt_count_score = {
 value = 0 every_in_list = { variable = wj_mt_pending add = 1 }
 subtract = 1 min = 0 max = 6 multiply = -4
}
WJ_mt_fraction_value = {
 value = 0
 if = { limit = { var:wj_mt_source_troops > 0 }
  value = var:wj_mt_pending_troops divide = var:wj_mt_source_troops multiply = 100
 }
}
WJ_mt_fraction_score = {
 value = 0
 if = { limit = { WJ_mt_fraction_value > 75 } value = -30 }
 else_if = { limit = { WJ_mt_fraction_value > 50 } value = -15 }
 else_if = { limit = { WJ_mt_fraction_value > 25 } value = -5 }
}
WJ_mt_warlord_score = { value = 0 if = { limit = { exists = var:wj_mt_a var:wj_mt_a = { vassal_contract_has_flag = hd_warlord } } value = -15 } }
WJ_mt_receiver_relation_score = {
 value = 0 save_temporary_scope_as = wj_mt_operator
 if = { limit = { exists = var:wj_mt_a exists = var:wj_mt_b NOT = { var:wj_mt_b = this } }
  var:wj_mt_a = {
   if = { limit = { has_relation_best_friend = scope:wj_mt_operator.var:wj_mt_b } add = 20 }
   else_if = { limit = { has_relation_friend = scope:wj_mt_operator.var:wj_mt_b } add = 10 }
   if = { limit = { has_relation_nemesis = scope:wj_mt_operator.var:wj_mt_b } add = -30 }
   else_if = { limit = { has_relation_rival = scope:wj_mt_operator.var:wj_mt_b } add = -20 }
  }
 }
}
# Read the actual top liege law: native indices 0..3 are displayed levels 1..4.
WJ_mt_crown_score = {
 value = 0
 top_liege = {
  if = { limit = { has_realm_law = celestial_bureaucracy_3 } value = 30 }
  else_if = { limit = { has_realm_law = celestial_bureaucracy_2 } value = 15 }
  else_if = { limit = { has_realm_law = celestial_bureaucracy_1 } value = 5 }
  else_if = { limit = { has_realm_law = celestial_bureaucracy_0 } value = -10 }
 }
}
WJ_mt_order_score = {
 value = 50
 add = WJ_mt_trait_score add = WJ_mt_opinion_score add = WJ_mt_trust_score add = WJ_mt_hostility_score
 add = WJ_mt_authority_score add = WJ_mt_crown_score add = WJ_mt_martial_score
}
WJ_mt_consequence_score = { value = WJ_mt_war_score add = WJ_mt_reinforcement_score add = WJ_mt_count_score add = WJ_mt_fraction_score add = WJ_mt_warlord_score add = WJ_mt_receiver_relation_score }
WJ_mt_acceptance = {
 value = WJ_mt_order_score add = WJ_mt_consequence_score
 min = 5 max = 100
 if = { limit = { exists = var:wj_mt_a var:wj_mt_a = this } value = 100 }
}
# 20 gold per full size step; paid to A even when A is the issuing player.
WJ_mt_cost = {
 value = 0
 if = { limit = { var:wj_mt_mode = 2 } every_in_list = { variable = wj_mt_pending add = { value = maa_size multiply = 20 } } }
}
