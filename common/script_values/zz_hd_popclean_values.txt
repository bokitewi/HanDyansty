# 开局人口清除优先级：数值越高越先清，0 = 不清。
# 3：霸权外的伯爵/男爵宫廷、部落/封建/游牧/牧民宫廷，以及霸权外的游荡者
# 2：霸权外其余宫廷
# 1：霸权内的伯爵/男爵宫廷与霸权内的游荡者
# 0：霸权内公爵及以上的宫廷
hd_popclean_tier_value = {
	value = 0
	if = {
		limit = { hd_popclean_in_hegemony_trigger = no }
		if = {
			limit = {
				OR = {
					hd_popclean_low_host_trigger = yes
					hd_popclean_rough_gov_host_trigger = yes
					hd_popclean_wanderer_trigger = yes
				}
			}
			value = 3
		}
		else = { value = 2 }
	}
	else_if = {
		limit = {
			OR = {
				hd_popclean_low_host_trigger = yes
				hd_popclean_wanderer_trigger = yes
			}
		}
		value = 1
	}
}

# 排序键：档位×1000 + 0~999 随机，同档内随机挑。
hd_popclean_key_value = {
	value = hd_popclean_tier_value
	multiply = 1000
	add = { integer_range = { min = 0 max = 999 } }
}
