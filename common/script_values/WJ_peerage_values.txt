# Same estate ingredients and 5..50 yearly range; merged royal rank uses the former king coefficient.
WJ_peer_annual_gold = {
 value = 0
 if = { limit = { var:wj_peer_rank = 1 }
  add = { value = var:wj_peer_food.title_province.combined_building_level multiply = 0.4 }
  add = { value = var:wj_peer_food.county.development_level multiply = 0.3 }
 }
 else = {
  var:wj_peer_food = { every_county_province = { add = { value = combined_building_level multiply = 0.4 } } }
  add = { value = var:wj_peer_food.development_level multiply = 0.3 }
  if = { limit = { var:wj_peer_rank = 3 } multiply = 1.25 }
  else_if = { limit = { var:wj_peer_rank >= 4 } multiply = 2 }
 }
 min = 5 max = 50 floor = yes
}
