# 厚生慎死·极端爆发 —— 关键数值
# 消费方：on_action zgrc_tianxia_xiangying_effect（on_war_started）
# 事件：zgrc_common.0003 - 0007
# ============================================================
# 外敌入侵：触发臣子表态的基础概率（%）
zgrc_extreme_foreign_chance = 25
# 动态概率：基础概率 × 攻防军力比（钳制 [基础, 100]）——军力越悬殊，臣子越可能响应
zgrc_extreme_foreign_chance_scaled = {
	value = zgrc_extreme_foreign_chance
	multiply = {
		value = scope:attacker.current_military_strength
		divide = {
			value = scope:defender.current_military_strength
			min = 1
		}
	}
	min = zgrc_extreme_foreign_chance
	max = 100
}

# 军力悬殊比例阈值（按场景分档；判断时内联：攻方战力 ≥ 防方战力 × 阈值）
zgrc_extreme_ratio_peasant  = 0.75   # 农民起义（民心补偿，阈值低）
