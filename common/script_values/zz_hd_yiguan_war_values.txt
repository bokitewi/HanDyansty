# 衣冠南渡·战争侧数值（docs/衣冠南渡_重做设计.md §6 #10–14；战争侧模块维护）

############################
# 举兵起义（§6 #12）
############################

# 威望花费。Scope：起义者。冲突期 100，其他阶段 200；
# 直属领主或最高宗主推行胡化期间（角色变量 hd_yiguan_huhua，胡人侧设，20 年）减半
hd_yiguan_qiyi_prestige_cost_value = {
	value = 200
	if = {
		limit = { hd_yiguan_phase_chongtu_trigger = yes }
		value = 100
	}
	if = {
		limit = {
			OR = {
				liege ?= { has_variable = hd_yiguan_huhua }
				top_liege ?= { has_variable = hd_yiguan_huhua }
			}
		}
		multiply = 0.5
	}
}

# 金币花费。Scope：起义者。冲突期不花金币，其他阶段 100；胡化期间减半
hd_yiguan_qiyi_gold_cost_value = {
	value = 100
	if = {
		limit = { hd_yiguan_phase_chongtu_trigger = yes }
		value = 0
	}
	if = {
		limit = {
			OR = {
				liege ?= { has_variable = hd_yiguan_huhua }
				top_liege ?= { has_variable = hd_yiguan_huhua }
			}
		}
		multiply = 0.5
	}
}

# 义军事件兵。Scope：起义者。500 + 每个直辖县 100（世族等无地县级头衔不算），上限 3000
hd_yiguan_qiyi_army_value = {
	value = 500
	every_held_title = {
		limit = {
			tier = tier_county
			is_landless_type_title = no
		}
		add = 100
	}
	max = 3000
}

# AI 起义的兵力上限。Scope：起义者。直属领主兵力超过（自身兵力 + 义军事件兵）× 4 时不起义
hd_yiguan_qiyi_ai_strength_cap_value = {
	value = max_military_strength
	add = hd_yiguan_qiyi_army_value
	multiply = 4
}

############################
# 招集流民（§6 #13）
############################

# 威望等级，最低按 1 级。Scope：招集者
hd_yiguan_liumin_level_value = {
	value = prestige_level
	min = 1
}

# 流民军人数 = 600 × 威望等级。Scope：招集者
hd_yiguan_liumin_army_value = {
	value = hd_yiguan_liumin_level_value
	multiply = 600
}

# 归附南朝（§6 #14）
############################

# 实领最高爵级（只算有地头衔；官职、世族等无地头衔会抬高 highest_held_title_tier，见 zz_hd_yiguan_triggers.txt 开头说明）
# Scope：角色。伯 2 / 公 3 / 王 4 / 帝 5 / 霸权 6，无地 0
hd_yiguan_war_landed_tier_value = {
	value = 0
	if = {
		limit = {
			any_held_title = {
				tier = tier_county
				is_landless_type_title = no
			}
		}
		value = 2
	}
	if = {
		limit = {
			any_held_title = {
				tier = tier_duchy
				is_landless_type_title = no
			}
		}
		value = 3
	}
	if = {
		limit = {
			any_held_title = {
				tier = tier_kingdom
				is_landless_type_title = no
			}
		}
		value = 4
	}
	if = {
		limit = {
			any_held_title = {
				tier = tier_empire
				is_landless_type_title = no
			}
		}
		value = 5
	}
	if = {
		limit = {
			any_held_title = {
				tier = tier_hegemony
				is_landless_type_title = no
			}
		}
		value = 6
	}
}

# 候选南方政权的排序分：先比实领爵级，再比兵力。Scope：候选者
hd_yiguan_guifu_score_value = {
	value = hd_yiguan_war_landed_tier_value
	multiply = 100000
	add = {
		value = max_military_strength
		max = 99999
	}
}
