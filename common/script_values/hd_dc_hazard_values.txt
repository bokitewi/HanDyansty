# Character scope. F changes initiation hazard, never scheme success or target
# selection weights. The era source multiplies F once, with a combined cap 3.
hd_dc_scheme_hazard_multiplier_value = {
	value = 1
	if = {
		limit = {
			exists = situation:dynastic_cycle
			any_character_situation = { this = situation:dynastic_cycle }
		}
		switch = {
			trigger = situation:dynastic_cycle.var:hd_dc_pressure_band
			2 = { value = 1.15 }
			3 = { value = 1.4 }
			4 = { value = 1.8 }
			5 = { value = 2.3 }
		}
		if = {
			limit = { situation:dynastic_cycle = { situation_current_phase = situation_dynastic_cycle_phase_chaos } }
			multiply = 1.5
		}
	}
	max = 3
}

# Inputs are saved numeric scopes hd_dc_hazard_base (percentage, actual native
# probability after its modifiers) and hd_dc_hazard_multiplier. This evaluates
# 100*(1-(1-p0/100)^m), using only verified script arithmetic and temporary
# numeric scopes. It consumes no random numbers and stores no character state.
# User approved absolute approximation error <= 0.01 percentage points.
# The fixed-point model remains an offline assumption, not a runtime claim.
hd_dc_hazard_probability_value = {
	value = scope:hd_dc_hazard_base
	min = 0
	max = 100
	if = {
		limit = {
			scope:hd_dc_hazard_base > 0
			scope:hd_dc_hazard_base < 100
			scope:hd_dc_hazard_multiplier > 1
		}
		if = {
			limit = {
				save_temporary_scope_value_as = { name = hd_dc_hz_q value = { value = 100 subtract = scope:hd_dc_hazard_base multiply = 100 } }
				save_temporary_scope_value_as = { name = hd_dc_hz_k value = 0 }
			}
		}
		# Binary range reduction. Under the audited .001 percentage input model,
		# q is a multiple of .1; rounded cutoffs cannot change the branch.
		if = {
			limit = { scope:hd_dc_hz_q < 0.152 }
			if = {
				limit = {
					save_temporary_scope_value_as = { name = hd_dc_hz_q value = { value = scope:hd_dc_hz_q multiply = 65536 } }
					save_temporary_scope_value_as = { name = hd_dc_hz_k value = 16 }
				}
			}
		}
		if = {
			limit = { scope:hd_dc_hz_q < 39.062 }
			if = {
				limit = {
					save_temporary_scope_value_as = { name = hd_dc_hz_q value = { value = scope:hd_dc_hz_q multiply = 256 } }
					save_temporary_scope_value_as = { name = hd_dc_hz_k value = { value = scope:hd_dc_hz_k add = 8 } }
				}
			}
		}
		if = {
			limit = { scope:hd_dc_hz_q < 625 }
			if = {
				limit = {
					save_temporary_scope_value_as = { name = hd_dc_hz_q value = { value = scope:hd_dc_hz_q multiply = 16 } }
					save_temporary_scope_value_as = { name = hd_dc_hz_k value = { value = scope:hd_dc_hz_k add = 4 } }
				}
			}
		}
		if = {
			limit = { scope:hd_dc_hz_q < 2500 }
			if = {
				limit = {
					save_temporary_scope_value_as = { name = hd_dc_hz_q value = { value = scope:hd_dc_hz_q multiply = 4 } }
					save_temporary_scope_value_as = { name = hd_dc_hz_k value = { value = scope:hd_dc_hz_k add = 2 } }
				}
			}
		}
		if = {
			limit = { scope:hd_dc_hz_q < 5000 }
			if = {
				limit = {
					save_temporary_scope_value_as = { name = hd_dc_hz_q value = { value = scope:hd_dc_hz_q multiply = 2 } }
					save_temporary_scope_value_as = { name = hd_dc_hz_k value = { value = scope:hd_dc_hz_k add = 1 } }
				}
			}
		}
		if = {
			limit = {
				save_temporary_scope_value_as = {
					name = hd_dc_hz_y
					value = { value = scope:hd_dc_hz_q subtract = 10000 multiply = 10000 divide = { value = scope:hd_dc_hz_q add = 10000 } }
				}
				save_temporary_scope_value_as = { name = hd_dc_hz_r value = { value = scope:hd_dc_hz_y multiply = scope:hd_dc_hz_y divide = 10000 } }
				save_temporary_scope_value_as = {
					name = hd_dc_hz_x
					value = {
						value = hd_dc_hazard_scaled_log_value
						subtract = { value = scope:hd_dc_hz_k multiply = 6931.472 }
						multiply = { value = scope:hd_dc_hazard_multiplier min = 1 max = 3 }
					}
				}
				save_temporary_scope_value_as = { name = hd_dc_hz_j value = { value = 0 subtract = scope:hd_dc_hz_x divide = 6931.472 floor = yes } }
				save_temporary_scope_value_as = { name = hd_dc_hz_z value = { value = scope:hd_dc_hz_j multiply = 6931.472 add = scope:hd_dc_hz_x } }
			}
		}
		if = {
			limit = { scope:hd_dc_hz_j >= 20 }
			# Remaining failure probability < 100/2^20 percentage points.
			value = 100
		}
		else = {
			value = 10000
			subtract = {
				value = hd_dc_hazard_scaled_exp_value
				divide = hd_dc_hazard_binary_divisor_value
			}
			# Round once to .001 of one percentage point.
			multiply = 10
			round = yes
			divide = 1000
			min = scope:hd_dc_hazard_base
			max = 100
		}
	}
}
