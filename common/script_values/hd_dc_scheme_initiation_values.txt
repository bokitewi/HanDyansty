# The installed MOD's evaluate_action_decreasing_house_unity with VALUE=100.
# Retain its current scope and every source factor; only blackmail calls this
# numeric mirror, so other unity consumers and their arithmetic are unchanged.
hd_dc_blackmail_unity_add_value = {
	value = 100
	if = { limit = { ai_wants_high_unity = yes } multiply = 0.8 }
	else_if = { limit = { ai_wants_low_unity = yes } multiply = 1.2 }
	if = {
		limit = {
			house ?= {
				OR = {
					has_house_unity_stage = harmonious
					has_house_unity_stage = friendly
					has_house_unity_stage = impassive
				}
			}
			AND = {
				OR = { has_trait = ambitious has_trait = deceitful has_trait = greedy }
				NOT = { is_heir_of = house.house_head }
			}
		}
		multiply = 1.2
	}
	if = { limit = { has_trait = fp3_struggle_detractor } multiply = 1.5 }
	else_if = { limit = { has_trait = fp3_struggle_supporter } multiply = 0.5 }
	if = { limit = { house ?= { has_house_unity_stage = harmonious } } multiply = 1.2 }
	else_if = { limit = { house ?= { has_house_unity_stage = friendly } } multiply = 1.1 }
	else_if = { limit = { house ?= { has_house_unity_stage = competitive } } multiply = 0.9 }
	else_if = {
		limit = { house ?= { has_house_unity_stage = antagonistic } }
		multiply = {
			value = 0.8
			if = { limit = { house ?= { house_unity_value <= critical_house_unity } } multiply = 0.5 }
		}
	}
}

hd_dc_blackmail_base_probability_value = {
	value = 100
	if = {
		limit = {
			scope:actor = {
				dm_compat_uses_house_unity_trigger = yes
				exists = scope:recipient.house
				house ?= scope:recipient.house
			}
		}
		add = hd_dc_blackmail_unity_add_value
	}
	if = { limit = { scope:actor.ai_honor >= 75 } multiply = 0 }
	if = { limit = { scope:actor.ai_honor >= 30 } multiply = 0.1 }
	if = {
		limit = {
			scope:actor.ai_honor >= -30
			scope:recipient = { has_relation_lover = scope:actor }
		}
		multiply = 0
	}
	if = {
		limit = {
			scope:recipient = {
				OR = {
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
		}
		multiply = 0
	}
}

hd_dc_abduct_initiation_probability_value = {
	if = {
		limit = {
			hd_dc_scheme_hazard_inputs_trigger = { BASE = hd_dc_abduct_base_probability_value ACTOR = scope:actor }
		}
	}
	value = hd_dc_hazard_probability_value
}

# Current scope is the selected target; ROOT remains the would-be murderer.
# Mirrors only murder_scheme_maintenance.1's final random chance, in its exact
# add/factor order. The separate target pool, eligibility and start_scheme are
# left in their original engine blocks. Do not clamp intermediate values.
hd_dc_murder_base_probability_value = {
	value = 25
	if = { limit = { has_relation_friend = root } add = -100 }
	if = { limit = { has_relation_lover = root } add = -25 }
	if = {
		limit = { OR = { has_relation_best_friend = root has_relation_soulmate = root } }
		add = -1000
	}
	if = {
		limit = { root.ai_compassion > 0 }
		add = { value = 0 subtract = root.ai_compassion divide = 3 }
	}
	if = {
		limit = { root.ai_honor > 0 }
		add = { value = 0 subtract = root.ai_honor divide = 2 }
	}
	if = {
		limit = { root.ai_honor < 0 }
		add = { value = root.ai_honor multiply = -1 divide = 2 }
	}
	if = {
		limit = {
			root.ai_vengefulness > 0
			OR = {
				has_relation_rival = root
				has_relation_nemesis = root
				exposed_cheating_on_spouse_trigger = { SPOUSE = root }
			}
		}
		add = { value = root.ai_vengefulness multiply = 1.5 }
	}
	if = {
		limit = {
			root.ai_rationality > 0
			save_temporary_scope_as = the_murder_target
			root = {
				is_landed = yes
				primary_title.tier >= tier_duchy
				any_targeting_faction = {
					faction_type = claimant_faction
					special_character = { this = scope:the_murder_target }
				}
			}
		}
		add = { value = root.ai_rationality multiply = 0.5 }
	}
	if = {
		limit = {
			root = { OR = { has_perk = a_job_done_right_perk has_focus = intrigue_skulduggery_focus } }
			save_temporary_scope_as = the_murder_target
			root = {
				is_landed = yes
				primary_title.tier >= tier_duchy
				any_targeting_faction = {
					faction_type = claimant_faction
					special_character = { this = scope:the_murder_target }
				}
			}
		}
		add = 50
	}
	if = {
		limit = {
			root.ai_greed > 0
			OR = {
				any_held_title = {
					tier >= root.highest_held_title_tier
					place_in_line_of_succession = { target = root value <= 3 }
				}
				any_heir_title = {
					holder != root
					tier >= root.highest_held_title_tier
					place_in_line_of_succession = { target = root value <= 3 }
				}
			}
		}
		add = { value = root.ai_greed divide = 2 }
	}
	if = {
		limit = { root = { has_dread_level_towards = { target = prev level >= 1 } } }
		multiply = 0.5
	}
	if = {
		limit = {
			NOT = { exists = dynasty }
			NOR = {
				has_relation_rival = root
				has_relation_nemesis = root
				exposed_cheating_on_spouse_trigger = { SPOUSE = root }
			}
		}
		add = -1000
	}
	if = {
		limit = {
			root = { any_character_struggle = { involvement = involved } }
			is_important_or_vip_struggle_character = yes
		}
		add = {
			value = 0
			if = {
				limit = {
					root = {
						any_character_struggle = { phase_has_catalyst = catalyst_gain_claim_on_title_fp2 }
						has_character_flag = agenda_towards_escalation
					}
				}
				add = 100
			}
			else_if = {
				limit = { root = { any_character_struggle = { phase_has_catalyst = catalyst_gain_claim_on_title_fp2 } } }
				add = -1000
			}
		}
	}
}
