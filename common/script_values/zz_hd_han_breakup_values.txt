# Character scope, evaluated before this round changes any vassal relation.
hd_hb_independence_score = {
	value = -20
	if = { limit = { has_trait = just } add = -5 }
	if = { limit = { has_trait = chaste } add = -5 }
	if = { limit = { has_trait = honest } add = -5 }
	if = { limit = { has_trait = content } add = -10 }
	if = { limit = { has_trait = disloyal } add = 10 }
	if = { limit = { has_trait = paranoid } add = 10 }
	if = { limit = { has_trait = deceitful } add = 5 }
	# Boolean existence: owning several developed counties still adds only five.
	if = {
		limit = { any_held_title = { tier = tier_county development_level > 10 } }
		add = 5
	}
	if = { limit = { current_military_strength > liege.current_military_strength } add = 10 }
	# 2026-10-14：信仰非国家信仰（最高领主信仰）+10，取代旧的 +3。
	if = { limit = { NOT = { faith = top_liege.faith } } add = 10 }
	if = { limit = { NOT = { faith = liege.faith } has_trait = zealous } add = 10 }
	if = { limit = { culture = { NOT = { has_same_culture_heritage = scope:hd_hb_round_liege.culture } } } add = 10 }
	if = { limit = { liege = { prestige_level >= 4 } } add = -10 }
	if = { limit = { liege = { prestige_level <= 1 } } add = 10 }
	if = { limit = { liege = { gold < 0 } } add = 5 }
	# 2026-10-14：领主行政法律（天朝/科举官僚制同名两套）。
	if = {
		limit = { liege = { OR = { has_realm_law = celestial_bureaucracy_0 has_realm_law = meritocratic_bureaucracy_0 } } }
		add = 50
	}
	else_if = {
		limit = { liege = { OR = { has_realm_law = celestial_bureaucracy_1 has_realm_law = meritocratic_bureaucracy_1 } } }
		add = 30
	}
	else_if = {
		limit = { liege = { OR = { has_realm_law = celestial_bureaucracy_3 has_realm_law = meritocratic_bureaucracy_3 } } }
		add = -15
	}
	# 最高领主正统性低于本人预期。
	if = {
		limit = {
			top_liege = {
				has_legitimacy = yes
				NOT = { meets_legitimacy_expectation_of = prev }
			}
		}
		add = 20
	}
	# 头衔等级：公国 +30，王国及以上 +50。
	if = { limit = { highest_held_title_tier = tier_duchy } add = 30 }
	else_if = { limit = { highest_held_title_tier >= tier_kingdom } add = 50 }
	# 恐惧上级领主。
	if = {
		limit = { has_dread_level_towards = { target = liege level = 1 } }
		add = -10
	}
	# Party state is relative to this vassal's CURRENT court, not any party
	# modifier and not a cached allegiance to an unrelated former court.
	if = { limit = { hd_hb_party_state_trigger = { KIND = attitude STATE = support } } add = -20 }
	if = { limit = { hd_hb_party_state_trigger = { KIND = stance STATE = zunwang } } add = -20 }
	if = { limit = { hd_hb_party_state_trigger = { KIND = attitude STATE = oppose } } add = 20 }
}

hd_hb_independence_chance = {
	value = 50
	add = hd_hb_independence_score
	min = 0
	max = 100
}
