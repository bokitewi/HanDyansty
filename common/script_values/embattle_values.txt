# Character scope; return zero if the combat/army reference is no longer valid.
# Preserve the original average / single -2 / absent -5 formulas and exemptions.
embattle_side_commander_advantage_value = {
    value = 0
    if = {
        limit = { hd_embattle_bonus_eligible_trigger = yes }
        value = martial
        commanding_army = { save_temporary_scope_as = hd_embattle_value_army }
        if = {
            limit = { scope:hd_embattle_value_army = { hd_embattle_left_valid_trigger = yes } }
            add = scope:hd_embattle_value_army.var:embattle_left_flank_commander.martial
        }
        if = {
            limit = { scope:hd_embattle_value_army = { hd_embattle_right_valid_trigger = yes } }
            add = scope:hd_embattle_value_army.var:embattle_right_flank_commander.martial
        }
        if = {
            limit = { scope:hd_embattle_value_army = { hd_embattle_both_valid_trigger = yes } }
            divide = 3
        }
        else_if = {
            limit = {
                scope:hd_embattle_value_army = {
                    OR = { hd_embattle_left_valid_trigger = yes hd_embattle_right_valid_trigger = yes }
                }
            }
            divide = 2
            subtract = 2
        }
        else_if = {
            limit = {
                NOR = {
                    is_leading_faction_type = populist_faction
                    is_leading_faction_type = peasant_faction
                    is_leading_faction_type = escalated_peasant_faction
                    max_number_of_knights <= 0
                }
            }
            subtract = 5
        }
        subtract = martial
    }
    desc = MODIFIER_DEFINITION_MAIN_DESC_SIDE_ADVANTAGE
}
