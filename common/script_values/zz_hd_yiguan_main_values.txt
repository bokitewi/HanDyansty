# 衣冠南渡·核心逻辑数值（模块 A2，见 docs/衣冠南渡_重做设计.md §1 第 8、9 轮，§6 第 28、29 条）
# 结局决议、200 年兜底用：北方八州（hd_yiguan_north_region）的县数与占比
# 写法先例：原版 00_dlc_fp3_script_values.txt:276-292（every_county_in_region 计数、按区域求百分比）、
#           00_decision_values.txt:147-155（every_realm_county 计数）
# 各方"合计"按县的独立宗主族属计：汉人 = 华夏且非汉化；汉化 = 王朝带 hd_yiguan_sinicized；胡人 = 非华夏且非汉化

############################
# 门槛
############################
hd_yiguan_end_guangfu_pct_value = 60			# 光复中原：本人势力占北方八州县数 ≥60%
hd_yiguan_end_heliu_pct_value = 60				# 胡汉合流：汉化政权合计 ≥60%
hd_yiguan_end_lunsang_pct_value = 70			# 衣冠沦丧：本人势力 ≥70%
hd_yiguan_end_lunsang_han_max_pct_value = 10	# 衣冠沦丧：汉人合计须 <10%
hd_yiguan_fallback_years_value = 200			# 满 200 年自动兜底

############################
# 计数
############################

# 北方八州县总数（至少 1，防除零）
hd_yiguan_north_total_value = {
	value = 0
	every_county_in_region = {
		region = hd_yiguan_north_region
		add = 1
	}
	min = 1
}

# Scope：角色。本人势力（含全部封臣）在北方八州的县数
hd_yiguan_realm_north_counties_value = {
	value = 0
	every_realm_county = {
		limit = { hd_yiguan_county_in_north_trigger = yes }
		add = 1
	}
}

# Scope：角色。本人势力占北方八州县数的百分比
hd_yiguan_realm_north_pct_value = {
	value = hd_yiguan_realm_north_counties_value
	multiply = 100
	divide = hd_yiguan_north_total_value
}

# 北方八州中独立宗主为汉人（华夏、非汉化）的县数
hd_yiguan_north_han_counties_value = {
	value = 0
	every_county_in_region = {
		region = hd_yiguan_north_region
		limit = {
			holder ?= {
				top_liege = { hd_yiguan_is_han_trigger = yes }
			}
		}
		add = 1
	}
}

# 北方八州中独立宗主为汉化王朝的县数
hd_yiguan_north_sin_counties_value = {
	value = 0
	every_county_in_region = {
		region = hd_yiguan_north_region
		limit = {
			holder ?= {
				top_liege = { hd_yiguan_is_sinicized_trigger = yes }
			}
		}
		add = 1
	}
}

# 北方八州中独立宗主为胡人（非华夏、非汉化）的县数
hd_yiguan_north_hu_counties_value = {
	value = 0
	every_county_in_region = {
		region = hd_yiguan_north_region
		limit = {
			holder ?= {
				top_liege = {
					hd_yiguan_is_hu_trigger = yes
					NOT = { hd_yiguan_is_sinicized_trigger = yes }
				}
			}
		}
		add = 1
	}
}

# 三方占比（百分比）
hd_yiguan_north_han_pct_value = {
	value = hd_yiguan_north_han_counties_value
	multiply = 100
	divide = hd_yiguan_north_total_value
}

hd_yiguan_north_sin_pct_value = {
	value = hd_yiguan_north_sin_counties_value
	multiply = 100
	divide = hd_yiguan_north_total_value
}

hd_yiguan_north_hu_pct_value = {
	value = hd_yiguan_north_hu_counties_value
	multiply = 100
	divide = hd_yiguan_north_total_value
}
