hd_peace_objective_price_value = {
	value = 0
	if = {
		limit = { exists = var:hd_peace_treaty_war }
		if = { limit = { var:hd_peace_treaty_war = { hd_peace_independence_war_trigger = yes } } add = 75 }
		else_if = { limit = { var:hd_peace_treaty_war = { hd_peace_subjugation_war_trigger = yes } } add = 100 }
		else_if = { limit = { var:hd_peace_treaty_war = { hd_peace_special_objective_war_trigger = yes } } add = 100 }
		else = {
			# Only target land still held by the opponent's realm changes hands;
			# land the proposer's side already occupies is priced at half.
			every_in_list = {
				variable = hd_peace_cb_reserved_counties
				limit = { hd_peace_county_in_opponent_realm_trigger = yes }
				add = hd_peace_occupied_county_price_value
			}
			min = 12
			# Titular claims must still pay for their political rank even when
			# the title has no de-jure counties to price.
			min = hd_peace_objective_title_floor_value
		}
		# The declared war goal costs a fifth of its full clause price.
		multiply = 0.2
	}
	round = yes
}

hd_peace_objective_title_floor_value = {
	value = 0
	if = {
		limit = { var:hd_peace_treaty_war = { OR = { using_cb = claim_cb using_cb = crusading_claim_cb } } }
		every_in_list = {
			variable = hd_peace_objective_display_titles
			if = { limit = { tier >= tier_empire } add = 100 }
			else_if = { limit = { tier = tier_kingdom } add = 75 }
			else_if = { limit = { tier = tier_duchy } add = 30 }
			else = { add = 12 }
		}
	}
}

# Display result, not an extra cost. Non-objective negotiated trades are peace.
hd_peace_result_preview_value = {
	value = 0
	if = { limit = { OR = { has_variable = hd_peace_objective hd_peace_actor_warscore_value >= 100 } } add = 1 }
}
