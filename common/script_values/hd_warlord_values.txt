# 郡国制·军阀契约（2026-10-24）
# 军阀每季度折入自家国库的数额 = 本应上缴的赋税 × 3 个月
# 本应上缴 ≈ 月收入 × 0.8（契约税率 0.75 + 郡国制政体 celestial_government_tax_contribution_add 0.05）
# 普通封臣的国库收入是上缴额 × 75%（政体 monthly_treasury_from_liege_mult = -0.25）；军阀钱留在自己手里，按全额折入
hd_warlord_quarterly_treasury_value = {
	value = monthly_character_income
	multiply = 0.8
	multiply = 3
	min = 0
}
# AI 军阀应召参战的机率（0~100；root = 军阀，scope:hd_wl_summoner = 召唤者，scope:hd_wl_beneficiary = 被援者）
# 用户拍板「看好感和战况」：基础 30，对召唤者好感 ×0.5，敌方疆域与自家接壤 +30；忠诚特质必应召
hd_warlord_summon_accept_chance = {
	value = 30
	add = {
		value = "opinion(scope:hd_wl_summoner)"
		multiply = 0.5
	}
	if = {
		limit = {
			any_neighboring_top_liege_realm_owner = { is_at_war_with = scope:hd_wl_beneficiary }
		}
		add = 30
	}
	min = 0
	max = 100
	if = {
		limit = { has_trait = loyal }
		value = 100
	}
}
hd_warlord_summon_refuse_chance = {
	value = 100
	subtract = hd_warlord_summon_accept_chance
}

# 「拥兵自重」的强力封臣支持度（root = 自请者）：支持者 +1、不支持者 -1；≥ 0 即至少半数支持
# 没有强力封臣时为 0，视为满足（用户拍板）
hd_warlord_powerful_vassal_balance = {
	value = 0
	every_vassal = {
		limit = { is_powerful_vassal = yes }
		if = {
			limit = { hd_warlord_supporter_trigger = { CANDIDATE = root } }
			add = 1
		}
		else = {
			add = -1
		}
	}
}

# 实际转入额：从金钱里扣出等额转进国库（用户拍板「金钱转国库」），金钱不足时有多少转多少
hd_warlord_quarterly_transfer_value = {
	value = hd_warlord_quarterly_treasury_value
	max = gold
	min = 0
}
