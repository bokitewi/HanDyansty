# Overrides vanilla create_title_counties_percentage. The engine sometimes
# evaluates this value with root set to the NO_CHARACTER sentinel (logged as
# "Scoped object of type 'character' is not valid"), which makes the
# has_legitimacy check error on every evaluation. exists = root is null-safe
# and filters that sentinel out; the no-discount fallback matches vanilla.
# Vanilla source: common/script_values/00_title_tiers_values.txt
create_title_counties_percentage = {
	scope:title = {
		if = {
			limit = {
				OR = {
					tier = tier_duchy
					tier = tier_kingdom
				}
			}
			add = {
				value = 0.5
				desc = CREATE_TITLE_PERCENTAGE_BASE_DUCHY_KINGDOM
			}
			root ?= {
				if = {
					limit = {
						exists = root
						has_legitimacy = yes
					}
					if = {
						limit = {
							has_legitimacy_flag = reduced_title_creation_cost
						}
						add = {
							value = -0.15 #35%
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_REDUCED
						}
					}
					else_if = {
						limit = {
							has_legitimacy_flag = very_reduced_title_creation_cost
						}
						add = {
							value = -0.20 #30%
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_VERY_REDUCED
						}
					}
					else_if = {
						limit = {
							has_legitimacy_flag = extra_reduced_title_creation_cost
						}
						add = {
							value = -0.25 #25%
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_EXTRA_REDUCED
						}
					}
					else = {
						#If you're level 2/1/0 or don't have Legitimacy you get no discounts
						add = {
							value = 0.0
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_NONE
						}
					}
				}
				else = {
					#If you're level 2/1/0 or don't have Legitimacy you get no discounts
					add = {
						# We add 0 with a 'desc', so we show up in the GUI and people understand legitimacy is a way to improve this
						value = 0.0
						desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_NONE
					}
				}
			}
		}
		else_if = { #Smaller discounts for Empires
			limit = {
				OR = {
					tier = tier_empire
					tier = tier_hegemony
				}
			}
			add = {
				value = 0.8
				desc = CREATE_TITLE_PERCENTAGE_BASE_EMPIRE
			}
			root ?= {
				if = {
					limit = {
						exists = root
						has_legitimacy = yes
					}
					if = {
						limit = {
							has_legitimacy_flag = reduced_title_creation_cost
						}
						add = {
							value = -0.05 #75%
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_REDUCED
						}
					}
					else_if = {
						limit = {
							has_legitimacy_flag = very_reduced_title_creation_cost
						}
						add = {
							value = -0.10 #70%
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_VERY_REDUCED
						}
					}
					else_if = {
						limit = {
							has_legitimacy_flag = extra_reduced_title_creation_cost
						}
						add = {
							value = -0.15 #65%
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_EXTRA_REDUCED
						}
					}
					else = {
						#If you're level 1/0 or don't have Legitimacy you get no discounts
						add = {
							value = 0.0
							desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_NONE
						}
					}
				}
				else = {
					#If you're level 1/0 or don't have Legitimacy you get no discounts
					add = {
						# We add 0 with a 'desc', so we show up in the GUI and people understand legitimacy is a way to improve this
						value = 0.0
						desc = CREATE_TITLE_PERCENTAGE_LEGITIMACY_DISCOUNT_NONE
					}
				}
			}
		}
	}
}
