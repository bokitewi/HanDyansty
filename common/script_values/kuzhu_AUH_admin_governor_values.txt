# root - charater that is evaluated for the governorship

kuzhu_governor_political_power = {
	value = {
		# default_ai_legitimacy_expectation 要求 scope:liege 指向当前领主
		save_temporary_scope_as = liege

		# 基础值
		add = {
			desc = BASE
			value = 0
		}

		# 性格特质
		if = {
			limit = { has_trait = loyal }
			add = { value = -50 desc = kuzhu_rebellion_loyal }
		}
		if = {
			limit = { has_trait = ambitious }
			add = { value = 15 desc = kuzhu_rebellion_ambitious }
		}
		if = {
			limit = { has_trait = disloyal }
			add = { value = 10 desc = kuzhu_rebellion_disloyal }
		}
		if = {
			limit = { has_trait = content }
			add = { value = -20 desc = kuzhu_rebellion_content }
		}
		if = {
			limit = { has_trait = craven }
			add = { value = -10 desc = kuzhu_rebellion_craven }
		}

		# 与直属领主的关系；各关系独立叠加
		if = {
			limit = {
				exists = liege
				liege = { has_relation_best_friend = root }
			}
			add = { value = -30 desc = kuzhu_rebellion_best_friend }
		}
		if = {
			limit = {
				exists = liege
				liege = { has_relation_friend = root }
			}
			add = { value = -15 desc = kuzhu_rebellion_friend }
		}
		if = {
			limit = {
				exists = liege
				liege = { has_relation_rival = root }
			}
			add = { value = 20 desc = kuzhu_rebellion_rival }
		}
		if = {
			limit = {
				exists = liege
				liege = { has_relation_blood_brother = root }
			}
			add = { value = 50 desc = kuzhu_rebellion_blood_brother }
		}

		# 忠顺不再豁免群雄割据判定，只作为数值因素
		if = {
			limit = {
				exists = obedience_target
				is_obedient = yes
			}
			add = { value = -60 desc = kuzhu_rebellion_obedient }
		}

		# 非法理封臣
		if = {
			limit = {
				exists = top_liege
				top_liege = {
					NOR = {
						government_has_flag = government_is_nomadic
						government_has_flag = government_is_steppe_admin
					}
				}
				primary_title = {
					NOT = {
						target_is_de_jure_liege_or_above = prev.top_liege.primary_title
					}
				}
			}
			add = { value = 30 desc = kuzhu_rebellion_illegal_vassal }
		}

		# 拥护霸主派
		if = {
			limit = {
				top_liege = title:h_china.holder
				top_participant_group:dynastic_cycle ?= {
					participant_group_type = pro_hegemon_movement
				}
			}
			add = { value = -10 desc = kuzhu_rebellion_pro_hegemon }
		}

		# 好感度独立叠加；直属领主就是最高领主时两项都会生效
		if = {
			limit = { exists = top_liege }
			add = {
				value = "opinion(top_liege)"
				multiply = -0.1
				desc = kuzhu_rebellion_top_liege_opinion
			}
		}
		if = {
			limit = { exists = liege }
			add = {
				value = "opinion(liege)"
				multiply = -0.2
				desc = kuzhu_rebellion_liege_opinion
			}
		}

		#封臣关系
		add = {
			desc = kuzhu_vassal_opinion
			every_vassal = {
				limit = { 
					is_landed = yes
					tier_difference = {
						target = root
						value = -1
					}
					opinion = {
						target = root
						value < 0
					}
				}
				add = -3
			}
			every_vassal = {
				limit = { 
					is_landed = yes
					tier_difference = {
						target = root
						value = -1
					}
					opinion = {
						target = root
						value >= 50
					}
				}
				add = 3
			}			
		}
		
		#影响力和威望取平均值，默认两边最大值都是5		
		add = {
			desc = kuzhu_prestige_level
			culture = {
				if = {
					limit = { has_cultural_era_or_later = culture_era_late_medieval }
					add = { value = root.prestige_level multiply = 8 }
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_high_medieval }
					add = { value = root.prestige_level multiply = 7 }
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_early_medieval }
					add = { value = root.prestige_level multiply = 6 }
				}
				else = {
					add = { value = root.prestige_level multiply = 5 }
				}
			}
		}
		add = {
			desc = kuzhu_influence_level
			culture = {
				if = {
					limit = { has_cultural_era_or_later = culture_era_late_medieval }
					add = { value = root.influence_level multiply = 8 }
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_high_medieval }
					add = { value = root.influence_level multiply = 7 }
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_early_medieval }
					add = { value = root.influence_level multiply = 6 }
				}
				else = {
					add = { value = root.influence_level multiply = 5 }
				}
			}
		}
		#私兵数量
		add = {
			desc = kuzhu_maa_num
			culture = {
				if = {
					limit = { has_cultural_era_or_later = culture_era_late_medieval }
					add = { value = root.maa_regiments_count multiply = 2 }
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_high_medieval }
					add = { value = root.maa_regiments_count multiply = 3 }
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_early_medieval }
					add = { value = root.maa_regiments_count multiply = 4 }
				}
				else = {
					add = { value = root.maa_regiments_count multiply = 5 }
				}
			}
		}
		
		# 直属领主正统性相对其直属封臣预期的平均值；空集合按0计算
		add = {
			value = {
				value = legitimacy_level
				subtract = {
					value = 0
					every_vassal = {
						limit = { is_landed = yes }
						add = default_ai_legitimacy_expectation
					}
					if = {
						limit = { any_vassal = { is_landed = yes } }
						divide = {
							value = 0
							every_vassal = {
								limit = { is_landed = yes }
								add = 1
							}
						}
						round = yes
					}
				}
				multiply = 5
			}
			desc = kuzhu_rebellion_legitimacy
		}

		# 直属领主君权等级：第一级为0，之后每级+5
		if = {
			limit = { liege = { has_realm_law = celestial_bureaucracy_3 } }
			add = { value = 15 desc = kuzhu_rebellion_liege_authority }
		}
		else_if = {
			limit = { liege = { has_realm_law = celestial_bureaucracy_2 } }
			add = { value = 10 desc = kuzhu_rebellion_liege_authority }
		}
		else_if = {
			limit = { liege = { has_realm_law = celestial_bureaucracy_1 } }
			add = { value = 5 desc = kuzhu_rebellion_liege_authority }
		}
		else_if = {
			limit = { liege = { has_realm_law = crown_authority_3 } }
			add = { value = 15 desc = kuzhu_rebellion_liege_authority }
		}
		else_if = {
			limit = { liege = { has_realm_law = crown_authority_2 } }
			add = { value = 10 desc = kuzhu_rebellion_liege_authority }
		}
		else_if = {
			limit = { liege = { has_realm_law = crown_authority_1 } }
			add = { value = 5 desc = kuzhu_rebellion_liege_authority }
		}

		max = 100
		min = 0
	}
}
