# Whole-title picker. ROOT is the player character and each list row supplies
# its landed title as scope:county, as in the county picker.

hd_peace_begin_demand_title_selection_gui = {
	scope = character
	is_valid = {
		hd_peace_treaty_is_editable_trigger = yes
		highest_held_title_tier >= tier_duchy
	}
	effect = {
		set_variable = { name = hd_peace_land_selection_mode value = 3 }
		# Realms and ranks may have changed since the tab was opened.
		hd_peace_populate_demand_title_candidates_effect = yes
	}
}

hd_peace_demand_title_available_gui = {
	scope = character
	is_shown = { highest_held_title_tier >= tier_duchy }
}

hd_peace_map_demand_title_toggle_gui = {
	scope = character
	saved_scopes = { county }
	is_shown = {
		hd_peace_treaty_is_editable_trigger = yes
		exists = scope:county
	}
	is_valid = {
		hd_peace_treaty_is_editable_trigger = yes
		exists = scope:county
		OR = {
			is_target_in_variable_list = { name = hd_peace_demand_titles target = scope:county }
			scope:county = {
				hd_peace_demand_title_candidate_trigger = yes
				NOT = { hd_peace_title_covered_by_demand_titles_trigger = yes }
			}
		}
	}
	effect = {
		if = {
			limit = { is_target_in_variable_list = { name = hd_peace_demand_titles target = scope:county } }
			remove_list_variable = { name = hd_peace_demand_titles target = scope:county }
		}
		else = {
			scope:county = { save_scope_as = hd_peace_new_demand_title }
			hd_peace_absorb_demands_into_title_effect = yes
			add_to_variable_list = { name = hd_peace_demand_titles target = scope:county }
		}
	}
}

hd_peace_map_demand_title_selected_gui = {
	scope = character
	saved_scopes = { county }
	is_shown = { is_target_in_variable_list = { name = hd_peace_demand_titles target = scope:county } }
}

# Shown on a title that a larger selected title already includes.
hd_peace_map_demand_title_covered_gui = {
	scope = character
	saved_scopes = { county }
	is_shown = {
		exists = scope:county
		scope:county = { hd_peace_title_covered_by_demand_titles_trigger = yes }
	}
}

hd_peace_demand_title_claim_filter_gui = { scope = character saved_scopes = { county } is_shown = { exists = scope:county root = { has_claim_on = scope:county } } }
hd_peace_demand_title_border_filter_gui = {
	scope = character
	saved_scopes = { county }
	is_shown = {
		exists = scope:county
		scope:county = {
			any_de_jure_county = {
				hd_peace_county_in_opponent_realm_trigger = yes
				any_neighboring_county = { exists = holder holder = { OR = { this = root is_vassal_or_below_of = root } } }
			}
		}
	}
}

hd_peace_demand_title_rank_discount_gui = {

hd_peace_demand_county_occupied_gui = {
	scope = character
	saved_scopes = { county }
	is_shown = { exists = scope:county scope:county = { hd_peace_county_occupied_by_proposer_side_trigger = yes } }

hd_peace_clear_demand_titles_gui = {
	scope = character
	is_valid = { hd_peace_treaty_is_editable_trigger = yes }
	effect = { if = { limit = { has_variable_list = hd_peace_demand_titles } clear_variable_list = hd_peace_demand_titles } }
}
