scorched_earth = {
	scope = army
	effect = {
		location.county = {
			hd_scorched_earth_development_loss_effect = yes
			change_county_control = -100
			add_county_modifier = {
				modifier = scorched_earth_county_modifier
				years = 5
			}
			every_county_province = {
				add_province_modifier = {
					modifier = scorched_earth_province_modifier
					years = 5
				}
			}
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_barony
					title_province = {
						has_holding = yes
					}
				}
				holder = {
					if = {
						limit = {
							NOT = { this = root.army_owner }
						}
						if = {
							limit = {
								has_opinion_modifier = {
									modifier = scorched_my_holding
									target = root.army_owner
								}
							}
							remove_opinion = {
								modifier = scorched_my_holding
								target = root.army_owner
							}
						}
						add_opinion = {
							modifier = scorched_my_holding
							target = root.army_owner
						}
					}
				}
			}
		}
		army_owner = {
			if = {
				limit = {
					root.location.county.holder = {
						OR = {
							this = root.army_owner
							AND = {
								NOT = {
									is_at_war_with = root.army_owner
								}
								is_vassal_or_below_of = root.army_owner
								NOT = {
									any_liege_or_above = {
										is_at_war_with = root.army_owner
										is_vassal_or_below_of = root.army_owner
									}
								}
							}
						}
					}
				}
				add_prestige = -100
			}
			if = {
				limit = {
					faith = {
						faith_hostility_level = {
							target = root.location.county.faith
							value = faith_fully_accepted_level
						}
						NOT = {
							has_doctrine = tenet_warmonger
						}
					}
				}
				add_piety = -50
			}
			if = {
				limit = {
					has_trait = sadistic
					root.location.county.holder = {
						OR = {
							is_at_war_with = root.army_owner
							any_liege_or_above = {
								is_at_war_with = root.army_owner
								NOT = {
									prev = {
										is_vassal_or_below_of = root.army_owner
									}
									target_is_liege_or_above = root.army_owner
								}
							}
						}
					}
				}
				stress_impact = {
					base = minor_stress_impact_loss
				}
			}
			if = {
				limit = {
					has_trait = vengeful
					root.location.county.holder = {
						OR = {
							is_at_war_with = root.army_owner
							any_liege_or_above = {
								is_at_war_with = root.army_owner
								NOT = {
									prev = {
										is_vassal_or_below_of = root.army_owner
									}
									target_is_liege_or_above = root.army_owner
								}
							}
						}
					}
				}
				stress_impact = {
					base = miniscule_stress_impact_loss
				}
			}
			if = {
				limit = {
					has_trait = wrathful
					root.location.county.holder = {
						OR = {
							is_at_war_with = root.army_owner
							any_liege_or_above = {
								is_at_war_with = root.army_owner
								NOT = {
									prev = {
										is_vassal_or_below_of = root.army_owner
									}
									target_is_liege_or_above = root.army_owner
								}
							}
						}
					}
				}
				stress_impact = {
					base = miniscule_stress_impact_loss
				}
			}
			if = {
				limit = {
					has_trait = zealous
					faith = {
						faith_hostility_level = {
							target = root.location.county.faith
							value >= faith_hostile_level
						}
					}
					root.location.county.holder = {
						OR = {
							is_at_war_with = root.army_owner
							any_liege_or_above = {
								is_at_war_with = root.army_owner
								NOT = {
									prev = {
										is_vassal_or_below_of = root.army_owner
									}
									target_is_liege_or_above = root.army_owner
								}
							}
						}
					}
				}
				stress_impact = {
					base = miniscule_stress_impact_loss
				}
			}
			if = {
				limit = {
					has_trait = paranoid
					root.location.county.holder = {
						OR = {
							this = root.army_owner
							AND = {
								any_liege_or_above = {
									this = root.army_owner
								}
								NOT = {
									any_liege_or_above = {
										is_at_war_with = root.army_owner
										is_vassal_or_below_of = root.army_owner
									}
								}
							}
						}
					}
				}
				stress_impact = {
					base = miniscule_stress_impact_loss
				}
			}
			if = {
				limit = {
					has_trait = forgiving
				}
				stress_impact = {
					base = miniscule_stress_impact_gain
				}
			}
			if = {
				limit = {
					has_trait = compassionate
				}
				stress_impact = {
					base = minor_stress_impact_gain
				}
			}
			if = {
				limit = {
					has_trait = just
					root.location.county.holder = {
						OR = {
							this = root.army_owner
							AND = {
								any_liege_or_above = {
									this = root.army_owner
								}
								NOT = {
									any_liege_or_above = {
										is_at_war_with = root.army_owner
										is_vassal_or_below_of = root.army_owner
									}
								}
							}
						}
					}
				}
				stress_impact = {
					base = miniscule_stress_impact_gain
				}
			}
		}
	}

	is_shown = {
		army_owner = { is_local_player = yes }
		exists = location.county
		OR = {
			has_game_rule = scorched_earth_enabled
			has_game_rule = scorched_earth_player_only
			has_game_rule = scorched_earth_independent_and_player_only
			AND = {
				army_owner = {
					is_independent_ruler = yes
				}
				OR = {
					has_game_rule = scorched_earth_independent_player_only
					has_game_rule = scorched_earth_independent_only
				}
			}
		}
		army_owner.faith = {
			NOR = {
				has_doctrine = tenet_pacifism
				has_doctrine = tenet_dharmic_pacifism
			}
		}
		army_owner.culture = {
			NOT = {
				has_cultural_tradition = tradition_pacifism
			}
		}
	}

	is_valid = {
		custom_description = {
			text = hd_scorched_earth_army_ready_tt
			is_army_in_combat = no
			is_army_in_raid = no
		}
		custom_description = {
			text = hd_scorched_earth_at_war_tt
			army_owner = { is_at_war = yes }
		}
		custom_description = {
			text = hd_scorched_earth_county_exists_tt
			exists = location.county
		}
		custom_description = {
			text = hd_scorched_earth_territory_tt
			OR = {
				AND = {
					location.county.county_controller = army_owner
					NOT = {
						location.county.holder = army_owner
					}
				}
				location.county.holder = {
					OR = {
						this = root.army_owner
						AND = {
							is_vassal_or_below_of = root.army_owner
							NOR = {
								is_at_war_with = root.army_owner
								any_liege_or_above = {
									is_at_war_with = root.army_owner
									is_vassal_or_below_of = root.army_owner
								}
							}
						}
					}
					root.location.county.county_controller = this
					root.army_owner = {
						OR = {
							any_character_war = {
								OR = {
									primary_attacker = prev
									primary_defender = prev
								}
							}
							any_liege_or_above = {
								is_at_war_with = prev
							}
						}
					}
				}
			}
		}
		custom_description = {
			text = hd_scorched_earth_not_scorched_tt
			NOT = {
				location.county = {
					has_county_modifier = scorched_earth_county_modifier
				}
			}
		}
	}
}
