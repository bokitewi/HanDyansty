hd_peace_objective_gui = {
	scope = character
	is_valid = {
		hd_peace_treaty_is_editable_trigger = yes
		OR = { has_variable = hd_peace_objective hd_peace_objective_available_trigger = yes }
	}
	effect = {
		if = { limit = { has_variable = hd_peace_objective } remove_variable = hd_peace_objective }
		else = { set_variable = { name = hd_peace_objective value = 1 } }
	}
}
hd_peace_scripted_context_gui = {
	scope = character
	saved_scopes = { war }
	is_shown = { exists = scope:war scope:war = { hd_peace_is_scripted_war_trigger = yes } }
}
hd_peace_scripted_settle_gui = {
	scope = character
	saved_scopes = { war }
	is_shown = {
		exists = scope:war
		scope:war = { hd_peace_is_scripted_war_trigger = yes OR = { primary_attacker = root primary_defender = root } }
	}
	is_valid = {
		exists = scope:war
		scope:war = {
			hd_peace_is_scripted_war_trigger = yes
			OR = { primary_attacker = root primary_defender = root }
			OR = { attacker_war_score >= 100 defender_war_score >= 100 }
			NOT = { has_variable = hd_peace_resolution_started }
		}
	}
	effect = {
		scope:war = {
			if = { limit = { attacker_war_score >= 100 } hd_peace_end_scripted_war_effect = { RESULT = attacker } }
			else_if = { limit = { defender_war_score >= 100 } hd_peace_end_scripted_war_effect = { RESULT = defender } }
		}
	}
}
