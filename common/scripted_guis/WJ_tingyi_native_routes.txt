# IsShown owns interception even when the operator cannot deliberate.
# Never fall back to a native appointment merely because IsValid is false.
WJ_ty_native_appoint = {
 scope = character
 saved_scopes = { wj_ty_native_owner wj_ty_native_task wj_ty_native_court_type wj_ty_native_candidate }
 is_shown = { WJ_ty_native_covered = yes }
 is_valid = { custom_tooltip = { text = WJ_ty_session_valid WJ_ty_native_appoint_valid = yes } }
 effect = { WJ_ty_native_appoint_effect = yes }
}
WJ_ty_native_dismiss = {
 scope = character
 saved_scopes = { wj_ty_native_owner wj_ty_native_task wj_ty_native_court_type }
 is_shown = { WJ_ty_native_covered = yes }
 is_valid = { custom_tooltip = { text = WJ_ty_session_valid WJ_ty_native_dismiss_valid = yes } }
 effect = { WJ_ty_native_dismiss_effect = yes }
}

WJ_ty_native_court_protected = {
 scope = character saved_scopes = { wj_ty_native_owner wj_ty_native_court_type }
 is_shown = { WJ_ty_native_court_slot = { OWNER = scope:wj_ty_native_owner TYPE = scope:wj_ty_native_court_type } }
}


WJ_ty_native_court_draft = {
 scope = character saved_scopes = { wj_ty_native_owner wj_ty_native_court_type candidate wj_ty_native_incumbent }
 is_valid = {
  WJ_ty_native_court_slot = { OWNER = scope:wj_ty_native_owner TYPE = scope:wj_ty_native_court_type }
  WJ_ty_can_open = yes exists = scope:candidate
  scope:candidate = { WJ_ty_native_court_candidate_valid = yes }
  trigger_if = { limit = { exists = scope:wj_ty_native_incumbent }
   scope:wj_ty_native_incumbent = { WJ_ty_court_position_employed = { TYPE = scope:wj_ty_native_court_type OWNER = scope:wj_ty_native_owner } }
  }
 }
 effect = { WJ_ty_native_court_draft_effect = yes }
}
