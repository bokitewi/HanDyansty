 scope = character
 is_valid = {
  OR = {
   AND = { var:wj_mufu_confirm_kind = 1 var:wj_mufu_confirm_target = this WJ_mufu_can_resign = yes var:wj_mufu_owner = var:wj_mufu_view }
   AND = { var:wj_mufu_confirm_kind = 2 var:wj_mufu_view = this WJ_mufu_owner_eligible = yes var:wj_mufu_confirm_target = { WJ_mufu_member_valid = yes var:wj_mufu_owner = root } }
  }
 }
 effect = {
  if = {
   limit = {
    OR = {
     AND = { var:wj_mufu_confirm_kind = 1 var:wj_mufu_confirm_target = this WJ_mufu_can_resign = yes var:wj_mufu_owner = var:wj_mufu_view }
     AND = { var:wj_mufu_confirm_kind = 2 var:wj_mufu_view = this WJ_mufu_owner_eligible = yes var:wj_mufu_confirm_target = { WJ_mufu_member_valid = yes var:wj_mufu_owner = root } }
    }
   }
   var:wj_mufu_confirm_target = { WJ_mufu_leave = yes }
  }
  remove_variable = wj_mufu_confirm_target
  remove_variable = wj_mufu_confirm_kind
 }
}
