WJ_ty_local_filter_reset = {
 scope = character
 effect = { if = { limit = { WJ_ty_local_picker_context = yes } remove_variable ?= wj_ty_local_filter_relations set_variable = { name = wj_ty_local_filter_tier value = 0 } set_variable = { name = wj_ty_local_filter_grade value = 0 } WJ_ty_local_filter = yes } }
}
WJ_ty_local_filter_relations = {
 scope = character
 effect = {
  if = {
   limit = { WJ_ty_local_picker_context = yes has_variable = wj_ty_picker }
   if = { limit = { has_variable = wj_ty_local_filter_relations } remove_variable = wj_ty_local_filter_relations }
   else = { set_variable = { name = wj_ty_local_filter_relations value = 1 } }
   WJ_ty_local_filter = yes
  }
 }
}
