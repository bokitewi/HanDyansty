WJ_mt_ui_central = {
 scope = character
 effect = {
  clear_variable_list = wj_mt_generals clear_variable_list = wj_mt_officials
  remove_variable = wj_mt_realm remove_variable = wj_mt_viewer
  if = { limit = { WJ_han_jin_law_trigger = yes top_liege = { WJ_inner_emperor_trigger = yes } }
   set_variable = { name = wj_mt_realm value = top_liege }
   set_variable = { name = wj_mt_viewer value = this }
   save_temporary_scope_as = wj_mt_viewer
   top_liege = {
    ordered_vassal = {
     limit = { WJ_mt_military_person = yes NOT = { has_title = title:e_minister_grand_marshal } } order_by = highest_held_title_tier
     max = { value = 0 every_vassal = { limit = { WJ_mt_military_person = yes NOT = { has_title = title:e_minister_grand_marshal } } add = 1 } } check_range_bounds = no
     if = { limit = { highest_held_title_tier >= tier_kingdom }
      scope:wj_mt_viewer = { add_to_variable_list = { name = wj_mt_generals target = prev } }
     }
     else = { scope:wj_mt_viewer = { add_to_variable_list = { name = wj_mt_officials target = prev } } }
    }
   }
  }
 }
}
WJ_mt_ui_subordinates = {
 scope = character
 effect = {
  if = { limit = { WJ_han_jin_law_trigger = yes scope:person = { is_alive = yes top_liege = root.top_liege } }
   set_variable = { name = wj_mt_subordinate_head value = scope:person }
   clear_variable_list = wj_mt_subordinates
   clear_variable_list = wj_mt_subordinates_kingdom
   clear_variable_list = wj_mt_subordinates_duchy
   clear_variable_list = wj_mt_subordinates_county
   save_temporary_scope_as = wj_mt_viewer
   scope:person = { ordered_vassal_or_below = {
    limit = { WJ_mt_military_person = yes } order_by = highest_held_title_tier
    max = { value = 0 every_vassal_or_below = { limit = { WJ_mt_military_person = yes } add = 1 } } check_range_bounds = no
    scope:wj_mt_viewer = { add_to_variable_list = { name = wj_mt_subordinates target = prev } }
    save_temporary_scope_as = wj_mt_subordinate
    if = { limit = { highest_held_title_tier >= tier_kingdom }
     scope:wj_mt_viewer = { add_to_variable_list = { name = wj_mt_subordinates_kingdom target = scope:wj_mt_subordinate } }
    }
    else_if = { limit = { highest_held_title_tier = tier_duchy }
     scope:wj_mt_viewer = { add_to_variable_list = { name = wj_mt_subordinates_duchy target = scope:wj_mt_subordinate } }
    }
    else_if = { limit = { highest_held_title_tier = tier_county }
     scope:wj_mt_viewer = { add_to_variable_list = { name = wj_mt_subordinates_county target = scope:wj_mt_subordinate } }
    }
   } }
  }
 }
}
