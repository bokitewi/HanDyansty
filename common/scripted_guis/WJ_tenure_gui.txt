# Only a manual UI choice marks these characters. The success callback consumes it.
WJ_tenure_mark_council = {
 scope = character
 is_shown = { scope:recipient = { WJ_chancellor_council_change_allowed = yes } }
 is_valid = {
  scope:recipient = { WJ_tenure_free = yes WJ_regency_manual_councillor_change_allowed = { OWNER = root } }
  trigger_if = {
   limit = { exists = scope:incumbent }
   scope:incumbent = { WJ_tenure_free = yes WJ_regency_manual_councillor_change_allowed = { OWNER = root } }
  }
 }
 effect = {
  if = {
   limit = { is_ai = no WJ_tenure_emperor = yes }
   scope:recipient = { set_variable = { name = wj_tenure_manual_emperor value = root days = 1 } }
   if = {
    limit = { exists = scope:incumbent }
    scope:incumbent = { set_variable = { name = wj_tenure_manual_emperor value = root days = 1 } }
   }
  }
 }
}
