WJ_ty_gui_sync = {
 scope = character
 is_shown = {
  trigger_if = { limit = { WJ_ty_session = yes } WJ_ty_roster_current = no }
  trigger_else = { always = no }
 }
 is_valid = {
  trigger_if = { limit = { WJ_ty_session = yes } WJ_ty_roster_current = no }
  trigger_else = { always = no }
 }
 effect = {
  if = { limit = { WJ_ty_session = yes }
   if = { limit = { WJ_ty_roster_current = no }
    WJ_ty_refresh_seats = yes
    # Only resolve the selected seat; leave the reviewed appointment snapshot intact.
    clear_saved_scope = wj_ty_sync_lobby
    if = { limit = { var:wj_ty_role = 0 exists = var:wj_ty_seat_0 }
     var:wj_ty_seat_0 = { save_scope_as = wj_ty_sync_lobby }
    }
    else_if = { limit = { var:wj_ty_role = 1 exists = var:wj_ty_seat_1 }
     var:wj_ty_seat_1 = { save_scope_as = wj_ty_sync_lobby }
    }
    else_if = { limit = { var:wj_ty_role = 2 exists = var:wj_ty_seat_2 }
     var:wj_ty_seat_2 = { save_scope_as = wj_ty_sync_lobby }
    }
    else_if = { limit = { var:wj_ty_role = 3 exists = var:wj_ty_seat_3 }
     var:wj_ty_seat_3 = { save_scope_as = wj_ty_sync_lobby }
    }
    else_if = { limit = { var:wj_ty_role = 4 exists = var:wj_ty_seat_4 }
     var:wj_ty_seat_4 = { save_scope_as = wj_ty_sync_lobby }
    }
    else_if = { limit = { var:wj_ty_role = 5 exists = var:wj_ty_seat_5 }
     var:wj_ty_seat_5 = { save_scope_as = wj_ty_sync_lobby }
    }
    else_if = { limit = { var:wj_ty_role = 6 exists = var:wj_ty_seat_6 }
     var:wj_ty_seat_6 = { save_scope_as = wj_ty_sync_lobby }
    }
    # Compare only existing pointers. Empty -> empty is unchanged.
    if = { limit = {
     trigger_if = { limit = { exists = var:wj_ty_lobby }
      trigger_if = { limit = { exists = scope:wj_ty_sync_lobby }
       NOT = { var:wj_ty_lobby = scope:wj_ty_sync_lobby }
      }
      trigger_else = { always = yes }
     }
     trigger_else = { exists = scope:wj_ty_sync_lobby }
    }
     remove_variable = wj_ty_choice_gold
     remove_variable = wj_ty_choice_prestige
     remove_variable = wj_ty_choice_influence
     remove_variable = wj_ty_choice_weak
     remove_variable = wj_ty_choice_strong
     remove_variable = wj_ty_choice_hook
    }
    if = { limit = { exists = scope:wj_ty_sync_lobby }
     set_variable = { name = wj_ty_lobby value = scope:wj_ty_sync_lobby }
    }
    else = { remove_variable = wj_ty_lobby }
    clear_saved_scope = wj_ty_sync_lobby
   }
  }
 }
}

      is_councillor_of = scope:wj_ty_operator.var:wj_ty_emperor

    WJ_ty_removal_snapshot = yes
   }
   remove_variable = wj_ty_picker clear_variable_list = wj_ty_candidates WJ_ty_picker_clear = yes
  }
 }
