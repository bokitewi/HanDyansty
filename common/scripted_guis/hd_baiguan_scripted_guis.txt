# Null-tolerant GUI entry points. Do not declare a mandatory character root:
# CouncilWindow can supply an empty GuiScope before a character is available.
# Character-only triggers/effects are reached exclusively through root ?=.
# Keep the two public IDs unchanged so existing GUI callers remain compatible.
hd_baiguan_open_effect = {
    is_shown = {
        exists = root
        root ?= { hd_audit_baiguan_available_trigger = yes }
    }
    is_valid = {
        exists = root
        root ?= { hd_audit_baiguan_available_trigger = yes }
    }
    effect = {
        root ?= {
            if = {
                limit = { hd_audit_baiguan_available_trigger = yes }
                hd_baiguan_rebuild_office_holders_effect = yes
            }
        }
    }
}

hd_baiguan_rebuild_effect = {

    }

                hd_baiguan_rebuild_office_holders_effect = yes
            }
        }
    }
}
