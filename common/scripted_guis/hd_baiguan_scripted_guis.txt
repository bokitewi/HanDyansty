hd_baiguan_open_effect = {
	scope = character

	is_shown = {
		is_ruler = yes
		OR = {
			highest_held_title_tier = tier_hegemony
			AND = {
				highest_held_title_tier = tier_empire
				is_independent_ruler = yes
			}
		}
	}

	effect = {
		hd_baiguan_rebuild_office_holders_effect = yes
	}
}

hd_baiguan_rebuild_effect = {
	scope = character

	is_valid = {
		is_ruler = yes
		OR = {
			highest_held_title_tier = tier_hegemony
			AND = {
				highest_held_title_tier = tier_empire
				is_independent_ruler = yes
			}
		}
	}

	effect = {
		hd_baiguan_rebuild_office_holders_effect = yes
	}
}
