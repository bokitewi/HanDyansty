# Override the master's existing entry names; preserve its imperial eligibility.
# is_valid also guards execution after death, loss of the qualifying title, or switching away from Han/Jin.
hd_baiguan_open_effect = {
	scope = character
	is_shown = {
		exists = this
		is_alive = yes
		is_ruler = yes
		WJ_han_jin_law_trigger = yes
		OR = {
			highest_held_title_tier = tier_hegemony
			AND = {
				highest_held_title_tier = tier_empire
				is_independent_ruler = yes
			}
		}
	}
	is_valid = {
		is_alive = yes
		is_ruler = yes
		WJ_han_jin_law_trigger = yes
		OR = {
			highest_held_title_tier = tier_hegemony
			AND = {
				highest_held_title_tier = tier_empire
				is_independent_ruler = yes
			}
		}
	}
	effect = {
		WJ_baiguan_refresh_display_effect = yes
	}
}

WJ_baiguan_display_current = {
	scope = character
	is_shown = {
		trigger_if = {
			limit = { exists = primary_spouse primary_spouse ?= { is_alive = yes top_liege = root has_trait = SanGong_GongWei_1 } }
			exists = var:wj_baiguan_empress
			var:wj_baiguan_empress = primary_spouse
		}
		trigger_else = { NOT = { has_variable = wj_baiguan_empress } }
		trigger_if = {
			limit = { exists = var:WJ_royal_heir var:WJ_royal_heir = { WJ_royal_crown_prince = yes } }
			exists = var:wj_baiguan_crown_prince
			var:wj_baiguan_crown_prince ?= var:WJ_royal_heir
			var:wj_baiguan_crown_prince = { WJ_royal_crown_prince = yes }
		}
		trigger_else = { NOT = { has_variable = wj_baiguan_crown_prince } }
		trigger_if = {
			limit = { exists = capital_county.kingdom.holder capital_county.kingdom.holder ?= { is_alive = yes top_liege = root } }
			exists = var:wj_baiguan_capital_governor
			var:wj_baiguan_capital_governor = capital_county.kingdom.holder
		}
		trigger_else = { NOT = { has_variable = wj_baiguan_capital_governor } }
	}
}

			}

		WJ_baiguan_refresh_display_effect = yes
	}
}

# Snapshot validity is checked in the viewed owner's scope, never in a possibly empty portrait scope.

			var:wj_baiguan_crown_prince = { top_liege = root WJ_royal_crown_prince = yes }
		}
		trigger_else = { always = no }
	}
}
