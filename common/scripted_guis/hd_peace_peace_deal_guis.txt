# All actions use the player character as ROOT. The active war is supplied only
# to the opening and sending actions; editing actions use the persisted context.

hd_peace_open_treaty_gui = {
	scope = character
	saved_scopes = { war }
	is_valid = {
		exists = scope:war
		custom_tooltip = {
			text = hd_peace_open_no_pending_tt
			NOT = { has_variable = hd_peace_treaty_locked }
		}
		trigger_if = {
			limit = { exists = scope:war }
			scope:war = {
				custom_tooltip = {
					text = hd_peace_open_leader_tt
					OR = { primary_attacker = root primary_defender = root }
				}
				custom_tooltip = {
					text = hd_peace_open_ordinary_tt
					hd_peace_is_ordinary_war_trigger = yes
				}
			}
		}
		trigger_else = { always = no }
	}
	effect = { hd_peace_open_treaty_effect = yes }
}

hd_peace_map_offer_county_toggle_gui = {
	scope = character
	saved_scopes = { county }
	is_shown = {
		hd_peace_treaty_is_editable_trigger = yes
		exists = scope:county
		scope:county = {
			tier = tier_county
			is_landless_type_title = no
			is_noble_family_title = no
			exists = holder
			holder = { OR = { this = root is_vassal_or_below_of = root } }
			NOT = { this = root.capital_county }
		}
	}
	is_valid = {
		hd_peace_treaty_is_editable_trigger = yes
		exists = scope:county
		OR = {
			is_target_in_variable_list = { name = hd_peace_offer_counties target = scope:county }
			AND = {
    scope:county = {
     tier = tier_county is_landless_type_title = no is_noble_family_title = no exists = holder
     holder = { OR = { this = root is_vassal_or_below_of = root } }
     NOT = { this = root.capital_county }
    }
				# No count cap; a ruler cannot sign away every county.
				hd_peace_offer_county_count_value < realm_size
			}
		}
	}
	effect = {
		if = {
			limit = { is_target_in_variable_list = { name = hd_peace_offer_counties target = scope:county } }
			remove_list_variable = { name = hd_peace_offer_counties target = scope:county }
		}
		else = { add_to_variable_list = { name = hd_peace_offer_counties target = scope:county } }
	}
}

	is_shown = {
		hd_peace_treaty_is_editable_trigger = yes
		has_variable = hd_peace_treaty_opponent
		exists = scope:county
		scope:county = {
			tier = tier_county
			is_landless_type_title = no
			is_noble_family_title = no
			exists = holder
		}
		trigger_if = {
			limit = { has_variable = hd_peace_treaty_opponent }
			scope:county = { holder = { OR = { this = root.var:hd_peace_treaty_opponent is_vassal_or_below_of = root.var:hd_peace_treaty_opponent } } }
		}
		trigger_else = { always = no }
	}
	is_valid = {
		hd_peace_treaty_is_editable_trigger = yes
		has_variable = hd_peace_treaty_opponent
		exists = scope:county
		OR = {
			is_target_in_variable_list = { name = hd_peace_demand_counties target = scope:county }
			AND = {
    scope:county = {
     tier = tier_county is_landless_type_title = no is_noble_family_title = no exists = holder
     holder = { OR = { this = root.var:hd_peace_treaty_opponent is_vassal_or_below_of = root.var:hd_peace_treaty_opponent } }
     # Already taken by a selected whole title.
     NOT = { hd_peace_county_covered_by_demand_titles_trigger = yes }
    }
				# No count cap; never more counties than the opponent's realm holds.
				trigger_if = {
					limit = { has_variable = hd_peace_treaty_opponent }
					var:hd_peace_treaty_opponent = { root.hd_peace_demand_county_count_value < realm_size }
				}
				trigger_else = { always = no }
			}
		}
	}
	effect = {
		if = {
			limit = { is_target_in_variable_list = { name = hd_peace_demand_counties target = scope:county } }
			remove_list_variable = { name = hd_peace_demand_counties target = scope:county }
