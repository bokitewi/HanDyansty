WJ_capital_open_kingdom = {
 scope = character
 saved_scopes = { hd_gui_player }
 is_shown = {
  exists = scope:hd_gui_player
  this = scope:hd_gui_player
  WJ_capital_grant_actor_valid = yes
  exists = capital_province.kingdom
  capital_province.kingdom = { OR = { NOT = { exists = holder } holder = root } }
 }
 is_valid = {
  exists = scope:hd_gui_player
  this = scope:hd_gui_player
  is_ai = no
  WJ_capital_grant_actor_valid = yes
  exists = capital_province.kingdom
  NOT = { has_variable = wj_capital_pending_title }
  capital_province.kingdom = { OR = { NOT = { exists = holder } holder = root } }
  trigger_if = {
   limit = { capital_province.kingdom = { NOT = { exists = holder } } }
   custom_tooltip = { text = WJ_capital_create_cost_tt gold >= 50 prestige >= 50 }
  }
 }
 effect = {
  capital_province.kingdom = { save_scope_as = WJ_capital_grant_selected_title }
  trigger_event = WJ_capital_grant.1
 }
}

WJ_capital_select = {
 scope = character
 saved_scopes = { recipient }
 is_shown = { exists = var:wj_capital_selected var:wj_capital_selected = scope:recipient }
 is_valid = {
  is_ai = no
  exists = scope:recipient
  exists = var:wj_capital_pending_title
  save_temporary_scope_as = actor
  var:wj_capital_pending_title = { save_temporary_scope_as = WJ_capital_grant_selected_title }
  is_target_in_variable_list = { name = wj_capital_candidates target = scope:recipient }
  WJ_capital_grant_selected_valid = yes
 }
 effect = { set_variable = { name = wj_capital_selected value = scope:recipient } }
}
