WJ_dasima_open_picker = {
 scope = character
 is_shown = { WJ_inner_emperor_trigger = yes is_ai = no }
 is_valid = { WJ_inner_emperor_trigger = yes is_ai = no }
 effect = {
  set_variable = { name = wj_dasima_sort value = 10 }
  remove_variable ?= wj_dasima_sort_ascending
  # Stable native-dropdown indices; labels/actions share this ordering.
  clear_variable_list = wj_dasima_sort_options
  add_to_variable_list = { name = wj_dasima_sort_options target = 0 }
  add_to_variable_list = { name = wj_dasima_sort_options target = 6 }
  add_to_variable_list = { name = wj_dasima_sort_options target = 7 }
  add_to_variable_list = { name = wj_dasima_sort_options target = 8 }
  add_to_variable_list = { name = wj_dasima_sort_options target = 9 }
  add_to_variable_list = { name = wj_dasima_sort_options target = 10 }
  clear_variable_list = wj_dasima_filter_options
  add_to_variable_list = { name = wj_dasima_filter_options target = 0 }
  add_to_variable_list = { name = wj_dasima_filter_options target = 1 }
  add_to_variable_list = { name = wj_dasima_filter_options target = 2 }
  set_variable = { name = wj_dasima_filter_faith value = 0 }
  set_variable = { name = wj_dasima_filter_culture value = 0 }
  WJ_dasima_sort_candidates_effect = yes
  set_variable = { name = wj_dasima_picker_open value = 1 }
 }
}
WJ_dasima_close_picker = {
 scope = character
 effect = {
  remove_variable ?= wj_dasima_picker_open
  clear_variable_list = wj_dasima_candidates
 }
}
WJ_dasima_select_candidate = {
 scope = character
 saved_scopes = { candidate wj_inner_emperor }
 is_shown = { WJ_inner_emperor_trigger = yes is_ai = no }
 is_valid = {
  WJ_inner_emperor_trigger = yes
  is_ai = no
  has_variable = wj_dasima_picker_open
  exists = scope:candidate
  trigger_if = {
   limit = { exists = var:wj_dasima_holder }
   NOT = { var:wj_dasima_holder = scope:candidate }
  }
  exists = scope:wj_inner_emperor
  scope:wj_inner_emperor = this
  scope:candidate = { WJ_dasima_candidate_trigger = yes }
 }
 effect = {
  WJ_dasima_appoint_effect = yes
  remove_variable ?= wj_dasima_picker_open
  clear_variable_list = wj_dasima_candidates
 }
}
