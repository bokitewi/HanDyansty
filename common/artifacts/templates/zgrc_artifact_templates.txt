# 普通宝剑（无合璧）
# 勇武过人、或身怀武勇秉性，方能发挥宝剑之利；否则只余虚名（仅威望）
zgrc_artifact_ordinary_template = {
	can_equip = { always = yes }
	can_benefit = {
		OR = {
			prowess >= 12
			has_trait = brave
			has_trait = strong
			has_trait = gallant
			has_trait = aggressive_attacker
			has_trait = berserker
			has_trait = varangian
		}
	}
	fallback = {
		monthly_prestige = 0.1
	}
	can_reforge = { always = no }
}

# 干将（雄）—— 拥有莫邪时享受完整 modifier
zgrc_artifact_ganjiang_template = {
	can_equip = { always = yes }
	can_benefit = {
		any_character_artifact = { has_variable = zgrc_artifact_moye }
	}
	fallback = {
		prowess_no_portrait = 11
	}
	can_reforge = { always = no }
}

# 莫邪（雌）—— 拥有干将时享受完整 modifier
zgrc_artifact_moye_template = {
	can_equip = { always = yes }
	can_benefit = {
		any_character_artifact = { has_variable = zgrc_artifact_ganjiang }
	}
	fallback = {
		prowess_no_portrait = 10
	}
	can_reforge = { always = no }
}
