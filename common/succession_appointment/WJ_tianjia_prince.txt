SanGong_Jichengren = {
    default_candidates = { holder_house_member }
    allow_children = yes

    candidate_score = {
        value = {
            # 加分项
            if = {                                         # 太孙：父亲是太子
                limit = {
                    has_trait = SanGong_Taizi
                    AND = { exists = father
                        father = { has_trait = SanGong_Taizi }
                    }
                }
                add = { value = 4000 desc = SanGong_JiCheng_TaiSun }
            }

            if = {                                         # 太子
                limit = {
                    is_child_of = scope:title.holder
                    has_trait = SanGong_Taizi
                }
                add = { value = 3000 desc = SanGong_JiCheng_TaiZi }
            }

            if = {                                         # 皇长子
                limit = {
                    is_child_of = scope:title.holder
                    any_sibling = { count = 0 age > prev.age }
                }
                add = { value = 1000 desc = SanGong_JiCheng_DiZhangZi }
            }

            if = {#指定继承人
                limit = { 
                    is_child_of = scope:title.holder
                    OR = {
                        always = no # HanFan modifier not imported 
                        scope:title.holder.designated_heir ?= this
                    }
                }
                add = { value = 5000 desc = SanGong_JiCheng_zhiding }
            }

            if = {                                         # 皇子保底（排除太子）
                limit = {
                    is_child_of = scope:title.holder
                    NOT = { has_trait = SanGong_Taizi }
                }
                add = { value = 500 desc = SanGong_JiCheng_HuangZi }
            }

            if = {                                         # 皇孙
                limit = {
                    OR = {
                        AND = { exists = father
                            father = { is_child_of = scope:title.holder }
                        }
                        AND = { exists = mother
                            mother = { is_child_of = scope:title.holder }
                        }
                    }
                }
                add = { value = 100 desc = SanGong_JiCheng_HuangSun }
            }

            if = {                                         # 兄终弟及（皇帝无子时）
                limit = {
                    scope:title.holder = {
                        any_child = { is_alive = yes is_male = yes count = 0 }
                    }
                    is_close_family_of = scope:title.holder
                }
                add = { value = 1000 desc = SanGong_JiCheng_XiongZhongDiJi }
            }

            if = {                                         # 宗室近亲（皇帝无子时）
                limit = {
                    scope:title.holder = {
                        any_child = { is_alive = yes is_male = yes count = 0 }
                    }
                    NOT = { is_child_of = scope:title.holder }
                }
                add = { value = 500 desc = SanGong_JiCheng_ZongShi_JinQin }
            }

            if = {                                         # 母位份1级（皇后，在世）
                limit = {
                    is_child_of = scope:title.holder
                    exists = mother
                    mother = { is_alive = yes has_trait = SanGong_GongWei_1 }
                }
                add = { value = 1000 desc = SanGong_JiCheng_mother_WeiFen }
            }

            

            

            

            

            if = {
                limit = {
                    is_female = yes
                    scope:title.holder = {
                        OR = { has_realm_law = male_only_law has_realm_law = male_preference_law }
                    }
                }
                add = { value = -500 desc = DM_appointment_male_preference_penalty }
            }
            if = {
                limit = {
                    is_male = yes
                    scope:title.holder = {
                        OR = { has_realm_law = female_only_law has_realm_law = female_preference_law }
                    }
                }
                add = { value = -500 desc = DM_appointment_female_preference_penalty }
            }

            if = {                                         # 绝嗣能力分（皇帝无子且年满16）
                limit = {
                    scope:title.holder = {
                        any_child = { is_alive = yes is_male = yes count = 0 }
                    }
                    age >= 16
                }
                add = { value = learning multiply = 10 desc = SanGong_JiCheng_JueSi_XueShi }
                add = { value = diplomacy multiply = 10 desc = SanGong_JiCheng_JueSi_WaiJiao }
                add = { value = stewardship multiply = 10 desc = SanGong_JiCheng_JueSi_GuanLi }
                add = { value = intrigue multiply = 10 desc = SanGong_JiCheng_JueSi_MouLve }
                add = { value = prestige_level multiply = 50 desc = SanGong_JiCheng_Lixian_weiwang }
            }
            # 扣分项

            if = {                                         # 嗣子：皇帝有其他儿子时减分
                limit = {
                    is_child_of = scope:title.holder
                    has_trait = SanGong_Sizi
                    scope:title.holder = {
                        any_child = { is_alive = yes is_male = yes count > 1 }
                    }
                }
                add = { value = -2500 desc = SanGong_JiCheng_SIZi }
            }

            if = {                                         # 形貌鄙陋（仅直系）
                limit = { bad_WaiMao_trigger = yes is_child_of = scope:title.holder }
                add = { value = -500 desc = SanGong_JiCheng_FuMian_XingMao }
            }

            if = {                                         # 资质庸懦（仅直系）
                limit = { bad_ShiShang_trigger = yes is_child_of = scope:title.holder }
                add = { value = -300 desc = SanGong_JiCheng_FuMian_ZiZhi }
            }

            if = {                                         # 肢体有缺（仅直系）
                limit = { bad_ShenTi_trigger = yes is_child_of = scope:title.holder }
                add = { value = -300 desc = SanGong_JiCheng_FuMian_ShenTi }
            }

            if = {                                         # 皇帝有子则旁支出局
                limit = {
                    scope:title.holder = {
                        any_child = { is_alive = yes is_male = yes count > 0 }
                    }
                    NOT = {
                        OR = {
                            is_child_of = scope:title.holder
                            AND = { exists = father
                                father = { is_child_of = scope:title.holder }
                            }
                            AND = { exists = mother
                                mother = { is_child_of = scope:title.holder }
                            }
                        }
                    }
                }
                add = { value = -9999 desc = SanGong_JiCheng_ZongFa }
            }

            if = {                                         # 宦官不得继位
                limit = {
                    OR = { has_trait = eunuch }
                }
                subtract = { value = 1000 desc = SanGong_JiCheng_PaiChu_HuanGuan }
            }
        }
    }
}
