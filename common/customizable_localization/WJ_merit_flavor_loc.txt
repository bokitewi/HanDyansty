# 禄秩说明按“实际官职—正式入仕资格—贤望任官上限”三层关系分别解释。
WJ_LuzhiCurrentDesc = {
	type = character

	text = {
		trigger = { WJ_luzhi_status_shou_trigger = yes }
		localization_key = WJ_LUZHI_DESC_SHOU
	}
	text = {
		trigger = {
			WJ_luzhi_status_shi_trigger = yes
			WJ_has_entry_road_trigger = yes
		}
		localization_key = WJ_LUZHI_DESC_SHI_WITH_ROAD
	}
	text = {
		trigger = {
			WJ_luzhi_status_shi_trigger = yes
			WJ_has_entry_road_trigger = no
		}
		localization_key = WJ_LUZHI_DESC_SHI_NO_ROAD
	}
	text = {
		trigger = {
			WJ_luzhi_status_chu_trigger = yes
			WJ_has_entry_road_trigger = yes
		}
		localization_key = WJ_LUZHI_DESC_CHU_WITH_ROAD
	}
	text = {
		trigger = {
			WJ_luzhi_status_chu_trigger = yes
			WJ_has_entry_road_trigger = no
		}
		localization_key = WJ_LUZHI_DESC_CHU_NO_ROAD
	}
	text = {
		fallback = yes
		localization_key = WJ_LUZHI_DESC_OFFICIAL
	}
}
