# ============================================================================
# WJ 称谓显示顺序机制 —— 自定义本地化链(重写版)
# xinsanguo-MOD 2026-08-17
#
# 机制原理(与原版/参考mod完全一致的接线方式):
#   原版链路: GUI文本 "CHARACTER_VIEW_NAME" → 本地化键值 [Character.GetUINameNotMeNoTooltip]
#            (GetUINameNotMeNoTooltip 是引擎内置数据函数, 无法被同名custom loc覆盖)
#   本mod链路: GUI文件不动 → 由 localization\simp_chinese\gui\WJ_name_view_l_simp_chinese.yml
#            覆盖 CHARACTER_VIEW_NAME 等键值 → [Character.Custom('WJ_GetUIName')]
#            → 本文件的 WJ_GetUIName 分支链 → 各分支 localization_key → 模板yml
#
# 参考依据:
#   - 游戏本体: localization\simp_chinese\gui\character_window_l_simp_chinese.yml
#   - 三公mod(3684849062): SanGong_view_name 链 + replace yml 覆盖同名GUI键(同法)
#   - jiaohu-mod: OE_GetUINameNotMeNoTooltip 链 + 00_oe_gui yml(同法, 无同名覆盖)
#   结论: "自定义链 + GUI本地化键覆盖"是原版验证过的唯一生效路径。
# ============================================================================

# ============ 皇室称谓链(移植自 GetRulerRelationFrom/Name, 适配主mod) ============
WJ_GetRulerRelationFrom = {
	type = character
	text = { # 太后/皇后: 有王级以上子女且有配偶
		trigger = {
			WJ_han_name_style_trigger = yes
			is_female = yes
			any_child = {
				primary_title.tier >= tier_kingdom
			}
			exists = primary_spouse
		}
		localization_key = WJ_relation_from_spouse
	}
	text = { # 有王级以上子女(无配偶): 用领主头衔
		trigger = {
			WJ_han_name_style_trigger = yes
			is_female = yes
			any_child = {
				primary_title.tier >= tier_kingdom
			}
			NOT = { exists = primary_spouse }
		}
		localization_key = WJ_relation_from_liege
	}
	text = { # 父为汉帝: 用父之国号
		trigger = {
			exists = father
			father = { primary_title = title:h_china }
		}
		localization_key = WJ_relation_from_father_guohao
	}
	text = { # 母为汉帝: 用母之国号
		trigger = {
			exists = mother
			mother = { primary_title = title:h_china }
		}
		localization_key = WJ_relation_from_mother_guohao
	}
	text = { # 父为王级以上
		trigger = {
			exists = father
			father = { primary_title.tier >= tier_kingdom }
		}
		localization_key = WJ_relation_from_father
	}
	text = { # 母为帝国级以上
		trigger = {
			exists = mother
			mother = { primary_title.tier > tier_kingdom }
		}
		localization_key = WJ_relation_from_mother
	}
	text = {
		fallback = yes
		localization_key = WJ_relation_from_fallback
	}
}
