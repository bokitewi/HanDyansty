wj_huangdi_zhaoshu_interaction = {
	icon = icon_contract_modification_single
	category = interaction_category_diplomacy

	use_diplomatic_range = yes

	name = wj_huangdi_zhaoshu_interaction
	desc = wj_huangdi_zhaoshu_interaction_desc

	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = {
			is_ai = no
			has_title = title:h_china
			government_has_flag = government_is_celestial
		}
	}

	# 下诏入宫
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_zhengpirugong_invalid
				scope:recipient = {
					is_imprisoned = no
					is_landed_or_landless_administrative = no
					NOT = { is_courtier_of = scope:actor }
					primary_title.tier < tier_barony
				}
			}
		}
		flag = xiazhao_zhengpirugong
		starts_enabled = { always = yes }
	}

	# 赏赐
	send_option = {
		flag = luoyang_shangci_interaction
		starts_enabled = { always = yes }
	}

	# 赐予同进士出身
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_jinshichushen_invalid
				scope:recipient = {
					OR = {
						AND = {
							is_vassal_or_below_of = scope:actor
							is_landed_or_landless_administrative = yes
							NOT = { has_character_modifier = tgp_passed_metropolitan_exam_modifier }
							NOT = { has_character_modifier = mod_jinshichushen_1 }
							NOT = { has_character_modifier = dynasty_celestial_favor_interaction_modifier }
						}
						AND = {
							NOT = { has_character_modifier = tgp_passed_metropolitan_exam_modifier }
							NOT = { has_character_modifier = mod_jinshichushen_1 }
							NOT = { has_character_modifier = dynasty_celestial_favor_interaction_modifier }
							is_courtier_of = scope:actor
						}
					}
				}
			}
		}
		flag = mod_jinshichushen
		starts_enabled = { always = no }
	}

	# 破格拔擢
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_add_major_invalid
				scope:recipient = {
					OR = {
						AND = {
							is_landed_or_landless_administrative = yes
							is_vassal_or_below_of = scope:actor
						}
						is_courtier_of = scope:actor
					}
				}
			}
		}
		flag = add_major_xianneng_option
		localization = "add_major_xianneng_option_desc"
	}

	# 官品拔擢
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_add_medium_invalid
				scope:recipient = {
					OR = {
						AND = {
							is_landed_or_landless_administrative = yes
							is_vassal_or_below_of = scope:actor

		}

		localization = "add_medium_xianneng_option_desc"
	}

	# 官品贬谪
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_reduce_major_invalid
				scope:recipient = {
					OR = {
						AND = {
							is_landed_or_landless_administrative = yes
							is_vassal_or_below_of = scope:actor
						}
						AND = {
							is_imprisoned_by = scope:actor
							is_courtier_of = scope:actor
						}
					}
				}
			}
		}
		flag = reduce_major_xianneng_option
		localization = "reduce_major_xianneng_option_desc"
	}

	# 小惩大戒
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_reduce_medium_invalid
				scope:recipient = {
					OR = {
						AND = {
							is_landed_or_landless_administrative = yes
							is_vassal_or_below_of = scope:actor
						}
						AND = {
							is_imprisoned_by = scope:actor
							is_courtier_of = scope:actor
						}
					}
				}
			}
		}
		flag = reduce_medium_xianneng_option
		localization = "reduce_medium_xianneng_option_desc"
	}

	# 构陷
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_random_negative_trait_invalid
				scope:recipient = {
					OR = {
						is_vassal_or_below_of = scope:actor
						is_courtier_of = scope:actor
					}
					NOR = {
						has_trait = witch
						has_trait = adulterer
						has_trait = fornicator

						has_trait = kinslayer_1
						has_trait = kinslayer_2
						has_trait = kinslayer_3
						has_trait = deviant
					}
				}
			}
		}
		flag = huangdi_random_negative_trait
	}

	# 罢免
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_cuihui_title_invalid
				scope:recipient = {
					is_vassal_or_below_of = scope:actor
					highest_held_title_tier >= tier_barony
					is_landless_ruler = no

						has_trait = witch

	# 永不录用
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_yongbuluyong_invalid
				scope:recipient = {
					is_landed = no
					NOT = { has_character_modifier = mod_yongbuluyong_1 }
				}
			}
		}
		flag = huangdi_yongbuluyong
	}

	# 流放
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_bianweishumin_invalid
				scope:recipient = {
					highest_held_title_tier <= tier_duchy
					is_landless_ruler = yes
					is_vassal_or_below_of = scope:actor
					dynasty != scope:actor.dynasty
				}
				scope:actor = {
					government_has_flag = government_is_celestial
					is_tributary = no
				}
			}
		}
		flag = mod_huangdi_bianweishumin
	}

	# 逮捕九族
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = huangdi_zhujiuzu_invalid
				scope:actor = {
					NOT = { exists = involved_activity }
					is_imprisoned = no
				}
				scope:recipient = {
					OR = {
						is_vassal_or_below_of = scope:actor
						is_courtier_of = scope:actor
					}
					OR = {
						has_trait = witch
						has_trait = adulterer
						has_trait = fornicator
						has_trait = sodomite
						has_trait = kinslayer_1
						has_trait = kinslayer_2
						has_trait = kinslayer_3
						has_trait = deviant
					}
				}
			}
		}
		flag = mod_huangdi_zhujiuzu
	}

	on_accept = {
		scope:recipient = { save_scope_as = wj_zhaoshu_target }
		switch = {
			trigger = yes

			# 破格拔擢
			scope:add_major_xianneng_option = {
				scope:actor = {
					change_influence = -800
					add_hook = {
						target = scope:recipient
						type = strong_test_hook
					}
				}
				scope:recipient = {
					change_merit = 10000
					add_opinion = {
						target = scope:actor
						modifier = grateful_opinion
						opinion = 100
					}
				}
			}

			# 官品拔擢
			scope:add_medium_xianneng_option = {
				scope:actor = {
					change_influence = -150
					add_hook = {
						target = scope:recipient
						type = favor_hook
					}
				}
				scope:recipient = {
					change_merit = 2000
					add_opinion = {
						target = scope:actor
						modifier = grateful_opinion
						opinion = 30
					}
				}
			}

			# 官品贬谪
			scope:reduce_major_xianneng_option = {
				scope:actor = {
					change_influence = -500
				}
				scope:recipient = {
					change_merit = -20000
					add_opinion = {
						target = scope:actor
						modifier = disgusted_opinion
						opinion = -30
					}
				}
			}

			# 小惩大戒
			scope:reduce_medium_xianneng_option = {
				scope:actor = {
					change_influence = -100
				}
				scope:recipient = {
					change_merit = -3000
					add_opinion = {
						target = scope:actor
						modifier = disgusted_opinion
						opinion = -20
					}
				}
			}

			# 下诏入宫
			scope:xiazhao_zhengpirugong = {
				scope:actor = {
					add_courtier = scope:recipient
					add_hook = {
						target = scope:recipient
						type = favor_hook
					}
				}
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = grateful_opinion
						opinion = 30
					}
				}
			}

			# 赏赐
			scope:luoyang_shangci_interaction = {
				scope:actor = {
					send_interface_message = {
						type = event_generic_neutral
						title = gift_interaction_notification
						right_icon = scope:recipient
						remove_short_term_gold = {
							value = 500
							multiply = 1
						}
						add_hook = {
							target = scope:recipient
							type = favor_hook
						}
					}
				}
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = gift_opinion
						opinion = send_gift_opinion
					}
					add_gold = 500
					add_opinion = {
						target = scope:actor
						modifier = grateful_opinion
						opinion = 50
					}
				}
			}

			# 同进士出身
			scope:mod_jinshichushen = {
				scope:actor = {
					add_hook = {
						target = scope:recipient
						type = perpetual_test_hook
					}
				}
				scope:recipient = {
					add_character_modifier = {
						modifier = mod_jinshichushen_1
					}
					add_opinion = {
						target = scope:actor
						modifier = loyalty_opinion
						opinion = 100
					}
				}
			}

			# 永不录用
			scope:huangdi_yongbuluyong = {
				scope:actor = {
					change_influence = -200
				}
				scope:recipient = {
					add_character_modifier = {
						modifier = mod_yongbuluyong_1
					}
					add_opinion = {
						target = scope:actor
						modifier = villain_opinion
						opinion = -100
					}
				}
			}

			# 罢免
			scope:mod_huangdi_cuihui_title_interaction = {
				scope:actor = {
					change_influence = -300
				}
				scope:recipient = {
					every_held_title = {
						limit = {
							NOT = { is_noble_family_title = yes }
						}
						create_title_and_vassal_change = {
							type = granted
							save_scope_as = title_change
							add_claim_on_loss = no
						}
						change_title_holder = {
							holder = scope:actor
							change = scope:title_change
						}
						resolve_title_and_vassal_change = scope:title_change
					}
					add_opinion = {
						target = scope:actor
						modifier = villain_opinion
						opinion = -50
					}
				}
			}

			# 逮捕九族
			scope:mod_huangdi_zhujiuzu = {
				scope:recipient.house = {
					every_house_member = {
						custom = every_house_member_tt
						custom_tooltip = {
							text = imprisoned_by_you
							scope:actor = {
								imprison = {
									target = prev
									type = house_arrest
								}
							}
						}
						add_opinion = {
							modifier = imprisoned_me
							years = 10
							target = scope:actor
						}
						change_merit = -10000
						add_opinion = {
							target = scope:actor
							modifier = villain_opinion
							opinion = -50
						}
					}
				}
				scope:actor = {
					change_influence = -1000
				}
			}

			# 构陷
			scope:huangdi_random_negative_trait = {
				scope:actor = {
					change_influence = -200
					send_interface_message = {
						type = event_generic_neutral
						title = huangdi_trait_assignment_notification
						right_icon = scope:recipient
					}
				}
				scope:recipient = {
					random_list = {
						1 = { add_trait = witch }
						1 = { add_trait = adulterer }
						1 = { add_trait = fornicator }
						1 = { add_trait = sodomite }
						1 = { add_trait = kinslayer_1 }
						1 = { add_trait = kinslayer_2 }
						1 = { add_trait = kinslayer_3 }
						1 = { add_trait = deviant }
					}
					add_opinion = {
						target = scope:actor
						modifier = disgusted_opinion
						opinion = -50
					}
				}
			}

			# 流放
			scope:mod_huangdi_bianweishumin = {
				scope:actor = {
					change_influence = -50
				}
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = disgusted_opinion
						opinion = -100
					}
					add_truce_both_ways = {
						character = scope:actor
						days = 3650
						name = TRUCE_GRANT_INDEPENDENCE
					}
					create_title_and_vassal_change = {
						type = independency
						save_scope_as = change
						add_claim_on_loss = yes
					}
					becomes_independent = {
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
					save_scope_as = adventurer
					if = {
						limit = { top_liege != this }
						liege = {
							save_scope_as = laamp_liege
							save_scope_as = laamp_inheritor
							custom_tooltip = become_landless_adventurer_decision_title_scope_tt
						}
					}
					else = {
						custom_tooltip = become_landless_adventurer_decision_title_created_tt
						if = {
							limit = {
								NOT = { exists = scope:laamp_inheritor }
							}
							hidden_effect_new_object = {
								create_character = {
									template = laamp_inheritor_template
									culture = root.capital_county.culture
									rite = root.capital_county.rite
									dynasty = generate
									location = root.capital_province
									after_creation = {
										save_scope_as = laamp_inheritor
										add_character_flag = new_laamp_inheritor
									}
								}
							}
						}
					}
					create_landless_adventurer_title_tooltip_effect = yes
					trigger_event = { id = ep3_laamps.0030 }
					stress_impact = {
						craven = minor_stress_impact_gain
						content = minor_stress_impact_gain
						lazy = minor_stress_impact_gain
					}
				}
			}
		}
	}

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
