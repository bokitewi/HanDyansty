# 乱世霸权·天子反击（第 4 期，2026-10-10，L4）
# 发出者为被挟持的天子（hd_court_defender_is_held_emperor_trigger）。

# 结纳：收买权臣的门客、骑士，为政变同党。玩家专用。
hd_court_suborn_interaction = {
	category = interaction_category_hostile
	icon = icon_scheme_befriend
	interface_priority = 60
	desc = hd_court_suborn_interaction_desc
	notification_text = hd_court_suborn_interaction_notification
	greeting = positive
	cooldown = { months = 6 }

	is_shown = {
		hd_court_privileges_active_trigger = yes
		scope:actor = {
			is_ai = no
			hd_court_defender_is_held_emperor_trigger = yes
		}
		scope:recipient = {
			is_adult = yes
			is_ruler = no
			employer ?= title:h_china.var:hd_court_regent
			NOT = { var:hd_court_coup_for ?= scope:actor }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_description = {
				text = hd_court_closely_watched_tt
				NOT = { has_character_modifier = hd_court_closely_watched_modifier }
			}
		}
	}

	cost = { gold = 50 }

	on_accept = {
		random_list = {
			90 = {
				scope:recipient = {
					set_variable = { name = hd_court_coup_for value = scope:actor years = 10 }
				}
				scope:actor = {
					add_to_variable_list = { name = hd_court_coup_supporters target = scope:recipient }
				}
			}
			10 = {
				custom_tooltip = hd_court_suborn_exposed_tt
				title:h_china.var:hd_court_regent ?= {
					add_opinion = { target = scope:actor modifier = hd_court_suspicious_opinion }
				}
			}
		}
	}

	ai_accept = {
		base = 0
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = title:h_china.var:hd_court_regent
			multiplier = -1
		}
		modifier = {
			desc = hd_court_reason_loyal
			add = -50
			scope:recipient = { has_trait = loyal }
		}
		modifier = {
			desc = hd_court_reason_greedy
			add = 20
			scope:recipient = { has_trait = greedy }
		}
		modifier = {
			desc = hd_court_reason_ambitious
			add = 10
			scope:recipient = { has_trait = ambitious }
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

# 衣带诏：密诏军阀勤王。奉诏者可免费、不论接壤，用挟天子 CB 讨伐权臣，胜后可还政或挟持。
hd_court_secret_edict_interaction = {
	category = interaction_category_diplomacy
	icon = icon_scheme_claim_throne
	interface_priority = 59
	desc = hd_court_secret_edict_interaction_desc
	notification_text = hd_court_secret_edict_interaction_notification
	greeting = positive
	ai_min_reply_days = 1
	ai_max_reply_days = 5
	cooldown = { years = 3 }
	use_diplomatic_range = no

	is_shown = {
		hd_court_privileges_active_trigger = yes
		scope:actor = { hd_court_defender_is_held_emperor_trigger = yes }
		scope:recipient = {
			hd_court_warlord_trigger = yes
			NOT = { hd_court_is_regent_trigger = yes }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_description = {
				text = hd_court_closely_watched_tt
				NOT = { has_character_modifier = hd_court_closely_watched_modifier }
			}
		}
	}

	on_accept = {
		random_list = {
			80 = {
				scope:recipient = {
					set_variable = { name = hd_court_qinwang_edict value = scope:actor years = 5 }
				}
			}
			20 = {
				custom_tooltip = hd_court_secret_edict_exposed_tt
				scope:actor = {
					add_character_modifier = { modifier = hd_court_closely_watched_modifier years = 5 }
					random_courtier = {
						limit = {
							is_adult = yes
							is_ruler = no
							NOT = { is_close_or_extended_family_of = scope:actor }
							NOT = { is_spouse_of = scope:actor }
						}
						death = { death_reason = death_execution }
					}
				}
				title:h_china.var:hd_court_regent ?= {
					add_opinion = { target = scope:actor modifier = hd_court_plotted_against_opinion }
				}
			}
		}
	}

	ai_accept = {
		base = 0
		modifier = {
			desc = hd_court_reason_zunhan
			add = 60
			scope:recipient = { hd_court_stance_zunhan_trigger = yes }
		}
		modifier = {
			desc = hd_court_reason_moni
			add = -30
			scope:recipient = { hd_court_stance_moni_trigger = yes }
		}
		modifier = {
			desc = hd_court_reason_rival_of_regent
			add = 40
			scope:recipient = {
				OR = {
					has_relation_rival = title:h_china.var:hd_court_regent
					has_relation_nemesis = title:h_china.var:hd_court_regent
				}
			}
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
	}

	ai_targets = { ai_recipients = neighboring_rulers }
	ai_frequency = 36
	ai_potential = {
		hd_court_defender_is_held_emperor_trigger = yes
		title:h_china.var:hd_court_regent = { hd_court_stance_moni_trigger = yes }
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 50
			scope:recipient = { hd_court_stance_zunhan_trigger = yes }
		}
	}
}

# 出奔：天子潜出，投奔一位军阀。投奔之主接任辅政，旧权臣带地独立，双方互得讨伐之名。
hd_court_flee_interaction = {
	category = interaction_category_diplomacy
	icon = icon_retreat
	interface_priority = 58
	desc = hd_court_flee_interaction_desc
	notification_text = hd_court_flee_interaction_notification
	greeting = positive
	ai_min_reply_days = 1
	ai_max_reply_days = 5
	cooldown = { years = 10 }
	use_diplomatic_range = no

	is_shown = {
		hd_court_privileges_active_trigger = yes
		scope:actor = { hd_court_defender_is_held_emperor_trigger = yes }
		scope:recipient = {
			hd_court_warlord_trigger = yes
			NOT = { hd_court_is_regent_trigger = yes }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_description = {
				text = hd_court_closely_watched_tt
				NOT = { has_character_modifier = hd_court_closely_watched_modifier }
			}
			is_imprisoned = no
		}
		scope:recipient = {
			NOT = { is_at_war_with = scope:actor }
			NOT = { is_at_war_with = title:h_china.var:hd_court_regent }
		}
	}

	on_accept = {
		scope:actor = { save_scope_as = hd_court_flee_emperor }
		random_list = {
			50 = {
				modifier = {
					add = {
						value = scope:actor.intrigue
						subtract = title:h_china.var:hd_court_regent.intrigue
						multiply = 2
					}
				}
				min = 20
				custom_tooltip = hd_court_flee_success_tt
				scope:recipient = { hd_court_flee_success_effect = yes }
				scope:actor = { trigger_event = hd_court.0014 }
			}
			50 = {
				modifier = {
					add = {
						value = title:h_china.var:hd_court_regent.intrigue
						subtract = scope:actor.intrigue
						multiply = 2
					}
				}
				min = 20
				custom_tooltip = hd_court_flee_failure_tt
				scope:actor = {
					add_character_modifier = { modifier = hd_court_closely_watched_modifier years = 10 }
					trigger_event = hd_court.0015
				}
				title:h_china.var:hd_court_regent ?= {
					add_opinion = { target = scope:actor modifier = hd_court_plotted_against_opinion }
				}
			}
		}
	}

	ai_accept = {
		base = -20
		modifier = {
			desc = hd_court_reason_zunhan
			add = 60
			scope:recipient = { hd_court_stance_zunhan_trigger = yes }
		}
		modifier = {
			desc = hd_court_reason_moni
			add = 20
			scope:recipient = { hd_court_stance_moni_trigger = yes }
		}
		modifier = {
			desc = hd_court_reason_weaker_than_regent
			add = -40
			scope:recipient = {
				current_military_strength < title:h_china.var:hd_court_regent.current_military_strength
			}
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
	}

	ai_targets = { ai_recipients = neighboring_rulers }
	ai_frequency = 60
	ai_potential = {
		hd_court_defender_is_held_emperor_trigger = yes
		title:h_china.var:hd_court_regent = { hd_court_stance_moni_trigger = yes }
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 20
			scope:recipient = {
				hd_court_stance_zunhan_trigger = yes
				current_military_strength >= title:h_china.var:hd_court_regent.current_military_strength
			}
		}
	}
}
