# 请求归附：弱小的 AI 独立诸侯主动请求成为相邻强者的封臣。
# 仅由 hd_petition_try_effect（每年一次）发出；玩家自己投靠请用原版“宣誓效忠”。
# actor = 投靠者，recipient = 被投靠者；结算沿用原版 offer_fealty_interaction_effect。
hd_petition_vassalage_interaction = {
    category = interaction_category_vassal
    icon = icon_liege
    common_interaction = no
    desc = hd_petition_vassalage_interaction_desc
    greeting = positive
    notification_text = hd_petition_request_notification
    auto_accept = no
    ai_maybe = no
    popup_on_receive = yes
    pause_on_receive = yes
    ai_min_reply_days = 5
    ai_max_reply_days = 10
    cooldown = { years = 1 }
    cooldown_against_recipient = { years = 2 }
    ai_targets = { ai_recipients = neighboring_rulers }
    ai_frequency_by_tier = {
        barony = 0
        county = 0
        duchy = 0
        kingdom = 0
        empire = 0
        hegemony = 0
    }
    is_shown = {
        scope:actor = {
            is_ai = yes
            this != scope:recipient
            is_landed = yes
            is_independent_ruler = yes
        }
        scope:recipient = {
            hd_submission_liege_trigger = yes
            highest_held_title_tier > scope:actor.highest_held_title_tier
        }
    }
    is_valid_showing_failures_only = {
        scope:actor = {
            hd_submission_recipient_legal_trigger = { LIEGE = scope:recipient }
            NOT = { hd_submission_ai_refuses_trigger = { LIEGE = scope:recipient } }
        }
    }
    can_send = {
        scope:actor = {
            NOT = { has_character_flag = hd_petition_outgoing_pending }
        }
    }
    ai_accept = {
        base = 50
        modifier = {
            desc = hd_petition_reason_faith
            trigger = { scope:actor.faith != scope:recipient.faith }
            add = hd_petition_faith_value
        }
        modifier = {
            desc = hd_petition_reason_rival
            trigger = {
                scope:recipient = {
                    OR = {
                        has_relation_rival = scope:actor
                        has_relation_nemesis = scope:actor
                    }
                }
            }
            add = -1000
        }
    }
    send_options_exclusive = yes
    send_option = {
        is_shown = { scope:actor = { government_has_flag = government_is_feudal } }
        flag = low_obligations
        localization = low_obligations
    }
    send_option = {
        is_shown = { scope:actor = { government_has_flag = government_is_feudal } }
        flag = normal_obligations
        localization = normal_obligations
        starts_enabled = { always = yes }
    }
    send_option = {
        is_shown = { scope:actor = { government_has_flag = government_is_feudal } }
        flag = high_obligations
        localization = high_obligations
    }
    send_option = {
        is_shown = { scope:actor = { government_has_flag = government_is_feudal } }
        is_valid = { scope:recipient.faith != scope:actor.faith }
        flag = religious_taxation
        localization = religious_taxation
    }
    send_option = {
        is_shown = { scope:actor = { government_has_flag = government_is_clan } }
        is_valid = { scope:recipient.faith != scope:actor.faith }
        flag = religious_exemption_clan
        localization = religious_exemption
    }
    send_option = {
        is_shown = { scope:actor = { government_has_flag = government_is_clan } }
        flag = no_exemption_clan
        localization = no_exemption
        starts_enabled = { always = yes }
    }
    on_send = { hidden_effect = { hd_petition_request_sent_effect = yes } }
    on_accept = {
        custom_tooltip = hd_petition_accept_summary_tt
        show_as_tooltip = {
            offer_fealty_interaction_effect = yes
            scope:recipient = { add_prestige = 50 }
        }
        hidden_effect = { hd_petition_request_accepted_effect = yes }
    }
    on_decline = {
        custom_tooltip = hd_petition_decline_summary_tt
        hidden_effect = { hd_petition_request_declined_effect = yes }
    }
    on_blocked_effect = { hidden_effect = { hd_petition_request_declined_effect = yes } }
    ai_potential = { is_ai = yes }
    ai_will_do = { base = 100 }
}
