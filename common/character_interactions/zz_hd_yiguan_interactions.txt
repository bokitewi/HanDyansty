# 流亡者请资助（流亡者 → 南方政权 / 北方汉人）

hd_yiguan_request_funding_interaction = {
	category = interaction_category_diplomacy
	icon = request_funding_interaction
	desc = hd_yiguan_request_funding_interaction_desc
	use_diplomatic_range = yes

	is_shown = {
		hd_yiguan_active_trigger = yes
		NOT = { scope:actor = scope:recipient }
		scope:actor = { hd_yiguan_gt_exile_base_trigger = yes }
		scope:recipient = { hd_yiguan_gt_funder_base_trigger = yes }
	}
	is_valid_showing_failures_only = {
		scope:actor = { hd_yiguan_gt_fundable_trigger = { FUNDER = scope:recipient } }
		scope:recipient = { treasury_or_gold >= hd_yiguan_gt_fund_unit_value }	# 国库或金币，写法同原版 00_demand_church_succession_interactions.txt:34
	}

	send_option = {
		flag = hd_yiguan_gt_tier_gold
		localization = hd_yiguan_gt_tier_gold
		starts_enabled = { always = yes }
		is_valid = { scope:recipient = { treasury_or_gold >= hd_yiguan_gt_fund_unit_value } }
	}
	send_option = {
		flag = hd_yiguan_gt_tier_troops
		localization = hd_yiguan_gt_tier_troops
		is_valid = { scope:recipient = { treasury_or_gold >= hd_yiguan_gt_fund_unit_value } }
	}
	send_option = {
		flag = hd_yiguan_gt_tier_both
		localization = hd_yiguan_gt_tier_both
		is_valid = {
			scope:recipient = {
				treasury_or_gold >= {
					value = hd_yiguan_gt_fund_unit_value
					multiply = 2
				}
			}
		}
	}
	send_options_exclusive = yes

	on_accept = {
		if = {
			limit = { scope:hd_yiguan_gt_tier_both = yes }
			hidden_effect = { hd_yiguan_gt_fund_effect = { FUNDER = scope:recipient EXILE = scope:actor GOLD = yes TROOPS = yes PRESTIGE = 150 } }
		}
		else_if = {
			limit = { scope:hd_yiguan_gt_tier_troops = yes }
			hidden_effect = { hd_yiguan_gt_fund_effect = { FUNDER = scope:recipient EXILE = scope:actor GOLD = no TROOPS = yes PRESTIGE = 100 } }
		}
		else = {
			hidden_effect = { hd_yiguan_gt_fund_effect = { FUNDER = scope:recipient EXILE = scope:actor GOLD = yes TROOPS = no PRESTIGE = 50 } }
		}
		custom_tooltip = hd_yiguan_gt_fund_accept_tt
	}

	ai_accept = {
		base = -10
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_conflict
			trigger = { hd_yiguan_phase_chongtu_trigger = yes }
			add = 15
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_own_court
			trigger = { scope:actor = { liege_or_court_owner ?= scope:recipient } }
			add = 10
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_martial
			trigger = { scope:actor = { martial >= 12 } }
			add = 10
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_generous
			trigger = { scope:recipient = { has_trait = generous } }
			add = 15
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_greedy
			trigger = { scope:recipient = { has_trait = greedy } }
			add = -20
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_big_ask
			trigger = { scope:hd_yiguan_gt_tier_both = yes }
			add = -15
		}
		modifier = {
			desc = hd_yiguan_gt_ai_reason_rival
			trigger = {
				scope:recipient = {
					OR = {
						has_relation_rival = scope:actor
						has_relation_nemesis = scope:actor
					}
				}
			}
			add = -100
		}
	}

	cooldown_against_recipient = { years = 3 }
	ai_targets = {
		ai_recipients = liege
		ai_recipients = top_liege
	}
	ai_frequency = 36
	ai_potential = {
		hd_yiguan_gt_exile_base_trigger = yes
		hd_yiguan_phase_chongtu_trigger = yes
	}
	ai_will_do = {
		base = 20
		modifier = {
			add = 20
			scope:actor = {
				has_trait = ambitious
			}
		}
		modifier = {
			add = -15
			scope:actor = {
				has_trait = content
			}
		}
		# 档位：AI 取 ai_will_do 最高的档（原版 _character_interactions.info:611），各档须分开打分（先例 原版 pam_interactions.txt:1884）
		# 默认求钱财；通晓兵事者求兵马；通晓兵事又进取或勇敢、对方钱多者求兵钱兼有
		modifier = {
			add = 5
			scope:hd_yiguan_gt_tier_gold = yes
		}
		modifier = {
			add = 10
			scope:hd_yiguan_gt_tier_troops = yes
			scope:actor = { martial >= 12 }
		}
		modifier = {
			add = 15
			scope:hd_yiguan_gt_tier_both = yes
			scope:actor = {
				martial >= 12
				OR = {
					has_trait = ambitious
					has_trait = brave
				}
			}
			scope:recipient = {
				treasury_or_gold >= {
					value = hd_yiguan_gt_fund_unit_value
					multiply = 4
				}
			}
		}
	}
}
