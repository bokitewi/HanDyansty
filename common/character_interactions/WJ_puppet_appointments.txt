# Native puppet actions keep actor=controller and puppet_or_actor=employer.
# Reuse the original council-task/court-position UI and engine eligibility.
WJ_puppet_council_appointment_interaction = {
 category = interaction_category_vassal
 interface_priority = 120
 icon = council
 special_interaction = invite_to_council_interaction
 interface = council_task_interaction
 popup_on_receive = yes
 pause_on_receive = yes
 desc = WJ_puppet_council_appointment_interaction_desc
 notification_text = WJ_puppet_council_appointment_notification
 is_shown = {
  scope:puppet_or_actor = { is_ruler = yes puppeteer = scope:actor has_council = yes }
  scope:recipient = { liege_or_court_owner = scope:puppet_or_actor this != scope:puppet_or_actor }
 }
 is_valid_showing_failures_only = {
  NOT = { scope:puppet_or_actor = { is_at_war_with = scope:recipient } }
  NOT = { scope:recipient.council_task = scope:target }
  trigger_if = {
   limit = { exists = scope:target.councillor }
   scope:target = { can_fire_position = yes }
   scope:target.councillor = { NOT = { is_blocked_from_being_fired_from_council_trigger = { COURT_OWNER = scope:puppet_or_actor } } }
  }
 }
 on_accept = {
  scope:puppet_or_actor = { assign_council_task = { target = scope:recipient council_task = scope:target } }
 }
 auto_accept = { scope:recipient = { is_ai = yes } }
 ai_frequency = 0
}

WJ_puppet_court_appointment_interaction = {
 category = interaction_category_vassal
 interface_priority = 120
 icon = seneschal_court_position
 special_interaction = grant_court_position
 interface = court_task_interaction
 target_type = court_position_type
 desc = WJ_puppet_court_appointment_interaction_desc
 is_shown = {
  scope:puppet_or_actor = { is_ruler = yes puppeteer = scope:actor }
  scope:recipient = {
   this != scope:puppet_or_actor
   liege = scope:puppet_or_actor
   tgp_is_ceremonial_liege_trigger = no
   NOT = { has_character_flag = travel_option_added_character }
  }
 }
 is_valid_showing_failures_only = {
  scope:recipient = { NOT = { is_imprisoned_by = scope:puppet_or_actor } is_adult = yes }
 }
 on_accept = {
  scope:puppet_or_actor = { appoint_court_position = { recipient = scope:recipient court_position = scope:target } }
 }
 auto_accept = yes
 ai_frequency = 0
}
