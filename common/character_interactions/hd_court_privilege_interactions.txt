# 授军职：沿用 WJ 军号
hd_court_grant_rank_interaction = {
	category = interaction_category_diplomacy
	icon = honorary_titles
	interface_priority = 60
	desc = hd_court_grant_rank_interaction_desc
	notification_text = hd_court_grant_rank_interaction_notification
	greeting = positive
	ai_min_reply_days = 1
	ai_max_reply_days = 5
	cooldown_against_recipient = { years = 5 }

	is_shown = {
		hd_court_privileges_active_trigger = yes
		scope:actor = { hd_court_controller_trigger = yes }
		scope:recipient = {
			hd_court_warlord_trigger = yes
			this != scope:actor
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_description = {
				text = hd_court_already_has_rank_tt
				NOT = { WJ_mr_has_rank = yes }
			}
			primary_title = { tier <= tier_kingdom }
		}
	}

	cost = { prestige = 50 }

	on_accept = {
		scope:recipient = {
			hd_court_grant_rank_effect = yes
			add_opinion = { target = scope:actor modifier = hd_court_honored_by_court_opinion }
		}
	}

	ai_accept = {
		base = 0
		modifier = {
			desc = hd_court_reason_zunhan
			add = 100
			scope:recipient = { hd_court_stance_zunhan_trigger = yes }
		}
		modifier = {
			desc = hd_court_reason_moni
			add = -60
			scope:recipient = { hd_court_stance_moni_trigger = yes }
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
	}

	ai_targets = { ai_recipients = neighboring_rulers }
	ai_frequency = 36
	ai_potential = {
		hd_court_controller_trigger = yes
		prestige >= 300
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 60
			scope:recipient = { hd_court_stance_zunhan_trigger = yes }
		}
	}
}

# 献地于朝廷：尊汉军阀把领地交还天子，由朝廷按法理另行分封
hd_court_return_land_interaction = {
	category = interaction_category_diplomacy
	icon = grant_titles_interaction
	interface_priority = 55
	desc = hd_court_return_land_interaction_desc
	notification_text = hd_court_return_land_interaction_notification
	greeting = positive
	target_type = title
	target_filter = actor_domain_titles
	auto_accept = yes

	is_shown = {
		hd_court_privileges_active_trigger = yes
		scope:recipient = { has_title = title:h_china }
		scope:actor = {
			hd_court_warlord_trigger = yes
			hd_court_stance_zunhan_trigger = yes
		}
	}

	can_be_picked_title = {
		scope:target = {
			tier >= tier_county
			tier <= tier_duchy
			is_landless_type_title = no
			is_noble_family_title = no
			NOT = { this = scope:actor.capital_county }
			NOT = { this = scope:actor.primary_title }
		}
	}
	has_valid_target_showing_failures_only = {
		scope:target = { tier >= tier_county }
	}

	on_accept = {
		scope:target = { save_scope_as = hd_court_returned_title }
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_court_return_change
			add_claim_on_loss = no
		}
		scope:target = {
			change_title_holder = {
				holder = scope:recipient
				change = scope:hd_court_return_change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:hd_court_return_change
		scope:actor = {
			if = {
				limit = { scope:hd_court_returned_title = { tier >= tier_duchy } }
				add_prestige = 250
			}
			else = { add_prestige = 100 }
		}
		scope:recipient = {
			add_opinion = { target = scope:actor modifier = hd_court_returned_land_opinion }
			if = {
				limit = { legitimacy > 0 }
				add_legitimacy = 50
			}
		}
		title:h_china.var:hd_court_regent ?= {
			if = {
				limit = { is_alive = yes }
				add_opinion = { target = scope:actor modifier = hd_court_returned_land_opinion }
			}
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
