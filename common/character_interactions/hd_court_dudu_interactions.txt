# 乱世霸权·都督区（第 6 期，2026-10-10，L7）
# 遥授宣称：延续"有宣称就有 CB"——天子/都督中外/大将军（或挟天子的权臣）给境内都督一个相邻境外郡的宣称，
# 由都督自行择机用宣称战争讨伐（对象可以是独立领主，也可以是别家的封臣）。与"指定征讨"并存，不替换。
# 目标判定与收件人复用 WJ_conquest_*（common/scripted_triggers/WJ_conquest_triggers.txt）。
hd_court_dudu_grant_claim_interaction = {
	category = interaction_category_diplomacy
	interface_priority = 79
	icon = icon_claim_pressed
	desc = hd_court_dudu_grant_claim_interaction_desc
	notification_text = hd_court_dudu_grant_claim_interaction_notification
	filter_tags = { admin_governor }
	auto_accept = yes
	cooldown_against_recipient = { years = 5 }

	target_type = title
	target_filter = secondary_recipient_de_jure_titles
	can_be_picked_title = { WJ_conquest_target_trigger = yes }

	populate_recipient_list = {
		scope:recipient = {
			every_character_to_title_neighboring_duchy = {
				limit = {
					holder ?= { top_liege != scope:actor.top_liege }
				}
				holder ?= { add_to_list = characters }
			}
		}
	}
	redirect = {
		scope:recipient = {
			random_character_to_title_neighboring_duchy = {
				limit = {
					holder ?= { top_liege != scope:actor.top_liege }
				}
				holder = { save_scope_as = secondary_recipient }
			}
		}
	}
	has_valid_target = { WJ_conquest_target_trigger = yes }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			OR = {
				# 天子未被挟持时：天子、都督中外、大将军
				AND = {
					WJ_conquest_commander_trigger = yes
					NOT = {
						AND = {
							has_title = title:h_china
							hd_court_has_regent_trigger = yes
						}
					}
				}
				# 天子被挟持时：权臣代行
				hd_court_is_regent_trigger = yes
			}
		}
		scope:recipient = {
			WJ_conquest_authorized_governor_trigger = yes
			top_liege = scope:actor.top_liege
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = { is_adult = yes }
	}

	cost = { prestige = 100 }

	on_accept = {
		scope:recipient = {
			add_pressed_claim = scope:target
			add_opinion = { target = scope:actor modifier = hd_court_honored_by_court_opinion }
		}
	}

	ai_targets = { ai_recipients = vassals }
	ai_frequency = 36
	ai_potential = {
		OR = {
			WJ_conquest_commander_trigger = yes
			hd_court_is_regent_trigger = yes
		}
		prestige >= 300
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 20
			scope:recipient = { is_at_war = no }
			scope:secondary_recipient = { hd_court_stance_moni_trigger = yes }
		}
	}
}
