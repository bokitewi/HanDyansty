# Native military-aid character picker, scoped to the head's district.
wj_dd_deploy_aid_interaction = {
 icon = icon_soldier_survivor category = interaction_category_wj_district
 common_interaction = yes desc = wj_dd_deploy_aid_interaction_desc
 filter_tags = { wj_district_management }
 shows_military_strength = yes auto_accept = yes
 is_shown = {
  scope:actor = { WJ_dd_head_trigger = yes }
  scope:recipient = { WJ_dd_member_of_core_trigger = { CORE = scope:actor.primary_title } }
 }
 is_valid_showing_failures_only = {
  custom_tooltip = { text = WJ_dd_dispatch_available_tt WJ_dd_dispatch_available_trigger = yes }
 }
 populate_recipient_list = {
  scope:actor.top_liege = {
   every_vassal_or_below = {
    limit = { WJ_dd_aid_officer_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient } }
    add_to_list = characters
   }
  }
 }
 can_send = {
  WJ_dd_aid_pair_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient }
  exists = scope:secondary_recipient
  scope:secondary_recipient = { WJ_dd_aid_officer_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient } }
 }
 on_accept = {
  scope:actor = { save_scope_as = wj_dd_commander }
  scope:recipient = { save_scope_as = wj_dd_beneficiary }
  scope:secondary_recipient = { save_scope_as = wj_dd_target }
  WJ_dd_join_aid_wars_effect = yes
 }
 ai_potential = { root = { WJ_dd_head_trigger = yes } }
 ai_frequency = 3
 ai_targets = { ai_recipients = realm_characters }
 ai_will_do = {
  base = 0
  modifier = { add = 20 WJ_dd_dispatch_available_trigger = yes }
  modifier = { add = 80 scope:recipient = { NOT = { any_war_ally = { always = yes } } } }
  modifier = { add = 30 scope:recipient = { any_war_enemy = { current_military_strength > scope:recipient.current_military_strength } } }
  modifier = { factor = 2 scope:secondary_recipient = { is_at_war = no } }
 }
}

# Same deployment result, with the commander himself as the joiner. No officer picker.
wj_dd_join_wars_01 = {
 icon = icon_soldier_survivor category = interaction_category_wj_district
 filter_tags = { wj_district_management }
 common_interaction = yes desc = wj_dd_join_wars_01_desc
 shows_military_strength = yes auto_accept = yes
 is_shown = {
  scope:actor = { WJ_dd_head_trigger = yes }
  scope:recipient = {
   this != scope:actor
   WJ_dd_member_of_core_trigger = { CORE = scope:actor.primary_title }
   any_character_war = { is_war_leader = scope:recipient }
  }
 }
 is_valid_showing_failures_only = {
  custom_tooltip = {
   text = WJ_dd_join_personally_available_tt
   WJ_dd_aid_pair_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient }
   scope:actor = { WJ_dd_aid_joiner_trigger = { BENEFICIARY = scope:recipient } }
  }
 }
 can_send = {
  WJ_dd_aid_pair_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient }
  scope:actor = { WJ_dd_aid_joiner_trigger = { BENEFICIARY = scope:recipient } }
 }
 on_accept = {
  scope:actor = { save_scope_as = wj_dd_commander save_scope_as = wj_dd_target }
  scope:recipient = { save_scope_as = wj_dd_beneficiary }
  WJ_dd_join_aid_wars_effect = yes
 }
 ai_potential = { root = { WJ_dd_head_trigger = yes } }
 ai_frequency = 3
 ai_targets = { ai_recipients = realm_characters }
 ai_will_do = {
  base = 10
  modifier = { add = 70 scope:recipient = { NOT = { any_war_ally = { always = yes } } } }
  modifier = { add = 30 scope:recipient = { any_war_enemy = { current_military_strength > scope:recipient.current_military_strength } } }
  modifier = { factor = 0.25 scope:actor = { is_at_war = yes } }
 }
}
