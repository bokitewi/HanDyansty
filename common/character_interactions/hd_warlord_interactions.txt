# 郡国制·军阀契约：「收回权力」互动（2026-10-24 用户指定）
# - 行为人：军阀的直属主君；对象：直属军阀封臣
# - 同意：变回普通武官（军事编制名额已满则退为普通文官），继承改回天朝流官任命；该封臣 5 年内不得再自请为军阀
# - 拒绝：主君获得对其的「讨不臣」宣战理由（5 年），胜利则收回权力、夺其主头衔并下狱
# - 对同一军阀冷却 5 年
# - AI 军阀是否同意：对主君好感 ×0.5，实力（军阀/主君）≥1 -100、≥0.5 -50、<0.25 +50，性格各 ±10~50
# - AI 主君少用：每 60 个月评估，自身兵力须达军阀两倍以上

hd_warlord_revoke_interaction = {
	category = interaction_category_vassal
	icon = revoke_governorship
	interface_priority = 100
	desc = hd_warlord_revoke_interaction_desc
	greeting = negative
	notification_text = hd_warlord_revoke_interaction_notification
	popup_on_receive = yes
	pause_on_receive = yes
	# 军阀拒绝后主君才拿到讨不臣，故对方不愿接受时也必须能发出
	can_send_despite_rejection = yes

	cooldown_against_recipient = { years = 5 }

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = {
			liege = scope:actor
			hd_warlord_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}
	}

	on_accept = {
		scope:recipient = { hd_warlord_revoke_effect = yes }
	}

	on_decline = {
		scope:recipient = { hd_warlord_defy_effect = yes }
	}

	# AI 军阀是否同意（大于 0 即同意）。拒绝即招来讨不臣，故自身越弱越倾向同意
	ai_accept = {
		base = 0
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}
		# 军力比 = 军阀 / 主君（max_military_strength）；用乘法比较，避免除以 0
		modifier = {
			add = -100
			scope:recipient.max_military_strength >= scope:actor.max_military_strength
			desc = hd_warlord_revoke_ai_strength_strong
		}
		modifier = {
			add = -50
			scope:recipient.max_military_strength < scope:actor.max_military_strength
			scope:recipient.max_military_strength >= {
				value = scope:actor.max_military_strength
				multiply = 0.5
			}
			desc = hd_warlord_revoke_ai_strength_half
		}
		modifier = {
			add = 50
			scope:recipient.max_military_strength < {
				value = scope:actor.max_military_strength
				multiply = 0.25
			}
			desc = hd_warlord_revoke_ai_strength_weak
		}
		# 性格：同意
		modifier = {
			add = 50
			scope:recipient = { has_trait = loyal }
			desc = trait_loyal
		}
		modifier = {
			add = 25
			scope:recipient = { has_trait = craven }
			desc = trait_craven
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = content }
			desc = trait_content
		}
		modifier = {
			add = 15
			scope:recipient = { has_trait = humble }
			desc = trait_humble
		}
		modifier = {
			add = 10
			scope:recipient = { has_trait = trusting }
			desc = trait_trusting
		}
		# 性格：拒绝
		modifier = {
			add = -30
			scope:recipient = { has_trait = ambitious }
			desc = trait_ambitious
		}
		modifier = {
			add = -20
			scope:recipient = { has_trait = arrogant }
			desc = trait_arrogant
		}
		modifier = {
			add = -20
			scope:recipient = { has_trait = stubborn }
			desc = trait_stubborn
		}
		modifier = {
			add = -10
			scope:recipient = { has_trait = brave }
			desc = trait_brave
		}
		modifier = {
			add = -10
			scope:recipient = { has_trait = paranoid }
			desc = trait_paranoid
		}
	}

	ai_potential = {
		any_vassal = { hd_warlord_trigger = yes }
	}
	ai_targets = {
		ai_recipients = vassals
	}
	ai_frequency = 60

	# AI 主君是否发起（root = scope:actor = 主君）：自身兵力须达军阀两倍以上
	ai_will_do = {
		base = 20
		# 剧本开局设定的军阀有保护期（hd_warlord_revoke_ai_grace，如 190 讨董的笮融/臧霸/黄祖 5 年）：AI 主君不发起
		modifier = {
			factor = 0
			scope:recipient = { has_character_flag = hd_warlord_revoke_ai_grace }
		}
		modifier = {
			add = 20
			scope:actor.max_military_strength >= {
				value = scope:recipient.max_military_strength
				multiply = 3
			}
		}
		modifier = {
			add = 10
			has_trait = ambitious
		}
		modifier = {
			add = 10
			has_trait = paranoid
		}
		modifier = {
			add = 10
			has_trait = arrogant
		}
		modifier = {
			add = -10
			has_trait = content
		}
		modifier = {
			add = -10
			has_trait = craven
		}
		# （置于末尾，避免被 add 抵消）
		modifier = {
			factor = 0
			scope:actor.max_military_strength < {
				value = scope:recipient.max_military_strength
				multiply = 2
			}
		}
	}
}
