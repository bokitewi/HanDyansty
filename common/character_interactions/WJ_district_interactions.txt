# Initial construction and management use native interaction windows.
wj_dd_create_interaction = {
 filter_tags = { wj_district_management }
 icon = icon_contract_modification_single
 category = interaction_category_wj_district
 common_interaction = yes
 desc = wj_dd_create_interaction_desc
 auto_accept = yes
 is_shown = { WJ_dd_create_visible_trigger = yes }
 is_valid_showing_failures_only = { WJ_dd_create_trigger = yes }
 send_options_exclusive = yes
 send_option = { flag = wj_dd_jiajie localization = WJ_dd_choose_jiajie starts_enabled = { always = no } can_invalidate_interaction = yes }
 send_option = { flag = wj_dd_chijie localization = WJ_dd_choose_chijie starts_enabled = { always = no } can_invalidate_interaction = yes }
 send_option = { flag = wj_dd_shichijie localization = WJ_dd_choose_shichijie starts_enabled = { always = no } can_invalidate_interaction = yes }
 can_send = {
  WJ_dd_create_trigger = yes
  custom_tooltip = {
   text = WJ_dd_choose_jie_tt
   OR = { scope:wj_dd_jiajie = yes scope:wj_dd_chijie = yes scope:wj_dd_shichijie = yes }
  }
 }
 on_accept = {
  hidden_effect = {
  if = {
   limit = { WJ_dd_create_trigger = yes }
   if = { limit = { scope:wj_dd_jiajie = yes } WJ_dd_create_effect = { JIE = 1 } }
   else_if = { limit = { scope:wj_dd_chijie = yes } WJ_dd_create_effect = { JIE = 2 } }
   else_if = { limit = { scope:wj_dd_shichijie = yes } WJ_dd_create_effect = { JIE = 3 } }
  }
  scope:actor = { if = { limit = { is_ai = no } WJ_dd_refresh_view_effect = yes } }
  }
 }
 ai_potential = { root = { WJ_zw_central_authority = yes } }
 ai_targets = { ai_recipients = realm_characters }
 ai_frequency = 12
 ai_will_do = {
  base = 10
  modifier = { add = 15 scope:wj_dd_chijie = yes }
  modifier = { add = 25 scope:wj_dd_shichijie = yes scope:recipient = { dynasty = scope:actor.dynasty } }
 }
}
wj_dd_manage_interaction = {
 filter_tags = { wj_district_management }
 hidden = yes # Keep the explicit district-page entry, not the right-click menu.
 icon = icon_contract_modification_single
 category = interaction_category_wj_district
 common_interaction = yes
 desc = wj_dd_manage_interaction_desc
 auto_accept = yes
 is_shown = { WJ_dd_manage_visible_trigger = yes }
 is_valid_showing_failures_only = { WJ_dd_manage_trigger = yes }
 populate_recipient_list = {
  scope:actor.top_liege = {
   every_vassal_or_below = {
    limit = {
     WJ_dd_king_official_trigger = yes
     this != scope:recipient
     primary_title = {
      NOT = { has_variable = wj_dd_core }
      WJ_dd_touches_core_trigger = { CORE = scope:recipient.primary_title }
     }
    }
    add_to_list = characters
   }
  }
 }
 can_send = { WJ_dd_add_trigger = yes }
 on_accept = {
  hidden_effect = {
  WJ_dd_add_effect = yes
  scope:actor = { if = { limit = { is_ai = no } WJ_dd_refresh_view_effect = yes } }
  }
 }
 ai_potential = { root = { WJ_zw_central_authority = yes } }
 ai_targets = { ai_recipients = realm_characters }
 ai_frequency = 12
 ai_will_do = { base = 0 modifier = { add = 10 WJ_dd_add_trigger = yes } }
}
# Active replacement asks for a fresh jie; natural succession is separate.
wj_dd_withdraw_interaction = {
 filter_tags = { wj_district_management }
 icon = icon_contract_modification_single
 category = interaction_category_wj_district
 common_interaction = yes
 desc = wj_dd_withdraw_interaction_desc
 auto_accept = yes
 is_shown = { WJ_dd_withdraw_candidate_trigger = yes }
 is_valid_showing_failures_only = {
  WJ_dd_withdraw_candidate_trigger = yes
  custom_tooltip = {
   text = WJ_dd_remove_connected_tt
   scope:recipient.primary_title = { has_variable = wj_dd_ui_can_remove }
  }
 }
 on_accept = {
  hidden_effect = {
   if = {
    limit = { WJ_dd_withdraw_candidate_trigger = yes }
    scope:recipient = {
     save_scope_as = secondary_recipient
     primary_title.var:wj_dd_core.holder = { save_scope_as = wj_dd_previous_head }
    }
    scope:wj_dd_previous_head = { save_scope_as = recipient }
    WJ_dd_remove_effect = yes
    scope:secondary_recipient = { save_scope_as = recipient }
    scope:actor = { WJ_dd_refresh_view_effect = yes }
   }
  }
 }
 ai_potential = { always = no }
 ai_frequency = 12
 ai_targets = { ai_recipients = realm_characters }
}
wj_dd_promote_interaction = {
 filter_tags = { wj_district_management }
 icon = icon_contract_modification_single
 category = interaction_category_wj_district
 common_interaction = yes
 desc = wj_dd_promote_interaction_desc
 auto_accept = yes
 is_shown = { WJ_dd_promote_trigger = yes }
 is_valid_showing_failures_only = { WJ_dd_promote_trigger = yes }
 send_options_exclusive = yes
 send_option = { flag = wj_dd_jiajie localization = WJ_dd_choose_jiajie starts_enabled = { always = no } can_invalidate_interaction = yes }
 send_option = { flag = wj_dd_chijie localization = WJ_dd_choose_chijie starts_enabled = { always = no } can_invalidate_interaction = yes }
 send_option = { flag = wj_dd_shichijie localization = WJ_dd_choose_shichijie starts_enabled = { always = no } can_invalidate_interaction = yes }
 can_send = {
  WJ_dd_promote_trigger = yes
  custom_tooltip = {
   text = WJ_dd_choose_jie_tt
   OR = { scope:wj_dd_jiajie = yes scope:wj_dd_chijie = yes scope:wj_dd_shichijie = yes }
  }
 }
 on_accept = {
  hidden_effect = {
   if = {
    limit = { WJ_dd_promote_trigger = yes }
    if = { limit = { scope:wj_dd_jiajie = yes } save_scope_value_as = { name = wj_dd_selected_jie value = 1 } }
    else_if = { limit = { scope:wj_dd_chijie = yes } save_scope_value_as = { name = wj_dd_selected_jie value = 2 } }
    else_if = { limit = { scope:wj_dd_shichijie = yes } save_scope_value_as = { name = wj_dd_selected_jie value = 3 } }
    # Capture both people before rebinding the existing transaction's recipient.
    scope:recipient = {
     save_scope_as = secondary_recipient
     primary_title.var:wj_dd_core.holder = { save_scope_as = wj_dd_previous_head }
    }
    scope:wj_dd_previous_head = { save_scope_as = recipient }
    WJ_dd_replace_effect = yes
    scope:secondary_recipient = { save_scope_as = recipient }
    scope:actor = { WJ_dd_refresh_view_effect = yes }
   }
  }
 }
 ai_potential = { always = no }
 ai_frequency = 12
 ai_targets = { ai_recipients = realm_characters }
}

# Same district status strip, native send-option window; no general contract rights.
wj_dd_adjust_jie_interaction = {
 filter_tags = { wj_district_management } hidden = yes
 icon = icon_contract_modification_single category = interaction_category_wj_district
 common_interaction = yes auto_accept = yes
 desc = wj_dd_adjust_jie_interaction_desc
 is_shown = { scope:actor = { WJ_zw_valid = yes } WJ_dd_manage_visible_trigger = yes }
 is_valid_showing_failures_only = { scope:actor = { WJ_zw_valid = yes } WJ_dd_manage_trigger = yes }
 send_options_exclusive = yes
 send_option = { flag = wj_dd_jiajie localization = WJ_dd_choose_jiajie starts_enabled = { always = no } }
 send_option = { flag = wj_dd_chijie localization = WJ_dd_choose_chijie starts_enabled = { always = no } }
 send_option = { flag = wj_dd_shichijie localization = WJ_dd_choose_shichijie starts_enabled = { always = no } }
 can_send = {
  scope:actor = { WJ_zw_valid = yes } WJ_dd_manage_trigger = yes
  custom_tooltip = { text = WJ_dd_choose_jie_tt OR = { scope:wj_dd_jiajie = yes scope:wj_dd_chijie = yes scope:wj_dd_shichijie = yes } }
 }
 on_accept = {
  hidden_effect = {
   if = { limit = { scope:actor = { WJ_zw_valid = yes } WJ_dd_manage_trigger = yes }
    scope:recipient = {
     if = { limit = { scope:wj_dd_jiajie = yes } vassal_contract_set_obligation_level = { type = WJ_sanjie level = 1 } }
     else_if = { limit = { scope:wj_dd_chijie = yes } vassal_contract_set_obligation_level = { type = WJ_sanjie level = 2 } }
     else_if = { limit = { scope:wj_dd_shichijie = yes } vassal_contract_set_obligation_level = { type = WJ_sanjie level = 3 } }
     WJ_dd_remember_jie_effect = yes WJ_dd_ensure_kaifu_effect = yes
    }
   }
  }
 }
 ai_potential = { always = no } ai_frequency = 12 ai_will_do = { base = 0 }
}
