# One new-game pass, after immediate startup and day-one qualification work.
on_game_start_after_lobby = {
	on_actions = { WJ_starting_vassal_entry_roads_queue }
}

WJ_starting_vassal_entry_roads_queue = {
	# Keep the delay inside this child so other startup handlers are unaffected.
	on_actions = {
		delay = { days = 2 }
		WJ_starting_vassal_entry_roads_apply
	}
}

WJ_starting_vassal_entry_roads_apply = {
	trigger = {
		NOT = { exists = global_var:wj_starting_vassal_entry_road_stage }
	}
	effect = {
		set_global_variable = { name = wj_starting_vassal_entry_road_stage value = 2 }
		WJ_grant_starting_vassal_entry_roads_effect = yes
	}
}
