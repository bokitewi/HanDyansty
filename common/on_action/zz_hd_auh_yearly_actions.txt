hd_auh_dynastic_cycle_yearly_on_action = {
	effect = {
		# 无主状态须经独立观察器持续确认，不在年度回调直接改时代。
		if = {
			limit = {
				exists = situation:dynastic_cycle
			}
			situation:dynastic_cycle = {
				hd_dc_request_observation_effect = yes
			}
		}

	}
}

kuzhu_warlord_on_yearly_check = {
	effect = {
		if = {
			limit = {
				highest_held_title_tier >= tier_duchy
				has_realm_law = kuzhu_warlord_elective_succession_law
			}
			every_held_title = {
				limit = {
					tier >= tier_duchy
					is_titular = no
					NOT = { has_title_law = kuzhu_warlord_appointment_title_law }
				}
				add_title_law = kuzhu_warlord_appointment_title_law
			}
		}
		else_if = {
			limit = {
				highest_held_title_tier >= tier_duchy
				NOT = { has_realm_law = kuzhu_warlord_elective_succession_law }
			}
			every_held_title = {
				limit = {
					tier >= tier_duchy
					has_title_law = kuzhu_warlord_appointment_title_law
				}
				remove_title_law = kuzhu_warlord_appointment_title_law
			}
		}
	}
}

hd_auh_puppet_diarchy_cleanup = {
	effect = {
		if = {
			limit = { has_trait = kuzhu_puppet_ruler }
			remove_trait = kuzhu_puppet_ruler
		}
		scope:old_diarch ?= {
			if = {
				limit = { has_trait = kuzhu_empire_overseer }
				remove_trait = kuzhu_empire_overseer
			}
		}
	}
}
