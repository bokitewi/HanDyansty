# 郡国制·军阀契约（2026-10-24）

on_title_gain = { on_actions = { hd_warlord_on_title_gain } }
quarterly_playable_pulse = { on_actions = { hd_warlord_quarterly_pulse } }

# root = 新持有者；scope:title；scope:transfer_type
# 只有经欢呼继承接位（inheritance / appointment_succession）才保留军阀身份；
# 被原版「剥夺头衔」、改封、征服、篡夺等方式易手，该头衔即不再是军阀（用户拍板剥夺头衔照原版）
hd_warlord_on_title_gain = {
	trigger = {
		OR = {
			scope:title = { has_variable = hd_warlord }
			any_held_title = { has_variable = hd_warlord }
		}
	}
	effect = {
		if = {
			limit = {
				scope:title = { has_variable = hd_warlord }
				NOR = {
					scope:transfer_type = flag:inheritance
					scope:transfer_type = flag:appointment_succession
				}
			}
			scope:title = { remove_variable = hd_warlord }
		}
		hd_warlord_reconcile_effect = yes
	}
}

hd_warlord_quarterly_pulse = {
	trigger = { hd_warlord_trigger = yes }
	effect = {
		hd_warlord_reconcile_effect = yes
		hd_warlord_quarterly_treasury_effect = yes
	}
}
