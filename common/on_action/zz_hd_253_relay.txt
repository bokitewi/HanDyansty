on_game_start = {
	on_actions = {
		hd_sg_ptr_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd253_relay_start
	}
}

# 三国三个开局：开局当天先刷人物指针（本文件名排在 253、263 的 startup 之前，开局效果用得到）
hd_sg_ptr_on_game_start = {
	trigger = {
		OR = {
			game_start_date = 218.1.1
			game_start_date = 253.3.1
			game_start_date = 263.9.20
		}
	}
	effect = {
		hd_sg_ptr_refresh_effect = yes
	}
}

hd253_relay_start = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		game_start_date = 218.1.1
		NOT = { exists = global_var:hd253_relay_watch_started }
	}
	effect = {
		set_global_variable = hd253_relay_watch_started
		title:h_china = { trigger_event = { on_action = hd253_relay_watch days = 30 } }
	}
}

# 每 30 天看一次：到 253.3.1 启动接力
hd253_relay_watch = {
	trigger = {
		game_start_date = 218.1.1
		NOT = { exists = global_var:hd253_relay_on }
	}
	effect = {
		if = {
			limit = { current_date >= 253.3.1 }
			hd253_relay_activate_effect = yes
		}
		else_if = {
			limit = { current_date < 270.1.1 }
			title:h_china = { trigger_event = { on_action = hd253_relay_watch days = 30 } }
		}
	}
}
