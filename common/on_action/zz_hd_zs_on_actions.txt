# 诸葛亮、司马懿事件链（boki 2026-10-31，设计见 docs/诸葛亮司马懿事件链_设计草案.md）入口。
# 218 开局 after_lobby 启动；218 年以前的开局在岁首查二人是否已入刘氏、曹氏麾下，入场即启动。启动后每 30 天心跳，到 260.1.1 停。

on_game_start_after_lobby = {
	on_actions = {
		hd_zs_start_218
	}
}

yearly_global_pulse = {
	on_actions = {
		hd_zs_entry_check
	}
}

on_death = {
	on_actions = {
		hd_zs_on_death
	}
}

hd_zs_start_218 = {
	trigger = {
		game_start_date = 218.1.1
		hd_history_events_enabled_trigger = yes
		NOT = { exists = global_var:hd_zs_on }
	}
	effect = {
		hd_zs_activate_effect = yes
	}
}

hd_zs_entry_check = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		NOT = { exists = global_var:hd_zs_on }
		game_start_date < 218.1.1
		current_date < 250.1.1
		OR = {
			character:zhu_ge_liang ?= { is_alive = yes is_adult = yes hd_zs_serves_liu_trigger = yes }
			character:si_ma_yi ?= { is_alive = yes is_adult = yes hd_zs_serves_cao_trigger = yes }
		}
	}
	effect = {
		hd_zs_activate_effect = yes
	}
}

hd_zs_pulse = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		exists = global_var:hd_zs_on
	}
	effect = {
		hd_zs_pulse_effect = yes
		if = {
			limit = { current_date < 260.1.1 }
			title:h_china = { trigger_event = { on_action = hd_zs_pulse days = 30 } }
		}
	}
}

# 诸葛亮、曹丕、曹叡之死只设标记，下次心跳接后续。Root：死者（on_death 在死亡结算之前）
hd_zs_on_death = {
	trigger = {
		exists = global_var:hd_zs_on
		OR = {
			AND = { exists = global_var:hd_zs_p_zhu_ge_liang this = global_var:hd_zs_p_zhu_ge_liang }
			AND = { exists = global_var:hd_zs_p_cao_pi this = global_var:hd_zs_p_cao_pi }
			AND = { exists = global_var:hd_zs_p_cao_rui this = global_var:hd_zs_p_cao_rui }
		}
	}
	effect = {
		if = {
			limit = { exists = global_var:hd_zs_p_zhu_ge_liang this = global_var:hd_zs_p_zhu_ge_liang }
			set_global_variable = hd_zgl_dead
			hd_zs_save_lords_effect = yes
			if = {
				limit = { hd_zs_beifa_active_trigger = yes }
				set_global_variable = hd_zgl_died_campaign
			}
			debug_log = "HD_ZGL DIED"
		}
		else_if = {
			limit = { exists = global_var:hd_zs_p_cao_pi this = global_var:hd_zs_p_cao_pi }
			set_global_variable = hd_smy_pi_dead
			debug_log = "HD_SMY CAO_PI DIED"
		}
		else_if = {
			limit = { exists = global_var:hd_zs_p_cao_rui this = global_var:hd_zs_p_cao_rui }
			set_global_variable = hd_smy_rui_dead
			debug_log = "HD_SMY CAO_RUI DIED"
		}
	}
}
