# 开局无地世族家主（郡国制世族）宫廷里被引擎塞满的无地史实人物 -> 放回人才池
# 现象：鲁肃开局宫廷76名廷臣，全是无头衔无封地的历史人物（引擎把无雇主人物统一塞进同一个无地统治者宫廷）。
# 做法：对所有无地统治者，把与其无亲缘、非同家族、无头衔的史实廷臣移入人才池，之后由正常雇佣逻辑招募。
on_game_start = {
	on_actions = {
		hd_landless_courtier_pool_startup
	}
}

hd_landless_courtier_pool_startup = {
	trigger = {
		NOT = { exists = global_var:hd_landless_courtier_pool_done }
	}
	effect = {
		set_global_variable = hd_landless_courtier_pool_done
		every_ruler = {
			limit = {
				is_landed = no
				any_courtier = { count >= 8 }
			}
			save_scope_as = hd_pool_host
			every_courtier = {
				limit = {
					is_ruler = no
					is_landed = no
					has_trait = historical_character
					NOT = { is_close_or_extended_family_of = scope:hd_pool_host }
					NOT = { house ?= scope:hd_pool_host.house }
					NOT = { is_councillor = yes }
				}
				move_to_pool = yes
			}
		}
	}
}
