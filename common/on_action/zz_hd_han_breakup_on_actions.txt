# Script-triggered callbacks only. No pulse, every_ruler or recurring scan.
hd_hb_recheck_after_three_days = {
	effect = { hd_hb_recheck_duchies_effect = yes }
}

hd_hb_legacy_after_thirty_days = {
	effect = {
		if = {
			limit = { NOT = { exists = global_var:hd_hb_d30_done } }
			set_global_variable = hd_hb_d30_done
			# Preserve this public D30 evidence at D30, not at D0 or D3.
			set_global_variable = hd_qxzl_scan_done
			hd_lb_gongsun_d30_effect = yes
			# Fallback for the historical warlords pass; hd_hw_run_effect itself runs only once.
			if = {
				limit = { exists = global_var:hd_hb_spatial_done }
				hd_hw_run_effect = yes
			}
		}
	}
}
