# 衣冠南渡 on_action（见 docs/衣冠南渡_设计草案.md）

# 触发检测：每月，挂在天朝循环局势月度脉冲上（只追加，不改原定义）
dynastic_cycle_monthly_pulse = {
	on_actions = { hd_yiguan_monthly_check }
}

hd_yiguan_monthly_check = {
	effect = {
		hd_yiguan_try_start_effect = yes
	}
}

# 年度：无天朝循环局势时兜底检测；侨郡与土断善后
yearly_global_pulse = {
	on_actions = { hd_yiguan_yearly_pulse }
}

hd_yiguan_yearly_pulse = {
	effect = {
		if = {
			limit = { NOT = { exists = situation:dynastic_cycle } }
			hd_yiguan_try_start_effect = yes
		}
		# 局势结束后 10 年内，持有侨郡的宗主仍可能收到土断事件
		if = {
			limit = {
				exists = global_var:hd_yiguan_outcome
				exists = global_var:hd_yiguan_tuduan_window
			}
			every_in_global_list = {
				variable = hd_yiguan_qiaojun_list
				limit = { has_county_modifier = hd_yiguan_qiaojun_county_modifier }
				holder ?= {
					top_liege = {
						if = {
							limit = { NOT = { is_in_list = hd_yiguan_tuduan_lords } }
							add_to_list = hd_yiguan_tuduan_lords
						}
					}
				}
			}
			every_in_list = {
				list = hd_yiguan_tuduan_lords
				limit = {
					NOT = { has_variable = hd_yiguan_event_cd }
					is_alive = yes
				}
				random = {
					chance = 25
					trigger_event = hd_yiguan.1210
				}
			}
		}
		hd_yiguan_aftermath_yearly_effect = yes
	}
}
