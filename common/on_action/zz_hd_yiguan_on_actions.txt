############################################################
# 开局检测：每月，挂在天朝循环局势月度脉冲上
############################################################
dynastic_cycle_monthly_pulse = {
	on_actions = { hd_yiguan_monthly_check }
}

hd_yiguan_monthly_check = {
	effect = {
		hd_yiguan_try_start_effect = yes
	}
}

############################################################
# 年度全局脉冲：无天朝循环时兜底检测开局；200 年兜底结局；AI 北伐/讨胡宣战；结局后土断善后
############################################################
yearly_global_pulse = {
	on_actions = { hd_yiguan_yearly_pulse }
}

hd_yiguan_yearly_pulse = {
	effect = {
		if = {
			limit = { NOT = { exists = situation:dynastic_cycle } }
			hd_yiguan_try_start_effect = yes
		}
		hd_yiguan_fallback_check_effect = yes
		if = {
			limit = {
				exists = situation:hd_yiguan_nandu
				NOT = { exists = global_var:hd_yiguan_outcome }	# 兜底刚结局、局势尚未销毁时不再宣战
			}
			hd_yiguan_ai_war_pulse_effect = yes	# 战争侧定义（zz_hd_yiguan_war_effects.txt）
		}
		hd_yiguan_aftermath_yearly_effect = yes
	}
}

# 宣战：胡人政权向汉人政权宣战（原版 war_on_actions.txt:309-312：对所有 CB 触发）
on_war_started = {
	on_actions = { hd_yiguan_cat_war_started }
}

hd_yiguan_cat_war_started = {
	trigger = {
		exists = situation:hd_yiguan_nandu
		exists = scope:attacker
		exists = scope:defender
	}
	effect = {
		hd_yiguan_hook_war_started_effect = yes
	}
}

# 攻方胜（原版 war_on_actions.txt:1259：两个 won 只触发一个）
on_war_won_attacker = {
	on_actions = { hd_yiguan_cat_won_attacker }
}

hd_yiguan_cat_won_attacker = {
	trigger = {
		exists = situation:hd_yiguan_nandu
		exists = scope:attacker
		exists = scope:defender
	}
	effect = {
		hd_yiguan_hook_war_ended_effect = { WINNER = scope:attacker LOSER = scope:defender }
	}
}
