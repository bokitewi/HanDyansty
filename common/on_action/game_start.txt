# ============================================================================
# WJ 三条入仕途径开局授予 —— 同名文件替换原版/主mod的 game_start.txt
# xinsanguo-MOD 2026-08-19
#
# 背景(已核实):
# 1) 原版 game_start.txt 里的"考试通过授予"段落, 在装主mod后不生效,
#    原因 = 主mod 的 common/on_action/game_start.txt 同名文件把原版整个文件替换掉了
#    (主mod该文件只有3行注释: "Deliberate compatibility override"),
#    所以原版 on_game_start 里的 every_courtier 授予代码不存在于主mod环境。
# 2) 键名 = on_game_start(原版 game_start.txt L2; 主mod的zz_saas/WJ_east文件
#    也用它注册启动子动作 = 该键在主mod环境确实被引擎调度; 同名键跨文件合并,
#    我们的效果与主mod的启动子动作并存)。
# 3) 效果: 天朝皇帝(h_china持有者)的所有廷臣等概率随机授予辟召、举孝廉或举茂才，
#    替代原版"乡试+会试"双送；无入仕途径者不再由开局脚本批量排除。
# 4) 举茂才活动虽已退役，但举茂才入仕设定保留，开局与另两条入仕途径等概率生成。
# ============================================================================

on_game_start = {
	effect = {
		### 皇帝廷臣随机取得一种开局入仕资格 ###
			if = {
				limit = {
					exists = title:h_china.holder
					title:h_china.holder = { WJ_han_jin_law_trigger = yes }
				}

				title:h_china.holder = {
				every_courtier = {
					hidden_effect = {
						WJ_grant_random_starting_road_effect = yes
					}
				}

				### 无地朝臣：原版无地官职头衔持有者是皇帝封臣，不属于 every_courtier/every_ruler。 ###
					every_vassal = {
						limit = {
							is_landless_ruler = yes
							government_has_flag = government_is_celestial
							WJ_has_substantive_office_trigger = yes
							WJ_has_entry_road_trigger = no
						}
					hidden_effect = {
						WJ_grant_random_starting_road_effect = yes
					}
				}
			}

			### 有地官员：原版 every_ruler 单池统一覆盖同一天朝伯爵及以上官员。 ###
			every_ruler = {
				limit = {
					is_landed = yes
					highest_held_title_tier >= tier_county
					top_liege = title:h_china.holder
					government_has_flag = government_is_celestial
				}
				hidden_effect = {
					WJ_grant_random_starting_road_effect = yes
				}
			}
		}
	}
}
