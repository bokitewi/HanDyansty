# ============================================================================
# WJ 三条入仕途径开局授予 —— 同名文件替换原版/主mod的 game_start.txt
# xinsanguo-MOD 2026-08-19
#
# 背景(已核实):
# 1) 原版 game_start.txt 里的"考试通过授予"段落, 在装主mod后不生效,
#    原因 = 主mod 的 common/on_action/game_start.txt 同名文件把原版整个文件替换掉了
#    (主mod该文件只有3行注释: "Deliberate compatibility override"),
#    所以原版 on_game_start 里的 every_courtier 授予代码不存在于主mod环境。
# 2) 键名 = on_game_start(原版 game_start.txt L2; 主mod的zz_saas/WJ_east文件
#    也用它注册启动子动作 = 该键在主mod环境确实被引擎调度; 同名键跨文件合并,
#    我们的效果与主mod的启动子动作并存)。
# 3) 效果: 只发给 WJ_starting_career_eligible 名单（与宦海沉浮同一批）：h_china 各层级天朝有实职封臣，
#    以及皇帝/天朝封臣的有内阁或宫廷职位廷臣；皇帝本人与无职廷臣不发。等概率辟召、举孝廉或举茂才。
# 4) 举茂才活动虽已退役，但举茂才入仕设定保留，开局与另两条入仕途径等概率生成。
# ============================================================================

on_game_start = {
	effect = {
		### h_china 体系开局入仕资格：与宦海沉浮共用 WJ_starting_career_eligible 名单 ###
		if = {
			limit = {
				NOT = { game_start_date = 190.1.1 } # 190 runs once after its final historical assignments.
				exists = title:h_china.holder
				title:h_china.holder = { government_has_flag = government_is_celestial }
			}
			title:h_china.holder = {
				every_courtier = {
					limit = { WJ_starting_career_eligible = yes }
					hidden_effect = { WJ_grant_random_starting_road_effect = yes }
				}
				every_vassal_or_below = {
					if = {
						limit = { WJ_starting_career_eligible = yes }
						hidden_effect = { WJ_grant_random_starting_road_effect = yes }
					}
					every_courtier = {
						limit = { WJ_starting_career_eligible = yes }
						hidden_effect = { WJ_grant_random_starting_road_effect = yes }
					}
				}
			}
		}
	}
}

# Native 1.21 merchant setup only. The custom map keeps its own other startup paths.
on_game_start_after_lobby = {
 effect = {
  every_independent_ruler = {
   limit = { government_has_flag = government_is_merchant }
if = {
						limit = { gold > 0 }
						remove_short_term_gold = gold
					}
add_gold = 50
if = {
						limit = { prestige > 25 }
						add_prestige = {
							value = 25
							subtract = prestige
						}
					}
save_temporary_scope_as = merchant_scope
primary_title = {
						set_coa = scope:merchant_scope.house
						set_variable = { name = merchant_creation_reason value = flag:historical }
						set_variable = gdp_starting_merchant_company
					}
random_held_title = {
						limit = { is_noble_family_title = yes }
						set_coa = holder.top_liege.house
					}
add_to_global_variable_list = {
						name = merchants_tally
						target = this.primary_title
					}
if = {
						limit = {
							is_from_ruler_designer = yes
							exists = location.barony.trade_market.trade_market_owner.title_province
						}
						location.barony.trade_market.trade_market_owner.title_province = { save_temporary_scope_as = merchant_market_capital }
						random_held_title = {
							limit = { is_noble_family_title = yes }
							set_capital_county = scope:merchant_market_capital.county
						}
						domicile ?= { move_domicile = scope:merchant_market_capital }
						set_location = scope:merchant_market_capital
					}
start_tributary_interaction_effect = {
						TRIBUTARY = this
						SUZERAIN = domicile.domicile_location.county.holder # county holder
					}
if = {
						limit = {
							is_ai = no
						}
						add_character_flag = free_starting_merchant_purpose_change
						if = {
							limit = {
								OR = {
									has_realm_law = company_purpose_artisan
									has_realm_law = company_purpose_caravaneer
									has_realm_law = company_purpose_creditor
								}
							}
							add_realm_law_skip_effects = company_purpose_family
							merchant_company_law_remove_purpose_effect = yes
						}
						if = {
							limit = {
								OR = {
									has_realm_law = company_size_medium
									has_realm_law = company_size_large
									has_realm_law = company_size_huge
								}
							}
							add_realm_law_skip_effects = company_size_small
						}
					}
if = {
						limit = {
							is_ai = yes
							NOR = {
								has_realm_law = company_purpose_artisan
								has_realm_law = company_purpose_caravaneer
								has_realm_law = company_purpose_creditor
							}
						}
						# Set some starting Company purposes
						switch = {
							trigger = highest_skill
							learning = { add_realm_law_skip_effects = company_purpose_artisan }
							diplomacy = { add_realm_law_skip_effects = company_purpose_caravaneer }
							martial = { add_realm_law_skip_effects = company_purpose_creditor }
							intrigue = { add_realm_law_skip_effects = company_purpose_creditor }
						}
						# Courtiers.
						trigger_event = gdp_merchant.1001
					}
gdp_update_usury_reputation_effect = yes
trigger_event = gdp_merchant.1003
if = {
						limit = { is_ai = yes }
						set_starting_company_size_effect = yes
						set_starting_company_acumen_effect = yes
					}
  }
 }
}
