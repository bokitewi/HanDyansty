# Native on_death root = dead character; additive child on_action.
on_death = { on_actions = { WJ_mufu_death } }
WJ_mufu_death = {
 trigger = {
  OR = { has_variable = wj_mufu_owner has_variable = wj_mufu_slot_1 has_variable = wj_mufu_slot_2 has_variable = wj_mufu_slot_3 has_variable = wj_mufu_slot_4 has_variable = wj_mufu_slot_5 has_variable = wj_yangwang_mufu_pending has_variable = wj_mufu_pending_owner }
 }
 effect = {
  WJ_mufu_cancel_annual_for_character = yes
  WJ_mufu_leave = yes
  WJ_mufu_dissolve = yes
 }
}

# Delay title/court transitions until the transaction has completed, avoiding false
# dissolution between two valid minister appointments. No AI departure roll here.
on_title_gain = { on_actions = { WJ_mufu_changed } }
on_title_lost = { on_actions = { WJ_mufu_changed } }
on_join_court = { on_actions = { WJ_mufu_changed } }
WJ_mufu_changed = {
 trigger = { OR = { has_variable = wj_mufu_owner has_variable = wj_mufu_slot_1 has_variable = wj_mufu_slot_2 has_variable = wj_mufu_slot_3 has_variable = wj_mufu_slot_4 has_variable = wj_mufu_slot_5 } }
 effect = { trigger_event = { id = WJ_mufu.0002 days = 1 } }
}
