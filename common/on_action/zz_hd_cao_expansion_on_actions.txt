on_death = { on_actions = { hd_cao_expansion_on_death } }

hd_cao_expansion_on_death = {
	trigger = { this = character:cao_cao }
	effect = { hd_cao_expansion_owner_death_effect = yes }
}

on_game_start_after_lobby = {
	on_actions = { hd_cao_expansion_after_lobby }
}

hd_cao_expansion_after_lobby = {
	trigger = { game_start_date = 184.1.1 }
	effect = {
		character:cao_cao = {
			if = {
				limit = {
					is_alive = yes
					NOT = { any_owned_story = { story_type = hd_story_historical_ai_controller } }
				}
				create_story = hd_story_historical_ai_controller
			}
		}
	}
}
