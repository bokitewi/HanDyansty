quarterly_playable_pulse = { on_actions = { WJ_pr_quarter } }
on_death = { on_actions = { WJ_pr_person_death } }
on_diarch_change = { on_actions = { WJ_pr_regent_change } }
on_title_lost = { on_actions = { WJ_pr_throne_change } }
WJ_pr_quarter = {
 trigger = { is_ai = no has_title = title:h_china }
 effect = {
  if = { limit = { exists = var:wj_pr_petitioner }
   var:wj_pr_petitioner = {
    if = { limit = { OR = { is_alive = no NOT = { is_diarch_of_target = root } top_liege != root } } WJ_pr_end = yes }
   }
  }
  diarch ?= { if = { limit = { is_ai = yes } trigger_event = WJ_regent_petition.100 } }
 }
}
WJ_pr_person_death = {
 trigger = { OR = { has_variable = wj_pr_emperor has_variable = wj_pr_petitioner } }
 effect = { var:wj_pr_petitioner ?= { WJ_pr_end = yes } WJ_pr_end = yes }
}
WJ_pr_regent_change = {
 trigger = { has_variable = wj_pr_petitioner }
 effect = {
  var:wj_pr_petitioner = { if = { limit = { NOT = { is_diarch_of_target = root } } WJ_pr_end = yes } }
 }
}
WJ_pr_throne_change = {
 trigger = { scope:title = title:h_china exists = var:wj_pr_petitioner }
 effect = { var:wj_pr_petitioner = { WJ_pr_end = yes } }
}
