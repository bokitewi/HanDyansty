# 农民义军：开战时与每年扫描，为有农民领袖特质的统治者开启故事

on_war_started = {
	on_actions = { hd_peasant_army_on_war_started }
}
hd_peasant_army_on_war_started = {
	effect = {
		scope:war ?= {
			every_war_participant = {
				limit = { has_trait = peasant_leader }
				hd_peasant_army_try_start_effect = yes
			}
		}
	}
}

yearly_global_pulse = {
	on_actions = { hd_peasant_army_yearly_scan }
}
hd_peasant_army_yearly_scan = {
	effect = {
		every_ruler = {
			limit = { has_trait = peasant_leader }
			hd_peasant_army_try_start_effect = yes
		}
	}
}
