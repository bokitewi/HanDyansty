# 晋制时间条件设置
on_game_start = {
    on_actions = {
        jin_time_setup_action
    }
}

jin_time_setup_action = {
    effect = {
        # 官制：汉制（280年之后）
        if = {
            limit = {
                current_date >= 184.1.1
               # current_date < 420.1.1  # 晋朝时间范围：280-420年
            }
            title:h_china.holder = {
                add_realm_law_skip_effects = han_province_title_law
                #add_realm_law_skip_effects = jin_councillor_title_law
                every_vassal_or_below = {
                    limit = {
                        highest_held_title_tier >= tier_county
                    }
                    add_realm_law_skip_effects = han_province_title_law
                    #add_realm_law_skip_effects = jin_councillor_title_law
                }
            }
        }
    }
}
