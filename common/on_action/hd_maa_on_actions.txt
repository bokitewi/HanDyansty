# ==========================================================================
# HD 特色兵种系统 on_action 挂钩（阶段0-4 技术原型）
# 依据 _on_actions.info：全部通过子 on_action 追加到原版 on_action，
# 不复制、不覆盖任何原版 on_action 文件的 trigger/effect 块。
# ==========================================================================

on_title_gain = {
	on_actions = { hd_maa_on_title_gain }
}

on_title_gain_inheritance = {
	on_actions = { hd_maa_on_title_gain_inheritance }
}

on_title_lost = {
	on_actions = { hd_maa_on_title_lost }
}

on_death = {
	on_actions = { hd_maa_on_death }
}

on_character_culture_change = {
	on_actions = { hd_maa_on_character_culture_change }
}

on_county_culture_change = {
	on_actions = { hd_maa_on_county_culture_change }
}

quarterly_playable_pulse = {
	on_actions = { hd_maa_quarterly_refresh }
}

on_game_start_after_lobby = {
	on_actions = { hd_maa_game_start_refresh }
}

# --- 头衔获得/失去/继承：即时刷新相关统治者 ---
hd_maa_on_title_gain = {
	trigger = { root = { hd_has_any_hd_regional_maa_trigger = yes } }
	effect = { root = { hd_refresh_all_hd_maa_sources_effect = yes } }
}

hd_maa_on_title_gain_inheritance = {
	trigger = { root = { hd_has_any_hd_regional_maa_trigger = yes } }
	effect = { root = { hd_refresh_all_hd_maa_sources_effect = yes } }
}

hd_maa_on_title_lost = {
	trigger = { root = { hd_has_any_hd_regional_maa_trigger = yes } }
	effect = { root = { hd_refresh_all_hd_maa_sources_effect = yes } }
}

# --- 死亡：军制只传主继承人，并立即重算其兵源 ---
hd_maa_on_death = {
	trigger = { root = { hd_has_any_hd_regional_maa_trigger = yes } }
	effect = { root = { hd_inherit_hd_maa_effect = yes } }
}

# --- 统治者文化变化：重算兵源（同文化判定依赖统治者文化） ---
hd_maa_on_character_culture_change = {
	trigger = { root = { hd_has_any_hd_regional_maa_trigger = yes } }
	effect = { root = { hd_refresh_all_hd_maa_sources_effect = yes } }
}

# --- 县文化变化：重算该县领主顶级统治者的兵源（root = 县头衔） ---
hd_maa_on_county_culture_change = {
	trigger = {
		root = {
			exists = holder
			holder = { exists = top_liege }
		}
	}
	effect = {
		root = {
			holder = {
				top_liege = { hd_refresh_all_hd_maa_sources_effect = yes }
			}
		}
	}
}

# --- 季度容错刷新：捕捉控制力/接受度漂移等不伴随头衔变化的状态 ---
hd_maa_quarterly_refresh = {
	trigger = { root = { hd_has_any_hd_regional_maa_trigger = yes } }
	effect = { root = { hd_refresh_all_hd_maa_sources_effect = yes } }
}

# --- 新游戏/读档初始化：对已建立军制者重算兵源（幂等） ---
# 只遍历独立统治者（every_playable_character 在 1.19 不是有效效果）；
# 持有军制的封臣由季度脉冲覆盖。
hd_maa_game_start_refresh = {
	effect = {
		every_independent_ruler = {
			limit = { hd_has_any_hd_regional_maa_trigger = yes }
			hd_refresh_all_hd_maa_sources_effect = yes
		}
	}
}
