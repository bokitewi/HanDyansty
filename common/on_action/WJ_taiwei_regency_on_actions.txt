# Replaces the old Taiwei-only title-gain hook. Title gain is handled in title_on_actions.
on_game_start_after_lobby = { on_actions = { WJ_grand_secretariat_game_start } }
WJ_grand_secretariat_game_start = {
 effect = { title:h_china.holder ?= { WJ_sync_grand_secretariat_effect = yes } }
}

# One constant-time check on the hegemon each quarter; no global office scan.
quarterly_playable_pulse = { on_actions = { WJ_chengxiang_vacancy_check } }
WJ_chengxiang_vacancy_check = {
 trigger = {
  has_title = title:h_china
  government_has_flag = government_is_celestial
  has_realm_law = WJ_grand_chengxiang_law
  NOT = { exists = title:e_WJ_minister_cheng_xiang.holder }
 }
 effect = {
  add_realm_law_skip_effects = grand_marshal_law
  WJ_sync_grand_secretariat_effect = yes
 }
}
