# 617 年剧本「晋阳起兵」史实主线入口（/mnt/project-files/617隋末群雄剧本_计划.md 第 6.4、10、13 节）。全部带 617.6.23 闸门，其余开局不受影响。
# 效果在 common/scripted_effects/zz_hd_617_history_effects.txt，事件在 events/zz_hd_617_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_617_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hd_22_pulse），到 630 年为止
hd_617_history_start = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		game_start_date = 617.6.23
		NOT = { exists = global_var:hd_617_pulse_started }
	}
	effect = {
		set_global_variable = hd_617_pulse_started
		title:h_china = { trigger_event = { on_action = hd_617_pulse days = 1 } }
	}
}

hd_617_pulse = {
	trigger = {
		game_start_date = 617.6.23
	}
	effect = {
		hd_617_pulse_effect = yes
		if = {
			limit = { current_date < 630.1.1 }
			title:h_china = { trigger_event = { on_action = hd_617_pulse days = 30 } }
		}
	}
}
