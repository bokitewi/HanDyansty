# D11 定德运
############################################################
on_title_gain = { on_actions = { hd_hanfl_dewu_title_gain } }

# root = 新持有人；新旧持有人王朝不同才算改朝（受禅用 type = created，不按转移类型判断）
# 前朝德运在得头衔当下就取好（受禅时可能没有 previous_holder，7 天后天子已是自己）：
# 先看前持有人王朝，再看当时天子的王朝，得天下头衔时最后看 global_var:hd_hanfl_last_tianzi_dewu（上一位天子王朝的德运）
hd_hanfl_dewu_title_gain = {
	trigger = {
		exists = global_var:hd_hanfl_dewu_armed
		exists = scope:title
		government_has_flag = government_is_celestial
		exists = dynasty
		NOT = { dynasty = { has_variable = hd_hanfl_dewu } }
		# 同时得天下和地方帝国头衔时只排一次
		NOT = { has_character_flag = hd_hanfl_dewu_scheduled }
		scope:title = {
			OR = {
				this = title:h_china
				WJ_cr_local_empire_title = yes
			}
		}
		OR = {
			NOT = { exists = scope:previous_holder }
			NOT = { exists = scope:previous_holder.dynasty }
			NOT = { scope:previous_holder.dynasty = root.dynasty }
		}
	}
	effect = {
		add_character_flag = { flag = hd_hanfl_dewu_scheduled days = 30 }
		scope:title = { save_scope_as = hd_hanfl_dewu_title }
		scope:previous_holder ?= {
			dynasty ?= {
				if = {
					limit = {
						has_variable = hd_hanfl_dewu
						NOT = { this = root.dynasty }
					}
					save_scope_value_as = { name = hd_hanfl_dewu_prev value = var:hd_hanfl_dewu }
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_hanfl_dewu_prev } }
			title:h_china.holder ?= {
				dynasty ?= {
					if = {
						limit = {
							has_variable = hd_hanfl_dewu
							NOT = { this = root.dynasty }
						}
						save_scope_value_as = { name = hd_hanfl_dewu_prev value = var:hd_hanfl_dewu }
					}
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:hd_hanfl_dewu_prev }
				scope:title = title:h_china
				exists = global_var:hd_hanfl_last_tianzi_dewu
			}
			save_scope_value_as = { name = hd_hanfl_dewu_prev value = global_var:hd_hanfl_last_tianzi_dewu }
		}
		if = {
			limit = { exists = scope:hd_hanfl_dewu_prev }
			debug_log = "HD_HANFL DEWU SCHEDULED WITH PREV"
		}
		else = {
			debug_log = "HD_HANFL DEWU SCHEDULED NO PREV"
		}
		trigger_event = { id = hd_hanfl.1301 days = 7 }
	}
}
