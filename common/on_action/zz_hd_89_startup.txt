# 89 年剧本「燕然勒石」（89.1.30，永元元年正月朔）开局。计划见 /mnt/project-files/89窦宪北伐剧本_计划.md 第 6、13 节。
# 全部以 game_start_date = 89.1.30 为闸门，其余开局不受影响。效果写在 common/scripted_effects/zz_hd_89_startup_effects.txt。
# 去重、政体、天朝循环、国号在 on_game_start 里做（大厅前就生效）；朝官、宫廷、朝贡、年号、停战放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_89_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_89_after_lobby
	}
}

hd_89_on_game_start = {
	trigger = {
		game_start_date = 89.1.30
		current_date = 89.1.30
		NOT = { exists = global_var:hd_89_startup_done }
	}
	effect = {
		set_global_variable = hd_89_startup_done
		hd_89_game_start_effect = yes
	}
}

hd_89_after_lobby = {
	trigger = {
		game_start_date = 89.1.30
		current_date = 89.1.30
		NOT = { exists = global_var:hd_89_after_lobby_done }
	}
	effect = {
		hd_89_after_lobby_effect = yes
		set_global_variable = hd_89_after_lobby_done
	}
}

# 开局次日再清一次天朝循环进度（吃掉开局首日事件触发的催化剂），照 22。Situation scope
hd_89_dc_zero_progress = {
	trigger = {
		game_start_date = 89.1.30
		situation_current_phase = situation_dynastic_cycle_phase_stability_expansion
	}
	effect = {
		hd_89_zero_progress_effect = yes
	}
}
