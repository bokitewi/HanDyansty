# 253「姜维北伐」剧本开局（延熙十六年正月，费祎遇刺之后）。全部以 game_start_date = 253.3.1 为闸门，其他开局不受影响。
# 效果写在 common/scripted_effects/zz_hd_253_startup_effects.txt，计划见 docs/姜维剧本_计划.md。

on_game_start = {
	on_actions = {
		hd253_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd253_after_lobby
	}
}

on_death = {
	on_actions = {
		hd253_on_death
	}
}

hd253_on_game_start = {
	trigger = {
		game_start_date = 253.3.1
		current_date = 253.3.1
		NOT = { exists = global_var:hd253_game_start_ran }
	}
	effect = {
		set_global_variable = hd253_game_start_ran
		hd253_game_start_effect = yes
	}
}

hd253_after_lobby = {
	trigger = {
		game_start_date = 253.3.1
		NOT = { exists = global_var:hd253_after_lobby_done }
	}
	effect = {
		hd_sg_ptr_refresh_effect = yes
		set_global_variable = hd253_after_lobby_done
		hd253_after_lobby_effect = yes
		title:h_china = { trigger_event = { on_action = hd253_pulse days = 1 } }
	}
}

# 每 30 天一次，到 270 年为止：魏吴大事件链（10-18 加固 C：由 266 延到 270）。218 局接力见 zz_hd_253_relay.txt
hd253_pulse = {
	trigger = {
		hd253_chain_on_trigger = yes
	}
	effect = {
		hd_sg_ptr_refresh_effect = yes
		hd253_pulse_effect = yes
		if = {
			limit = { current_date < 270.1.1 }
			title:h_china = { trigger_event = { on_action = hd253_pulse days = 30 } }
		}
	}
}

# 司马师死后第 3 天，司马昭接任权臣（终身辅政锁随司马师之死清除，在此补上）。Root：死者
hd253_on_death = {
	trigger = {
		hd253_chain_on_trigger = yes
		this = character:si_ma_shi
		title:h_china.var:hd_court_regent ?= this # 死时仍是权臣才接任
	}
	effect = {
		title:h_china = { trigger_event = { on_action = hd253_zhao_regency days = 3 } }
	}
}

hd253_zhao_regency = {
	trigger = {
		hd253_chain_on_trigger = yes
	}
	effect = {
		hd253_zhao_regency_route_effect = yes
	}
}
