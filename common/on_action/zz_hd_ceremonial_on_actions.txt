# 天子礼仪君主（2026-10-24，历史志向任务卡 D48）。设计见 docs/历史志向任务卡_设计草案.md §3.1。

on_title_gain = {
	on_actions = { hd_cer_on_title_gain }
}

# 礼仪头衔易手（「另立天子」派系、夺位计谋等）：在世的旧礼仪君主得"前帝"特质，新礼仪君主去掉"前帝"。
# root = 新持有者；scope:title；scope:previous_holder
hd_cer_on_title_gain = {
	trigger = {
		scope:title.var:ceremonial_title ?= title:h_china
	}
	effect = {
		scope:previous_holder ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { this = root }
				}
				add_trait = former_emperor
			}
		}
		if = {
			limit = {
				has_trait = former_emperor
				NOT = { has_title = title:h_china }
			}
			remove_trait = former_emperor
		}
	}
}
