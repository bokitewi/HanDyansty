yearly_global_pulse = {
	on_actions = { hd_qin_lun_yearly_pulse }
}

hd_qin_lun_yearly_pulse = {
	trigger = { hd_history_events_enabled_trigger = yes }
	effect = {
		if = {
			limit = { NOT = { exists = global_var:hd_qin_lun_visited } }
			random_ruler = {
				limit = { hd_qin_lun_visit_trigger = yes }
				trigger_event = { id = hd_qin_lun.0001 days = { 5 60 } }
			}
		}
	}
}
