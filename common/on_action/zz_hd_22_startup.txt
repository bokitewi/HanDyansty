# 22 年剧本「舂陵起兵」（22.11.13，新地皇三年十一月朔）开局。计划见 /mnt/project-files/22舂陵起兵剧本_计划.md 第 6、13 节。
# 全部以 game_start_date = 22.11.13 为闸门，其余开局不受影响。效果写在 common/scripted_effects/zz_hd_22_startup_effects.txt。
# 去重、政体、天朝循环、国号在 on_game_start 里做（大厅前就生效）；宣战、年号、宫廷、朝官等放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_22_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_22_after_lobby
	}
}

hd_22_on_game_start = {
	trigger = {
		game_start_date = 22.11.13
		current_date = 22.11.13
		NOT = { exists = global_var:hd_22_startup_done }
	}
	effect = {
		set_global_variable = hd_22_startup_done
		hd_22_game_start_effect = yes
	}
}

hd_22_after_lobby = {
	trigger = {
		game_start_date = 22.11.13
		current_date = 22.11.13
		NOT = { exists = global_var:hd_22_after_lobby_done }
	}
	effect = {
		hd_22_after_lobby_effect = yes
		set_global_variable = hd_22_after_lobby_done
	}
}

# 开局次日再清一次天朝循环进度（吃掉开局首日事件触发的催化剂），照 saas 的 184 写法。Situation scope
hd_22_dc_zero_progress = {
	trigger = {
		game_start_date = 22.11.13
		situation_current_phase = situation_dynastic_cycle_phase_instability
	}
	effect = {
		hd_dc_zero_start_progress_effect = yes
	}
}
