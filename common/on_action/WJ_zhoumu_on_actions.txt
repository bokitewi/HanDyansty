# Append handlers; one-day deferral lets native title/vassal transfers settle.
on_title_gain = { on_actions = { WJ_zhoumu_identity_changed } }
on_title_lost = { on_actions = { WJ_zhoumu_identity_changed } }
on_vassal_change = { on_actions = { WJ_zhoumu_identity_changed } }
on_government_change = { on_actions = { WJ_zhoumu_identity_changed } }
on_liege_government_change = { on_actions = { WJ_zhoumu_identity_changed } }
on_realm_capital_change = { on_actions = { WJ_zhoumu_capital_changed } }
on_death = { on_actions = { WJ_zhoumu_member_died } }
quarterly_playable_pulse = { on_actions = { WJ_zhoumu_holder_check } }
WJ_zhoumu_identity_changed = {
 trigger = {
  is_alive = yes
  OR = { WJ_zhoumu_has_state = yes any_vassal_or_below = { WJ_zhoumu_has_state = yes } }
 }
 effect = { trigger_event = { on_action = WJ_zhoumu_after_change days = 1 } }
}
# Native on_realm_capital_change root is the new capital barony, not its holder.
WJ_zhoumu_capital_changed = {
 trigger = { holder ?= { WJ_inner_emperor_trigger = yes } }
 effect = { holder ?= { trigger_event = { on_action = WJ_zhoumu_after_change days = 1 } } }
}
WJ_zhoumu_after_change = {
 trigger = { is_alive = yes }
 effect = {
  if = {
   limit = { character:dong_zhuo_2 = { has_character_flag = hd_palace_sili_trace_active } }
   debug_log = "HD_HISTORY SILI_TRACE ZHOUMU_RECONCILE_BEGIN"
   debug_log_scopes = yes
  }
  WJ_zhoumu_reconcile_subrealm = yes
  if = {
   limit = { character:dong_zhuo_2 = { has_character_flag = hd_palace_sili_trace_active } }
   debug_log = "HD_HISTORY SILI_TRACE ZHOUMU_RECONCILE_DONE"
  }
 }
}
WJ_zhoumu_member_died = {
 trigger = { has_variable = wj_zhoumu_opinion_emperor }
 effect = { WJ_zhoumu_clear_opinion = yes }
}
# Native per-character pulse is a saved-game/direct-setter safety net, no world scan.
WJ_zhoumu_holder_check = {
 trigger = { WJ_zhoumu_has_state = yes }
 effect = { WJ_zhoumu_reconcile = yes }
}
