# SWORN BROTHERHOOD OATH-BREAKING TRIGGERS
# Fires when blood brother relation is removed
# 原版自动钩子名为 on_remove_relation_[relation]（root = 执行 remove_relation 的一方，scope:target = 另一方）。
# 原版该钩子已有 effect 块，不能再追加 effect，故按 _on_actions.info 的写法挂一个子 on_action。

on_remove_relation_blood_brother = {
    on_actions = { hd_sworn_brotherhood_oath_broken }
}

# 仅在真正背誓时触发：原版（及本 mod 覆盖的）拒绝应召参战分支会先给背誓者加
# broke_blood_brotherhood_modifier，再由背誓者执行 remove_relation_blood_brother。
# 结盟自然解除、PAM 交易、MPO 事件等非背誓的解除不带此修正，不触发。
hd_sworn_brotherhood_oath_broken = {
    trigger = {
        has_character_modifier = broke_blood_brotherhood_modifier
        exists = scope:target
        scope:target = { is_alive = yes }
    }
    effect = {
        save_scope_as = betrayer
        scope:target = {
            trigger_event = sworn_brotherhood_events.0003  # Fires for the betrayed character
        }
    }
}
