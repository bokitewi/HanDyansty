# Native title_on_actions: gain root = new holder; lost root = old holder;
# vassal change root = the changed vassal. ep3/mpo government hooks also use character root.
# Recheck the affected holder one day after the native transaction completes.
on_title_gain = { on_actions = { WJ_contract_privilege_changed } }
on_title_lost = { on_actions = { WJ_contract_privilege_changed } }
on_vassal_change = { on_actions = { WJ_contract_privilege_changed } }
on_government_change = { on_actions = { WJ_contract_privilege_changed } }
on_liege_government_change = { on_actions = { WJ_contract_privilege_changed } }

WJ_contract_privilege_changed = {
	trigger = {
		is_alive = yes
		OR = {
			WJ_contract_six_offices_trigger = yes
			WJ_has_contract_privilege_state_trigger = yes
		}
	}
	effect = {
		WJ_clear_invalid_maojie_effect = yes
		trigger_event = { on_action = WJ_contract_privilege_after_change days = 1 }
	}
}

WJ_contract_privilege_after_change = {
	trigger = { is_alive = yes }
	effect = { WJ_reconcile_contract_privileges_effect = yes }
}
