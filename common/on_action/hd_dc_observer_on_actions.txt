dynastic_cycle_monthly_pulse = {
	on_actions = { hd_dc_request_monthly_observation }
}

hd_dc_request_monthly_observation = {
	effect = {
		hd_dc_maintain_record_registry_effect = yes
		hd_dc_maintain_royal_events_effect = yes
		hd_dc_maintain_party_events_effect = yes
		hd_dc_request_observation_effect = yes
	}
}

# Scope remains the situation supplied by trigger_event.on_action.
hd_dc_observe_political_state = {
	effect = {
		remove_variable = hd_dc_observation_queued
		hd_dc_refresh_snapshot_effect = yes
		hd_dc_sync_pressure_effect = yes
		hd_dc_apply_pressure_sources_effect = yes
		hd_dc_observe_ordinary_pool_effect = yes
		hd_dc_evaluate_era_effect = yes
		every_in_list = {
			variable = hd_dc_instances
			limit = { NOT = { has_variable = hd_dc_reward_committed } }
			hd_dc_observe_royal_war_resolution_effect = yes
			hd_dc_audit_royal_governance_effect = yes
			hd_dc_observe_governance_effect = yes
		}
		hd_dc_observe_annual_crisis_pool_effect = yes
	}
}

# Also catch a regional war that has not yet been claimed by a crisis adapter.
on_war_started = { on_actions = { hd_dc_interrupt_governance_on_war } }
hd_dc_interrupt_governance_on_war = {
	trigger = { exists = situation:dynastic_cycle exists = scope:war scope:war = { hd_dc_war_disrupts_governance_trigger = yes } }
	effect = {
		situation:dynastic_cycle = {
			every_in_list = {
				variable = hd_dc_instances
				limit = { NOT = { has_variable = hd_dc_reward_committed } }
				hd_dc_interrupt_governance_stability_effect = yes
			}
		}
	}
}

# Title callbacks provide scope:title and a character ROOT. Appending children
# preserves every vanilla/mod effect in the main on_action definition.
on_title_gain = {
	on_actions = { hd_dc_observe_title_transfer }
}
on_title_lost = {
	on_actions = { hd_dc_observe_title_transfer }
}
on_title_destroyed = {
	on_actions = { hd_dc_observe_hegemony_destruction }
}
on_vassal_change = {
	on_actions = { hd_dc_observe_vassal_change }
}

hd_dc_observe_title_transfer = {
	trigger = {
		exists = situation:dynastic_cycle
		exists = scope:title
		scope:title = {
			OR = {
				this = title:h_china
				has_variable = hd_dc_core_county_v1
			}
		}
	}
	effect = {
		situation:dynastic_cycle = { hd_dc_request_observation_effect = yes }
	}
}

hd_dc_observe_hegemony_destruction = {
	trigger = {
		exists = situation:dynastic_cycle
		scope:landed_title = title:h_china
	}
	effect = {
		situation:dynastic_cycle = { hd_dc_request_observation_effect = yes }
	}
}

hd_dc_observe_vassal_change = {
	trigger = {
		exists = situation:dynastic_cycle
		any_realm_county = { has_variable = hd_dc_core_county_v1 }
	}
	effect = {
		situation:dynastic_cycle = { hd_dc_request_observation_effect = yes }
	}
}
