# 531 年剧本「信都起兵」史实主线入口（/mnt/project-files/531高欢剧本_计划.md 第 10、13 节）。全部带 531.7.1 闸门，其余开局不受影响。
# 效果在 common/scripted_effects/zz_hd_531_history_effects.txt，事件在 events/zz_hd_531_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_531_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hd_304_pulse），到 552 年为止
hd_531_history_start = {
	trigger = {
		game_start_date = 531.7.1
		NOT = { exists = global_var:hd_531_pulse_started }
	}
	effect = {
		set_global_variable = hd_531_pulse_started
		title:h_china = { trigger_event = { on_action = hd_531_pulse days = 1 } }
	}
}

hd_531_pulse = {
	trigger = {
		game_start_date = 531.7.1
	}
	effect = {
		hd_531_pulse_effect = yes
		if = {
			limit = { current_date < 552.1.1 }
			title:h_china = { trigger_event = { on_action = hd_531_pulse days = 30 } }
		}
	}
}
