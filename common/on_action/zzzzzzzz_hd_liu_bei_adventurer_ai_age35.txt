# Start Liu Bei's vanilla story_adventurer_ai on/after his 35th birthday.
# The normal yearly pulse remains a recovery path after age 35.
on_birthday = {
	on_actions = {
		hd_liu_bei_story_adventurer_ai_35th_birthday
	}
}

hd_liu_bei_story_adventurer_ai_35th_birthday = {
	trigger = {
		game_start_date = 184.1.1
		this = character:liu_bei
		is_alive = yes
		is_ai = yes
		age >= 35
		ep3_adventurer_ai_logic_valid_trigger = yes
		NOT = { owns_story_of_type = story_adventurer_ai }
	}
	effect = {
		create_story = { type = story_adventurer_ai }
		add_character_flag = hd_liu_bei_story_adventurer_ai_started
		debug_log = "HD_LIU_BEI STORY_ADVENTURER_AI_STARTED_AT_35"
	}
}
