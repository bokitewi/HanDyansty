# 304 年剧本「左国称王」（304.11.20，晋建武元年十月、汉元熙元年）开局。计划见 docs/304刘渊剧本_计划.md 第 6、13、14 节。
# 全部以 game_start_date = 304.11.20 为闸门，其余开局不受影响。效果写在 common/scripted_effects/zz_hd_304_startup_effects.txt。
# 头衔、天朝循环、朝贡在 on_game_start 里做；宣战、年号、国号等原版与 TGP 开局初始化完，放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_304_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_304_after_lobby
	}
}

hd_304_on_game_start = {
	trigger = {
		game_start_date = 304.11.20
		current_date = 304.11.20
		NOT = { exists = global_var:hd_304_startup_done }
	}
	effect = {
		hd_304_game_start_effect = yes
		set_global_variable = hd_304_startup_done
	}
}

hd_304_after_lobby = {
	trigger = {
		game_start_date = 304.11.20
		current_date = 304.11.20
		NOT = { exists = global_var:hd_304_after_lobby_done }
	}
	effect = {
		hd_304_after_lobby_effect = yes
		set_global_variable = hd_304_after_lobby_done
	}
}

# 晋帝庙号昵称段在 after_lobby 里先于 hbirth 指针运行，延后 1 天补跑一次
hd_304_tianjia_history_delayed = {
	trigger = { game_start_date = 304.11.20 }
	effect = { hd_nbc_jin_emperor_nicknames_effect = yes }
}

# 群雄逐鹿开局后第 2 天把通往对峙的进度清零（写法同 hd218_dc_zero_progress）
hd_304_dc_zero_progress = {
	trigger = {
		game_start_date = 304.11.20
		situation_current_phase = situation_dynastic_cycle_phase_chaos
	}
	effect = {
		every_situation_sub_region = {
			hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_hd_standoff }
		}
	}
}
