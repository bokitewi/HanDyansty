# 218 剧本历史大事件的入口（第 4 步）。全部带 game_start_date = 218.1.1，184 开局不受影响。
# 效果在 common/scripted_effects/zz_hd_218_history_effects.txt，事件在 events/zz_hd_218_history_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_sg218_history_start
	}
}

on_death = {
	on_actions = {
		hd_sg218_on_death
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hds_heartbeat）
hd_sg218_history_start = {
	trigger = {
		game_start_date = 218.1.1
		NOT = { exists = global_var:hd_sg218_pulse_started }
	}
	effect = {
		set_global_variable = hd_sg218_pulse_started
		title:h_china = { trigger_event = { on_action = hd_sg218_pulse days = 1 } }
	}
}

hd_sg218_pulse = {
	trigger = {
		game_start_date = 218.1.1
	}
	effect = {
		hd_sg218_pulse_effect = yes
		if = {
			limit = { current_date < 229.1.1 }
			title:h_china = { trigger_event = { on_action = hd_sg218_pulse days = 30 } }
		}
		else = {
			title:h_china = { trigger_event = { on_action = hd_sg218_late_pulse days = 30 } }
		}
	}
}

# 229 年后只续跑称帝这条线（boki 10-17：曹操活过 229 时受禅、称帝不能永久跳过）；三家都称完帝或到 260 年停
hd_sg218_late_pulse = {
	trigger = {
		game_start_date = 218.1.1
	}
	effect = {
		hd_sg218_check_shouchan_effect = yes
		hd_sg218_check_emperor_effect = yes
		hd_sg218_check_sun_quan_effect = yes
		hd_sg218_jianzuo_fallback_effect = yes
		if = {
			limit = {
				current_date < 260.1.1
				NOT = {
					AND = {
						has_global_variable = hd_sg218_shouchan_done
						has_global_variable = hd_sg218_shu_crowned
						has_global_variable = hd_sg218_wu_crowned
					}
				}
			}
			title:h_china = { trigger_event = { on_action = hd_sg218_late_pulse days = 30 } }
		}
	}
}

# 曹操、刘备、夏侯渊之死。Root：死者（on_death 在死亡结算之前触发，头衔还在本人手里）
hd_sg218_on_death = {
	trigger = {
		game_start_date = 218.1.1
		OR = {
			this = character:cao_cao
			this = character:liu_bei
			this = character:xia_hou_yuan
			this = character:sun_quan
		}
	}
	effect = {
		if = {
			limit = { this = character:cao_cao }
			hd_sg218_cao_cao_death_effect = yes
		}
		else_if = {
			limit = { this = character:liu_bei }
			hd_sg218_liu_bei_death_effect = yes
		}
		else_if = {
			limit = { this = character:sun_quan }
			title:h_china = { trigger_event = { on_action = hd_sg218_after_sun_quan_death days = 3 } }
		}
		else = {
			hd_sg218_xhy_death_effect = yes
		}
	}
}

# 第 11 条：曹操死后第 3 天（hd_court.0003 第 2 天先接续权臣）
hd_sg218_after_cao_cao_death = {
	trigger = {
		game_start_date = 218.1.1
	}
	effect = {
		character:cao_pi ?= {
			if = {
				limit = { is_alive = yes }
				trigger_event = hd_sg218.0110
			}
		}
	}
}

# 第 18 条：刘备死后第 3 天，新蜀主为汉中王号（或"汉"）的持有者
hd_sg218_after_liu_bei_death = {
	trigger = {
		game_start_date = 218.1.1
	}
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = hd_sg218_tuogu_done }
				global_var:hd218_hanzhong_title.holder ?= {
					is_alive = yes
					NOT = { this = character:zhu_ge_liang }
				}
				character:zhu_ge_liang ?= {
					is_alive = yes
					is_imprisoned = no
					hd_sg218_in_shu_trigger = yes
				}
			}
			set_global_variable = hd_sg218_tuogu_done
			global_var:hd218_hanzhong_title.holder = { trigger_event = hd_sg218.0180 }
		}
		# 刘备已许称帝、大典未成而死：继承人接践祚许可
		if = {
			limit = {
				has_global_variable = hd_sg218_liu_bei_emperor
				NOT = { has_global_variable = hd_sg218_shu_crowned }
			}
			hd_sg218_jianzuo_inherit_effect = { HOLDER = global_var:hd218_hanzhong_title.holder NAME = dynn_title_han }
		}
	}
}

# 孙权已许称帝、大典未成而死：死后第 3 天由吴国继承人接践祚许可

		}
	}
}
