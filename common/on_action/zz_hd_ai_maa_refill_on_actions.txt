# 汉晋春秋：AI 出钱补员
# 玩家靠"出钱补员"交互；AI 不会用该交互，这里按同样条件（同政体、6 个月一次、军队不在敌境）
# 替 AI 花钱补员，有国库扣国库、否则扣金币，留 100 底（见 zz_hd_ai_maa_refill_effects.txt）。
##################

quarterly_playable_pulse = {
	on_actions = { hd_ai_maa_gold_refill_pulse }
}

hd_ai_maa_gold_refill_pulse = {
	trigger = {
		is_ai = yes
		OR = {
			has_government = celestial_government
			has_government = tribal_government
			has_government = feudal_government
			has_government = herder_government
		}
		NOT = { has_character_flag = hd_ai_maa_gold_refill_cd }
		treasury_or_gold > 100
		# 2026-10-31：私兵 + 自己各头衔的头衔兵（含借出的）；借来的由原主补
		OR = {
			any_maa_regiment = {
				is_personal_maa_regiment = yes
				maa_regiments_valid_to_refill_trigger = yes
			}
			any_held_title = {
				any_owned_title_maa_regiment = { maa_regiments_valid_to_refill_trigger = yes }
			}
		}
		NOT = {
			any_army = {
				# 2026-10-25 日志：军队在海上时 location 无 county，原写法刷 Failed context switch；改 ?= 后海上视为不在敌境
				location.county.holder ?= {
					OR = {
						is_at_war_with = root
						any_liege_or_above = { is_at_war_with = root }
					}
				}
			}
		}
	}
	effect = {
		add_character_flag = { flag = hd_ai_maa_gold_refill_cd months = 6 }
		hd_ai_refill_maa_with_gold_effect = yes
	}
}
