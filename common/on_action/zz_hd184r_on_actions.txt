# 2026-10-22：董卓无论怎么死（诛董、战死、病死），两天后检查：结算链没接手时，自行还原法理、销毁郿国，未进入割据则进入。
on_death = { on_actions = { hd184r_dong_death } }
hd184r_dong_death = {
	trigger = {
		game_start_date = 190.1.1
		root = { this = character:dong_zhuo_2 }
	}
	effect = {
		title:h_china = { trigger_event = { on_action = hd184r_dong_death_followup days = 2 } }
	}
}
hd184r_dong_death_followup = {
	effect = {
		if = {
			limit = {
				NOT = { exists = global_var:hd_dong_settlement_pending }
				NOT = { exists = global_var:hd_dong_settlement_processed }
			}
			# 只在董卓已入京（郿国已建）或讨董联盟已成立后才进入割据；入京前就死的交给 193 年兜底
			if = {
				limit = {
					hd_history_events_enabled_trigger = yes
					OR = {
						exists = global_var:hd184r_meiguo
						exists = global_var:hd_anti_regent_founded
						exists = global_var:hd_anti_regent_chain_resolved
					}
				}
				hd184r_meiguo_destroy_effect = yes
				hd184r_warlord_era_begin_effect = yes
			}
			else = { hd184r_meiguo_destroy_effect = yes }
			debug_log = "HD184R P3 DONG_DEATH_FOLLOWUP"
		}
	}
}
