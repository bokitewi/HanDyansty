# 2026-10-17：孙策首次获得王国级或更高头衔 → 小霸王（hd_sc_bawang.0001），一生一次。
on_title_gain = {
	on_actions = { hd_sc_bawang_on_title_gain }
}

hd_sc_bawang_on_title_gain = {
	trigger = {
		scope:title.tier >= tier_kingdom
		NOT = { has_character_flag = hd_sc_bawang_done }
		OR = {
			AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
			AND = { exists = character:sun_ce this = character:sun_ce }
		}
	}
	effect = {
		# 同一天拿到多个头衔时只发一次
		add_character_flag = hd_sc_bawang_done
		trigger_event = hd_sc_bawang.0001
	}
}
