# 原版 on_title_gain 只枚举九个内阁头衔；丞相补充同样的授职回调。
on_title_gain = {
	on_actions = {
		WJ_cheng_xiang_title_gain
	}
}

WJ_cheng_xiang_title_gain = {
	trigger = {
		scope:title = title:e_WJ_minister_cheng_xiang
		is_alive = yes
		is_independent_ruler = no
		government_has_flag = government_is_celestial
		scope:title.holder = root
		exists = title:h_china.holder
		liege = top_liege
		top_liege = title:h_china.holder
		root != title:h_china.holder
		title:h_china.holder = { WJ_han_jin_law_trigger = yes }
	}
	effect = {
		title:h_china.holder = {
			save_scope_as = councillor_liege
		}
		root = {
			got_minister_position_effect = { MINISTER_TITLE = e_WJ_minister_cheng_xiang MINISTER_POSITION = WJ_minister_cheng_xiang }
		}
	}
}
