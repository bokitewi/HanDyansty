# 531 年剧本「信都起兵」（531.7.1，魏普泰元年六月庚子、梁中大通三年六月）开局。计划见 /mnt/project-files/531高欢剧本_计划.md 第 6、13 节。
# 全部以 game_start_date = 531.7.1 为闸门，其余开局不受影响。效果写在 common/scripted_effects/zz_hd_531_startup_effects.txt。
# 清理、头衔、天朝循环、朝贡在 on_game_start 里做；宣战、年号、国号、权臣等原版与 TGP 开局初始化完，放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_531_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_531_after_lobby
	}
}

hd_531_on_game_start = {
	trigger = {
		game_start_date = 531.7.1
		current_date = 531.7.1
		NOT = { exists = global_var:hd_531_startup_done }
	}
	effect = {
		# 先设：hd_court_regent_effects.txt、hd_dc_kit_ten_effects.txt 凭它先 granted 授丞相（计划复核必改 7）
		set_global_variable = hd_531_startup_done
		hd_531_game_start_effect = yes
	}
}

hd_531_after_lobby = {
	trigger = {
		game_start_date = 531.7.1
		current_date = 531.7.1
		NOT = { exists = global_var:hd_531_after_lobby_done }
	}
	effect = {
		hd_531_after_lobby_effect = yes
		set_global_variable = hd_531_after_lobby_done
	}
}

# 群雄逐鹿开局后第 2 天把通往对峙的进度清零（写法同 hd_304_dc_zero_progress）
hd_531_dc_zero_progress = {
	trigger = {
		game_start_date = 531.7.1
		situation_current_phase = situation_dynastic_cycle_phase_chaos
	}
	effect = {
		every_situation_sub_region = {
			hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_hd_standoff }
		}
	}
}
