# 184 开局：宦官任史实宫廷职位（2026-10-08）
# 张让 → 中常侍（chief_eunuch）；毕岚 → 掖庭令（keeper_of_the_harem）；宋典 → 钩盾令（court_gardener）。
# 宦官的阉人特质与入宫（employer = liu_hong_3）写在人物历史里，这里只做任职。
on_game_start = {
	on_actions = { hd_eunuch_court_184_startup }
}

hd_eunuch_court_184_startup = {
	trigger = {
		game_start_date = 184.1.1
		exists = character:liu_hong_3
	}
	effect = {
		character:liu_hong_3 = {
			save_scope_as = hd_eunuch_emperor
			hd_eunuch_appoint_effect = { WHO = zhang_rang POS = chief_eunuch_court_position }
			hd_eunuch_appoint_effect = { WHO = bi_lan POS = keeper_of_the_harem_court_position }
			hd_eunuch_appoint_effect = { WHO = song_dian POS = court_gardener_court_position }
		}
		character:zhang_rang ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = officer_zhong_chang_shi
			}
		}
	}
}
