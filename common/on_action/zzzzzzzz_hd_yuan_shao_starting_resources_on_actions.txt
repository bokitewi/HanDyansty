# 184.1.1 Yuan Shao scenario resource normalization.
# Loaded late so the final after-lobby value is deterministic.
on_game_start_after_lobby = {
	on_actions = {
		hd_yuan_shao_184_starting_resources
	}
}

hd_yuan_shao_184_starting_resources = {
	trigger = { game_start_date = 184.1.1 }
	effect = {
		character:yuan_shao ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = hd_yuan_shao_184_resources_initialized }
				}
				# Exact starting treasury: current gold + (8132 - current gold).
				add_gold = {
					value = 8132
					subtract = gold
				}
				# Fame level 3 = 赫赫有名. Normalize instead of blindly adding 3.
				add_prestige_level = {
					value = 3
					subtract = prestige_level
				}
				add_character_flag = hd_yuan_shao_184_resources_initialized
			}
		}
	}
}
