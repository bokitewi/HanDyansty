# The engine staggers this annual pulse across characters, including landless courtiers.
random_yearly_everyone_pulse = {
 on_actions = { WJ_ty_reward_ai_annual }
}

WJ_ty_reward_ai_annual = {
 trigger = {
  is_ai = yes
  is_imprisoned = no
  exists = title:h_china.holder
  WJ_ty_memorial_in_realm = yes
  OR = {
   AND = { gold >= 10 prestige >= 20 }
   AND = { highest_held_title_tier >= tier_empire prestige >= 50 }
  }
  NOT = { has_variable = wj_ty_open }
 }
 effect = {
  # Most local officials never prepare a case on a given annual opportunity.
  random_list = {
   199 = {}
   1 = {
    modifier = { factor = 24 WJ_ty_reward_ai_can_open = yes }
    modifier = { factor = 2 WJ_ty_reward_ai_recent_service = yes }
    WJ_ty_reward_ai_consider = yes
   }
  }
 }
}
