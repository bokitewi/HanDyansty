# 218 年（建安二十三年）剧本开局。全部以 game_start_date = 218.1.1 为闸门，184 开局不受影响。
# 效果写在 common/scripted_effects/zz_hd_218_startup_effects.txt。
# 头衔、封臣关系在 on_game_start 里改好（进大厅前，书签和立后、立太子都按改好的头衔判断）；
# 朝官、契约、都督区、宣战等要等原版和 TGP 开局初始化完，放在 on_game_start_after_lobby（同 184 的汉廷开局）。

on_game_start = {
	on_actions = {
		hd218_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd218_after_lobby
	}
}

yearly_global_pulse = {
	on_actions = {
		hd218_yearly
	}
}

hd218_on_game_start = {
	trigger = {
		game_start_date = 218.1.1
		current_date = 218.1.1
		NOT = { exists = global_var:hd218_startup_done }
	}
	effect = {
		hd218_game_start_effect = yes
		set_global_variable = hd218_startup_done
	}
}

hd218_after_lobby = {
	trigger = {
		game_start_date = 218.1.1
		current_date = 218.1.1
		NOT = { exists = global_var:hd218_after_lobby_done }
	}
	effect = {
		hd218_after_lobby_effect = yes
		set_global_variable = hd218_after_lobby_done
	}
}

# 开局第 10 天检查，由 hd218_after_lobby_effect 从 title:h_china 延时触发
hd218_day10_check = {
	trigger = {
		game_start_date = 218.1.1
	}
	effect = {
		hd218_day10_check_effect = yes
	}
}


# 生日生成器在父亲死后补出的孩子（陆抗、张震等），把父亲接到保留的那个人身上
hd218_yearly = {
	trigger = {
		game_start_date = 218.1.1
	}
	effect = {
		hd218_father_patch_effect = yes
		hd218_cadet_sons_effect = yes
		hd218_cao_land_watch_effect = yes
		hd218_e_han_watch_effect = yes
	}
}

# 开局第 2 天再清一次对峙期进度，吃掉开局首日事件触发的催化剂。Situation scope
hd218_dc_zero_progress = {
	trigger = {
		situation_current_phase = situation_dynastic_cycle_phase_hd_standoff
	}
	effect = {
		every_situation_sub_region = {
			hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_chaos }
		}
	}
}
