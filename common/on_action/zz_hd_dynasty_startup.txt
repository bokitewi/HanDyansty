# 开局宗族配置 - Dynasty startup setup (184)
# 1. 汝南袁氏（dynasty=dyn_111010059004002 / house=house_111010059004013）：
#    家族族长 = 袁隗（yuan_kui，125-190）；宗族等级 = 名声显著（dynasty prestige level 6）
# 2. 汉室彭城刘氏（dynasty=dyn_111010250025002 / house=house_111010250025013）：
#    宗族等级 = 震古烁今（dynasty prestige level 9）
on_game_start = {
	on_actions = {
		hd_dynasty_startup
	}
}

hd_dynasty_startup = {
	trigger = {
		game_start_date = 184.1.1
		current_date = 184.1.1
		NOT = { exists = global_var:hd_dynasty_startup_initialized }
	}
	effect = {
		hd_setup_historical_ai_stories_effect = yes
		# 袁氏家族族长 → 袁隗（先压平再设等级，保证精确到达目标等级）
		character:yuan_kui ?= {
			house ?= {
				set_house_head = prev
			}
		}
		dynasty:dyn_111010059004002 = {
			add_dynasty_prestige_level = -20
			add_dynasty_prestige_level = 6
		}
		dynasty:dyn_111010250025002 = {
			add_dynasty_prestige_level = -20
			add_dynasty_prestige_level = 9
		}

		# 184 start relationships.
		character:cao_cao ?= {
			add_character_flag = hd_cao_personal_story_established
			if = {
				limit = { is_alive = yes NOT = { any_owned_story = { type = hd_story_cao_life } } }
				create_story = { type = hd_story_cao_life }
			}
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
			set_relation_friend = { reason = friend_common_ground target = character:yuan_shao }
			set_relation_friend = { reason = friend_common_ground target = character:xu_you }
		}
		character:lv_bu ?= {
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
		}
		character:he_jin ?= {
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
		}
		character:ding_yuan ?= {
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
		}
		character:liu_bian ?= {
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
		}
		character:liu_xie ?= {
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
		}
		character:he_huang_hou ?= {
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
		}
		character:yuan_shao ?= {
			add_character_flag = hd_yuan_personal_story_established
			if = {
				limit = { is_alive = yes NOT = { any_owned_story = { type = hd_story_yuan_life } } }
				create_story = { type = hd_story_yuan_life }
			}
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
			set_relation_friend = { reason = friend_common_ground target = character:xu_you }
			house ?= { set_house_head = prev }
		}
		character:liu_bei ?= {
			if = {
				limit = { is_alive = yes NOT = { any_owned_story = { type = hd_story_liu_bei_life } } }
				create_story = { type = hd_story_liu_bei_life }
			}
			set_relation_friend = { reason = friend_common_ground target = character:gong_sun_zan }
			set_relation_mentor = character:lu_zhi
		}
		character:han_fu ?= {
			set_relation_mentor = character:yuan_kui
		}
		character:gong_sun_zan ?= {
			set_relation_rival = { reason = rival_historical target = character:liu_yu_3 }
		}
		character:dong_zhuo_2 ?= {
			if = {
				limit = { is_alive = yes NOT = { any_owned_story = { type = hd_story_dong_zhuo_life } } }
				create_story = { type = hd_story_dong_zhuo_life }
			}
			if = {
				limit = { is_alive = yes NOT = { has_character_flag = hd_historical_actor_protection_received } }
				add_character_modifier = { modifier = hd_historical_actor_protection years = 10 }
				add_character_flag = hd_historical_actor_protection_received
			}
			set_relation_mentor = character:yuan_kui
		}
		character:jian_yong ?= {
			if = {
				limit = { is_alive = yes }
				destroy_title = title:c_hd_nf_famous_h_house_111010380001013
				if = {
					limit = { is_landed = no }
					set_employer = character:liu_bei
				}
				set_relation_friend = { reason = friend_common_ground target = character:liu_bei }
			}
		}

		# Gongsun Zan's permanent 184 White Horse Volunteers.
		character:gong_sun_zan ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = hd_gongsun_baima_start_army_received }
				}
				spawn_army = {
					name = hd_gongsun_baima_event_army
					men_at_arms = { type = hd_baima_yicong men = 6000 }
					location = capital_province
					uses_supply = yes
					inheritable = yes
				}
				add_character_flag = hd_gongsun_baima_start_army_received
			}
		}

		# Liu Biao assumes Jingzhou exactly on 188.1.1.
		character:liu_biao ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = hd_liu_biao_jingzhou_scheduled }
					NOT = { has_character_flag = hd_liu_biao_jingzhou_completed }
				}
				add_character_flag = hd_liu_biao_jingzhou_scheduled
				trigger_event = { id = hd_historical_world.0188 days = 1460 }
			}
		}

		# Keep the four historical officers as Sun Jian's unlanded courtiers.
		if = {
			limit = {
				exists = character:sun_jian
				character:sun_jian = { is_alive = yes }
				exists = title:d_lean.holder
				exists = title:d_yiyang.holder
			}
			create_title_and_vassal_change = { type = conquest save_scope_as = hd_sun_court_change }
			title:c_uuii_p_22 = {
				change_title_holder_include_vassals = { holder = title:d_lean.holder change = scope:hd_sun_court_change }
			}
			title:c_uuii_p_1944 = {
				change_title_holder_include_vassals = { holder = title:d_yiyang.holder change = scope:hd_sun_court_change }
			}
			resolve_title_and_vassal_change = scope:hd_sun_court_change

			character:cheng_pu ?= { destroy_title = title:c_hd_nf_famous_h_house_111010191003013 }
			character:huang_gai ?= { destroy_title = title:c_hd_nf_famous_h_house_111010094002013 }
			character:han_dang ?= { destroy_title = title:c_hd_nf_famous_h_house_111010015005013 }

			character:cheng_pu ?= { if = { limit = { is_alive = yes is_landed = no } set_employer = character:sun_jian } }
			character:huang_gai ?= { if = { limit = { is_alive = yes is_landed = no } set_employer = character:sun_jian } }
			character:han_dang ?= { if = { limit = { is_alive = yes is_landed = no } set_employer = character:sun_jian } }
			character:zu_mao ?= { if = { limit = { is_alive = yes is_landed = no } set_employer = character:sun_jian } }
		}

		# Protect the historical 189 transition from Liu Hong's low starting health.
		character:liu_hong_3 ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = hd_liu_hong_189_death_scheduled }
				}
				add_character_flag = hd_liu_hong_189_death_scheduled
				add_character_modifier = { modifier = hd_liu_hong_timeline_protection days = 1958 }
				trigger_event = { id = hd_palace_crisis.0090 days = 1957 }
			}
		}
		set_global_variable = hd_dynasty_startup_initialized
	}
}
