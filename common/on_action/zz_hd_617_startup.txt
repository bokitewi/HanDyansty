# 617 年剧本「晋阳起兵」（617.6.23，隋大业十三年五月甲子）开局。计划见 /mnt/project-files/617隋末群雄剧本_计划.md 第 6、13 节。
# 全部以 game_start_date = 617.6.23 为闸门，其余开局不受影响。效果写在 common/scripted_effects/zz_hd_617_startup_effects.txt。
# 清理、开关、天朝循环、朝贡、国号在 on_game_start 里做（大厅前就生效）；宣战、年号、宫廷、朝官、停战放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_617_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_617_after_lobby
	}
}

hd_617_on_game_start = {
	trigger = {
		game_start_date = 617.6.23
		current_date = 617.6.23
		NOT = { exists = global_var:hd_617_startup_done }
	}
	effect = {
		# 先设：hd_court_regent_effects.txt、hd_dc_kit_ten_effects.txt 凭它先 granted 授丞相（S7 王世充）
		set_global_variable = hd_617_startup_done
		hd_617_game_start_effect = yes
	}
}

hd_617_after_lobby = {
	trigger = {
		game_start_date = 617.6.23
		current_date = 617.6.23
		NOT = { exists = global_var:hd_617_after_lobby_done }
	}
	effect = {
		hd_617_after_lobby_effect = yes
		set_global_variable = hd_617_after_lobby_done
	}
}

# 群雄逐鹿开局后第 2 天把通往对峙的进度清零（写法同 hd_531_dc_zero_progress）
hd_617_dc_zero_progress = {
	trigger = {
		game_start_date = 617.6.23
		situation_current_phase = situation_dynastic_cycle_phase_chaos
	}
	effect = {
		every_situation_sub_region = {
			hd_dc_zero_phase_points_effect = { PHASE = situation_dynastic_cycle_phase_hd_standoff }
		}
	}
}
