# Phase 2 uses the project monthly heartbeat only to open state-based opportunities.
on_game_start_after_lobby = {
	on_actions = { hd_liu_bei_life_after_lobby }
}

hd_liu_bei_life_after_lobby = {
	trigger = {
		game_start_date = 184.1.1
	}
	effect = {
		hd_update_liu_bei_life_story_effect = yes
	}
}

hd_historical_world_monthly_check = {
	effect = {
		# The Liu Bei updater now exists only to maintain the pre-Yellow-Turban
		# Peach Garden story and to close it when the campaign resolves.
		hd_update_liu_bei_life_story_effect = yes
		hd_resolve_invalid_historical_world_chains_effect = yes
		character:cao_cao = {
			if = {
				limit = { has_character_flag = hd_cao_fengying_authorized }
				hd_cao_fengying_try_takeover_effect = yes
			}
		}
		if = {
			limit = { hd_fengying_world_ready_trigger = yes }
			hd_open_fengying_world_opportunity_effect = yes
		}
		if = {
			limit = { hd_guandu_world_ready_trigger = yes }
			hd_open_guandu_world_opportunity_effect = yes
		}
		if = {
			limit = { hd_chibi_world_ready_trigger = yes }
			hd_open_chibi_world_opportunity_effect = yes
		}
	}
}
