on_game_start = { on_actions = { WJ_peer_full_check } }
# Reuse the main mod's post-law-change callback, after on_pass's old-law scope has settled.
WJ_titles_after_change = { on_actions = { WJ_peer_after_title_change } }
yearly_global_pulse = { on_actions = { WJ_peer_yearly } }
on_title_gain = { on_actions = { WJ_peer_realm_changed } }
on_title_gain_inheritance = { on_actions = { WJ_peer_realm_changed } }
on_title_destroyed = { on_actions = { WJ_peer_title_destroyed } }
on_vassal_change = { on_actions = { WJ_peer_realm_changed } }
on_join_court = { on_actions = { WJ_peer_realm_changed } }
on_leave_court = { on_actions = { WJ_peer_court_departure } }
on_government_change = { on_actions = { WJ_peer_realm_changed } }
on_liege_government_change = { on_actions = { WJ_peer_realm_changed } }
# 每季度全量维护已移除：每年 WJ_peer_yearly 发俸前全量校验一次。
WJ_peer_yearly = { effect = {
 WJ_peer_maintain_all = yes
 every_in_global_list = { variable = wj_peer_families limit = { WJ_peer_family_valid = yes } holder = { WJ_peer_pay = yes } }
 every_in_global_list = { variable = wj_peer_personal_bearers limit = { WJ_peer_personal_valid = yes } WJ_peer_pay = yes }
} }
# 生命周期钩子只查当事人。
WJ_peer_realm_changed = { effect = { WJ_peer_maintain_self = yes } }
WJ_peer_full_check = { effect = { WJ_peer_maintain_all = yes } }
# 官称共用回调：法律变更（独立君主）保留全量，普通头衔变动只查当事人。
WJ_peer_after_title_change = { effect = {
 if = { limit = { is_independent_ruler = yes government_has_flag = government_is_celestial } WJ_peer_maintain_all = yes }
 else = { WJ_peer_maintain_self = yes }
} }
WJ_peer_title_destroyed = { trigger = { scope:landed_title = { has_variable = wj_peer_rank } } effect = { scope:landed_title = { WJ_peer_clear_family = yes } } }
WJ_peer_court_departure = {
 trigger = { has_variable = wj_peer_rank }
 # Native leave fires before placement. Join/vassal hooks are immediate; pool fallback is one day.
 effect = { trigger_event = { id = WJ_peer.900 days = 1 } }
}
