# 184历史职官：开局一次性配置（效果见 zz_hd_184_historical_offices_effects.txt）。
# 两校尉在大厅结束当天定号，早于第 1 天的随机授号；资格补足延到第 1 天，晚于宦海沉浮开局分配。
on_game_start_after_lobby = {
	on_actions = { hd_184_historical_offices_after_lobby }
}

hd_184_historical_offices_after_lobby = {
	trigger = {
		game_start_date = 184.1.1
		current_date = 184.1.1
		NOT = { exists = global_var:hd_184_historical_offices_initialized }
		exists = title:h_china.holder
		title:h_china.holder = { government_has_flag = government_is_celestial }
	}
	effect = {
		set_global_variable = hd_184_historical_offices_initialized
		hd_184_fixed_colonels_effect = yes
		title:h_china.holder = {
			trigger_event = { on_action = hd_184_historical_offices_qualify days = 1 }
		}
	}
}

hd_184_historical_offices_qualify = {
	effect = {
		hd_184_office_qualify_all_effect = yes
	}
}

hd_184_office_cap_step = {
	effect = {
		hd_184_office_cap_step_effect = yes
	}
}
