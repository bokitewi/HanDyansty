# 南北朝剧本史实主线的入口（docs/南北朝剧本_改进计划.md 3.7）。全部带 346.12.10 / 404.3.24 闸门，184、218 开局不受影响。
# 效果在 common/scripted_effects/zz_hd_nbc_history_effects.txt，事件在 events/zz_hd_nbc_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_nbc_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hd_sg218_pulse）；346 线到 386 年（淝水未结算时延后，最晚 400 年）、404 线到 421 年为止
hd_nbc_history_start = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		OR = {
			game_start_date = 346.12.10
			game_start_date = 404.3.24
		}
		NOT = { exists = global_var:hd_nbc_pulse_started }
	}
	effect = {
		set_global_variable = hd_nbc_pulse_started
		title:h_china = { trigger_event = { on_action = hd_nbc_pulse days = 1 } }
	}
}

hd_nbc_pulse = {
	trigger = {
		OR = {
			game_start_date = 346.12.10
			game_start_date = 404.3.24
		}
	}
	effect = {
		hd_nbc_pulse_effect = yes
		if = {
			limit = {
				OR = {
					# 346 线：心跳到 386.2.1（淝水入口最晚 386.1.1）；淝水已发事件但未结算时继续（等玩家点事件最多到 387 年，战争最多到 400 年），保证前秦瓦解能触发
					AND = {
						game_start_date = 346.12.10
						OR = {
							current_date < 386.2.1
							AND = {
								current_date < 400.1.1
								NOT = { has_global_variable = hd_nbc_feishui_result_done }
								OR = {
									has_global_variable = hd_nbc_feishui_active
									has_global_variable = hd_nbc_feishui_won
									has_global_variable = hd_nbc_feishui_lost
									has_global_variable = hd_nbc_feishui_white
									has_global_variable = hd_nbc_feishui_invalid
									AND = { has_global_variable = hd_nbc_feishui_done current_date < 387.1.1 }
								}
							}
						}
					}
					AND = { game_start_date = 404.3.24 current_date < 421.1.1 }
				}
			}
			title:h_china = { trigger_event = { on_action = hd_nbc_pulse days = 30 } }
		}
	}
}
