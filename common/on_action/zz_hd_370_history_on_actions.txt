# 370 年剧本「王猛伐燕」史实主线入口（docs/370苻坚剧本_计划.md 第 10、14 节）。全部带 370.2.13 闸门，其余开局不受影响。
# 效果在 common/scripted_effects/zz_hd_370_history_effects.txt，事件在 events/zz_hd_370_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_370_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hd_531_pulse），到 386 年为止
hd_370_history_start = {
	trigger = {
		game_start_date = 370.2.13
		NOT = { exists = global_var:hd_370_pulse_started }
	}
	effect = {
		set_global_variable = hd_370_pulse_started
		title:h_china = { trigger_event = { on_action = hd_370_pulse days = 1 } }
	}
}

hd_370_pulse = {
	trigger = {
		game_start_date = 370.2.13
	}
	effect = {
		hd_370_pulse_effect = yes
		if = {
			limit = { current_date < 386.1.1 }
			title:h_china = { trigger_event = { on_action = hd_370_pulse days = 30 } }
		}
	}
}
