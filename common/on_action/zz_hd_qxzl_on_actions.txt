# Script-triggered on_actions only. No daily, monthly or global pulse hook.
# trigger_event { on_action = ... days = ... } accepts non-character scopes.
hd_qxzl_scan_after_30_days = {
	effect = { hd_qxzl_scan_current_rulers_effect = yes }
}

# Saved title scope is captured separately for each queued call. A dead D30
# holder is never an event recipient: resolve the title's CURRENT holder.
hd_qxzl_deliver_duke_ticket = {
	effect = {
		scope:hd_qxzl_ticket_title ?= {
			if = {
				limit = {
					has_variable = hd_qxzl_duke_ticket
					NOT = { has_variable = hd_qxzl_ticket_consumed }
				}
				set_variable = hd_qxzl_ticket_consumed
				holder ?= {
					if = {
						limit = {
							highest_held_title_tier = tier_duchy
							hd_qxzl_vassal_eligible_trigger = yes
						}
						hd_qxzl_present_event_effect = yes
					}
				}
			}
		}
	}
}

hd_qxzl_deliver_count_ticket = {
	effect = {
		scope:hd_qxzl_ticket_title ?= {
			if = {
				limit = {
					has_variable = hd_qxzl_count_ticket
					NOT = { has_variable = hd_qxzl_ticket_consumed }
				}
				set_variable = hd_qxzl_ticket_consumed
				holder ?= {
					if = {
						limit = {
							highest_held_title_tier = tier_county
							hd_qxzl_vassal_eligible_trigger = yes
						}
						hd_qxzl_present_event_effect = yes
					}
				}
			}
		}
	}
}

# Bookkeeping only; never a rescan, another release pass or forced response.
# A player's already-open event can still be answered after dispatch ends.
hd_qxzl_close_distribution = {
	effect = {
		if = {
			limit = { exists = global_var:hd_qxzl_started }
			set_global_variable = hd_qxzl_distribution_closed
			debug_log = "HD_QXZL DISTRIBUTION_WINDOW_CLOSED"
		}
	}
}
