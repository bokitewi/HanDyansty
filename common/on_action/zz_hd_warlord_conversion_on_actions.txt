on_game_start = {
	on_actions = {
		hd_warlord_conversion_on_game_start
	}
}

#HD: 军阀政体转换不再挂头衔/封臣/文化/信仰/政体变更等高频事件(裂解与战争结算期每次头衔转移都会重评触发器)，
#改挂原版 random_yearly_playable_pulse——每个角色每年在随机日期触发一次，新独立者最迟一年内随机时点转换，天然错开运算
random_yearly_playable_pulse = {
	on_actions = {
		hd_warlord_conversion_random_yearly
	}
}

#HD: 独立时刻挂1825天(五年)稳定期flag，供逐鹿中原CB做"裂解稳定期"判定；
#开局(on_game_start)不设flag——只保护游戏开始后新独立的统治者
on_vassal_change = {
	on_actions = {
		hd_warlord_stabilization_on_vassal_change
	}
}

hd_warlord_conversion_on_game_start = {
	effect = {
		every_independent_ruler = {
			limit = {
				hd_should_convert_to_independent_warlord_trigger = yes
			}
			hd_convert_to_independent_warlord_effect = yes
		}
	}
}

hd_warlord_conversion_random_yearly = {
	trigger = {
		hd_should_convert_to_independent_warlord_trigger = yes
	}
	effect = {
		hd_convert_to_independent_warlord_effect = yes
	}
}

hd_warlord_stabilization_on_vassal_change = {
	trigger = {
		is_independent_ruler = yes
	}
	effect = {
		add_character_flag = {
			flag = hd_consolidation_stabilization
			days = 1825
		}
	}
}
