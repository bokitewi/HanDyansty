hd_dong_preserve_historical_regent = {
	trigger = {
		root = {
			this = title:h_china.holder
			hd_dong_historical_regent_active_trigger = yes
			NOT = { has_character_flag = hd_dong_regency_repair_queued }
			OR = {
				NOT = { has_diarchy_type = grand_secretariat }
				NOT = { diarch = var:hd_dong_fixed_regent }
				AND = { var:hd_dong_fixed_regent = character:wang_yun NOT = { has_realm_law = grand_chancellor_law } }
				AND = { NOT = { var:hd_dong_fixed_regent = character:wang_yun } NOT = { has_realm_law = WJ_grand_chengxiang_law } }
			}
		}
	}
	effect = {
		root = {
			add_character_flag = hd_dong_regency_repair_queued
			trigger_event = { id = hd_dong_aftermath.0005 days = 1 }
		}
	}
}

hd_dong_aftermath_actor_death = {
	trigger = { root = { OR = { this = character:dong_zhuo_2 this = character:wang_yun this = character:li_jue this = character:cao_cao this = character:diao_chan this = title:h_china.holder } } }
	effect = {
		if = {
			limit = { root = title:h_china.holder exists = global_var:hd_dong_settlement_pending }
			hd_dong_abort_settlement_effect = yes
		}
		if = {
			limit = { root = character:wang_yun exists = global_var:hd_dong_murder_pending }
			remove_global_variable = hd_dong_murder_pending
			debug_log = "HD_DONG MURDER_ABORTED_WANG_DIED"
		}
		if = {
			limit = { root = character:li_jue OR = { exists = global_var:hd_li_jue_war_pending exists = global_var:hd_li_jue_war_active } }
			hd_li_jue_end_war_effect = yes
		}
		title:h_china.holder ?= {
			if = {
				limit = { has_variable = hd_dong_fixed_regent var:hd_dong_fixed_regent = root }
				remove_variable = hd_dong_fixed_regent
			}
		}
		root = { remove_character_flag = hd_dong_diaochan_union_pending }
	}
}
