# No world scan: only changed character, direct old/new imperial holder, and title succession.
on_vassal_change = { on_actions = { WJ_inner_vassal_changed } }
on_death = { on_actions = { WJ_inner_member_died } }
on_title_gain = { on_actions = { WJ_inner_title_changed } }
on_title_lost = { on_actions = { WJ_inner_title_changed } }
quarterly_playable_pulse = { on_actions = { WJ_inner_holder_check } }
WJ_inner_holder_check = {
 trigger = { has_variable = wj_dasima_emperor }
 effect = { WJ_dasima_reconcile_effect = yes }
}
WJ_inner_member_died = {
 trigger = { has_variable = wj_dasima_emperor }
 effect = { WJ_dasima_clear_effect = yes }
}
WJ_inner_vassal_changed = {
 effect = {
  # Let native succession finish transferring titles/vassals before judging loss.
  if = { limit = { has_variable = wj_dasima_emperor } trigger_event = { on_action = WJ_inner_holder_check days = 1 } }
  var:wj_dasima_holder ?= { trigger_event = { on_action = WJ_inner_holder_check days = 1 } }
 }
}
WJ_inner_title_changed = {
 effect = {
  if = { limit = { has_variable = wj_dasima_emperor } trigger_event = { on_action = WJ_inner_holder_check days = 1 } }
  scope:previous_holder ?= {
   if = { limit = { has_variable = wj_dasima_emperor } trigger_event = { on_action = WJ_inner_holder_check days = 1 } }
  }
  # Native succession may finalize vassal transfer after title callback.
  if = {
   limit = { exists = scope:previous_holder.var:wj_dasima_holder WJ_inner_emperor_trigger = yes }
   scope:previous_holder.var:wj_dasima_holder = {
    trigger_event = { on_action = WJ_inner_holder_check days = 1 }
   }
  }
 }
}
