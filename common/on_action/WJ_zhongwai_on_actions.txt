# Native diarchy hooks: ROOT = liege, scope:old_diarch = outgoing incumbent.
on_left_diarchy = { on_actions = { WJ_zw_diarchy_changed } }
on_diarch_change = { on_actions = { WJ_zw_diarchy_changed } }
WJ_zw_diarchy_changed = {
 effect = {
  scope:old_diarch ?= { WJ_zw_reconcile = yes }
  var:wj_zw_holder ?= { WJ_zw_reconcile = yes }
 }
}
on_vassal_change = { on_actions = { WJ_zw_holder_changed } }
on_title_gain = { on_actions = { WJ_zw_holder_changed } }
on_title_lost = { on_actions = { WJ_zw_holder_changed } }
on_government_change = { on_actions = { WJ_zw_holder_changed } }
on_liege_government_change = { on_actions = { WJ_zw_holder_changed } }
quarterly_playable_pulse = { on_actions = { WJ_zw_holder_changed } }
WJ_zw_holder_changed = {
 trigger = { OR = { has_trait = WJ_zhongwai has_variable = wj_zw_emperor vassal_contract_obligation_level:WJ_huangyue > 0 } }
 effect = { WJ_zw_reconcile = yes }
}
on_death = { on_actions = { WJ_zw_holder_died } }
WJ_zw_holder_died = {
 trigger = { OR = { has_trait = WJ_zhongwai has_variable = wj_zw_emperor } }
 effect = { WJ_zw_clear = yes }
}
