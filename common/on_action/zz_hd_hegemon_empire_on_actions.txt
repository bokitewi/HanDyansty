# 霸权者只能拥有一个实地帝国头衔：获得 h_china 或获得帝国头衔时清理多余的实地帝国头衔。
# 只向 on_title_gain 追加子 on_action，不覆盖原版。
on_title_gain = {
	on_actions = { hd_hegemon_empire_on_title_gain }
}

hd_hegemon_empire_on_title_gain = {
	trigger = {
		exists = scope:title
		root = { has_title = title:h_china }
		OR = {
			scope:title = title:h_china
			scope:title.tier = tier_empire
		}
	}
	effect = {
		root = { hd_cycle_hegemon_empire_purge_effect = yes }
	}
}
