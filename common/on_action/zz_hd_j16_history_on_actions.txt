# 救援口（01 §1.4）：旧档里旧心跳早已停排、转发壳永远不会执行时，由每年一次的全局脉冲把它接进新心跳。
# 新档开局已设 hd_j16_pulse_started，这里不会触发
yearly_global_pulse = {
	on_actions = {
		hd_j16_yearly_rescue
	}
}

hd_j16_history_start = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		hd_j16_start_trigger = yes
		NOT = { has_global_variable = hd_j16_pulse_started }
	}
	effect = {
		hd_j16_bootstrap_effect = yes
	}
}

# 载体 title:h_china：霸权头衔无持有者时也存在，可一直挂排期（01 §1.2）。每 30 天一次，到 421.1.1 为止
hd_j16_pulse = {
	trigger = {
		hd_j16_start_trigger = yes
	}
	effect = {
		hd_j16_pulse_effect = yes
		if = {
			limit = { hd_j16_continue_trigger = yes }
			title:h_china = { trigger_event = { on_action = hd_j16_pulse days = 30 } }
		}
	}
}

hd_j16_yearly_rescue = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		hd_j16_start_trigger = yes
		hd_j16_continue_trigger = yes
		NOT = { has_global_variable = hd_j16_pulse_started }
	}
	effect = {
		hd_j16_bootstrap_effect = yes
		debug_log = "HD_J16 boot FIRE yearly_rescue"
	}
}
