# 89 年剧本「燕然勒石」史实主线入口（/mnt/project-files/89窦宪北伐剧本_计划.md 第 10、13 节）。全部带 89.1.30 闸门，其余开局不受影响。
# 效果在 common/scripted_effects/zz_hd_89_history_effects.txt，事件在 events/zz_hd_89_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_89_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 22），到 108 年为止（卒年表最后一人刘庆卒于 107.7.1，复核必改 7）
hd_89_history_start = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		game_start_date = 89.1.30
		NOT = { exists = global_var:hd_89_pulse_started }
	}
	effect = {
		set_global_variable = hd_89_pulse_started
		title:h_china = { trigger_event = { on_action = hd_89_pulse days = 1 } }
	}
}

hd_89_pulse = {
	trigger = {
		game_start_date = 89.1.30
	}
	effect = {
		hd_89_pulse_effect = yes
		if = {
			limit = { current_date < 108.1.1 }
			title:h_china = { trigger_event = { on_action = hd_89_pulse days = 30 } }
		}
	}
}
