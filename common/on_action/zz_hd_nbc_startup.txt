# 南北朝剧本（346.12.10「戡定岷峨」、404.3.24「京口起兵」）开局。计划见 docs/南北朝剧本_整合计划.md 第 5.8 节。
# 全部以 game_start_date 为闸门，184、218 开局不受影响。效果写在 common/scripted_effects/zz_hd_nbc_startup_effects.txt。
# 头衔、天朝循环在 on_game_start 里改好；宣战、雇主、京城等等原版与 TGP 开局初始化完，放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_nbc_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_nbc_after_lobby
	}
}

hd_nbc_on_game_start = {
	trigger = {
		OR = {
			AND = { game_start_date = 346.12.10 current_date = 346.12.10 }
			AND = { game_start_date = 404.3.24 current_date = 404.3.24 }
		}
		NOT = { exists = global_var:hd_nbc_startup_done }
	}
	effect = {
		hd_nbc_game_start_effect = yes
		set_global_variable = hd_nbc_startup_done
	}
}

hd_nbc_after_lobby = {
	trigger = {
		OR = {
			AND = { game_start_date = 346.12.10 current_date = 346.12.10 }
			AND = { game_start_date = 404.3.24 current_date = 404.3.24 }
		}
		NOT = { exists = global_var:hd_nbc_after_lobby_done }
	}
	effect = {
		hd_nbc_after_lobby_effect = yes
		set_global_variable = hd_nbc_after_lobby_done
	}
}
