# 1.20 base-clothing fallback; Faith/Rite permissions are loaded from history/faiths.
on_game_start = {
	on_actions = { hd_120_clothing_initialization }
}

hd_120_clothing_initialization = {
	effect = {
		if = {
			limit = { NOT = { exists = global_var:chinese_fashion } }
			set_global_variable = { name = chinese_fashion value = flag:tang }
		}
	}
}
