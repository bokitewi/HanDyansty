			limit = {

				hd_create_wolong_cao_cao_armor_effect = { OWNER = this }
			}
			set_global_variable = hd_wolong_cao_cao_armor_initialized
		}

		if = {
			limit = {
				NOT = { exists = global_var:hd_wolong_guan_yu_armor_initialized }

			}
			set_global_variable = hd_wolong_guan_yu_armor_initialized
		}

		if = {
			limit = {
				NOT = { exists = global_var:hd_wolong_liu_bei_armor_initialized }

	}
