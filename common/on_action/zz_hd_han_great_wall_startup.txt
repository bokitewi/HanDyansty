# Initializes the Eastern Han Great Wall on the route selected for the Han Dynasty map.
# Keep this in a project-owned child on_action so the compatibility override of
# vanilla game_start.txt does not need to duplicate vanilla startup logic.
on_game_start = {
	on_actions = {
		hd_initialize_han_great_wall
	}
}

hd_initialize_han_great_wall = {
	trigger = {
		has_tgp_dlc_trigger = yes
		OR = {
			game_start_date = 190.1.1 # 184 开局已于 2026-10-22 撤，改 190 讨董
			game_start_date = 218.1.1 # 218 年剧本同样修好东汉长城
			game_start_date = 346.12.10 # 南北朝剧本同上
			game_start_date = 404.3.24 # 南北朝剧本同上
			game_start_date = 304.11.20 # 304 刘渊剧本同上
			game_start_date = 253.3.1 # 253 姜维剧本同上
			game_start_date = 263.9.20 # 263 钟会伐蜀剧本同上
			game_start_date = 531.7.1 # 531 高欢剧本同上
			game_start_date = 370.2.13 # 370 苻坚剧本同上
			game_start_date = 22.11.13 # 22 光武中兴剧本同上
			game_start_date = 89.1.30 # 89 燕然勒石剧本同上（东汉长城）
			game_start_date = 617.6.23 # 617 隋末群雄剧本同上（隋修长城）
		}
		NOT = { exists = global_var:hd_han_great_wall_initialized }
	}

	effect = {
		# Hexi Corridor main line
		province:5097 = { set_great_district = the_great_wall_01_district } # Yang Pass
		province:5007 = { set_great_district = the_great_wall_01_district } # Dunhuang
		province:6769 = { set_great_district = the_great_wall_01_district } # Yumen Pass
		province:5805 = { set_great_district = the_great_wall_01_district } # Yumen
		province:6780 = { set_great_district = the_great_wall_01_district } # Jiuquan
		province:1566 = { set_great_district = the_great_wall_01_district } # Zhangye
		province:1564 = { set_great_district = the_great_wall_01_district } # Guzang
		province:1551 = { set_great_district = the_great_wall_01_district } # Lingju

		# Juyan branch
		province:5923 = { set_great_district = the_great_wall_01_district } # Eastern Juyan

		# Hetao and Dai Commandery line
		province:1577 = { set_great_district = the_great_wall_01_district } # Shuofang
		province:4725 = { set_great_district = the_great_wall_01_district } # Wuyuan
		province:1332 = { set_great_district = the_great_wall_01_district } # Yunzhong
		province:1338 = { set_great_district = the_great_wall_01_district } # Dingxiang
		province:1321 = { set_great_district = the_great_wall_01_district } # Wuzhou Fort
		province:1314 = { set_great_district = the_great_wall_01_district } # Gaoliu

		# Yanshan and Liaoxi line
		province:1283 = { set_great_district = the_great_wall_01_district } # Juyong Pass
		province:1245 = { set_great_district = the_great_wall_01_district } # Yuyang
		province:1258 = { set_great_district = the_great_wall_01_district } # Wuzhong
		province:2609 = { set_great_district = the_great_wall_01_district } # Junmi
		province:2558 = { set_great_district = the_great_wall_01_district } # Lingzhi
		province:3489 = { set_great_district = the_great_wall_01_district } # Linyu
		province:3482 = { set_great_district = the_great_wall_01_district } # Tuhe
		province:3480 = { set_great_district = the_great_wall_01_district } # Yangle

		set_global_variable = hd_han_great_wall_initialized
	}
}
