on_title_lost = { on_actions = { WJ_tenure_job_changed } }
on_title_destroyed = { on_actions = { WJ_tenure_job_changed } }
on_government_change = { on_actions = { WJ_tenure_job_changed } }
on_vassal_change = { on_actions = { WJ_tenure_job_changed } }
on_councillor_left = { on_actions = { WJ_tenure_job_changed } }
on_leave_council = { on_actions = { WJ_tenure_council_changed } }
on_fired_from_council = { on_actions = { WJ_tenure_council_changed } }
on_councillors_swapped = { on_actions = { WJ_tenure_swapped } }
on_death = { on_actions = { WJ_tenure_died } }
WJ_tenure_job_changed = {
 trigger = { has_variable = wj_tenure_realm }
 effect = { trigger_event = { id = WJ_tenure.3 days = 1 } }
}
WJ_tenure_council_changed = {
 trigger = { scope:councillor ?= { has_variable = wj_tenure_realm } }
 effect = { scope:councillor = { trigger_event = { id = WJ_tenure.3 days = 1 } } }
}
WJ_tenure_swapped = {
 effect = {
  if = { limit = { has_variable = wj_tenure_realm } trigger_event = { id = WJ_tenure.3 days = 1 } }
  if = { limit = { scope:councillor ?= { has_variable = wj_tenure_realm } } scope:councillor = { trigger_event = { id = WJ_tenure.3 days = 1 } } }
 }
}
WJ_tenure_died = { trigger = { has_variable = wj_tenure_realm } effect = { WJ_tenure_clear = yes } }
