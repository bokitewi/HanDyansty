# 新存档开局：为拥有庄园且达到规模条件的不见经传宗族分步提升 1–3 级。
# 在世成员 > 8，或家族总数 > 3（包含始祖家族）；每个宗族只执行一次。
# 必升 1 级，再以 60% 概率升第 2 级；成功后以 40% 概率升第 3 级。
on_game_start_after_lobby = {
	on_actions = {
		hd_dynasty_prestige_startup
	}
}

hd_dynasty_prestige_startup = {
	effect = {
		every_dynasty = {
			limit = {
				dynasty_prestige_level = 0
				any_dynasty_member = {
					is_alive = yes
					any_held_title = {
						is_noble_family_title = yes
					}
				}
				OR = {
					any_dynasty_member = {
						count > 8
						is_alive = yes
					}
					any_dynasty_house = {
						count > 3
					}
				}
			}
			add_dynasty_prestige_level = 1
			random = {
				chance = 60
				add_dynasty_prestige_level = 1
				random = {
					chance = 40
					add_dynasty_prestige_level = 1
				}
			}
		}
	}
}
