on_death = {
	on_actions = {
		hd_expedition_cleanup_on_death
	}
}

hd_expedition_cleanup_on_death = {
	effect = {
		if = {
			limit = { has_character_flag = hd_far_west_expedition_active }
			remove_global_variable = hd_far_west_exploration_active
			remove_variable = hd_far_west_failures
			remove_character_flag = hd_far_west_expedition_active
		}
		if = {
			limit = { has_character_flag = hd_yizhou_expedition_active }
			remove_global_variable = hd_yizhou_exploration_active
			remove_variable = hd_yizhou_failures
			remove_character_flag = hd_yizhou_expedition_active
		}
	}
}
