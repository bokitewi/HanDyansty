# Project-owned children append to vanilla on_actions without replacing them.
on_death = {
	on_actions = {
		hd_palace_crisis_on_death
		hd_anti_regent_on_death
		hd_anti_regent_invitee_on_death
		hd_historical_migration_on_death
	}
}

hd_palace_crisis_on_death = {
	trigger = {
		root = {
			this = character:liu_hong_3
		}
	}
	effect = {
		# on_death runs before title succession and may still report the
		# character as alive. This marker distinguishes the death callback.
		root = {
			add_character_flag = hd_palace_crisis_liu_hong_death_signal
			hd_try_start_palace_crisis_effect = yes
		}
	}
}

hd_anti_regent_on_death = {
	trigger = {
		root = {
			OR = {
				has_character_flag = hd_anti_regent_league_leader
				has_character_flag = hd_current_han_regent
			}
		}
	}
	effect = {
		if = {
			limit = { root = { has_character_flag = hd_current_han_regent } }
			root = { hd_handle_dong_regent_death_effect = yes }
		}
		else_if = {
			limit = { exists = global_var:hd_palace_crisis_completed }
			root = { hd_anti_regent_failure_effect = yes }
		}
		else = {
			# Leave Dong's founding/declaration timers and the surviving league intact.
			root = { remove_character_flag = hd_anti_regent_league_leader }
		}
	}
}

hd_anti_regent_invitee_on_death = {
	trigger = { root = { has_variable = hd_anti_regent_inviter } }
	effect = { root = { hd_anti_regent_invitation_resolved_effect = yes } }
}

hd_historical_migration_on_death = {
	trigger = {
		root = {
			OR = {
				AND = {
					this = character:cao_cao
					OR = {
						has_character_flag = hd_cao_early_story_pending
						has_character_flag = hd_cao_early_story_started
						has_character_flag = hd_cao_chenliu_migration_pending
						has_character_flag = hd_cao_chenliu_migration_started
					}
				}
				AND = {
					this = character:yuan_shao
					OR = {
						AND = {
							exists = global_var:hd_anti_regent_pending_leader
							global_var:hd_anti_regent_pending_leader = root
						}
						has_character_flag = hd_yuan_bohai_migration_started
						has_character_flag = hd_yuan_bohai_migration_pending
						has_character_flag = hd_yuan_bohai_arrived
						has_character_flag = hd_anti_regent_creation_pending
						has_variable = hd_palace_deposition_regent
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = { root = { this = character:cao_cao } }
			root = { hd_cao_abort_chenliu_migration_effect = yes }
		}
		else = { root = { hd_yuan_abort_bohai_migration_effect = yes } }
	}
}
