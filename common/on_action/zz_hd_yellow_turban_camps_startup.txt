# Configure the fixed 184 camps and their permanent, inheritable event troops.
on_game_start = {
	on_actions = {
		hd_setup_yellow_turban_historical_camps
	}
}

hd_setup_yellow_turban_historical_camps = {
	trigger = {
		game_start_date = 184.1.1
		current_date = 184.1.1
		NOT = { exists = global_var:hd_yellow_turban_historical_camps_initialized }
	}
	effect = {
		# Thirteen historical Yellow Turban wandering camps.
		character:guan_hai ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:zhang_lei_gong ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:xu_he ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:zhou_cang ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:pei_yuan_shao ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:zhao_hong_2 ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:han_zhong ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:sun_xia ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:peng_tuo ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:he_yi ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:huang_shao ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:liu_pi ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }
		character:gong_dou ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_wanderers } }

		# Independent bandit camps.
		character:chu_yan ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_brigands } }
		character:zhang_niu_jiao ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_brigands } }
		character:hd_zheng_jiang ?= { hd_yellow_turban_setup_armed_camp_effect = { PURPOSE = camp_purpose_brigands } }

		# Zhang Lu has a scholarly religious camp but no event troops.
		character:zhang_lu ?= { hd_yellow_turban_setup_unarmed_camp_effect = { PURPOSE = camp_purpose_scholars } }

		set_global_variable = hd_yellow_turban_historical_camps_initialized
	}
}
