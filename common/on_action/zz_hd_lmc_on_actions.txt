# 冒险者落地：旧部改编为事件兵（2026-11-01 boki）。效果见 zz_hd_lmc_effects.txt。
# 只追加，不占用原版钩子的 effect。两个钩子都只排队，延后 1 天统一执行（避开头衔/政体变更途中）。
on_government_change = { on_actions = { hd_lmc_on_government_change } }
on_title_gain = { on_actions = { hd_lmc_on_title_gain } }

# 冒险者政体换成别的政体。ROOT = 角色；scope:old_government = 原政体。
hd_lmc_on_government_change = {
	trigger = {
		is_alive = yes
		scope:old_government = { government_type_has_flag = government_is_landless_adventurer }
	}
	effect = { hd_lmc_queue_effect = yes }
}

# 兜底：拿到有地头衔时仍是冒险者政体（脚本授地不一定换政体）。ROOT = 新持有人；scope:title = 头衔。
hd_lmc_on_title_gain = {
	trigger = {
		is_alive = yes
		government_has_flag = government_is_landless_adventurer
		scope:title = { is_landless_type_title = no }
	}
	effect = { hd_lmc_queue_effect = yes }
}

# 延后执行。ROOT = 角色。
hd_lmc_convert = {
	trigger = {
		is_alive = yes
		has_character_flag = hd_lmc_pending
	}
	effect = {
		remove_character_flag = hd_lmc_pending
		if = {
			limit = { is_landed = yes }
			hd_lmc_convert_effect = yes
		}
		else = {
			debug_log = "hd_lmc: queued but not landed, skipped"
		}
	}
}
