# ============================================================================
# WJ 汉晋状态跨制度校正
# 复用原版迁廷、宾客和转封臣 on_action；不做旧档兼容或开局／年度扫描。
# ============================================================================

# on_leave_court 早于 on_join_court，延迟一天后才能读取角色最终归属。
on_leave_court = {
	on_actions = {
		WJ_han_jin_state_leave_court_action
	}
}

WJ_han_jin_state_leave_court_action = {
	trigger = {
		WJ_has_han_jin_conversion_state_trigger = yes
	}
	effect = {
		trigger_event = {
			on_action = WJ_han_jin_state_after_leave_action
			days = 1
		}
	}
}

WJ_han_jin_state_after_leave_action = {
	effect = {
		WJ_reconcile_han_jin_character_state_effect = yes
	}
}

# 原版宾客到访会提供 characters 列表，包含根角色及一起迁移的人物。
on_guest_arrived_from_pool = {
	on_actions = {
		WJ_han_jin_state_guest_arrival_action
	}
}

WJ_han_jin_state_guest_arrival_action = {
	effect = {
		every_in_list = {
			list = characters
			limit = { WJ_has_han_jin_conversion_state_trigger = yes }
			WJ_reconcile_han_jin_character_state_effect = yes
		}
	}
}

# 原版 on_vassal_change 的 root 是改变领主的封臣；整个子领地的 top_liege
# 会一起变化，因此连同其下级封臣、廷臣和宾客一次校正。
on_vassal_change = {
	on_actions = {
		WJ_han_jin_state_vassal_change_action
	}
}

WJ_han_jin_state_vassal_change_action = {
	trigger = {
		OR = {
			WJ_has_han_jin_conversion_state_trigger = yes
			any_courtier_or_guest = {
				WJ_has_han_jin_conversion_state_trigger = yes
			}
			any_vassal_or_below = {
				OR = {
					WJ_has_han_jin_conversion_state_trigger = yes
					any_courtier_or_guest = {
						WJ_has_han_jin_conversion_state_trigger = yes
					}
				}
			}
		}
	}
	effect = {
		WJ_reconcile_han_jin_subrealm_state_effect = yes
	}
}
