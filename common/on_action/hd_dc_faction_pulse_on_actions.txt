# Extra opportunities only; native faction creation remains enabled.
dynastic_cycle_yearly_pulse = {
	on_actions = { hd_dc_extra_faction_yearly }
}
hd_dc_extra_faction_yearly = {
	trigger = {
		exists = situation:dynastic_cycle
		situation:dynastic_cycle = { NOT = { has_variable = hd_dc_extra_faction_cycle_lock } }
	}
	effect = {
		situation:dynastic_cycle = {
			# Skip all participant/county work between ten-year rounds.
			set_variable = { name = hd_dc_extra_faction_cycle_lock value = yes years = 10 }
			every_situation_participant = {
				limit = { is_alive = yes is_landed = yes exists = primary_title }
				hd_dc_check_extra_faction_year_effect = yes
			}
		}
	}
}
