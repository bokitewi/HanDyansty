# 轲比能寇边（210–235，史实 218）
hd_hi_yearly_kebineng = {
	trigger = {
		has_game_rule = hd_hi_kebineng_on
		current_year >= 210
		NOT = { has_global_variable = hd_hi_done_kebineng }
	}
	effect = {
		if = {
			limit = {
				OR = {
					current_year > 235
					AND = {
						character:ke_bi_neng = { is_alive = no }
					}
				}
			}
			set_global_variable = { name = hd_hi_done_kebineng value = yes }
		}
		else_if = {
			limit = {
				character:ke_bi_neng ?= { hd_hi_leader_ready_trigger = yes }
			}
			random = {
				chance = {
					value = 5
					if = {
						limit = { current_year >= 216 }
						value = 15
					}
					if = {
						limit = { current_year >= 218 }
						value = 40
					}
					if = {
						limit = { current_year >= 235 }
						value = 100
					}
				}
				character:ke_bi_neng = { hd_hi_begin_effect = { KEY = kebineng } }
			}
		}
	}
}
