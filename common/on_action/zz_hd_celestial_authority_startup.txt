# 新存档开局：每位有领地的天朝制领主独立抽取集权等级，玩家与 AI 使用相同概率。
# 界面第 1–4 级对应 celestial_bureaucracy_0–3，权重依次为 20、50、20、10。
# 保留中华霸权持有者的阶段限制：动荡、崩溃、征服阶段抽中第 4 级时改为第 3 级。
on_game_start_after_lobby = {
	on_actions = {
		hd_celestial_authority_startup
	}
}

hd_celestial_authority_startup = {
	effect = {
		every_ruler = {
			limit = {
				is_landed = yes
				has_government = celestial_government
			}
			random_list = {
				20 = {
					if = { limit = { NOT = { has_realm_law = celestial_bureaucracy_0 } } add_realm_law_skip_effects = celestial_bureaucracy_0 } # 已有同级法律时再加会报错
				}
				50 = {
					if = { limit = { NOT = { has_realm_law = celestial_bureaucracy_1 } } add_realm_law_skip_effects = celestial_bureaucracy_1 } # 已有同级法律时再加会报错
				}
				20 = {
					if = { limit = { NOT = { has_realm_law = celestial_bureaucracy_2 } } add_realm_law_skip_effects = celestial_bureaucracy_2 } # 已有同级法律时再加会报错
				}
				10 = {
					if = {
						limit = {
							exists = title:h_china.holder
							this = title:h_china.holder
							exists = situation:dynastic_cycle
							situation:dynastic_cycle = {
								OR = {
									situation_current_phase = situation_dynastic_cycle_phase_instability
									situation_current_phase = situation_dynastic_cycle_phase_hd_collapse
									situation_current_phase = situation_dynastic_cycle_phase_instability_conquest
								}
							}
						}
						if = { limit = { NOT = { has_realm_law = celestial_bureaucracy_2 } } add_realm_law_skip_effects = celestial_bureaucracy_2 } # 已有同级法律时再加会报错
					}
					else = {
						if = { limit = { NOT = { has_realm_law = celestial_bureaucracy_3 } } add_realm_law_skip_effects = celestial_bureaucracy_3 } # 已有同级法律时再加会报错
					}
				}
			}
		}
	}
}
