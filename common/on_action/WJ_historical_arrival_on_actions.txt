# 无角色root的全局日期直接创建；不弹出场事件，也不覆盖主包既有effect。
on_game_start_after_lobby = { on_actions = { wj_hchar_check20 } }
yearly_global_pulse = { on_actions = { wj_hchar_check20 wj_hchar_expire_protection } }

# 原保护截止均为1月1日，四个固定指针即可清理，无全世界遍历。
wj_hchar_expire_protection = {
	trigger = { game_start_date = 184.1.1 current_year >= 204 current_year <= 224 }
	effect = { wj_hchar_expire_protection_effect = yes }
}

wj_hchar_check20 = {
	trigger = {
		game_start_date = 184.1.1
		current_year >= 184
		NOT = { has_global_variable = wj_hchar_all20_done }
	}
	effect = { wj_hchar_check20_effect = yes }
}
