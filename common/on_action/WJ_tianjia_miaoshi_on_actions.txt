# on_death is BEFORE loss of life/titles/variables; .009 consumes these traits three days later.
on_game_start_after_lobby = { on_actions = { WJ_miaoshi_living_start } }
on_title_gain = { on_actions = { WJ_miaoshi_accession } }
WJ_miaoshi_living_start = {
 effect = { every_ruler = { limit = { WJ_inner_emperor_trigger = yes } WJ_miaoshi_initialize_effect = yes } }
}
WJ_miaoshi_accession = {
 trigger = { WJ_inner_emperor_trigger = yes NOT = { has_variable = Dengji_sub_realm_var } }
 effect = {
  WJ_miaoshi_initialize_effect = yes
  if = {
   limit = { scope:title = root.primary_title exists = scope:previous_holder scope:previous_holder = { NOT = { dynasty = root.dynasty } } }
   add_trait = SanGong_KaiGuo
  }
 }
}
on_death = { on_actions = { WJ_miaoshi_death } }
WJ_miaoshi_death = { #皇帝死亡。盖棺定论
    trigger = {
        has_game_rule = MiaoShi_on
        WJ_inner_emperor_trigger = yes
        primary_heir = { dynasty = root.dynasty }
	}
    effect = {
        WJ_miaoshi_initialize_effect = yes
        save_scope_as = XianDi
        if = {#开国君主
            limit = {
                OR = {
                    has_trait = SanGong_KaiGuo 
                    has_trait = SanGong_ZhongXing
                } 
            }
            primary_heir = { trigger_event = { id = HuangDi_miaoshi.009 days = 3 } }
        }
        else_if = {#中兴之主
            limit = {
                prestige_level >= 4
                any_character_situation = { this = situation:dynastic_cycle }
                has_variable = SanGong_Zhongxing_var
                situation:dynastic_cycle ?= {
                    OR = { 
                        situation_current_phase = situation_dynastic_cycle_phase_stability_expansion
                        situation_current_phase = situation_dynastic_cycle_phase_stability_advancement
                    }
                }      
            }
            add_trait = SanGong_ZhongXing
            primary_heir = { trigger_event = { id = HuangDi_miaoshi.009 days = 3 } }
        }
        else_if = {#明君
            limit = {
                prestige_level >= 4
                sub_realm_size >= var:SanGong_XiongZhu_var
            }
            add_trait = SanGong_XiongZhu
            primary_heir = { trigger_event = { id = HuangDi_miaoshi.009 days = 3 } }
        }
        else_if = {#幼主
            limit = {
                age < 16
            }
            add_trait = SanGong_YouZhu
            primary_heir = { trigger_event = { id = HuangDi_miaoshi.009 days = 3 } }
        }
        else_if = {#昏君
            limit = {
                OR = {
                    sub_realm_size < var:SanGong_HunJun_var
                    AND = {
                        has_variable = SanGong_HunJun_var2
                        any_character_situation = { this = situation:dynastic_cycle }
                        situation:dynastic_cycle ?= {
                            OR = { 
                                situation_current_phase = situation_dynastic_cycle_phase_chaos            
                                situation_current_phase = situation_dynastic_cycle_phase_instability_conquest 
                                situation_current_phase = situation_dynastic_cycle_phase_instability       
                            }
                        }
                    }
                }
            }
            add_trait = SanGong_HunJun
            primary_heir = { trigger_event = { id = HuangDi_miaoshi.009 days = 3 } }
        }
        else = {#兜底：守成
            add_trait = SanGong_ShouCheng
            primary_heir = { trigger_event = { id = HuangDi_miaoshi.009 days = 3 } }
        }
    }
}
