on_death = {
	on_actions = {
		hd_wuhu_liao_on_death
	}
}

hd_wuhu_liao_on_death = {
	effect = {
		# 内附标记传给主继承人（修原版塞尔柱继承人丢旗标的问题）
		if = {
			limit = {
				has_variable = hd_wuhu_settled
				exists = primary_heir
			}
			primary_heir = {
				set_variable = {
					name = hd_wuhu_settled
					value = root.var:hd_wuhu_settled
				}
			}
		}
		# 僚人入蜀：决定已做、尚未宣战时，旗标传给玩家继承人（照原版）
		if = {
			limit = {
				has_character_flag = hd_liao_undertaking
				exists = player_heir
			}
			player_heir = { add_character_flag = hd_liao_undertaking }
		}
	}
}
