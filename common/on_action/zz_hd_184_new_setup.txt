# 184 剧本新增设置：
# 1. 黄巾三兄弟绰号（张角=大贤良师、张宝=地公将军、张梁=人公将军）；
# 2. 堂吉诃德绰号 + 开局加入维林诺冒险者营地（d_hd_weilinor，营地主 bokitewi）；
# 3. 诸葛亮成年得"卧龙"、司马懿成年得"冢虎"（挂原版 on_birthday_adulthood，追加模式）。
on_game_start = {
	on_actions = {
		hd_184_nickname_startup
	}
}

on_birthday_adulthood = {
	on_actions = {
		hd_adulthood_nicknames
	}
}

hd_184_nickname_startup = {
	trigger = {
		game_start_date = 184.1.1
		NOT = { exists = global_var:hd_184_nickname_initialized }
	}
	effect = {
		character:zhang_jiao ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_nickname = nick_hd_daxianliangshi }
				}
				give_nickname = nick_hd_daxianliangshi
			}
		}
		character:zhang_bao ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_nickname = nick_hd_digong_jiangjun }
				}
				give_nickname = nick_hd_digong_jiangjun
			}
		}
		character:zhang_liang ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_nickname = nick_hd_rengong_jiangjun }
				}
				give_nickname = nick_hd_rengong_jiangjun
			}
		}
		character:alonso_quijano ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_nickname = nick_hd_don_quixote }
				}
				give_nickname = nick_hd_don_quixote
			}
			if = {
				limit = {
					is_alive = yes
					character:bokitewi ?= { is_alive = yes }
				}
				set_employer = character:bokitewi
			}
		}
		set_global_variable = hd_184_nickname_initialized
	}
}

hd_adulthood_nicknames = {
	effect = {
		character:zhu_ge_liang ?= {
			if = {
				limit = {
					is_adult = yes
					NOT = { has_nickname = nick_hd_wolong }
				}
				give_nickname = nick_hd_wolong
			}
		}
		character:si_ma_yi ?= {
			if = {
				limit = {
					is_adult = yes
					NOT = { has_nickname = nick_hd_zhonghu }
				}
				give_nickname = nick_hd_zhonghu
			}
		}
	}
}
