on_title_gain_inheritance = { on_actions = { WJ_cr_inherit_king } }
on_title_gain = { on_actions = { WJ_cr_gain_empire } }
on_vassal_change = { on_actions = { WJ_cr_independent_empire } }
on_game_start_after_lobby = { on_actions = { WJ_cr_existing_empires } }
WJ_cr_gain_empire = {
 trigger = { is_alive = yes government_has_flag = government_is_celestial exists = scope:title scope:title = { WJ_cr_local_empire_title = yes } }
 effect = {
  # Native transactions may set primary title after on_title_gain; dispatch after the transaction.
  WJ_cr_queue_empire_settlement = yes
 }
}
WJ_cr_independent_empire = {
 trigger = { WJ_cr_empire_unsettled = yes WJ_cr_empire_in_china = yes }
 effect = { WJ_cr_queue_empire_settlement = yes }
}
WJ_cr_existing_empires = {
 effect = { every_independent_ruler = {
  limit = { WJ_cr_emperor = yes primary_title = { WJ_cr_local_empire_title = yes } }
  set_variable = wj_cr_empire_settled
 } }
}
WJ_cr_inherit_king = {
 trigger = { is_alive = yes government_has_flag = government_is_celestial exists = scope:title scope:title = { tier = tier_kingdom has_variable = wj_cr_hereditary_king holder = root } }
 effect = { WJ_title_style_reconcile = yes WJ_miaoshi_initialize_effect = yes }
}
