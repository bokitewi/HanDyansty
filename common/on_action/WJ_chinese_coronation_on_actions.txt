 trigger = { NOT = { has_global_variable = wj_cr_legacy_oath_cleaned } }
 effect = {
  set_global_variable = { name = wj_cr_legacy_oath_cleaned value = 1 }
  every_living_character = {
   limit = { OR = {
    AND = { has_variable = diplomacy_oath_1 NOT = { has_variable = diplomacy_oath_1_timer } }
    AND = { has_variable = stewardship_oath_1 NOT = { has_variable = stewardship_oath_1_timer } }
   } }
   if = { limit = { has_variable = diplomacy_oath_1 NOT = { has_variable = diplomacy_oath_1_timer } } remove_variable = diplomacy_oath_1 }
   if = { limit = { has_variable = stewardship_oath_1 NOT = { has_variable = stewardship_oath_1_timer } } remove_variable = stewardship_oath_1 }
  }
 }
