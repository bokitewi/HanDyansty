# 304 年剧本「左国称王」史实主线入口（docs/304刘渊剧本_计划.md 第 10、13、14 节）。全部带 304.11.20 闸门，其余开局不受影响。
# 效果在 common/scripted_effects/zz_hd_304_history_effects.txt，事件在 events/zz_hd_304_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_304_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hd_nbc_pulse），到 320 年为止
hd_304_history_start = {
	trigger = {
		game_start_date = 304.11.20
		NOT = { exists = global_var:hd_304_pulse_started }
	}
	effect = {
		set_global_variable = hd_304_pulse_started
		title:h_china = { trigger_event = { on_action = hd_304_pulse days = 1 } }
	}
}

hd_304_pulse = {
	trigger = {
		game_start_date = 304.11.20
	}
	effect = {
		hd_304_pulse_effect = yes
		if = {
			limit = { current_date < 320.1.1 }
			title:h_china = { trigger_event = { on_action = hd_304_pulse days = 30 } }
		}
	}
}
