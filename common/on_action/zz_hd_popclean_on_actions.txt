# 开局人口清除：新开局大厅结束后自动执行一次。
on_game_start_after_lobby = {
	on_actions = { hd_popclean_game_start }
}

# on_action 里没有随机数上下文（integer_range / days 区间会失效），
# 所以交给一名角色的隐藏事件去跑；全局变量保证只排一次。
hd_popclean_game_start = {
	effect = {
		every_player = {
			if = {
				limit = { NOT = { has_global_variable = hd_popclean_queued } }
				set_global_variable = { name = hd_popclean_queued value = yes days = 30 }
				trigger_event = hd_popclean.2
			}
		}
		if = {
			limit = { NOT = { has_global_variable = hd_popclean_queued } }
			every_ruler = {
				limit = { is_alive = yes }
				if = {
					limit = { NOT = { has_global_variable = hd_popclean_queued } }
					set_global_variable = { name = hd_popclean_queued value = yes days = 30 }
					trigger_event = hd_popclean.2
				}
			}
		}
	}
}
