on_title_gain = { on_actions = { hd_special_military_contract_reconcile_on_action } }
on_title_lost = { on_actions = { hd_special_military_contract_reconcile_on_action } }
on_government_change = { on_actions = { hd_special_military_contract_reconcile_on_action } }
quarterly_playable_pulse = { on_actions = { hd_special_military_contract_reconcile_on_action } }

# These hooks have the affected character as root. Never run a global realm scan here.
hd_special_military_contract_reconcile_on_action = {
	trigger = {
		OR = {
			vassal_contract_has_flag = hd_tuntian
			vassal_contract_has_flag = hd_military_colony
			has_character_flag = hd_special_contract_timeout_owned
			has_realm_law = hd_military_colony_succession_law
		}
	}
	effect = { hd_special_military_contract_reconcile_effect = yes }
}
