# 22 年剧本「光武中兴」史实主线入口（/mnt/project-files/22光武中兴剧本_计划.md 第 10、13 节）。全部带 22.11.13 闸门，其余开局不受影响。
# 效果在 common/scripted_effects/zz_hd_22_history_effects.txt，事件在 events/zz_hd_22_events.txt。

on_game_start_after_lobby = {
	on_actions = {
		hd_22_history_start
	}
}

# 开局第 1 天起，每 30 天由 title:h_china 跑一次心跳（写法同 hd_370_pulse），到 57 年为止（卒年心跳要跑到比单于 56 年，复核必改 9）
hd_22_history_start = {
	trigger = {
		hd_history_events_enabled_trigger = yes
		game_start_date = 22.11.13
		NOT = { exists = global_var:hd_22_pulse_started }
	}
	effect = {
		set_global_variable = hd_22_pulse_started
		title:h_china = { trigger_event = { on_action = hd_22_pulse days = 1 } }
	}
}

hd_22_pulse = {
	trigger = {
		game_start_date = 22.11.13
	}
	effect = {
		hd_22_pulse_effect = yes
		if = {
			limit = { current_date < 57.1.1 }
			title:h_china = { trigger_event = { on_action = hd_22_pulse days = 30 } }
		}
	}
}
