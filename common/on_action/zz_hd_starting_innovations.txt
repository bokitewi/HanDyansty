# ==========================================================================
# 汉传承（heritage_chinese）文化开局革新
# 通过子 on_action 追加到原版 on_game_start_after_lobby（不覆盖原文件）。
# 写法与原版一致：every_culture_global + add_innovation（参考原版 game_start.txt
# 中 heritage_arabic 解锁 innovation_ghilman 的写法）。
# 注意：床弩/夯土/坞堡的 _chinese key 是原版基础革新的中文显示变体
# （asset 变体），实际添加使用基础革新 key：
#   innovation_catapult_chinese -> innovation_catapult
#   innovation_motte_chinese    -> innovation_motte
#   innovation_burhs_chinese    -> innovation_burhs
# ==========================================================================

on_game_start_after_lobby = {
	on_actions = {
		hd_starting_innovations_on_game_start
	}
}

hd_starting_innovations_on_game_start = {
	trigger = {
		NOT = { exists = global_var:hd_starting_innovations_initialized }
	}
	effect = {
		every_culture_global = {
			limit = {
				has_cultural_pillar = heritage_chinese
			}
			# 部落时代 · 军事
			if = {
				limit = { NOT = { has_innovation = innovation_mustering_grounds } }
				add_innovation = innovation_mustering_grounds	# 校阅场
			}
			if = {
				limit = { NOT = { has_innovation = innovation_quilted_armor } }
				add_innovation = innovation_quilted_armor		# 棉甲
			}
			if = {
				limit = { NOT = { has_innovation = innovation_catapult } }
				add_innovation = innovation_catapult			# 床弩（中文变体 innovation_catapult_chinese）
			}
			if = {
				limit = { NOT = { has_innovation = innovation_motte } }
				add_innovation = innovation_motte				# 夯土（中文变体 innovation_motte_chinese）
			}
			if = {
				limit = { NOT = { has_innovation = innovation_bannus } }
				add_innovation = innovation_bannus				# 军队征召
			}
			if = {
				limit = { NOT = { has_innovation = innovation_barracks } }
				add_innovation = innovation_barracks			# 兵营
			}
			# 部落时代 · 民政
			if = {
				limit = { NOT = { has_innovation = innovation_currency_01 } }
				add_innovation = innovation_currency_01			# 货币
			}
			# 中世纪早期 · 军事
			if = {
				limit = { NOT = { has_innovation = innovation_burhs } }
				add_innovation = innovation_burhs				# 坞堡（中文变体 innovation_burhs_chinese）
			}
			if = {
				limit = { NOT = { has_innovation = innovation_house_soldiers } }
				add_innovation = innovation_house_soldiers		# 家兵
			}
			# 中世纪早期 · 民政
			if = {
				limit = { NOT = { has_innovation = innovation_hereditary_rule } }
				add_innovation = innovation_hereditary_rule		# 世袭统治
			}
			if = {
				limit = { NOT = { has_innovation = innovation_manorialism } }
				add_innovation = innovation_manorialism			# 庄园制度
			}
			# 天朝 DLC（TGP）
			if = {
				limit = { NOT = { has_innovation = innovation_waterworks } }
				add_innovation = innovation_waterworks			# 供水系统
			}
			if = {
				limit = { NOT = { has_innovation = innovation_coking } }
				add_innovation = innovation_coking				# 炼焦工艺
			}
			if = {
				limit = { NOT = { has_innovation = innovation_pharmacopoeia } }
				add_innovation = innovation_pharmacopoeia		# 药典
			}
			if = {
				limit = { NOT = { has_innovation = innovation_sericulture } }
				add_innovation = innovation_sericulture			# 蚕桑业
			}
			if = {
				limit = { NOT = { has_innovation = innovation_champa_rice } }
				add_innovation = innovation_champa_rice			# 占城稻
			}
			if = {
				limit = { NOT = { has_innovation = innovation_compass } }
				add_innovation = innovation_compass				# 司南
			}
		}
		# ------------------------------------------------------------------
		# 百越传承（heritage_baiyue）—— 象兵革新
		# 【修改包新增】百越文化开局自带象骑兵革新。
		#
		# 为何走脚本而不走 history/cultures 文件：
		#   · 原版 history/cultures 的日期块最早为 865 年，在 184 剧本中不生效；
		#   · history/cultures/_culture.info 明示「若某文化有独立文件，则只用
		#     独立文件，不读传承组文件」——viet 有独立 viet.txt，会被漏掉。
		#   脚本遍历可覆盖全部百越文化（含 viet）。
		# ------------------------------------------------------------------
		every_culture_global = {
			limit = {
				has_cultural_pillar = heritage_baiyue
			}
			if = {
				limit = { NOT = { has_innovation = innovation_elephantry } }
				add_innovation = innovation_elephantry		# 象兵
			}
		}

		set_global_variable = hd_starting_innovations_initialized
	}
}
