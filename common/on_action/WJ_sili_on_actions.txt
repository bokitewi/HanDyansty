# Native title_on_actions.txt: gain/lost root is holder; capital root is barony.
# Append child handlers, do not overwrite native effects. No periodic scan.
on_title_gain = { on_actions = { WJ_sili_title_changed } }
on_title_lost = { on_actions = { WJ_sili_title_changed } }
on_vassal_change = { on_actions = { WJ_sili_identity_changed } }
on_government_change = { on_actions = { WJ_sili_identity_changed } }
on_liege_government_change = { on_actions = { WJ_sili_identity_changed } }
on_realm_capital_change = { on_actions = { WJ_sili_capital_changed } }

WJ_sili_title_changed = {
	trigger = {
		is_alive = yes
		OR = {
			vassal_contract_has_flag = WJ_celestial_province_sili
			AND = {
				scope:title.tier >= tier_kingdom
				WJ_han_jin_law_trigger = yes
			}
		}
	}
	effect = { trigger_event = { on_action = WJ_sili_after_change days = 1 } }
}
WJ_sili_identity_changed = {
	trigger = {
		is_alive = yes
		OR = {
			vassal_contract_has_flag = WJ_celestial_province_sili
			any_vassal_or_below = { vassal_contract_has_flag = WJ_celestial_province_sili }
		}
	}
	effect = { trigger_event = { on_action = WJ_sili_after_change days = 1 } }
}
WJ_sili_capital_changed = {
	trigger = {
		holder ?= {
			is_alive = yes
			this = top_liege
			government_has_flag = government_is_celestial
			WJ_han_jin_law_trigger = yes
		}
	}
	effect = {
		holder ?= { trigger_event = { on_action = WJ_sili_after_change days = 1 } }
	}
}
WJ_sili_after_change = {
	trigger = { is_alive = yes }
	effect = {
		if = {
			limit = { character:dong_zhuo_2 = { has_character_flag = hd_palace_sili_trace_active } }
			debug_log = "HD_HISTORY SILI_TRACE SILI_RECONCILE_BEGIN"
			debug_log_scopes = yes
		}
		WJ_sili_reconcile_subrealm_effect = yes
		if = {
			limit = { character:dong_zhuo_2 = { has_character_flag = hd_palace_sili_trace_active } }
			debug_log = "HD_HISTORY SILI_TRACE SILI_RECONCILE_DONE"
		}
	}
}
