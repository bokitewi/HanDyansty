# 263「钟会伐蜀」剧本开局（魏景元四年八月朔）。全部以 game_start_date = 263.9.20 为闸门，其他开局不受影响。
# 效果写在 common/scripted_effects/zz_hd_263_startup_effects.txt 与 zz_hd_263_history_effects.txt，
# 计划 /mnt/project-files/263钟会伐蜀剧本_计划.md。

on_game_start = {
	on_actions = {
		hd263_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd263_after_lobby
	}
}

on_death = {
	on_actions = {
		hd263_on_death
	}
}

hd263_on_game_start = {
	trigger = {
		game_start_date = 263.9.20
		current_date = 263.9.20
		NOT = { exists = global_var:hd263_game_start_ran }
	}
	effect = {
		set_global_variable = hd263_game_start_ran
		hd263_game_start_effect = yes
	}
}

hd263_after_lobby = {
	trigger = {
		game_start_date = 263.9.20
		NOT = { exists = global_var:hd263_after_lobby_done }
	}
	effect = {
		hd_sg_ptr_refresh_effect = yes
		set_global_variable = hd263_after_lobby_done
		hd263_after_lobby_effect = yes
		title:h_china = { trigger_event = { on_action = hd263_pulse days = 30 } }
	}
}

# 每 30 天一次，到 266.6.1 为止：伐蜀至禅代的史实主线（253 局接力时到 270.1.1，见 zz_hd_263_relay.txt）
hd263_pulse = {
	trigger = {
		hd263_chain_on_trigger = yes
	}
	effect = {
		hd_sg_ptr_refresh_effect = yes
		hd263_pulse_effect = yes
		if = {
			limit = {
				OR = {
					AND = { NOT = { hd263_relay_trigger = yes } current_date < 266.6.1 }
					AND = { hd263_relay_trigger = yes current_date < 270.1.1 }
				}
			}
			title:h_china = { trigger_event = { on_action = hd263_pulse days = 30 } }
		}
	}
}

# 司马昭死后第 3 天：晋国与河内交司马炎，司马炎接任权臣（S7 的通用路径）。Root：死者
hd263_on_death = {
	trigger = {
		hd263_chain_on_trigger = yes
		this = character:si_ma_zhao
		NOT = { exists = global_var:hd263_s8_done }
		# 253 局接力：司马昭须是天子封臣兼权臣才由司马炎接任（玩家司马师仍为权臣时不顶替）
		OR = {
			NOT = { hd263_relay_trigger = yes }
			AND = {
				NOT = { exists = global_var:hd263_zhao_dead } # S7 接力分支已自行交接
				title:h_china.var:hd_court_regent ?= this
				liege ?= title:h_china.holder
			}
		}
	}
	effect = {
		set_global_variable = hd263_zhao_dead
		title:h_china = { trigger_event = { on_action = hd263_yan_succession days = 3 } }
	}
}

hd263_yan_succession = {
	trigger = {
		hd263_chain_on_trigger = yes
	}
	effect = {
		hd263_yan_succession_route_effect = yes
	}
}
