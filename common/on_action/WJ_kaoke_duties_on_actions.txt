# 五年考课：独立军功与生日年度事件入口。
# 只读来源（相对 CK3 game）：
# common/on_action/combat_on_actions.txt — combat side / every_side_commander / every_side_knight.
# common/on_action/yearly_on_actions.txt — actual yearly_playable_pulse definition.
# common/on_action/_on_actions.info — additive on_actions;
# random_events selects one event; an on_action effect is a separate chain.
# events/court_events/01_ep3_court_events.txt — every_in_list + remove_from_list.
# 主MOD events/rp_merit_events（无后缀）仅参考真实参与域，不接入任何 RP 事件。
# 胜利每人每场 +10、失败 -10；参战次数均 +1，胜败分类与军功分限幅由 core 处理。
# 不计未亲临的 side_primary_participant；既有 XP、战斗效果和事件池不覆盖。

on_combat_end_winner = {
	on_actions = { WJ_kaoke_duties_battle_winner }
}

WJ_kaoke_duties_battle_winner = {
	effect = {
		# Temporary script-chain list, cleared per combat side; no character flags.
		every_in_list = {
			list = wj_kaoke_battle_participants
			remove_from_list = wj_kaoke_battle_participants
		}
		every_side_commander = {
			limit = { NOT = { is_in_list = wj_kaoke_battle_participants } }
			add_to_list = wj_kaoke_battle_participants
		}
		every_side_knight = {
			limit = { NOT = { is_in_list = wj_kaoke_battle_participants } }
			add_to_list = wj_kaoke_battle_participants
		}
		every_in_list = {
			list = wj_kaoke_battle_participants
			WJ_kaoke_reconcile_effect = yes
			if = {
				limit = {
					WJ_kaoke_eligible_trigger = yes
					has_variable = wj_kaoke_start
					has_variable = wj_kaoke_realm
					var:wj_kaoke_realm = top_liege.primary_title
				}
				# Core records participation +1, the corresponding outcome, and score.
				WJ_kaoke_battle_effect = { WON = 1 POINTS = 10 }
			}
			remove_from_list = wj_kaoke_battle_participants
		}
	}
}

# 使用当前本体实际生日入口；不采用旧.info中的 on_yearly_playable 名称。
yearly_playable_pulse = {
	on_actions = { WJ_kaoke_duties_yearly }
}

WJ_kaoke_duties_yearly = {
	trigger = {
		OR = {
			has_variable = wj_kaoke_start
			WJ_kaoke_eligible_trigger = yes
		}
	}
	# Birthday core due/reconcile owns today's changes. Only dispatch tomorrow;
	# do not combine an effect and random_events in parallel chains here.
	effect = {
		if = {
			# AI 不进年度公务事件，次日直接按性格权重随机加分。
			limit = { is_ai = yes }
			trigger_event = {
				on_action = WJ_kaoke_duties_ai_dispatch
				days = 1
			}
		}
		else = {
			# 原版拜占庭总督任务写法：每年 70%，触发后 3 年冷却期内降为 50%。
			random = {
				chance = {
					value = 70
					if = {
						limit = { has_character_flag = wj_kaoke_duties_cooldown }
						subtract = 20
					}
				}
				trigger_event = {
					on_action = WJ_kaoke_duties_yearly_dispatch
					days = 1
				}
			}
		}
	}
}

WJ_kaoke_duties_ai_dispatch = {
	trigger = {
		is_ai = yes
		WJ_kaoke_eligible_trigger = yes
		has_variable = wj_kaoke_start
		WJ_kaoke_days_value < 1825
		has_variable = wj_kaoke_realm
		var:wj_kaoke_realm = top_liege.primary_title
	}
	effect = { WJ_kaoke_ai_annual_effect = yes }
}

WJ_kaoke_duties_yearly_dispatch = {
	trigger = {
		is_ai = no
		WJ_kaoke_eligible_trigger = yes
		has_variable = wj_kaoke_start
		WJ_kaoke_days_value < 1825
		has_variable = wj_kaoke_realm
		var:wj_kaoke_realm = top_liege.primary_title
	}
	# No 0 entry: exactly one eligible dilemma each annual pulse.
	# Snapshot start/realm/epoch in event immediate, never in a parallel effect.
	random_events = {
		100 = wj_kaoke_duties.1001
		100 = wj_kaoke_duties.1002
		100 = wj_kaoke_duties.1003
	}
}
