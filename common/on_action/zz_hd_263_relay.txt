# 253「姜维北伐」局接上 263「钟会伐蜀」后半段（三国剧本事件衔接，2026-10-18）。只在 253 开局起作用（218 局接上 253 后也起作用，见 zz_hd_253_relay.txt），263 开局不受影响。
# 263.8.1 起设 hd263_relay_on 并点起 hd263_pulse（zz_hd_263_startup.txt）；伐蜀由 253 发兵，出降起走 263 的 S2–S8。
# 效果在 common/scripted_effects/zz_hd_263_relay_effects.txt，计划 /mnt/project-files/三国剧本事件衔接/253接263_计划.md。

on_game_start_after_lobby = {
	on_actions = {
		hd263_relay_start
	}
}

hd263_relay_start = {
	trigger = {
		game_start_date = 253.3.1
		NOT = { exists = global_var:hd263_relay_watch_started }
	}
	effect = {
		set_global_variable = hd263_relay_watch_started
		title:h_china = { trigger_event = { on_action = hd263_relay_watch days = 30 } }
	}
}

# 每 30 天看一次：到 263.8.1（早于 253 最早发兵的 263.9.1）启动接力
hd263_relay_watch = {
	trigger = {
		hd253_chain_on_trigger = yes # 218 局由 hd253_relay_activate_effect 排起
		NOT = { exists = global_var:hd263_relay_on }
	}
	effect = {
		if = {
			limit = { current_date >= 263.8.1 }
			hd263_relay_activate_effect = yes
		}
		else_if = {
			limit = { current_date < 270.1.1 }
			title:h_china = { trigger_event = { on_action = hd263_relay_watch days = 30 } }
		}
	}
}
