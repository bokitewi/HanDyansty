# 370 年剧本「王猛伐燕」（370.2.13，晋太和五年、秦建元六年、燕建熙十一年正月朔）开局。计划见 docs/370苻坚剧本_计划.md 第 6、14 节。
# 全部以 game_start_date = 370.2.13 为闸门，其余开局不受影响。效果写在 common/scripted_effects/zz_hd_370_startup_effects.txt。
# 头衔、天朝循环、衣冠南渡在 on_game_start 里做；宣战、年号、国号、权臣等原版与 TGP 开局初始化完，放在 on_game_start_after_lobby。

on_game_start = {
	on_actions = {
		hd_370_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		hd_370_after_lobby
	}
}

hd_370_on_game_start = {
	trigger = {
		game_start_date = 370.2.13
		current_date = 370.2.13
		NOT = { exists = global_var:hd_370_startup_done }
	}
	effect = {
		# 先设：hd_court_regent_effects.txt、hd_dc_kit_ten_effects.txt 凭它先 granted 授丞相
		set_global_variable = hd_370_startup_done
		hd_370_game_start_effect = yes
	}
}

hd_370_after_lobby = {
	trigger = {
		game_start_date = 370.2.13
		current_date = 370.2.13
		NOT = { exists = global_var:hd_370_after_lobby_done }
	}
	effect = {
		hd_370_after_lobby_effect = yes
		set_global_variable = hd_370_after_lobby_done
	}
}
