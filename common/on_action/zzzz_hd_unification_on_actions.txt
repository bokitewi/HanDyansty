# Append child on-actions only. Never replace existing native effect blocks.
on_game_start_after_lobby = {
	on_actions = { hd_unification_startup }
}
quarterly_playable_pulse = {
	on_actions = { hd_unification_refresh_character }
}
on_title_gain = {
	on_actions = { hd_unification_refresh_character }
}
on_title_gain_inheritance = {
	on_actions = { hd_unification_refresh_character }
}

# No root in startup; scan only independent rulers, not every living character.
hd_unification_startup = {
	effect = {
		every_independent_ruler = {
			limit = {
				is_ai = yes
				hd_unification_celestial_attacker_trigger = yes
			}
			hd_unification_ai_ensure_story_effect = yes
		}
	}
}

# Root is a character for the three registered character hooks above.
# Quarterly refresh bootstraps existing saves and catches later independence
# or government changes without a permanent character-flag/loop lock.
hd_unification_refresh_character = {
	trigger = {
		is_ai = yes
		hd_unification_celestial_attacker_trigger = yes
	}
	effect = { hd_unification_ai_ensure_story_effect = yes }
}
