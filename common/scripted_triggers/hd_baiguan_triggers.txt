# Current main-MOD ownership test, without the removed official-system law gate.
# The HUD calls this often: cheap checks precede the held-title fallback.
hd_baiguan_owner_trigger = {
	is_alive = yes
	is_ruler = yes
	is_independent_ruler = yes
	government_has_flag = government_is_celestial
	highest_held_title_tier >= tier_empire
	OR = {
		primary_title ?= {
			tier >= tier_empire
			is_landless_type_title = no
			is_noble_family_title = no
		}
		any_held_title = {
			tier >= tier_empire
			is_landless_type_title = no
			is_noble_family_title = no
		}
	}
}
