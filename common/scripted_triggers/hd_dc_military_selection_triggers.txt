# The social snapshot is necessary, not sufficient, for MIL-BOOT. P016 still
# needs a separately verified normal-army count; never call this full eligibility.
hd_dc_military_social_snapshot_valid_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = yes
	is_imprisoned = no
	NOT = { has_trait = incapable }
	exists = title:h_china.holder
	exists = liege
	liege = { government_allows = obedience }
	obedience_target = title:h_china.holder
	var:hd_dc_obedience_snapshot_target = title:h_china.holder
	var:hd_dc_obedience_snapshot_year = current_year
	situation:dynastic_cycle = {
		var:hd_dc_military_snapshot_ruler = title:h_china.holder
		var:hd_dc_military_snapshot_year = current_year
		has_variable = hd_dc_obedience_cutoff
	}
	var:hd_dc_obedience_snapshot <= situation:dynastic_cycle.var:hd_dc_obedience_cutoff
	var:hd_dc_military_opinion_snapshot <= -30
	opinion = { target = title:h_china.holder value <= -30 }
	has_vassal_stance = belligerent
	OR = { has_trait = ambitious has_trait = brave has_trait = arrogant }
	any_liege_or_above = { this = title:h_china.holder }
	NOT = { is_at_war_with = title:h_china.holder }
}
