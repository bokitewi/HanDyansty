# 全线共用的触发器在 zz_hd_j16_triggers.txt（核心，已冻结）。本文件只放 4-B 节点用到的、前缀 hd_j16_b4_ 的小触发器。

# 石氏之主：在世、独立、上党石氏（dynasty:shi_2）。Scope：角色
hd_j16_b4_shi_lord_trigger = {
	is_alive = yes
	is_independent_ruler = yes
	dynasty ?= dynasty:shi_2
}

# 冉魏之主：在世、独立、内黄冉氏（dynasty:neihuang_ran_dyn，第 2 期已改挂）。Scope：角色
hd_j16_b4_ran_lord_trigger = {

	is_independent_ruler = yes
	is_landed = yes
	dynasty ?= dynasty:neihuang_ran_dyn
}

# $A$、$B$ 两人所在的顶层国土接壤（两人可以是封臣，按各自 top_liege 判）。$A$、$B$ 须已存在。Scope：任意
hd_j16_b4_adjacent_trigger = {
	$A$.top_liege = {
		NOT = { this = $B$.top_liege }
		any_neighboring_top_liege_realm_owner = { this = $B$.top_liege }
	}
}

# $WHO$ 在 $LORD$ 一方（是其廷臣或下属，且不是 $LORD$ 本人）。$WHO$ 已死或不存在为假。Scope：任意
hd_j16_b4_under_trigger = {
	$WHO$ ?= {
		is_alive = yes
		NOT = { this = $LORD$ }
		OR = {
			top_liege = $LORD$
			employer ?= $LORD$
		}
	}
}

# 晋臣：$WHO$ 在世、有地，最高领主是晋天子（h_china 持有者，河内司马氏），不是天子本人。Scope：任意
hd_j16_b4_jin_vassal_trigger = {
	exists = title:h_china.holder
	title:h_china.holder = { house ?= house:henei_sima }
	$WHO$ ?= {
		is_alive = yes
		is_landed = yes
		top_liege = title:h_china.holder
		NOT = { this = title:h_china.holder }
	}
}
