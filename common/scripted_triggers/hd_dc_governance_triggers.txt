# War scope. A territorial target inside the actual regional county set, or
# combat between two regional principals, interrupts regional stability.
hd_dc_war_disrupts_governance_trigger = {
	OR = {
		hd_dc_is_mandate_war_trigger = yes
		any_in_list = {
			list = target_titles
			OR = {
				has_variable = hd_dc_core_county_v1
				any_de_jure_county = { has_variable = hd_dc_core_county_v1 }
			}
		}
		AND = {
			primary_attacker = { any_character_situation = { this = situation:dynastic_cycle } }
			primary_defender = { any_character_situation = { this = situation:dynastic_cycle } }
		}
	}
}

# Main instance. A missing war receipt or incomplete audit cannot prove peace.
hd_dc_governance_peace_trigger = {
	hd_dc_all_core_wars_resolved_trigger = yes
	exists = var:hd_dc_region
	var:hd_dc_region = {
		hd_dc_snapshot_valid_trigger = yes
		NOT = {
			any_situation_participant = {
				any_character_war = { hd_dc_war_disrupts_governance_trigger = yes }
			}
		}
	}
}

# Each ending adapter must audit and publish its real security/obligation
# result. Neither a political victory nor zero Q constitutes that audit.
hd_dc_governance_accounts_ready_trigger = {
	var:hd_dc_record_kind = flag:instance
	has_variable = hd_dc_political_result
	has_variable = hd_dc_governance_audit_version
	var:hd_dc_governance_audit_version = var:hd_dc_obligation_version
	var:hd_dc_open_obligations = 0
	NOT = { has_variable = hd_dc_obligation_count_error }
	NOT = { has_variable = hd_dc_reward_committed }
	NOT = { has_variable = hd_dc_points_committed }
	NOT = {
		any_in_list = {
			variable = hd_dc_actor_stories
			OR = { has_variable = hd_dc_pending_proposal has_variable = hd_dc_pending_action }
		}
	}
	hd_dc_governance_peace_trigger = yes
}

# ROY-E3 explicitly caps suppression alone at the low award. The current
# resolution adapter proves victory and ROY-01 fulfillment, but has no actual
# military/pension reform audit. Open this gate only with that future adapter's
# real completed operations; elapsed peace and Q are not reform evidence.
hd_dc_governance_substantive_settlement_trigger = {
	NOT = { var:hd_dc_political_result = flag:roy_e3 }
}

hd_dc_governance_full_fulfillment_trigger = {
	hd_dc_governance_accounts_ready_trigger = yes
	hd_dc_governance_substantive_settlement_trigger = yes
	var:hd_dc_core_security_completed = yes
	var:hd_dc_full_fulfillment_completed = yes
	NOT = { var:hd_dc_major_uncorrected_breach = yes }
	NOT = {
		any_in_list = {
			variable = hd_dc_actor_stories
			exists = var:hd_dc_ritual_pledge
			NOT = { var:hd_dc_ritual_pledge_result = flag:accepted }
		}
	}
}

# Character scope, current living hegemon only. This is authority to choose an
# early final award, not ownership of predecessor debts or historical Q.
hd_dc_governance_reviewer_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	this = title:h_china.holder
}

hd_dc_governance_window_valid_trigger = {
	hd_dc_governance_reviewer_trigger = yes
	exists = scope:hd_dc_governance_instance
	scope:hd_dc_governance_instance = {
		var:hd_dc_instance_id = scope:hd_dc_governance_id
		var:hd_dc_obligation_version = scope:hd_dc_governance_obligation_version
		var:hd_dc_war_epoch = scope:hd_dc_governance_war_epoch
		hd_dc_governance_accounts_ready_trigger = yes
	}
}
