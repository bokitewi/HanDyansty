tgp_war_ban_all_ministers_trigger = {
	trigger_if = {
		limit = {
			scope:attacker.liege = {
				tgp_has_access_to_ministry_trigger = yes
			}
			scope:attacker = { hd_court_is_regent_trigger = no }
		}
		scope:attacker = {
			tgp_has_minister_title = no
		}
	}
}
