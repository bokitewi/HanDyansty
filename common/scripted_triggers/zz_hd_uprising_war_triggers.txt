# Shared triggers for the uprising war occupation rules.

hd_is_dynamic_uprising_war_trigger = {
	using_cb = hd_yellow_turban_rebellion_war
}

hd_uprising_siege_can_transfer_county_trigger = {
	is_alive = yes

	exists = scope:war
	exists = scope:county
	scope:war = { hd_is_dynamic_uprising_war_trigger = yes }
	scope:county = {
		tier = tier_county
		exists = holder
		county_controller ?= { this = root }
	}
	OR = {
		AND = {
			is_attacker_in_war = scope:war
			scope:county.holder = {
				OR = {
					is_defender_in_war = scope:war
					this = scope:war.primary_defender
					is_vassal_or_below_of = scope:war.primary_defender
				}
			}
		}
		AND = {
			is_defender_in_war = scope:war
			scope:county.holder = {
				OR = {
					is_attacker_in_war = scope:war
					this = scope:war.primary_attacker
					is_vassal_or_below_of = scope:war.primary_attacker
				}
			}
		}
	}
}

hd_uprising_peasant_war_grants_land_trigger = {
	situation:dynastic_cycle ?= {
		OR = {
			situation_current_phase = situation_dynastic_cycle_phase_instability
			situation_current_phase = situation_dynastic_cycle_phase_hd_collapse
			situation_current_phase = situation_dynastic_cycle_phase_chaos
			situation_current_phase = situation_dynastic_cycle_phase_hd_standoff
		}
	}
}

hd_uprising_attacker_cannot_continue_trigger = {
	$WAR$ = { NOT = { using_cb = hd_yellow_turban_rebellion_war } }
	$ATTACKER$ = {
		NOT = { any_realm_county = { always = yes } }
	}
}

hd_uprising_defender_cannot_continue_trigger = {
	$DEFENDER$ = {
		NOT = { any_realm_county = { always = yes } }
	}
}
