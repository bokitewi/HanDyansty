# Main instance scope. Missing/destroyed war references remain unresolved until
# their exact receipt has an actual callback. Zero wars permits negotiated peace
# only if the separate political and obligation checks also pass.
hd_dc_all_core_wars_resolved_trigger = {
	NOT = { has_variable = hd_dc_unbound_core_war }
	NOT = { has_variable = hd_dc_war_result_conflict }
	OR = {
		NOT = { has_variable = hd_dc_unresolved_wars }
		var:hd_dc_unresolved_wars = 0
	}
	NOT = {
		any_in_list = {
			variable = hd_dc_war_records
			NOT = { has_variable = hd_dc_core_war_result }
		}
	}
}

# Main instance, evaluated again by each automatic escalation/spawn caller.
# Do not rely on monthly cached status at the twenty-year boundary.
hd_dc_automatic_escalation_allowed_trigger = {
	var:hd_dc_stage = flag:acute
	NOT = { has_variable = hd_dc_reward_committed }
	NOT = { has_variable = hd_dc_unbound_core_war }
	NOT = { has_variable = hd_dc_war_result_conflict }
	NOT = {
		any_in_list = {
			variable = hd_dc_war_records
			NOT = { has_variable = hd_dc_core_war_result }
			OR = {
				NOT = { exists = var:hd_dc_core_war_ref }
				NOT = { has_variable = hd_dc_core_war_escalation_wait }
			}
		}
	}
}
