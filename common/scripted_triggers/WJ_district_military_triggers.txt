WJ_dd_crime_current_trigger = {
 scope:wj_dd_target = {
  has_variable = wj_dd_crime_realm
  var:wj_dd_crime_realm = top_liege.primary_title
 }
}
WJ_dd_justice_available_trigger = {
 exists = scope:wj_dd_commander.top_liege.cp:minister_justice
 scope:wj_dd_commander.top_liege.cp:minister_justice = {
  is_alive = yes
  is_imprisoned = no
  has_council_position = minister_justice
  is_councillor_of = scope:wj_dd_commander.top_liege
  top_liege = scope:wj_dd_commander.top_liege
  this != scope:wj_dd_target
 }
}

 WJ_dd_crime_current_trigger = yes
 WJ_dd_justice_available_trigger = yes
 scope:wj_dd_target = { is_alive = yes is_imprisoned = no }
}
