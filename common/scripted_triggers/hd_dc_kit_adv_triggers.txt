hd_dc_kit_adv_emperor_trigger = {
	has_title = title:h_china
	hd_dc_phase_is_trigger = { PHASE = stability_advancement }
}

# Scope：h_china 头衔。正在修大典或修律令
hd_dc_kit_adv_compiling_trigger = {
	OR = {
		has_variable = hd_dc_adv_canon_years
		has_variable = hd_dc_adv_code_years
	}
}

# Scope：活动。任一花费类别选了最高档（E7b "奢华活动"）
# 按活动类型分开判定，避免对不含该类别的活动查询选项
hd_dc_kit_adv_luxurious_activity_trigger = {
	OR = {
		AND = {
			has_activity_type = activity_feast
			OR = {
				has_activity_option = { category = feast_option_food option = feast_food_good }
				has_activity_option = { category = feast_option_courses option = feast_courses_good }
				has_activity_option = { category = feast_option_wine option = feast_wine_extensive }
			}
		}
		AND = {
			has_activity_type = activity_hunt
			OR = {
				has_activity_option = { category = hunt_option_attendants option = hunt_attendants_many }
				has_activity_option = { category = hunt_option_party_size option = hunt_party_size_many }
			}
		}
		AND = {
			has_activity_type = activity_wedding
			OR = {
				has_activity_option = { category = wedding_option_entertainment option = wedding_entertainment_good }
				has_activity_option = { category = wedding_option_food option = wedding_food_good }
				has_activity_option = { category = wedding_option_decoration option = wedding_decoration_good }
			}
		}
		AND = {
			has_activity_type = activity_tour
			OR = {
				has_activity_option = { category = tour_entourage option = tour_entourage_grand }
				has_activity_option = { category = tour_luxuries option = tour_furniture_good }
			}
		}
		AND = {
			has_activity_type = activity_pilgrimage
			has_activity_option = { category = pilgrimage_option_pomp option = pilgrimage_pomp_throng }
		}
	}
}

# Scope：角色（辩论讨论阶段，debate_discussion_events 追加）。进取期诗会/清谈事件的前提
hd_dc_kit_adv_debate_flavor_trigger = {
	is_ai = no
	hd_dc_phase_is_trigger = { PHASE = stability_advancement }
	NOT = { has_character_flag = hd_dc_adv_debate_flavor_cd }
	involved_activity ?= {
		has_activity_type = activity_debate
		any_attending_character = {
			NOT = { this = root }
		}
	}
}
