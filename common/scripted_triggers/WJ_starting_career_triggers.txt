# 开局入仕之途与宦海沉浮共用名单：只发给 h_china 持有者的天朝体系，皇帝本人不发。
# (a) 各层级天朝封臣且有实职；(b) 皇帝或天朝封臣的廷臣中，有内阁或宫廷职位者。
WJ_starting_career_eligible = {
 is_alive = yes
 exists = title:h_china.holder
 this != title:h_china.holder
 top_liege ?= title:h_china.holder
 government_has_flag = government_is_celestial
 OR = {
  AND = {
   is_ruler = yes
   government_has_flag = government_is_celestial
   WJ_has_substantive_office_trigger = yes
  }
  AND = {
   is_courtier = yes
   OR = { is_councillor = yes has_any_court_position = yes }
   court_owner ?= {
    OR = {
     this = title:h_china.holder
     AND = {
      top_liege ?= title:h_china.holder
      government_has_flag = government_is_celestial
     }
    }
   }
  }
 }
}
