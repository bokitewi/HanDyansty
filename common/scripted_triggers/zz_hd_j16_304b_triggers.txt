# E1-T10 猗卢总摄三部：郡 $T$ 落在猗卢（scope:hd_j16_zsh_yilu）以外的拓跋族 AI 手里。Scope：任意
hd_j16_304b_zs_target_trigger = {
	title:$T$.holder ?= {
		is_alive = yes
		is_ai = yes
		NOT = { this = scope:hd_j16_zsh_yilu }
		dynasty ?= scope:hd_j16_zsh_yilu.dynasty
	}
}

# 在 $RULER$（人物 scope）的国中：其最高领主是 $RULER$，或雇主是 $RULER$ 或其下属。Scope：角色
hd_j16_304b_in_realm_trigger = {
	OR = {
		top_liege = $RULER$
		employer ?= {
			OR = {
				this = $RULER$
				top_liege = $RULER$
			}
		}
	}
}
