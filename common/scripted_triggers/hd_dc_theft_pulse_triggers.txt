# Character scope. Native AI potential plus the scheme owner requirements.
hd_dc_theft_pulse_actor_trigger = {
	any_character_situation = { situation_type = dynastic_cycle }
	is_ai = yes
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	is_landed = yes
	has_treasury = yes
	NOT = { has_character_flag = flag_hostile_actions_disabled_delay }
	OR = { treasury < 0 AND = { ai_greed >= 100 ai_honor < 0 } }
	NOR = {
		any_scheme = { is_scheme_category = hostile }
		ai_honor >= 25
		has_trait = honest
		has_trait = generous
		has_trait = just
		has_trait = content
		has_trait = loyal
	}
}

# Candidate character scope; actor is explicitly saved by the annual handler.
# Business eligibility is separate from the engine gate for truthful diagnostics.
hd_dc_theft_pulse_target_trigger = {
	is_alive = yes
	exists = primary_title
	has_treasury = yes
	treasury > 0
	OR = { is_landed = yes tgp_is_any_minister = yes }
	top_liege = scope:actor.top_liege
	this != scope:actor
	NOT = { is_in_list = hd_dc_theft_pulse_targets }
	scope:actor = {
		NOT = { is_at_war_with = prev }
		trigger_if = {
			limit = { has_variable = no_scheming_allowed_var }
			var:no_scheming_allowed_var != prev
		}
	}
}
