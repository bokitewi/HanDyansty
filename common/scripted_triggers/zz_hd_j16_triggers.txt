# 两晋十六国统一史实线（j16）核心触发器。设计：docs/两晋十六国统一史实线/01_架构.md §1.1、§4.1；约定见 08_实施约定.md。
# 本文件只放全线共用的触发器；各节自己的触发器放 zz_hd_j16_<节>_triggers.txt。

# 四个 j16 开局，显式 OR（不用日期区间：190、218 等开局一路玩到 4 世纪不算 j16 局）
hd_j16_start_trigger = {
	OR = {
		game_start_date = 304.11.20
		game_start_date = 346.12.10
		game_start_date = 370.2.13
		game_start_date = 404.3.24
	}
}

# 心跳续排的截止：420 年最后一跳续排出的那一跳落在 421 年 1 月，正好给 E5-J90 收尾
hd_j16_continue_trigger = {
	current_date < 421.1.1
}

# 人物在世：先认 history 原生人物，再认 hbirth 出场指针（指针在载体死后仍指着死人，所以两头都判 is_alive）
# 用法：hd_j16_alive_trigger = { ID = si_ma_rui }
hd_j16_alive_trigger = {
	OR = {
		character:$ID$ ?= { is_alive = yes }
		hd_hb_alive = { ID = $ID$ }
	}
}

# 人物在库中且已死。判断「某人已死」一律用它，不读卒年的 done 变量（01 §2.2-7）。
# 两头都取不到（hbirth 还没出场）时为假：没出场的人不算死
hd_j16_dead_trigger = {
	OR = {
		character:$ID$ ?= { is_alive = no }
		AND = {
			has_global_variable = hd_hb_p_$ID$
			global_var:hd_hb_p_$ID$ = { is_alive = no }
		}
	}
}
