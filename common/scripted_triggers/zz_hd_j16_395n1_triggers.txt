# Character：属 $R$ 一系（本人，或其 top_liege 是 $R$；廷臣按雇主一系算）
hd_j16_n1_in_realm_trigger = {
	OR = {
		this = $R$
		top_liege ?= $R$
	}
}

# Character：燕系之主（M35/M50/M60/M65 换主时的现任、M45 燕主须满足）：慕容氏、长乐冯氏（冯跋）、昌黎兰氏（兰汗），

#   不能按王朝或家族认；按头衔认，高云死后由其继承人接位时仍能通过）。防止龙城被北魏、高句丽等外人取得时把外国整个转给史实之人
hd_j16_n1_yan_system_trigger = {
	OR = {
		dynasty ?= dynasty:murong
		house ?= house:house_changle_feng
		house ?= house:changli_lan
		AND = {
			exists = global_var:hd_j16_n1_yan_title
			has_title = global_var:hd_j16_n1_yan_title
		}
	}
}

# $A$ 一系的国土与 $B$ 一系的国土接壤（$A$、$B$ 为独立君主 scope，各调用处都是 top_liege）。无 scope 要求
# 用引擎的邻国表 any_neighboring_top_liege_realm_owner（原版 183 处），不再逐县扫描 $B$ 全境 × 邻县（窗口内每跳求值，大国开销过高）。
hd_j16_n1_borders_realm_trigger = {
	$A$ ?= {
		any_neighboring_top_liege_realm_owner = { this = $B$ }
	}
}

# $SIDE$ 一系与郡 $DUCHY$ 接壤：郡内有不属 $SIDE$ 一系的县，与 $SIDE$ 一系的县相邻。无 scope 要求
hd_j16_n1_borders_duchy_trigger = {
	$DUCHY$ = {
		any_de_jure_county = {
			NOT = { holder ?= { target_is_same_character_or_above = $SIDE$ } }
			any_neighboring_county = {
				holder ?= { target_is_same_character_or_above = $SIDE$ }
			}
		}
	}
}

# 称藩战可以开打（E5-Y20、E5-Y30）：$A$（攻方）、$B$ 都在、不是同一人、未交战、接壤
hd_j16_n1_vassal_war_ready_trigger = {
	exists = $A$
	exists = $B$
	NOT = { $A$ = $B$ }
	$A$ = {
		is_imprisoned = no
		NOT = { is_at_war_with = $B$ }
	}
	hd_j16_n1_borders_realm_trigger = { A = $A$ B = $B$ }
}

# 西燕已不存在：慕容永 changli_murong_064 不再是有地的独立君主（原生或 hbirth 指针）

			is_alive = yes

			is_alive = yes
			is_landed = yes
			is_independent_ruler = yes
		}
	}
}
