# 强秦（04 §2.1）：长安 c_uuii_p_867 之主的 top_liege 存在、苻氏（dynasty:fu_1）、独立，领土不少于 60 县（R6 自拟值）。
# 由 E4-W50 在 383.6.1 后首跳检查一次，成立即设 hd_j16_qin_existed；历史入侵侧（zz_hd_hi_j16_triggers.txt）只读 existed。无 scope 要求
############################################################
hd_j16_strong_qin_trigger = {
	title:c_uuii_p_867.holder.top_liege ?= {
		is_alive = yes
		dynasty ?= dynasty:fu_1
		is_independent_ruler = yes
		realm_size >= 60
	}
}

############################################################
# 某人属秦主一系（本人、或其 top_liege 是秦主；廷臣按雇主一系算）。Scope：角色。参数 QIN：秦主 scope
############################################################
hd_j16_n370_in_realm_trigger = {
	OR = {
		this = $QIN$
		top_liege ?= $QIN$
	}
}

############################################################
# $QIN$ 一系的国土与郡 $DUCHY$ 接壤：郡内有不属 $QIN$ 一系的县，与 $QIN$ 一系的县相邻。无 scope 要求。
# 用法：hd_j16_n370_borders_duchy_trigger = { DUCHY = title:d_wudou QIN = scope:x }。只在本节窗口内的节点里调用（清单：时代段外不做 neighboring）
############################################################
hd_j16_n370_borders_duchy_trigger = {
	$DUCHY$ = {
		any_de_jure_county = {
			NOT = { holder ?= { target_is_same_character_or_above = $QIN$ } }
			any_neighboring_county = {
				holder ?= { target_is_same_character_or_above = $QIN$ }
			}
		}
	}
}

############################################################
# $QIN$ 一系的国土与 $OTHER$（独立君主）的国土接壤。Scope 无要求。用于代、前凉（k_suozhou 静态不挂郡，按持有者的国土判）
############################################################
hd_j16_n370_borders_realm_trigger = {
	$OTHER$ = {
		any_sub_realm_county = {
			any_neighboring_county = {
				holder ?= { target_is_same_character_or_above = $QIN$ }
			}
		}
	}
}

# E4-Q20 取梁益可以开战。读调用方已存的 scope:hd_j16_q20_qin（秦主）、hd_j16_q20_def（天子）。无 scope 要求
############################################################
hd_j16_liangyi_ready_trigger = {
	hd_j16_n370_jin_emperor_trigger = yes
	exists = scope:hd_j16_q20_def
	scope:hd_j16_q20_qin ?= {
		is_independent_ruler = yes
		is_imprisoned = no
		NOT = { this = scope:hd_j16_q20_def }
		NOT = { any_character_war = { var:hd_nbc_war_kind ?= flag:liangyi } }
	}
	title:d_hanzhong.title_capital_county.holder ?= {
		OR = {
			this = scope:hd_j16_q20_def
			top_liege ?= scope:hd_j16_q20_def
		}
	}
	hd_j16_n370_borders_duchy_trigger = { DUCHY = title:d_hanzhong QIN = scope:hd_j16_q20_qin }
	OR = {
		has_global_variable = hd_j16_qiuchi_done
		title:d_wudou.title_capital_county.holder ?= { hd_j16_n370_in_realm_trigger = { QIN = scope:hd_j16_q20_qin } }
	}
}
