# 佛教东传（hd_fo，2026-10-25）：触发器。设计见 docs/佛教东传_设计草案.md。

# 角色或县：信的是当前日期还没开放的宗派（第 46 项）。开放年份与 history/faiths/00_buddhism.txt 一致；
# 律、华严、唯识要到 617 以后才由事件开放，开局一律算未开放。
hd_fo_has_closed_sect_trigger = {
	OR = {
		rite = rite:vinaya
		rite = rite:avatamsaka
		rite = rite:yogacara
		AND = { rite = rite:sukhavati current_date < 402.1.1 }
		AND = { rite = rite:dhyana current_date < 527.1.1 }
		AND = { rite = rite:pundarika current_date < 575.1.1 }
	}
}

# 角色：信的是此刻还没开放的宗派（第 46 项扩展到开局后按年出场的人，2026-10-25）。
# 开局晚于史实开宗年份的，宗派开局即由 history 开放；否则要等开宗事件设下 hd_fo_open_<rite>。
hd_fo_has_closed_sect_now_trigger = {
	OR = {
		AND = { rite = rite:vinaya NOT = { has_global_variable = hd_fo_open_vinaya } }
		AND = { rite = rite:avatamsaka NOT = { has_global_variable = hd_fo_open_avatamsaka } }
		AND = { rite = rite:yogacara NOT = { has_global_variable = hd_fo_open_yogacara } }
		AND = { rite = rite:sukhavati NOT = { has_global_variable = hd_fo_open_sukhavati } game_start_date < 402.1.1 }
		AND = { rite = rite:dhyana NOT = { has_global_variable = hd_fo_open_dhyana } game_start_date < 527.1.1 }
		AND = { rite = rite:pundarika NOT = { has_global_variable = hd_fo_open_pundarika } game_start_date < 575.1.1 }
	}
}

# 史实事件到期（§10.1 R2）：到了年份、开局早于该年、还没做过
hd_fo_due_trigger = {
	current_year >= $YEAR$
	game_start_date < $YEAR$.1.1
	NOT = { has_global_variable = hd_fo_done_$E$ }
}

# 某位高僧已出场且在世
hd_fo_monk_alive_trigger = {
	exists = global_var:hd_fo_p_$ID$
	global_var:hd_fo_p_$ID$ = { is_alive = yes }
}

# 寺院、石窟一级的可建条件（§10.1 R3）：事件建成之前（原版已放建筑位的三处），或灭佛毁掉之后，锁住正常建造。
hd_fo_can_build_trigger = {
	custom_description = {
		text = hd_fo_can_build_locked
		NOT = { has_global_variable = hd_fo_lock_$KEY$ }
	}
}

# 角色：能接纳 scope:hd_fo_monk（第 13、21 项）。灭佛标记 hd_fo_miefo 是角色旗标，随本人死亡消失。
hd_fo_can_receive_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	NOT = { has_character_flag = hd_fo_miefo }
	trigger_if = {
		limit = { exists = scope:hd_fo_monk }
		NOT = { this = scope:hd_fo_monk }
		NOT = { scope:hd_fo_monk = { is_target_in_variable_list = { name = hd_fo_refused target = prev } } }
	}
}
