# 从原版恢复的、先前被同名文件覆盖而丢失的触发器

holding_is_feudalized = {
	OR = {
		has_holding_type = castle_holding
		has_holding_type = city_holding
		has_holding_type = church_holding
		has_holding_type = temple_citadel_holding
	}
}

holding_is_church = {
	OR = {
		has_holding_type = church_holding
		has_holding_type = temple_citadel_holding
	}
}

holding_is_city = {
	has_holding_type = city_holding
}

holding_is_castle = {
	OR = {
		has_holding_type = castle_holding
		has_holding_type = temple_citadel_holding
	}
}

court_position_can_be_combatant_tt_trigger = {
	custom_tooltip = {
		text = NONE_ARE_ALLOWED_TO_BE_WARRIORS_IN_CULTURE
		can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = $EMPLOYER$ }
	}
}

wet_nurse_court_position_unlocked_trigger = {
	OR = {
		has_bp2_dlc_trigger = yes
		rite = { rite_has_parameter = free_wet_nurse_court_position }
	}
}

court_scholar_court_position_unlocked_trigger = {
	has_fp3_dlc_trigger = yes
	OR = {
		culture = { has_cultural_tradition = tradition_fp3_beacon_of_learning }
		rite = { rite_has_parameter = tenet_scholasticism_court_scholar }
	}
}

hre_valid_prince_elector_trigger = {
	is_landed = yes
	faith = root.faith
	OR = { # must be de-jure vassal of empire
		AND = {
			capital_province.kingdom.holder ?= root
			capital_province.empire != title:e_italy
		}
		capital_province.empire.holder ?= root
	}
	NOT = { # not already selected
		any_held_title = {
			is_target_in_global_variable_list = {
				name = hre_elector_list
				target = this
			}
		}
	}
}

marriage_clergy_can_marry_trigger = {
	#Monks can never marry
	NOR = {
		has_trait = devoted
		has_trait = order_member
	}
	OR = {
		rite = { rite_has_parameter = clergy_can_marry }
		AND = {
			is_ruler = yes
			top_participant_group:the_christian_church ?= { has_participant_group_parameter = clergy_can_marry }
			trigger_if = {
				limit = { is_ai = yes }
				saeculum_obscurum_clergy_would_marry_trigger = yes
			}
		}
	}
}

fp2_struggle_secure_iberian_foothold_neighbor_with_outsider_kingdom_trigger = {
	any_de_jure_county = {
		any_title_to_title_neighboring_and_across_water_county = {
			fp2_struggle_secure_iberian_foothold_outsider_kingdom_trigger = yes
		}
	}
}

is_eastern_roman_emperor_trigger = {
	OR = {
		has_title = title:e_byzantium
		has_title = title:h_eastern_roman_empire
	}
}

show_byzantion_throne_room_background = {
	save_temporary_scope_as = background_check
	OR = {
		location ?= title:b_constantinople.title_province
		AND = {
			is_roman_emperor_primary_title_trigger = yes
			title:c_byzantion = {
				any_de_jure_top_liege = { this = scope:background_check }
			}
		}
	}
}

culture_use_chinese_character_trigger = {
	OR = {
		culture_use_standard_chinese_character_trigger = yes
		culture_use_variant_chinese_character_trigger = yes
	}
}

culture_use_variant_chinese_character_trigger = {
	OR = {
		has_cultural_pillar = language_khitan
		has_cultural_pillar = language_tungusic  # all 3 cultures using this has different chinese character variant
		has_cultural_tradition = tradition_sinophilic # assume they have do same thing
	}
}

culture_use_standard_chinese_character_trigger = {
	OR = {
		has_cultural_pillar = language_chinese
		has_cultural_pillar = language_korean
		has_cultural_pillar = language_japonic
		has_cultural_pillar = language_vietnamese
	}
}

should_use_christian_music_trigger = {
	religion = religion:christianity_religion
}
