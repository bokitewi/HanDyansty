# Actual primary-title change caused by this instance's completed claimant
# war. A normal inheritance, foreign claimant or an unrelated war is excluded.
hd_dc_royal_war_enthronement_trigger = {
	exists = title:h_china.holder
	title:h_china.holder = { is_alive = yes }
	NOT = { title:h_china.holder = var:hd_dc_initial_ruler }
	any_in_list = {
		variable = hd_dc_war_records
		var:hd_dc_core_war_result = 1
		var:hd_dc_war_cb = flag:claimant_faction_war
		var:hd_dc_war_target = title:h_china
		exists = var:hd_dc_actor_story_ref
		var:hd_dc_war_beneficiary = title:h_china.holder
		var:hd_dc_actor_story_ref.var:hd_dc_actor = title:h_china.holder
	}
}

hd_dc_royal_war_defended_trigger = {
	exists = title:h_china.holder
	title:h_china.holder = { is_alive = yes }
	title:h_china.holder = var:hd_dc_initial_ruler
	any_in_list = {
		variable = hd_dc_war_records
		var:hd_dc_core_war_result = 2
		var:hd_dc_war_cb = flag:claimant_faction_war
		var:hd_dc_war_target = title:h_china
		var:hd_dc_core_war_defender = title:h_china.holder
	}
}

# Current authored ROY politics only create claimant factions and claimant
# wars. Any additional sovereignty route must first supply its E1 priority
# audit; this gate prevents future independence work silently losing priority.
hd_dc_royal_supported_war_resolution_trigger = {
	var:hd_dc_type = flag:roy
	OR = { var:hd_dc_stage = flag:acute AND = { var:hd_dc_stage = flag:aftermath has_variable = hd_dc_roy01_node_closed } }
	NOT = { has_variable = hd_dc_political_result }
	NOT = { has_variable = hd_dc_reward_committed }
	hd_dc_all_core_wars_resolved_trigger = yes
	NOT = { has_variable = hd_dc_independence_started }
	NOT = { has_variable = hd_dc_sovereignty_audit_pending }
	NOT = {
		any_in_list = {
			variable = hd_dc_war_records
			NOT = { var:hd_dc_war_cb = flag:claimant_faction_war }
		}
	}
}
