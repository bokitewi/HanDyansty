# 普通吞并的占位目标：守方主头衔可用作目标（2026-10-25）。Scope：守方主头衔；$DEF$ = 守方
# 主头衔须有法理县在守方疆域内，否则不会出现在宣战目标列表里——袁术僭号新建的动态帝国没有法理县，
# 吞并因此整条消失。不满足时改用守方首府县（见 CB valid_to_start 与 hd_strat_declare_war_effect）
hd_tb_primary_title_target_trigger = {
	tier >= tier_county
	tier < tier_hegemony
	is_landless_type_title = no
	OR = {
		tier = tier_county
		any_de_jure_county = { holder.top_liege ?= $DEF$ }
	}
}

# 僭号帝国（hd_court_proclaim_emperor_effect 所立）：吞并胜利时销毁，不归胜者。Scope：头衔；$DEF$ = 持有者
# 旧档的僭号帝国没有标记变量，按"僭号称帝者持有、无法理县的帝国"兜底认定
hd_tb_pseudo_empire_trigger = {
	OR = {
		has_variable = hd_court_pseudo_empire
		AND = {
			tier = tier_empire
			$DEF$ = { has_character_modifier = hd_court_pseudo_emperor_modifier }
			NOT = { any_de_jure_county = { always = yes } }
		}
	}
}
