# Character scope. Ordinary noble-family titles do not keep an office active.
hd_special_military_contract_active_trigger = {
	is_alive = yes
	is_landed = yes
	is_landless_ruler = no
	is_independent_ruler = no
	government_has_flag = government_is_celestial
	OR = {
		vassal_contract_has_flag = hd_tuntian
		vassal_contract_has_flag = hd_military_colony
	}
}

hd_military_colony_contract_trigger = {
	hd_special_military_contract_active_trigger = yes
	vassal_contract_has_flag = hd_military_colony
}
