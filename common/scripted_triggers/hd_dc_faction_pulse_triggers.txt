# Candidate character. The native trigger owns government, membership,
# relationship and action restrictions; this pulse never changes those rules.
hd_dc_extra_faction_creator_trigger = {
	is_ai = yes
	is_alive = yes
	this != scope:hd_dc_extra_faction_target
	scope:hd_dc_extra_faction_target = { NOT = { any_targeting_faction = { faction_is_type = $TYPE$ } } }
	can_create_faction = { type = $TYPE$ target = scope:hd_dc_extra_faction_target }
}
hd_dc_extra_faction_county_trigger = {
	scope:hd_dc_extra_faction_target = { NOT = { any_targeting_faction = { faction_is_type = $TYPE$ } } }
	can_title_create_faction = { type = $TYPE$ target = scope:hd_dc_extra_faction_target }
}
hd_dc_extra_character_faction_available_trigger = {
	NOT = { any_targeting_faction = { faction_is_type = $TYPE$ } }
	situation:dynastic_cycle = {
		any_situation_participant = { hd_dc_extra_faction_creator_trigger = { TYPE = $TYPE$ } }
	}
}
hd_dc_extra_county_faction_available_trigger = {
	NOT = { any_targeting_faction = { faction_is_type = $TYPE$ } }
	situation:dynastic_cycle = {
		any_in_list = {
			variable = hd_dc_counties
			hd_dc_extra_faction_county_trigger = { TYPE = $TYPE$ }
		}
	}
}
