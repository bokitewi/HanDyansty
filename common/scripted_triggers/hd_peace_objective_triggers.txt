# The objective clause always fulfils the primary attacker's declared aim.
# Defenders negotiate concessions through the normal terms instead.
hd_peace_objective_available_trigger = {
	hd_peace_ordinary_context_trigger = yes
	var:hd_peace_treaty_war = {
		primary_attacker = root
		OR = {
			AND = {
				hd_peace_special_objective_war_trigger = yes
				root = { hd_peace_special_objective_available_trigger = yes }
			}
			hd_peace_independence_war_trigger = yes
			AND = {
				hd_peace_subjugation_war_trigger = yes
				primary_attacker = { highest_held_title_tier > root.var:hd_peace_treaty_opponent.highest_held_title_tier }
			}
			AND = {
				OR = { using_cb = claim_cb using_cb = crusading_claim_cb }
				claimant ?= { is_alive = yes }
				exists = claimant
				claimant = root.var:hd_peace_objective_claimant
				casus_belli = { any_target_title = { exists = holder } }
			}
			AND = {
				NOT = { OR = { using_cb = claim_cb using_cb = crusading_claim_cb } }
				NOT = { hd_peace_independence_war_trigger = yes }
				NOT = { hd_peace_subjugation_war_trigger = yes }
				NOT = { hd_peace_special_objective_war_trigger = yes }
				NOT = { hd_peace_ordinary_annex_war_trigger = yes }	# 2026-10-29：普通吞并不给完整目标（否则五分之一价即可吞全境）
				root = { has_variable_list = hd_peace_cb_reserved_counties }
			}
		}
	}
	# The full objective and ad-hoc land exchanges must not transfer the same land.
	hd_peace_offer_county_count_value = 0
	hd_peace_demand_county_count_value = 0
	hd_peace_demand_title_count_value = 0
	hd_peace_offer_release_count_value = 0
	hd_peace_demand_release_count_value = 0
	NOT = { has_variable = hd_peace_offer_suzerainty }
	NOT = { has_variable = hd_peace_demand_suzerainty }
}
