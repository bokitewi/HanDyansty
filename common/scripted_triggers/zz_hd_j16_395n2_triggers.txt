# 某人属 $RULER$ 一系（本人，或其 top_liege 是 $RULER$；廷臣按雇主一系算）。Scope：角色
############################################################
hd_j16_n2_in_realm_trigger = {
	OR = {
		this = $RULER$
		top_liege ?= $RULER$
	}
}

############################################################
# $A$ 一系的国土与 $B$ 一系的国土接壤。无 scope 要求。只在本节窗口内的节点里调用（清单：时代段外不做 neighboring）
############################################################
hd_j16_n2_realms_border_trigger = {
	$A$ = {
		any_sub_realm_county = {
			any_neighboring_county = {
				holder ?= { target_is_same_character_or_above = $B$ }
			}
		}
	}
}

############################################################
# $RULER$ 一系已持郡 $DUCHY$ 的县，或与郡内某县相邻。无 scope 要求。只在本节窗口内调用
############################################################
hd_j16_n2_realm_borders_duchy_trigger = {
	$DUCHY$ = {
		any_de_jure_county = {
			OR = {
				holder ?= { target_is_same_character_or_above = $RULER$ }
				any_neighboring_county = {
					holder ?= { target_is_same_character_or_above = $RULER$ }
				}
			}
		}
	}
}
