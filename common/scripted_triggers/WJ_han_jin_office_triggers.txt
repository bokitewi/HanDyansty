WJ_han_jin_law_trigger = {
	top_liege ?= {
		OR = {
			has_realm_law = han_province_title_law
			has_realm_law = jin_province_title_law
		}
	}
}

WJ_tang_law_trigger = {
	top_liege ?= {
		has_realm_law = tang_province_title_law
	}
}

# 当前封臣契约必须实际包含天朝省区契约组，才能安全切换为武官省区。
WJ_has_celestial_province_contract_trigger = {
	OR = {
		vassal_contract_has_flag = celestial_province_standard
		vassal_contract_has_flag = celestial_province_industrial
		vassal_contract_has_flag = celestial_province_metropolitan
		vassal_contract_has_flag = celestial_province_military
		vassal_contract_has_flag = celestial_province_protectorate
		vassal_contract_has_flag = WJ_celestial_province_sili
	}
}

# 三条正式入仕途径。郎官尚在候考，不属于正式入仕。
WJ_has_entry_road_trigger = {
	OR = {
		has_character_modifier = WJ_bizhao_modifier
		has_character_modifier = WJ_xiaolian_modifier
		has_character_modifier = WJ_maocai_modifier
	}
}

# 原版任命决议在这一身份下改用“隐居养望/出山”风味。
WJ_uses_yinju_appointment_flavor_trigger = {
	WJ_han_jin_law_trigger = yes
	WJ_is_landless_shizu_without_office_trigger = yes
}

# 实际官职：原版天朝部长，或男爵级以上非家族头衔。
WJ_has_substantive_office_trigger = {
	OR = {
		tgp_is_any_minister = yes
		any_held_title = {
			tier >= tier_barony
			is_noble_family_title = no
		}
	}
}

# 有现行官职的角色：实际官职或处于试守期的假吏。
WJ_has_official_position_trigger = {
	OR = {
		WJ_has_substantive_office_trigger = yes
		has_character_modifier = WJ_temp_official_modifier
	}
}

# 事件驱动试守：汉晋天朝内已有实职，但尚无正式入仕资格。
WJ_is_valid_temp_official_trigger = {
	government_has_flag = government_is_celestial
	is_independent_ruler = no
	is_adult = yes
	WJ_han_jin_law_trigger = yes
	WJ_has_substantive_office_trigger = yes
	WJ_has_entry_road_trigger = no
}

# 隐居/出山适用人群：无地士族且没有任何现行官职。
WJ_is_landless_shizu_without_office_trigger = {
	any_held_title = {
		is_noble_family_title = yes
	}
	NOR = {
		any_held_title = {
			tier >= tier_barony
			is_noble_family_title = no
		}
		tgp_is_any_minister = yes
		has_character_modifier = WJ_temp_official_modifier
# 		has_character_modifier = WJ_langguan_modifier
	}
}

# 禄秩人物块的唯一状态判定。

	has_character_modifier = WJ_temp_official_modifier
}

WJ_luzhi_status_shi_trigger = {
	NOT = { has_character_modifier = WJ_temp_official_modifier }
	WJ_is_landless_shizu_without_office_trigger = yes
}

WJ_luzhi_status_chu_trigger = {
	NOT = { has_character_modifier = WJ_temp_official_modifier }
	WJ_has_substantive_office_trigger = no
	WJ_is_landless_shizu_without_office_trigger = no
}

WJ_luzhi_status_gong_trigger = {
	NOT = { has_character_modifier = WJ_temp_official_modifier }
	OR = {
		has_title = title:e_WJ_minister_tai_fu
		has_title = title:e_WJ_minister_cheng_xiang
		has_title = title:e_minister_grand_marshal
	}
}

WJ_luzhi_status_chao_trigger = {
	NOT = { has_character_modifier = WJ_temp_official_modifier }
	NOT = { WJ_luzhi_status_gong_trigger = yes }
	any_held_title = {
		is_noble_family_title = no
		tier >= tier_empire
	}
}

# 举孝廉候选人：必须尚未由任何途径入仕、无地且无官。
# RECOMMENDER = 将成为举主的公国/王国级官员。
WJ_can_be_recommended_xiaolian_candidate_trigger = {
	government_has_flag = government_is_celestial
	top_liege = $RECOMMENDER$.top_liege
	is_adult = yes
	is_incapable = no
	is_imprisoned = no
	WJ_has_official_position_trigger = no
	NOR = {
		has_character_modifier = WJ_bizhao_modifier
		has_character_modifier = WJ_xiaolian_modifier
		has_character_modifier = WJ_maocai_modifier
# 		has_character_modifier = WJ_langguan_modifier
	}
}

# 举茂才候选人：允许现任官员、孝廉与辟召出身者，唯独不得重复举茂才。
# RECOMMENDER = 将成为举主的王国级以上官员。
WJ_can_be_recommended_maocai_candidate_trigger = {
	WJ_langguan_real_emperor_trigger = no
	$RECOMMENDER$ = { WJ_langguan_real_emperor_trigger = no }
	government_has_flag = government_is_celestial
	top_liege = $RECOMMENDER$.top_liege
	is_adult = yes
	is_incapable = no
	is_imprisoned = no
	NOT = { has_character_modifier = WJ_maocai_modifier }
}

# 从举主视角检查原版天朝门生限制。已有的本府门生无需重复建立关系。
# CANDIDATE = 将被举荐的角色。
WJ_can_recommend_candidate_as_disciple_trigger = {
	OR = {
		has_relation_disciple = $CANDIDATE$
		eligible_for_elder_trigger = {
			DISCIPLE = $CANDIDATE$
		}
	}
}

# Real sovereign: excludes kingdom rulers and empire-tier minister vassals.
WJ_langguan_real_emperor_trigger = {
	is_ruler = yes
	OR = {
		highest_held_title_tier = tier_hegemony
		AND = { highest_held_title_tier = tier_empire is_independent_ruler = yes }
	}
}

WJ_mufu_has_open_slot_trigger = {
	WJ_mufu_empty = yes
}
