# 某人在世，且在你宫廷或为你封臣（任意层级）
hd_asp3_subject_trigger = {
	OR = {
		global_var:wj_hchar_$ID$ ?= { is_alive = yes target_is_liege_or_above = root }
		global_var:hd_hb_p_$ID$ ?= { is_alive = yes target_is_liege_or_above = root }
		character:$ID$ ?= { is_alive = yes target_is_liege_or_above = root }
	}
}

hd_asp3_dead_or_subject_trigger = {
	OR = {
		hd_asp2_dead_trigger = { ID = $ID$ }
		hd_asp3_subject_trigger = { ID = $ID$ }
	}
}

# 已死（尚未出生按已死算），或在世而无地
hd_asp3_gone_or_landless_trigger = {
	OR = {
		hd_asp2_dead_trigger = { ID = $ID$ }

		global_var:hd_hb_p_$ID$ ?= { is_alive = yes is_landed = no }
		character:$ID$ ?= { is_alive = yes is_landed = no }
	}
}

# 某州的每个法理郡都在你领内（郡头衔或郡治县在你领内即算，口径同 D26）
hd_asp3_prov_all_duchies_trigger = {
	title:$P$ = {
		any_in_de_jure_hierarchy = {
			tier = tier_duchy
			count >= 1
		}
		NOT = {
			any_in_de_jure_hierarchy = {
				tier = tier_duchy
				NOT = { hd_asp_title_in_root_realm_trigger = yes }
			}
		}
	}
}
