# HanDynasty / 群雄逐鹿. All triggers below use CHARACTER scope unless noted.
# "Under the Han hegemony" follows the current mod's owned-realm geography:
# at least one sub-realm county has h_china as its de jure hegemony.
# A landless hegemony holder with landed vassals is NOT excluded here.
hd_qxzl_top_liege_eligible_trigger = {
	is_alive = yes
	is_ruler = yes
	is_independent_ruler = yes
	government_has_flag = government_is_celestial
	any_sub_realm_county = {
		tier = tier_county
		exists = hegemony
		hegemony = title:h_china
	}
}

# Cao Cao and Yuan Shao breakup exemption. Character scope.
# True for Cao/Yuan themselves and for any ruler anywhere below either of them
# in the current de facto vassal hierarchy.
hd_qxzl_cao_yuan_breakup_exempt_trigger = {
	OR = {
		this = character:cao_cao
		this = character:yuan_shao
		target_is_liege_or_above = character:cao_cao
		target_is_liege_or_above = character:yuan_shao
	}
}

# No capital-only test and no sub-realm development test here.
hd_qxzl_rank_eligible_trigger = {
	OR = {
		highest_held_title_tier = tier_duchy
		AND = {
			highest_held_title_tier = tier_county
			OR = {
				any_held_title = {
					tier = tier_county
					development_level >= 8
				}
				has_trait = ambitious
			}
		}
	}
}

hd_qxzl_vassal_eligible_trigger = {
	is_alive = yes
	is_ruler = yes
	is_landed = yes
	is_independent_ruler = no
	exists = liege
	exists = primary_title
	NOT = { hd_qxzl_cao_yuan_breakup_exempt_trigger = yes }
	hd_qxzl_rank_eligible_trigger = yes
	top_liege = { hd_qxzl_top_liege_eligible_trigger = yes }
}

# Used at visible-event delivery AND when accepting option A.
# The queue's rank cannot silently migrate into the other rank's time window.
# The ticket still has to belong to this holder; revocation while a window is
# open cannot let a former holder act on the new holder's ticket.
hd_qxzl_ticket_eligible_trigger = {
	hd_qxzl_vassal_eligible_trigger = yes
	exists = scope:hd_qxzl_ticket_title
	scope:hd_qxzl_ticket_title = { holder = root }
	OR = {
		AND = {
			highest_held_title_tier = tier_duchy
			scope:hd_qxzl_ticket_title = { has_variable = hd_qxzl_duke_ticket }
		}
		AND = {
			highest_held_title_tier = tier_county
			scope:hd_qxzl_ticket_title = { has_variable = hd_qxzl_count_ticket }
		}
	}
}

# Direct post-coalition Yuzhou/Qingzhou/Yanzhou split. Use the ruler's
# primary-title de jure hierarchy so a ruler is not released merely for holding
# one stray county in these kingdoms. Counts and above are valid targets; barons
# are intentionally excluded.
hd_qxzl_yu_qing_yan_vassal_trigger = {
	is_alive = yes
	is_ruler = yes
	is_landed = yes
	highest_held_title_tier >= tier_county
	is_independent_ruler = no
	exists = liege
	exists = primary_title
	NOT = { hd_qxzl_cao_yuan_breakup_exempt_trigger = yes }
	primary_title = {
		OR = {
			this = title:k_uuii_yuzhou
			de_jure_liege = title:k_uuii_yuzhou
			de_jure_liege.de_jure_liege = title:k_uuii_yuzhou
			this = title:k_uuii_qingzhou
			de_jure_liege = title:k_uuii_qingzhou
			de_jure_liege.de_jure_liege = title:k_uuii_qingzhou
			this = title:k_uuii_yanzhou
			de_jure_liege = title:k_uuii_yanzhou
			de_jure_liege.de_jure_liege = title:k_uuii_yanzhou
		}
	}
}

# Literal global immediate-release scope: not restricted to Han geography.
# Native every_ruler iterates count+ rulers; barony holders are not promoted.
hd_qxzl_non_celestial_vassal_trigger = {
	is_alive = yes
	is_ruler = yes
	is_landed = yes
	highest_held_title_tier >= tier_county
	is_independent_ruler = no
	exists = liege
	NOT = { hd_qxzl_cao_yuan_breakup_exempt_trigger = yes }
	NOT = { government_has_flag = government_is_celestial }
}
