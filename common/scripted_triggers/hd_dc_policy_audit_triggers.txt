# PAR-01 A/D share real local signatures, payments and two personal audits.
hd_dc_par01_audit_available_trigger = {
	hd_dc_par01_window_valid_trigger = yes
	is_imprisoned = no gold >= 30
	scope:hd_dc_instance.var:hd_dc_par01_county = {
		hd_dc_party_case_county_trigger = yes
		holder = scope:hd_dc_instance.var:hd_dc_par01_executor
	}
}

# The original disputed county is the relief side; select a genuinely
# comparable second county, not the same county or a fabricated reference.
hd_dc_par01_comparison_county_trigger = {
	hd_dc_party_case_county_trigger = yes
	this != scope:hd_dc_instance.var:hd_dc_par01_county
	development_level >= { value = scope:hd_dc_instance.var:hd_dc_par01_county.development_level subtract = 3 }
	development_level <= { value = scope:hd_dc_instance.var:hd_dc_par01_county.development_level add = 3 }
	county_control >= { value = scope:hd_dc_instance.var:hd_dc_par01_county.county_control subtract = 10 }
	county_control <= { value = scope:hd_dc_instance.var:hd_dc_par01_county.county_control add = 10 }
}

hd_dc_par01_comparison_available_trigger = {
	hd_dc_par01_audit_available_trigger = yes gold >= 100
	OR = {
		stewardship >= 16
		scope:hd_dc_instance.var:hd_dc_par01_executor = { stewardship >= 16 opinion = { target = title:h_china.holder value >= 60 } }
	}
	situation:dynastic_cycle = { any_in_list = { variable = hd_dc_counties hd_dc_par01_comparison_county_trigger = yes } }
}

# The political signatories stay frozen. Neither death nor replacement can
# supply a missing signature; affectedness is required before execution only.
hd_dc_par01_audit_context_trigger = {
	NOT = { has_variable = hd_dc_par01_result }
	var:hd_dc_stage = flag:acute
	var:hd_dc_audit_author = title:h_china.holder
	var:hd_dc_audit_author = { is_alive = yes is_imprisoned = no }
	hd_dc_par01_parties_valid_trigger = yes
}

hd_dc_par01_executor_window_trigger = {
	is_alive = yes is_imprisoned = no
	exists = scope:hd_dc_instance
	scope:hd_dc_instance = {
		hd_dc_par01_audit_context_trigger = yes
		var:hd_dc_par01_version = scope:hd_dc_par01_version
		var:hd_dc_audit_state = flag:offered has_variable = hd_dc_audit_wait
	}
	OR = {
		AND = { scope:hd_dc_audit_side = flag:first this = scope:hd_dc_instance.var:hd_dc_audit_first_executor scope:hd_dc_instance = { NOT = { has_variable = hd_dc_audit_first_answer } } }
		AND = { scope:hd_dc_audit_side = flag:second this = scope:hd_dc_instance.var:hd_dc_audit_second_executor scope:hd_dc_instance = { NOT = { has_variable = hd_dc_audit_second_answer } } }
	}
}
