hd_title_is_in_realm_trigger = {
	$TITLE$.holder ?= {
		OR = {
			this = root
			any_liege_or_above = { this = root }
		}
	}
}

hd_is_chinese_heritage_trigger = {
	culture = { has_cultural_pillar = heritage_chinese }
}

hd_controls_entire_e_han_trigger = {
	title:e_han = {
		any_de_jure_county = {
			percent = 1
			holder ?= {
				OR = {
					this = root
					any_liege_or_above = { this = root }
				}
			}
		}
	}
}

hd_valid_partition_child_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	dynasty = root.dynasty
	NOR = {
		highest_held_title_tier = tier_empire
		highest_held_title_tier = tier_hegemony
	}
}

# 头衔持有者是 root 本人（直辖），或是 root 的直属封臣。
hd_title_held_by_root_or_direct_vassal_trigger = {
	$TITLE$.holder ?= {
		OR = {
			this = root
			liege ?= root
		}
	}
}
