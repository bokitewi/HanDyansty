# Eastern Han adaptation: real consorts/selected palace women; no post-Han rank ladder.
not_involved_activity_trigger = {
 is_imprisoned = no is_travelling = no is_at_war = no is_adult = yes
 NOT = { exists = involved_activity }
}
WJ_harem_has_role = {
 OR = { has_trait = SanGong_GongWei_1 has_trait = SanGong_Taihou has_trait = SanGong_TaiHuangTaihou has_trait = WJ_harem_guiren has_trait = WJ_harem_meiren has_trait = WJ_harem_gongren has_trait = WJ_harem_cainv }
}
WJ_harem_member = {
 is_alive = yes is_female = yes WJ_harem_has_role = yes
 exists = var:wj_harem_realm exists = var:wj_harem_realm.holder
 var:wj_harem_realm.holder = { WJ_inner_emperor_trigger = yes }
 top_liege = var:wj_harem_realm.holder
 OR = { has_variable = wj_harem_widow has_trait = SanGong_Taihou has_trait = SanGong_TaiHuangTaihou has_trait = WJ_harem_cainv is_consort_of = var:wj_harem_realm.holder }
}
WJ_harem_candidate = {
 is_alive = yes is_female = yes is_adult = yes
 top_liege = $EMPEROR$
 NOR = { has_trait = SanGong_Taihou has_trait = SanGong_TaiHuangTaihou has_variable = wj_harem_widow }
 OR = { is_consort_of = $EMPEROR$ AND = { has_trait = WJ_harem_cainv var:wj_harem_realm = $EMPEROR$.primary_title } }
}
WJ_harem_can_crown = {
 WJ_inner_emperor_trigger = yes
 save_temporary_scope_as = wj_crowning_emperor
 exists = scope:huanghou
 scope:huanghou = {
  WJ_harem_candidate = { EMPEROR = scope:wj_crowning_emperor }
  NOT = { has_trait = SanGong_GongWei_1 }
  OR = { primary_spouse = scope:wj_crowning_emperor has_trait = WJ_harem_guiren }
 }
 NOT = { any_consort = { is_alive = yes has_trait = SanGong_GongWei_1 } }
}
WJ_harem_intimacy_allowed = {
 WJ_inner_emperor_trigger = yes age >= 18 is_imprisoned = no not_involved_activity_trigger = yes
 exists = scope:ShiQie
 scope:ShiQie = {
  is_alive = yes age >= 18 is_female = yes is_imprisoned = no not_involved_activity_trigger = yes
  is_consort_of = root WJ_harem_member = yes
  NOR = { has_variable = wj_harem_widow has_trait = pregnant has_character_flag = ShiQin_flag }
 }
}
