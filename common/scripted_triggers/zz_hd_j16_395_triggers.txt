# 正以独立战争攻其主君（E5-J51 覆舟山）。Scope：角色
hd_j16_395_indep_attacker_trigger = {
	any_character_war = {
		primary_attacker = prev
		using_cb = independence_war
	}
}

# 刘裕可出征（E5-J70、J80；原 H4/H5 条件，is_at_war = no 改为「不是任何 hd_nbc 场次的主攻方」，06 §2.1）。Scope：刘裕
hd_j16_395_liuyu_campaign_ready_trigger = {
	is_alive = yes
	is_imprisoned = no
	NOT = {
		any_character_war = {
			primary_attacker = prev
			has_variable = hd_nbc_war_kind
		}
	}
	OR = {
		title:h_china.var:hd_court_regent ?= this
		AND = {
			is_independent_ruler = yes
			highest_held_title_tier >= tier_kingdom
		}
	}
}

# 某场次的战争已结束：节点存下的攻守双方都已没有该场次的战争（存的人已死也算没有；TEMPLATE §5-6）。用法：hd_j16_395_war_over_trigger = { KIND = nanyan }
hd_j16_395_war_over_trigger = {
	NOT = { global_var:hd_j16_$KIND$_att ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:$KIND$ } } }
	NOT = { global_var:hd_j16_$KIND$_def ?= { any_character_war = { var:hd_nbc_war_kind ?= flag:$KIND$ } } }
}

# 关键县归攻方一系：县主是攻方本人、攻方是其 top_liege，或县主与在世攻方同属一个 top_liege（攻方是天子的封臣时，战果可能落在天子名下）。
# 用法：hd_j16_395_side_holds_trigger = { COUNTY = c_uuii_p_2978 KIND = nanyan }
hd_j16_395_side_holds_trigger = {
	exists = global_var:hd_j16_$KIND$_att
	title:$COUNTY$.holder ?= {
		OR = {
			this = global_var:hd_j16_$KIND$_att
			top_liege = global_var:hd_j16_$KIND$_att
			AND = {
				global_var:hd_j16_$KIND$_att ?= { is_alive = yes }
				top_liege = global_var:hd_j16_$KIND$_att.top_liege
			}
		}
	}
}

# hd_nbc.0210 覆舟山的触发兜底：刘裕（原生人物或 hbirth 出场者）正以独立战争攻其主君。不依赖事件携带的 scope
hd_j16_395_liuyu_indep_war_trigger = {
	OR = {
		character:suiyuli_liu_013 ?= {
			is_alive = yes
			hd_j16_395_indep_attacker_trigger = yes
		}
		AND = {
			has_global_variable = hd_hb_p_suiyuli_liu_013
			global_var:hd_hb_p_suiyuli_liu_013 = {
				is_alive = yes
				hd_j16_395_indep_attacker_trigger = yes
			}
		}
	}
}
