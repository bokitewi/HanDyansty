# 人物身份入口：不保存官职，不把无地朝臣混入地方考课。
WJ_kaoke_eligible_trigger = {
 is_alive = yes
 is_adult = yes
 is_landed = yes
 is_independent_ruler = no
 highest_held_title_tier >= tier_county
 highest_held_title_tier <= tier_kingdom
 government_has_flag = government_is_celestial
 WJ_han_jin_law_trigger = yes
 OR = {
  WJ_has_celestial_province_contract_trigger = yes
  vassal_contract_has_flag = celestial_civil_appointment
  vassal_contract_has_flag = celestial_military_appointment
 }
}
WJ_kaoke_grade_usable_trigger = {
 WJ_kaoke_eligible_trigger = yes
 has_variable = wj_kaoke_last_grade
 has_variable = wj_kaoke_realm
 var:wj_kaoke_realm = top_liege.primary_title
 OR = {
  WJ_has_entry_road_trigger = yes
  # 试守也承受新考第的固定负分；不放宽正向考赏，不追罚旧制历史。
  AND = {
   var:wj_kaoke_schema_version = 2
   NOT = { has_variable = wj_kaoke_last_legacy }
   var:wj_kaoke_last_grade <= 2
  }
 }
}
WJ_kaoke_due_trigger = {
 WJ_kaoke_eligible_trigger = yes
 has_variable = wj_kaoke_start
 var:wj_kaoke_realm = top_liege.primary_title
 WJ_kaoke_days_value >= 1825
}
WJ_kaoke_reward_available_trigger = {
 WJ_kaoke_eligible_trigger = yes
 has_variable = wj_kaoke_reward_pending
 var:wj_kaoke_reward_pending = 1
 var:wj_kaoke_last_grade >= 4
 var:wj_kaoke_realm = top_liege.primary_title
}
WJ_kaoke_reward_claimable_trigger = {
 WJ_kaoke_reward_available_trigger = yes
 exists = scope:wj_kaoke_reward_date
 exists = scope:wj_kaoke_reward_epoch
 exists = scope:wj_kaoke_reward_realm
 var:wj_kaoke_last_date = scope:wj_kaoke_reward_date
 var:wj_kaoke_window_epoch = scope:wj_kaoke_reward_epoch
 var:wj_kaoke_realm = scope:wj_kaoke_reward_realm
 top_liege.primary_title = scope:wj_kaoke_reward_realm
}
WJ_kaoke_xp_reward_available_trigger = {
 var:wj_kaoke_last_trial = 0
 WJ_has_entry_road_trigger = yes
 has_trait = WJ_mingshi_trait
 OR = {
  AND = {
   var:wj_kaoke_last_military_award = 1
   has_trait_xp = { trait = WJ_mingshi_trait track = jungong value < 100 }
  }
  AND = {
   var:wj_kaoke_last_military_award = 0
   has_trait_xp = { trait = WJ_mingshi_trait track = zhengji value < 100 }
  }
 }
}
