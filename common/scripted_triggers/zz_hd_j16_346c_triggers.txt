# 两晋十六国统一史实线 346 节 · 四名可选人物与刘玄（03 §4、§5；08 §9 的 4-C）触发器。写入者：p4-C。
# 节内共用触发器（zz_hd_j16_346_triggers.txt）归 4-B；本路只放自己用到的，前缀 hd_j16_346c_。

# 某郡里有县与对方接壤：$DUCHY$ 内有县属 $DEF$ 一方（持有者即 $DEF$ 或其下属），且该县邻接 $ATT$ 一方（持有者即 $ATT$ 或其下属）的县。
# $ATT$/$DEF$ 为人物 scope（须为顶层君主）。用于 E3-L08 龙黎（d_tianshui）等「先判接壤再问玩家」的边境战。Scope：任意
hd_j16_346c_duchy_front_trigger = {
	$DUCHY$ = {
		any_de_jure_county = {
			holder ?= {
				OR = {
					this = $DEF$
					is_vassal_or_below_of = $DEF$
				}
			}
			any_neighboring_county = {
				holder ?= {
					OR = {
						this = $ATT$
						is_vassal_or_below_of = $ATT$
					}
				}
			}
		}
	}
}

# 两个顶层君主的领土接壤（$A$ 一方某县邻接 $B$ 一方的县）。Scope：任意。只在 E3 段节点里、条件都已满足后最后判（开销较大）
hd_j16_346c_realms_adjacent_trigger = {
	$A$ = {
		any_realm_county = {
			any_neighboring_county = {
				holder ?= {
					OR = {
						this = $B$
						is_vassal_or_below_of = $B$
					}
				}
			}
		}
	}
}

# 人物在某君主的国境内：本人、或其雇主，是该君主或其下属。Scope：人物
hd_j16_346c_in_realm_trigger = {
	OR = {
		this = $RULER$
		is_vassal_or_below_of = $RULER$
		employer ?= {
			OR = {
				this = $RULER$
				is_vassal_or_below_of = $RULER$
			}
		}
	}
}

# 主头衔已称王以上（读 hd_nbc_style：wang / tianwang / huangdi；gong 不算，与 3-D zgui_xizhou 同口径）。Scope：人物
hd_j16_346c_styled_king_trigger = {
	primary_title ?= {
		OR = {
			var:hd_nbc_style ?= flag:wang
			var:hd_nbc_style ?= flag:tianwang
			var:hd_nbc_style ?= flag:huangdi
		}
	}
}

# 刘玄幻想卡在手（野望 lxuan_xinghan 进行中，玩家）。Scope：人物
hd_j16_346c_lxuan_card_trigger = {

	var:hd_asp_aspiration ?= flag:lxuan_xinghan
}

# 人在益州（汉代静态区域 uuii_han_yizhou_region；报错清单「汉代州归属别用 k_uuii_* 法理」）。Scope：人物
hd_j16_346c_in_yizhou_trigger = {
	location ?= { geographical_region = uuii_han_yizhou_region }
}

# 持有益州（同上静态区域）任一县。Scope：人物
hd_j16_346c_holds_yizhou_county_trigger = {
	any_held_title = {
		tier = tier_county
		title_province ?= { geographical_region = uuii_han_yizhou_region }
	}
}

# 「蜀中起兵」可取之县：在益州（汉代静态区域）、持有者是 AI、非独立君主、不是决议者本人，且不是持有者的首府县。Scope：县头衔（root = 决议者）
hd_j16_346c_qibing_county_trigger = {
	title_province ?= { geographical_region = uuii_han_yizhou_region }
	holder ?= {
		is_ai = yes
		is_independent_ruler = no
		NOT = { this = root }
		NOT = { capital_county ?= prev }
	}
}
