# 清议 / 月旦评触发器（清议重制，2026-10-08）。

# Character scope。可选月旦评意图：
# 15 年冷却外，且（威望等级 ≥4 或 名士特质清议轨道满级）。
WJ_qingyi_yuedan_host_eligible_trigger = {
	NOT = { has_variable = wj_yuedan_host_cooldown }
	OR = {
		prestige_level >= 4
		AND = {
			has_trait = WJ_mingshi_trait
			has_trait_xp = {
				trait = WJ_mingshi_trait
				track = qingtan
				value >= 100
			}
		}
	}
}

# Character scope（候选人），scope:host = 评主。
# 在世成年、威望等级 ≥2、在评主外交范围内（评主本人可入选），
# 未持名士特质，且不在另一场进行中的月旦评名单里。
WJ_qingyi_yuedan_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	prestige_level >= 2
	NOT = { has_trait = hd_yuedan_mingshi }
	NOT = { has_variable = wj_yd_host }
	OR = {
		this = scope:host
		in_diplomatic_range = scope:host
	}
}

# Character scope。属于朋党（参与者组为党派类，而非天子/诸侯/四夷统治者组）。
WJ_qingyi_has_party_trigger = {
	top_participant_group:dynastic_cycle ?= {
		OR = {
			participant_group_type = pro_hegemon_movement
			participant_group_type = undecided_movement
			participant_group_type = expansion_movement
			participant_group_type = conservative_movement
			participant_group_type = advancement_movement
			participant_group_type = hd_siyi_movement
		}
	}
}

# Character scope（候选人），scope:activity。仍在本场当前轮次中。
# var:wj_yd_round：1 = 提名（8 人），2 = 入第三轮（4 人），3 = 当选。
WJ_qingyi_yuedan_in_round_trigger = {
	is_alive = yes
	has_variable = wj_yd_host
	var:wj_yd_host = scope:activity.activity_host
	has_variable = wj_yd_round
	var:wj_yd_round >= $ROUND$
}

# Character scope。品评模板分类（按顺序取第一个符合的），提名时写入 var:wj_yd_class。
# 1 雄主 2 奸雄 3 名将 4 能臣 5 谋士 6 名儒
WJ_qingyi_yd_class_xiongzhu_trigger = {
	highest_held_title_tier >= tier_kingdom
}
WJ_qingyi_yd_class_jianxiong_trigger = {
	OR = {
		has_trait = deceitful
		has_trait = ambitious
	}
	intrigue >= WJ_qingyi_max_skill
}

# 任意 scope；scope:activity。本场月旦评正在进行（已提名、尚未定评）。
WJ_qingyi_yuedan_running_trigger = {
	scope:activity = {
		has_variable = wj_yd_active
		NOT = { has_variable = wj_yd_done }
	}
}

# Character scope（候选人），scope:activity。仍在本轮的有效候选人：
# 第一、二阶段为 8 人名单，第三阶段为 4 人终评。
WJ_qingyi_yuedan_live_candidate_trigger = {
	WJ_qingyi_yuedan_running_trigger = yes
	WJ_qingyi_yuedan_in_round_trigger = { ROUND = 1 }
	trigger_if = {
		limit = { scope:activity = { has_variable = wj_yd_vote_round } }
		var:wj_yd_round >= scope:activity.var:wj_yd_vote_round
	}
}
