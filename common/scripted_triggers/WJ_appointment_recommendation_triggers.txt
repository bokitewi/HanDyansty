WJ_appointment_nominee_range = {
 is_alive = yes
 NOT = { this = $EMPLOYER$ }
 custom_tooltip = {
  text = WJ_appointment_nominee_relation_tt
  OR = {
   this = $RECOMMENDER$
   $RECOMMENDER$ = { has_relation_disciple = prev }
   AND = { exists = puppeteer puppeteer = $RECOMMENDER$ }
   AND = { WJ_mufu_member_valid = yes exists = var:wj_mufu_owner var:wj_mufu_owner = $RECOMMENDER$ }
   AND = { exists = house exists = $RECOMMENDER$.house house = $RECOMMENDER$.house }
  }
 }
 custom_tooltip = {
  text = WJ_appointment_nominee_realm_tt
  OR = {
   top_liege = $RECOMMENDER$.top_liege
   AND = { is_landed = no NOT = { exists = liege_or_court_owner } }
  }
 }
}

  AND = { is_diarch_of_target = top_liege top_liege = { OR = { WJ_name_celestial_secretariat_trigger = yes highest_held_title_tier < tier_hegemony } } }

}
