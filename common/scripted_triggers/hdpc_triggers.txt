# All custom triggers are character-scoped unless stated otherwise.
hdpc_is_celestial_trigger = {
    OR = {
        has_government = celestial_government
        government_has_flag = government_is_celestial
        hdpc_extra_celestial_trigger = yes
    }
}

# Only the two approved court types are opted in.
hdpc_cleanup_government_trigger = {
    OR = {
        has_government = tribal_government
        has_government = nomad_government
    }
}

hdpc_court_allowed_trigger = {
    is_alive = yes
    is_ai = yes
    is_landed = yes
    highest_held_title_tier >= tier_county
    hdpc_cleanup_government_trigger = yes
    hdpc_is_celestial_trigger = no
    NOT = { has_character_flag = hdpc_protected_realm }
    # Every intermediate liege is checked, not just the top liege.
    NOT = {
        any_liege_or_above = {
            OR = {
                is_ai = no
                hdpc_cleanup_government_trigger = no
                hdpc_is_celestial_trigger = yes
                has_character_flag = hdpc_protected_realm
            }
        }
    }
}

# Shared protection for both pool and court death candidates.
# No broad genealogy search, no spouse-candidate search.
hdpc_unimportant_adult_trigger = {
    is_alive = yes
    is_ai = yes
    is_adult = yes
    age >= hdpc_min_age_value
    is_ruler = no
    is_landed = no
    is_lowborn = yes
    is_married = no
    NOT = { any_child = { is_alive = yes } }
    hdpc_is_celestial_trigger = no
    hdpc_extra_character_protection_trigger = no
    is_available_ai_adult = yes
    is_pregnant = no
    is_betrothed = no
    is_concubine = no
    has_any_court_position = no
    is_diarch = no
    is_designated_diarch = no
    is_theocratic_lessee = no
    has_owned_scheme = no
    owns_a_story = no
    diplomacy < hdpc_talent_threshold_value
    martial < hdpc_talent_threshold_value
    stewardship < hdpc_talent_threshold_value
    intrigue < hdpc_talent_threshold_value
    learning < hdpc_talent_threshold_value
    prowess < hdpc_talent_threshold_value
    NOR = {
        has_character_flag = hdpc_protected
        has_character_flag = hdpc_scheme_agent_guard
        has_character_flag = blocked_from_leaving
        has_character_flag = easteregg
        has_character_flag = is_in_task_contract_event_chain
        has_trait = historical_character
        has_trait = adventurer
        has_trait = adventurer_follower
        has_trait = heresiarch
        has_trait = populist_leader
        exists = inspiration
        exists = involved_activity
        any_claim = { }
        any_heir_title = { }
        any_character_artifact = { }
        any_pinning_character = { }
        any_targeting_scheme = { }
        any_concubine = { }
        any_relation = { type = friend }
        any_relation = { type = best_friend }
        any_relation = { type = lover }
        any_relation = { type = soulmate }
        any_relation = { type = rival }
        any_relation = { type = nemesis }
        any_relation = { type = guardian }
        any_relation = { type = ward }
        any_relation = { type = mentor }
        any_relation = { type = student }
        any_close_or_extended_family_member = {
            OR = {
                is_ruler = yes
                is_ai = no
                hdpc_is_celestial_trigger = yes
                has_character_flag = hdpc_protected
                has_trait = historical_character
            }
        }
    }
}

# Requires scope:hdpc_court and scope:liege, both saved by the court executor.
# This check is run once when queued and again immediately before death.
hdpc_court_death_candidate_trigger = {
    is_courtier_of = scope:hdpc_court
    hdpc_unimportant_adult_trigger = yes
    NOT = { is_councillor_of = scope:hdpc_court }
    NOT = { is_knight_of = scope:hdpc_court }
    NOT = { is_tax_collector_of = scope:hdpc_court }
    # Preserve vanilla departure constraints, including liege-heir protections.
    courtier_allowed_to_leave_trigger = yes
}

# Pool characters only, not all landless characters or adventurer camps.
hdpc_pool_death_candidate_trigger = {
    is_pool_character = yes
    hdpc_unimportant_adult_trigger = yes
    NOT = {
        any_liege_or_above = {
            OR = {
                is_ai = no
                hdpc_is_celestial_trigger = yes
                has_character_flag = hdpc_protected_realm
            }
        }
    }
    NOT = { has_character_flag = hdpc_protected_realm }
}

# Rechecked after every death; never exceed the cap or cross the stop target.
hdpc_death_budget_available_trigger = {
    global_var:hdpc_killed_last < global_var:hdpc_death_budget
    global_var:hdpc_living > hdpc_target_low_value
}
