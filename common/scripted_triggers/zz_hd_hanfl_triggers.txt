# 6.1 考课公务（hd_hanfl.1001~1011）
############################################################
# Scope：角色。事件 trigger，照 wj_kaoke_duties.1001
hd_hanfl_kaoke_event_trigger = {
	WJ_kaoke_eligible_trigger = yes
	has_variable = wj_kaoke_start
	WJ_kaoke_days_value < 1825
	has_variable = wj_kaoke_window_epoch
	has_variable = wj_kaoke_realm
	var:wj_kaoke_realm = top_liege.primary_title
}

# Scope：角色。选项重验，照 wj_kaoke_duties.1001 选项里的 14 行
hd_hanfl_kaoke_choice_valid_trigger = {
	WJ_kaoke_eligible_trigger = yes
	has_variable = wj_kaoke_start
	WJ_kaoke_days_value < 1825
	has_variable = wj_kaoke_window_epoch
	has_variable = wj_kaoke_realm
	exists = scope:wj_kaoke_choice_start
	exists = scope:wj_kaoke_choice_epoch
	exists = scope:wj_kaoke_choice_realm
	var:wj_kaoke_start = scope:wj_kaoke_choice_start
	var:wj_kaoke_window_epoch = scope:wj_kaoke_choice_epoch
	var:wj_kaoke_realm = scope:wj_kaoke_choice_realm
	top_liege.primary_title = scope:wj_kaoke_choice_realm
	exists = scope:wj_kaoke_duty_county
	scope:wj_kaoke_duty_county.holder = root
}

# 身份与地区
############################################################
# Scope：角色。天子本人
hd_hanfl_is_tianzi_trigger = {
	exists = title:h_china.holder
	this = title:h_china.holder
}

# Scope：角色。天子或天朝独立皇帝（D5）
hd_hanfl_is_emperor_trigger = {
	is_independent_ruler = yes
	government_has_flag = government_is_celestial
	exists = primary_title
	primary_title = { tier >= tier_empire }
}

# Scope：角色。首府的法理郡是 $DUCHY$
hd_hanfl_capital_in_duchy_trigger = {
	exists = capital_county
	capital_county.duchy ?= title:$DUCHY$
}

# Scope：角色。首府的法理州是 $KINGDOM$
hd_hanfl_capital_in_kingdom_trigger = {
	exists = capital_county
	capital_county.kingdom ?= title:$KINGDOM$
}

# Scope：县。首府省份不是楚巫仪轨（D9 不碰楚巫）
hd_hanfl_county_not_chuwu_trigger = {
	title_province = {
		NOT = { rite ?= rite:chuwu }
	}
}

############################################################
# D5 天人感应
############################################################
# Scope：角色。本人是地震、洪水、饥荒局势里的受灾统治者（照原版 10_tgp_natural_disaster_triggers.txt:3）
hd_hanfl_in_disaster_trigger = {
	any_character_situation = {
		OR = {
			situation_type = natural_disaster_earthquake
			situation_type = natural_disaster_flood
			situation_type = natural_disaster_hd_famine
		}
		any_participant_group = {
			participant_group_type = affected_ruler
			participant_group_has_character = root
		}
	}
}

# Scope：角色。持有《伤寒杂病论》，或张机在宫（或就是本人）
hd_hanfl_has_shanghan_trigger = {
	OR = {
		any_character_artifact = { has_variable = han_artifact_shanghan_zabing_lun }
		AND = {
			exists = global_var:hd_hanfl_zhang_ji
			global_var:hd_hanfl_zhang_ji = {
				is_alive = yes
				OR = {
					this = root
					is_courtier_of = root
				}
			}
		}
	}
}

# Player-local ordinary case only. This is not a central judicial or appointment shortcut.
hd_hanfl_revenge_case_valid_trigger = {
 hd_hanfl_kaoke_choice_valid_trigger = yes
 is_ai = no
 is_alive = yes
 is_imprisoned = no
 is_incapable = no
 exists = scope:hd_hanfl_revenge_judge
 this = scope:hd_hanfl_revenge_judge
 exists = scope:hd_hanfl_revenge_woman
 scope:hd_hanfl_revenge_woman = {
  is_alive = yes
  is_adult = yes
  is_ruler = no
  has_character_flag = hd_hanfl_revenge_subject
  var:hd_hanfl_revenge_judge = root
  has_variable = hd_hanfl_revenge_judge
  is_imprisoned = yes
  is_imprisoned_by = root
 }
}

# Ordinary local participants, bound to this scene and official; no real ministers.
WJ_duty_person_available_trigger = {
	is_available_adult = yes is_alive = yes is_ruler = no is_imprisoned = no this != root
	location = root.location
	OR = { NOT = { exists = employer } employer = root }
	has_character_flag = WJ_duty_subject has_variable = wj_duty_official var:wj_duty_official = root
}
WJ_duty_people_valid_trigger = {
	hd_hanfl_kaoke_choice_valid_trigger = yes
	is_ai = no is_available_adult = yes
	exists = scope:wj_duty_official this = scope:wj_duty_official
	exists = scope:wj_duty_person
	scope:wj_duty_person = { WJ_duty_person_available_trigger = yes }
	trigger_if = { limit = { OR = { exists = scope:wj_duty_land_case exists = scope:wj_duty_pair_required } } exists = scope:wj_duty_other scope:wj_duty_other = { WJ_duty_person_available_trigger = yes } }
}

# Rite case has no generated actor. Its saved one-use token belongs to this window.
WJ_duty_ritual_valid_trigger = {
	hd_hanfl_kaoke_choice_valid_trigger = yes is_ai = no is_available_adult = yes
	exists = scope:wj_duty_official this = scope:wj_duty_official
	exists = scope:wj_duty_ritual_open scope:wj_duty_ritual_open = 1
	scope:wj_kaoke_duty_county = { hd_hanfl_county_not_chuwu_trigger = yes }
}

# A current captive must not pass through the FREE-person availability trigger.
WJ_duty_prisoner_valid_trigger = {
	hd_hanfl_kaoke_choice_valid_trigger = yes is_ai = no is_available_adult = yes
	exists = scope:wj_duty_official this = scope:wj_duty_official
	exists = scope:wj_duty_prison_case scope:wj_duty_prison_case = 1
	exists = scope:wj_duty_person
	scope:wj_duty_person = {
		is_alive = yes is_adult = yes is_ruler = no has_character_flag = WJ_duty_subject has_variable = wj_duty_official var:wj_duty_official = root
		is_imprisoned = yes is_imprisoned_by = root
	}
}
