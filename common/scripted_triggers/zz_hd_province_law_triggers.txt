# 2026-10-31 擅兴律/疆界之制：原版行省宣战（province_war_law）、行省边界（province_border_law）两组法律放宽给郡国制
# 设计见 docs/郡国制_擅兴律与疆界之制_设计.md

# 能持有这两组法律的政体：行政制（帝国官僚）或郡国制（天朝官僚）
hd_province_law_valid_trigger = {
	OR = {
		realm_law_use_imperial_bureaucracy = yes
		realm_law_use_celestial_bureaucracy = yes
	}
}

# 不受擅兴律约束的封臣（攻方角色作用域）：军阀契约、旄节（假节/持节/使持节）、假黄钺
hd_province_war_law_exempt_trigger = {
	OR = {
		hd_warlord_trigger = yes
		vassal_contract_has_flag = WJ_jiajie
		vassal_contract_has_flag = WJ_chijie
		vassal_contract_has_flag = WJ_shichijie
		vassal_contract_has_flag = WJ_huangyue_granted
	}
}

# 封臣须与直属主君同档；主君本身不能持有这两组法律时不同步（封臣的档位不影响宣战，宣战只看最高主君）
hd_province_law_sync_trigger = {
	top_liege != this
	liege = { hd_province_law_valid_trigger = yes }
}
