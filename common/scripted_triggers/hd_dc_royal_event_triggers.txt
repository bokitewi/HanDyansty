# Event recipient and saved delivery version; rechecked by each action.
hd_dc_roy01_window_valid_trigger = {
	is_alive = yes
	this = title:h_china.holder
	exists = scope:hd_dc_instance
	exists = scope:hd_dc_roy01_version
	scope:hd_dc_instance = {
		var:hd_dc_type = flag:roy
		var:hd_dc_stage = flag:acute
		var:hd_dc_roy01_version = scope:hd_dc_roy01_version
		has_variable = hd_dc_roy01_delivery_wait
		var:hd_dc_roy01_author = title:h_china.holder
		NOT = { has_variable = hd_dc_roy01_result }
		NOT = { has_variable = hd_dc_roy01_action }
		hd_dc_roy01_parties_valid_trigger = yes
	}
}

# Main instance scope. A current dispute needs two living, free claimants.
hd_dc_roy01_seat_claim_trigger = {
	save_temporary_scope_as = hd_dc_seat_petitioner
	OR = {
		var:hd_dc_ritual_old_rank > 1
		scope:hd_dc_instance = {
			any_in_list = {
				variable = hd_dc_actor_stories
				this != scope:hd_dc_seat_petitioner
				var:hd_dc_ritual_old_rank = 1
				var:hd_dc_actor = { is_alive = yes }
			}
		}
	}
}

hd_dc_roy01_parties_valid_trigger = {
	exists = title:h_china.holder
	title:h_china.holder = { is_alive = yes }
	exists = var:hd_dc_roy01_favored.var:hd_dc_actor
	exists = var:hd_dc_roy01_displaced.var:hd_dc_actor
	var:hd_dc_roy01_favored != var:hd_dc_roy01_displaced
	var:hd_dc_roy01_favored = {
		var:hd_dc_instance_ref = scope:hd_dc_instance
		var:hd_dc_actor = {
			is_alive = yes
			is_imprisoned = no
			NOT = { is_at_war_with = title:h_china.holder }
		}
	}
	var:hd_dc_roy01_displaced = {
		var:hd_dc_instance_ref = scope:hd_dc_instance
		var:hd_dc_actor = {
			is_alive = yes
			is_imprisoned = no
			NOT = { is_at_war_with = title:h_china.holder }
		}
	}
}

# Dynasty-member character scope; no invented minister or presumed consent.
hd_dc_roy01_neutral_elder_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	NOT = { has_trait = incapable }
	this != title:h_china.holder
	dynasty = title:h_china.holder.dynasty
	age > scope:hd_dc_favored.age
	age > scope:hd_dc_displaced.age
	NOT = { has_claim_on = title:h_china }
	NOT = { is_at_war_with = title:h_china.holder }
	opinion = { target = scope:hd_dc_favored value >= 20 }
	opinion = { target = scope:hd_dc_displaced value >= 20 }
	scope:hd_dc_favored = { opinion = { target = prev value >= 20 } }
	scope:hd_dc_displaced = { opinion = { target = prev value >= 20 } }
}

hd_dc_royal_invitation_delivery_valid_trigger = {
	is_alive = yes
	exists = title:h_china.holder
	exists = scope:hd_dc_actor_story
	exists = scope:hd_dc_invitation_version
	scope:hd_dc_actor_story = {
		var:hd_dc_actor = root
		var:hd_dc_invitation_version = scope:hd_dc_invitation_version
		NOT = { has_variable = hd_dc_invitation_answer }
		var:hd_dc_inviter = title:h_china.holder
		var:hd_dc_instance_ref = {
			var:hd_dc_type = flag:roy
			NOT = { var:hd_dc_stage = flag:closed }
		}
	}
	NOT = { is_at_war_with = title:h_china.holder }
	is_imprisoned = no
}

	}
