# Extra claimant pulse: native interaction eligibility, not its disabled UI can_send.
# Character = possible creator; the target/title/claimant are frozen real scopes.
hd_dc_extra_claimant_creator_trigger = {
	is_ai = yes
	is_alive = yes
	is_a_faction_member = no
	highest_held_title_tier > tier_barony
	liege = scope:hd_dc_extra_faction_target
	scope:hd_dc_extra_faction_target = { NOT = { any_targeting_faction = { faction_is_type = $TYPE$ } } }
	NOT = { has_truce = scope:hd_dc_extra_faction_target }
	scope:hd_dc_extra_faction_target = { NOT = { has_strong_hook = prev } }
	can_join_or_create_faction_against = {
		who = scope:hd_dc_extra_faction_target
		faction = $TYPE$
		check_in_a_faction = no
	}
	trigger_if = {
		limit = { government_allows = administrative scope:hd_dc_extra_faction_target = { government_allows = administrative } }
		scope:hd_dc_extra_faction_target = { is_independent_ruler = yes }
	}
}

# Character = actual claimant enumerated from the selected title.
# Mirrors the native interaction and the claimant faction's continued validity.
hd_dc_extra_claimant_person_trigger = {
	save_temporary_scope_as = hd_dc_extra_probe_claimant
	is_alive = yes
	has_claim_on = scope:hd_dc_extra_probe_title
	NOR = {
		this = scope:hd_dc_extra_faction_target
		has_trait = incapable
		is_at_war_with = scope:hd_dc_extra_probe_creator
	}
	scope:hd_dc_extra_probe_title = { holder = scope:hd_dc_extra_faction_target }
	trigger_if = {
		limit = { scope:hd_dc_extra_probe_title.holder.culture ?= { has_cultural_parameter = physical_disfigurement_blocks_inheritance } }
		valid_for_byz_succession_trigger = yes
	}
	trigger_if = {
		limit = { is_imprisoned_by = scope:hd_dc_extra_faction_target }
		time_in_prison_type = { years < 5 }
	}
	trigger_if = {
		limit = { OR = { tgp_is_any_minister = yes tgp_has_minister_title = yes } }
		any_held_title = { tgp_is_minister_title = yes de_jure_liege = scope:hd_dc_extra_probe_title }
	}
	trigger_if = {
		limit = { tgp_realm_has_ceremonial_liege_trigger = yes }
		tgp_is_ceremonial_regent_trigger = no
	}
	trigger_if = {
		limit = { scope:hd_dc_extra_probe_creator.top_liege.primary_title.var:administrative_ui_special_title ?= scope:hd_dc_extra_probe_title }
		tgp_is_in_ceremonial_house_trigger = yes
		trigger_if = {
			limit = {
				scope:hd_dc_extra_probe_title = {
					holder.faith = { has_doctrine = doctrine_gender_male_dominated }
					any_claimant = { is_male = yes }
				}
			}
			is_male = yes
		}
		trigger_else_if = {
			limit = {
				scope:hd_dc_extra_probe_title = {
					holder.faith = { has_doctrine = doctrine_gender_female_dominated }
					any_claimant = { is_female = yes }
				}
			}
			is_female = yes
		}
	}
	scope:hd_dc_extra_faction_target = {
		NOT = {
			any_targeting_faction = {
				faction_is_type = $TYPE$
				special_character = scope:hd_dc_extra_probe_claimant
				special_title = scope:hd_dc_extra_probe_title
			}
		}
	}
}

# A type gets one ticket only after prepare froze a complete valid tuple.
hd_dc_extra_claimant_available_trigger = {
	exists = scope:hd_dc_extra_$TYPE$_creator
	exists = scope:hd_dc_extra_$TYPE$_title
	exists = scope:hd_dc_extra_$TYPE$_claimant
}
