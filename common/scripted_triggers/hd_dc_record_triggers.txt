# Only outstanding work remains in the monthly queue. Archived records retain
# their identity, actual receipts and references; archive never means forgiven.
hd_dc_record_needs_monthly_trigger = {
	OR = {
		has_variable = hd_dc_keeper_missing
		AND = { var:hd_dc_record_kind = flag:party_case NOT = { has_variable = hd_dc_case_result } }
		AND = { var:hd_dc_record_kind = flag:policy_trial NOT = { has_variable = hd_dc_policy_result } }
		AND = { var:hd_dc_record_kind = flag:proposal var:hd_dc_proposal_state = flag:waiting }
		AND = { var:hd_dc_record_kind = flag:war NOT = { has_variable = hd_dc_core_war_result } }
		AND = {
			var:hd_dc_record_kind = flag:ordinary
			OR = {
				NOT = { var:hd_dc_stage = flag:closed }
				var:hd_dc_ransom_payment ?= { var:hd_dc_unreturned_debit > 0 }
			}
		}
		AND = {
			var:hd_dc_record_kind = flag:royal_support
			OR = {
				NOT = { has_variable = hd_dc_support_answer }
				AND = { has_variable = hd_dc_pause_started NOT = { has_variable = hd_dc_pause_result } }
				AND = { var:hd_dc_support_answer = flag:protection_requested NOT = { has_variable = hd_dc_protection_answer } }
				AND = { var:hd_dc_protection_answer = flag:accepted NOT = { has_variable = hd_dc_protection_result } }
				AND = { var:hd_dc_support_reward_due = yes NOT = { has_variable = hd_dc_support_reward_result } }
			}
		}
		AND = {
			var:hd_dc_record_kind = flag:payment
			OR = { NOT = { var:hd_dc_payment_commit = 2 } var:hd_dc_unreturned_debit > 0 }
		}
		AND = { var:hd_dc_record_kind = flag:theft NOT = { var:hd_dc_payment_commit = 2 } }
	}
}
