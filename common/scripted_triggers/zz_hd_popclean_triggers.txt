# 开局人口清除：清理无王朝、无封地的廷臣与游荡者。分档见 zz_hd_popclean_values.txt。

# 游荡者：人才池角色或游历中的客人。
hd_popclean_wanderer_trigger = {
	OR = {
		is_pool_character = yes
		is_pool_guest = yes
	}
}

# 宿主：廷臣看 court_owner，客人看 host。
hd_popclean_host_in_hegemony_trigger = {
	OR = {
		court_owner ?= { OR = { this = title:h_china.holder top_liege = title:h_china.holder } }
		AND = {
			is_foreign_court_guest = yes
			host ?= { OR = { this = title:h_china.holder top_liege = title:h_china.holder } }
		}
	}
}

# 霸权范围：宿主是 h_china 持有者或其臣属；游荡者看所在郡县的领主。
hd_popclean_in_hegemony_trigger = {
	exists = title:h_china.holder
	OR = {
		hd_popclean_host_in_hegemony_trigger = yes
		AND = {
			hd_popclean_wanderer_trigger = yes
			location.county.holder ?= {
				OR = { this = title:h_china.holder top_liege = title:h_china.holder }
			}
		}
	}
}

# 伯爵、男爵的宫廷。
hd_popclean_low_host_trigger = {
	OR = {
		court_owner ?= { highest_held_title_tier <= tier_county }
		AND = {
			is_foreign_court_guest = yes
			host ?= { highest_held_title_tier <= tier_county }
		}
	}
}

# 部落、封建、游牧、牧民政体的宫廷。
hd_popclean_rough_gov_trigger = {
	OR = {
		has_government = tribal_government
		has_government = feudal_government
		has_government = nomad_government
		has_government = herder_government
	}
}

hd_popclean_rough_gov_host_trigger = {
	OR = {
		court_owner ?= { hd_popclean_rough_gov_trigger = yes }
		AND = {
			is_foreign_court_guest = yes
			host ?= { hd_popclean_rough_gov_trigger = yes }
		}
	}
}

# 保护名单：统治者的配偶/妾室/子女、现任议员/官职/军官、历史人物、稀有特质/高能力者等。
hd_popclean_protected_trigger = {
	OR = {
		is_ai = no
		is_ruler = yes
		any_held_title = { }
		has_character_flag = hd_popclean_protect
		has_character_flag = hd_historical_actor_protection_received
		any_consort = { is_ruler = yes }
		any_parent = { is_ruler = yes }
		is_councillor = yes
		has_any_court_position = yes
		tgp_is_any_minister = yes
		has_character_modifier = WJ_temp_official_modifier
		is_knight = yes
		is_commanding_army = yes
		has_trait_with_flag = WJ_mr_rank
		is_imprisoned = yes
		exists = involved_activity
		any_scheme = { }
		num_of_good_genetic_traits > 0
		diplomacy >= 15
		martial >= 15
		stewardship >= 15
		intrigue >= 15
		learning >= 15
		prowess >= 15
	}
}

# 选取时的候选人：无王朝平民，在宫廷或游荡，且不在保护名单。
hd_popclean_candidate_trigger = {
	is_lowborn = yes
	OR = {
		is_courtier = yes
		is_foreign_court_guest = yes
		hd_popclean_wanderer_trigger = yes
	}
	hd_popclean_protected_trigger = no
}

# 执行时复查：排期后半年内可能已婚给统治者、得封地或成为统治者子女的家长等。
hd_popclean_still_valid_trigger = {
	is_alive = yes
	is_ruler = no
	is_ai = yes
	NOR = {
		any_held_title = { }
		has_character_flag = hd_popclean_protect
		has_character_flag = hd_historical_actor_protection_received
		any_consort = { is_ruler = yes }
		any_parent = { is_ruler = yes }
	}
}
