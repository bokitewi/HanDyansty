# Faction scope. The pause binds this actual faction and its original claimant.
hd_dc_claimant_mobilization_paused_trigger = {
	faction_is_type = claimant_faction
	has_variable = hd_dc_mobilization_wait
	exists = var:hd_dc_pause_record
	var:hd_dc_pause_record.var:hd_dc_support_faction = this
	var:hd_dc_pause_record.var:hd_dc_support_claimant = special_character
	var:hd_dc_pause_record.var:hd_dc_support_ruler = faction_target
	NOT = { exists = faction_war }
}

hd_dc_can_pause_royal_support_trigger = {
	hd_dc_royal_offer_response_trigger = yes
	has_usable_hook = scope:hd_dc_support.var:hd_dc_support_claimant
	scope:hd_dc_support.var:hd_dc_support_faction = { NOT = { has_variable = hd_dc_mobilization_wait } }
}
