# ROY selection is exhaustive. Personality changes motives/AI weights, not an
# undocumented numeric cap on the number of eligible imperial relatives.
hd_dc_royal_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = yes
	exists = title:h_china.holder
	exists = dynasty
	dynasty = title:h_china.holder.dynasty
	this != title:h_china.holder
	highest_held_title_tier > tier_barony
	NOT = { has_trait = incapable }
	NOT = { is_imprisoned = yes }
	NOT = { is_at_war_with = title:h_china.holder }
	OR = {
		opinion = { target = title:h_china.holder value <= -20 }
		has_variable = hd_dc_verified_royal_breach
	}
	OR = {
		AND = {
			exists = joined_faction
			joined_faction = {
				faction_is_type = claimant_faction
				special_title = title:h_china
				special_character = prev
				faction_target = title:h_china.holder
				NOT = { exists = faction_war }
			}
		}
		AND = {
			is_a_faction_member = no
			can_join_or_create_faction_against = {
				who = title:h_china.holder
				faction = claimant_faction
				check_in_a_faction = yes
			}
		}
	}
}

# Actor record scope: all three references must still identify the same politics.
hd_dc_royal_story_has_faction_trigger = {
	exists = var:hd_dc_actor
	exists = var:hd_dc_claimant_faction
	var:hd_dc_actor = {
		is_alive = yes
		has_claim_on = title:h_china
	}
	var:hd_dc_claimant_faction = {
		faction_is_type = claimant_faction
		special_character = prev.var:hd_dc_actor
		special_title = title:h_china
		faction_target = title:h_china.holder
	}
}
