		tier = tier_kingdom

		is_noble_family_title = no
	}
}

# D7 豁免：当朝权臣（董卓）与剧本指定人选。作用域：角色。

		has_character_flag = hd_cp_merit_exempt
	}
}

# 第 3 期 群雄落位补足（zz_hd184r_placement_effects.txt）。需 scope:hd_hw_actor / hd_hw_duchy / hd_hw_cand。
############################################################

# 郡（公国）作用域：有县与首领本人或其领内接壤。
hd184r_duchy_adjacent_to_actor_trigger = {
	any_de_jure_county = {
		any_neighboring_county = {
			holder ?= {
				OR = {
					this = scope:hd_hw_actor
					any_liege_or_above = { this = scope:hd_hw_actor }
				}
			}
		}
	}
}

# 人物作用域（县或郡的持有人）：虚构人物，且所在势力的顶层领主也是虚构人物（已独立的虚构郡守领内）。
hd184r_hw_fict_realm_holder_trigger = {
	is_ai = yes
	has_character_flag = hd_hw_fict
	NOT = { has_character_flag = hd_hw_leader }
	NOT = { this = scope:hd_hw_actor }
	NOT = { any_liege_or_above = { this = scope:hd_hw_actor } }
	NOT = { hd_hw_player_above_trigger = yes }
	top_liege = {
		is_ai = yes
		has_character_flag = hd_hw_fict
		NOT = { has_character_flag = hd_hw_leader }
		NOT = { has_character_flag = hd_hw_keep }
		NOT = { title:h_china.holder ?= this }
	}
}

# 县作用域：照 hd_hw_grab_county_trigger，另允许夺取已独立虚构人物领内虚构人物的县。
hd184r_hw_grab_county_trigger = {
	OR = {
		hd_hw_grab_county_trigger = yes
		AND = {
			NOT = { hd_hw_royal_capital_trigger = yes }
			holder ?= { hd184r_hw_fict_realm_holder_trigger = yes }
		}
	}
}

# 郡作用域：郡头衔可接手（照 hd_hw_duchy_takeable_trigger，另允许已独立虚构人物领内的虚构郡守）。
hd184r_hw_duchy_takeable_trigger = {
	tier = tier_duchy
	OR = {
		hd_hw_duchy_takeable_trigger = yes
		holder ?= { hd184r_hw_fict_realm_holder_trigger = yes }
	}
}

# 郡作用域：至少有一个可夺取的县。
hd184r_hw_duchy_usable_trigger = {
	tier = tier_duchy
	any_in_de_jure_hierarchy = {
		tier = tier_county
		hd184r_hw_grab_county_trigger = yes
	}
}

# 第 4 期 导演扩张引擎（zz_hd184r_director_effects.txt）。需 scope:hd184r_actor。
############################################################

# 名单人物可由导演层开战。Scope：名单人物。
hd184r_director_actor_ready_trigger = {
	is_alive = yes
	is_ai = yes
	is_adult = yes
	is_landed = yes
	is_imprisoned = no
	hd_court_independent_or_regent_trigger = yes
	NOT = { government_has_flag = cannot_be_vassal_or_liege }
	hd_dong_transition_pending_trigger = no
	# 冀州交接未完时，袁绍、公孙瓒不另开战（同天意青睐引擎）
	trigger_if = {
		limit = { exists = global_var:hd_yuan_jizhou_pending }
		NOT = { this = character:yuan_shao }
		NOT = { this = character:gong_sun_zan }
	}
}

# 进攻战（本人为主攻方）少于 4 场（D23）。Scope：名单人物。
hd184r_director_war_slot_trigger = {
	NOT = {
		any_character_war = {
			count >= 4
			primary_attacker = prev
		}
	}
}

# 合法的守方（前线县的顶层领主）。Scope：角色。
hd184r_target_ok_trigger = {
	is_alive = yes
	is_landed = yes
	NOT = { this = scope:hd184r_actor }
	NOT = { is_allied_to = scope:hd184r_actor }
	NOT = { is_at_war_with = scope:hd184r_actor }
	NOT = { scope:hd184r_actor = { has_truce = prev } }
	NOT = { has_title = title:h_china }
	NOT = { character:liu_xie ?= this }
	current_military_strength <= scope:hd184r_actor.hd_strat_strength_ceiling_value
	# 官渡之前曹操、袁绍互不攻伐（同天意青睐引擎）
	trigger_if = {
		limit = { NOT = { exists = global_var:hd_guandu_world_resolved } }
		NOT = { AND = { character:yuan_shao ?= this character:cao_cao ?= scope:hd184r_actor } }
		NOT = { AND = { character:cao_cao ?= this character:yuan_shao ?= scope:hd184r_actor } }
	}
}

# 前线县：在别人领内，与 scope:hd184r_actor 领内的县接壤，且守方合法。Scope：县。
hd184r_county_valid_front_trigger = {
	exists = holder
	NOT = {
		holder = {
			OR = {
				this = scope:hd184r_actor
				is_vassal_or_below_of = scope:hd184r_actor
			}
		}
	}
	any_neighboring_county = {
		holder ?= {
			OR = {
				this = scope:hd184r_actor
				is_vassal_or_below_of = scope:hd184r_actor
			}
		}
	}
	holder = { top_liege = { hd184r_target_ok_trigger = yes } }
}

# 郡里至少有一个前线县。Scope：郡。
hd184r_duchy_has_valid_defender_trigger = {
	any_de_jure_county = { hd184r_county_valid_front_trigger = yes }
}

# D37 韩馥让冀州：公孙瓒有地，且领地与冀州接壤（原为持有幽州王国头衔）。Scope：公孙瓒。

# D44-1 曹操已拿下兖州：持有兖州王国头衔、导演层兖州阶段已完成，或实控兖州七成县（玩家曹操不走导演层）。Scope：曹操。
hd184r_cao_yanzhou_secured_trigger = {
	OR = {
		has_title = title:k_uuii_yanzhou
		AND = {
			has_variable = hd184r_stage
			var:hd184r_stage >= 2
		}
		title:k_uuii_yanzhou = {
			any_de_jure_county = {
				percent >= 0.7
				holder ?= { OR = { this = character:cao_cao is_vassal_or_below_of = character:cao_cao } }
			}
		}
	}
}

# D44-1 陶谦领内、与曹操领地接壤的县。Scope：县。
hd184r_cao_tao_border_county_trigger = {
	holder.top_liege ?= character:tao_qian
	any_neighboring_county = {
		holder ?= { OR = { this = character:cao_cao is_vassal_or_below_of = character:cao_cao } }
	}
}

# D44-1 曹嵩遇害可触发：曹操已据兖州，且与陶谦的徐州接壤（2026-10-24 boki：原先只看曹操独立有地，190 开局即满足）。Scope：任意。
hd184r_caosong_ready_trigger = {
	character:cao_song ?= { is_alive = yes }
	character:cao_cao ?= {
		is_alive = yes
		is_landed = yes
		hd_court_independent_or_regent_trigger = yes
		hd184r_cao_yanzhou_secured_trigger = yes
	}
	character:tao_qian ?= { is_alive = yes is_landed = yes }
	title:k_uuii_xuzhou = { any_de_jure_county = { hd184r_cao_tao_border_county_trigger = yes } }
}

# D44-1 曹嵩遇害作罢：陶谦已让徐州、三人有一不在，或到 200 年仍未触发。Scope：任意。
hd184r_caosong_abort_trigger = {
	OR = {
		exists = global_var:hd_lb_bequest_resolved
		NOT = { character:cao_song ?= { is_alive = yes } }
		NOT = { character:cao_cao ?= { is_alive = yes } }
		NOT = { character:tao_qian ?= { is_alive = yes is_landed = yes } }
		current_date >= 200.1.1
	}
}

# D44-2 袁术可称帝：独立有地、未称帝，且据有寿春一带（寿春、九江、淮南）或领县 >= 30。Scope：袁术。
hd184r_yuanshu_can_proclaim_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	highest_held_title_tier < tier_empire
	NOT = { has_character_modifier = hd_court_pseudo_emperor_modifier }
	OR = {
		any_realm_county = {
			OR = {
				duchy = title:d_anfeng
				duchy = title:d_guangling_yangzhou
				duchy = title:d_huainan
			}
		}
		realm_size >= 30
	}
}
