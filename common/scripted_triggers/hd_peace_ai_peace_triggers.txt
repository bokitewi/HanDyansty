# Clause availability for AI-authored proposals.

hd_peace_ai_can_offer_county_trigger = {
	realm_size > 1
	has_variable_list = hd_peace_offer_candidates
}

hd_peace_ai_can_demand_county_trigger = {
	has_variable = hd_peace_treaty_opponent
	has_variable_list = hd_peace_demand_candidates
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = { realm_size > 1 }
	}
}

hd_peace_ai_can_offer_tribute_trigger = {
	NOT = { has_variable = hd_peace_tribute_receiver }
}

hd_peace_ai_can_demand_tribute_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			NOT = { has_variable = hd_peace_tribute_receiver }
		}
	}
}

hd_peace_ai_can_offer_herd_trigger = {
	hd_peace_can_offer_herd_trigger = yes
}

hd_peace_ai_can_demand_herd_trigger = {
	hd_peace_can_demand_herd_trigger = yes
}
hd_peace_ai_can_offer_claims_trigger = {
	has_variable = hd_peace_treaty_opponent
	any_claim = {
		explicit = yes
		exists = holder
		holder = { OR = { this = root.var:hd_peace_treaty_opponent is_vassal_or_below_of = root.var:hd_peace_treaty_opponent } }
	}
}

hd_peace_ai_can_demand_claims_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			any_claim = {
				explicit = yes
				exists = holder
				holder = { OR = { this = root is_vassal_or_below_of = root } }
			}
		}
	}
}

hd_peace_ai_can_offer_hostage_trigger = {
	any_close_family_member = {
		bp2_valid_to_offer_as_hostage_trigger = { SENDER = root }
	}
}

hd_peace_ai_can_demand_hostage_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			any_close_family_member = {
				bp2_valid_to_offer_as_hostage_trigger = {
					SENDER = root.var:hd_peace_treaty_opponent
				}
			}
		}
	}
}

hd_peace_ai_can_offer_break_alliances_trigger = {
	any_ally = { always = yes }
}

hd_peace_ai_can_demand_break_alliances_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			any_ally = { always = yes }
		}
	}
}

hd_peace_ai_can_offer_prisoners_trigger = {
	has_variable = hd_peace_treaty_opponent
	any_prisoner = {
		is_close_family_of = root.var:hd_peace_treaty_opponent
	}
}

hd_peace_ai_can_demand_prisoners_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			any_prisoner = { is_close_family_of = root }
		}
	}
}

hd_peace_ai_can_offer_conversion_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			NOT = { faith = root.faith }
		}
	}
}

hd_peace_ai_can_demand_conversion_trigger = {
	hd_peace_ai_can_offer_conversion_trigger = yes
}
