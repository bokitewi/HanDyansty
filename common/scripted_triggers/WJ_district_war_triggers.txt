# Native aid joins the beneficiary's eligible leader wars, never a saved selection.
# Character scope; used for both offensive and defensive wars, as vanilla.
WJ_dd_aid_wars_trigger = {
 save_temporary_scope_as = wj_dd_checked_beneficiary
 is_alive = yes
 is_at_war = yes
 is_at_war_with_liege = no
 NOT = { any_war_enemy = { top_liege = scope:wj_dd_checked_beneficiary.top_liege } }
 any_character_war = { scope:wj_dd_checked_beneficiary = { is_leader_in_war = prev } }
}
# Character scope, explicit beneficiary. Mirrors vanilla conflict checks.
WJ_dd_aid_joiner_trigger = {
 is_alive = yes
 is_imprisoned = no
 NOT = { has_trait = incapable }
 top_liege = $BENEFICIARY$.top_liege
 save_temporary_scope_as = wj_dd_checked_joiner
 NOR = {
  this = $BENEFICIARY$
  any_war_ally = { this = $BENEFICIARY$ }
  is_at_war_with = $BENEFICIARY$
  $BENEFICIARY$ = {
   any_war_enemy = {
    OR = {
     is_vassal_of = scope:wj_dd_checked_joiner
     is_liege_or_above_of = scope:wj_dd_checked_joiner
     is_allied_to = scope:wj_dd_checked_joiner
     is_tributary_of = scope:wj_dd_checked_joiner
     is_at_war_with = scope:wj_dd_checked_joiner
    }
   }
  }
 }
}
WJ_dd_aid_officer_trigger = {
 WJ_contract_sanjie_visible_trigger = yes
 primary_title = { tier >= tier_duchy tier <= tier_kingdom }
 WJ_dd_member_of_core_trigger = { CORE = $HEAD$.primary_title }
 this != $HEAD$
 NOT = { is_at_war_with = $HEAD$ }
 WJ_dd_aid_joiner_trigger = { BENEFICIARY = $BENEFICIARY$ }
}
WJ_dd_aid_pair_trigger = {
 $HEAD$ = { WJ_dd_head_trigger = yes is_imprisoned = no NOT = { has_trait = incapable } NOT = { primary_title = { has_variable = wj_dd_grant_pending } } }
 $BENEFICIARY$ = {
  WJ_dd_member_of_core_trigger = { CORE = $HEAD$.primary_title }
  top_liege = $HEAD$.top_liege
  WJ_dd_aid_wars_trigger = yes
 }
}
WJ_dd_request_available_trigger = {
 WJ_dd_aid_pair_trigger = { HEAD = scope:recipient BENEFICIARY = scope:actor }
 OR = {
  scope:recipient = { WJ_dd_aid_joiner_trigger = { BENEFICIARY = scope:actor } }
  scope:recipient.top_liege = {
   any_vassal_or_below = { WJ_dd_aid_officer_trigger = { HEAD = scope:recipient BENEFICIARY = scope:actor } }
  }
 }
}
WJ_dd_dispatch_available_trigger = {
 WJ_dd_aid_pair_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient }
 scope:actor.top_liege = {
  any_vassal_or_below = { WJ_dd_aid_officer_trigger = { HEAD = scope:actor BENEFICIARY = scope:recipient } }
 }
}
# The original central selector uses this in addition to its native candidate check.
WJ_dd_central_candidate_trigger = {
 trigger_if = {
  limit = { $BENEFICIARY$ = { WJ_dd_member_trigger = yes NOT = { WJ_dd_head_trigger = yes } } }
  WJ_dd_head_trigger = yes
  $BENEFICIARY$ = { WJ_dd_member_of_core_trigger = { CORE = prev.primary_title } }
 }
}
