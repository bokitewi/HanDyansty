# 218 年开局接上 253「姜维北伐」心跳（三国剧本事件衔接，2026-10-18）。文件名排在 zz_hd_253_triggers.txt 之前，那里要用这里的触发器。
# 计划 /mnt/project-files/三国剧本事件衔接/218接253_计划.md；on_action 见 zz_hd_253_relay.txt。

# 218 局已到 253.3.1，接上 253 心跳
hd253_relay_trigger = {
	exists = global_var:hd253_relay_on
}

# 253 事件链开着：253 开局，或 218 局接力
hd253_chain_on_trigger = {
	OR = {
		game_start_date = 253.3.1
		hd253_relay_trigger = yes
	}
}

# 当前 scope 的人在魏天子境内：天子本人、天子治下的统治者，或雇主是其中之一
hd253_relay_in_wei_trigger = {
	OR = {
		this = title:h_china.holder
		top_liege ?= title:h_china.holder
		employer ?= {
			OR = {
				this = title:h_china.holder
				top_liege ?= title:h_china.holder
			}
		}
	}
}

# 接力局里可以被拉进天子宫中的人：在世、在魏境内；无地的须雇主是电脑（不从玩家宫中拉人）
hd253_relay_pullable_trigger = {
	is_alive = yes
	hd253_relay_in_wei_trigger = yes
	OR = {
		is_landed = yes
		employer ?= { is_ai = yes }
	}
}

# 伐蜀的对象：巴蜀帝国持有者以它为主头衔，不兼天子、不兼吴（任意 scope）
hd253_relay_shu_ok_trigger = {
	title:e_bashu.holder ?= {
		primary_title = title:e_bashu
		NOT = { has_title = title:h_china }
		NOT = { has_title = title:e_jiangdong }
		NOT = { this = title:h_china.holder }
	}
}

# 孙峻诛诸葛恪的局面：诸葛恪是吴帝治下的首辅或议政直属封臣，孙峻在吴帝治下（任意 scope）
hd253_relay_zhuge_ke_pos_trigger = {
	exists = title:e_jiangdong.holder
	character:zhu_ge_ke ?= {
		is_independent_ruler = no
		top_liege ?= title:e_jiangdong.holder
	}
	OR = {
		title:e_jiangdong.holder = { diarch ?= character:zhu_ge_ke }
		character:zhu_ge_ke ?= {
			liege ?= title:e_jiangdong.holder
			is_councillor_of = title:e_jiangdong.holder
		}
	}
	global_var:hd_sg_sun_jun ?= {
		is_independent_ruler = no
		OR = {
			top_liege ?= title:e_jiangdong.holder
			employer ?= {
				OR = {
					this = title:e_jiangdong.holder
					top_liege ?= title:e_jiangdong.holder
				}
			}
		}
	}
}
