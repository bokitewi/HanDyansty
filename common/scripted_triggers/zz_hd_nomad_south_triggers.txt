# Triggers for the nomad southern-invasion boost (see common/on_action/zz_hd_nomad_south_invasion.txt).

# Global scope or any scope: the dynastic cycle is currently in the chaos / standoff phase.
hd_dynastic_cycle_is_luanshi_trigger = {
	situation:dynastic_cycle ?= {
		OR = {
			situation_current_phase = situation_dynastic_cycle_phase_chaos
			situation_current_phase = situation_dynastic_cycle_phase_hd_standoff
		}
	}
}

# Character scope: an AI nomad ruler allowed to be rolled for a southern invasion.
hd_nomad_south_attacker_trigger = {
	is_ai = yes
	is_landed = yes
	is_independent_ruler = yes
	is_at_war = no
	highest_held_title_tier >= tier_duchy # county-level nomads never take part
	government_has_flag = government_is_nomadic
	NOT = { has_realm_law = uncrowned }
	OR = {
		has_realm_law = nomadic_authority_2
		has_realm_law = nomadic_authority_3
		has_realm_law = nomadic_authority_4
		has_realm_law = nomadic_authority_5
	}
}

# Character scope (defender candidate), ROOT/prev = the nomad: independent settled Han-realm ruler it may invade.
hd_nomad_south_defender_trigger = {
	is_independent_ruler = yes
	is_landed = yes
	NOT = { government_has_flag = government_is_nomadic }
	exists = capital_county
	capital_province = { geographical_region = world_asia_china }
	in_diplomatic_range = $NOMAD$
	# Strength gate: defender may be at most 1.5x the nomad (defender*2 <= nomad*3).
	hd_strength_x2 <= $NOMAD$.hd_strength_x3
	NOR = {
		is_allied_to = $NOMAD$
		is_tributary_of = $NOMAD$
		has_truce = $NOMAD$
		has_any_good_relationship_with_character_trigger = { CHARACTER = $NOMAD$ }
		$NOMAD$ = { is_tributary_of_suzerain_or_above = prev }
	}
}

# County scope: a county bordering the nomad's realm whose top liege is a valid Han-realm defender.
hd_nomad_south_border_county_trigger = {
	holder ?= {
		top_liege = {
			hd_nomad_south_defender_trigger = { NOMAD = $NOMAD$ }
		}
	}
}
