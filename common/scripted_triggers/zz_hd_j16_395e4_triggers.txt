# 两晋十六国统一史实线 · E4 补遗（386–394 北方诸国互动，06_395-421.md §1.8）触发器。写入者：第 6 期 6-E4（本期五路分文件，汇总人 10-27）。
# 本路名字一律带 hd_j16_e4b_ 前缀：06 §0.2 列的 hd_j16_get_<线>_ruler_effect 留给 6-N1/6-N2，本路不定义，免同名冲突。
# 只在本节窗口内的节点里调用（清单：时代段外不做 neighboring）。

############################################################
# 某王朝的独立君主。Scope：角色。参数 DYN：王朝键（murong、tuoba、yao、qifu、lueyang_lv_dyn）
############################################################
hd_j16_e4b_dyn_ruler_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	dynasty ?= dynasty:$DYN$
}

############################################################
# 秦余部（苻氏）的独立君主：苻坚一系 fu_1 或临渭苻氏 linwei_fu_dyn（06 §0.3）。Scope：角色
############################################################
hd_j16_e4b_fu_ruler_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	OR = {
		dynasty ?= dynasty:fu_1
		dynasty ?= dynasty:linwei_fu_dyn
	}
}

############################################################
# 当前 scope 不是 $X$（$X$ 是可能未存的 saved scope；未存时视为「不是」）。Scope：角色
############################################################
hd_j16_e4b_not_trigger = {
	OR = {
		NOT = { exists = $X$ }
		NOT = { this = $X$ }
	}
}

############################################################
# 某人属 $R$ 一系：本人、top_liege 是 $R$，或其雇主属 $R$ 一系（廷臣）。Scope：角色
############################################################
hd_j16_e4b_in_realm_trigger = {
	OR = {
		this = $R$
		top_liege ?= $R$
		employer ?= {
			OR = {
				this = $R$
				top_liege ?= $R$
			}
		}
	}
}

############################################################
# $A$ 一系的国土与 $B$ 一系的国土接壤。Scope 无要求
############################################################
hd_j16_e4b_borders_trigger = {
	$A$ = {
		any_sub_realm_county = {
			any_neighboring_county = {
				holder ?= { target_is_same_character_or_above = $B$ }
			}
		}
	}
}

############################################################
# E4-M55 灭西燕可以开战。读调用方已存的 scope:hd_j16_e4b_yan（燕主）、hd_j16_e4b_xy（西燕之主）
############################################################
hd_j16_e4b_xiyan_ready_trigger = {
	exists = scope:hd_j16_e4b_yan
	exists = scope:hd_j16_e4b_xy
	scope:hd_j16_e4b_yan = {
		is_imprisoned = no
		NOT = { this = scope:hd_j16_e4b_xy }
		NOT = { is_at_war_with = scope:hd_j16_e4b_xy }
		NOT = { any_character_war = { var:hd_nbc_war_kind ?= flag:xiyan } }
	}
	hd_j16_e4b_borders_trigger = { A = scope:hd_j16_e4b_xy B = scope:hd_j16_e4b_yan }
}

############################################################
# E4-T40 破刘卫辰可以开战。读 scope:hd_j16_e4b_wei（魏主）、hd_j16_e4b_lwc（刘卫辰）
# 卫辰须独立（自拟：打封臣会把其宗主拖进战争），其一系据有朔方、拓跋或铁弗郡的首府县
############################################################
hd_j16_e4b_tiefu_ready_trigger = {
	exists = scope:hd_j16_e4b_wei
	scope:hd_j16_e4b_wei = {
		is_imprisoned = no
		NOT = { any_character_war = { var:hd_nbc_war_kind ?= flag:tiefu } }
	}
	scope:hd_j16_e4b_lwc ?= {
		is_landed = yes
		is_independent_ruler = yes
		is_imprisoned = no
		NOT = { this = scope:hd_j16_e4b_wei }
		NOT = { is_at_war_with = scope:hd_j16_e4b_wei }
	}
	OR = {
		title:d_shuofang.title_capital_county.holder ?= { target_is_same_character_or_above = scope:hd_j16_e4b_lwc }
		title:d_tuoba.title_capital_county.holder ?= { target_is_same_character_or_above = scope:hd_j16_e4b_lwc }
		title:d_tiefu.title_capital_county.holder ?= { target_is_same_character_or_above = scope:hd_j16_e4b_lwc }
	}
	hd_j16_e4b_borders_trigger = { A = scope:hd_j16_e4b_lwc B = scope:hd_j16_e4b_wei }
}
