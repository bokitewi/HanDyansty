# Actual territorial titles, not titular ministries or noble-family names.
WJ_cr_real_realm = {
 is_alive = yes government_has_flag = government_is_celestial is_landed = yes
 primary_title = {
  tier >= tier_kingdom is_noble_family_title = no tgp_is_minister_title = no
  NOT = { has_variable = SanGong_DongGong_title }
  any_de_jure_county = { always = yes }
 }
}
WJ_cr_king = { WJ_cr_real_realm = yes is_independent_ruler = yes primary_title = { tier = tier_kingdom } }
WJ_cr_emperor = { WJ_cr_real_realm = yes is_independent_ruler = yes primary_title = { tier >= tier_empire } }
WJ_cr_can_plan = {
 has_ach_dlc_trigger = yes NOT = { has_game_rule = coronation_laws_off }
 is_ai = no
 OR = { WJ_cr_king = yes WJ_cr_emperor = yes }
 NOR = { has_realm_law = crowned_emperor AND = { WJ_cr_king = yes has_realm_law = crowned_king } }
}
# The law engine also calls this predicate: excluding Chinese rulers here would clear their laws.
# Non-Chinese branches retain the installed game's complete original qualification.
coronation_trigger = {
 has_ach_dlc_trigger = yes is_landed = yes highest_held_title_tier >= tier_kingdom
 trigger_if = {
  limit = { government_has_flag = government_is_celestial }
  OR = { WJ_cr_king = yes WJ_cr_emperor = yes }
 }
 trigger_else = {
  trigger_if = { limit = { tgp_realm_has_ceremonial_liege_trigger = yes }
   custom_tooltip = { text = tgp_ceremonial_liege_coronation_trigger tgp_has_ceremonial_liege_title_trigger = yes }
  }
  trigger_if = { limit = { government_has_mechanic = administrative } is_independent_ruler = yes }
  trigger_else = { NOR = {
   government_has_flag = government_is_theocracy government_has_flag = government_is_republic
   government_has_flag = government_is_mercenary government_has_flag = government_is_holy_order
  } }
 }
}
