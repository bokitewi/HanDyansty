# Broad dates are opportunity windows only. Every trigger also requires current world facts.

hd_fengying_world_ready_trigger = {
	NOT = { exists = global_var:hd_fengying_world_open }
	NOT = { exists = global_var:hd_fengying_world_claimed }
	NOT = { exists = global_var:hd_fengying_world_resolved }
	current_date >= 195.1.1
	current_date <= 197.12.31
	character:liu_xie = { is_alive = yes }
	OR = {
		title:h_china.holder = character:liu_xie
		title:e_han.holder = character:liu_xie
	}
	OR = {
		character:cao_cao = { is_alive = yes is_landed = yes is_independent_ruler = yes }
		character:yuan_shao = { is_alive = yes is_landed = yes is_independent_ruler = yes }
	}
}

hd_guandu_world_ready_trigger = {
	NOT = { exists = global_var:hd_guandu_world_open }
	NOT = { exists = global_var:hd_guandu_world_active }
	NOT = { exists = global_var:hd_guandu_world_resolved }
	current_date >= 198.1.1
	current_date <= 203.12.31
	character:cao_cao = {
		is_alive = yes
		is_landed = yes
		is_independent_ruler = yes
		realm_size >= 10
		NOT = { is_allied_to = character:yuan_shao }
	}
	character:yuan_shao = {
		is_alive = yes
		is_landed = yes
		is_independent_ruler = yes
		realm_size >= 10
		NOT = { is_allied_to = character:cao_cao }
	}
	NOT = { character:cao_cao.top_liege = character:yuan_shao.top_liege }
	OR = {
		title:k_uuii_jizhou.holder.top_liege = character:yuan_shao
		character:yuan_shao = { realm_size >= 25 }
	}
	OR = {
		title:k_uuii_yuzhou.holder.top_liege = character:cao_cao
		character:cao_cao = { realm_size >= 25 }
	}
}

hd_chibi_world_ready_trigger = {
	NOT = { exists = global_var:hd_chibi_world_open }
	NOT = { exists = global_var:hd_chibi_world_active }
	NOT = { exists = global_var:hd_chibi_world_resolved }
	current_date >= 207.1.1
	current_date <= 211.12.31
	character:cao_cao = { is_alive = yes is_landed = yes is_independent_ruler = yes realm_size >= 25 }
	character:sun_quan = { is_alive = yes is_landed = yes is_independent_ruler = yes realm_size >= 8 }
	character:liu_bei = { is_alive = yes is_landed = yes realm_size >= 1 }
	OR = {
		title:k_uuii_yangzhou.holder.top_liege = character:sun_quan
		character:sun_quan = { realm_size >= 18 }
	}
	OR = {
		exists = global_var:hd_chibi_sun_liu_alliance
		character:sun_quan = { is_allied_to = character:liu_bei }
	}
	OR = {
		title:k_uuii_jingzhou.holder = { is_alive = yes }
		title:d_jiangxia.holder = { is_alive = yes }
	}
}
