	}
}

# 奉迎宣战对象：挟持天子之人（通用权臣优先，其次剧本摄政），无人挟持则为天子本人。Scope：被检查的角色
hd_cao_fengying_target_trigger = {
	trigger_if = {
		limit = { hd_court_has_regent_trigger = yes }
		this = title:h_china.var:hd_court_regent
	}
	trigger_else_if = {
		limit = { hd_dong_historical_regent_active_trigger = yes }
		this = title:h_china.holder.var:hd_dong_fixed_regent
	}
	trigger_else = {
		has_title = title:h_china
	}
}

	is_alive = yes
	NOT = { this = character:cao_cao }
	NOT = { character:cao_cao = { has_truce = prev } }
	NOT = { is_allied_to = character:cao_cao }
}

hd_cao_fengying_can_declare_trigger = {
	hd_cao_fengying_actor_valid_trigger = yes
	has_character_flag = hd_cao_fengying_authorized
	NOT = { exists = global_var:hd_fengying_world_resolved }
	hd_dong_transition_pending_trigger = no
	is_at_war = no
	is_imprisoned = no
	trigger_if = {
		limit = { hd_court_has_regent_trigger = yes }
		title:h_china.var:hd_court_regent = { hd_cao_fengying_target_ok_trigger = yes }
		can_declare_war = { defender = title:h_china.var:hd_court_regent casus_belli = hd_cao_fengying_cb }
	}
	trigger_else_if = {
		limit = { hd_dong_historical_regent_active_trigger = yes }
		title:h_china.holder.var:hd_dong_fixed_regent = { hd_cao_fengying_target_ok_trigger = yes }
		can_declare_war = { defender = title:h_china.holder.var:hd_dong_fixed_regent casus_belli = hd_cao_fengying_cb }
	}
	trigger_else = {
		exists = title:h_china.holder
		title:h_china.holder = { hd_cao_fengying_target_ok_trigger = yes }
		can_declare_war = { defender = title:h_china.holder casus_belli = hd_cao_fengying_cb }
	}
}
