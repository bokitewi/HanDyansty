# Character scope: Cao Cao. These conditions also apply after the invitation.
hd_cao_fengying_actor_valid_trigger = {
	this = character:cao_cao
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	character:liu_xie = { is_alive = yes }
	title:h_china.holder = character:liu_xie
}

hd_cao_fengying_invitation_trigger = {
	hd_cao_fengying_actor_valid_trigger = yes
	current_date > 196.1.1
	current_military_strength > 10000
	hd_cao_controls_xuchang_trigger = yes
	hd_dong_transition_pending_trigger = no
}

	NOT = { exists = global_var:hd_fengying_world_resolved }
	hd_dong_transition_pending_trigger = no
	is_at_war = no
	is_imprisoned = no
	exists = title:d_jingzhao.holder
	title:d_jingzhao.holder = { is_alive = yes NOT = { this = character:cao_cao } }
	NOT = { has_truce = title:d_jingzhao.holder }
	NOT = { is_allied_to = title:d_jingzhao.holder }
	can_declare_war = { defender = title:d_jingzhao.holder casus_belli = hd_cao_fengying_cb }
}
