# 218 剧本历史大事件用的判断（第 4 步）。

# "在战中"：本人在 $SIDE$ 一方的军中——是 $SIDE$ 本人、其廷臣或封臣，且此刻作为统帅或骑士在军中。Scope：人物
# 封臣的廷臣进不了宗主的军队，所以庞德、于禁等的雇主必须是交战君主本人（第 1 步雇主表已照此写）。
hd_sg218_in_army_for_trigger = {
	is_alive = yes
	is_in_army = yes
	OR = {
		this = $SIDE$
		employer ?= $SIDE$
		liege ?= $SIDE$
	}
}

# "在蜀"：本人位于益州州内任一县。Scope：人物
hd_sg218_in_shu_trigger = {
	location ?= {
		county ?= {
			kingdom ?= title:k_uuii_yizhou
		}
	}
}

# 某场 218 战争正在进行（开战时设、结局时删的全局标记）。无作用域要求
hd_sg218_war_active_trigger = {
	exists = global_var:hd_sg218_$KIND$_active
}

# 诸葛亮受遗诏辅政（白帝托孤事件设的旗标，不依赖首辅二元统治是否建成）
hd_sg218_zgl_regent_trigger = {
	character:zhu_ge_liang ?= {
		is_alive = yes
		has_character_flag = hd_sg218_tuogu
	}
}

# 南中叛乱仍在：雍闿、高定、朱褒有人仍是独立统治者
hd_sg218_nanzhong_rebels_active_trigger = {
	OR = {
		character:yong_kai ?= { is_alive = yes is_independent_ruler = yes }
		character:gao_ding ?= { is_alive = yes is_independent_ruler = yes }
		character:zhu_bao ?= { is_alive = yes is_independent_ruler = yes }
	}
}

# 刘备一系的继承人：汉中王号（称帝后为"汉"）的现任持有者
hd_sg218_shu_ruler_trigger = {
	global_var:hd218_hanzhong_title.holder ?= this
}

# 头衔下有县归 $DEF$ 一方。Scope：头衔
hd_sg218_title_held_by_trigger = {
	any_in_de_jure_hierarchy = {
		tier = tier_county
		holder ?= {
			OR = {
				this = $DEF$
				target_is_liege_or_above = $DEF$
			}
		}
	}
}

# 218 剧本随机郡守入选：$RULER$ 直持的郡，首都所在郡、颍川（许昌，复核建议 2 留给曹操）、郡内没有他直持县的郡（如洛阳）除外。Scope：头衔
hd218_governor_duchy_trigger = {
	tier = tier_duchy
	is_noble_family_title = no
	is_landless_type_title = no
	NOT = { this = title:d_yinchuan }
	NOT = { $RULER$.capital_county.duchy ?= this }
	any_in_de_jure_hierarchy = {
		tier = tier_county
		holder ?= $RULER$
	}
}

# 218 剧本空郡补郡守：首府县在 $DUCHY$ 郡、只持县级、独立的地方首领
hd218_empty_duchy_chief_trigger = {
	is_independent_ruler = yes
	highest_held_title_tier = tier_county
	capital_county ?= { duchy ?= title:$DUCHY$ }
}

# 218 剧本独立郡自检：持有者独立、天朝政体，郡内没有曹操直持的县
hd218_ind_duchy_ok_trigger = {
	title:$DUCHY$ = {
		holder ?= {
			is_independent_ruler = yes
			has_government = celestial_government
		}
		NOT = {
			any_in_de_jure_hierarchy = {
				tier = tier_county
				holder ?= character:cao_cao
			}
		}
	}
}
