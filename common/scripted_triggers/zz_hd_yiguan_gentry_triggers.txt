# 任意作用域。全局年度计数未满：$NAME$ 计数名，$MAX$ 上限。存的年份不是今年即视为已清零
# 计数写法见 hd_yiguan_gt_bump_counter_effect；年份比较先例 zz_hd_confederation_on_actions.txt:16
hd_yiguan_gt_counter_below_trigger = {
	OR = {
		NOT = { has_global_variable = hd_yiguan_gt_$NAME$_year }
		NOT = { has_global_variable = hd_yiguan_gt_$NAME$_count }
		NOT = { global_var:hd_yiguan_gt_$NAME$_year = current_year }
		global_var:hd_yiguan_gt_$NAME$_count < $MAX$
	}
}

# 角色（候选目的地）。$MIGRANT$ 为迁徙者。独立的南方政权或北方汉人；玩家只能投天朝制（§6 第 17 条）
hd_yiguan_gt_nanqian_dest_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	NOT = { this = $MIGRANT$ }
	hd_yiguan_gt_han_group_trigger = yes
	NOT = { is_at_war_with = $MIGRANT$ }
	trigger_if = {
		limit = { $MIGRANT$ = { is_ai = no } }
		custom_tooltip = {
			text = hd_yiguan_gt_nanqian_dest_celestial_tt
			government_has_flag = government_is_celestial
		}
		# 改投须主君高一级：持公国级世族（如行政制下）者只能投王国级以上
		trigger_if = {
			limit = {
				$MIGRANT$ = {
					any_held_title = {
						is_noble_family_title = yes
						tier >= tier_duchy
					}
				}
			}
			custom_tooltip = {
				text = hd_yiguan_gt_nanqian_dest_tier_tt
				any_held_title = {
					tier >= tier_kingdom
					is_landless_type_title = no
				}
			}
		}
	}
}

# 任意作用域。$REGION$ 州内有县归属可作守方的胡人政权
# 北方八州一律按汉代原州的静态地理区域（uuii_han_*_region）判定，不看 k_uuii_* 的运行时法理：
# 南北朝剧本 280.1.1 起（zz_hd_nbc_dejure.txt）河南、洛阳、河内等郡改挂洛州、司州等新州，而地理区域不随法理改挂变化
# （原版 map_data/geographical_regions/_geographical_regions.info）
hd_yiguan_gt_zhou_has_hu_trigger = {
	any_county_in_region = {	# 写法同原版 03_dlc_fp2_script_values.txt:710
		region = $REGION$
		holder ?= {
			top_liege = { hd_yiguan_gt_bh_def_ok_trigger = yes }
		}
	}
}

# 任意。$A$、$B$ 两家各有一名可成婚的族人，且一男一女（粗查：各家逐一过一遍，不做两家人两两配对）
# 精确配对（血亲、信仰等，原版 can_marry_character_trigger）只在 hd_yiguan_gt_qiaowu_marry_effect 里做一次；配不成对时效果里给玩家提示
# 互动 is_valid 会被 AI 定期求值、玩家右键时也求值，大族上百人时两两配对太重，故此处只粗查
hd_yiguan_gt_qiaowu_pair_exists_trigger = {
	OR = {
		AND = {
			$A$.house ?= {
				any_house_member = {
					is_male = yes
					hd_yiguan_gt_marriage_candidate_trigger = yes
				}
			}
			$B$.house ?= {
				any_house_member = {
					is_female = yes
					hd_yiguan_gt_marriage_candidate_trigger = yes
				}
			}
		}
		AND = {
			$A$.house ?= {
				any_house_member = {
					is_female = yes
					hd_yiguan_gt_marriage_candidate_trigger = yes
				}
			}
			$B$.house ?= {
				any_house_member = {
					is_male = yes
					hd_yiguan_gt_marriage_candidate_trigger = yes
				}
			}
		}
	}
}
