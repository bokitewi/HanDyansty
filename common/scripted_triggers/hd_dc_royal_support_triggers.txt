# Recipient character scope; scope:hd_dc_support is the exact invitation record.
# Acceptance is recorded by the recipient event, never by the joining executor.
hd_dc_royal_support_identity_trigger = {
	is_alive = yes
	exists = title:h_china.holder
	exists = scope:hd_dc_support
	scope:hd_dc_support = {
		var:hd_dc_record_kind = flag:royal_support
		var:hd_dc_support_recipient = scope:hd_dc_support_actor
		var:hd_dc_support_ruler = title:h_china.holder
		exists = var:hd_dc_support_claimant
		var:hd_dc_support_claimant = { is_alive = yes has_claim_on = title:h_china }
		exists = var:hd_dc_support_faction
		var:hd_dc_support_faction = {
			faction_is_type = claimant_faction
			special_character = scope:hd_dc_support.var:hd_dc_support_claimant
			special_title = title:h_china
			faction_target = title:h_china.holder
			NOT = { exists = faction_war }
		}
		exists = var:hd_dc_actor_story
		var:hd_dc_actor_story = {
			var:hd_dc_actor = scope:hd_dc_support.var:hd_dc_support_claimant
			var:hd_dc_claimant_faction = scope:hd_dc_support.var:hd_dc_support_faction
			var:hd_dc_instance_ref = scope:hd_dc_support.var:hd_dc_instance_ref
			var:hd_dc_node = scope:hd_dc_support.var:hd_dc_origin_node
			var:hd_dc_node_version = scope:hd_dc_support.var:hd_dc_origin_version
		}
		var:hd_dc_instance_ref = {
			var:hd_dc_type = flag:roy
			OR = { var:hd_dc_stage = flag:acute AND = { var:hd_dc_stage = flag:aftermath has_variable = hd_dc_roy01_node_closed NOT = { has_variable = hd_dc_political_result } } }
			NOT = { has_variable = hd_dc_reward_committed }
		}
	}
}

hd_dc_royal_support_join_authorized_trigger = {
	hd_dc_royal_support_identity_trigger = yes
	NOT = { scope:hd_dc_support.var:hd_dc_support_faction = { hd_dc_claimant_mobilization_paused_trigger = yes } }
	is_imprisoned = no
	NOT = { has_trait = incapable }
	NOT = { is_at_war_with = title:h_china.holder }
	NOT = { is_at_war_with = scope:hd_dc_support.var:hd_dc_support_claimant }
	this != scope:hd_dc_support.var:hd_dc_support_claimant
	this != title:h_china.holder
	scope:hd_dc_support = {
		var:hd_dc_support_answer = flag:accepted
		var:hd_dc_support_signature = scope:hd_dc_support_actor
		has_variable = hd_dc_support_wait
		NOT = { has_variable = hd_dc_support_commit }
	}
	NOT = { joined_faction = scope:hd_dc_support.var:hd_dc_support_faction }
	can_join_faction = scope:hd_dc_support.var:hd_dc_support_faction
	NOT = { has_variable = hd_dc_royal_support_commit_lock }
}
