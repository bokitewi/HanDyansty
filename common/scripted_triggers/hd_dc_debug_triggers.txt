# Choose a real claimant who can make this player a real support offer.
hd_dc_debug_epidemic_county_trigger = {
	title_tier = county
	exists = holder
	holder = { is_alive = yes is_imprisoned = no is_vassal_or_below_of = title:h_china.holder }
	NOT = { has_province_with_epidemic = { intensity = any } }
	NOT = { has_variable = hd_dc_policy_trial }
	NOT = { has_county_modifier = ep3_governance_tax_relief }
	NOT = { has_county_modifier = hd_dc_par_tax_relief_modifier }
	NOT = { has_game_rule = epidemic_frequency_disabled }
}

# Province scope. Read the native zero/nonzero contract, not a second RNG pool.
hd_dc_debug_epidemic_intensity_trigger = {
	exists = scope:epidemic_type
	save_temporary_scope_value_as = {
		name = hd_dc_debug_epidemic_base
		value = {
			value = outbreak_chance_$INTENSITY$_default_value
			if = {
				limit = { scope:epidemic_type = epidemic_type:consumption has_variable = consumption_endemic_variable }
				if = { limit = { scope:hd_dc_debug_epidemic_intensity = flag:hd_dc_debug.epidemic.minor } add = outbreak_increase_add_high }
				else = { add = outbreak_increase_add_low }
			}
			if = {
				limit = { scope:epidemic_type = epidemic_type:measles has_variable = measles_endemic_variable }
				if = { limit = { scope:hd_dc_debug_epidemic_intensity = flag:hd_dc_debug.epidemic.minor } add = outbreak_increase_add_high }
				else_if = { limit = { scope:hd_dc_debug_epidemic_intensity = flag:hd_dc_debug.epidemic.major } add = outbreak_increase_add_medium }
				else = { add = outbreak_increase_add_low }
			}
			if = { limit = { scope:epidemic_type = epidemic_type:dysentery } value = outbreak_chance_dysentery_default_value }
			if = { limit = { scope:epidemic_type = epidemic_type:bubonic_plague scope:hd_dc_debug_epidemic_intensity = flag:hd_dc_debug.epidemic.apocalyptic } value = outbreak_chance_black_plague_default_value multiply = outbreak_chance_black_plague_mult_value }
			else = { multiply = outbreak_chance_$INTENSITY$_mult_value }
			multiply = recent_epidemics_outbreak_mult_value
		}
	}
	scope:hd_dc_debug_epidemic_base > 0
	trigger_if = {
		limit = { OR = { scope:epidemic_type = epidemic_type:measles scope:epidemic_type = epidemic_type:dysentery scope:epidemic_type = epidemic_type:ergotism } }
		has_dlc_feature = legends_of_the_dead
	}
	trigger_if = {
		limit = { scope:epidemic_type = epidemic_type:ergotism }
		NOT = { is_target_in_variable_list = { name = epidemic_cooldown target = epidemic_type:ergotism } }
	}
	trigger_if = {
		limit = { scope:epidemic_type = epidemic_type:bubonic_plague scope:hd_dc_debug_epidemic_intensity = flag:hd_dc_debug.epidemic.apocalyptic }
		NOT = { has_global_variable = black_death }
		NOT = { has_game_rule = bd_frequency_never }
		NOT = { has_game_rule = bd_frequency_once has_global_variable = black_death_happened }
		# The loaded custom-map historical geographic branch is explicitly false.
		NOT = { has_game_rule = bd_occurrence_historical }
	}
}
