# 历史人物崛起：名单、资格、终止条件、开战槽位。均为角色作用域。
hd_rise_roster_trigger = {
	# 22 光武中兴剧本：刘縯、刘秀；刘玄 26 年以前。22 开局时 184 一代人尚未出生、人物不存在，不求值下面的名单
	trigger_if = {
		limit = { game_start_date = 22.11.13 }
		OR = {
			this = character:liu_yan
			this = character:liu_xiu
			AND = { this = character:WJ_history_gengshi current_year < 26 }
		}
	}
	# 89 燕然勒石剧本：不给任何人天意青睐；184 一代人尚未出生，不求值下面的名单（计划复核建议 8）
	trigger_else_if = {
		limit = { game_start_date = 89.1.30 }
		always = no
	}
	trigger_else = { hd_rise_roster_main_trigger = yes }
}
hd_rise_roster_main_trigger = {
	OR = {
		this = character:cao_cao
		this = character:liu_bei
		this = character:yuan_shu
		this = character:yuan_shao
		this = character:liu_biao
		this = character:lv_bu
		this = character:sun_jian
		AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
		this = character:ma_teng
		this = character:han_sui
		this = character:tao_qian
		this = character:liu_zhang_4
		AND = { game_start_date = 304.11.20 this = character:xiongnu_liu_020 } # 304 刘渊剧本：汉王刘渊
		AND = { game_start_date = 531.7.1 this = character:hd_hi_gaohuan } # 531 高欢剧本：渤海王高欢
		AND = { game_start_date = 370.2.13 this = character:lueyang_fu_020 } # 370 苻坚剧本：大秦天王苻坚
		AND = { game_start_date = 617.6.23 this = character:han_268 } # 617 隋末群雄剧本：唐国公李渊
	}
}

# 持有 >= 3 个王国级头衔后永久失去加成（由脉冲打 hd_rise_retired 标记）。2026-10-10：由 2 改为 3。
hd_rise_retired_trigger = {
	any_held_title = {
		count >= 3
		tier = tier_kingdom
		is_noble_family_title = no
		is_landless_type_title = no
	}
}

hd_rise_eligible_trigger = {
	is_alive = yes
	is_ai = yes
	is_landed = yes
	NOT = { has_character_flag = hd_rise_retired }
	# 2026-10-24 boki：190 开局刘备徐州时期（托付徐州后、失徐州前）不享天意青睐，投刘表后恢复
	trigger_if = {
		limit = {
			game_start_date = 190.1.1
			exists = global_var:hd_lb_bequest_completed
			NOT = { exists = global_var:hd_lbxz_seized }
			exists = character:liu_bei
			this = character:liu_bei
		}
		NOT = { has_title = title:k_uuii_xuzhou }
	}
}

# 主动进攻战（作为主攻方）少于 3 场。
hd_rise_has_war_slot_trigger = {
	NOT = {
		any_character_war = {
			count >= 3
			primary_attacker = prev
		}
	}
}

hd_rise_war_ready_trigger = {
	hd_rise_eligible_trigger = yes
	is_independent_ruler = yes
	is_imprisoned = no
	exists = global_var:hd_anti_regent_chain_resolved
	hd_dong_transition_pending_trigger = no
	hd_rise_has_war_slot_trigger = yes
	trigger_if = {
		limit = { exists = global_var:hd_yuan_jizhou_pending }
		NOT = { this = character:yuan_shao }
		NOT = { this = character:gong_sun_zan }
	}
}

# 目标作用域：被选中的对手。root 为进攻者（在 hd_rise.0001 事件内调用）。
hd_rise_target_valid_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	NOT = { character:liu_xie ?= this } # 22 开局时献帝等尚未出生，用 ?= 免报错（其他开局等价）
	trigger_if = {
		limit = {
			NOT = { exists = global_var:hd_guandu_world_open }
			NOT = { exists = global_var:hd_guandu_world_active }
		}
		NOT = { AND = { character:yuan_shao ?= this character:cao_cao ?= root } }
		NOT = { AND = { character:cao_cao ?= this character:yuan_shao ?= root } }
	}
}
