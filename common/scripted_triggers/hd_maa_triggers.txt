# --- 幽州突骑：当前有效兵源 ---
# 统治者合法领域（sub_realm，天然排除临时占领县）内的幽州兵源区（7 郡 47 县）中，
# 控制力 ≥ 50，且（与统治者同文化，或两文化间接受度 > 20）的县 ≥ 24 个。
# 统治者一律通过 prev 引用（不依赖 ROOT 上下文），
# 因此在 can_recruit、决议、on_action 等任意上下文中求值结果一致。
# 接受度从统治者文化一侧求值（prev.culture 域内，prev = 县，target = 县文化）。
hd_valid_youzhou_tuqi_source_trigger = {
	any_sub_realm_county = {
		count >= 24
		title_province = { geographical_region = hd_region_youzhou_tuqi }
		county_control >= 50
		OR = {
			culture = prev.culture
			prev.culture = {
				cultural_acceptance = {
					target = prev.culture
					value > 20
				}
			}
		}
	}
}

# --- 幽州突骑：可招募（已建立军制 且 当前仍有有效兵源）---
# 不含任何日期条件：公元 300 年限制只约束首次建立决议（is_shown），
# 已建立军制者在 300 年后仍可招募、扩编与在复地后恢复。
hd_can_recruit_youzhou_tuqi_trigger = {
	has_character_flag = hd_established_youzhou_tuqi
	hd_valid_youzhou_tuqi_source_trigger = yes
}

# --- 刷新守卫：是否已建立任一地域军制 ---
# 挂在所有刷新类 on_action 的 trigger 上，未建立任何军制者刷新零开销。
hd_has_any_hd_regional_maa_trigger = {
	OR = {

		has_character_flag = hd_established_xiandeng_ruishi
		has_character_flag = hd_established_qingzhou_bing
		has_character_flag = hd_established_liujun_liangjiazi
		has_character_flag = hd_established_huangzhong_yicong
		has_character_flag = hd_established_bandun_congbing
		has_character_flag = hd_established_wudang_feijun
		has_character_flag = hd_established_wuxi_manbing
		has_character_flag = hd_established_shanyue_jiashi

		has_character_flag = hd_established_fuling_liannushi
