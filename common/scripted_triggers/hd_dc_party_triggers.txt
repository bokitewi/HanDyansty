# Native movement groups are the organization authority. Hegemon, unaffiliated
# and other situation groups cannot masquerade as political parties.
hd_dc_par_named_pair_trigger = {
	exists = $FIRST$
	exists = $SECOND$
	$FIRST$ = {
		hd_dc_par_group_qualified_trigger = yes
		save_temporary_scope_as = hd_dc_par_first_group
	}
	$SECOND$ = { hd_dc_par_hostile_pair_trigger = yes }
}

hd_dc_par_native_movement_trigger = {
	OR = {
		participant_group_type = pro_hegemon_movement
		participant_group_type = advancement_movement
		participant_group_type = expansion_movement
		participant_group_type = conservative_movement
	}
}

hd_dc_par_office_member_trigger = {
	is_alive = yes
	OR = { is_landed = yes is_councillor = yes has_any_court_position = yes is_diarch_of_target = title:h_china.holder }
}

# Group scope after one annual snapshot of all four comparable native powers.
hd_dc_par_group_qualified_trigger = {
	hd_dc_par_native_movement_trigger = yes
	var:hd_dc_par_snapshot_year = current_year
	var:hd_dc_par_office_members >= 3
	var:hd_dc_par_power_snapshot > 0
	exists = var:hd_dc_par_leader_snapshot
	var:movement_leader = var:hd_dc_par_leader_snapshot
	var:hd_dc_par_leader_snapshot = {
		save_temporary_scope_as = hd_dc_par_checked_leader
		is_alive = yes
		is_adult = yes is_imprisoned = no
		NOT = { has_trait = incapable }
	}
	any_situation_group_participant = { this = scope:hd_dc_par_checked_leader }
	situation:dynastic_cycle = {
		var:hd_dc_par_power_readable = yes
		var:hd_dc_par_total_power > 0
	}
	# Exact >=20% comparison, with no division or rounded percentage.
	hd_dc_par_power_fivefold_value >= situation:dynastic_cycle.var:hd_dc_par_total_power
}

hd_dc_par_snapshot_valid_trigger = {
	var:hd_dc_par_snapshot_year = current_year
	exists = title:h_china.holder
	var:hd_dc_par_snapshot_ruler = title:h_china.holder
	var:hd_dc_par_power_readable = yes
	var:hd_dc_par_total_power > 0
	var:hd_dc_par_snapshot_status = flag:organizational_pair_found
	# Annual storage is not proof that the same actors remain eligible at use.
	any_participant_group = {
		count = hd_dc_par_snapshot_group_count_value
		hd_dc_par_native_movement_trigger = yes
	}
	NOT = {
		any_participant_group = {
			hd_dc_par_native_movement_trigger = yes
			NOT = { hd_dc_par_group_snapshot_current_trigger = yes }
		}
	}
	any_participant_group = {
		hd_dc_par_group_qualified_trigger = yes
		save_temporary_scope_as = hd_dc_par_first_group
		situation:dynastic_cycle = {
			any_participant_group = { hd_dc_par_hostile_pair_trigger = yes }
		}
	}
}

# Both membership directions are checked; new entrants cannot silently inherit
# a frozen member's identity. At most four native groups, without a world scan.
hd_dc_par_group_snapshot_current_trigger = {
	save_temporary_scope_as = hd_dc_par_live_group
	var:hd_dc_par_snapshot_year = current_year
	has_variable = movement_power
	var:movement_power = var:hd_dc_par_power_snapshot
	OR = {
		AND = { NOT = { has_variable = movement_leader } NOT = { has_variable = hd_dc_par_leader_snapshot } }
		AND = { exists = var:hd_dc_par_leader_snapshot var:movement_leader = var:hd_dc_par_leader_snapshot }
	}
	NOT = {
		any_in_list = {
			variable = hd_dc_par_member_snapshot
			save_temporary_scope_as = hd_dc_par_live_member
			OR = {
				NOT = { hd_dc_par_office_member_trigger = yes }
				NOT = { scope:hd_dc_par_live_group = { participant_group_has_character = scope:hd_dc_par_live_member } }
			}
		}
	}
	NOT = {
		any_situation_group_participant = {
			hd_dc_par_office_member_trigger = yes
			save_temporary_scope_as = hd_dc_par_live_member
			NOT = {
				scope:hd_dc_par_live_group = {
					is_target_in_variable_list = { name = hd_dc_par_member_snapshot target = scope:hd_dc_par_live_member }
				}
			}
		}
	}
}

# Two real organizations, two distinct current leaders. Private rivalry may
# supply hostility only after the independent organizational tests above.
hd_dc_par_hostile_pair_trigger = {
	exists = scope:hd_dc_par_first_group
	this != scope:hd_dc_par_first_group
	hd_dc_par_group_qualified_trigger = yes
	var:hd_dc_par_leader_snapshot != scope:hd_dc_par_first_group.var:hd_dc_par_leader_snapshot
	var:hd_dc_par_leader_snapshot = { save_temporary_scope_as = hd_dc_par_second_leader }
	scope:hd_dc_par_first_group.var:hd_dc_par_leader_snapshot = {
		save_temporary_scope_as = hd_dc_par_first_leader
		OR = {
			has_relation_rival = scope:hd_dc_par_second_leader
			has_relation_nemesis = scope:hd_dc_par_second_leader
			AND = {
				opinion = { target = scope:hd_dc_par_second_leader value <= -30 }
				scope:hd_dc_par_second_leader = { opinion = { target = scope:hd_dc_par_first_leader value <= -30 } }
			}
		}
	}
}
