# changed_obligations entries are native levels passed to the native setter.
# Contract preview must identify the office without relying on prospective flags.
WJ_dd_contract_head_lock_trigger = {
 WJ_contract_privilege_realm_trigger = yes
 primary_title = {
  tier = tier_kingdom
  is_noble_family_title = no
  trigger_if = {
   limit = { exists = var:wj_dd_core exists = var:wj_dd_realm exists = holder }
   var:wj_dd_core = this
   exists = var:wj_dd_realm.holder
   holder.primary_title = this
   holder.top_liege.primary_title = var:wj_dd_realm
  }
  trigger_else = { always = no }
 }
}
# Compare obligation objects, never a numeric index (see event_targets.log).
WJ_dd_contract_change_allowed_trigger = {
 # Only district military privileges are locked; unrelated native contracts remain unchanged.
 NAND = {
  $SUBJECT$ = { WJ_dd_contract_head_lock_trigger = yes WJ_zw_realm_taken_over = yes }
  OR = { vassal_contract_type = vassal_contract:WJ_sanjie vassal_contract_type = vassal_contract:WJ_kaifu }
 }
 NAND = {
  $SUBJECT$ = { WJ_dd_contract_head_lock_trigger = yes }
  vassal_contract_type = vassal_contract:celestial_provinces
  this = vassal_contract:celestial_provinces.obligation:celestial_province_standard
 }
}
