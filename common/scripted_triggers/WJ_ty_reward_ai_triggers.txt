 var:wj_kaoke_last_grade >= 4

 var:wj_kaoke_last_date >= { value = current_date subtract = 365 }
}

}

# Check the already computed attitude against the actual five voting bands.
WJ_ty_reward_ai_can_gain_vote = {
 save_temporary_scope_value_as = {
  name = wj_ty_ai_offer_result
  value = { value = scope:wj_ty_ai_attitude add = $GAIN$ }
 }
 OR = {
  AND = { scope:wj_ty_ai_attitude <= -75 scope:wj_ty_ai_offer_result > -75 }
  AND = { scope:wj_ty_ai_attitude <= -25 scope:wj_ty_ai_offer_result > -25 }
  AND = { scope:wj_ty_ai_attitude < 25 scope:wj_ty_ai_offer_result >= 25 }
  AND = { scope:wj_ty_ai_attitude < 75 scope:wj_ty_ai_offer_result >= 75 }
 }
}
