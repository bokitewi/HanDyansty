# 253「姜维北伐」剧本触发器。计划见 docs/姜维剧本_计划.md。

# 开局运行时生成郡守：君主直持、含其直持县的郡，首都所在郡除外（写法同 hd218_governor_duchy_trigger，不排除颍川）
hd253_governor_duchy_trigger = {
	tier = tier_duchy
	is_noble_family_title = no
	is_landless_type_title = no
	NOT = { $RULER$.capital_county.duchy ?= this }
	any_in_de_jure_hierarchy = {
		tier = tier_county
		holder ?= $RULER$
	}
}

# 北伐中原的攻方：253 开局（或 218 局接上 253 后）、汉中（梁州）都督，最高领主为汉（巴蜀帝国为主头衔）
hd253_beifa_attacker_trigger = {
	hd253_chain_on_trigger = yes
	is_independent_ruler = no
	government_has_flag = government_is_celestial
	OR = {
		has_title = title:k_liaozhou
		has_title = title:d_hanzhong
	}
	top_liege = {
		primary_title = title:e_bashu
		NOT = { has_title = title:h_china }
	}
}

# 北伐中原的守方：魏国境内的封臣（郡级以上），不打天子本人
hd253_beifa_defender_trigger = {
	is_independent_ruler = no
	highest_held_title_tier >= tier_duchy
	top_liege = {
		OR = {
			has_title = title:h_china
			has_title = title:e_hd218_wei
		}
	}
}

# 开局边地首领刷新：独立、年过七十、首府不在三国境内的地方首领（史实人物除外）
hd253_old_chief_trigger = {
	is_independent_ruler = yes
	is_ai = yes
	age > 70
	highest_held_title_tier <= tier_kingdom
	NOT = { this = character:tuoba_liwei }
	NOT = { this = character:bokitewi }
	NOT = { has_title = title:h_china }
	exists = capital_county
}

# 10-18 加固 B：当朝权臣是司马师或司马昭（在世、仍是魏天子封臣）。任意 scope
hd253_sima_regent_trigger = {
	title:h_china.var:hd_court_regent ?= {
		is_alive = yes
		OR = {
			this = character:si_ma_shi
			this = character:si_ma_zhao
		}
		liege ?= title:h_china.holder
	}
}

# 魏将来降的人选：成年、无地、未被囚、有武艺；史实上不会入蜀的司马氏、邓艾、钟会、钟毓除外
hd253_defector_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = no
	is_imprisoned = no
	martial >= 8
	NOT = { house ?= house:henei_sima }
	NOT = { this = character:deng_ai }
	NOT = { global_var:hd_sg_yingchuan_zhong_024 ?= this }
	NOT = { this = character:yingchuan_zhong_022 }
}
