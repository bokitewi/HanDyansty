# Interaction actor/recipient scopes. The interface sends an invitation only;
# its automatic delivery acceptance is never political acceptance by the target.
hd_dc_royal_offer_pair_trigger = {
	exists = situation:dynastic_cycle
	exists = scope:actor
	exists = scope:recipient
	scope:actor = {
		is_alive = yes
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = incapable }
		in_diplomatic_range = scope:recipient
	}
	scope:recipient = {
		is_alive = yes
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = incapable }
		this != scope:actor
		this != title:h_china.holder
		NOT = { is_at_war_with = scope:actor }
		NOT = { is_at_war_with = title:h_china.holder }
	}
	situation:dynastic_cycle = {
		any_in_list = {
			variable = hd_dc_instances
			var:hd_dc_type = flag:roy
			OR = { var:hd_dc_stage = flag:acute AND = { var:hd_dc_stage = flag:aftermath has_variable = hd_dc_roy01_node_closed NOT = { has_variable = hd_dc_political_result } } }
			this = situation:dynastic_cycle.var:hd_dc_latest_roy_instance
			NOT = { has_variable = hd_dc_reward_committed }
			any_in_list = {
				variable = hd_dc_actor_stories
				var:hd_dc_actor = scope:actor
				hd_dc_royal_story_has_faction_trigger = yes
				NOT = { var:hd_dc_claimant_faction = { hd_dc_claimant_mobilization_paused_trigger = yes } }
				NOT = { var:hd_dc_claimant_faction = { exists = faction_war } }
				save_temporary_scope_as = hd_dc_offer_actor_story
				scope:recipient = {
					NOT = { joined_faction = scope:hd_dc_offer_actor_story.var:hd_dc_claimant_faction }
					can_join_faction = scope:hd_dc_offer_actor_story.var:hd_dc_claimant_faction
				}
				NOT = {
					any_in_list = {
						variable = hd_dc_support_offers
						var:hd_dc_support_recipient = scope:recipient
						OR = { has_variable = hd_dc_support_wait var:hd_dc_support_commit = flag:joined has_variable = hd_dc_support_retry_wait }
					}
				}
			}
		}
	}
}

hd_dc_royal_offer_response_trigger = {
	save_temporary_scope_as = hd_dc_support_actor
	hd_dc_royal_support_identity_trigger = yes
	NOT = { scope:hd_dc_support.var:hd_dc_support_faction = { hd_dc_claimant_mobilization_paused_trigger = yes } }
	in_diplomatic_range = scope:hd_dc_support.var:hd_dc_support_claimant
	scope:hd_dc_support = {
		var:hd_dc_record_version = scope:hd_dc_support_delivery_version
		has_variable = hd_dc_support_wait
		NOT = { has_variable = hd_dc_support_answer }
		NOT = { has_variable = hd_dc_support_commit }
	}
}
