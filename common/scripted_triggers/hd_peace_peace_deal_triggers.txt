hd_peace_treaty_context_is_valid_trigger = {
	root = {
		exists = var:hd_peace_treaty_war
		exists = var:hd_peace_treaty_opponent
		trigger_if = {
			limit = {
				exists = var:hd_peace_treaty_war
				exists = var:hd_peace_treaty_opponent
			}
			var:hd_peace_treaty_war = {
				OR = {
					AND = {
						primary_attacker = root
						primary_defender = root.var:hd_peace_treaty_opponent
					}
					AND = {
						primary_defender = root
						primary_attacker = root.var:hd_peace_treaty_opponent
					}
				}
			}
		}
	}
}

hd_peace_treaty_is_editable_trigger = {
	hd_peace_treaty_context_is_valid_trigger = yes
	hd_peace_ordinary_context_trigger = yes
	NOT = { has_variable = hd_peace_treaty_locked }
}

hd_peace_nomadic_herd_exchange_available_trigger = {
	has_variable = hd_peace_treaty_opponent
	government_has_flag = government_is_nomadic
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			government_has_flag = government_is_nomadic
		}
	}
}

hd_peace_can_offer_herd_trigger = {
	hd_peace_nomadic_herd_exchange_available_trigger = yes
	domicile.herd > 0
}

hd_peace_can_demand_herd_trigger = {
	hd_peace_nomadic_herd_exchange_available_trigger = yes
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = { domicile.herd > 0 }
	}
}
hd_peace_can_become_tributary_of_opponent_trigger = {
	is_independent_ruler = yes
	NOT = { is_tributary = yes }
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			is_independent_ruler = yes
			NOT = { is_tributary = yes }
		}
	}
}

hd_peace_opponent_can_become_tributary_trigger = {
	has_variable = hd_peace_treaty_opponent
	trigger_if = {
		limit = { has_variable = hd_peace_treaty_opponent }
		var:hd_peace_treaty_opponent = {
			is_independent_ruler = yes
			NOT = { is_tributary = yes }
		}
	}
	is_independent_ruler = yes
	NOT = { is_tributary = yes }
}
