# Any scope. D30 evidence is separate from a completed invitation.
# The QXZL scan flag is written by the existing thirty-day callback.
# Explicit player refusals and later Xuzhou progress are never overridden.
hd_lb_gongsun_window_open_trigger = {
	game_start_date = 184.1.1
	exists = global_var:hd_anti_regent_chain_resolved
	OR = {
		exists = global_var:hd_lb_gongsun_d30_processed
		exists = global_var:hd_qxzl_scan_done
	}
	NOT = { exists = global_var:hd_lb_bequest_completed }
	exists = character:liu_bei
	exists = character:gong_sun_zan
	character:liu_bei ?= {
		is_alive = yes
		NOT = {
			OR = {
				has_character_flag = hd_lb_gongsun_player_declined
				has_character_flag = hd_lb_tao_accepted
				has_character_flag = hd_lb_mi_visit_done
			}
		}
	}
	character:gong_sun_zan ?= {
		is_alive = yes
		NOT = { has_character_flag = hd_lb_gongsun_sender_declined }
	}
}

# Preserve the existing actor, capital ownership, prison and war requirements.
# Only evaluate the original non-optional character links when both exist.
hd_lb_gongsun_attempt_valid_trigger = {
	hd_lb_gongsun_window_open_trigger = yes
	trigger_if = {
		limit = { hd_lb_gongsun_window_open_trigger = yes }
		hd_lb_gongsun_offer_valid_trigger = yes
	}
}
