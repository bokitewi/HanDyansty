# A county where a named local office holder can answer the emperor's policy
# inquiry. The emperor's own demesne is not relabeled as a separate governor.
hd_dc_party_case_county_trigger = {
	hd_dc_policy_affected_county_trigger = yes
	exists = holder
	holder = {
		is_alive = yes
		is_imprisoned = no
		is_vassal_or_below_of = title:h_china.holder
	}
	NOT = { has_variable = hd_dc_policy_trial }
	NOT = { has_county_modifier = ep3_governance_tax_relief }
	NOT = { has_county_modifier = hd_dc_par_tax_relief_modifier }
}

# Case record. A personal grievance does not stand in for a policy statement.
hd_dc_party_case_current_trigger = {
	var:hd_dc_record_kind = flag:party_case
	NOT = { has_variable = hd_dc_case_result }
	var:hd_dc_case_year = current_year
	var:hd_dc_case_ruler = title:h_china.holder
	exists = var:hd_dc_case_county
	var:hd_dc_case_county = {
		hd_dc_party_case_county_trigger = yes
		holder = scope:hd_dc_party_case.var:hd_dc_case_executor
	}
	situation:dynastic_cycle = {
		hd_dc_par_snapshot_valid_trigger = yes
		NOT = { has_variable = hd_dc_acute_instance_id }
		NOT = { has_variable = hd_dc_pending_instance }
		var:hd_dc_pending_party_case = scope:hd_dc_party_case
		any_participant_group = {
			var:hd_dc_par_group_type = scope:hd_dc_party_case.var:hd_dc_case_first_type
			var:movement_leader = scope:hd_dc_party_case.var:hd_dc_case_first_leader
			hd_dc_par_group_qualified_trigger = yes
			save_temporary_scope_as = hd_dc_par_first_group
			situation:dynastic_cycle = {
				any_participant_group = {
					var:hd_dc_par_group_type = scope:hd_dc_party_case.var:hd_dc_case_second_type
					var:movement_leader = scope:hd_dc_party_case.var:hd_dc_case_second_leader
					hd_dc_par_hostile_pair_trigger = yes
				}
			}
		}
	}
}

hd_dc_party_case_window_valid_trigger = {
	exists = scope:hd_dc_party_case
	scope:hd_dc_party_case = { hd_dc_party_case_current_trigger = yes has_variable = hd_dc_case_wait }
	OR = {
		AND = {
			scope:hd_dc_case_side = flag:first
			this = scope:hd_dc_party_case.var:hd_dc_case_first_leader
			scope:hd_dc_party_case = { NOT = { has_variable = hd_dc_case_first_vote } }
		}
		AND = {
			scope:hd_dc_case_side = flag:second
			this = scope:hd_dc_party_case.var:hd_dc_case_second_leader
			scope:hd_dc_party_case = { NOT = { has_variable = hd_dc_case_second_vote } }
		}
	}
}
