# 请求归附（弱者主动投靠）。角色作用域。
# 与“劝其归附”方向相反：actor = 投靠者（弱），recipient = 被投靠者（强）。

# 投靠者自身资格：便宜的条件在前。
hd_petition_actor_trigger = {
    is_ai = yes
    is_alive = yes
    is_landed = yes
    is_independent_ruler = yes
    sub_realm_size <= 5
    highest_held_title_tier >= tier_county
    highest_held_title_tier < tier_kingdom
    is_at_war = no
    is_physically_able_adult = yes
    NOR = {
        has_trait = ambitious
        has_trait = arrogant
        has_character_flag = hd_petition_outgoing_pending
        has_character_flag = hd_submission_incoming_pending
        has_character_flag = hd_petition_cooldown
    }
    # 历史主角及其子女不主动投靠
    NOT = { hd_rise_roster_trigger = yes }
    NOT = { any_parent = { hd_rise_roster_trigger = yes } }
    NOT = { AND = { this = character:tao_qian has_character_flag = hd4_tao_activated } }
    hd_submission_world_active_trigger = yes
    hd_unification_realm_in_china_trigger = yes
}

# 被投靠者 scope；$ACTOR$ 必须是已保存的 scope（投靠者）。
hd_petition_candidate_trigger = {
    hd_submission_liege_trigger = yes
    highest_held_title_tier > $ACTOR$.highest_held_title_tier
    current_military_strength > 0
    current_military_strength >= hd_petition_required_strength
    trigger_if = {
        limit = { is_ai = no }
        NOT = { has_character_flag = hd_petition_player_inbox_cooldown }
    }
    trigger_if = {
        limit = { exists = $ACTOR$.var:hd_petition_declined_liege }
        NOT = { $ACTOR$.var:hd_petition_declined_liege = this }
    }
}
