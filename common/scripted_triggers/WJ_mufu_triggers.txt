# A.45/A.46. Five slots are the only membership authority.
WJ_mufu_owner_eligible = {
 is_alive = yes
 highest_held_title_tier >= tier_kingdom
 WJ_has_kaifu_privilege_trigger = yes
 OR = {
  AND = { tgp_is_any_minister = yes vassal_contract_has_flag = celestial_minister_obligation }
  AND = {
   WJ_contract_local_official_trigger = yes
   OR = { vassal_contract_has_flag = celestial_province_military vassal_contract_has_flag = WJ_celestial_province_sili }
  }
 }
}
WJ_mufu_military = {
 OR = {
  vassal_contract_has_flag = celestial_grand_marshal_obligation
  vassal_contract_has_flag = celestial_province_military
  vassal_contract_has_flag = WJ_celestial_province_sili
 }
}
# HOST is the actual patron, never physical location (travel/prison).
WJ_mufu_same_realm = {
 OR = {
  AND = { is_pool_guest = yes exists = host host.top_liege = $HOST$.top_liege }
  AND = { is_pool_guest = no top_liege = $HOST$.top_liege }
 }
}
WJ_mufu_empty = {
 NAND = {
  has_variable = wj_mufu_slot_1
  has_variable = wj_mufu_slot_2
  has_variable = wj_mufu_slot_3
  has_variable = wj_mufu_slot_4
  has_variable = wj_mufu_slot_5
 }
}
WJ_mufu_candidate = {
 is_alive = yes
 is_adult = yes
 is_female = no
 this != $HOST$
 NOT = { has_variable = wj_mufu_owner }
 WJ_mufu_same_realm = { HOST = $HOST$ }
 highest_held_title_tier <= $HOST$.highest_held_title_tier
 OR = {
  AND = { government_has_flag = government_is_celestial OR = { tgp_is_any_minister = yes WJ_contract_local_official_trigger = yes } }
  is_courtier = yes
  is_pool_guest = yes
  WJ_yangwang_eligible_trigger = yes
 }
 # Transitive descendant index is maintained on every attach/detach, not a staff pool.
 NOT = { is_target_in_variable_list = { name = wj_mufu_descendants target = $HOST$ } }
 NOT = { is_target_in_variable_list = { name = wj_mufu_rejected_owners target = $HOST$ } }
}
WJ_mufu_member_valid = {
 is_alive = yes
 exists = var:wj_mufu_owner
 var:wj_mufu_owner = { WJ_mufu_owner_eligible = yes }
 WJ_mufu_same_realm = { HOST = var:wj_mufu_owner }
 highest_held_title_tier <= var:wj_mufu_owner.highest_held_title_tier
 OR = {
  AND = { var:wj_mufu_slot = 1 var:wj_mufu_owner.var:wj_mufu_slot_1 = this }
  AND = { var:wj_mufu_slot = 2 var:wj_mufu_owner.var:wj_mufu_slot_2 = this }
  AND = { var:wj_mufu_slot = 3 var:wj_mufu_owner.var:wj_mufu_slot_3 = this }
  AND = { var:wj_mufu_slot = 4 var:wj_mufu_owner.var:wj_mufu_slot_4 = this }
  AND = { var:wj_mufu_slot = 5 var:wj_mufu_owner.var:wj_mufu_slot_5 = this }
 }
}
WJ_mufu_request_valid = {
 scope:actor = { WJ_mufu_owner_eligible = yes WJ_mufu_empty = yes }
 scope:recipient = { WJ_mufu_candidate = { HOST = scope:actor } }
}
WJ_mufu_selected_empty = {
 OR = {
  AND = { scope:mufu_role_1 = yes scope:actor = { NOT = { has_variable = wj_mufu_slot_1 } } scope:mufu_role_2 = no scope:mufu_role_3 = no scope:mufu_role_4 = no scope:mufu_role_5 = no }
  AND = { scope:mufu_role_2 = yes scope:actor = { NOT = { has_variable = wj_mufu_slot_2 } } scope:mufu_role_1 = no scope:mufu_role_3 = no scope:mufu_role_4 = no scope:mufu_role_5 = no }
  AND = { scope:mufu_role_3 = yes scope:actor = { NOT = { has_variable = wj_mufu_slot_3 } } scope:mufu_role_1 = no scope:mufu_role_2 = no scope:mufu_role_4 = no scope:mufu_role_5 = no }
  AND = { scope:mufu_role_4 = yes scope:actor = { NOT = { has_variable = wj_mufu_slot_4 } } scope:mufu_role_1 = no scope:mufu_role_2 = no scope:mufu_role_3 = no scope:mufu_role_5 = no }
  AND = { scope:mufu_role_5 = yes scope:actor = { NOT = { has_variable = wj_mufu_slot_5 } } scope:mufu_role_1 = no scope:mufu_role_2 = no scope:mufu_role_3 = no scope:mufu_role_4 = no }
 }
}
WJ_mufu_can_resign = {
 WJ_mufu_member_valid = yes
 NOT = { has_variable = wj_mufu_join_lock }
}

WJ_mufu_terms_valid = {
 scope:actor.influence >= WJ_mufu_total_influence_cost
 trigger_if = { limit = { scope:prestige = yes } scope:actor.prestige >= prestige_bribe_value }
 trigger_if = { limit = { scope:gift = yes } scope:actor.gold > 0 }
 trigger_if = { limit = { scope:hook = yes } scope:actor = { has_usable_hook = scope:recipient } }
 trigger_if = { limit = { scope:offer_hook = yes } scope:recipient = { NOT = { has_hook = scope:actor } } }
}
