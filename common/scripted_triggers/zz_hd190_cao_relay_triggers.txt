# 190 曹操陈留—兖州接力（2026-10-26）。效果见 common/scripted_effects/zz_hd190_cao_relay_effects.txt。

# 曹操还能接着打：在世、有地、成年、未被囚、独立（或挟天子的权臣）。Scope：曹操
hd190_cao_relay_actor_ok_trigger = {
	this = character:cao_cao
	is_alive = yes
	is_landed = yes
	is_imprisoned = no
	hd_court_independent_or_regent_trigger = yes
}

# 可作接力对象：独立君主、非天子、未结盟、曹操对其无停战、未与曹操交战、本轮未宣战失败过。Scope：候选人
hd190_cao_relay_valid_def_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	NOT = { this = character:cao_cao }
	NOT = { has_title = title:h_china }
	NOT = { has_title = title:e_han }
	NOT = { government_has_flag = cannot_be_vassal_or_liege }
	NOT = { is_allied_to = character:cao_cao }
	NOT = { is_at_war_with = character:cao_cao }
	NOT = { character:cao_cao = { has_truce = prev } }
	NOT = { character:cao_cao = { is_target_in_variable_list = { name = hd190_cao_relay_failed target = prev } } }
}

# 县与曹操疆域接壤，且其最高领主可作接力对象。Scope：县
hd190_cao_relay_border_county_trigger = {
	holder ?= { top_liege = { hd190_cao_relay_valid_def_trigger = yes } }
	any_neighboring_county = { holder ?= { top_liege = character:cao_cao } }
}

# 张邈可打：可作接力对象，且领内有县与曹操接壤。Scope：张邈
hd190_cao_relay_zhang_miao_ok_trigger = {
	hd190_cao_relay_valid_def_trigger = yes
	any_sub_realm_county = { any_neighboring_county = { holder ?= { top_liege = character:cao_cao } } }
}

# 正在结算的战争是接力战争：攻方曹操在接力中，守方是记在他身上的接力对象。Scope：吞并 CB 结算（scope:attacker / scope:defender）
hd190_cao_relay_war_trigger = {
	exists = scope:attacker
	exists = scope:defender
	scope:attacker = {
		this = character:cao_cao
		has_character_flag = hd190_cao_relay_active
		var:hd190_cao_relay_def ?= scope:defender
	}
}
