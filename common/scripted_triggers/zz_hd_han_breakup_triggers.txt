# Current character must still be a territorial vassal. No geography rescan here:
# D0 membership is captured from h_china's holder and independent kingdom/empire
# rulers whose primary title is de jure Han, including their vassal trees.
hd_hb_live_vassal_trigger = {
	is_alive = yes
	is_landed = yes
	highest_held_title_tier >= tier_county
	is_independent_ruler = no
	exists = liege
}

# Force and block are independent booleans. Their intersection uses the same
# weighted branch as neither; do not put either side ahead of the other.
hd_hb_force_independence_trigger = {
	OR = {
		has_trait = ambitious
		# 2026-10-16：伯爵不因非天朝政体被强制独立，只走概率。
		AND = {
			highest_held_title_tier >= tier_duchy
			NOT = { government_has_flag = government_is_celestial }
		}
		has_relation_rival = liege
		AND = {
			OR = { has_trait = just has_trait = zealous }
			liege = { dread > 20 }
			opinion = { target = liege value < 30 }
		}
		AND = {
			highest_held_title_tier = tier_duchy
			liege = { highest_held_title_tier = tier_kingdom governor_efficiency < 0.25 }
		}
		AND = {
			highest_held_title_tier = tier_county
			liege = { highest_held_title_tier = tier_duchy governor_efficiency < 0.15 }
		}
	}
}

hd_hb_block_independence_trigger = {
	OR = {
		is_obedient_to = liege
		has_trait = loyal
		has_relation_best_friend = liege
		has_relation_soulmate = liege
		has_relation_blood_brother = liege
	}
}

# Both scopes exist when the player choice is created. If its liege changes,
# the original choice cannot detach the player from an unrelated new liege.
hd_hb_player_choice_valid_trigger = {
	hd_hb_live_vassal_trigger = yes
	is_ai = no
	has_character_flag = hd_hb_choice_pending
	exists = scope:hd_hb_choice_liege
	liege = scope:hd_hb_choice_liege
}

# Member character scope. The authoritative attitude/stance belongs to the
# matching party on the current court ruler; never inspect another party.
hd_hb_party_group_state_trigger = {
	var:hd_party_my_group = flag:$G$
	exists = top_liege.var:hd_party_$G$_$KIND$
	top_liege.var:hd_party_$G$_$KIND$ = flag:$STATE$
}

hd_hb_party_state_trigger = {
	exists = top_liege
	exists = var:hd_party_my_court
	exists = var:hd_party_my_group
	var:hd_party_my_court = top_liege
	top_liege = { is_alive = yes has_variable = hd_party_court }
	OR = {
		hd_hb_party_group_state_trigger = { G = huanguan KIND = $KIND$ STATE = $STATE$ }
		hd_hb_party_group_state_trigger = { G = waiqi KIND = $KIND$ STATE = $STATE$ }
		hd_hb_party_group_state_trigger = { G = wuxun KIND = $KIND$ STATE = $STATE$ }
		hd_hb_party_group_state_trigger = { G = shizu KIND = $KIND$ STATE = $STATE$ }
		hd_hb_party_group_state_trigger = { G = hanmen KIND = $KIND$ STATE = $STATE$ }
	}
}
