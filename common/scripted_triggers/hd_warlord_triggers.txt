# 郡国制·军阀契约（2026-10-24 用户指定）
# 军阀身份记在主头衔变量 hd_warlord 上：人死换人也跟着头衔走，只有经欢呼继承接位才保留
# （见 hd_warlord_on_title_gain）。契约档位、继承法、免税免兵都以此为准，不认契约 flag，
# 因为继承人尚未拿到契约时引擎就会先校验档位。

# root = 角色
hd_warlord_trigger = {
	primary_title ?= { has_variable = hd_warlord }
}

# 自请为军阀的基本资格（决议 is_shown 用）
hd_warlord_declare_shown_trigger = {
	government_has_flag = government_is_celestial
	is_independent_ruler = no
	is_landed = yes
	exists = liege
	primary_title ?= {
		tier >= tier_county
		is_noble_family_title = no
	}
	NOT = { hd_warlord_trigger = yes }
}

# 强力封臣是否支持自请者（2026-10-25 用户指定）：忠顺、朋友/至交、好感 ≥ 70、或对其「畏惧」（威慑 2 级）
# root = 强力封臣；$CANDIDATE$ = 自请为军阀者
hd_warlord_supporter_trigger = {
	OR = {
		is_obedient_to = $CANDIDATE$
		has_relation_friend = $CANDIDATE$
		has_relation_best_friend = $CANDIDATE$
		opinion = {
			target = $CANDIDATE$
			value >= 70
		}
		has_dread_level_towards = {
			target = $CANDIDATE$
			level = 2
		}
	}
}

# 收回权力后能否落到第 3 档军事编制：与原版档位 is_valid 同口径（关隘县、直属最高领主或领主本身是武官；
# 直属最高领主时受军事编制名额限制）。不满足则退为第 0 档普通文官
# root = 被收回的封臣
hd_warlord_can_be_military_trigger = {
	OR = {
		WJ_name_pass_county_trigger = yes
		liege = top_liege
		liege = { vassal_contract_has_flag = celestial_military_appointment }
	}
	trigger_if = {
		limit = { liege = top_liege }
		liege = {
			any_vassal = {
				count < celestial_province_military_max_value
				vassal_contract_has_flag = celestial_province_military
			}
		}
	}
}
