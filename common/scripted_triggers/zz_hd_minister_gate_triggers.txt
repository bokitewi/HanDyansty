# 公卿席位（原版九职 + 太傅/宗正/丞相/光禄勋）共用的判断。

# 是否为十三个内阁头衔之一（原版 tgp_is_minister_title 只认九职）。
hd_is_any_ministry_title_trigger = {
	OR = {
		tgp_is_minister_title = yes
		this = title:e_WJ_minister_tai_fu
		this = title:e_WJ_minister_zong_zheng
		this = title:e_WJ_minister_cheng_xiang
		this = title:e_WJ_minister_guang_lu_xun
	}
}

hd_holds_any_ministry_title_trigger = {
	any_held_title = { hd_is_any_ministry_title_trigger = yes }
}

# 贤望须达卿档（海内称誉，merit_level 7）方可列入省部任命之选。
# 现任者（已持此头衔或已在此席）不受影响；开局首月留给剧本脚本布置；
# 剧本指定人选另以 hd_cp_merit_exempt 放行。
hd_minister_merit_gate_trigger = {
	trigger_if = {
		limit = {
			exists = liege_or_court_owner
			liege_or_court_owner = { tgp_has_access_to_ministry_trigger = yes }
			months_from_game_start >= 1
			NOR = {
				has_title = title:$MINISTER_TITLE$
				has_council_position = $POSITION$
				has_character_flag = hd_cp_merit_exempt
			}
		}
		custom_tooltip = {
			text = hd_minister_merit_gate_tt
			merit_level >= 7
		}
	}
	# 184 剧本流程重制 D7：董卓死前，州牧不入公卿（入朝会辞掉本州并引发连锁调任）。现任者不受影响。
	trigger_if = {
		limit = {
			hd184r_governor_freeze_active_trigger = yes
			NOR = {
				has_title = title:$MINISTER_TITLE$
				has_council_position = $POSITION$
				hd184r_freeze_exempt_trigger = yes
			}
		}
		custom_tooltip = {
			text = hd184r_freeze_minister_tt
			NOT = { hd184r_holds_kingdom_land_trigger = yes }
		}
	}
}

# 卸任后由头衔继承人顺补时，继承人同样须达卿档。
hd_minister_heir_merit_ok_trigger = {
	OR = {
		merit_level >= 7
		months_from_game_start < 1
		has_character_flag = hd_cp_merit_exempt
	}
}
