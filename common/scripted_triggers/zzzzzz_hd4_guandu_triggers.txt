hd4_guandu_actors_trigger = {
	AND = {
		exists = character:cao_cao
		character:cao_cao = {
			is_alive = yes
			is_landed = yes
			hd_cao_historical_war_leader_trigger = yes
		}
	}
	AND = {
		exists = character:yuan_shao
		character:yuan_shao = {
			is_alive = yes
			is_landed = yes
			is_independent_ruler = yes
		}
	}
	NOT = {
		character:cao_cao.top_liege = character:yuan_shao.top_liege
	}
}
hd4_north_county_in_two_realms_trigger = {
	AND = {
		exists = holder
		holder = {
			OR = {
				this = character:cao_cao
				is_vassal_or_below_of = character:cao_cao
				this = character:yuan_shao
				is_vassal_or_below_of = character:yuan_shao
			}
		}
	}
}
hd4_north_only_two_trigger = {
	hd4_guandu_actors_trigger = yes
	title:h_china = {
		any_in_list = {
			variable = hd4_north_counties
			always = yes
		}
		NOT = {
			any_in_list = {
				variable = hd4_north_counties
				NOT = {
					hd4_north_county_in_two_realms_trigger = yes
				}
			}
		}
		any_in_list = {
			variable = hd4_north_counties
			AND = {
				exists = holder
				holder = {
					OR = {
						this = character:cao_cao
						is_vassal_or_below_of = character:cao_cao
					}
				}
			}
		}
		any_in_list = {
			variable = hd4_north_counties
			AND = {
				exists = holder
				holder = {
					OR = {
						this = character:yuan_shao
						is_vassal_or_below_of = character:yuan_shao
					}
				}
			}
		}
	}
}
hd4_guandu_can_attempt_trigger = {
	this = character:cao_cao
	is_alive = yes
	is_landed = yes
	is_imprisoned = no
	has_character_flag = hd4_guandu_authorized
	hd4_guandu_actors_trigger = yes
	NOT = {
		title:h_china = {
			has_variable = hd4_guandu_done
		}
	}
	NOT = {
		title:h_china = {
			has_variable = hd4_legacy_guandu
		}
	}
	NOT = {
		has_truce = character:yuan_shao
	}
	NOT = {
		is_allied_to = character:yuan_shao
	}
	NOT = {
		is_at_war_with = character:yuan_shao
	}
	NOT = {
		any_character_war = {
			count >= 3
			always = yes
		}
	}
}
