# Check the council's actual employer, or the actual court containing a courtier.
# A character's own title tier does not determine the rank of their employment.
hd_han_low_court_membership_trigger = {
	OR = {
		any_court_position_employer = {
			highest_held_title_tier <= tier_duchy
		}
		AND = {
			is_councillor = yes
			exists = liege
			is_councillor_of = liege
			liege = { highest_held_title_tier <= tier_duchy }
		}
		AND = {
			is_councillor = yes
			exists = court_owner
			is_councillor_of = court_owner
			court_owner = { highest_held_title_tier <= tier_duchy }
		}
		AND = {
			is_courtier = yes
			exists = court_owner
			court_owner != this
			court_owner = { highest_held_title_tier <= tier_duchy }
		}
	}
}

# Real central seat requires the current emperor, council appointment and title.
hd_han_central_seat_trigger = {
	has_council_position = $POSITION$
	has_title = title:$TITLE$
	is_councillor_of = top_liege

}

# Faction affiliation is a current camp/realm relation or an actual attacking war.
# Rite alone does not establish Yellow Turban membership.
hd_han_yellow_turban_member_trigger = {
	exists = this
	is_male = yes
	is_adult = yes
	hd_han_pool_royal_trigger = no
	OR = {
		AND = { exists = character:zhang_jiao character:zhang_jiao = { is_alive = yes } OR = { this = character:zhang_jiao top_liege = character:zhang_jiao host = character:zhang_jiao } }
		AND = { exists = character:zhang_man_cheng character:zhang_man_cheng = { is_alive = yes } OR = { this = character:zhang_man_cheng top_liege = character:zhang_man_cheng host = character:zhang_man_cheng } }
		AND = { exists = character:bo_cai character:bo_cai = { is_alive = yes } OR = { this = character:bo_cai top_liege = character:bo_cai host = character:bo_cai } }
		any_character_war = { using_cb = hd_yellow_turban_rebellion_war is_attacker = prev }
	}
}
