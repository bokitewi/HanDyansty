# Position validation also runs for incumbents. Refusal must not dismiss them.
# Call only from real minister positions; never from the shared minister basics.
WJ_ministry_position_acceptance_trigger = {
	# Keep the actual chancellor valid in his own seat, but out of other pools.
	trigger_if = {
		limit = { OR = { has_title = title:e_WJ_minister_cheng_xiang has_council_position = WJ_minister_cheng_xiang } }
		title:$MINISTER_TITLE$ = { this = title:e_WJ_minister_cheng_xiang }
	}
	# 190初始化只允许历史官署持有人入对应席位：不让旧184任职记录或
	# 引擎随机人选覆盖190史官；无持有人的大将军自然留空。进局后正常任命。
	trigger_if = {
		limit = {
			game_start_date = 190.1.1
			current_date = 190.1.1
			NOT = { has_global_variable = hd190_after_lobby_done }
			exists = root.liege_or_court_owner
			root.liege_or_court_owner = { has_title = title:h_china }
		}
		has_title = title:$MINISTER_TITLE$
	}
	trigger_if = {
		limit = {
			exists = root.liege_or_court_owner
			root.liege_or_court_owner = { tgp_has_access_to_ministry_trigger = yes }
		}
		OR = {
			WJ_accepts_new_ministry_appointment_trigger = yes
			has_title = title:$MINISTER_TITLE$
			AND = {
				has_council_position = $POSITION$
				is_councillor_of = root.liege_or_court_owner
			}
		}
	}
}

# Candidate scope, explicit target title. Ordinary local titles remain available.
WJ_ministry_target_acceptance_trigger = {
	trigger_if = {
		limit = { $TITLE$ = { tgp_is_minister_title = yes } }
		WJ_accepts_new_ministry_appointment_trigger = yes
	}
}

# Real state office, not a titular family or ministry empire.
WJ_ministry_major_local_authority_trigger = {
	government_has_flag = government_is_celestial
	OR = {
		any_held_title = {
			tier = tier_kingdom
			is_noble_family_title = no
			tgp_is_minister_title = no
			OR = { is_landless_type_title = no AND = { has_variable = wj_rf_rank holder = { is_landed = yes } } }
			any_de_jure_county = { always = yes }
		}
		WJ_dd_head_trigger = yes
	}
}
