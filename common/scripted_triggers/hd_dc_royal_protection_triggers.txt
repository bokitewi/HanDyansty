# Receipt scope. Reuse actual political identity without treating a request
# for protection as either entry into the faction or a paid support contract.
hd_dc_royal_protection_context_trigger = {
	var:hd_dc_support_recipient = { save_temporary_scope_as = hd_dc_support_actor }
	var:hd_dc_support_recipient = { hd_dc_royal_support_identity_trigger = yes is_imprisoned = no is_adult = yes }
	var:hd_dc_support_claimant = { is_alive = yes is_imprisoned = no in_diplomatic_range = scope:hd_dc_support_actor }
}

hd_dc_royal_protection_response_trigger = {
	is_alive = yes is_imprisoned = no
	exists = scope:hd_dc_support
	this = scope:hd_dc_support.var:hd_dc_support_claimant
	scope:hd_dc_support = {
		var:hd_dc_record_version = scope:hd_dc_support_delivery_version
		var:hd_dc_support_answer = flag:protection_requested
		has_variable = hd_dc_protection_reply_wait
		NOT = { has_variable = hd_dc_protection_answer }
		hd_dc_royal_protection_context_trigger = yes
	}
}
