# Political membership plus physical estates/courts; foreign prisons take priority.
hd_dong_inside_realm_trigger = {
	is_alive = yes
	trigger_if = {
		limit = { is_imprisoned = yes }
		imprisoner = { OR = { this = $RULER$ is_vassal_or_below_of = $RULER$ } }
	}
	trigger_else = {
		OR = {
			this = $RULER$
			is_vassal_or_below_of = $RULER$
			is_courtier_of = $RULER$
			AND = { exists = employer employer = { OR = { this = $RULER$ is_vassal_or_below_of = $RULER$ } } }
			AND = { exists = location.county.holder location.county.holder = { OR = { this = $RULER$ is_vassal_or_below_of = $RULER$ } } }
			AND = { exists = domicile.domicile_location.county.holder domicile.domicile_location.county.holder = { OR = { this = $RULER$ is_vassal_or_below_of = $RULER$ } } }
		}
	}
}

# Han's own court and personal domain are under Dong's regency, not all Han vassals.
hd_dong_yuan_execution_target_trigger = {
	is_alive = yes
	OR = {
		hd_dong_inside_realm_trigger = { RULER = character:dong_zhuo_2 }
		AND = {
			exists = title:h_china.holder
			character:dong_zhuo_2 = { has_character_flag = hd_current_han_regent }
			OR = {
				AND = { is_imprisoned = yes imprisoner = title:h_china.holder }
				AND = {
					is_imprisoned = no
					OR = {
						is_courtier_of = title:h_china.holder
						AND = { exists = employer employer = title:h_china.holder }
						AND = { exists = location.county.holder location.county.holder = title:h_china.holder }
						AND = { exists = domicile.domicile_location.county.holder domicile.domicile_location.county.holder = title:h_china.holder }
					}
				}
			}
		}
	}
}

hd_dong_historical_regent_active_trigger = {
	exists = title:h_china.holder
	title:h_china.holder = {
		has_variable = hd_dong_fixed_regent
		var:hd_dong_fixed_regent = {
			is_alive = yes
			OR = { this = character:wang_yun this = character:li_jue this = character:cao_cao }
		}
	}
}

hd_dong_transition_pending_trigger = {
	OR = {
		# Some CB endings mark resolution a day before the cleanup event runs.
		AND = {
			exists = global_var:hd_anti_regent_chain_resolved
			exists = global_var:hd_anti_regent_first_war_confirmed
			NOT = { exists = global_var:hd_dong_murder_scheduled }
			character:dong_zhuo_2 = { is_alive = yes }
		}
		exists = global_var:hd_dong_murder_pending
		exists = global_var:hd_dong_settlement_pending
		exists = global_var:hd_li_jue_war_pending
		exists = global_var:hd_li_jue_war_active
	}
}

# Office initialization must still respect an incumbent historical regent.
hd_dong_blocks_fengying_trigger = {
	OR = {
		hd_dong_transition_pending_trigger = yes
		hd_dong_historical_regent_active_trigger = yes
	}
}
