# 天朝循环·时代特色重制：判定（见 docs/天朝循环_时代特色重制计划.md）

# Scope：角色。VIP（Q8）：天子及其配偶/继承人；王国级以上统治者；派系/朋党领袖与宰辅；公爵级独立统治者
hd_dc_is_vip_trigger = {
	is_alive = yes
	OR = {
		# 天子及配偶/继承人
		AND = {
			exists = title:h_china.holder
			OR = {
				this = title:h_china.holder
				is_spouse_of = title:h_china.holder
				title:h_china.holder.primary_heir ?= this
			}
		}
		# 王国级以上统治者
		AND = {
			is_ruler = yes
			highest_held_title_tier >= tier_kingdom
		}
		# 公爵级独立统治者
		AND = {
			is_independent_ruler = yes
			highest_held_title_tier >= tier_duchy
		}
		# 原版运动领袖
		top_participant_group:dynastic_cycle.var:movement_leader ?= this
		# 朋党领袖（任一帝国级朝廷）
		AND = {
			exists = var:hd_party_my_court
			hd_party_is_any_leader_of_court_trigger = { COURT = var:hd_party_my_court }
		}
		# 宰辅：天朝大臣或天子议事会成员
		tgp_has_minister_title = yes
		AND = {
			exists = title:h_china.holder
			is_councillor_of = title:h_china.holder
		}
	}
}

# Scope：角色。位于天下（h_china 法理）之内
hd_dc_in_china_trigger = {
	OR = {
		capital_county ?= { target_is_de_jure_liege_or_above = title:h_china }
		# 2026-10-10 乱世霸权：被挟持的 0 县天子按权臣的首都判定
		AND = {
			hd_court_defender_is_held_emperor_trigger = yes
			title:h_china.var:hd_court_regent.capital_county ?= { target_is_de_jure_liege_or_above = title:h_china }
		}
	}
}

# Scope：角色。天下之外的势力（"四夷"判定口径）
hd_dc_is_outsider_trigger = {
	is_ruler = yes
	NOT = { hd_dc_in_china_trigger = yes }
}

# Scope：君主（朝廷）。$G$ 党势力最强（并列算最强）
hd_dc_strongest_party_is_trigger = {
	exists = var:hd_party_$G$_share
	var:hd_party_$G$_share >= var:hd_party_huanguan_share
	var:hd_party_$G$_share >= var:hd_party_waiqi_share
	var:hd_party_$G$_share >= var:hd_party_wuxun_share
	var:hd_party_$G$_share >= var:hd_party_shizu_share
	var:hd_party_$G$_share >= var:hd_party_hanmen_share
}

# Scope：任意。当前处于治世三期之一
hd_dc_phase_is_zhishi_trigger = {
	situation:dynastic_cycle ?= {
		OR = {
			situation_current_phase = situation_dynastic_cycle_phase_stability_expansion
			situation_current_phase = situation_dynastic_cycle_phase_stability_advancement
			situation_current_phase = situation_dynastic_cycle_phase_hd_recovery
		}
	}
}

# Scope：任意。当前处于乱世五期之一（A1/A2 口径：紧张/崩溃/斗争/对峙/新朝）
hd_dc_phase_is_luanshi_trigger = {
	situation:dynastic_cycle ?= {
		OR = {
			situation_current_phase = situation_dynastic_cycle_phase_instability
			situation_current_phase = situation_dynastic_cycle_phase_instability_conquest
			situation_current_phase = situation_dynastic_cycle_phase_hd_collapse
			situation_current_phase = situation_dynastic_cycle_phase_chaos
			situation_current_phase = situation_dynastic_cycle_phase_hd_standoff
		}
	}
}

# Scope：任意。当前阶段为 $PHASE$（短名：stability_expansion / hd_recovery / ...）
hd_dc_phase_is_trigger = {
	situation:dynastic_cycle ?= { situation_current_phase = situation_dynastic_cycle_phase_$PHASE$ }
}

# Scope：角色。文化属中华传承
hd_dc_is_chinese_heritage_trigger = {
	culture ?= { has_cultural_pillar = heritage_chinese }
}

# Scope：角色。信仰属礼教/道教/佛教组（E23）
hd_dc_is_chinese_faith_family_trigger = {
	OR = {
		religion = religion:confucianism_religion
		religion = religion:taoism_religion
		religion = religion:buddhism_religion
	}
}
