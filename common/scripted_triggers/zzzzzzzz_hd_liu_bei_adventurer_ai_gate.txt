	is_ai = yes
	government_has_flag = government_is_landless_adventurer
	domicile ?= { is_domicile_type = camp }
	# 吕布寄居徐州、刘备失徐州待投刘表期间不启用原版冒险者 AI（zz_hd_lbxz_triggers.txt，仅 184 开局）。

		NOT = { this = character:liu_bei }
		AND = {
			age >= 35
			OR = {
				exists = global_var:hd_lb_bequest_resolved
				exists = global_var:hd_lb_bequest_failed
			}
		}
	}
