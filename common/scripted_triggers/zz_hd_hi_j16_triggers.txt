# 历史入侵 × 两晋十六国统一史实线（j16）：入侵侧读取的判定（p5-C；docs/两晋十六国统一史实线/04 §2、06 §3）。
# 秦况标记 hd_j16_qin_existed / _collapse / _collapse_m / _weak / _size_383 的「设置」归 370 节点 E4-Q51、E4-W50，本文件只读。
# 开闸 hd_j16_hi_go_<KEY>（及到期兜底标记 hd_j16_hi_go_<KEY>_late）、阶段闸门 hd_j16_hi_gate_<KEY>_<N> 由 395 节点设，本文件只读。
# 凡新逻辑都以 hd_j16_hi_mode_trigger（统一心跳在运行）包裹：190、218 等非 j16 局与史实事件规则关闭的 j16 局一律走旧逻辑。

# j16 模式：统一心跳在运行（06 §3.2-1）
hd_j16_hi_mode_trigger = {
	has_global_variable = hd_j16_pulse_started
}

# 秦况模式（04 §2.1）：j16 心跳在运行，且 E4-W50 在 383.6.1 后判定确有强秦（existed）；
# 瓦解标记只由 E4-Q51 (a) 设上、existed 未设的极端情形也算
hd_j16_hi_qin_mode_trigger = {
	has_global_variable = hd_j16_pulse_started
	OR = {
		has_global_variable = hd_j16_qin_existed
		has_global_variable = hd_j16_qin_collapse
	}
}

# 秦况已开始：前秦正式瓦解，或苻坚胜淝水后身死、前秦易主仍完整（04 §2.3）
hd_hi_qin_started_trigger = {
	OR = {
		has_global_variable = hd_j16_qin_collapse
		has_global_variable = hd_j16_qin_weak
	}
}

# 秦况月计数达标：正式瓦解时 M >= $M$；只有 qin_weak 时 M >= $MW$（照 04 §2.3 每档加 12）
# 用法：hd_hi_qin_m_trigger = { M = 2 MW = 14 }
hd_hi_qin_m_trigger = {
	has_global_variable = hd_j16_qin_collapse_m
	OR = {
		AND = {
			has_global_variable = hd_j16_qin_collapse
			global_var:hd_j16_qin_collapse_m >= $M$
		}
		AND = {
			NOT = { has_global_variable = hd_j16_qin_collapse }
			has_global_variable = hd_j16_qin_weak
			global_var:hd_j16_qin_collapse_m >= $MW$
		}
	}
}

# 秦况模式下条目了结（04 §2.2-1）：计数满 120（约 10 年），或统一心跳截止（D1，421.1.1）
hd_hi_qin_end_trigger = {
	OR = {
		current_date >= 421.1.1
		AND = {
			has_global_variable = hd_j16_qin_collapse_m
			global_var:hd_j16_qin_collapse_m >= 120
		}
	}
}

# Character（本人，需 scope:hd_hi_story）：环境评估直接判「合适」（04 §2.2-3；06 §3.2-4 扩展）
# ① 前秦已正式瓦解（qin_weak 不算，04 §2.3），且 KEY 属秦况 ①–⑤（含西燕 murongchong）；慕容垂请往河北被拒（E4-M25 b）时照常评估
# ② 心跳已为本 KEY 开闸（hd_j16_hi_go_<KEY>，KEY = tufawugu、juqumengxun、helianbobo、murongde，06 §0.2）；开闸节点到窗口末兜底开闸时另设 hd_j16_hi_go_<KEY>_late，不短路（06 §3.2-3）
hd_j16_hi_env_ok_trigger = {
	hd_j16_hi_mode_trigger = yes
	exists = scope:hd_hi_story
	OR = {
		AND = {
			has_global_variable = hd_j16_qin_collapse
			OR = {
				AND = {
					scope:hd_hi_story.var:hd_hi_key ?= flag:murongchui
					NOT = { has_global_variable = hd_j16_chui_north_denied }
				}
				scope:hd_hi_story.var:hd_hi_key ?= flag:murongchong
				scope:hd_hi_story.var:hd_hi_key ?= flag:yaochang
				scope:hd_hi_story.var:hd_hi_key ?= flag:qifuguoren
				scope:hd_hi_story.var:hd_hi_key ?= flag:lvguang
			}
		}
		AND = {
			scope:hd_hi_story.var:hd_hi_key ?= flag:tufawugu
			has_global_variable = hd_j16_hi_go_tufawugu
			NOT = { has_global_variable = hd_j16_hi_go_tufawugu_late }
		}
		AND = {
			scope:hd_hi_story.var:hd_hi_key ?= flag:juqumengxun
			has_global_variable = hd_j16_hi_go_juqumengxun
			NOT = { has_global_variable = hd_j16_hi_go_juqumengxun_late }
		}
		AND = {
			scope:hd_hi_story.var:hd_hi_key ?= flag:helianbobo
			has_global_variable = hd_j16_hi_go_helianbobo
			NOT = { has_global_variable = hd_j16_hi_go_helianbobo_late }
		}
		AND = {
			scope:hd_hi_story.var:hd_hi_key ?= flag:murongde
			has_global_variable = hd_j16_hi_go_murongde
			NOT = { has_global_variable = hd_j16_hi_go_murongde_late }
		}
	}
}

# Story（入侵故事）：下一阶段的闸门未开（06 §3.2-5、§3.3）。只在 j16 模式下拦；非 j16 局永远为假

			var:hd_hi_stage = 2

			var:hd_hi_key ?= flag:helianbobo
			var:hd_hi_stage = 2

		}
		# shelun_3：E5-X50 在 404 开局是「前标」，不会设闸门，而 shelun 在 404 局仍然有效（06 §3.4）→ 404 开局视为已开
		AND = {

# Character（本人 = root，需 scope:hd_hi_crown）：j16 下王冠不篡夺、只改国号（06 §3.2-6，A6-F5）
# 受保护：持有者（≠ 本人）或其 top_liege（≠ 本人）是 hd_hi 首领。「首领」三种认法，任一即可：
#   ① 身上有 hd_hi_key（hd_hi_begin_effect 设，含已了结的）；
#   ② 身上有 hd_hi_j16_leader（hd_hi_j16_mark_leaders_effect 每年给在世的 383–421 首领设，不论是否起过兵）；
#   ③ 其王朝有 hd_hi_j16_leader_dyn（同一效果只给拓跋氏与凉州四家的王朝设）——覆盖 404 开局：拓跋珪从没 begin、
#      409 年后由拓跋嗣继位，北魏封臣拓跋遵持并州 k_uuii_bingzhou（06 §4.3）时夏仍不篡夺；凉州四家的继位者之间也不互夺。
# 前秦苻氏、东晋司马氏不在名单里：秦况各家照旧从苻坚手里篡夺王冠（如西燕取 k_yongzhou）。
hd_j16_hi_crown_protected_trigger = {
	hd_j16_hi_mode_trigger = yes
	scope:hd_hi_crown.holder ?= {
		NOT = { this = root }
		OR = {
			hd_hi_j16_is_leader_trigger = yes
			top_liege ?= {
				NOT = { this = root }
				hd_hi_j16_is_leader_trigger = yes
			}
		}
	}
}

# Character：是 hd_hi 首领或其王朝成员（见上）
hd_hi_j16_is_leader_trigger = {
	OR = {
		has_variable = hd_hi_key
		has_variable = hd_hi_j16_leader
		dynasty ?= { has_variable = hd_hi_j16_leader_dyn }
	}
}
