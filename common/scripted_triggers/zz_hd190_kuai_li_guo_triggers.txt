# 蒯越献策、平定江南
############################################################

# 角色作用域：仍可由刘表收服的长沙、南平、襄阳、南乡之主。
# 有地、不是刘表本人；独立，或已在刘表领内（被第三方收编的不动）。
hd190_kuai_target_valid_trigger = {
	is_alive = yes
	is_landed = yes
	character:liu_biao ?= { is_alive = yes is_landed = yes }
	NOT = { this = character:liu_biao }
	OR = {
		is_independent_ruler = yes
		is_vassal_or_below_of = character:liu_biao
	}
}

############################################################
# 李傕郭汜相攻
############################################################

# 县作用域：持有人是李傕本人或其封臣（scope:hd190_lg_li）。
hd190_lg_county_in_li_realm_trigger = {
	exists = holder
	holder = {
		OR = {
			this = scope:hd190_lg_li
			is_vassal_or_below_of = scope:hd190_lg_li
		}
	}
}

# 参数 DUCHY：李傕领内持有该郡法理下至少一半的县。
hd190_lg_majority_trigger = {
	$DUCHY$ = {
		any_de_jure_county = {
			percent >= 0.5
			hd190_lg_county_in_li_realm_trigger = yes
		}
	}
}

# 可以分裂：李傕已攻破长安、在世有地且独立；郭汜在世，且不是李傕领外的有地统治者。
hd190_li_guo_ready_trigger = {
	exists = global_var:hd_hev_184_changan_fall
	character:li_jue ?= { is_alive = yes is_landed = yes is_independent_ruler = yes }
	character:guo_si ?= {
		is_alive = yes
		OR = {
			is_landed = no
			hd_dong_inside_realm_trigger = { RULER = character:li_jue }
		}
	}
}
