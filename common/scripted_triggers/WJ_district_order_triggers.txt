WJ_dd_order_range_trigger = {
 custom_description = { text = WJ_failure_14
OR = {
 WJ_zw_law_range = yes
 AND = {
 scope:actor = { WJ_dd_head_trigger = yes vassal_contract_obligation_level:WJ_sanjie > 0 NOT = { primary_title = { has_variable = wj_dd_grant_pending } } }

  is_alive = yes this != scope:actor
  WJ_contract_sanjie_visible_trigger = yes
  WJ_dd_member_of_core_trigger = { CORE = scope:actor.primary_title }
  top_liege = scope:actor.top_liege
 }
 # Reject only a rebellious commander ordering a loyal official.
 NAND = {
  scope:actor = { is_at_war_with = top_liege }
  scope:recipient = { NOT = { is_at_war_with = top_liege } }
 }
 }
 }
}
}
