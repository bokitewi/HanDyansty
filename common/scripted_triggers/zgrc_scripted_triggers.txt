# 亡国级战争判定（War 作用域）—— 天下响应判定③ 之 S2 路
# 此类 CB 不瞄准具体头衔（针对整个国家），只能按 CB 白名单识别
zgrc_luanshi_annihilation_cb_trigger = {
	exists = casus_belli
	OR = {
		# 异族 / 游牧入侵
		using_cb = invasion_war
		using_cb = clan_invasion_war
		using_cb = mpo_nomad_invasion_cb
		# 蒙古灭国（打王国 / 打帝国两版）
		using_cb = mongol_invasion_war
		using_cb = mongol_realm_invasion_war
		using_cb = mpo_gok_onslaught_cb
		# 无地冒险者夺国
		using_cb = ep3_laamp_adventurer_empire_invasion
		# 历史剧本入侵（诺曼 / 挪威）
		using_cb = norman_conquest_cb
		using_cb = norwegian_invasion_cb
		# 大圣战（异教征服）
		using_cb = undirected_great_holy_war
		using_cb = directed_great_holy_war
		using_cb = major_religious_war
		# 天朝征伐（TGP）
		using_cb = china_hegemon_seize_duchy_cb
		# 帝国政治战
		using_cb = imperial_reconquest_cb
		using_cb = dissolve_empire_war
		using_cb = tribal_subjugation_cb
	}
}

# 蛮夷判定：① 自身非华夏礼乐 ② 最高领主亦非 ③ 非（通汉语 且 信公羊学）
zgrc_is_manyi_character_trigger = {
	culture = {
		NOT = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
	}
	NOT = {
		top_liege = {
			culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
		}
	}
	NOT = {
		AND = {
			knows_language = language_chinese
			rite = { rite_has_tenet = tenet_zgrc_zunwangrangyi }
		}
	}
}

# 「广张教化」CB 授权：领主为华夏礼乐，或通汉语受儒学者；且奉公羊学
zgrc_can_use_jiaohua_cb_trigger = {
	OR = {
		top_liege = {
			culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
		}
		AND = {
			knows_language = language_chinese
			has_trait = confucian_education
		}
	}
	rite = { rite_has_parameter = zgrc_jiaohua_cb_enabled }
}

# 大复仇资格：有此个人信条、成年、且有立足之地（有地或冒险者）
zgrc_can_revenge_trigger = {
	is_alive = yes
	is_adult = yes
	has_personal_tenet_flag = zgrc_dafuchou_enabled
	OR = {
		is_ruler = yes
		government_has_flag = government_is_landless_adventurer
	}
}
