# AI-only admission and letter guards. Player interactions keep their own options.
WJ_ai_offer_possible = {
 OR = { scope:actor.gold >= 300 scope:recipient = { NOT = { has_hook = scope:actor } } }
}
WJ_ai_recommender_xiaolian = { primary_title.tier = tier_duchy }
WJ_ai_recommender_maocai = { primary_title.tier >= tier_kingdom WJ_langguan_real_emperor_trigger = no }
WJ_ai_offer_valid = {
 OR = {
  AND = { scope:wj_ai_offer = 1 scope:actor.gold >= 300 }
  AND = { scope:wj_ai_offer = 2 scope:recipient = { NOT = { has_hook = scope:actor } } }
 }
}

 scope:recipient = {
  is_pool_guest = no
  OR = { AND = { exists = liege liege = scope:actor } AND = { is_courtier = yes exists = employer employer = scope:actor } }
 }
}

 NOR = {
  has_variable = wj_ai_bizhao_pending
  has_variable = wj_ai_bizhao_month
  has_variable = wj_yangwang_mufu_pending
  has_pending_interaction_of_type = wj_request_bizhao_interaction
 }
}

}

  var:wj_ai_bizhao_serial = scope:wj_ai_serial

}
# RECOMMENDER qualification is checked on the recipient; the candidate is always actor.
WJ_ai_nomination_valid = {
 scope:actor = {
  is_ai = yes is_alive = yes is_male = yes government_has_flag = government_is_celestial
  NOT = { is_at_war_with = scope:recipient }
  WJ_can_be_recommended_$ROAD$_candidate_trigger = { RECOMMENDER = scope:recipient }
  OR = {
   any_liege_or_above = { this = scope:recipient }
   AND = { exists = employer employer = { OR = { this = scope:recipient any_liege_or_above = { this = scope:recipient } } } }
  }
 }
 scope:recipient = {
  is_alive = yes is_landed = yes government_has_flag = government_is_celestial
  is_imprisoned = no is_incapable = no
  WJ_ai_recommender_$ROAD$ = yes
  NOT = { has_character_flag = wj_$ROAD$_nomination_yearly_quota }
 }
 trigger_if = { limit = { scope:recipient = { is_ai = no } } WJ_ai_offer_possible = yes }
}
WJ_ai_nomination_reply_valid = {
 WJ_ai_nomination_valid = { ROAD = $ROAD$ }
 WJ_ai_offer_valid = yes
 scope:recipient = {
  var:wj_ai_$ROAD$_requester = scope:actor
  var:wj_ai_$ROAD$_serial = scope:wj_ai_nomination_serial
  has_character_flag = wj_$ROAD$_nomination_pending
 }
}
