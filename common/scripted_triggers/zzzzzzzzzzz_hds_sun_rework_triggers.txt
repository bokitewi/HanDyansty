# 2026-10-10 孙家线重制：新触发器（只放新键）。

# 角色作用域：当前是接续孙策的孙权（global_var:hds_sq_leader）。
hds_sq_leader_trigger = {
	trigger_if = {
		limit = { has_global_variable = hds_sq_leader }
		this = global_var:hds_sq_leader
	}
	trigger_else = { always = no }
}

# 任意作用域：ACTOR 持有丹阳郡，且丹阳每个法理县都在 ACTOR 本人或其封臣手里。
hds_sun_danyang_control_trigger = {
	$ACTOR$ = {
		is_alive = yes
		is_landed = yes
		has_title = title:d_danyang
	}
	title:d_danyang = {
		any_de_jure_county = { always = yes }
		NOT = {
			any_de_jure_county = {
				NOT = {
					AND = {
						exists = holder
						holder ?= {
							is_alive = yes
							OR = {
								this = $ACTOR$
								is_vassal_or_below_of = $ACTOR$
							}
						}
					}
				}
			}
		}
	}
}
