# County scope, shared by real spawn effects and the debug scene.
hd_dc_earthquake_county_eligible_trigger = {
	development_level >= 2
	title_province = { geographical_region = world_asia_china }
	NOT = { any_county_situation = { situation_type = natural_disaster_earthquake } }
}

hd_dc_flood_county_eligible_trigger = {
	development_level >= 2
	title_province = { geographical_region = global_flood_region }
	save_temporary_scope_as = hd_dc_disaster_county
	any_in_global_list = {
		variable = flood_region_list
		save_temporary_scope_as = hd_dc_disaster_river
		scope:hd_dc_disaster_county.title_province = { geographical_region = scope:hd_dc_disaster_river }
		NOT = {
			any_in_global_list = {
				variable = ongoing_floods
				var:river_region ?= scope:hd_dc_disaster_river
			}
		}
	}
}
