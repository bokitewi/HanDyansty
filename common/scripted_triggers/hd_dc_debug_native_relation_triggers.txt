# Exact global copies of the native event-local eligibility blocks. The engine
# forbids calling those inline names outside their declaring event files.
hd_dc_debug_adultery_unfaithful_trigger = {
	save_temporary_scope_as = unfaithful
	OR = {
		any_relation = {
			type = lover
			NOR = {
				any_consort = { this = scope:unfaithful }
				has_relation_soulmate = scope:unfaithful
			}
		}
		any_secret = {
			type = secret_lover
			secret_target = {
				NOR = {
					any_consort = { this = scope:unfaithful }
					has_relation_soulmate = scope:unfaithful
				}
			}
		}
		trigger_if = {
			limit = { exists = var:had_recent_sex_with }
			var:had_recent_sex_with ?= {
				NOR = {
					any_consort = { this = scope:unfaithful }
					has_relation_soulmate = scope:unfaithful
				}
			}
		}
	}
}

hd_dc_debug_adultery_valid_partner_trigger = {
	is_physically_able_adult = yes
	NOT = { has_character_flag = was_recently_suspected_of_adultery }
}

hd_dc_debug_yearly_1010_abductee_trigger = {
	basic_is_available_ai = yes
	is_adult = yes
	NOT = { has_trait = wounded_3 }
	OR = {
		is_spouse_of = root
		is_concubine_of = root
		is_child_of = root
		is_sibling_of = root
		is_parent_of = root
		any_relation = { type = lover this = root }
	}
	NOT = { has_relation_rival = root }
}

hd_dc_debug_yearly_1010_abductor_trigger = {
	save_temporary_scope_as = potential_abductor
	basic_is_available_ai = yes
	is_adult = yes
	NOT = { has_trait = wounded_3 }
	OR = {
		is_landed = yes
		AND = {
			exists = liege
			is_player_heir_of = scope:potential_abductor.liege
		}
	}
	exists = liege
	liege != root
}
