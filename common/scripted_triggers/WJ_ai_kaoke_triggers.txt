# Only an actual kingdom superior changes the request route; central manual scope stays national.
WJ_kaoke_under_king_trigger = {
 highest_held_title_tier < tier_kingdom
 any_liege_or_above = { highest_held_title_tier = tier_kingdom }
}
WJ_kaoke_request_reviewer_trigger = {
 OR = {
  AND = {
   scope:actor = { WJ_kaoke_under_king_trigger = yes any_liege_or_above = { this = scope:recipient } }
   scope:recipient = { WJ_kaoke_eligible_trigger = yes highest_held_title_tier = tier_kingdom is_incapable = no is_imprisoned = no }
  }
  AND = {
   scope:actor = { NOT = { WJ_kaoke_under_king_trigger = yes } }
   scope:recipient = { has_council_position = minister_personnel is_councillor_of = scope:actor.top_liege }
  }
 }
}
WJ_ai_kaoke_request_valid = {
 WJ_kaoke_request_participants_trigger = yes WJ_kaoke_request_period_trigger = yes
 scope:actor = { is_ai = yes }
}
WJ_ai_kaoke_reply_current = {
 exists = scope:actor exists = scope:recipient
 scope:actor = {
  var:wj_ai_kaoke_serial = scope:wj_ai_kaoke_serial
  var:wj_ai_kaoke_pending = scope:recipient
  var:wj_kaoke_window_epoch = scope:wj_kaoke_request_epoch
  var:wj_kaoke_start = scope:wj_kaoke_request_start
 }
}
WJ_ai_kaoke_reply_valid = {
 WJ_ai_kaoke_reply_current = yes WJ_ai_kaoke_request_valid = yes
 scope:actor = {
  top_liege = scope:wj_kaoke_request_emperor
  var:wj_kaoke_realm = scope:wj_kaoke_request_realm
  liege = scope:wj_kaoke_request_liege
 }
}
WJ_ai_kaoke_award_valid = {
 scope:actor = {
  is_ai = yes WJ_kaoke_table_viewer_trigger = yes
  NOT = { has_variable = wj_ai_kaoke_award_year }
  OR = { this = top_liege AND = { has_council_position = minister_personnel is_councillor_of = top_liege } }
 }
 scope:recipient = {
  WJ_kaoke_eligible_trigger = yes top_liege = scope:actor.top_liege
  has_variable = wj_kaoke_start
  var:wj_kaoke_realm = top_liege.primary_title
  WJ_kaoke_days_value < 1825
  NOT = { has_variable = wj_kaoke_central_request_used }
  OR = {
   AND = { exists = dynasty exists = scope:actor.dynasty dynasty = scope:actor.dynasty }
   scope:actor = { has_relation_disciple = prev }
   AND = { exists = var:wj_mufu_owner var:wj_mufu_owner = scope:actor }
  }
 }
}
