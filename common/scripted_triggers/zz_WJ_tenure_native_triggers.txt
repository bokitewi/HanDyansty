# Native 1.20 helpers with their original conditions retained. Only routine AI
# dismissal and the chancellor's non-dismissable office are added.
is_blocked_from_being_fired_from_council_trigger = {
 custom_description = {
  text = "is_blocked_from_being_fired_from_council"
  subject = this object = $COURT_OWNER$
  OR = {
   NOT = { WJ_chancellor_council_change_allowed = yes }
   council_task ?= { can_fire_position = no }
   var:block_fire_councillor ?= $COURT_OWNER$
   AND = {
    has_character_modifier = WJ_tenure_modifier
    $COURT_OWNER$ = { is_ai = yes }
    OR = { NOT = { exists = scope:actor } scope:actor = { is_ai = yes } }
   }
  }
 }
}
title_revocation_standard_valid_showing_failures_only_trigger = {
 scope:recipient = { WJ_tenure_can_dismiss = { ACTOR = scope:actor } }
 scope:puppet_or_actor = {
  NOR = { is_at_war_with = scope:recipient has_truce = scope:recipient }
  custom_description = {
   text = "character_interactions_revoke_title_from_ally_without_reason"
   object = scope:recipient
   OR = { NOT = { is_allied_to = scope:recipient } has_revoke_title_reason = scope:recipient }
  }
 }
 scope:recipient = { NOT = { has_strong_hook = scope:puppet_or_actor } }
}
