# 雇主scope。只有已定名表命中的内阁才覆盖；其余沿原版，不猜名称。
WJ_council_name_override_trigger = {
 is_alive = yes government_has_flag = government_is_celestial
 OR = {
  primary_title ?= { has_variable = hd218_shu_chengxiang } # 218 蜀汉丞相
  AND = { WJ_cr_emperor = yes highest_held_title_tier = tier_empire }
  tgp_has_access_to_ministry_trigger = yes
  WJ_name_family_head_trigger = yes
  
  AND = { has_title = title:e_WJ_minister_cheng_xiang has_council_position = WJ_minister_cheng_xiang }
  AND = { has_title = title:e_minister_chancellor has_council_position = councillor_chancellor }
  AND = { has_title = title:e_minister_censor has_council_position = councillor_spymaster }
  AND = { has_title = title:e_minister_of_war has_council_position = councillor_marshal }
  AND = { has_title = title:e_minister_grand_marshal has_council_position = minister_grand_marshal }
  
  AND = { has_title = title:e_minister_of_rites has_council_position = councillor_court_chaplain }
  AND = { has_title = title:e_minister_of_justice has_council_position = minister_justice }
  AND = { has_title = title:e_minister_of_works has_council_position = minister_works }
  AND = { has_title = title:e_minister_of_revenue has_council_position = councillor_steward }
  
  WJ_council_local_ruler_trigger = yes
 }
}

# Display only: a landed local ruler keeps the county/commandery/province cabinet, including independent rulers.
WJ_council_local_ruler_trigger = {
 is_alive = yes
 government_has_flag = government_is_celestial
 is_landed = yes
 highest_held_title_tier >= tier_county
 highest_held_title_tier <= tier_kingdom
}
