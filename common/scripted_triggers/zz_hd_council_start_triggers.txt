# 开局议会预填闸门：引擎在开局 setup 阶段会给所有空的议会席位生成平民填满。
# 新开局时，在 after_lobby 开闸前只放行对应历史官署的持有人；不预填随机平民。
# 旧存档不带该变量，但已过开局一个月以上，照常放行。
# 另：被人口清除排期的人（hd_popclean_marked）不得入席，免得 AI 把待清的人任命进去。
hd_council_start_open_trigger = {
	NOT = { has_character_flag = hd_popclean_marked }
	OR = {
		exists = global_var:hd_council_start_open
		months_from_game_start >= 1
		AND = {
			game_start_date = 190.1.1
			current_date = 190.1.1
			has_title = title:$MINISTER_TITLE$
			exists = liege_or_court_owner
			liege_or_court_owner = title:h_china.holder
		}
	}
}
