# One personal tenure, bound to a real office and its court, never inherited.
WJ_tenure_emperor = {
 is_alive = yes government_has_flag = government_is_celestial
 OR = { has_title = title:h_china WJ_cr_emperor = yes }
}
# Title scope. Peerages, noble families and honorary offices are not appointments.
WJ_tenure_office_title = {
 is_noble_family_title = no
 NOT = { has_variable = wj_rf_rank }
 NOT = { has_variable = SanGong_DongGong_title }
 OR = {
  tgp_is_minister_title = yes
  AND = { tier >= tier_county tier <= tier_kingdom is_landless_type_title = no }
 }
}
WJ_tenure_free = {
 custom_tooltip = { text = WJ_tenure_release_first_tt NOT = { has_character_modifier = WJ_tenure_modifier } }
}
# Explicit player dismissal is still permitted; AI daily removal is not.
WJ_tenure_can_dismiss = {
 OR = { $ACTOR$ = { is_ai = no } WJ_tenure_free = yes }
}
WJ_tenure_job_valid = {
 is_alive = yes government_has_flag = government_is_celestial
 exists = var:wj_tenure_realm.holder
 top_liege = var:wj_tenure_realm.holder
 OR = {
  AND = { exists = var:wj_tenure_office has_title = var:wj_tenure_office }
  AND = {
   has_variable = wj_tenure_council
   is_councillor_of = var:wj_tenure_realm.holder
   council_task = var:wj_tenure_council
  }
 }
}
WJ_tenure_can_begin = {
 is_alive = yes government_has_flag = government_is_celestial WJ_tenure_free = yes
 top_liege = { WJ_tenure_emperor = yes }
 OR = {
  AND = { exists = scope:wj_tenure_office scope:wj_tenure_office = { WJ_tenure_office_title = yes } has_title = scope:wj_tenure_office }
  AND = { exists = scope:wj_tenure_council is_councillor_of = top_liege council_task = scope:wj_tenure_council }
 }
}
# Event scopes are captured from the successful native appointment, not searched globally.
WJ_tenure_offer_valid = {
 is_ai = no WJ_tenure_emperor = yes
 NOT = { has_character_flag = wj_tenure_prompts_off }
 exists = scope:wj_tenure_appointee
 scope:wj_tenure_appointee = {
  is_alive = yes government_has_flag = government_is_celestial
  this != $RULER$ top_liege = $RULER$ WJ_tenure_free = yes
  OR = {
   AND = { exists = scope:wj_tenure_office has_title = scope:wj_tenure_office }
   AND = {
    exists = scope:wj_tenure_council
    is_councillor_of = $RULER$ council_task = scope:wj_tenure_council
   }
  }
 }
}
