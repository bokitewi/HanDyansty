# Manual emperor actions only. Recommendations use the chancellor guard separately.
WJ_regency_authority_three = {
 custom_tooltip = {
  text = WJ_regency_authority_three_tt
  OR = { has_realm_law = celestial_bureaucracy_2 has_realm_law = celestial_bureaucracy_3 }
 }
}

# Character scope. Ordinary appointment never moves or replaces the chancellor.
# Do not use this as validity for his own occupied seat.
WJ_chancellor_council_change_allowed = {
 custom_tooltip = {
  text = WJ_chancellor_council_protected_tt
  NOR = {
   has_title = title:e_WJ_minister_cheng_xiang
   has_council_position = WJ_minister_cheng_xiang
  }
 }
}

# Character scope = the incumbent/candidate being dismissed, replaced or moved.
WJ_regency_manual_councillor_change_allowed = {
 WJ_chancellor_council_change_allowed = yes
 custom_tooltip = {
  text = WJ_regency_protected_office_tt
  trigger_if = {
   limit = {
    $OWNER$ = { government_has_flag = government_is_celestial WJ_name_actual_emperor_trigger = yes }
    is_alive = yes
    is_councillor_of = $OWNER$
   }
   $OWNER$ = {
    WJ_regency_authority_three = yes
    NOT = { has_trait = WJ_lost_imperial_authority }
   }
  }
 }
}

# Native invitations can wait for a player's answer: recheck both posts on acceptance too.
WJ_regency_manual_invitation_allowed = {
 scope:recipient = { WJ_regency_manual_councillor_change_allowed = { OWNER = scope:actor } }
 trigger_if = {
  limit = { exists = scope:target.councillor }
  scope:target.councillor = { WJ_regency_manual_councillor_change_allowed = { OWNER = scope:actor } }
 }
}
