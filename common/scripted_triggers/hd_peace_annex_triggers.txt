# 和约：普通吞并与法理已占领县（2026-10-29 boki）
# - 普通吞并（吞并 CB 且不带史实情境 hd_tb_kind）不再强行全取：守方全境可在和约中索取，索地价格 ×0.5（与已占领半价叠加），
#   不提供「完整战争目标」条款。带史实情境的吞并战照旧专用结算。
# - 所有走和约的战争：自己法理内（县的法理公国、王国或帝国由提案人持有，霸权头衔不算）且已被己方占领的县价格为 0，
#   不受宣战目标限制可强制索取；玩家只索要这类县时不看战分即可强制签署（AI 不用这条，只按零价报价）。

# 战争 scope
hd_peace_ordinary_annex_war_trigger = {
	using_cb = hd_dc_tunbing_cb
	NOT = { has_variable = hd_tb_kind }
}

# 县 scope，ROOT = 提案人
hd_peace_free_dejure_county_trigger = {
	tier = tier_county
	hd_peace_county_occupied_by_proposer_side_trigger = yes
	any_this_title_or_de_jure_above = {
		tier >= tier_duchy
		tier <= tier_empire
		holder ?= root
	}
}

# 县 scope，ROOT = 提案人：普通吞并战里守方的县（和约打开时缓存进 hd_peace_cb_reserved_counties）
hd_peace_annex_discount_county_trigger = {
	exists = root.var:hd_peace_treaty_war
	root.var:hd_peace_treaty_war = { hd_peace_ordinary_annex_war_trigger = yes }
	root = { is_target_in_variable_list = { name = hd_peace_cb_reserved_counties target = prev } }
}

# 提案人 scope：所索要的县与整片头衔全部是法理已占领的零价地（不要求有索地）
hd_peace_all_demanded_land_free_trigger = {
	NOT = {
		any_in_list = {
			variable = hd_peace_demand_counties
			NOT = { hd_peace_free_dejure_county_trigger = yes }
		}
	}
	NOT = {
		any_in_list = {
			variable = hd_peace_demand_titles
			any_de_jure_county = {
				hd_peace_county_in_opponent_realm_trigger = yes
				NOT = { hd_peace_free_dejure_county_trigger = yes }
			}
		}
	}
}

# 提案人 scope：有索地，且全部是法理已占领的零价地
hd_peace_free_land_only_demand_trigger = {

		hd_peace_demand_county_count_value > 0
		hd_peace_demand_title_count_value > 0
	}
	hd_peace_all_demanded_land_free_trigger = yes
}
