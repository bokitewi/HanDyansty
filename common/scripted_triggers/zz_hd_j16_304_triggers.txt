# 天子的朝贡国，或天子的有地封臣（02 §1.2 is_shown ④）。Scope：角色

	exists = title:h_china.holder
	is_landed = yes
	OR = {
		is_tributary_of_suzerain_or_above = title:h_china.holder
		liege ?= title:h_china.holder
	}
}

# 县 $COUNTY$ 在 $WHO$ 一方：持有者是 $WHO$、或其最高领主是 $WHO$、或与 $WHO$ 同一最高领主。$WHO$ 已死为假。Scope：任意
hd_j16_county_on_side_trigger = {
	$WHO$ ?= { is_alive = yes }
	$COUNTY$.holder ?= {
		OR = {
			this = $WHO$
			top_liege = $WHO$
			top_liege = $WHO$.top_liege
		}
	}
}
