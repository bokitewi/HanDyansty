# Append once, after the complete native monthly ultimatum chance. Creation
# scores are later changed by engine personality code and are NOT probabilities.
hd_dc_faction_demand_hazard_modifier = {
	modifier = {
		exists = faction_target
		faction_target.hd_dc_faction_hazard_multiplier_value > 1
		add = {
			if = {
				limit = {
					save_temporary_scope_value_as = { name = hd_dc_hazard_raw value = $BASE$ }
					save_temporary_scope_value_as = { name = hd_dc_hazard_base value = { value = scope:hd_dc_hazard_raw min = 0 max = 100 } }
					save_temporary_scope_value_as = { name = hd_dc_hazard_multiplier value = faction_target.hd_dc_faction_hazard_multiplier_value }
				}
			}
			value = hd_dc_hazard_probability_value
			subtract = scope:hd_dc_hazard_raw
		}
	}
}
