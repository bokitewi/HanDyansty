# Append after all original native modifiers at an initiation probability.
# Keep their original engine evaluation, then add f(clamp(raw)) - raw once.
# F=1 leaves the original block completely untouched. Not a target-pool weight.
hd_dc_scheme_hazard_modifier = {
	modifier = {
		$ACTOR$.hd_dc_scheme_hazard_multiplier_value > 1
		add = {
			if = {
				limit = { hd_dc_scheme_hazard_inputs_trigger = { BASE = $BASE$ ACTOR = $ACTOR$ } }
			}
			value = hd_dc_hazard_probability_value
			subtract = scope:hd_dc_hazard_raw
		}
	}
}
