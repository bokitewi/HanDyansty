# 天朝循环大修：崩溃期"军阀称帝"派系。只由危机脚本创建，玩家/AI 不能自行创建或加入。
# 借鉴 All Under Heaven Crisis Overhaul 的 auhco_warlord_revolt_faction（已获授权）。
hd_collapse_warlord_faction = {
	casus_belli = hd_collapse_warlord_war

	short_effect_desc = hd_collapse_warlord_faction_desc

	sort_order = 1

	show_special_title = yes
	name = hd_collapse_warlord_faction

	claimant = yes

	is_shown = { always = no }

	discontent_progress = { base = 0 }

	power_threshold = { base = 80 }

	is_valid = {
		trigger_if = {
			limit = {
				OR = {
					exists = special_character
					exists = faction_war
				}
			}
			special_title.holder ?= faction_target
			OR = {
				special_character ?= {
					is_alive = yes
					NOR = {
						has_trait = incapable
						this = root.faction_target
					}
				}
				exists = faction_war
			}
		}
	}

	demand = { faction_start_war = { title = title:h_china } }

	ai_demand_chance = { base = 0 }

	is_character_valid = {
		is_alive = yes
		is_landed = yes
		OR = {
			scope:faction = { faction_is_at_war = yes }
			AND = {
				hd_collapse_warlord_member_trigger = yes
				liege = scope:faction.faction_target
			}
		}
	}

	can_character_create_ui = { always = no }

	can_character_create = {
		liege = scope:target
		scope:target = { has_character_flag = hd_collapse_warlord_assembling }
	}

	can_character_join = {
		liege = scope:faction.faction_target
		scope:faction = { has_variable = hd_collapse_warlord_assembling }
	}

	can_character_become_leader = {
		is_alive = yes
		is_adult = yes
		is_imprisoned = no
		is_landed = yes
	}

	ai_create_score = { base = -100000 }

	ai_join_score = { base = -100000 }

	is_county_valid = {
		scope:faction.faction_target = holder
	}

	can_county_join = {
		exists = scope:faction.special_title
		target_is_de_jure_liege_or_above = scope:faction.special_title
	}

	county_join_score = { base = -100000 }

	on_creation = { set_variable = { name = hd_collapse_warlord_assembling value = yes } }

	county_allow_join = no
	county_allow_create = no
	county_power = county_levies_to_raise

	character_allow_create = yes

	character_allow_join = yes

	multiple_targeting = yes

	special_character_title = "FACTIONS_WINDOW_CLAIMANT"

	ignore_soft_block = yes
}
