# Vanilla 1.20.1.1 01_grant_title_rules.txt:1-28, with one pre-grant check.
# ROOT is recipient; liege is granter; landed_title is the specific title.
allowed_to_be_granted_titles_by = {
	WJ_ministry_target_acceptance_trigger = { TITLE = scope:landed_title }
	trigger_if = {
		limit = {
			scope:landed_title = {
				tier <= tier_county
				title_province = { has_holding_type = church_holding }
			}
			scope:liege.rite = {
				NOT = { rite_has_parameter = lay_clergy }
			}
		}
		custom_tooltip = {
			text = grant_title_clerical_gender_tt
			can_be_granted_theocratic_titles_by = { RULER = scope:liege }
		}
	}
	trigger_else = {
		can_be_granted_titles_by = { RULER = scope:liege }
	}
}
