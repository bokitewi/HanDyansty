# HD 地域军制兵：幽州突骑（技术原型）
# 军制由"组建幽州突骑"决议一次性建立（hd_established_youzhou_tuqi 旗标，只传主继承人）；
# 此后只要仍掌握幽州有效兵源（hd_valid_youzhou_tuqi_source_trigger）即可招募与扩编。
# 规模上限为静态 max = 6，失地后由 can_recruit 失效拦截新建与扩编，已有军团保留。

@hd_youzhou_tuqi_low_maintenance_cost = 0.45

hd_youzhou_tuqi = {
	type = light_cavalry

	damage = 32
	toughness = 18
	pursuit = 40
	screen = 32

	terrain_bonus = {
		plains = { damage = 10 pursuit = 8 }
		drylands = { damage = 6 }
		mountains = { damage = -8 pursuit = -12 }
	}

	counters = {
		archers = 1
		skirmishers = 1
	}

		hd_can_recruit_youzhou_tuqi_trigger = yes
	}

	max_regiments = 2

	buy_cost = {
		gold = @hd_youzhou_tuqi_buy_cost
	}
	low_maintenance_cost = {
		gold = @hd_youzhou_tuqi_low_maintenance_cost
	}
	high_maintenance_cost = {
		gold = @hd_youzhou_tuqi_high_maintenance_cost
	}

			should_use_asian_maa_graphics = yes

}
