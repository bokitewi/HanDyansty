@admin_law_cooldown_value = 5

# 官制法律 - 格式对齐原版 realm laws
province_title_flavour_law = {
	default = han_province_title_law
	cumulative = no
	flag = realm_law

	han_province_title_law = {
		flag = province_title_flavour_law
		flag = han_province_title
		flag = hanzhi

		modifier = {
			courtly_opinion = 5 # Pro-hegemon movement
			glory_hound_opinion = 5 # Conservative movement
			barons_and_minor_landholders_opinion = 5
			celestial_government_tax_contribution_add = 0.05
		}
		can_keep = {
			realm_law_use_celestial_bureaucracy = yes
			trigger_if = {
				limit = {
					is_independent_ruler = no
				}
				liege = { has_realm_law = han_province_title_law }
			}
		}

		can_pass = {
			trigger_if = {
				limit = {
					is_independent_ruler = no
					liege = { has_realm_law = celestial_bureaucracy_0 }
				}
				custom_tooltip = {
					text = imperial_bureaucracy_pending_law_change
					always = no
				}
			}
			custom_tooltip = {
				text = admin_law_cooldown_desc
				NOT = { has_variable = admin_law_cooldown }
			}
		}

		pass_cost = {
			prestige = {
				if = {
					limit = {
						NOR = {
							has_realm_law = celestial_bureaucracy_2
							has_realm_law = celestial_bureaucracy_3
						}
					}
					add = increase_crown_authority_prestige_cost
				}
			}
		}

		on_pass = {
			## Set cooldown
			set_variable = {
				name = admin_law_cooldown
				years = @admin_law_cooldown_value
			}
		}

		ai_will_do = {
			value = 1
		}
	}
