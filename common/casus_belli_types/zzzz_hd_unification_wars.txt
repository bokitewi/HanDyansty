# Late, object-level override. Do NOT delete 10_tgp_china_wars.txt.
# chinese_consolidation_cb (the existing plunder-style CB) is NOT redefined here.

chinese_reunification_cb = {
	icon = celestial
	group = celestial
	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		hd_unification_celestial_attacker_trigger = yes
	}
	allowed_against_character = {
		scope:defender = { hd_unification_realm_in_china_trigger = yes }
		scope:attacker = { top_liege != scope:defender.top_liege }
	}
	allowed_for_character_display_regardless = { }
	allowed_against_character_display_regardless = { }

	target_title_tier = kingdom
	target_titles = all
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder
	ai_only_against_neighbors = no
	ai_can_target_all_titles = { always = yes }
	ai_score_mult = { value = 20 }

	valid_to_start = {
		scope:attacker = { hd_unification_celestial_attacker_trigger = yes }
		scope:defender = { hd_unification_realm_in_china_trigger = yes }
		exists = scope:target
		scope:target = {
			tier = tier_kingdom
			exists = hegemony
			hegemony = title:h_china
			hd_unification_target_has_defender_land_trigger = yes
		}
		hd_unification_ai_cb_start_allowed_trigger = {
			ATTACKER = scope:attacker
			DEFENDER = scope:defender
		}
	}

	# Only structural validity. Changing phase, military strength, currency,
	# AI cooldown or war count during a war must NOT invalidate that war.
	should_invalidate = {
		OR = {
			scope:attacker = { NOT = { hd_unification_celestial_attacker_trigger = yes } }
			NOT = {
				any_in_list = {
					list = target_titles
					exists = hegemony
					hegemony = title:h_china
					hd_unification_target_has_defender_land_trigger = yes
				}
			}
		}
	}
	on_invalidated_desc = HD_UNIFICATION_REGIONAL_INVALIDATED
	on_invalidated = { }

	# Preserve the existing conquering CB's normal costs.
	cost = {
		piety = {
			add = common_cb_impious_piety_cost
			if = {
				limit = { scope:defender.faith = scope:attacker.faith }
				add = { value = 100 desc = CB_SAME_FAITH_COST }
				add = {
					value = 10
					multiply = scope:attacker.realm_size
					desc = CB_REALM_SIZE_COST
					max = 800
				}
			}
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = 0
			if = {
				limit = { NOT = { scope:defender.faith = scope:attacker.faith } }
				add = { value = 150 desc = CB_COST_DIFFERENT_FAITH }
				add = {
					value = 10
					multiply = scope:attacker.realm_size
					desc = CB_REALM_SIZE_COST
					max = 800
				}
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}
	on_declaration = {
		on_declared_war = yes
		scope:attacker = { hd_unification_ai_set_cooldown_effect = yes }
	}
	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = invasion_war_victory_desc_attacker
			}
			desc = invasion_war_victory_desc
		}
	}
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		add_influence_attacker_victory_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_high_effect = yes }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		show_as_tooltip = {
			every_in_list = {
				list = target_titles
				custom_tooltip = invasion_title_transfer_tt
				custom_tooltip = invasion_title_transfer_2_tt
				every_in_de_jure_hierarchy = {
					limit = { holder.top_liege = scope:defender }
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
			scope:defender = {
				every_sub_realm_county = {
					limit = { county_controller = scope:attacker }
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
		}
		hidden_effect = {
			setup_invasion_cb = {
				titles = target_titles
				attacker = scope:attacker
				defender = scope:defender
				change = scope:change
				take_occupied = yes
			}
			resolve_title_and_vassal_change = scope:change
		}
		modify_all_participants_fame_values = {
			WINNER = scope:attacker
			LOSER = scope:defender
			FAME_BASE = scope:cb_prestige_factor
			IS_RELIGIOUS_WAR = no
			WINNER_FAME_SCALE = 10
			LOSER_FAME_SCALE = -10
			WINNER_ALLY_FAME_SCALE = 10
			LOSER_ALLY_FAME_SCALE = 10
		}
		add_truce_attacker_victory_effect = yes
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }
		laamp_as_mercenary_payout_tooltip_effect = yes
		mandala_war_victory_effects = yes
	}
	on_white_peace_desc = HD_UNIFICATION_WHITE_PEACE
	on_white_peace = { hd_unification_white_peace_effect = yes }
	on_defeat_desc = HD_UNIFICATION_DEFEAT
	on_defeat = { hd_unification_defeat_effect = yes }

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = HD_UNIFICATION_REGIONAL_WAR_NAME
	war_name_base = HD_UNIFICATION_REGIONAL_WAR_NAME_BASE
	cb_name = HD_UNIFICATION_REGIONAL_CB_NAME
	interface_priority = 120
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.5
	defender_ticking_warscore = 0.05
	attacker_ticking_warscore = 0.05
	defender_ticking_warscore_delay = { months = 24 }
	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150
	max_defender_score_from_battles = 100
	max_attacker_score_from_battles = 100
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

# A CHARACTER-targeted, realm-wide war; not a single-kingdom target list.
hd_unification_cb = {
	icon = celestial
	group = celestial
	should_show_war_goal_subview = no
	ai_only_against_neighbors = no

	allowed_for_character = {
		hd_unification_empire_attacker_trigger = yes
	}
	allowed_against_character = {
		scope:defender = {
			is_independent_ruler = yes
			hd_unification_realm_in_china_trigger = yes
		}
		scope:attacker = { top_liege != scope:defender.top_liege }
	}
	allowed_for_character_display_regardless = { }
	allowed_against_character_display_regardless = { }
	valid_to_start = {
		scope:attacker = { hd_unification_empire_attacker_trigger = yes }
		scope:defender = {
			is_independent_ruler = yes
			hd_unification_realm_in_china_trigger = yes
		}
		hd_unification_ai_cb_start_allowed_trigger = {
			ATTACKER = scope:attacker
			DEFENDER = scope:defender
		}
	}
	should_invalidate = {
		OR = {
			scope:attacker = { NOT = { hd_unification_empire_attacker_trigger = yes } }
			scope:defender = {
				OR = {
					is_independent_ruler = no
					NOT = { hd_unification_realm_in_china_trigger = yes }
				}
			}
		}
	}
	on_invalidated_desc = HD_UNIFICATION_FULL_INVALIDATED
	on_invalidated = { }
	ai_score_mult = { value = 100 }
	cost = {
		prestige = {
			value = hd_unification_full_cb_prestige_cost

		}
	}
	on_declaration = {

	}

		add_legitimacy_attacker_victory_effect = yes

		hd_unification_full_annex_effect = yes
		scope:attacker = { add_prestige_experience = major_prestige_value }
		scope:defender = { add_prestige = { value = major_prestige_value multiply = -1 } }
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_victory_effect = yes
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }
		laamp_as_mercenary_payout_tooltip_effect = yes
		mandala_war_victory_effects = yes
	}
	on_white_peace_desc = HD_UNIFICATION_WHITE_PEACE
	on_white_peace = { hd_unification_white_peace_effect = yes }
	on_defeat_desc = HD_UNIFICATION_DEFEAT
	on_defeat = { hd_unification_defeat_effect = yes }

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = HD_UNIFICATION_FULL_WAR_NAME
	war_name_base = HD_UNIFICATION_FULL_WAR_NAME_BASE
	cb_name = HD_UNIFICATION_FULL_CB_NAME
	interface_priority = 121
	use_de_jure_wargoal_only = no
	ticking_war_score_targets_entire_realm = yes
	attacker_wargoal_percentage = 0.8
