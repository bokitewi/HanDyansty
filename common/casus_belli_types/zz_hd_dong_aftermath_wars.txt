hd_li_jue_changan_cb = {
	icon = claim_throne
	group = civil_war
	target_titles = none
	target_top_liege_if_outside_realm = no
	ticking_war_score_targets_entire_realm = yes
	allowed_for_character = {
		scope:attacker = { this = character:li_jue }
		OR = { exists = global_var:hd_li_jue_war_pending exists = global_var:hd_li_jue_war_active }
		NOT = { exists = global_var:hd_li_jue_war_resolved }
	}
	allowed_against_character = { scope:defender = { has_title = title:d_jingzhao } }
	on_declaration = { on_declared_war = yes }
	on_victory_desc = { desc = hd_li_jue_changan_victory_desc }
	on_victory = {
		# Record the actual primary matchup BEFORE the original title settlement.
		if = {
			limit = {
				scope:attacker = { this = character:li_jue is_alive = yes }
				scope:defender = { this = character:lv_bu is_alive = yes }
			}
			scope:defender = { add_character_flag = hd_lbx_defeated_at_changan }
			# Protect named courtiers from the native camp constructor's old-court cleanup.
			character:zhang_liao ?= { hd_lbx_preserve_old_retainer_effect = yes }
			character:gao_shun ?= { hd_lbx_preserve_old_retainer_effect = yes }
			character:chen_gong ?= { hd_lbx_preserve_old_retainer_effect = yes }
		}
		scope:attacker = { hd_li_jue_victory_effect = yes }
		# Keep land loss and camp creation in this SAME victory effect, not a delayed event.
		character:lv_bu ?= {
			if = {
				limit = { is_alive = yes has_character_flag = hd_lbx_defeated_at_changan }
				hd_lbx_begin_exile_effect = yes
			}
		}
	}
	on_white_peace_desc = { desc = hd_li_jue_changan_no_victory_desc }
	on_white_peace = { hd_li_jue_end_war_effect = yes add_truce_white_peace_effect = yes }
	on_defeat_desc = { desc = hd_li_jue_changan_no_victory_desc }
	on_defeat = { hd_li_jue_end_war_effect = yes add_truce_attacker_defeat_effect = yes }
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = { hd_li_jue_end_war_effect = yes }
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	interface_priority = 75
	war_name = HD_LI_JUE_CHANGAN_WAR_NAME
}
