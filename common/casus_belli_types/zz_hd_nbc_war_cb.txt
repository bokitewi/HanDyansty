# 南北朝剧本史实战争（开局的桓温伐蜀、谯纵起兵，以及 zz_hd_nbc_history_effects.txt 的各场）。
# 结局一律调 hd_nbc_war_result_effect，设全局标记 hd_nbc_<场次>_won / _lost / _white / _invalid（以攻方论胜负，见战争变量 hd_nbc_war_kind）。
# 写法照 zz_hd_sg218_war_cb.txt：只能由脚本宣战，胜利时转移目标头衔。
hd_nbc_war_cb = {
	icon = conquest
	group = conquest
	combine_into_one = no
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no
	allowed_for_character = {}
	allowed_against_character = {}
	target_titles = all
	target_title_tier = all
	target_de_jure_regions_above = yes
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder
	should_check_for_interface_availability = no
	valid_to_start = { always = no }
	cost = {}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = { OR = { this = scope:defender target_is_liege_or_above = scope:defender } }
				}
			}
		}
	}

	on_invalidated_desc = msg_religious_war_invalidation_region_message
	on_invalidated = {
		hd_nbc_war_result_effect = { RESULT = invalid }
	}
	on_declaration = { on_declared_war = yes }

	on_victory_desc = { desc = hd_nbc_war_victory_desc }
	on_victory = {
		create_title_and_vassal_change = { type = conquest save_scope_as = hd_nbc_conquest_change add_claim_on_loss = no }
		every_in_list = {
			list = target_titles
			conquest_cb_title_transfer = { RELIGIOUS_WAR = no }
		}
		# 被灭之国的称谓变量与国号随头衔留下，胜者不应显示旧国的天王、皇帝和国号
		every_in_list = {
			list = target_titles
			limit = { has_variable = hd_nbc_style }
			remove_variable = hd_nbc_style
			reset_title_name = yes
		}
		every_in_list = { list = vassals_taken change_liege = { liege = scope:attacker change = scope:hd_nbc_conquest_change } }
		every_in_list = { list = titles_taken change_title_holder = { holder = scope:attacker change = scope:hd_nbc_conquest_change take_baronies = yes } }
		resolve_title_and_vassal_change = scope:hd_nbc_conquest_change
		add_truce_attacker_victory_effect = yes
		hd_nbc_war_result_effect = { RESULT = won }
	}

	on_white_peace_desc = { desc = hd_nbc_war_white_peace_desc }
	on_white_peace = {
		add_truce_white_peace_effect = yes
		hd_nbc_war_result_effect = { RESULT = white }
	}
	on_defeat_desc = { desc = hd_nbc_war_defeat_desc }
	on_defeat = {
		add_truce_attacker_defeat_effect = yes
		hd_nbc_war_result_effect = { RESULT = lost }
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	transfer_behavior = transfer
	attacker_allies_inherit = no
	defender_allies_inherit = yes
	war_name = "HD_NBC_WAR_NAME"
	interface_priority = 80
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	landless_attacker_needs_armies = no
	full_occupation_by_attacker_gives_victory = no
	full_occupation_by_defender_gives_victory = no
	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 50
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	max_ai_diplo_distance_to_title = 500
}
