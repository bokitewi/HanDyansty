# 衣冠南渡：义军起义 / 举州归南 宣战理由（只由脚本发动，玩家与 AI 都不能手动选用）

############################
# 义军起义：无地的义军首领对胡人宗主（或作乱的流民帅对南方宗主）起兵，胜则据目标县自立
############################
hd_yiguan_uprising_cb = {
	icon = peasant_war
	group = civil_war
	combine_into_one = no
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no

	allowed_for_character = {}
	allowed_against_character = {}
	target_titles = all
	target_title_tier = county
	target_de_jure_regions_above = no
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder
	should_check_for_interface_availability = no

	valid_to_start = {
		always = no
	}
	cost = {}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				holder ?= {
					OR = {
						this = scope:defender
						target_is_liege_or_above = scope:defender
					}
				}
			}
		}
	}

	on_invalidated_desc = msg_religious_war_invalidation_region_message

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = hd_yiguan_uprising_victory_desc
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yiguan_uprising_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = hd_yiguan_uprising_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yiguan_uprising_failed_allies_effect = yes
	}

	on_defeat_desc = {
		desc = hd_yiguan_uprising_defeat_desc
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yiguan_uprising_failed_allies_effect = yes
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	transfer_behavior = transfer
	attacker_allies_inherit = no
	defender_allies_inherit = yes

	war_name = "HD_YIGUAN_UPRISING_WAR_NAME"
	interface_priority = 80
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8

	landless_attacker_needs_armies = no
	full_occupation_by_attacker_gives_victory = yes
	full_occupation_by_defender_gives_victory = no

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 50
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	max_ai_diplo_distance_to_title = 500
}

############################
# 举州归南：沦陷区的汉人封臣对直属宗主起兵，胜后改投接壤的南方宗主
############################
hd_yiguan_guinan_cb = {
	icon = independence_faction_war
	group = independence
	combine_into_one = no
	allow_hostages = no

	allowed_for_character = {}
	allowed_against_character = {}
	target_titles = none
	target_top_liege_if_outside_realm = no
	should_check_for_interface_availability = no

	valid_to_start = {
		always = no
	}
	cost = {}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = hd_yiguan_guinan_victory_desc
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yiguan_guinan_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = hd_yiguan_guinan_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
	}

	on_defeat_desc = {
		desc = hd_yiguan_guinan_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_prestige = medium_prestige_loss
		}
		scope:defender = {
			add_opinion = {
				target = scope:attacker
				modifier = hd_yiguan_rebel_ally_opinion
			}
		}
	}

	on_primary_attacker_death = invalidate
	on_primary_defender_death = inherit

	transfer_behavior = transfer
	attacker_allies_inherit = no
	defender_allies_inherit = yes

	war_name = "HD_YIGUAN_GUINAN_WAR_NAME"
	interface_priority = 80

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	max_ai_diplo_distance_to_title = 500
}
