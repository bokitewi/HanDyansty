# 义军起义：无地的义军首领对胡人宗主（或作乱的流民帅、北府将领对南方宗主）起兵，胜则据目标县自立
# 守方为胡人时计入催化剂"沦陷区汉人起义"，失败计入"起义或北还失败"

hd_yiguan_uprising_cb = {
	icon = peasant_war
	group = civil_war
	combine_into_one = no
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no

	allowed_for_character = {}
	allowed_against_character = {}
	target_titles = all
	target_title_tier = county
	target_de_jure_regions_above = no
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder
	should_check_for_interface_availability = no

	valid_to_start = {
		always = no
	}
	cost = {}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				holder ?= {
					OR = {
						this = scope:defender
						target_is_liege_or_above = scope:defender
					}
				}
			}
		}
	}

	on_invalidated_desc = hd_yiguan_uprising_invalidated_desc

	# 催化剂"沦陷区汉人起义"由 hd_yiguan_spawn_uprising_effect 在战争确实开始后投（只对胡人宗主），此处不重复投
	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = hd_yiguan_uprising_victory_desc
	}

	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yiguan_uprising_victory_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_white_peace_desc = {
		desc = hd_yiguan_uprising_white_peace_desc
	}

	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yiguan_uprising_failed_allies_effect = yes
		if = {
			limit = {
				scope:defender = {
					hd_yiguan_is_hu_trigger = yes
					NOT = { hd_yiguan_is_sinicized_trigger = yes }
				}
			}
			hd_yiguan_rising_failed_effect = yes
		}
	
		}
		else = { hd_peace_cb_result_effect = yes }

	on_defeat_desc = {
		desc = hd_yiguan_uprising_defeat_desc
	}

			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		if = {
			limit = {

					NOT = { hd_yiguan_is_sinicized_trigger = yes }

	on_primary_defender_death = inherit

	attacker_allies_inherit = no
	defender_allies_inherit = yes

# 北伐（§1 第 5 轮、§6 #10）：南方政权专用，冲突期开放（组参数 hd_yiguan_can_beifa）
# 目标：北方一州（王国级法理州，可无人持有），须有县与攻方领地相邻；守方为胡人政权或汉化政权且在该州的北方县中持县
# "北方"按区域 hd_yiguan_north_region 判定，不比对 k_uuii_* 头衔：南北朝剧本 280.1.1 起（zz_hd_nbc_dejure.txt）及建国决议会把
# 河南、洛阳、河内等郡改挂洛州、司州等新州，而地理区域按原始法理划定、不随运行时改挂变化（原版 _geographical_regions.info）
# 花威望 150（只对玩家计；AI 由年度脉冲扣）。胜利夺取守方体系在该州的全部县、王师北定、威望 +中；白和与战败都算北伐失败
############################
hd_yiguan_beifa_cb = {
	icon = celestial
	group = celestial

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	ai = no

	allowed_for_character = {
		top_participant_group:hd_yiguan_nandu ?= { has_participant_group_parameter = hd_yiguan_can_beifa }
	}

	allowed_against_character = {
		scope:defender = {
			OR = {
				hd_yiguan_is_hu_regime_trigger = yes
				hd_yiguan_is_sinicized_regime_trigger = yes
			}
		}
	}

	target_titles = all
	target_title_tier = kingdom
	target_de_jure_regions_above = yes
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	valid_to_start = {
		scope:target = {
			# 北方之州：守方体系在州内持有位于北方八州区域的县（按区域判定，见上方说明）
			any_de_jure_county = {
				hd_yiguan_county_in_north_trigger = yes
				holder ?= { target_is_same_character_or_above = scope:defender }
			}
			# 州内有县与攻方领地相邻（或攻方已在州内持县）
			any_de_jure_county = {
				OR = {
					holder ?= { target_is_same_character_or_above = scope:attacker }
					any_neighboring_county = {
						holder ?= { target_is_same_character_or_above = scope:attacker }
					}
				}
			}
		}
	}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					continue = { tier > tier_county }	# 不下钻男爵领（原版 00_conquest.txt:495）
					tier = tier_county
					holder ?= { target_is_same_character_or_above = scope:defender }
				}
			}
		}
	}

	on_invalidated_desc = msg_duchy_conquest_war_invalidated_message
	on_invalidated = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		}
		else = { hd_peace_cb_invalidated_effect = yes }
	}

	# 按设计 §6 #10 固定 150，不乘原版花费系数（与 AI 脉冲所扣一致）
	cost = {
		prestige = {
			value = 0
			if = {
				limit = { scope:attacker = { is_ai = no } }
				add = {
					value = 150
					desc = CB_BASE_COST
				}
			}
		}
	}

	on_declaration = {
		on_declared_war = yes
		hd_yiguan_fire_catalyst_effect = {
			CATALYST = catalyst_hd_yiguan_beifa_declared
			CHAR = scope:attacker
		}
		# 冲突期拖延（负值，点数不足不扣）
		hd_yiguan_fire_delay_catalyst_effect = {
			CATALYST = catalyst_hd_yiguan_beifa_declared_delay
			TOWARD = hd_yiguan_phase_tingzhi
			VALUE = 50
			CHAR = scope:attacker
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = hd_yiguan_beifa_cb_victory_desc_attacker
			}
			desc = hd_yiguan_beifa_cb_victory_desc
		}
	}

	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }
		hd_yiguan_beifa_victory_effect = yes
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_victory_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_white_peace_desc = {
		desc = hd_yiguan_beifa_cb_white_peace_desc
	}

	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_white_peace_effect = yes
		hd_yiguan_beifa_failed_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = hd_yiguan_beifa_cb_defeat_desc_defender
			}
			desc = hd_yiguan_beifa_cb_defeat_desc
		}
	}

	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_defeat_effect = yes
		scope:defender = {
			accolade_defender_war_end_glory_gain_med_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = major_prestige_value
			}
		}
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_defeat_effect = yes
		scope:attacker = { save_temporary_scope_as = loser }
		on_lost_aggression_war_discontent_loss = yes
		hd_yiguan_beifa_failed_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "HD_YIGUAN_BEIFA_WAR_NAME"
	war_name_base = "HD_YIGUAN_BEIFA_WAR_NAME_BASE"
	cb_name = "HD_YIGUAN_BEIFA_CB_NAME"

	interface_priority = 85

	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.5

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
