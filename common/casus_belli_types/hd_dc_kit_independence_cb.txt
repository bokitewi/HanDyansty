# 「要求独立」被拒后发动的独立战争：所有层级的封臣胜利后完全独立。
hd_dc_demand_independence_war = {
	icon = independence_faction_war
	group = independence
	ai = no
	target_top_liege_if_outside_realm = no
	allow_hostages = no

	allowed_for_character = {
		is_landed = yes
		highest_held_title_tier >= tier_county
	}
	allowed_against_character = {
		scope:attacker = { liege = scope:defender }
	}
	cost = { prestige = 0 }
	on_declaration = { on_declared_war = yes }

	on_victory_desc = hd_peace_cb_result_desc
	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:attacker = {
			becomes_independent = { change = scope:change }
		}
		resolve_title_and_vassal_change = scope:change
		scope:attacker = {
			add_character_flag = { flag = hd_dc_recently_independent years = 3 }
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = root.war
				result = victory
			}
		}
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_white_peace_desc = hd_peace_cb_result_desc
	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = {
			show_pow_release_message_effect = yes
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = root.war
				result = white_peace
			}
		}
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_defeat_desc = hd_peace_cb_result_desc
	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = {
			show_pow_release_message_effect = yes
			hard_imprison_character_effect = {
				TARGET = this
				IMPRISONER = scope:defender
			}
			scope:defender = {
				add_opinion = {
					target = prev
					modifier = vassal_lost_faction_revolt_war
				}
			}
		}
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	should_invalidate = { scope:attacker.liege != scope:defender }
	on_invalidated_desc = msg_invalidate_war_title
	check_defender_inheritance_validity = no
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	transfer_behavior = transfer
	war_name = INDEPENDENCE_WAR_NAME
	interface_priority = 120
	attacker_wargoal_percentage = 0.8
	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 50
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	max_ai_diplo_distance_to_title = 500
}
