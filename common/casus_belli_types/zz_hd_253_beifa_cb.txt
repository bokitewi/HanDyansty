# 253「姜维北伐」：北伐中原宣战理由。改写自原版 china_seize_county_cb（10_tgp_china_wars.txt）。计划见 docs/姜维剧本_计划.md 4.6。
# 攻方：253 开局里汉的梁州（汉中）都督，作汉帝封臣（hd253_beifa_attacker_trigger）。
# 守方：魏国境内郡级以上的封臣（郭淮、陈泰、各郡守等），不打天子本人，魏全国兵力不会自动压上（target_top_liege_if_outside_realm = no）。
# 边境战争（boki 10-14）：只与边郡守将交战，不另给事件兵；羌胡响应、粮尽、魏将来降见 events/zz_hd_253_events.txt。
# 战后两年内不得再用（停战两年 + 北伐冷却修正）。
hd253_beifa_cb = {
	icon = county_conquest_cb
	group = celestial

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		hd253_beifa_attacker_trigger = yes
		NOT = { has_character_modifier = hd253_beifa_rest_modifier }
	}

	allowed_against_character = {
		scope:defender = { hd253_beifa_defender_trigger = yes }
		scope:attacker.top_liege != scope:defender.top_liege
	}

	target_titles = neighbor_land
	target_title_tier = county
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	ai_only_against_neighbors = yes

	# 电脑：守方兵力超过己方两倍就不打
	ai_score_mult = {
		value = 1
		if = {
			limit = { scope:defender.current_military_strength > scope:attacker.hd253_double_military_strength }
			multiply = 0
		}
	}

	valid_to_start = {
		scope:attacker = { hd253_beifa_attacker_trigger = yes }
	}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = {
						OR = {
							this = scope:defender
							target_is_liege_or_above = scope:defender
						}
					}
				}
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {
		debug_log = "HD253 BEIFA_RESULT invalid"
		scope:attacker = { add_character_modifier = { modifier = hd253_beifa_rest_modifier days = 730 } }
	}

	cost = {
		prestige = {
			add = {
				value = 50
				desc = CB_BASE_COST
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
		# 10-18 加固 F：北伐日志
		scope:attacker = {
			if = {
				limit = { is_ai = yes }
				debug_log = "HD253 BEIFA_DECLARED ai"
			}
			else = { debug_log = "HD253 BEIFA_DECLARED player" }
			if = {
				limit = { NOT = { this = character:jiang_wei } }
				debug_log = "HD253 BEIFA_DECLARED other_attacker"
			}
		}
		# 北伐事件：一月后羌胡响应，八月后粮秣告急（事件里再判战争是否还在）
		scope:attacker = {
			trigger_event = { id = hd253.0100 days = 30 }
			trigger_event = { id = hd253.0110 days = 240 }
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = hd253_beifa_cb_victory_desc_attacker
			}
			desc = hd253_beifa_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { accolade_attacker_war_end_glory_gain_low_effect = yes }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		every_in_list = {
			list = target_titles
			scope:attacker = { add_prestige_experience = minor_prestige_value }
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		hd253_beifa_end_effect = { RESULT = victory }
		debug_log = "HD253 BEIFA_RESULT won"
		scope:defender = { save_scope_as = hd253_beifa_loser }
		scope:attacker = { trigger_event = hd253.0120 }
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		desc = hd253_beifa_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		hd253_beifa_end_effect = { RESULT = white_peace }
		debug_log = "HD253 BEIFA_RESULT white"
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = hd253_beifa_cb_defeat_desc_defender
			}
			desc = hd253_beifa_cb_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = {
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		hd253_beifa_end_effect = { RESULT = defeat }
		debug_log = "HD253 BEIFA_RESULT lost"
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "HD253_BEIFA_WAR_NAME"
	war_name_base = "HD253_BEIFA_WAR_NAME_BASE"
	cb_name = "HD253_BEIFA_CB_NAME"

	interface_priority = 84

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
