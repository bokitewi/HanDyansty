hd_yuan_jizhou_cb = {
	icon = conquest
	group = conquest
	target_titles = all
	target_title_tier = kingdom
	target_top_liege_if_outside_realm = no
	allowed_for_character = {
		scope:attacker = {
			this = character:gong_sun_zan
			is_alive = yes
			has_title = title:k_uuii_youzhou
			has_character_flag = hd4_jizhou_authorized
		}
		NOT = {
			exists = global_var:hd4_jizhou_finished
		}
	}
	allowed_against_character = {
		scope:defender = {
			OR = {
				this = character:han_fu
				this = character:yuan_shao
			}
		}
	}
	valid_to_start = {
		scope:target = title:k_uuii_jizhou
		scope:defender = {
			has_title = title:k_uuii_jizhou
		}
		OR = {
			scope:defender = character:han_fu
			AND = {
				scope:defender = character:yuan_shao
				exists = global_var:hd_yuan_jizhou_handover_complete
			}
		}
	}
	cost = { }
	on_declaration = {
		on_declared_war = yes
	}
	should_invalidate = {
		OR = {
			character:yuan_shao = {
				is_alive = no
			}
			character:gong_sun_zan = {
				is_alive = no
			}
			NOT = {
				scope:attacker = character:gong_sun_zan
			}
			NOT = {
				scope:defender = {
					OR = {
						this = character:han_fu
						this = character:yuan_shao
					}
				}
			}
			NOT = {
				AND = {
					exists = title:k_uuii_jizhou.holder
					title:k_uuii_jizhou.holder = {
						OR = {
							this = character:han_fu
							this = character:yuan_shao
						}
					}
				}
			}
		}
	}
	on_victory_desc = {
		desc = hd_yuan_jizhou_victory_desc
	}
	on_victory = {
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = hd4_jizhou_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = target_titles
			add_to_list = titles_taken
			conquest_cb_title_transfer = {
				RELIGIOUS_WAR = no
			}
		}
		every_in_list = {
			list = vassals_taken
			change_liege = {
				liege = scope:attacker
				change = scope:hd4_jizhou_change
			}
		}
		every_in_list = {
			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:hd4_jizhou_change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:hd4_jizhou_change
		set_global_variable = hd4_jizhou_finished
		hd_yuan_jizhou_end_effect = yes
		add_truce_attacker_victory_effect = yes
	}
	on_white_peace_desc = {
		desc = hd_yuan_jizhou_no_conquest_desc
	}
	on_white_peace = {
		set_global_variable = hd4_jizhou_finished
		hd_yuan_jizhou_end_effect = yes
		add_truce_white_peace_effect = yes
	}
	on_defeat_desc = {
		desc = hd4_gsz_defeat_desc
	}
	on_defeat = {
		hd4_gsz_defeat_effect = yes
		set_global_variable = hd4_jizhou_finished
		hd_yuan_jizhou_end_effect = yes
		add_truce_attacker_defeat_effect = yes
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {
		hd_yuan_jizhou_end_effect = yes
	}
	transfer_behavior = transfer
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = HD_YUAN_JIZHOU_WAR_NAME
	interface_priority = 75
}
