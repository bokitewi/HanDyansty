hd_yuan_jizhou_cb = {
	icon = conquest
	group = conquest
	target_titles = all
	target_title_tier = kingdom
	target_top_liege_if_outside_realm = no
	allowed_for_character = { scope:attacker = { this = character:gong_sun_zan } exists = global_var:hd_yuan_jizhou_pending }
	allowed_against_character = { scope:defender = { OR = { this = character:han_fu this = character:yuan_shao } } }
	valid_to_start = { scope:target = title:k_uuii_jizhou scope:defender = { this = character:han_fu has_title = title:k_uuii_jizhou } }
	cost = { }
	on_declaration = { on_declared_war = yes }
	should_invalidate = {
		OR = {
			character:yuan_shao = { is_alive = no }
			character:gong_sun_zan = { is_alive = no }
			character:han_fu = { is_alive = no }
			NOT = { scope:attacker = character:gong_sun_zan }
			NOT = { scope:defender = { OR = { this = character:han_fu this = character:yuan_shao } } }
			NOT = { title:k_uuii_jizhou = { holder = { OR = { this = character:han_fu this = character:yuan_shao } } } }
		}
	}
	on_victory_desc = { desc = hd_yuan_jizhou_victory_desc }
	on_victory = {
		create_title_and_vassal_change = { type = conquest save_scope_as = hd_yuan_conquest_change add_claim_on_loss = no }
		every_in_list = { list = target_titles add_to_list = titles_taken conquest_cb_title_transfer = { RELIGIOUS_WAR = no } }
		every_in_list = { list = vassals_taken change_liege = { liege = scope:attacker change = scope:hd_yuan_conquest_change } }
		every_in_list = { list = titles_taken change_title_holder = { holder = scope:attacker change = scope:hd_yuan_conquest_change take_baronies = yes } }
		resolve_title_and_vassal_change = scope:hd_yuan_conquest_change
		debug_log = "HD_YUAN JIZHOU_WAR_ATTACKER_VICTORY"
		hd_yuan_jizhou_end_effect = yes
		add_truce_attacker_victory_effect = yes
	}
	on_white_peace_desc = { desc = hd_yuan_jizhou_no_conquest_desc }
	on_white_peace = { debug_log = "HD_YUAN JIZHOU_WAR_WHITE_PEACE" hd_yuan_jizhou_end_effect = yes add_truce_white_peace_effect = yes }
	on_defeat_desc = { desc = hd_yuan_jizhou_no_conquest_desc }
	on_defeat = { debug_log = "HD_YUAN JIZHOU_WAR_ATTACKER_DEFEAT" hd_yuan_jizhou_end_effect = yes add_truce_attacker_defeat_effect = yes }
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = { debug_log = "HD_YUAN JIZHOU_WAR_INVALIDATED" hd_yuan_jizhou_end_effect = yes }
	# Native transfer preserves the war object when its defender is subordinated.
	transfer_behavior = transfer
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = HD_YUAN_JIZHOU_WAR_NAME
	interface_priority = 75
}
