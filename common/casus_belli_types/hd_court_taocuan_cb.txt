# 乱世霸权·讨篡 CB（第 5 期，2026-10-10）
# - 权臣篡位（逼宫禅让、废为礼仪君主）后 10 年内，天下内的郡国制独立君主可讨伐篡位者
#   （h_china 的 var:hd_court_usurper 指向篡位者）
# - 花费 500 威望；胜利：讨篡者取天命（hd_court.0022 以讨篡者为 root 调用天命交接）
# - 停战与战败赔偿：原版征服 CB 标准
hd_court_taocuan_cb = {
	icon = celestial
	group = conquest
	ai_only_against_neighbors = no

	max_attacker_score_from_battles = 200
	attacker_capital_gives_war_score = yes
	imprisonment_by_attacker_give_war_score = yes

	allowed_for_character = {
		hd_court_warlord_trigger = yes
		exists = title:h_china.var:hd_court_usurper
	}

	allowed_against_character = {
		scope:defender = {
			has_title = title:h_china
			title:h_china.var:hd_court_usurper ?= this
		}
	}

	should_invalidate = {
		OR = {
			scope:defender = { NOT = { has_title = title:h_china } }
			NOT = { exists = title:h_china.var:hd_court_usurper }
		}
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	cost = {
		prestige = 500
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = hd_court_taocuan_cb_victory_desc
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		scope:attacker = {
			add_prestige_experience = major_prestige_value
			trigger_event = hd_court.0022
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = hd_court_taocuan_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = hd_court_taocuan_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { pay_short_term_gold_reparations_effect = { GOLD_VALUE = 3 } }
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = hd_court_taocuan_cb_war_name
	interface_priority = 95
}
