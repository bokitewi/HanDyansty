# A zero-cost, region-scoped war, not a claim war on a possibly uncreated
# kingdom. The claim is narrative/legal entitlement; this CB takes the land.
hd_yuan_expansion_cb = {
	icon = invasion
	group = invasion
	target_titles = all
	target_title_tier = kingdom
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	on_primary_attacker_death = invalidate
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	transfer_behavior = transfer
	war_name = hd_yuan_expansion_cb_war
	war_name_base = hd_yuan_expansion_cb_war_name
	cb_name = hd_yuan_expansion_cb_name
	interface_priority = 110
	# Automatic selection belongs to the 60-day controller, not generic AI.
	ai_score_mult = { value = 0 }
	allowed_for_character = {
		scope:attacker = {
			this = character:yuan_shao
			is_alive = yes
			is_landed = yes
			has_variable = hd_yuan_expansion_region
			NOT = { has_character_flag = hd_yuan_expansion_menu_pending }
		}
	}
	allowed_against_character = {
		scope:defender = { hd_yuan_expansion_strategic_target_trigger = yes }
		NOT = { scope:attacker.top_liege = scope:defender.top_liege }
	}
	valid_to_start = {
		scope:attacker = { hd_yuan_expansion_actor_ready_trigger = yes }
		scope:target = {
			this = scope:attacker.var:hd_yuan_expansion_region
			any_de_jure_county = {
				exists = holder
				holder.top_liege = scope:defender
			}
		}
	}
	cost = { prestige = { value = 0 } }
	on_declaration = { on_declared_war = yes }
	# DO NOT check strength, war count, current direction or menu-pending here.
	# Those are start gates, not reasons to invalidate an already legal war.
	should_invalidate = {
		OR = {
			scope:attacker = {
				OR = {
					NOT = { this = character:yuan_shao }
					is_alive = no
					is_landed = no
				}
			}
			scope:attacker.top_liege = scope:defender.top_liege
			NOT = { hd_yuan_expansion_war_has_land_trigger = yes }
		}
	}
	on_invalidated_desc = hd_yuan_expansion_cb_invalidated_desc
	on_invalidated = { }
	on_victory_desc = hd_yuan_expansion_cb_victory_desc
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		hd_yuan_expansion_victory_effect = yes
		add_truce_attacker_victory_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}
	on_white_peace_desc = hd_yuan_expansion_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}
	on_defeat_desc = hd_yuan_expansion_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_attacker_defeat_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}
}
