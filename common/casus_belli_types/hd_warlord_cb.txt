# 郡国制·军阀契约：「讨不臣」（2026-10-24 用户指定）
# 军阀拒绝「收回权力」后，其直属主君可用（标记 hd_warlord_defiant 记在军阀身上，值为主君，5 年）。
# 结构参照本模组「讨逆」（hd_dc_kit_taoni_cb.txt）。
# - 免费；胜利：收回权力（变回普通武官/文官）、夺其主头衔归主君、本人下狱；停战按原版内战 1825 天
# - AI 少用：主君兵力须达军阀 1.5 倍以上才宣战

hd_warlord_revoke_cb = {
	icon = vassalization
	group = vassalization

	allowed_for_character = {
		is_landed = yes
	}

	allowed_against_character = {
		scope:defender = {
			liege = scope:attacker
			hd_warlord_trigger = yes
			exists = var:hd_warlord_defiant
			var:hd_warlord_defiant = scope:attacker
		}
		trigger_if = {
			limit = { scope:attacker = { is_ai = yes } }
			scope:attacker.max_military_strength >= {
				value = scope:defender.max_military_strength
				multiply = 1.5
			}
		}
	}

	# 军阀不再是攻方的直属封臣、或已不是军阀（如被剥夺头衔）则作废
	should_invalidate = {
		scope:defender = {
			OR = {
				NOT = { liege = scope:attacker }
				NOT = { hd_warlord_trigger = yes }
			}
		}
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	cost = {
		prestige = 0
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = hd_warlord_revoke_cb_victory_desc
	on_victory = {
		hd_warlord_revoke_cb_victory_effect = yes
	}

	on_white_peace_desc = hd_warlord_revoke_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = root.war
				result = white_peace
			}
		}
		scope:defender = { remove_variable = hd_warlord_defiant }
	}

	on_defeat_desc = hd_warlord_revoke_cb_defeat_desc
	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = root.war
				result = defeat
			}
		}
		scope:defender = { remove_variable = hd_warlord_defiant }
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = hd_warlord_revoke_cb_war_name
	interface_priority = 80
}
