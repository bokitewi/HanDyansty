# 天朝循环大修：乱世战争。
# hd_siyi_raid_cb：四夷入寇（斗争期，非汉统治者），夺取汉地一个公国。

hd_siyi_raid_cb = {
	on_invalidated_desc = hd_peace_cb_invalidated_desc

	icon = invasion
	group = conquest
	combine_into_one = yes
	should_show_war_goal_subview = yes
	ai_only_against_neighbors = yes

	allowed_for_character = {
		hd_cycle_phase_is_struggle_trigger = yes
		is_independent_ruler = yes
		hd_cycle_non_han_trigger = yes
	}
	allowed_against_character = {
		scope:defender = { hd_cycle_in_china_trigger = yes }
	}

	target_titles = all
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			target_is_de_jure_liege_or_above = title:h_china
			any_in_de_jure_hierarchy = {
				tier = tier_county
				holder.top_liege ?= scope:defender
			}
		}
	}

	ai_score_mult = { value = 1 }

	cost = {
		prestige = {
			value = hd_siyi_raid_prestige_cost
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = hd_peace_cb_result_desc
	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			custom_tooltip = CONQUEST_CB_TITLE
			conquest_cb_title_transfer = {
				RELIGIOUS_WAR = no
			}
		}
		every_in_list = {
			list = vassals_taken
			change_liege = {
				liege = scope:attacker
				change = scope:change
			}
		}
		every_in_list = {
			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:change
		# 战时占领结算（历史入侵式）
		hd_occupation_settle_effect = yes
		# 奖励：牧群 = 发展度 x30（游牧），威望 = 发展度 x20
		every_in_list = {
			list = target_titles
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_county }
				save_temporary_scope_as = hd_raid_county
				scope:attacker = {
					add_prestige = { value = scope:hd_raid_county.development_level multiply = 20 }
					if = {
						limit = { government_has_flag = government_is_nomadic }
						domicile ?= { change_herd = { value = scope:hd_raid_county.development_level multiply = 30 } }
					}
				}
			}
		}
		# 天朝循环·时代特色：siyi_raid_won 改由通用战争结束钩子发射（hd_dc_kit_war_ended_effect），避免重复计分
		hd_cycle_count_conquest_effect = yes
		# 只生成一年和平
		scope:attacker = {
			add_truce_both_ways = {
				character = scope:defender
				years = 1
				war = root.war
				result = victory
			}
		}
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_white_peace_desc = hd_peace_cb_result_desc
	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		hd_occupation_clear_effect = yes
		add_truce_white_peace_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_defeat_desc = hd_peace_cb_result_desc
	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		scope:attacker = { show_pow_release_message_effect = yes }
		hd_occupation_clear_effect = yes
		add_truce_attacker_defeat_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = hd_siyi_raid_cb_war_name
	cb_name = hd_siyi_raid_cb_name
	interface_priority = 100
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 100
	defender_score_from_battles_scale = 100
	max_defender_score_from_battles = 100
	max_attacker_score_from_battles = 100
	max_defender_score_from_occupation = 100
	max_attacker_score_from_occupation = 150
}
