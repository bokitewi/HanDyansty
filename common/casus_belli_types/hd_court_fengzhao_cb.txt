# 乱世霸权·奉诏讨伐 CB（第 3 期，2026-10-10）
# - 由"令征伐""求诏讨伐"授予（var:hd_court_fengzhao_target，5 年）；令征伐时双方互得此 CB
# - 战争目标：防守方之国持有县的一个郡；胜利时取该郡内防守方之国持有的县（及郡头衔，若在其国内）
# - 免费；停战与战败赔偿用原版征服 CB 标准
hd_court_fengzhao_cb = {
	icon = celestial
	group = conquest
	combine_into_one = yes
	should_show_war_goal_subview = yes
	ai_only_against_neighbors = yes

	allowed_for_character = {
		hd_cycle_is_divided_trigger = yes
		hd_court_independent_or_regent_trigger = yes
		is_landed = yes
	}

	allowed_against_character = {
		hd_court_fengzhao_authorized_trigger = yes
		scope:defender = {
			is_independent_ruler = yes
			NOT = { has_title = title:h_china }
		}
		scope:attacker = { top_liege != scope:defender.top_liege }
	}

	target_titles = all
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			any_in_de_jure_hierarchy = {
				tier = tier_county
				holder.top_liege ?= scope:defender
			}
		}
	}

	should_invalidate = {
		scope:defender = { is_independent_ruler = no }
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = hd_court_fengzhao_cb_victory_desc
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_county
					holder.top_liege ?= scope:defender
				}
				change_title_holder = {
					holder = scope:attacker
					change = scope:change
					take_baronies = yes
				}
			}
			if = {
				limit = { holder.top_liege ?= scope:defender }
				change_title_holder = {
					holder = scope:attacker
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:attacker = {
			add_prestige_experience = medium_prestige_value
			remove_variable = hd_court_fengzhao_target
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = hd_court_fengzhao_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = hd_court_fengzhao_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { pay_short_term_gold_reparations_effect = { GOLD_VALUE = 3 } }
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = hd_court_fengzhao_cb_war_name
	interface_priority = 85
}
