# 天朝循环·时代特色：吞并（群雄割据期主力 CB，2026-10-08 用户指定）
# - 群雄割据（斗争期）期间，天下内独立统治者对接壤的天下内独立统治者使用
# - 花费：防守方持有县所在的每个郡（公爵领）200 威望，乘斗争期宣战折扣（hd_cycle_war_cost_mult_value，×0.4）
# - 战分：战斗分上限 200%，占领分上限 10%；首都（原版 10%）与俘虏统治者（引擎固定 100%）照原版开启
# - 胜利：防守方本人持有的全部头衔（霸权头衔除外，乱世只能由决议结束）归进攻方；其封臣归属交引擎默认处理
# - 停战与战败赔偿：原版征服 CB 标准
# 结构参照原版 conquest 类 CB 与本 mod 的 hd_strategic_annex_cb。

hd_dc_tunbing_cb = {
	icon = celestial
	group = conquest
	ai_only_against_neighbors = yes

	max_attacker_score_from_battles = 200
	max_attacker_score_from_occupation = 10
	attacker_capital_gives_war_score = yes
	imprisonment_by_attacker_give_war_score = yes

	allowed_for_character = {
		hd_cycle_phase_is_struggle_trigger = yes
		hd_court_independent_or_regent_trigger = yes	# 2026-10-10 乱世霸权（D28）：权臣视同独立君主
		# 2026-10-09：佣兵团/冒险者/骑士团不可用
		is_landed = yes
		NOT = { government_has_flag = cannot_be_vassal_or_liege }
		hd_dc_in_china_trigger = yes
	}

	allowed_against_character = {
		scope:defender = {
			is_independent_ruler = yes
			hd_dc_in_china_trigger = yes
		}
		scope:attacker = { top_liege != scope:defender.top_liege }
		hd_court_ai_respects_emperor_trigger = yes	# 2026-10-10 乱世霸权（D4）：AI 尊汉者不攻天子
		trigger_if = {	# AI 已由 ai_only_against_neighbors 预筛；接壤判定开销大，只对玩家检查
			limit = { scope:attacker = { is_ai = no } }
			scope:attacker = {
				any_neighboring_top_liege_realm_owner = { this = scope:defender }
			}
		}
	}

	should_invalidate = {
		OR = {
			hd_cycle_phase_is_struggle_trigger = no
			scope:defender = { is_independent_ruler = no }
			# 2026-10-09：清理旧存档中佣兵团长发起的吞并战争
			scope:attacker = {
				OR = {
					is_landed = no
					government_has_flag = cannot_be_vassal_or_liege
				}
			}
		}
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	cost = {
		prestige = {
			value = hd_dc_tunbing_cost_value
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = hd_dc_tunbing_cb_victory_desc
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		# 2026-10-10 乱世霸权（D29）：防守方为被挟持的天子时，以权臣为结算对象
		hd_court_save_settle_target_effect = yes
		scope:hd_court_settle = {
			every_held_title = {
				limit = {
					tier >= tier_county
					tier < tier_hegemony
					is_landless_type_title = no
				}
				add_to_temporary_list = hd_dc_tunbing_titles
			}
		}
		every_in_list = {
			list = hd_dc_tunbing_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:attacker = { add_prestige_experience = medium_prestige_value }
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = hd_dc_tunbing_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = hd_dc_tunbing_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { pay_short_term_gold_reparations_effect = { GOLD_VALUE = 3 } }
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = hd_dc_tunbing_cb_war_name
	interface_priority = 90
}
