	on_declaration = {

			limit = { scope:attacker = { has_variable = hd_tb_kind } }
			war ?= {
				if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:historical } set_variable = { name = hd_tb_kind value = flag:historical } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:cao } set_variable = { name = hd_tb_kind value = flag:cao } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:yuan } set_variable = { name = hd_tb_kind value = flag:yuan } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:jizhou } set_variable = { name = hd_tb_kind value = flag:jizhou } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:guandu } set_variable = { name = hd_tb_kind value = flag:guandu } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:xuzhou } set_variable = { name = hd_tb_kind value = flag:xuzhou } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:regional } set_variable = { name = hd_tb_kind value = flag:regional } }
				else_if = { limit = { scope:attacker.var:hd_tb_kind ?= flag:wu } set_variable = { name = hd_tb_kind value = flag:wu } }
			}
			if = {
				limit = { scope:attacker.var:hd_tb_kind ?= flag:guandu }
				set_variable = {

				type = conquest

				add_claim_on_loss = yes

					change = scope:change
					take_baronies = yes
				}
			}

				every_vassal = {
					limit = {
						this != scope:attacker
						is_landed = yes
						NOT = { government_has_flag = cannot_be_vassal_or_liege }
					}
					add_to_temporary_list = hd_dc_tunbing_vassals
				}
			}
			every_in_list = {
				list = hd_dc_tunbing_vassals
				change_liege = {
					liege = scope:attacker
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			every_in_list = {
				list = hd_dc_tunbing_pseudo_empires
				save_temporary_scope_as = hd_dc_tunbing_pseudo_empire
				scope:attacker = { destroy_title = scope:hd_dc_tunbing_pseudo_empire }
			}
			scope:attacker = { add_prestige_experience = medium_prestige_value }
			add_truce_attacker_victory_effect = yes
		}
	}
