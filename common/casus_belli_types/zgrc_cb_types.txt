# 尊王攘夷，参考原版的 征服周边
zgrc_county_rangyi_cb = {
	icon = county_conquest_cb        # 沿用原版图标（同 TGP china_seize_county_cb 做法）
	group = zgrc_jiaohua_cb_group

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	# —— 施教者：华夏礼乐之民，且奉公羊学「大一统」 ——
	allowed_for_character = {
		zgrc_can_use_jiaohua_cb_trigger = yes
		NOT = {
			government_has_flag = government_is_landless_adventurer
		}
	}

	# —— 受教者：蛮夷（闸②；本块 root 为被宣战者）——
	# 互斥判定：不与自身同宗、不以下犯上（「内服/汉化」者已由蛮夷判定排除）
	allowed_against_character = {
		scope:defender = { zgrc_is_manyi_character_trigger = yes }
		scope:attacker = {
			# 须「不同治下」：同一最高领主之下（内附胡人、同国封臣）不得兴师
			NOT = { top_liege = scope:defender.top_liege }
			# 不以下犯上
			NOT = { liege = scope:defender }
		}
	}

	target_titles = all
	target_title_tier = county
	ignore_effect = change_title_holder

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	ai_score_mult = {
		value = 1
		# 教化
		add = zgrc_jiaohua_ai_score_value
		add = house_relation_ai_score_value
	}

	# —— 开战条件：原版邻接判定 + 蛮夷闸①（所夺之县持有者须为蛮夷）——
	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			neighboring_county_or_viking_conquest_trigger = { CHARACTER = root }
			holder ?= { zgrc_is_manyi_character_trigger = yes }
		}
	}

	# —— 开战后失效条件（原版同款） ——
	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = {
						OR = {
							this = scope:defender
							target_is_liege_or_above = scope:defender
						}
					}
				}
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	# —— 成本（原版同款：同信仰耗虔诚，异信仰耗威望） ——
	cost = {
		piety = {
			value = 0
			if = {
				limit = {
					scope:defender.faith = scope:attacker.faith
				}
				add = {
					value = 25
					desc = CB_SAME_FAITH_COST
				}
				add = {
					value = 5
					multiply = scope:attacker.realm_size
					desc = CB_REALM_SIZE_COST
					max = 75
				}
			}
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = 0
			if = {
				limit = {
					scope:defender.faith != scope:attacker.faith
				}
				add = {
					value = 50
					desc = CB_COST_DIFFERENT_FAITH
				}
				add = {
					value = 5
					multiply = scope:attacker.realm_size
					desc = CB_REALM_SIZE_COST
					max = 100
				}
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	# —— 胜利：夺其县（原版同款；结算文案复用 county_conquest_cb_*） ——
	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = county_conquest_cb_victory_desc_attacker
			}
			desc = county_conquest_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_victory_effect = yes
		add_influence_attacker_victory_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_low_effect = yes }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		scope:defender = { ep3_war_loss_adventurer_tt_effect = yes }

		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
			}
		}

		resolve_title_and_vassal_change = scope:change

		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = minor_prestige_value
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_victory_effect = yes
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }
		laamp_as_mercenary_payout_tooltip_effect = yes
		mandala_war_victory_effects = yes
	}

	# —— 白和（原版同款） ——
	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_white_peace_desc_defender
			}
			desc = county_conquest_cb_white_peace_desc
		}
	}

				arrogant = medium_stress_impact_gain
			}
		}

	}

# ============================================================
# 雪耻复仇 —— 对「不共戴天」者宣战，胜则囚之
# ============================================================
# 授权：宣战方对目标持有「不共戴天」（zgrc_bugongdaitian_opinion）
#       该修正自带 imprisonment_reason / execute_reason / revoke_title_reason
#       → 囚禁（及日后处决）皆为「合法」，不担暴政
# 目标：打「人」不打「地」—— 攻其首都（CK3 需头衔目标），胜后不夺地
#       （注：原版 target_titles = none 者皆为派系/事件战争，无法在宣战界面手动选目标）
# 胜利：仅囚禁对方（不夺地）；仇家位在帝国及以上者，高概率得「烈丈夫」之名
# 断环：被合法处决者死因＝death_execution → 其亲属不复仇（见 zgrc_dachou_on_death_effect）
# ============================================================
zgrc_dafuchou_cb = {
	icon = county_conquest_cb
	group = zgrc_jiaohua_cb_group

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	# —— 打人：攻其首都（胜后不夺地） ——
	target_titles = all
	target_title_tier = county
	ignore_effect = change_title_holder

	# 施复仇者：对防守者持「不共戴天」，且非无地冒险者
	# 受复仇者：有地统治者；须「不同治下」（最高领主不同）方可以兵戎相见
	# 同一国家之内不得私动刀兵（内部之事另有国法），亦不以下犯上
	allowed_against_character = {
		scope:defender = { is_ruler = yes }
		scope:attacker = {
			has_opinion_modifier = { modifier = zgrc_bugongdaitian_opinion target = scope:defender }
			NOT = { government_has_flag = government_is_landless_adventurer }
			NOT = { top_liege = scope:defender.top_liege }
			NOT = { liege = scope:defender }
		}
	}

	# 战争目标只能是凶手的首都（依据：capital_county 为角色属性，00_activity_triggers.txt:254 同款比较）
	valid_to_start = {
		scope:target = scope:defender.capital_county
	}

	on_declaration = {
		on_declared_war = yes
	}

	# 开战后：仇恨若已不在（关系被清），则战争失其正当，作废
	should_invalidate = {
		NOT = {
			scope:attacker = {
				has_opinion_modifier = { modifier = zgrc_bugongdaitian_opinion target = scope:defender }
			}
		}
	}
	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	# —— 胜利：囚之（血仇得报，其人入狱） ——
	on_victory_desc = {
		desc = zgrc_dafuchou_cb_victory_desc
	}

	on_defeat_desc = {
		desc = zgrc_dafuchou_cb_defeat_desc
	}

	on_white_peace_desc = {
		desc = zgrc_dafuchou_cb_white_peace_desc
	}

	on_victory = {
		scope:attacker = {
			# 大仇得报：仇家位在帝国及以上（含帝国）→ 高概率得「烈丈夫」之名
			if = {
				limit = {
					scope:defender = { highest_held_title_tier >= tier_empire }
					NOT = { has_nickname = nick_zgrc_liezhangfu }
				}
				random = {
					chance = 75
					give_nickname = nick_zgrc_liezhangfu
				}
			}
			# 囚之
			imprison = {
				target = scope:defender
				type = dungeon
			}
		}
	}

	# —— 白和 ——
	on_white_peace = {
		scope:attacker = {
			add_prestige = { value = minor_prestige_value multiply = -1.0 }
		}
		add_truce_white_peace_effect = yes
	}

	# —— 战败 ——
	on_defeat = {
		scope:attacker = {
			add_prestige = { value = major_prestige_value multiply = -1.0 }
		}
		add_truce_attacker_defeat_effect = yes
	}

	# 存续与继承（原版同款）
	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	# 文案
	war_name = "ZGRC_XUECHOU_NAME"
	war_name_base = "ZGRC_XUECHOU_NAME_BASE"
	cb_name = "zgrc_dafuchou_cb_NAME"

	interface_priority = 79

	# 战争节奏（对齐原版 depose_war + 项目 CB）
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
