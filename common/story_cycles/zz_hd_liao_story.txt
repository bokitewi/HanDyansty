# 僚人入蜀：渐进僚化。迁徙战争胜利后创建在僚帅身上，传给主继承人。
# 每年 15% 改一块属于故事主领国、位于目标区域内、与僚人伯爵领接壤的伯爵领；累计 20 块后结束。

story_hd_liao_settlement = {
	on_setup = {
		set_variable = {
			name = hd_liao_converted
			value = 0
		}
	}

	on_end = {}

	on_owner_death = {
		if = {
			limit = {
				exists = story_owner.primary_heir
				story_owner.primary_heir = { is_alive = yes }
			}
			make_story_owner = story_owner.primary_heir
		}
		else = {
			end_story = yes
		}
	}

	effect_group = {
		days = 365
		chance = 15
		trigger = {
			var:hd_liao_converted < hd_liao_convert_cap
		}
		triggered_effect = {
			trigger = { always = yes }
			effect = {
				story_owner = { hd_liao_convert_one_county_effect = yes }
				if = {
					limit = { exists = scope:hd_liao_conv_county }
					change_variable = {
						name = hd_liao_converted
						add = 1
					}
				}
			}
		}
	}

	effect_group = {
		days = 30
		trigger = {
			var:hd_liao_converted >= hd_liao_convert_cap
		}
		triggered_effect = {
			trigger = { always = yes }
			effect = { end_story = yes }
		}
	}
}
