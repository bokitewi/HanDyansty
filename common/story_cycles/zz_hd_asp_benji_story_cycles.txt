# 马氏：马腾死后若马超在世，本纪随之传给马超（与野望名单链一致）
hd_story_ma_life = {
	visible = yes
	icon = { trigger = { always = yes } reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds" }
	on_setup = { set_variable = hd_benji }
	visualization = { custom_string_key = "HD_MA_LIFE_STORY_STATUS" }
	on_owner_death = {
		if = {
			limit = {
				trigger_if = {
					limit = {
						has_global_variable = wj_hchar_ma_teng
						has_global_variable = wj_hchar_ma_chao
					}
					story_owner = { this = global_var:wj_hchar_ma_teng }
					global_var:wj_hchar_ma_chao = { is_alive = yes }
				}
				trigger_else = { always = no }
			}
			make_story_owner = global_var:wj_hchar_ma_chao
		}
		else = { scope:story = { end_story = yes } }
	}
}
