# 农民义军：有农民领袖特质的统治者，战时每半年按外交×管理判定一次乡民来投。
# 不随继承传递；失去特质、死亡、或既无领地又不在战争中时结束。

story_hd_peasant_army = {
	on_setup = {}

	on_end = {}

	on_owner_death = {
		end_story = yes
	}

	effect_group = {
		days = 180
		trigger = {
			story_owner = {
				hd_peasant_army_eligible_trigger = yes
				is_at_war = yes
				is_imprisoned = no
			}
		}
		triggered_effect = {
			trigger = { always = yes }
			effect = {
				story_owner = {
					random = {
						chance = hd_peasant_army_chance_value
						trigger_event = hd_peasant_army.0001
					}
				}
			}
		}
	}

	effect_group = {
		days = 365
		trigger = {
			story_owner = {
				NOT = { hd_peasant_army_eligible_trigger = yes }
			}
		}
		triggered_effect = {
			trigger = { always = yes }
			effect = { end_story = yes }
		}
	}
}
