# Exactly one live instance belongs to Yuan. Ending the native story cancels
# its native timer; there is no chain of orphaned delayed pulse events.
hd_story_yuan_expansion = {
	visible = no
	icon = { trigger = { always = yes } reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds" }
	on_setup = {
		set_variable = { name = hd_yuan_expansion_story_region value = story_owner.var:hd_yuan_expansion_region }
	}
	visualization = { custom_string_key = "HD_YUAN_EXPANSION_STORY_STATUS" }
	on_owner_death = {
		story_owner = { hd_yuan_expansion_owner_death_effect = yes }
		scope:story = { end_story = yes }
	}
	effect_group = {
		days = 60
		triggered_effect = {
			trigger = { story_owner = { is_alive = yes } }
			effect = {
				save_scope_as = hd_yuan_expansion_tick_story
				story_owner = { hd_yuan_expansion_tick_effect = yes }
			}
		}
	}
}
