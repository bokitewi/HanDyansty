# One personal story for Lu Bu, independent of the native adventurer AI story.
# Completion leaves a visible record but disables the repair effect group.
hd_story_lubu_exile = {
	visible = yes
	icon = { trigger = { always = yes } reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds" }
	visualization = { custom_string_key = "HD_LUBU_EXILE_STORY_STATUS" }
	on_owner_death = {
		scope:story = { end_story = yes }
	}
	effect_group = {
		days = 30
		triggered_effect = {
			trigger = {
				story_owner = {
					is_alive = yes
					has_character_flag = hd_lbx_exile_pending
					NOT = { has_character_flag = hd_lbx_completed }
				}
			}
			effect = { story_owner = { hd_lbx_progress_effect = yes } }
		}
	}
}
