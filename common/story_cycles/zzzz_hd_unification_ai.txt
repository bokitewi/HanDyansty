# One story per eligible independent celestial AI. No immediate war inside title
# transfer on-actions; the first attempt is deferred to the 30-60 day group.
hd_unification_ai_story = {
	on_setup = { }
	on_end = { }
	on_owner_death = { end_story = yes }
	effect_group = {
		days = { 30 60 }
		triggered_effect = {
			trigger = {
				story_owner = {
					is_ai = yes
					hd_unification_celestial_attacker_trigger = yes
				}
			}
			effect = {
				story_owner = { hd_unification_ai_attempt_effect = yes }
			}
		}
		triggered_effect = {
			trigger = {
				story_owner = {
					OR = {
						is_ai = no
						NOT = { hd_unification_celestial_attacker_trigger = yes }
					}
				}
			}
			effect = { end_story = yes }
		}
	}
}
