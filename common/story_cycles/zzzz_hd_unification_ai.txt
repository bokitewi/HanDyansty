# Replaces this ONE small existing story file; retains the old story identity.
# Current saves and the existing creation helper therefore use the same story.
# Diplomatic follow-ups are event-driven; this group is only a 180-day fallback.
hd_unification_ai_story = {
    on_setup = { }
    on_end = { }
    on_owner_death = { end_story = yes }
    effect_group = {
        days = 180
        triggered_effect = {
            trigger = {
                story_owner = {
                    is_ai = yes
                    hd_unification_celestial_attacker_trigger = yes
                }
            }
            effect = {
                story_owner = {
                    if = {
                        limit = { hd_submission_world_active_trigger = yes }
                        hd_submission_dispatch_effect = yes
                    }
                    else = {
                        hd_unification_ai_attempt_effect = yes
                    }
                }
            }
        }
        triggered_effect = {
            trigger = {
                story_owner = {
                    OR = {
                        is_ai = no
                        NOT = { hd_unification_celestial_attacker_trigger = yes }
                    }
                }
            }
            effect = { end_story = yes }
        }
    }
}
