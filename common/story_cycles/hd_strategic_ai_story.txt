# 天朝循环大修：战略区 AI story（统一派发器）。
# 乱世（崩溃/斗争/对峙）中，独立公爵级以上 AI 每 40-80 天按战略区扩张一次。
hd_strategic_ai_story = {
	on_setup = {
		story_owner = { hd_strat_init_effect = yes }
	}
	on_end = {
		story_owner ?= { hd_strat_cleanup_effect = yes }
	}
	on_owner_death = { end_story = yes }

	effect_group = {
		days = { 40 80 }
		# 失效：离开乱世 / 不再满足条件 / 无战略区
		triggered_effect = {
			trigger = {
				story_owner = {
					OR = {
						hd_strat_eligible_trigger = no
						NOT = { has_variable = hd_strat_region }
					}
				}
			}
			effect = { end_story = yes }
		}
		triggered_effect = {
			trigger = {
				story_owner = {
					hd_strat_eligible_trigger = yes
					has_variable = hd_strat_region
				}
			}
			effect = {
				story_owner = { hd_strat_pulse_effect = yes }
				# 重启招降调度器：定期扫描可招降的邻国（内部自带 240 天防重入）。
				story_owner = { hd_submission_queue_followup_effect = { DELAY_MIN = 20 DELAY_MAX = 60 } }
			}
		}
	}
}
