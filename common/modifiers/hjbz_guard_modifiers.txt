# ────────────────── 角色修正 1：羽林军重建者 ──────────────────
# 草稿：军事+3；自然恐怖值+50；敌方阴谋成功概率上限-100%
# 持续：永久，不可继承（由事件 add_character_modifier 施加，不带 years；
#       角色死亡即消失，故天然「不可继承」）
hjbz_yulin_rebuilder_modifier = {
	icon = martial_positive

	martial = 3
	dread_baseline_add = 50

	# 「敌方阴谋成功率上限」无通用字段，只能按阴谋类型写。
	# 仅拦截「敌对阴谋」(hostile scheme)：谋杀 / 绑架 / 篡位 / 伪造牵制。
	# 取值 -100 = 上限归零，即此类阴谋对本人不可能成功。
	enemy_hostile_scheme_success_chance_max_add = -100
}


# ────────────────── 角色修正 2：羽林军 ──────────────────
# 草稿：兵士军团上限+1；兵士军团规模上限+1
# 持续：永久，继承给 h_china 的持有者
#
# ⚠️ CK3 角色修正不跨角色继承，故「继承」由 on_action 钩子维持：
#    见 common/on_action/hjbz_guard_on_actions.txt 的 on_title_gain / on_title_lost。
#    本修正是「朝廷之制」而非「君主之私」，故随 h_china 头衔转移。
hjbz_yulin_guard_modifier = {
	icon = martial_positive

	men_at_arms_cap = 1		# 兵士团数上限 +1
	men_at_arms_limit = 1	# 兵士每团规模 +1
}


# ────────────── 伯爵领修正 1：焕然一新的执金吾 ──────────────
# 草稿：每月控制力增长+100%；每月控制力下降-80%；旅行安全性+50；税收+10%；民众好感+20
# 持续：20 年（由事件 add_county_modifier 带 years = 20）
hjbz_zhijinwu_renewed_modifier = {
	icon = county_control_positive

	monthly_county_control_growth_factor = 1.0		# +100% 控制力增长
	monthly_county_control_decline_factor = -0.8	# -80% 控制力下降

	travel_danger = -50		# 旅行危险 -50（= 更安全）

	tax_mult = 0.1			# 税收 +10%
	county_opinion_add = 20	# 民众好感 +20
}


# ────────────── 伯爵领修正 2：崩溃的执金吾 ──────────────
# 草稿：每月控制力-0.5；每月控制力下降+100%；旅行安全性-50；民众好感-20
# 持续：20 年
hjbz_zhijinwu_collapsed_modifier = {
	icon = county_control_negative

	# 「每月控制力 -0.5」= 每月下降量 +0.5
	# （原版唯一实证：00_special_buildings.txt:10586 monthly_county_control_decline_add = 0.5）
	monthly_county_control_decline_add = 0.5
	monthly_county_control_decline_factor = 1.0		# +100% 控制力下降

	travel_danger = 50		# 旅行危险 +50

	county_opinion_add = -20
}
