# 丞相只由共用授相脚本启用；普通内阁任命不能凭空创建丞相。
WJ_minister_cheng_xiang = {
	skill = diplomacy

	name = {
		first_valid = {
			triggered_desc = {
				trigger = { government_has_flag = government_is_celestial }
				desc = WJ_minister_cheng_xiang_hanjin
			}
			desc = WJ_minister_cheng_xiang_tang
		}
	}

	valid_position = {
		# Native council positions must exist before the title-gain callback assigns them.
		# Real-title eligibility stays in valid_character; cabinet layout is GUI-only.
		tgp_has_access_to_ministry_trigger = yes
		government_has_flag = government_is_celestial
	}

	tooltip = WJ_minister_cheng_xiang_desc

	auto_fill = { always = no }

	modifier = {
		name = WJ_minister_cheng_xiang_modifier
		monthly_influence = 1
		scale = council_scaled_admin_value
	}

	council_owner_modifier = {
		name = learn_on_the_job_modifier
		diplomacy = 1
		scale = chancellor_learn_on_the_job_scale
	}

	valid_character = {
		has_title = title:e_WJ_minister_cheng_xiang
		WJ_yinshi_not_hermit_trigger = yes # 隐士不受任命
		exists = root.liege_or_court_owner
		tgp_can_be_a_minister_trigger = { COURT_OWNER = root.liege_or_court_owner }
		WJ_ministry_position_acceptance_trigger = { MINISTER_TITLE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang }
		hd_minister_merit_gate_trigger = { MINISTER_TITLE = e_WJ_minister_cheng_xiang POSITION = WJ_minister_cheng_xiang }
	}

	on_get_position = {
		got_council_position_effect = yes
		got_minister_position_effect = { MINISTER_TITLE = e_WJ_minister_cheng_xiang MINISTER_POSITION = WJ_minister_cheng_xiang }
	}

	on_fired_from_position = {
		if = {
			limit = { scope:councillor_liege = { government_has_flag = government_is_celestial } }
			fired_from_council_position_effect = yes
		}
	}

	on_lose_position = {
		# 改制撤销不走普通离职继任链，交由唐法生效后定点清理。
		if = {
			limit = { scope:councillor_liege = { government_has_flag = government_is_celestial } }
			# Native loss on death is not a dismissal when a valid family heir was saved.
			if = {
				limit = { NOT = { has_character_flag = wj_cx_death_handoff } }
				WJ_cx_abolish = yes
			}
		}
	}

	portrait_animation = happy_teacher

	barbershop_data = {
		position = { 834.0 172.0 }
		click_to_front = no
	}
}
