# OE 副本：在世子女先改挂到本 mod 那份（家族一并改成父亲的家族），再静默除去。只在 89 开局生效
hd_89_oe_dup_one_effect = {
	character:$OE$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_landed = no
				character:$KEEP$ ?= { is_alive = yes }
			}
			every_child = {
				limit = { is_alive = yes }
				if = {
					limit = { NOT = { father ?= character:$KEEP$ } }
					set_father = character:$KEEP$
				}
				if = {
					limit = {
						exists = character:$KEEP$.house
						NOT = { house ?= character:$KEEP$.house }
					}
					set_house = character:$KEEP$.house
				}
			}
			death = { death_reason = death_vanished }
			debug_log = "HD_89 OE_DUP_REMOVED $OE$"
		}
	}
}

hd_89_dynastic_cycle_effect = {
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = title:h_china.holder
			NOT = { exists = situation:dynastic_cycle }
		}
		start_situation = {
			type = dynastic_cycle
			start_phase = situation_dynastic_cycle_phase_stability_expansion
		}
	}
	if = {
		limit = {
			has_tgp_dlc_trigger = yes
			exists = situation:dynastic_cycle
			NOT = { exists = global_var:saas_dynastic_cycle_initialized }
		}
		situation:dynastic_cycle = {
			title:h_china = {
				set_variable = { name = years_since_imperial_examination value = 0 }
			}
			title:h_china.holder ?= {
				if = {
					limit = { legitimacy < legitimacy_level_6 }
					add_legitimacy = {
						value = legitimacy_level_6
						subtract = legitimacy
					}
				}
			}
			tgp_set_wuking_element_effect = { ELEMENT = fire }
			every_situation_participant = {
				recalculate_participant_group = situation:dynastic_cycle
			}
			hd_89_zero_progress_effect = yes
			trigger_event = { on_action = hd_89_dc_zero_progress days = 2 }
		}
		set_global_variable = {
			name = saas_dynastic_cycle_initialized
			value = yes
		}
		debug_log = "HD_89 DYNASTIC_CYCLE EXPANSION FIRE"
	}
}

# 疏勒、于阗、鄯善奉和帝为宗主（天朝朝贡合同：只有纳贡、册封，无参战义务）。

hd_89_tributary_one_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				is_independent_ruler = yes
				is_tributary = no
				NOT = { this = title:h_china.holder }
			}
			start_tributary = {
				contract_group = tributary_celestial
				suzerain = title:h_china.holder
			}
			if = {
				limit = { is_tributary = yes }
				hidden_effect = {
					tributary_contract_set_obligation_level = { type = celestial_tribute_gold level = 1 }
					tributary_contract_set_obligation_level = { type = celestial_tribute_prestige level = 1 }
				}
				add_subject_standing = 10
			}
			debug_log = "HD_89 TRIBUTARY $WHO$"
		}
	}
}
