# 太傅：新增真实内阁席位，复用原版文官省部任命链。
WJ_minister_tai_fu = {
	skill = learning

	name = WJ_minister_tai_fu

	valid_position = {
		tgp_has_access_to_ministry_trigger = yes
		WJ_han_jin_law_trigger = yes
	}

	tooltip = WJ_minister_tai_fu_desc

	auto_fill = { }

	modifier = {
		name = WJ_minister_tai_fu_modifier
		monthly_influence = 1
		scale = council_scaled_admin_value
	}

	council_owner_modifier = {
		name = learn_on_the_job_modifier
		diplomacy = 1
		scale = chancellor_learn_on_the_job_scale
	}

	valid_character = {
		exists = root.liege_or_court_owner
		tgp_can_be_a_minister_trigger = { COURT_OWNER = root.liege_or_court_owner }
		# auto_fill 空块已等于 no，仍须阻止 native 在184开局预填普通候选人。
		# 次日起正常任命；已有头衔者仍可入席，不清退旧档或显式授衔者。
		OR = {
			has_title = title:e_WJ_minister_tai_fu
			NOT = { game_start_date = 184.1.1 }
			NOT = { current_date = 184.1.1 }
		}
	}

	on_get_position = {
		got_council_position_effect = yes
		got_minister_position_effect = { MINISTER_TITLE = e_WJ_minister_tai_fu MINISTER_POSITION = WJ_minister_tai_fu }
	}

	on_fired_from_position = {
		if = {
			limit = { scope:councillor_liege = { WJ_han_jin_law_trigger = yes } }
			fired_from_council_position_effect = yes
		}
	}

	on_lose_position = {
		# 改制撤销不走普通离职继任链，交由唐法生效后定点清理。
		if = {
			limit = { scope:councillor_liege = { WJ_han_jin_law_trigger = yes } }
			fired_minister_position_effect = yes
		}
	}

	portrait_animation = happy_teacher

	barbershop_data = {
		position = { 750.0 172.0 }
		click_to_front = no
	}
}
