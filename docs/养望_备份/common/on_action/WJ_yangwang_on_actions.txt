# Native quarter is personal, not calendar Q1: quarters 1 and 3 are six months apart.
quarterly_playable_pulse = { on_actions = { WJ_yangwang_halfyear } }
on_title_gain = { on_actions = { WJ_yangwang_reconcile } }
on_title_lost = { on_actions = { WJ_yangwang_reconcile } }
on_join_court = { on_actions = { WJ_yangwang_reconcile } }
WJ_yangwang_reconcile = {
 trigger = { has_trait = WJ_yangwang_trait WJ_yangwang_eligible_trigger = no }
 effect = { WJ_yangwang_exit_effect = yes }
}
WJ_yangwang_halfyear = {
 effect = {
  if = {
   limit = { has_trait = WJ_yangwang_trait WJ_yangwang_eligible_trigger = no }
   WJ_yangwang_exit_effect = yes
  }
  if = {
   limit = {
    WJ_yangwang_able_trigger = yes
    OR = { scope:quarter = 1 scope:quarter = 3 }
   }
   WJ_yangwang_annual_effect = yes
  }
 }
}
