# ============================================================================
# WJ 清议活动 —— 以 CK3 原版宴会活动为模板
#
# 结构依据 game/common/activities/activity_types/feast.txt：
#   地点、荣誉嘉宾、活动选项、意图、预定义阶段、结算。
# 清议不再使用旧活动的运动、挑战者或正统性系统。
# ============================================================================

activity_wj_qingyi = {
	is_shown = {
		WJ_han_jin_law_trigger = yes
		has_tgp_dlc_trigger = yes
		government_has_flag = government_has_merit
		is_adult = yes
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_wj_qingyi }
		is_available_adult = yes
		is_at_war = no
	}

	can_always_plan = no

	is_valid = {
		scope:host = {
			WJ_han_jin_law_trigger = yes
			is_alive = yes
			is_capable_adult = yes
			is_imprisoned = no
		}
		trigger_if = {
			limit = { is_current_phase_active = yes }
			has_attending_activity_guests = yes
		}
	}

	on_invalidated = {
		every_attending_character = {
			WJ_qingyi_cleanup_participant_effect = yes
		}
		if = {
			limit = { has_attending_activity_guests = no }
			scope:activity = {
				activity_type = { save_scope_as = activity_type }
				activity_location = { save_scope_as = location }
			}
			scope:host = { trigger_event = activity_system.0100 }
		}
		if = {
			limit = { scope:host = { is_incapable = yes } }
			scope:activity = { activity_type = { save_scope_as = activity_type } }
			scope:host = { trigger_event = activity_system.0330 }
			every_attending_character = {
				limit = { this != scope:host }
				trigger_event = activity_system.0331
			}
		}
	}

	on_host_death = {
		every_attending_character = {
			limit = { this != scope:host }
			trigger_event = activity_system.0320
		}
	}

	###################
	# PARAMETERS
	###################

	is_single_location = yes

	province_filter = domicile_domain
	ai_province_filter = capital
	max_province_icons = 5

	is_location_valid = {
		has_holding = yes
	}

	province_description = {
		desc = activity_wj_qingyi_province_desc
	}

	cooldown = { years = 3 }

	ai_will_do = {
		value = 25
		if = {
			limit = { has_trait = gregarious }
			add = 25
		}
		if = {
			limit = { has_trait = scholar }
			add = 25
		}
		if = {
			limit = { has_trait = shy }
			subtract = 15
		}
	}

	###################
	# OPTIONS
	###################

	options = {
		# No feast/debate type-selection stage: these are the three direct
		# preparation choices shown on the activity planning window.
		qingyi_option_topic_depth = {
			qingyi_topic_plain = {
				default = yes
				ai_will_do = {
					value = 50
					add = activity_option_likes_cheap_expense_value
				}
			}
			qingyi_topic_deep = {
				cost = {
					gold = { value = WJ_yangwang_qingyi_paid_cost }
				}
				ai_will_do = {
					value = 50
					add = activity_option_likes_ostentatious_food_option_value
				}
			}
		}

		qingyi_option_tea = {
			qingyi_tea_simple = {
				default = yes
				ai_will_do = {
					value = 50
					add = activity_option_food_prefers_small_volume_value
				}
			}
			qingyi_tea_refined = {
				cost = {
					gold = { value = WJ_yangwang_qingyi_paid_cost }
				}
				ai_will_do = {
					value = 50
					add = activity_option_prefers_large_volume_value
				}
			}
		}

		qingyi_option_wine = {
			qingyi_wine_none = {
				default = yes
				ai_will_do = {
					value = 50
					add = activity_option_likes_cheap_expense_value
				}
			}
			qingyi_wine_generous = {
				cost = {
					gold = { value = WJ_yangwang_qingyi_paid_cost }
				}
				ai_will_do = {
					value = 50
					add = activity_option_likes_ostentatious_food_option_value
				}
			}
		}
	}

	###################
	# GUEST HANDLING
	###################

	max_guests = 20

	guest_invite_rules = {
		defaults = {
			1 = activity_invite_rule_close_family
			1 = activity_invite_rule_liege
			1 = activity_invite_rule_vassals
			2 = activity_invite_rule_friends
			2 = activity_invite_rule_courtiers
			2 = activity_invite_rule_guests
		}
	}

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		in_diplomatic_range = scope:host
	}

	host_intents = {
		intents = { qingyi_host_prestige_intent qingyi_relief_intent }
		default = qingyi_host_prestige_intent
	}

	guest_intents = {
		intents = { qingyi_host_prestige_intent qingyi_relief_intent }
		default = qingyi_host_prestige_intent
	}

	guest_join_chance = {
		base = 10
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
		activity_guest_feast_ai_accept_modifier = yes
	}

	special_guests = {
		ming_shi = {
			is_required = no
			can_pick = {
				this != scope:host
				is_adult = yes
				is_healthy = yes
				OR = {
					is_vassal_of = scope:host
					target_is_vassal_or_below = scope:host
					has_relation_friend = scope:host
					has_relation_lover = scope:host
				}
			}
			ai_will_do = { value = 10 }
		}
	}

	open_invite = yes
	max_guest_arrival_delay_time = { months = 5 }

	###################
	# PHASES
	###################

	phases = {
		qingyi_phase_gathering = {
			order = 1
			is_predefined = yes
			location_source = pickable

			on_phase_active = {
				WJ_qingyi_prepare_participant_effect = yes
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { months = 1 } }
					trigger_event = WJ_qingyi.0002
				}
				else = {
					trigger_event = WJ_qingyi.0003
				}
			}

			on_weekly_pulse = {
				if = {
					limit = { this = scope:host }
					trigger_event = { on_action = WJ_qingyi_host_event_selection }
				}
				else_if = {
					limit = { is_ai = no }
					trigger_event = { on_action = WJ_qingyi_guest_event_selection }
				}
				else = {
					random = {
						chance = 25
						trigger_event = { on_action = WJ_qingyi_guest_event_selection }
					}
				}
			}
		}

		qingyi_phase_conclusion = {
			order = 2
			is_predefined = yes

			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					trigger_event = WJ_qingyi.9001
					scope:activity = { progress_activity_phase_after = { weeks = 1 } }
				}
				else_if = {
					limit = {
						is_ai = no
						is_alive = yes
						is_imprisoned = no
					}
					trigger_event = WJ_qingyi.9002
				}
			}
		}
	}

	on_start = {
		add_activity_log_entry = {
			key = WJ_qingyi_opening_log
			character = scope:host
			score = 10
		}
	}

	on_enter_passive_state = {
		trigger_event = WJ_qingyi.0001
	}

	on_complete = {
		if = {
			limit = {
				scope:activity = {
					NOT = { has_variable = wj_qingyi_rewards_disbursed }
				}
			}
			scope:host = { WJ_qingyi_disburse_rewards_effect = yes }
		}
	}

	###################
	# WINDOW CHARACTERS
	###################

	# Native activity-window character data, following the vanilla feast and
	# camp-party pattern. The generic activity GUI consumes this list directly.
	window_characters = {
		guest = {
			camera = camera_body_right

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					every_attending_character = {
						limit = {
							location = scope:activity.activity_location
							this != scope:host
							NOT = { scope:activity.special_guest:ming_shi ?= this }
						}
						add_to_list = characters
					}
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { drink_goblet happiness throne_room_conversation_3 throne_room_conversation_1 }
				}
			}
		}

		host = {
			camera = camera_body

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { toast_goblet pondering personality_bold }
				}
			}
		}

		ming_shi = {
			camera = camera_body

			effect = {
				if = {
					limit = {
						scope:host.location = scope:activity.activity_location
						exists = scope:activity.special_guest:ming_shi
						scope:activity.special_guest:ming_shi.location = scope:activity.activity_location
					}
					special_guest:ming_shi = { add_to_list = characters }
				}
				else = {
					every_attending_character = {
						limit = {
							location = scope:activity.activity_location
							this != scope:host
						}
						add_to_list = characters
					}
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { personality_honorable pondering storyteller }
				}
			}
		}

		guest = {
			camera = camera_body_left

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					every_attending_character = {
						limit = {
							location = scope:activity.activity_location
							this != scope:host
							NOT = { scope:activity.special_guest:ming_shi ?= this }
						}
						add_to_list = characters
					}
				}
			}

			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { drink_goblet laugh throne_room_conversation_2 }
				}
			}
		}
	}

	###################
	# GRAPHICS
	###################

	background = {
		texture = "gfx/interface/illustrations/event_scenes/tgp_feast_asia.dds"
		environment = "environment_tgp_feast_asia"
		ambience = "event:/DLC/EP4/SFX/Events/Event_Backgrounds/tgp_feast_asia"
	}
}
