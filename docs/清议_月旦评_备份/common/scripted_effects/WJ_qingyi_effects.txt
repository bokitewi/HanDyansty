# 清议单次活动经验上限与结算奖励。
# 会期临时变量只负责限额；永久清议经验直接写入资望原生轨道。

WJ_qingyi_prepare_participant_effect = {
	remove_variable = wj_qingyi_session_xp_eligible
	remove_variable = wj_qingyi_session_event_xp
	if = {
		limit = { NOT = { has_variable = wj_qingyi_qingtan_xp_year_lock } }
		set_variable = { name = wj_qingyi_session_xp_eligible value = yes }
		set_variable = { name = wj_qingyi_session_event_xp value = 0 }
	}
}

WJ_qingyi_add_qingtan_xp_2_effect = {
	if = {
		limit = {
			has_variable = wj_qingyi_session_xp_eligible
			OR = {
				AND = { this = scope:host var:wj_qingyi_session_event_xp <= 12 }
				AND = { this != scope:host var:wj_qingyi_session_event_xp <= 9 }
			}
		}
		change_variable = { name = wj_qingyi_session_event_xp add = 2 }
		WJ_ziwang_qingyi_xp_effect = { VALUE = 2 }
		set_variable = { name = wj_qingyi_qingtan_xp_year_lock value = yes years = 1 }
	}
}

WJ_qingyi_host_reward_effect = {
	add_prestige = medium_prestige_gain
	change_merit = medium_merit_gain
	if = {
		limit = { has_variable = wj_qingyi_session_xp_eligible }
		WJ_ziwang_qingyi_xp_effect = { VALUE = 6 }
		set_variable = { name = wj_qingyi_qingtan_xp_year_lock value = yes years = 1 }
	}
	WJ_qingyi_cleanup_participant_effect = yes
}

WJ_qingyi_guest_reward_effect = {
	add_prestige = minor_prestige_gain
	change_merit = minor_merit_gain

WJ_qingyi_disburse_rewards_effect = {
	save_scope_as = wj_qingyi_reward_host
	scope:activity = {
		set_variable = { name = wj_qingyi_rewards_disbursed value = yes }

		add_activity_log_entry = {
			key = WJ_qingyi_host_reward_log
			tags = { completed }
			character = scope:wj_qingyi_reward_host
			show_in_conclusion = yes
			score = 50
			scope:wj_qingyi_reward_host = { WJ_qingyi_host_reward_effect = yes }
		}

		add_activity_log_entry = {
			key = WJ_qingyi_guest_reward_log
			tags = { completed }
			character = scope:wj_qingyi_reward_host
			show_in_conclusion = yes
			score = 40
			every_attending_character = {
				limit = {
					is_alive = yes
					this != scope:wj_qingyi_reward_host
				}
				custom = WJ_qingyi_all_attending_guests
				WJ_qingyi_guest_reward_effect = yes
			}
		}
	}
}

WJ_qingyi_cleanup_participant_effect = {
	remove_variable = wj_qingyi_session_xp_eligible
	remove_variable = wj_qingyi_session_event_xp
}
