# Display-only API. Call in the viewed imperial Character scope on open/manual refresh.
# These three variables are snapshots, not appointments or an automatically updated roster.
WJ_baiguan_refresh_display_effect = {
	remove_variable = wj_baiguan_empress
	remove_variable = wj_baiguan_crown_prince
	remove_variable = wj_baiguan_capital_governor
	save_scope_as = wj_baiguan_emperor

	if = {
		limit = {
			exists = primary_spouse
            primary_spouse ?= { has_trait = SanGong_GongWei_1 }
			primary_spouse ?= {
				is_alive = yes
				top_liege = scope:wj_baiguan_emperor
			}
		}
		set_variable = { name = wj_baiguan_empress value = primary_spouse }
	}
	if = { limit = { exists = var:WJ_royal_heir var:WJ_royal_heir = { WJ_royal_crown_prince = yes } } set_variable = { name = wj_baiguan_crown_prince value = var:WJ_royal_heir } }
	if = {
		limit = {
			exists = capital_county.kingdom.holder
			capital_county.kingdom.holder ?= {
				is_alive = yes
				top_liege = scope:wj_baiguan_emperor
			}
		}
		set_variable = { name = wj_baiguan_capital_governor value = capital_county.kingdom.holder }
	}
}
