# ============================================================
# 汉家冠带 · 全局分发（开局 + debug 决议复用）
# 遍历所有可玩角色（伯爵及以上，即有头衔者）；
# 其文化拥有华夏礼乐 → 该角色家族（house）全员添加特质
# ============================================================
zgrc_distribute_hanjia_guandai_effect = {
	every_ruler = {
		limit = {
			# NOT = { has_trait = zgrc_hanjia_guandai }
			culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
		}
		house ?= {
			every_house_member = {
				add_trait = zgrc_hanjia_guandai
			}
		}
	}
}

# ============================================================
# 地域风俗 · 公国转化（DUCHY=公国id，CULTURE=目标汉文化分支）
# 将公国领地 county 文化、及头衔持有者（中华传承）与其宫廷，改写为目标汉文化分支
# ============================================================
zgrc_apply_duchy_culture_effect = {
	title:$DUCHY$ = {
		every_de_jure_county = {
			# if = {
			# 	limit = { culture = { has_cultural_pillar = heritage_chinese } }
			# 	set_county_culture = culture:$CULTURE$
			# }
			set_county_culture = culture:$CULTURE$
			# 伯爵领级别头衔持有者及其廷臣
			holder ?= {
				if = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = culture:$CULTURE$
					every_courtier = {
						limit = { culture = { has_cultural_pillar = heritage_chinese } }
						set_culture = culture:$CULTURE$
					}
				}
			}
		}
		holder ?= {
			if = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = culture:$CULTURE$
				every_courtier = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = culture:$CULTURE$
				}
			}
		}
	}
}

# ============================================================
# 地域风俗 · 伯爵领转化（COUNTY=伯爵领id，CULTURE=目标汉文化分支）
# 用途：公国内部跨越汉志分区时逐县改写
# 顺序：与公国级互不重叠 —— 公国级只写「本文化独有的公国」，
#       跨文化的公国（▼）一律不用公国级、全部按县列出，故无先后依赖
# 语义：与公国级完全一致 —— county 文化强制写；
#       持有者与其宫廷仅"中华传承"者改写
# ============================================================
zgrc_apply_county_culture_effect = {
	title:$COUNTY$ = {
		set_county_culture = culture:$CULTURE$
		holder ?= {
			if = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = culture:$CULTURE$
				every_courtier = {
					limit = { culture = { has_cultural_pillar = heritage_chinese } }
					set_culture = culture:$CULTURE$
				}
			}
		}
	}
}

# 公羊学 · 角色转化（RITE=目标学派）：须信儒家；1178 起仅非统治者儒生
zgrc_set_gongyang_rite_effect = {
	if = {
		limit = {
			faith = faith:confucian_faith
			OR = {
				NOT = { game_start_date >= 1178.10.1 }
				is_ruler = no
			}
		}
		set_character_rite = rite:$RITE$
	}
	every_courtier = {
		if = {
			limit = {
				faith = faith:confucian_faith
				OR = {
					NOT = { game_start_date >= 1178.10.1 }
					is_ruler = no
				}
			}
			set_character_rite = rite:$RITE$
		}
	}
}

# 公羊学 · 公国转化（DUCHY / RITE）：县固定改；人只改信儒家者
zgrc_apply_duchy_rite_effect = {
	title:$DUCHY$ = {
		every_de_jure_county = {
			set_county_rite = rite:$RITE$
			holder ?= { zgrc_set_gongyang_rite_effect = { RITE = $RITE$ } }
		}
		holder ?= { zgrc_set_gongyang_rite_effect = { RITE = $RITE$ } }
	}
}

zgrc_apply_ruler_local_culture_effect = {
	every_ruler = {
		limit = {
			culture = culture:han
		}
		if = {
			limit = {
				highest_held_title_tier >= tier_empire
			}
			set_culture = title:h_china.holder.culture
			every_courtier = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = title:h_china.holder.culture
			}
			every_vassal = {
				limit = { culture = { has_cultural_pillar = heritage_chinese } }
				set_culture = title:h_china.holder.culture
			}
		}
		else = {
			if = {
				limit = {

				}

				every_courtier = {

				}
				every_vassal = {

# ============================================================
# 革新同步 —— 同步 effect
# 触发：yearly_culture_pulse + 全局计数为 0（见 on_action）
# 逻辑：随机选一个已知革新 → 同步给首领最高领主治下全部华夏礼乐文化
# ============================================================
zgrc_innovation_sync_effect = {
	# 随机选一个已知革新
	random_known_innovation = {
		save_scope_as = chosen_innovation
	}
	# 未选中（该文化无已知革新）→ 跳过同步
	if = {
		limit = { exists = scope:chosen_innovation }
		# 同步给首领最高领主治下全部华夏礼乐文化
		culture_head = {
			top_liege = {
				every_sub_realm_county = {
					culture = {
						if = {
							limit = {
								has_cultural_tradition = tradition_zgrc_hua_xia_li_yue
								NOT = { has_innovation = scope:chosen_innovation }
							}
							add_innovation = scope:chosen_innovation
							# 消息通知：该文化有领袖 → 通知其文化领袖
							if = {
								limit = { exists = culture_head }
								culture_head = {
									send_interface_toast = {
										type = event_toast_effect_good
										title = zgrc_innovation_sync_toast_title
										left_icon = this
										custom_tooltip = zgrc_innovation_sync_toast_desc
									}
								}
							}
						}
					}
				}
			}
		}
	}
}

# 天下响应 · 封臣表态（on_war_started，War 作用域）
# 设计基线：文化向（华夏礼乐 + 厚生慎死）。完全忽略政体：客观底线只用 is_landed。
# 流程：大事判定（2 条 AND + 战争性质三路 OR）→ 遍历防御方封臣 → 触发事件 1010
# 传参：var zgrc_luanshi_war（战争）
##############################
# 大事判定（War 作用域）：
#   ① 防御方为「独立君主」且**帝国（含霸主）级以上**、**正统性等级 ≥ 4**，且持「华夏礼乐」传统——社稷级战事。
#      注1：防御方若本身是藩属，其封臣的 top_liege 便不是防御方，故须排除，以免误认君父；
#      注2：防御方若无华夏礼乐，则属蛮夷内乱，与我华夏无涉（麾下华裔封臣亦不作响应）；
#   ② 来犯者成气候：攻击方军力 ≥ 防御方 × zgrc_luanshi_ratio_lo（0.8），上不封顶；
#   ③ 战争性质够重（三路 OR，兼收「瞄准头衔」与「针对整国」两类 CB）：
#      S1 社稷之争：目标含帝国级头衔 / 直指任一方首都（瞄准头衔型）；
#      S2 亡国之战：亡国级 CB 白名单（zgrc_luanshi_annihilation_cb_trigger）——
#         蒙古入侵、大圣战、绝罚解散帝国、天朝大征伐等「针对整国」之战，
#         其 target_titles 为空或仅「邻接领地」，既无帝国级头衔也不含首都，
#         无法用头衔层级判定捕获，只能按 CB 识别；
#      S3 主头衔被夺：目标直指防御方最高头衔（兜底）。
##############################
zgrc_tianxia_xiangying_effect = {
	if = {
		limit = {
			# ① 防御方为独立君主，且帝国（含霸主）级以上，且君父须有华夏礼乐：社稷级战事
			scope:war.primary_defender = {
				is_independent_ruler = yes
				highest_held_title_tier >= tier_empire
				# 正统性：君父须有足够正统（等级 ≥ 4）方可谓社稷所系
				legitimacy_level >= 4
				culture = { has_cultural_tradition = tradition_zgrc_hua_xia_li_yue }
			}
			# ② 来犯者成气候：攻击方 ≥ 防御方 × ratio_lo（愈强愈该响应，上不封顶）
			scope:war.primary_attacker.current_military_strength >= {
				value = {
					value = scope:war.primary_defender.current_military_strength
					multiply = zgrc_luanshi_ratio_lo
				}
			}
			# ③ 战争性质够重：三路 OR
			OR = {
				# S1 社稷之争：目标含帝国级头衔 / 直指任一方首都
				AND = {
					# 空安全：CB 弱引用失效时不下钻（on_war_started 边缘时序）
					scope:war = { exists = casus_belli }
					scope:war.casus_belli = {
						OR = {
							any_in_list = {
								list = target_titles
								tier >= tier_empire
							}
							any_in_list = {
								list = target_titles
								OR = {
									title_province ?= { this = scope:war.primary_defender.capital_province }
									title_province ?= { this = scope:war.primary_attacker.capital_province }
									any_de_jure_county = { this = scope:war.primary_defender.capital_county }
									any_de_jure_county = { this = scope:war.primary_attacker.capital_county }
								}
							}
						}
					}
				}
				# S2 亡国之战：亡国级 CB 白名单（针对整国而非头衔的战争）
				scope:war = { zgrc_luanshi_annihilation_cb_trigger = yes }
				# S3 主头衔被夺：目标直指防御方最高头衔
				AND = {
					# 空安全：CB 弱引用失效时不下钻（on_war_started 边缘时序）
					scope:war = { exists = casus_belli }
					scope:war.casus_belli = {
						any_in_list = {
							list = target_titles
							this = scope:war.primary_defender.primary_title
						}
					}
				}
			}
		}
		zgrc_luanshi_ask_effect = yes
	}
}

# 遍历合格封臣：传参 + 发问（全部 / 仅玩家 由 game_rule 决定）
zgrc_luanshi_ask_effect = {
	scope:war.primary_defender = {
		every_vassal_or_below = {
			limit = {
				# ① 客观底线：有实土（忽略政体）
				is_landed = yes
				# ② 文化底线：厚生慎死（机制绑定）
				culture = { has_cultural_tradition = tradition_zgrc_hou_sheng_shen_si }
				# ③ 冷却底线：同一人 10 年内不重复被问
				NOT = { has_character_flag = zgrc_luanshi_asked }
			}
			# ④ 规则判定：全部（含 AI）/ 仅玩家
			if = {
				limit = {
					OR = {
						has_game_rule = zgrc_luanshi_xiangying_enabled
						AND = {
							has_game_rule = zgrc_luanshi_xiangying_player_only
							is_ai = no
						}
					}
				}
				set_variable = { name = zgrc_luanshi_war value = scope:war }
				add_character_flag = { flag = zgrc_luanshi_asked years = zgrc_luanshi_cooldown_years }
				if = {
					limit = { is_ai = yes }
					random = {
						chance = zgrc_luanshi_ai_chance
						trigger_event = { id = zgrc_common.1010 }
					}
				}
				else = {
					trigger_event = { id = zgrc_common.1010 }
				}
			}
		}
	}
}

# 祖风 · 跨大区迁居（出生时判定，「后代风俗」的配套机制）
# 六大方区（汉书风俗大区）：
#   yanzhao 燕赵·北方（燕 赵 晋 河内 北地）
#   qilu    齐鲁·东方（齐 鲁 宋）

#   qinlong 秦陇·巴蜀（关中 河东 西北 河西 巴蜀）
#   jingwu  荆吴·南方（楚 汝南 吴 粤）
#   dongyi  东夷·辽海（东北）
# 调用顺序（同文化改属前记录，改文化后挂载）：
#   ① zgrc_zufeng_set_origin_effect = yes   ← 记录旧文化大区，须在 set_culture 之前
#   ② set_culture = ...                     ← 出生：location.county.culture
#                                             入乡随俗：capital_county.culture
#   ③ zgrc_zufeng_clear_effect = yes        ← 清旧祖风（全量、有则移除），二次迁居时才需要
#   ④ zgrc_zufeng_apply_effect = yes        ← 按新文化大区挂「某祖风」modifier
# 同大区内跨风俗不改祖风；不在六区内者（无大区归属）不挂；任何时刻至多 1 条祖风。
# 调用点（2 处）：
#   · on_action zgrc_hanjia_children_on_birth —— 新生儿随出生地风俗（① ② ④）
#   · 事件 zgrc_common.0002 A「入乡随俗」—— 久居他乡改从当地风俗（① ② ③ ④）
# ============================================================

# ① 记录祖上（继承）大区 → 变量 zgrc_zufeng_from（存 flag 值）
zgrc_zufeng_set_origin_effect = {
	# 默认：不在六区内（无祖风）
	set_variable = { name = zgrc_zufeng_from value = flag:none }
	# 燕赵·北方
	if = {
		limit = {
			OR = {
				culture = { this = culture:youzhou }
				culture = { this = culture:jizhou }
				culture = { this = culture:bingzhou }
				culture = { this = culture:hd_henei }
				culture = { this = culture:hd_beidi }
			}
		}
		set_variable = { name = zgrc_zufeng_from value = flag:yanzhao }
	}
	# 齐鲁·东方
	else_if = {
		limit = {
			OR = {
				culture = { this = culture:hd_qi_state }
				culture = { this = culture:qingzhou }
				culture = { this = culture:yanzhou }
			}
		}
		set_variable = { name = zgrc_zufeng_from value = flag:qilu }
	}
	# 三河·中原
	else_if = {
		limit = {
			OR = {
				culture = { this = culture:hd_zhou_state }
				culture = { this = culture:hd_han_state }
				culture = { this = culture:hd_zheng_state }
				culture = { this = culture:hd_wei_state }
				culture = { this = culture:hd_chen_state }
				culture = { this = culture:yuzhou }
			}
		}
		set_variable = { name = zgrc_zufeng_from value = flag:sanhe }
	}
	# 秦陇·巴蜀
	else_if = {
		limit = {
			OR = {
				culture = { this = culture:han }
				culture = { this = culture:hd_hedong }

# ============================================================
# 大复仇 · 亲人被害 → 直系亲属复仇抉择（root = 死者）
# ============================================================
# 闸①：死者死有余辜（被合法处决）→ 不复仇（断环，防无限复仇链）
# 闸②：凶手已知（scope:killer 存在，见 death.txt:3「If a killer is known」）
#        秘密谋杀未揭发时 killer 不存在 → 家人无从复仇
# 范围：父母 + 子女 + 兄弟姐妹
# 事件：zgrc_common.4001（玩家自选；AI 按 ai_chance 权重自选）
# ============================================================
zgrc_dachou_on_death_effect = {
	save_scope_as = zgrc_victim

	if = {
		limit = {
			exists = scope:killer
			NOT = { scope:killer = root }
			# 被合法处决者＝罪犯，死有余辜：亲属不得对行刑者复仇
			NOT = { scope:death_reason ?= flag:death_execution }
		}

		scope:killer = { save_scope_as = zgrc_killer }

		# —— 子嗣 ——
		every_child = {
			limit = { zgrc_can_revenge_trigger = yes }
			trigger_event = { id = zgrc_common.4001 days = 3 }
		}

		# —— 兄弟姐妹 ——
		every_sibling = {
			limit = { zgrc_can_revenge_trigger = yes }
			trigger_event = { id = zgrc_common.4001 days = 3 }
		}

		# —— 父亲 ——
		if = {
			limit = { exists = father }
			father = {
				if = {
					limit = { zgrc_can_revenge_trigger = yes }
					trigger_event = { id = zgrc_common.4001 days = 3 }
				}
			}
		}

		# —— 母亲 ——
		if = {
			limit = { exists = mother }
			mother = {
				if = {
					limit = { zgrc_can_revenge_trigger = yes }
					trigger_event = { id = zgrc_common.4001 days = 3 }
				}
			}
		}
	}
}
