# 天朝循环大修：斗争期内容。Scope：dynastic_cycle 局势（年度）

hd_struggle_yearly_effect = {
	# 游荡奇才：每 10 年 1-5 名
	if = {
		limit = { NOT = { has_variable = hd_struggle_talent_years } }
		set_variable = { name = hd_struggle_talent_years value = 0 }
	}
	change_variable = { name = hd_struggle_talent_years add = 1 }
	if = {
		limit = { var:hd_struggle_talent_years >= hd_struggle_talent_interval_years }
		set_variable = { name = hd_struggle_talent_years value = 0 }
		random_list = {
			20 = { hd_struggle_spawn_talent_effect = yes }
			20 = { hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes }
			20 = { hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes }
			20 = { hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes }
			20 = { hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes hd_struggle_spawn_talent_effect = yes }
		}
	}

	# 军事冒险者：没有怯懦/安于现状的 AI 冒险者补建 story_adventurer_ai
	every_independent_ruler = {
		limit = {
			is_ai = yes
			government_has_flag = government_is_landless_adventurer
			NOT = { owns_story_of_type = story_adventurer_ai }
			NOR = {
				has_trait = craven
				has_trait = content
			}
			ep3_adventurer_ai_logic_valid_trigger = yes
			location ?= { county ?= { target_is_de_jure_liege_or_above = title:h_china } }
		}
		create_story = { type = story_adventurer_ai }
	}
}

# 在随机汉地县生成一名高属性无地角色（进入角色池）
hd_struggle_spawn_talent_effect = {
	title:h_china = {
		random_de_jure_county = {
			limit = { development_level >= 5 }
			save_scope_as = hd_talent_county
		}
	}
	if = {
		limit = { exists = scope:hd_talent_county }
		create_character = {
			location = scope:hd_talent_county.title_province
			culture = scope:hd_talent_county.culture
			faith = scope:hd_talent_county.faith
			gender_female_chance = 0
			age = { 16 50 }
			diplomacy = { 6 30 }
			martial = { 6 30 }
			stewardship = { 6 30 }
			intrigue = { 6 30 }
			learning = { 6 30 }
			prowess = { 6 30 }
			random_traits = yes
			save_scope_as = hd_talent
		}
		scope:hd_talent = {
			random_list = {
				1 = { random_list = { 1 = { add_trait = education_diplomacy_3 } 1 = { add_trait = education_martial_3 } 1 = { add_trait = education_stewardship_3 } 1 = { add_trait = education_intrigue_3 } 1 = { add_trait = education_learning_3 } } }
				1 = { random_list = { 1 = { add_trait = education_diplomacy_4 } 1 = { add_trait = education_martial_4 } 1 = { add_trait = education_stewardship_4 } 1 = { add_trait = education_intrigue_4 } 1 = { add_trait = education_learning_4 } } }
				1 = { random_list = { 1 = { add_trait = education_diplomacy_5 } 1 = { add_trait = education_martial_5 } 1 = { add_trait = education_stewardship_5 } 1 = { add_trait = education_intrigue_5 } 1 = { add_trait = education_learning_5 } } }
			}
		}
		clear_saved_scope = hd_talent_county
	}
}
