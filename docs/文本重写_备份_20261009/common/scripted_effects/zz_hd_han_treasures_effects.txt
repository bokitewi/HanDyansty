# 汉代宝物（书籍奇珍）创建效果 - Han treasure artifact creation effects
# 依据《汉代宝物_书籍奇珍_CK3完整制作表》制作；接口：$OWNER$ = 持有者角色
# 首版美术全部复用原版visuals；开局发放仅限玉玺/王莽头/和氏璧/太平要术（见 zz_hd_han_treasures_startup.txt）
# 其余宝物的创建效果供后续事件链调用，暂无生成入口

# 编号1 《孙子兵法》善本 - 兵书
hd_han_create_sunzi_bingfa_effect = {
	$OWNER$ = {
		save_scope_as = owner
		create_artifact = {
			name = han_artifact_sunzi_bingfa_name
			description = han_artifact_sunzi_bingfa_description
			rarity = illustrious
			type = book
			template = hd_han_unique_book_template
			visuals = book
			wealth = 100
			quality = 100
			max_durability = 120
			decaying = no
			history = { type = created_before_history }
			modifier = artifact_martial_2_modifier
			save_scope_as = newly_created_artifact
		}
		scope:newly_created_artifact = {
			add_artifact_modifier = artifact_monthly_martial_lifestyle_xp_4_modifier
			add_artifact_modifier = artifact_controlled_province_advantage_3_modifier
			add_artifact_modifier = artifact_knight_effectiveness_7_modifier
			set_variable = { name = historical_unique_artifact value = yes }
			set_variable = han_artifact_sunzi_bingfa
		}
	}
}
