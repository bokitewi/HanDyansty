# Main instance scope: retain actual contracts until every pending operation ends.
hd_dc_close_completed_instance_effect = {
	if = {
		limit = {
			NOT = { var:hd_dc_stage = flag:closed }
			OR = { has_variable = hd_dc_political_result has_variable = hd_dc_par01_node_closed }
			var:hd_dc_open_obligations = 0
			NOT = { has_variable = hd_dc_obligation_count_error }
			hd_dc_all_core_wars_resolved_trigger = yes
			NOT = { any_in_list = { variable = hd_dc_actor_stories OR = { has_variable = hd_dc_pending_action has_variable = hd_dc_pending_proposal } } }
			NOT = { has_variable = hd_dc_audit_wait }
		}
		set_variable = { name = hd_dc_stage value = flag:closed }
		set_variable = { name = hd_dc_closed_date value = current_date }
		hd_dc_release_own_acute_lock_effect = yes
	}
}


hd_dc_release_own_acute_lock_effect = {
	save_scope_as = hd_dc_releasing_instance
	var:hd_dc_region ?= {
		if = {
			limit = {
				var:hd_dc_acute_instance = scope:hd_dc_releasing_instance
				var:hd_dc_acute_instance_id = scope:hd_dc_releasing_instance.var:hd_dc_instance_id
			}
			remove_variable = hd_dc_acute_instance
			remove_variable = hd_dc_acute_instance_id
			remove_variable = hd_dc_acute_type
		}
	}
}
