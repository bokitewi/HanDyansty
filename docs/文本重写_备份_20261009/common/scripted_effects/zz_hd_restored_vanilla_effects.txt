human_sacrifice_personal_execution_piety_bonus_effect = {
	if = {
		limit = { highest_held_title_tier >= tier_hegemony }
		scope:executioner = {
			add_piety = massive_piety_gain
			change_spiritual_fulfillment = major_spiritual_fulfillment_value
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_empire }
		scope:executioner = {
			add_piety = major_piety_gain
			change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		scope:executioner = {
			add_piety = {
				value = major_piety_gain
				multiply = 0.5
			}
			change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_duchy }
		scope:executioner = {
			add_piety = minor_piety_gain
			change_spiritual_fulfillment = miniscule_spiritual_fulfillment_value
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_county }
		scope:executioner = {
			add_piety = miniscule_piety_gain
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_barony }
		scope:executioner = {
			add_piety = {
				value = miniscule_piety_gain
				multiply = 0.75
			}
		}
	}
	else_if = {
		limit = { is_lowborn = no }
		scope:executioner = {
			add_piety = {
				value = miniscule_piety_gain
				multiply = 0.5
			}
		}
	}
	else = {
		scope:executioner = {
			add_piety = {
				value = miniscule_piety_gain
				multiply = 0.4
			}
		}
	}
}

clerical_kill_piety_effect = {
	if = {
		limit = {
			exists = faith.religious_head
			faith.religious_head = $VICTIM$
			NOT = { has_perk = divine_retribution_perk }
		}
		if = {
			limit = {
				situation:the_christian_church ?= {
					has_situation_top_phase_parameter = the_christian_church_clerical_violence_less_consequential
				}
			}
			add_piety_level = -1
		}
		else = {
			add_piety_level = -2
		}
	}
	else_if = {
		limit = {
			NOT = { has_perk = divine_retribution_perk }
			$VICTIM$ = {
				pam_is_senior_cleric_of_character_trigger = {
					CHARACTER = $KILLER$
				}
			}
		}
		if = {
			limit = {
				situation:the_christian_church ?= {
					has_situation_top_phase_parameter = the_christian_church_clerical_violence_less_consequential
				}
			}
			add_piety = medium_piety_loss
		}
		else = {
			add_piety = major_piety_loss
		}
	}
}

abduction_rivalry_progress_effect = {
	scope:target = {
		#Target is a ruler
		if = {
			limit = {
				abduct_ruler_rivalling_trigger = yes
			}
			if = {
				limit = {
					is_close_family_of = scope:owner
				}
				set_relation_rival = {
					target = scope:owner
					reason = rival_abducted_family
				}
			}
			else = {
				progress_towards_rival_effect = {
					CHARACTER = scope:owner
					REASON = rival_abducted_ruler
					OPINION = 0
				}
			}
		}
		#Target is close family
		if = {
			limit = {
				abduct_family_rivalling_trigger = yes
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:owner
				REASON = rival_abducted_family
				OPINION = 0
			}
		}
		#Target has parents that are rulers of superior power
		if = {
			limit = {
				any_parent = {
					abduct_rivalling_relative_trigger = yes
				}
			}
			every_parent = {
				limit = {
					abduct_rivalling_relative_trigger = yes
				}
				progress_towards_rival_effect = {
					CHARACTER = scope:owner
					REASON = rival_abducted_child
					OPINION = 0
				}
			}
		}
		#Target has children that are rulers of superior power
		if = {
			limit = {
				any_child = {
					abduct_rivalling_relative_trigger = yes
				}
			}
			every_child = {
				limit = {
					abduct_rivalling_relative_trigger = yes
				}
				progress_towards_rival_effect = {
					CHARACTER = scope:owner
					REASON = rival_abducted_parent
					OPINION = 0
				}
			}
		}
		#Target has siblings that are rulers of comparable power
		if = {
			limit = {
				any_sibling = {
					scope:target = {
						is_courtier_of = prev
					}
					#stupid incest making things complicated
					NOR = {
						is_parent_of = scope:target
						is_child_of = scope:target
					}
					abduct_rivalling_relative_trigger = yes
				}
			}
			liege = {
				progress_towards_rival_effect = {
					CHARACTER = scope:owner
					REASON = rival_abducted_sibling
					OPINION = 0
				}
			}
		}
		#Target has spouse that is of superior power
		if = {
			limit = {
				any_spouse = {
					abduct_rivalling_relative_trigger = yes
					#Prevent incest duplicating this effect
					NOR = {
						is_parent_of = scope:target
						is_child_of = scope:target
						is_sibling_of = scope:target
					}
				}
			}
			every_spouse = {
				limit = {
					abduct_rivalling_relative_trigger = yes
					#Prevent incest duplicating this effect
					NOR = {
						is_parent_of = scope:target
						is_child_of = scope:target
						is_sibling_of = scope:target
					}
				}
				progress_towards_rival_effect = {
					CHARACTER = scope:owner
					REASON = rival_abducted_spouse
					OPINION = 0
				}
			}
		}
	}
}

				location != prev.capital_province

			}
			debug_log = "Courtier not updated in migration"
			debug_log_scopes = no
			set_location_to_default = yes
		}
		every_prisoner = {
			limit = { location != prev.capital_province }
			debug_log = "Prisoner not updated in migration"
			debug_log_scopes = no
			set_location_to_default = yes
		}
	}
}
