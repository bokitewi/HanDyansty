# 天朝循环大修：阶段切换辅助 effect。Scope：dynastic_cycle 局势。

# 新阶段开始时的通用记录。$PHASE$ = 阶段 key 去掉前缀；$NUMBER$ = last_recorded_phase_change 编号
# 编号：6 斗争(chaos) / 7 恢复 / 8 崩溃 / 9 对峙
hd_cycle_phase_started_effect = {
	record_situation_special_event = {
		key = situation_dynastic_cycle_era_change_history_entry
		phase = situation_dynastic_cycle_phase_$PHASE$
	}
	set_variable = { name = last_recorded_phase_change value = $NUMBER$ }
	hd_dc_request_observation_effect = yes
	# 四夷旧 modifier 已由四夷参与者组取代，这里只清理旧存档残留
	hd_cycle_siyi_clear_effect = yes
	# 天朝循环大修·朋党：青睐复制效果按新时代刷新
	every_situation_participant = {
		limit = { is_alive = yes has_variable = hd_party_court }
		hd_party_apply_favor_copy_effect = yes
	}
	if = {
		limit = { hd_cycle_is_luanshi_trigger = yes }
		hd_strat_grant_all_effect = yes
	}
}

# 崩溃期节律计数器归零（危机 / 派系 / 灾害 / 冒险者）
hd_cycle_collapse_reset_counters_effect = {
	set_variable = { name = hd_collapse_crisis_years value = 0 }
	set_variable = { name = hd_collapse_faction_years value = 0 }
	set_variable = { name = hd_collapse_disaster_years value = 0 }
	set_variable = { name = hd_collapse_disaster_chance value = hd_collapse_disaster_base_chance }
	set_variable = { name = hd_collapse_adventurer_years value = 0 }
}

# 进入乱世阶段时立即发放战略 AI story（季度脉冲也会补发）
hd_strat_grant_all_effect = {
	every_independent_ruler = {
		limit = {
			is_ai = yes
			highest_held_title_tier >= tier_duchy
			hd_strat_eligible_trigger = yes
			NOT = { owns_story_of_type = hd_strategic_ai_story }
		}
		create_story = hd_strategic_ai_story
	}
}

# 乱世结束时结束全部战略 AI story
hd_strat_end_all_effect = {
	every_independent_ruler = {
		limit = { owns_story_of_type = hd_strategic_ai_story }
		every_owned_story = {
			limit = { story_type = hd_strategic_ai_story }
			end_story = yes
		}
	}
}

# Scope：霸权持有者。霸权者只能拥有一个实地帝国头衔：保留 e_han（优先）或任意一个，其余实地帝国头衔销毁。
hd_cycle_hegemon_empire_purge_effect = {
	if = {
		limit = { has_title = title:h_china }
		if = {
			limit = { has_title = title:e_han }
			title:e_han = { save_temporary_scope_as = hd_keep_empire }
		}
		else_if = {
			limit = {
				any_held_title = {
					tier = tier_empire
					is_titular = no
					is_landless_type_title = no # 天朝官职（e_minister_*）也是帝国级，不能被当作实地帝国
					is_noble_family_title = no
				}
			}
			random_held_title = {
				limit = {
					tier = tier_empire
					is_titular = no
					is_landless_type_title = no # 天朝官职（e_minister_*）也是帝国级，不能被当作实地帝国
					is_noble_family_title = no
				}
				save_temporary_scope_as = hd_keep_empire
			}
		}
		if = {
			limit = { exists = scope:hd_keep_empire }
			every_held_title = {
				limit = {
					tier = tier_empire
					is_titular = no
					is_landless_type_title = no # 天朝官职（e_minister_*）也是帝国级，不能被当作实地帝国
					is_noble_family_title = no
					NOT = { this = scope:hd_keep_empire }
				}
				root = { destroy_title = prev }
			}
		}
	}
}
