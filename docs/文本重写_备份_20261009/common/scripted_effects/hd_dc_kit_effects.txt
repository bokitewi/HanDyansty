# 天朝循环·时代特色重制：诱因发射包装（见 docs/天朝循环_时代特色重制计划.md §2、§6 Q5/Q9/Q10）
# 顺序：当前阶段列有该诱因（廉价短路）→ 行为人是 VIP → 行为人 3 年冷却 → 该诱因全局每年 10 次上限 → 发射

# 行为诱因。$CATALYST$ = 诱因 key；$ACTOR$ = 行为人 scope（如 root / scope:attacker）
hd_dc_kit_fire_effect = {
	# 先在调用处保存行为人：$ACTOR$ 可能是 this，进入局势 scope 后会指向局势
	clear_saved_scope = hd_dc_kit_actor
	$ACTOR$ ?= { save_scope_as = hd_dc_kit_actor }
	situation:dynastic_cycle ?= {
		if = {
			limit = {
				exists = scope:hd_dc_kit_actor
				situation_top_has_catalyst = $CATALYST$
				scope:hd_dc_kit_actor = {
					hd_dc_is_vip_trigger = yes
					NOT = { has_character_flag = hd_dc_cd_$CATALYST$ }
				}
				OR = {
					NOT = { has_variable = hd_dc_cap_$CATALYST$ }
					var:hd_dc_cap_$CATALYST$ < hd_dc_catalyst_yearly_cap
				}
			}
			scope:hd_dc_kit_actor = { add_character_flag = { flag = hd_dc_cd_$CATALYST$ years = hd_dc_vip_cooldown_years } }
			hd_dc_kit_count_cap_effect = { CATALYST = $CATALYST$ }
			trigger_situation_catalyst = {
				catalyst = $CATALYST$
				character = scope:hd_dc_kit_actor
			}
		}
	}
}

# 无行为人或非 VIP 口径的诱因（年度统计、局势级事件）。只受全局年上限约束
hd_dc_kit_fire_global_effect = {
	situation:dynastic_cycle ?= {
		if = {
			limit = {
				situation_top_has_catalyst = $CATALYST$
				OR = {
					NOT = { has_variable = hd_dc_cap_$CATALYST$ }
					var:hd_dc_cap_$CATALYST$ < hd_dc_catalyst_yearly_cap
				}
			}
			hd_dc_kit_count_cap_effect = { CATALYST = $CATALYST$ }
			trigger_situation_catalyst = $CATALYST$
		}
	}
}

# Scope：dynastic_cycle 局势。年上限计数（变量 1 年过期即清零）
hd_dc_kit_count_cap_effect = {
	if = {
		limit = { NOT = { has_variable = hd_dc_cap_$CATALYST$ } }
		set_variable = { name = hd_dc_cap_$CATALYST$ value = 0 years = 1 }
	}
	change_variable = { name = hd_dc_cap_$CATALYST$ add = 1 }
}
