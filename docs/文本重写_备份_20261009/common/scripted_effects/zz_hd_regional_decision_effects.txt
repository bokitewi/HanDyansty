hd_pacify_nanman_effect = {
	$TITLE$.title_capital_county = { save_scope_as = hd_nanman_capital }
	create_character = {
		template = new_commander_character
		age = 30
		gender_female_chance = 0
		culture = scope:hd_nanman_capital.culture
		rite = scope:hd_nanman_capital.rite
		location = scope:hd_nanman_capital.title_province
		save_scope_as = hd_nanman_ruler
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_nanman_change
		add_claim_on_loss = no
	}
	scope:hd_nanman_capital = {
		change_title_holder = {
			holder = scope:hd_nanman_ruler
			change = scope:hd_nanman_change
		}
	}
	$TITLE$ = {
		change_title_holder = {
			holder = scope:hd_nanman_ruler
			change = scope:hd_nanman_change
		}
	}
	$TITLE$ = {
		every_de_jure_county = {
			limit = {
				holder ?= {
					this != root
					this != scope:hd_nanman_ruler
					highest_held_title_tier = tier_county
					any_liege_or_above = { this = root }
				}
			}
			holder = {
				change_liege = {
					liege = scope:hd_nanman_ruler
					change = scope:hd_nanman_change
				}
			}
		}
	}
	resolve_title_and_vassal_change = scope:hd_nanman_change
	scope:hd_nanman_ruler = { change_government = tribal_government }
	$TITLE$ = {
		every_de_jure_county = {
			limit = {
				holder ?= {
					OR = {
						this = root
						any_liege_or_above = { this = root }
					}
				}
			}
			remove_county_modifier = hd_satisfied_nanman_modifier
			add_county_modifier = {
				modifier = hd_satisfied_nanman_modifier
				years = 30
			}
		}
	}
	scope:hd_nanman_ruler.culture = {
		if = {
			limit = {
				NOT = { has_variable = hd_nanman_acceptance_granted }
				this != root.culture
			}
			change_cultural_acceptance = {
				target = root.culture
				value = 3
				desc = hd_nanman_acceptance_gain
			}
			set_variable = { name = hd_nanman_acceptance_granted value = yes }
		}
	}
	add_character_flag = { flag = hd_pacify_nanman_cooldown years = 5 }
}

# 五胡内附·第一层（见 docs/五胡内附_僚人入蜀_设计草案.md 4.1）。
# ROOT = 决定使用者；需事先保存 scope:hd_wuhu_tribe。
hd_complete_inward_migration_effect = {
	$TITLE$ = { save_scope_as = hd_migration_target }
	hd_wuhu_save_tribe_culture_rite_effect = { COUNTY = scope:hd_migration_target }

	# 强割直属封臣的伯爵领：该封臣好感 -20，持续 5 年
	if = {
		limit = {
			exists = scope:hd_migration_target.holder
			NOT = { scope:hd_migration_target.holder = root }
		}
		scope:hd_migration_target.holder = {
			add_opinion = {
				target = root
				modifier = hd_wuhu_land_seized_opinion
				years = 5
			}
		}
	}

	hd_wuhu_create_chief_effect = { LOCATION = scope:hd_migration_target.title_province }
	scope:hd_wuhu_chief = { save_scope_as = hd_migrant_ruler }

	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_migration_change
		add_claim_on_loss = no
	}
	scope:hd_migration_target = {
		change_title_holder = {
			holder = scope:hd_migrant_ruler
			change = scope:hd_migration_change
			take_baronies = yes
		}
	}
	resolve_title_and_vassal_change = scope:hd_migration_change

	# 胡首成为决定使用者的直属封臣
	if = {
		limit = {
			NOT = { scope:hd_migrant_ruler.liege ?= root }
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_migration_liege_change
			add_claim_on_loss = no
		}
		scope:hd_migrant_ruler = {
			change_liege = {
				liege = root
				change = scope:hd_migration_liege_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_migration_liege_change
	}

	# 伯爵领文化与信仰都改为该胡族的
	scope:hd_migration_target = {
		hd_wuhu_flip_county_culture_rite_effect = yes
		remove_county_modifier = hd_inward_migration_modifier
		add_county_modifier = {
			modifier = hd_inward_migration_modifier
			years = 30
		}
	}
	scope:hd_migrant_ruler = {
		hd_wuhu_apply_chief_government_effect = yes
		hd_wuhu_mark_settled_effect = yes
	}
	add_character_flag = { flag = hd_inward_migration_cooldown years = 10 }
}

hd_partition_han_effect = {
	if = {
		limit = { has_character_flag = hd_partition_choose_west }
		root = { save_scope_as = hd_west_han_holder }
		$RECIPIENT$ = { save_scope_as = hd_east_han_holder }
	}
	else = {
		root = { save_scope_as = hd_east_han_holder }
		$RECIPIENT$ = { save_scope_as = hd_west_han_holder }
	}

	title:d_jingzhao = { set_de_jure_liege_title = title:k_hd_guanzhong }
	title:d_fufeng = { set_de_jure_liege_title = title:k_hd_guanzhong }
	title:d_xiabeidi = { set_de_jure_liege_title = title:k_hd_guanzhong }
	title:d_shangjun_sili = { set_de_jure_liege_title = title:k_hd_guanzhong }
	title:d_fengyi = { set_de_jure_liege_title = title:k_hd_guanzhong }
	title:k_hd_guanzhong = { set_de_jure_liege_title = title:e_hd_western_han }
	title:k_uuii_liangzhou = { set_de_jure_liege_title = title:e_hd_western_han }
	title:k_uuii_yizhou = { set_de_jure_liege_title = title:e_hd_western_han }

	create_title_and_vassal_change = {
		type = granted
		save_scope_as = hd_partition_change
		add_claim_on_loss = no
	}
	title:c_uuii_p_867 = {
		change_title_holder = {
			holder = scope:hd_west_han_holder
			change = scope:hd_partition_change
		}
	}
	title:c_luoyang = {
		change_title_holder = {
			holder = scope:hd_east_han_holder
			change = scope:hd_partition_change
		}
	}
	title:k_hd_guanzhong = {
		change_title_holder = {
			holder = scope:hd_west_han_holder
			change = scope:hd_partition_change
		}
	}
	title:e_hd_western_han = {
		change_title_holder = {
			holder = scope:hd_west_han_holder
			change = scope:hd_partition_change
		}
	}
	if = {
		limit = { root != scope:hd_east_han_holder }
		title:e_han = {
			change_title_holder = {
				holder = scope:hd_east_han_holder
				change = scope:hd_partition_change
			}
		}
	}

	if = {
		limit = { root = scope:hd_east_han_holder }
		every_held_title = {
			limit = {
				OR = { tier = tier_county tier = tier_duchy tier = tier_kingdom }
				target_is_de_jure_liege_or_above = title:e_hd_western_han
			}
			change_title_holder = {
				holder = scope:hd_west_han_holder
				change = scope:hd_partition_change
			}
		}
		every_vassal = {
			limit = {
				this != scope:hd_west_han_holder
				capital_county = { target_is_de_jure_liege_or_above = title:e_hd_western_han }
			}
			change_liege = {
				liege = scope:hd_west_han_holder
				change = scope:hd_partition_change
			}
		}
	}
	else = {
		every_held_title = {
			limit = {
				OR = { tier = tier_county tier = tier_duchy tier = tier_kingdom }
				NOT = { target_is_de_jure_liege_or_above = title:e_hd_western_han }
			}
			change_title_holder = {
				holder = scope:hd_east_han_holder
				change = scope:hd_partition_change
			}
		}
		every_vassal = {
			limit = {
				this != scope:hd_east_han_holder
				capital_county = { NOT = { target_is_de_jure_liege_or_above = title:e_hd_western_han } }
			}
			change_liege = {
				liege = scope:hd_east_han_holder
				change = scope:hd_partition_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:hd_partition_change

	title:e_han = { set_title_name = e_han_eastern }
	scope:hd_west_han_holder = { set_realm_capital = title:c_uuii_p_867 }
	scope:hd_east_han_holder = { set_realm_capital = title:c_luoyang }
	scope:hd_west_han_holder = {
		add_truce_both_ways = {
			character = scope:hd_east_han_holder
			years = 10
			name = TRUCE_COLLABORATION
		}
	}
	set_global_variable = { name = hd_han_partition_complete value = root }
	remove_character_flag = hd_partition_choose_east
	remove_character_flag = hd_partition_choose_west
}
