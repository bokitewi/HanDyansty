# Native references: variable lists (00_ce1_interactions), while (00_courtier_guest_management_effects),
# hook types (00_hook_types), modifiers (00_spouse_tasks). No real mentor/court movement.
# MEMBER carries its intact subtree; update all ancestors when the edge changes.
WJ_mufu_index_edge = {
 $MEMBER$ = { save_scope_as = wj_index_member }
 $HOST$ = { save_scope_as = wj_index_cursor }
 save_scope_value_as = { name = wj_index_done value = 0 }
 while = {
  limit = { scope:wj_index_done = 0 }
  scope:wj_index_cursor = {
   $OP$ = { name = wj_mufu_descendants target = scope:wj_index_member }
  }
  scope:wj_index_member = {
   every_in_list = {
    variable = wj_mufu_descendants
    save_scope_as = wj_index_descendant
    scope:wj_index_cursor = { $OP$ = { name = wj_mufu_descendants target = scope:wj_index_descendant } }
   }
  }
  if = {
   limit = { scope:wj_index_cursor = { has_variable = wj_mufu_owner } }
   scope:wj_index_cursor.var:wj_mufu_owner = { save_scope_as = wj_index_cursor }
  }
  else = { save_scope_value_as = { name = wj_index_done value = 1 } }
 }
}

# Read the real parent pointers as well as the cached descendant trigger.
# Tortoise/hare detects even a pre-existing corrupt cycle, without recursive triggers
# or a persistent traversal scratch list. All scratch scopes are overwritten on each call.
WJ_mufu_check_chain = {
 scope:actor = { save_scope_as = wj_chain_slow save_scope_as = wj_chain_fast }
 save_scope_value_as = { name = wj_chain_safe value = 1 }
 save_scope_value_as = { name = wj_chain_done value = 0 }
 save_scope_value_as = { name = wj_chain_fast_done value = 0 }
 while = {
  limit = { scope:wj_chain_done = 0 }
  if = {
   limit = { scope:wj_chain_slow = scope:recipient }
   save_scope_value_as = { name = wj_chain_safe value = 0 }
   save_scope_value_as = { name = wj_chain_done value = 1 }
  }
  else_if = {
   limit = { scope:wj_chain_slow = { NOT = { has_variable = wj_mufu_owner } } }
   save_scope_value_as = { name = wj_chain_done value = 1 }
  }
  else = {
   scope:wj_chain_slow.var:wj_mufu_owner = { save_scope_as = wj_chain_slow }
   if = {
    limit = { scope:wj_chain_fast = { has_variable = wj_mufu_owner } }
    scope:wj_chain_fast.var:wj_mufu_owner = { save_scope_as = wj_chain_fast }
   }
   else = { save_scope_value_as = { name = wj_chain_fast_done value = 1 } }
   if = {
    limit = { scope:wj_chain_fast = { has_variable = wj_mufu_owner } }
    scope:wj_chain_fast.var:wj_mufu_owner = { save_scope_as = wj_chain_fast }
   }
   else = { save_scope_value_as = { name = wj_chain_fast_done value = 1 } }
   if = {
    limit = { scope:wj_chain_fast_done = 0 scope:wj_chain_fast = scope:wj_chain_slow }
    save_scope_value_as = { name = wj_chain_safe value = 0 }
    save_scope_value_as = { name = wj_chain_done value = 1 }
   }
  }
 }
}
