hd4_bootstrap_effect = {
	if = {
		limit = {
			game_start_date = 184.1.1
		}
		title:h_china = {
			if = {
				limit = {
					NOT = {
						has_variable = hd4_installed
					}
				}
				set_variable = hd4_installed
				if = {
					limit = {
						exists = global_var:hd_guandu_world_active
					}
					set_variable = hd4_legacy_guandu
				}
				else_if = {
					limit = {
						exists = global_var:hd_guandu_world_resolved
					}
					set_variable = hd4_legacy_guandu
				}
			}
			hd4_schedule_tick_effect = yes
		}
		character:cao_cao ?= {
			if = {
				limit = {
					is_alive = yes
					AND = {
						exists = character:yuan_shao
						character:yuan_shao = {
							is_alive = yes
						}
					}
					NOT = {
						has_character_flag = hd4_cao_yuan_friend_initialized
					}
				}
				add_character_flag = hd4_cao_yuan_friend_initialized
				if = {
					limit = {
						NOT = {
							has_relation_rival = character:yuan_shao
						}
						NOT = {
							has_relation_nemesis = character:yuan_shao
						}
						NOT = {
							exists = global_var:hd_guandu_world_active
						}
						NOT = {
							exists = global_var:hd_guandu_world_resolved
						}
					}
					if = {
						limit = {
							NOT = {
								has_relation_friend = character:yuan_shao
							}
						}
						set_relation_friend = character:yuan_shao
					}
				}
			}
		}
	}
}
hd4_schedule_tick_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = hd4_tick_queued
			}
			OR = {
				AND = {
					exists = character:tao_qian
					character:tao_qian = {
						is_alive = yes
					}
				}
				AND = {
					exists = character:gong_sun_zan
					character:gong_sun_zan = {
						is_alive = yes
					}
				}
				AND = {
					exists = character:cao_cao
					character:cao_cao = {
						is_alive = yes
					}
				}
				AND = {
					exists = character:yuan_shao
					character:yuan_shao = {
						is_alive = yes
					}
				}
				has_variable = hd4_gsz_release_active
				has_variable = hd4_yuan_partition_active
			}
		}
		set_variable = {
			name = hd4_tick_queued
			value = yes
			days = 61
		}
		trigger_event = {
			on_action = hd4_heartbeat
			days = 60
		}
	}
}
hd4_tick_effect = {
	hd4_guandu_reconcile_land_effect = yes
	character:tao_qian ?= {
		if = {
			limit = {
				is_alive = yes
			}
			hd4_tao_tick_effect = yes
		}
	}
	character:cao_cao ?= {
		if = {
			limit = {
				is_alive = yes
			}
			hd4_cao_tick_effect = yes
		}
	}
	character:yuan_shao ?= {
		if = {
			limit = {
				is_alive = yes
			}
			hd4_yuan_recovery_effect = yes
		}
	}
	hd4_jizhou_watchdog_effect = yes
	hd4_guandu_tick_effect = yes
	if = {
		limit = {
			has_variable = hd4_gsz_release_active
			NOT = {
				has_variable = hd4_gsz_release_queued
			}
		}
		hd4_gsz_release_tick_effect = yes
	}
	if = {
		limit = {
			has_variable = hd4_yuan_partition_active
			NOT = {
				has_variable = hd4_yuan_partition_queued
			}
		}
		hd4_yuan_partition_tick_effect = yes
	}
}
hd4_release_character_effect = {
	if = {
		limit = {
			is_alive = yes
			is_landed = yes
			highest_held_title_tier >= tier_county
			exists = liege
		}
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = hd4_release_change
			add_claim_on_loss = no
		}
		becomes_independent = {
			change = scope:hd4_release_change
		}
		resolve_title_and_vassal_change = scope:hd4_release_change
	}
}
