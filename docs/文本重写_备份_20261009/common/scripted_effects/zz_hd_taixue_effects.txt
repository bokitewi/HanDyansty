# 太学：建造与年度联动

# 在目标省份立太学：补城市据点、开特殊建筑位、建成。作用域：角色（郡持有者）。
hd_taixue_build_effect = {
	title:$BARONY$.title_province = {
		if = {
			limit = { has_holding = no }
			set_holding_type = city_holding
		}
		if = {
			limit = { has_special_building_slot = no }
			add_special_building_slot = TK_HD_TaiXue
		}
		add_special_building = TK_HD_TaiXue
	}
}

hd_taixue_yearly_for_effect = {
	if = {
		limit = {
			title:$COUNTY$.holder = this
			title:$BARONY$.title_province = { has_building_or_higher = TK_HD_TaiXue }
		}
		WJ_ziwang_qingyi_xp_effect = { VALUE = 1 }
		every_child = {
			limit = {
				is_alive = yes
				age >= 3
				age < 16
			}
			add_character_modifier = {
				modifier = hd_taixue_student_modifier
				years = 1
			}
		}
		# 天朝循环·时代特色：进取期联动 ×2、太学专属事件、诱因"开科取士"（见 hd_dc_kit_adv_effects.txt）
		hd_dc_kit_adv_taixue_linkage_effect = yes
	}
}
