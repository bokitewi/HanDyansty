# Summoning uses a native war picker followed by a separate acceptance interaction.
WJ_dd_force_conviction_effect = {
 if = {
  limit = { WJ_dd_order_force_trigger = yes }
  scope:recipient = { set_variable = { name = wj_dd_crime_realm value = top_liege.primary_title } add_character_modifier = { modifier = WJ_dd_public_crime_modifier } }
  scope:actor = { add_tyranny = minor_tyranny_gain add_dread = minor_dread_gain }
 }
}
WJ_dd_self_imprison_effect = {
 if = {
  limit = { WJ_dd_order_self_trigger = yes }
  scope:actor = { add_opinion = { target = scope:recipient modifier = WJ_dd_lawful_detention_reason opinion = 0 } }
  imprison_character_effect = { TARGET = scope:recipient IMPRISONER = scope:actor }
  scope:actor = { remove_opinion = { target = scope:recipient modifier = WJ_dd_lawful_detention_reason } }
 }
}
WJ_dd_execute_effect = {
 if = {
  limit = { WJ_dd_order_execute_trigger = yes }
  scope:recipient = { death = { killer = scope:actor death_reason = death_execution } }
  if = {
   limit = { scope:recipient = { is_alive = no } }
   scope:actor = {
    remove_character_modifier = WJ_dd_execution_modifier
    add_character_modifier = { modifier = WJ_dd_execution_modifier years = 1 }
   }
  }
 }
}
