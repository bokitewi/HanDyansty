# Character scope. Mutually exclusive, non-stacking and safe to call repeatedly.
# Preserve the old intended bands: normal -0.30; already -0.50 governments
# another -0.15; zero-levy wild-state placeholders receive no additional reduction.
hd_audit_refresh_levy_balance_effect = {
    if = {
        limit = { has_government = hd_yeren_government }
        remove_character_modifier = hd_audit_levy_standard
        remove_character_modifier = hd_audit_levy_reduced_government
    }
    else_if = {
        limit = {
            OR = {
                has_government = administrative_government
                has_government = celestial_government
                has_government = steppe_admin_government
                has_government = meritocratic_government
            }
        }
        remove_character_modifier = hd_audit_levy_standard
        if = {
            limit = { NOT = { has_character_modifier = hd_audit_levy_reduced_government } }
            add_character_modifier = { modifier = hd_audit_levy_reduced_government }
        }
    }
    else = {
        remove_character_modifier = hd_audit_levy_reduced_government
        if = {
            limit = { NOT = { has_character_modifier = hd_audit_levy_standard } }
            add_character_modifier = { modifier = hd_audit_levy_standard }
        }
    }
}
