hd4_yuan_snapshot_death_effect = {

			NOT = {

				}
			}
		}
		add_character_flag = hd4_yuan_origin_member

			limit = {
				exists = primary_heir
			}
			primary_heir = {
				add_character_flag = hd4_yuan_origin_member
			}
		}
		title:h_china = {
			clear_variable_list = hd4_partition_sons
			clear_variable_list = hd4_partition_counties
			clear_variable_list = hd4_partition_upper_titles
		}
		every_child = {
			limit = {
				is_alive = yes
				is_adult = yes
				is_female = no
			}
			set_variable = {
				name = hd4_partition_allocated
				value = 0
			}
			add_character_flag = hd4_yuan_origin_member
			save_scope_as = hd4_partition_son
			title:h_china = {
				add_to_variable_list = {
					name = hd4_partition_sons
					target = scope:hd4_partition_son
				}
			}
		}
		every_vassal_or_below = {
			add_character_flag = hd4_yuan_origin_member
			if = {
				limit = {
					exists = primary_heir
				}
				primary_heir = {
					add_character_flag = hd4_yuan_origin_member
				}
			}
		}
		if = {
			limit = {
				title:h_china = {
					any_in_list = {
						variable = hd4_partition_sons
						is_alive = yes
					}
				}
			}
			every_realm_county = {
				limit = {
					NOT = {
						has_variable = hd4_guandu_land_ticket
					}
				}
				save_scope_as = hd4_partition_county
				if = {
					limit = {
						exists = holder
					}
					set_variable = {
						name = hd4_partition_original_holder
						value = holder
					}
				}
				title:h_china = {
					add_to_variable_list = {
						name = hd4_partition_counties
						target = scope:hd4_partition_county
					}
					hd4_partition_select_least_son_effect = yes
				}
				if = {
					limit = {
						exists = scope:hd4_partition_selected_son
					}
					set_variable = {
						name = hd4_partition_recipient
						value = scope:hd4_partition_selected_son
					}
					scope:hd4_partition_selected_son = {
						change_variable = {
							name = hd4_partition_allocated
							add = 1
						}
					}
				}
			}
			hd4_partition_snapshot_upper_effect = yes
			every_vassal_or_below = {
				hd4_partition_snapshot_upper_effect = yes
			}
			title:h_china = {
				set_variable = hd4_yuan_death_snapshotted
				set_variable = hd4_yuan_partition_active
				set_variable = {
					name = hd4_yuan_partition_queued
					value = yes
					days = 2
				}
				trigger_event = {
					on_action = hd4_yuan_partition_tick
					days = 1
				}
			}
		}
		else = {
			title:h_china = {
				set_variable = hd4_yuan_death_snapshotted
				set_variable = hd4_yuan_partition_no_adult_sons
			}
		}
	}
}
hd4_partition_select_least_son_effect = {
	clear_saved_scope = hd4_partition_selected_son
	every_in_list = {
		variable = hd4_partition_sons
		limit = {
			is_alive = yes
		}
		if = {
			limit = {
				NOT = {
					exists = scope:hd4_partition_selected_son
				}
			}
			save_scope_as = hd4_partition_selected_son
		}
		else_if = {
			limit = {
				var:hd4_partition_allocated < scope:hd4_partition_selected_son.var:hd4_partition_allocated
			}
			save_scope_as = hd4_partition_selected_son
		}
	}

hd4_yuan_partition_tick_effect = {
	remove_variable = hd4_yuan_partition_queued

		limit = {

			character:yuan_shao = {
				is_alive = no
			}
		}
		if = {
			limit = {
				any_in_list = {
					variable = hd4_partition_sons
					is_alive = yes
				}
			}
			every_in_list = {

				limit = {

						is_alive = no
					}
				}

				title:h_china = {
					hd4_partition_select_least_son_effect = yes
				}
				if = {
					limit = {
						exists = scope:hd4_partition_selected_son
					}
					set_variable = {
						name = hd4_partition_recipient
						value = scope:hd4_partition_selected_son
					}
					scope:hd4_partition_selected_son = {
						change_variable = {
							name = hd4_partition_allocated

						}
					}
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = hd4_partition_change
				add_claim_on_loss = no
			}
			every_in_list = {
				variable = hd4_partition_counties
				limit = {
					has_variable = hd4_partition_recipient
					var:hd4_partition_recipient = {
						is_alive = yes
					}
					NOT = {
						has_variable = hd4_partition_external
					}
				}
				save_scope_as = hd4_partition_current_title
				var:hd4_partition_recipient = {
					save_scope_as = hd4_partition_recipient
				}
				if = {
					limit = {
						NOT = {
							AND = {
								exists = holder
								holder = {
									OR = {
										this = scope:hd4_partition_recipient
										is_vassal_or_below_of = scope:hd4_partition_recipient
									}
								}
							}
						}
						hd4_partition_source_still_local_trigger = yes
					}
					change_title_holder = {
						holder = scope:hd4_partition_recipient
						change = scope:hd4_partition_change
						take_baronies = yes
					}
				}
				else_if = {
					limit = {
						NOT = {
							AND = {
								exists = holder
								holder = {
									OR = {
										this = scope:hd4_partition_recipient
										is_vassal_or_below_of = scope:hd4_partition_recipient
									}
								}
							}
						}
						NOT = {
							hd4_partition_source_still_local_trigger = yes
						}
					}
					set_variable = hd4_partition_external
				}
			}
			resolve_title_and_vassal_change = scope:hd4_partition_change
			every_in_list = {
				variable = hd4_partition_upper_titles
				hd4_partition_repair_upper_recipient_effect = yes
				limit = {
					has_variable = hd4_partition_recipient
					NOT = {
						has_variable = hd4_partition_upper_done
					}
				}
				if = {
					limit = {
						var:hd4_partition_recipient = {
							is_alive = no
						}
					}
					if = {
						limit = {
							AND = {
								exists = title_capital_county
								title_capital_county = {
									has_variable = hd4_partition_recipient
								}
							}
						}
						set_variable = {
							name = hd4_partition_recipient
							value = title_capital_county.var:hd4_partition_recipient
						}
					}
				}
				var:hd4_partition_recipient = {
					save_scope_as = hd4_partition_recipient
				}
				if = {
					limit = {
						scope:hd4_partition_recipient = {
							is_alive = yes
							any_held_title = {
								tier = tier_county
							}
						}
						hd4_partition_source_still_local_trigger = yes
					}
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = hd4_partition_upper_change
						add_claim_on_loss = no
					}
					change_title_holder = {
						holder = scope:hd4_partition_recipient
						change = scope:hd4_partition_upper_change
					}
					resolve_title_and_vassal_change = scope:hd4_partition_upper_change
				}
				if = {
					limit = {
						OR = {
							holder = scope:hd4_partition_recipient
							NOT = {
								hd4_partition_source_still_local_trigger = yes
							}
						}
					}
					set_variable = hd4_partition_upper_done
				}
			}
			every_in_list = {
				variable = hd4_partition_sons
				limit = {
					is_alive = yes
					is_landed = yes
					any_held_title = {
						tier = tier_county
					}
				}
				hd4_release_character_effect = yes
			}
			hd4_partition_reserve_anchors_effect = yes
			hd4_partition_restore_counts_effect = yes
			if = {
				limit = {
					NOT = {
						any_in_list = {
							variable = hd4_partition_counties
							NOT = {
								has_variable = hd4_partition_external
							}
							has_variable = hd4_partition_recipient
							NOT = {
								AND = {
									exists = holder

										OR = {
											this = prev.var:hd4_partition_recipient
											is_vassal_or_below_of = prev.var:hd4_partition_recipient
										}
									}
								}
							}
						}
					}
					NOT = {
						any_in_list = {
							variable = hd4_partition_sons
							is_alive = yes
							is_landed = yes
							any_held_title = {
								tier = tier_county
							}
							is_independent_ruler = no
						}
					}
					NOT = {
						any_in_list = {
							variable = hd4_partition_upper_titles
							has_variable = hd4_partition_recipient
							NOT = {
								has_variable = hd4_partition_upper_done
							}
						}
					}
				}
				remove_variable = hd4_yuan_partition_active
				set_variable = hd4_yuan_partition_complete
				every_in_list = {
					variable = hd4_partition_sons
					limit = {
						is_alive = yes
						is_landed = yes
					}
					trigger_event = hd4_yuan.0003
				}
			}
			else = {
				set_variable = {
					name = hd4_yuan_partition_queued
					value = yes
					days = 2
				}
				trigger_event = {
					on_action = hd4_yuan_partition_tick
					days = 1
				}
			}
		}
		else = {
			remove_variable = hd4_yuan_partition_active
			set_variable = hd4_yuan_partition_no_surviving_sons
		}
	}
}
hd4_partition_restore_counts_effect = {
	every_in_list = {
		variable = hd4_partition_counties
		limit = {
			has_variable = hd4_partition_original_holder
			has_variable = hd4_partition_recipient
			NOT = {
				has_variable = hd4_partition_external
			}
		}
		var:hd4_partition_original_holder = {
			save_scope_as = hd4_restore_count
		}
		var:hd4_partition_recipient = {
			save_scope_as = hd4_restore_liege
		}
		if = {
			limit = {
				scope:hd4_restore_count = {
					is_alive = yes
					NOT = {
						has_character_flag = hd4_partition_restore_done
					}
					NOT = {
						this = scope:hd4_restore_liege
					}
					OR = {
						is_landed = no
						AND = {
							has_variable = hd4_restore_started
							var:hd4_restore_started = scope:hd4_restore_liege
							NOT = {
								any_held_title = {
									tier = tier_county
									NOT = {
										var:hd4_partition_recipient = scope:hd4_restore_liege
									}
								}
							}
						}
					}
					highest_held_title_tier < scope:hd4_restore_liege.highest_held_title_tier
				}
				scope:hd4_restore_liege = {
					is_alive = yes
					is_landed = yes
					highest_held_title_tier >= tier_duchy
					has_variable = hd4_partition_anchor_county
				}
				title:h_china = {
					NOT = {
						any_in_list = {
							variable = hd4_partition_counties
							var:hd4_partition_original_holder = scope:hd4_restore_count
							NOT = {
								var:hd4_partition_recipient = scope:hd4_restore_liege
							}
						}
					}
					any_in_list = {
						variable = hd4_partition_counties
						var:hd4_partition_original_holder = scope:hd4_restore_count
						NOT = {
							this = scope:hd4_restore_liege.var:hd4_partition_anchor_county
						}
						NOT = {
							has_variable = hd4_partition_external
						}
						OR = {
							holder = scope:hd4_restore_liege
							holder = scope:hd4_restore_count
						}
					}
				}
			}
			scope:hd4_restore_count = {
				set_variable = {
					name = hd4_restore_started
					value = scope:hd4_restore_liege
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = hd4_restore_change
				add_claim_on_loss = no
			}
			title:h_china = {
				every_in_list = {
					variable = hd4_partition_counties
					limit = {
						var:hd4_partition_original_holder = scope:hd4_restore_count
						holder = scope:hd4_restore_liege
						NOT = {
							this = scope:hd4_restore_liege.var:hd4_partition_anchor_county
						}
						NOT = {
							has_variable = hd4_partition_external
						}
					}
					change_title_holder = {
						holder = scope:hd4_restore_count
						change = scope:hd4_restore_change
						take_baronies = yes
					}
				}
			}
			resolve_title_and_vassal_change = scope:hd4_restore_change
			scope:hd4_restore_count = {
				if = {
					limit = {
						is_landed = yes
						any_held_title = {
							tier = tier_county
						}
						NOT = {
							liege = scope:hd4_restore_liege
						}
					}
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = hd4_restore_fealty
						add_claim_on_loss = no
					}
					change_liege = {
						liege = scope:hd4_restore_liege
						change = scope:hd4_restore_fealty
					}
					resolve_title_and_vassal_change = scope:hd4_restore_fealty
				}
				if = {
					limit = {
						liege = scope:hd4_restore_liege
						title:h_china = {
							NOT = {
								any_in_list = {
									variable = hd4_partition_counties
									var:hd4_partition_original_holder = scope:hd4_restore_count
									holder = scope:hd4_restore_liege
									NOT = {
										this = scope:hd4_restore_liege.var:hd4_partition_anchor_county
									}

									}

	}
