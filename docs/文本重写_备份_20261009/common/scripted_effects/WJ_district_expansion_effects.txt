# ROOT may be either side; named scopes survive the native letter callback.
WJ_dd_petition_begin = {
 scope:actor = { save_scope_as = wj_dd_petitioner }
 scope:recipient = { save_scope_as = wj_dd_petition_emperor }
 scope:wj_dd_petitioner = {
  if = { limit = { has_variable = wj_dd_petition_counter } change_variable = { name = wj_dd_petition_counter add = 1 } }
  else = { set_variable = { name = wj_dd_petition_counter value = 1 } }
  save_scope_value_as = { name = wj_dd_petition_serial value = var:wj_dd_petition_counter }
  set_variable = { name = wj_dd_petition_pending value = yes days = 90 }
  set_variable = { name = wj_dd_petition_kind value = $KIND$ days = 90 }
  if = { limit = { $KIND$ = 2 } set_variable = { name = wj_dd_petition_target value = scope:secondary_recipient days = 90 } }
  else = {
   if = { limit = { is_ai = yes }
    random_list = {
     50 = { set_variable = { name = wj_dd_petition_jie value = 1 days = 90 } }
     40 = { set_variable = { name = wj_dd_petition_jie value = 2 days = 90 } }
     10 = {
      modifier = { factor = 3 exists = dynasty exists = top_liege.dynasty dynasty = top_liege.dynasty }
      modifier = { factor = 2 realm_size >= 40 }
      set_variable = { name = wj_dd_petition_jie value = 3 days = 90 }
     }
    }
   }
   else_if = { limit = { scope:wj_dd_jiajie = yes } set_variable = { name = wj_dd_petition_jie value = 1 days = 90 } }
   else_if = { limit = { scope:wj_dd_chijie = yes } set_variable = { name = wj_dd_petition_jie value = 2 days = 90 } }
   else = { set_variable = { name = wj_dd_petition_jie value = 3 days = 90 } }
  }
 }
}
WJ_dd_petition_clear = {
 scope:wj_dd_petitioner ?= {
  if = { limit = { var:wj_dd_petition_counter = scope:wj_dd_petition_serial }
   remove_variable = wj_dd_petition_pending remove_variable = wj_dd_petition_kind
   remove_variable = wj_dd_petition_target remove_variable = wj_dd_petition_jie
  }
 }
}
WJ_dd_petition_approve = {
 if = { limit = { WJ_dd_petition_reply_valid = yes }
  scope:actor ?= { save_scope_as = wj_dd_petition_old_actor }
  scope:recipient ?= { save_scope_as = wj_dd_petition_old_recipient }
  scope:wj_dd_petition_emperor = { save_scope_as = actor }
  scope:wj_dd_petitioner = { save_scope_as = recipient }
  if = { limit = { scope:wj_dd_petitioner = { var:wj_dd_petition_kind = 1 } }
   WJ_dd_create_effect = { JIE = scope:wj_dd_petitioner.var:wj_dd_petition_jie }
  }
  else = {
   scope:wj_dd_petitioner.var:wj_dd_petition_target = { save_scope_as = secondary_recipient }
   WJ_dd_add_effect = yes
  }
  scope:wj_dd_petitioner = {
   if = { limit = { WJ_dd_head_trigger = yes }
    add_opinion = { target = scope:wj_dd_petition_emperor modifier = WJ_dd_petition_approved_opinion }
    if = { limit = { is_ai = no } WJ_dd_refresh_view_effect = yes }
   }
  }
  scope:wj_dd_petition_emperor = { if = { limit = { is_ai = no } WJ_dd_refresh_view_effect = yes } }
  scope:wj_dd_petition_old_actor ?= { save_scope_as = actor }
  scope:wj_dd_petition_old_recipient ?= { save_scope_as = recipient }
 }
 WJ_dd_petition_clear = yes
}
WJ_dd_petition_decline = {
 if = { limit = { WJ_dd_petition_reply_valid = yes }
  scope:wj_dd_petitioner = { add_opinion = { target = scope:wj_dd_petition_emperor modifier = WJ_dd_petition_declined_opinion } }
 }
 WJ_dd_petition_clear = yes
}
WJ_dd_copy_custom_name = {
 if = { limit = { $CORE$ = { has_variable = wj_dd_custom_name_slot } }
  if = { limit = { $CORE$ = { var:wj_dd_custom_name_slot = 1 } }
   copy_localized_text = { key = wj_dd_custom_name_1 target = $FROM$ }
   store_localized_text_in_death = wj_dd_custom_name_1
  }
  else = {
   copy_localized_text = { key = wj_dd_custom_name_2 target = $FROM$ }
   store_localized_text_in_death = wj_dd_custom_name_2
  }
 }
}
