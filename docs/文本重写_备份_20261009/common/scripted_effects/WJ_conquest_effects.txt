# Saved interaction scopes never change the actor into an emperor proxy.
WJ_conquest_accept_effect = {
	if = {
		limit = { WJ_conquest_request_still_valid_trigger = yes }
		if = {
			limit = { scope:hook = yes }
			scope:actor = { use_hook = scope:recipient }
		}
		# Native frontier aid: control only, from the emperor's own primary title.
		# The outer validity check rejects Grand Marshal/raised/lent/stale aid first.

		scope:recipient = {
			start_war = {
				cb = influence_war_cb
				target = scope:secondary_recipient
				claimant = scope:recipient.top_liege
				target_title = scope:target
			}
		}
		scope:actor = { trigger_event = WJ_conquest.0001 }
	}
	else = { WJ_conquest_invalid_reply_effect = yes }
}

WJ_conquest_decline_effect = {
	if = {
		limit = { WJ_conquest_request_still_valid_trigger = yes }
		# Native refusal consequences only; influence is not merit/贤望.
		scope:recipient = { change_influence = major_influence_loss }
		scope:actor = {
			if = {
				limit = { is_ai = yes }
				add_opinion = {
					modifier = refused_summons_to_war_opinion
					target = scope:recipient
				}
			}
			trigger_event = WJ_conquest.0002
		}
	}
	else = { WJ_conquest_invalid_reply_effect = yes }
}

WJ_conquest_invalid_reply_effect = {
	scope:actor ?= {
		if = {
			limit = { is_alive = yes }
			trigger_event = WJ_conquest.0003
		}
	}
}
