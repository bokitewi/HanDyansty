# 天朝循环·时代特色：恢复期套件效果（见 docs/天朝循环_时代特色重制计划.md §3.7）
# 诱因一律经 hd_dc_kit_fire_effect（common/scripted_effects/hd_dc_kit_effects.txt）发射

# ---------------------------------------------------------------------------
# 轻徭薄赋
# ---------------------------------------------------------------------------
# Scope：天子
hd_dc_rec_light_taxes_apply_effect = {
	add_character_modifier = {
		modifier = hd_dc_rec_light_taxes_modifier
		years = 10
	}
	custom_tooltip = hd_dc_rec_light_taxes_county_tt
	hidden_effect = {
		every_sub_realm_county = {
			add_county_modifier = {
				modifier = hd_dc_rec_light_taxes_county_modifier
				years = 10
			}
		}
	}
	hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_light_taxes ACTOR = root }
}

# Scope：天子（或原颁布者）。中止轻徭薄赋：撤去天子与各县修正，计"苛政加税"
hd_dc_rec_light_taxes_cancel_effect = {
	remove_character_modifier = hd_dc_rec_light_taxes_modifier
	custom_tooltip = hd_dc_rec_cancel_light_taxes_county_tt
	hidden_effect = {
		every_sub_realm_county = {
			limit = { has_county_modifier = hd_dc_rec_light_taxes_county_modifier }
			remove_county_modifier = hd_dc_rec_light_taxes_county_modifier
		}
	}
	hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_harsh_tax ACTOR = root }
}

# ---------------------------------------------------------------------------
# 招抚流民。Scope：天子；scope:selected_item = 目标州（王国级法理头衔）
# ---------------------------------------------------------------------------
hd_dc_rec_resettle_refugees_effect = {
	scope:selected_item = {
		custom_tooltip = hd_dc_rec_resettle_refugees_county_tt
		hidden_effect = {
			every_de_jure_county = {
				limit = { holder.top_liege ?= root }
				add_county_modifier = {
					modifier = hd_dc_rec_resettle_refugees_county_modifier
					years = 10
				}
			}
		}
	}
	hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_resettle_refugees ACTOR = root }
}

# ---------------------------------------------------------------------------
# 大赦天下。Scope：天子
# ---------------------------------------------------------------------------
# 释放天子本人的囚犯
hd_dc_rec_release_own_prisoners_effect = {
	every_prisoner = {
		release_from_prison = yes
	}
}

# 封臣好感 +15（5 年）
hd_dc_rec_amnesty_vassal_opinion_effect = {
	every_vassal = {
		add_opinion = {
			target = root
			modifier = hd_dc_rec_amnesty_opinion
		}
	}
}

# 连下属封臣（含次级封臣）的囚犯一并释放；囚犯被放走的封臣对天子 −10
hd_dc_rec_release_vassal_prisoners_effect = {
	every_vassal_or_below = {
		limit = { any_prisoner = { always = yes } }
		every_prisoner = {
			release_from_prison = yes
		}
		add_opinion = {
			target = root
			modifier = hd_dc_rec_prisoners_released_opinion
		}
	}
}
