# Scope：天子。把 character:$WHO$ 任命为 $POS$（仅当其在世、是天子廷臣、职位空缺且可任）。
hd_eunuch_appoint_effect = {
	if = {
		limit = {
			exists = character:$WHO$
			character:$WHO$ = {
				is_alive = yes
				is_courtier_of = scope:hd_eunuch_emperor
			}
			is_valid_to_hire_court_position_type = $POS$
			NOT = { employs_court_position = $POS$ }
		}
		appoint_court_position = {
			recipient = character:$WHO$
			court_position = $POS$
		}
	}
}
