# --- 幽州突骑：兵源刷新 ---
# 已建立标记在头衔上；兵源旗标是当前持有者的缓存状态，供通知做边沿检测。
hd_refresh_youzhou_tuqi_source_effect = {
	if = {
		limit = { hd_has_established_youzhou_tuqi_trigger = yes }
		if = {
			limit = {
				hd_valid_youzhou_tuqi_source_trigger = yes
				NOT = { has_character_flag = hd_source_youzhou_tuqi }
			}
			add_character_flag = hd_source_youzhou_tuqi
			if = {
				limit = { is_ai = no }
				trigger_event = { id = hd_maa_source.0002 }
			}
		}
		if = {
			limit = {
				NOT = { hd_valid_youzhou_tuqi_source_trigger = yes }
				has_character_flag = hd_source_youzhou_tuqi
			}
			remove_character_flag = hd_source_youzhou_tuqi
			if = {
				limit = { is_ai = no }
				trigger_event = { id = hd_maa_source.0001 }
			}
		}
	}
	else = {
		remove_character_flag = hd_source_youzhou_tuqi
	}
}

# --- 陷阵营：选择最佳训练者 ---
# 评分最高者胜出；并列时取遍历顺序靠后者（每轮遍历顺序稳定，结果确定）。
# 选中者存为 scope:hd_maa_trainer，供训练事件使用。
hd_select_xianzhenying_trainer_effect = {
	set_variable = {
		name = hd_maa_best_score
		value = -1
	}
	every_knight = {
		limit = { hd_xianzhenying_trainer_trigger = yes }
		set_variable = {
			name = hd_maa_candidate_score
			value = hd_xianzhenying_trainer_score_value
		}
		if = {
			limit = { var:hd_maa_candidate_score >= root.var:hd_maa_best_score }
			save_scope_as = hd_maa_trainer
			root = {
				set_variable = {
					name = hd_maa_best_score
					value = prev.var:hd_maa_candidate_score
				}
			}
		}
		remove_variable = hd_maa_candidate_score
	}
	remove_variable = hd_maa_best_score
}

# --- 陷阵营：训练结算（$INVESTMENT$ = 投入训练修正 0/10/20） ---
# 训练值 = 训练者军事 + 武勇 + 投入修正 + 随机 0~40；四档数量分别随机为
# 800~1000、900~1200、1100~1400、1300~1500 人（陷阵营每组 100 人）。
hd_resolve_xianzhenying_training_effect = {
	set_variable = {
		name = hd_training_score
		value = {
			add = scope:hd_maa_trainer.martial
			add = scope:hd_maa_trainer.prowess
			add = $INVESTMENT$
		}
	}
	change_variable = {
		name = hd_training_score
		add = {
			integer_range = { min = 0 max = 40 }
		}
	}
	if = {
		limit = { var:hd_training_score >= 90 }
		set_variable = { name = hd_training_tier value = 4 }
		spawn_army = {
			name = hd_xianzhenying_army
			men_at_arms = {
				type = hd_xianzhenying
				stacks = { integer_range = { min = 13 max = 15 } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	else_if = {
		limit = { var:hd_training_score >= 70 }
		set_variable = { name = hd_training_tier value = 3 }
		spawn_army = {
			name = hd_xianzhenying_army
			men_at_arms = {
				type = hd_xianzhenying
				stacks = { integer_range = { min = 11 max = 14 } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	else_if = {
		limit = { var:hd_training_score >= 50 }
		set_variable = { name = hd_training_tier value = 2 }
		spawn_army = {
			name = hd_xianzhenying_army
			men_at_arms = {
				type = hd_xianzhenying
				stacks = { integer_range = { min = 9 max = 12 } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	else = {
		set_variable = { name = hd_training_tier value = 1 }
		spawn_army = {
			name = hd_xianzhenying_army
			men_at_arms = {
				type = hd_xianzhenying
				stacks = { integer_range = { min = 8 max = 10 } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	trigger_event = { id = hd_maa_training.1001 }
}

	}
	every_knight = {
		limit = { $TRIGGER$ = yes }
		set_variable = {
			name = hd_maa_candidate_score
			value = $SCORE$
		}
		if = {
			limit = { var:hd_maa_candidate_score >= root.var:hd_maa_best_score }
			save_scope_as = hd_maa_trainer
			root = {
				set_variable = {
					name = hd_maa_best_score
					value = prev.var:hd_maa_candidate_score
				}
			}
		}
		remove_variable = hd_maa_candidate_score
	}
	remove_variable = hd_maa_best_score
}

# $T1_MIN$/$T1_MAX$ ... $T4_MIN$/$T4_MAX$ 为兵种堆叠数量范围；调用方传入的范围
# 分别对应 800~1000、900~1200、1100~1400、1300~1500 人。
hd_resolve_elite_training_effect = {
	set_variable = {
		name = hd_training_score
		value = {
			add = scope:hd_maa_trainer.martial
			add = scope:hd_maa_trainer.prowess
			add = $INVESTMENT$
		}
	}
	change_variable = {
		name = hd_training_score
		add = {
			integer_range = { min = 0 max = 40 }
		}
	}
	if = {
		limit = { var:hd_training_score >= 90 }
		set_variable = { name = hd_training_tier value = 4 }
		spawn_army = {
			name = $ARMY_NAME$
			men_at_arms = {
				type = $MAA_TYPE$
				stacks = { integer_range = { min = $T4_MIN$ max = $T4_MAX$ } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	else_if = {
		limit = { var:hd_training_score >= 70 }
		set_variable = { name = hd_training_tier value = 3 }
		spawn_army = {
			name = $ARMY_NAME$
			men_at_arms = {
				type = $MAA_TYPE$
				stacks = { integer_range = { min = $T3_MIN$ max = $T3_MAX$ } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	else_if = {
		limit = { var:hd_training_score >= 50 }
		set_variable = { name = hd_training_tier value = 2 }
		spawn_army = {
			name = $ARMY_NAME$
			men_at_arms = {
				type = $MAA_TYPE$
				stacks = { integer_range = { min = $T2_MIN$ max = $T2_MAX$ } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	else = {
		set_variable = { name = hd_training_tier value = 1 }
		spawn_army = {
			name = $ARMY_NAME$
			men_at_arms = {
				type = $MAA_TYPE$
				stacks = { integer_range = { min = $T1_MIN$ max = $T1_MAX$ } }
			}
			location = capital_province
			uses_supply = yes
			inheritable = yes
		}
	}
	trigger_event = { id = $RESULT_EVENT$ }
}
