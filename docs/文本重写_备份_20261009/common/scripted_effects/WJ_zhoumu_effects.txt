# Scope is the official; the actual top emperor dislikes the appointment, not vice versa.
WJ_zhoumu_clear_opinion = {
 if = {
  limit = { exists = var:wj_zhoumu_opinion_emperor }
  save_scope_as = wj_zhoumu_official
  var:wj_zhoumu_opinion_emperor = {
   remove_opinion = { target = scope:wj_zhoumu_official modifier = WJ_zhoumu_imperial_distrust }
  }
 }
 remove_variable = wj_zhoumu_opinion_emperor
}
WJ_zhoumu_clear = {
 if = {
  limit = { vassal_contract_obligation_level:WJ_zhoumu > 0 }
  vassal_contract_set_obligation_level = { type = WJ_zhoumu level = 0 }
 }
 WJ_zhoumu_clear_opinion = yes
}
WJ_zhoumu_reconcile = {
 if = {
  limit = { WJ_zhoumu_active = yes }
  if = {
   limit = { has_variable = wj_zhoumu_opinion_emperor NOT = { var:wj_zhoumu_opinion_emperor = top_liege } }
   WJ_zhoumu_clear_opinion = yes
  }
  save_scope_as = wj_zhoumu_official
  set_variable = { name = wj_zhoumu_opinion_emperor value = top_liege }
  # Non-stacking, non-decaying; reapplication repairs a missing opinion after loading.
  top_liege = { add_opinion = { target = scope:wj_zhoumu_official modifier = WJ_zhoumu_imperial_distrust } }
 }
 else_if = { limit = { WJ_zhoumu_has_state = yes } WJ_zhoumu_clear = yes }
}
WJ_zhoumu_reconcile_subrealm = {
 WJ_zhoumu_reconcile = yes
 every_vassal_or_below = { limit = { WJ_zhoumu_has_state = yes } WJ_zhoumu_reconcile = yes }
}
