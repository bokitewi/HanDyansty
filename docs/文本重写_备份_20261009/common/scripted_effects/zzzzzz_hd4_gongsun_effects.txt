hd4_jizhou_watchdog_effect = {
	if = {
		limit = {
			NOT = {
				exists = global_var:hd4_jizhou_finished
			}
			exists = global_var:hd_anti_regent_chain_resolved
		}
		if = {
			limit = {
				OR = {
					AND = {
						exists = character:gong_sun_zan
						character:gong_sun_zan = {
							is_alive = no
						}
					}
					AND = {
						exists = character:yuan_shao
						character:yuan_shao = {
							is_alive = no
						}
					}
				}
			}
			set_global_variable = hd4_jizhou_finished
			hd_yuan_jizhou_end_effect = yes
		}
		else = {
			if = {
				limit = {
					exists = global_var:hd_yuan_jizhou_handover_started
					NOT = {
						exists = global_var:hd_yuan_jizhou_handover_complete
					}
					title:h_china = {
						has_variable = hd4_han_snapshot_taken
					}
					AND = {
						exists = character:han_fu
						character:han_fu = {
							is_alive = yes
						}
					}
					AND = {
						exists = character:yuan_shao
						character:yuan_shao = {
							is_alive = yes
							is_landed = yes
						}
					}
				}
				hd4_jizhou_handover_effect = yes
			}
			clear_saved_scope = hd4_live_jizhou_war
			character:gong_sun_zan ?= {
				random_character_war = {
					limit = {
						using_cb = hd_yuan_jizhou_cb
						primary_attacker = character:gong_sun_zan
					}
					save_scope_as = hd4_live_jizhou_war
				}
			}
			if = {
				limit = {
					exists = scope:hd4_live_jizhou_war
				}
				set_global_variable = {
					name = hd_yuan_jizhou_war
					value = scope:hd4_live_jizhou_war
				}
				set_global_variable = hd_yuan_jizhou_pending
				if = {
					limit = {
						NOT = {
							exists = global_var:hd_yuan_jizhou_handover_complete
						}
						NOT = {
							exists = global_var:hd_yuan_jizhou_handover_waiting
						}
						hd_yuan_jizhou_handover_ready_trigger = yes
					}
					hd4_jizhou_handover_effect = yes
				}
			}
			else_if = {
				limit = {
					AND = {
						exists = character:gong_sun_zan
						character:gong_sun_zan = {
							is_alive = yes
							has_title = title:k_uuii_youzhou
						}
					}
					AND = {
						exists = character:yuan_shao
						character:yuan_shao = {
							is_alive = yes
							is_landed = yes
						}
					}
					OR = {
						AND = {
							exists = title:k_uuii_jizhou.holder
							title:k_uuii_jizhou.holder = {
								this = character:han_fu
								is_alive = yes
							}
						}
						AND = {
							exists = global_var:hd_yuan_jizhou_handover_complete
							AND = {
								exists = title:k_uuii_jizhou.holder
								title:k_uuii_jizhou.holder = {
									this = character:yuan_shao
									is_alive = yes
								}
							}
						}
					}
				}
				remove_global_variable = hd_yuan_jizhou_war_attempted
				if = {
					limit = {
						character:gong_sun_zan = {
							has_character_flag = hd4_jizhou_authorized
						}
					}
					hd4_jizhou_try_start_effect = yes
				}
				else = {
					remove_global_variable = hd_yuan_jizhou_pending
					character:gong_sun_zan = {
						if = {
							limit = {
								NOT = {
									has_character_flag = hd4_jizhou_invitation_lease
								}
							}
							add_character_flag = {
								flag = hd4_jizhou_invitation_lease
								days = 180
							}
							trigger_event = hd_yuan_jizhou.0012
						}
					}
				}
			}
			else = {
				hd_yuan_jizhou_end_effect = yes
			}
		}
	}
}
hd4_jizhou_try_start_effect = {
	if = {
		limit = {
			NOT = {
				exists = global_var:hd4_jizhou_finished
			}
			AND = {
				exists = character:gong_sun_zan
				character:gong_sun_zan = {
					is_alive = yes
					is_landed = yes
					is_imprisoned = no
					has_title = title:k_uuii_youzhou
					has_character_flag = hd4_jizhou_authorized
					NOT = {
						any_character_war = {
							using_cb = hd_yuan_jizhou_cb
						}
					}
				}
			}
			AND = {
				exists = character:yuan_shao
				character:yuan_shao = {
					is_alive = yes
					is_landed = yes
				}
			}
			AND = {
				exists = title:k_uuii_jizhou.holder
				title:k_uuii_jizhou.holder = {
					is_alive = yes
					OR = {
						this = character:han_fu
						AND = {
							this = character:yuan_shao
							exists = global_var:hd_yuan_jizhou_handover_complete
						}
					}
				}
			}
		}
		title:k_uuii_jizhou.holder = {
			save_scope_as = hd4_jizhou_defender
		}
		character:gong_sun_zan = {
			if = {
				limit = {
					NOT = {
						is_at_war_with = scope:hd4_jizhou_defender
					}
					NOT = {
						is_allied_to = scope:hd4_jizhou_defender
					}
					NOT = {
						has_truce = scope:hd4_jizhou_defender
					}
					can_declare_war = {
						defender = scope:hd4_jizhou_defender
						casus_belli = hd_yuan_jizhou_cb
						target_titles = {
							title:k_uuii_jizhou
						}
					}
				}
				start_war = {
					cb = hd_yuan_jizhou_cb
					target = scope:hd4_jizhou_defender
					target_title = title:k_uuii_jizhou
				}
				random_character_war = {
					limit = {
						using_cb = hd_yuan_jizhou_cb
						primary_attacker = character:gong_sun_zan
						primary_defender = scope:hd4_jizhou_defender
					}
					save_scope_as = hd4_created_jizhou_war
				}
			}
		}
		if = {
			limit = {
				exists = scope:hd4_created_jizhou_war
			}
			set_global_variable = {
				name = hd_yuan_jizhou_war
				value = scope:hd4_created_jizhou_war
			}
			set_global_variable = {
				name = hd_yuan_jizhou_target
				value = title:k_uuii_jizhou
			}
			set_global_variable = hd_yuan_jizhou_pending
			set_global_variable = hd_yuan_jizhou_war_attempted
			if = {
				limit = {
					NOT = {
						exists = global_var:hd_yuan_jizhou_handover_complete
					}
				}
				set_global_variable = {
					name = hd_yuan_jizhou_handover_waiting
					value = yes
					days = 30
				}
				character:yuan_shao = {
					trigger_event = {
						id = hd_yuan_jizhou.0002
						days = 31
					}
				}
			}
			if = {
				limit = {
					NOT = {
						exists = global_var:hd4_jizhou_start_notified
					}
				}
				set_global_variable = hd4_jizhou_start_notified
				character:yuan_shao = {
					trigger_event = hd_yuan_jizhou.0010
				}
			}
		}
	}
}
hd4_jizhou_handover_effect = {
	if = {
		limit = {
			OR = {
				hd_yuan_jizhou_handover_ready_trigger = yes
				AND = {
					exists = global_var:hd_yuan_jizhou_handover_started
					title:h_china = {
						has_variable = hd4_han_snapshot_taken
					}
					AND = {
						exists = character:han_fu
						character:han_fu = {
							is_alive = yes
						}
					}
					AND = {
						exists = character:yuan_shao
						character:yuan_shao = {
							is_alive = yes
							is_landed = yes
						}
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {
					title:h_china = {
						has_variable = hd4_han_snapshot_taken
					}
				}
			}
			title:h_china = {
				clear_variable_list = hd4_han_titles
				clear_variable_list = hd4_han_vassals
			}
			character:han_fu = {
				every_held_title = {
					limit = {
						is_noble_family_title = no
					}
					save_scope_as = hd4_han_title
					title:h_china = {
						add_to_variable_list = {
							name = hd4_han_titles
							target = scope:hd4_han_title
						}
					}
				}
				every_vassal = {
					save_scope_as = hd4_han_vassal
					title:h_china = {
						add_to_variable_list = {
							name = hd4_han_vassals
							target = scope:hd4_han_vassal
						}
					}
				}
			}
			title:h_china = {
				set_variable = hd4_han_snapshot_taken
			}
		}
		set_global_variable = hd_yuan_jizhou_handover_started
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd4_han_rank_change
			add_claim_on_loss = no
		}
		title:h_china = {
			every_in_list = {
				variable = hd4_han_titles
				limit = {
					tier >= tier_kingdom
					holder = character:han_fu
				}
				change_title_holder = {
					holder = character:yuan_shao
					change = scope:hd4_han_rank_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd4_han_rank_change
		character:han_fu = {
			if = {
				limit = {
					is_landed = yes
					highest_held_title_tier < character:yuan_shao.highest_held_title_tier
					NOT = {
						liege = character:yuan_shao
					}
				}
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = hd4_han_fealty
					add_claim_on_loss = no
				}
				change_liege = {
					liege = character:yuan_shao
					change = scope:hd4_han_fealty
				}
				resolve_title_and_vassal_change = scope:hd4_han_fealty
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd4_han_realm_change
			add_claim_on_loss = no
		}
		title:h_china = {
			every_in_list = {
				variable = hd4_han_vassals
				limit = {
					is_alive = yes
					is_landed = yes
					liege = character:han_fu
					highest_held_title_tier < character:yuan_shao.highest_held_title_tier
				}
				change_liege = {
					liege = character:yuan_shao
					change = scope:hd4_han_realm_change
				}
			}
			every_in_list = {
				variable = hd4_han_titles
				limit = {
					holder = character:han_fu
				}
				change_title_holder = {
					holder = character:yuan_shao
					change = scope:hd4_han_realm_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hd4_han_realm_change
		character:han_fu = {
			every_held_title = {
				limit = {
					is_noble_family_title = yes
				}
				save_scope_as = hd4_han_family
				character:han_fu = {
					destroy_title = scope:hd4_han_family
				}
			}
			if = {
				limit = {
					is_imprisoned = yes
				}
				release_from_prison = yes
			}
			if = {
				limit = {
					is_ruler = no
				}
				set_employer = character:yuan_shao
			}
		}
		if = {
			limit = {
				title:h_china = {
					NOT = {
						any_in_list = {
							variable = hd4_han_titles
							holder = character:han_fu
						}
					}
					NOT = {
						any_in_list = {
							variable = hd4_han_vassals
							is_alive = yes
							liege = character:han_fu
						}
					}
				}
				title:k_uuii_jizhou.holder = character:yuan_shao
				OR = {
					NOT = {
						exists = global_var:hd_yuan_jizhou_war
					}
					AND = {
						exists = global_var:hd_yuan_jizhou_war
						global_var:hd_yuan_jizhou_war = {
							primary_defender = character:yuan_shao
						}
					}
				}
			}
			set_global_variable = hd_yuan_jizhou_handover_complete
			set_global_variable = hd_yuan_jizhou_realm_transferred
			character:yuan_shao = {
				if = {
					limit = {
						NOT = {
							has_relation_rival = character:gong_sun_zan
						}
					}
					set_relation_rival = {
						target = character:gong_sun_zan
						reason = rival_historical
					}
				}
				add_character_flag = hd_yuan_jizhou_claim_received
				if = {
					limit = {
						NOT = {
							has_character_flag = hd4_jizhou_handover_notified
						}
					}
					add_character_flag = hd4_jizhou_handover_notified
					trigger_event = hd_yuan_jizhou.0011
				}
			}
		}
	}
}
hd4_gsz_defeat_effect = {
	character:gong_sun_zan = {
		if = {
			limit = {
				is_alive = yes
				NOT = {
					has_character_flag = hd4_gsz_defeat_pending
				}
			}
			add_character_flag = hd4_gsz_defeat_pending
			hd4_gsz_snapshot_effect = yes
		}
	}
}
hd4_gsz_snapshot_effect = {
	if = {
		limit = {
			NOT = {
				title:h_china = {
					has_variable = hd4_gsz_snapshot_taken
				}
			}
		}
		title:h_china = {
			clear_variable_list = hd4_gsz_titles
		}
		if = {
			limit = {
				exists = primary_heir
			}
			primary_heir = {
				add_character_flag = hd4_gsz_release_member
			}
		}
		every_vassal_or_below = {
			limit = {
				is_alive = yes
				is_landed = yes
				highest_held_title_tier >= tier_county
			}
			add_character_flag = hd4_gsz_release_member
			save_scope_as = hd4_gsz_old_holder
			primary_title = {
				set_variable = {
					name = hd4_gsz_old_holder
					value = scope:hd4_gsz_old_holder
				}
				save_scope_as = hd4_gsz_ticket
				title:h_china = {
					add_to_variable_list = {
						name = hd4_gsz_titles
						target = scope:hd4_gsz_ticket
					}
				}
			}
		}
		title:h_china = {
			set_variable = hd4_gsz_snapshot_taken
			set_variable = hd4_gsz_release_active
			set_variable = {
				name = hd4_gsz_release_queued
				value = yes
				days = 2
			}
			trigger_event = {
				on_action = hd4_gongsun_release_tick
				days = 1
			}
			trigger_event = {
				on_action = hd4_gongsun_release_tick
				days = 15
			}
			trigger_event = {
				on_action = hd4_gongsun_release_tick
				days = 29
			}
		}
	}
}
hd4_gsz_release_tick_effect = {
	if = {
		limit = {
			has_variable = hd4_gsz_release_active
			NOT = {
				has_variable = hd4_gsz_ran_today
			}
		}
		set_variable = {
			name = hd4_gsz_ran_today
			value = yes
			days = 1
		}
		remove_variable = hd4_gsz_release_queued
		character:gong_sun_zan ?= {
			if = {
				limit = {
					is_alive = yes
					has_character_flag = hd4_gsz_defeat_pending
				}
				death = {
					death_reason = death_suicide
				}
			}
		}
		every_in_list = {
			variable = hd4_gsz_titles
			limit = {
				has_variable = hd4_gsz_old_holder
				NOT = {
					has_variable = hd4_gsz_ticket_done
				}
			}
			save_scope_as = hd4_gsz_ticket
			if = {
				limit = {
					AND = {
						exists = holder
						holder = {
							is_alive = yes
							is_landed = yes
						}
					}
				}
				holder = {
					if = {
						limit = {
							OR = {
								AND = {
									exists = liege
									liege = {
										this = character:gong_sun_zan
									}
								}
								any_liege_or_above = {
									has_character_flag = hd4_gsz_release_member
								}
							}
						}
						hd4_release_character_effect = yes
					}
					if = {
						limit = {
							OR = {
								is_independent_ruler = yes
								NOT = {
									any_liege_or_above = {
										has_character_flag = hd4_gsz_release_member
									}
								}
							}
						}
						scope:hd4_gsz_ticket = {
							set_variable = hd4_gsz_ticket_done
						}
					}
				}
			}
			else = {
				set_variable = hd4_gsz_ticket_done
			}
		}
		if = {
			limit = {
				NOT = {
					any_in_list = {
						variable = hd4_gsz_titles
						NOT = {
							has_variable = hd4_gsz_ticket_done
						}
					}
				}
			}
			remove_variable = hd4_gsz_release_active
			if = {
				limit = {
					NOT = {
						has_variable = hd4_gsz_end_notified
					}
				}
				set_variable = hd4_gsz_end_notified
				character:yuan_shao ?= {
					if = {
						limit = {
							is_alive = yes
						}
						trigger_event = hd_yuan_jizhou.0090
					}
				}
			}
		}
		else = {
			set_variable = {
				name = hd4_gsz_release_queued
				value = yes
				days = 2
			}
			trigger_event = {
				on_action = hd4_gongsun_release_tick
				days = 1
			}
		}
	}
}
