# Rebuild display pointers from actual titles; no monthly/global character sweep.
on_title_gain = { on_actions = { WJ_rf_refresh } }
on_title_gain_inheritance = { on_actions = { WJ_rf_inherit WJ_rf_refresh } }
on_death = { on_actions = { WJ_rf_before_death } }
on_title_destroyed = { on_actions = { WJ_rf_destroyed } }
on_title_lost = { on_actions = { WJ_rf_refresh } }
WJ_rf_refresh = { trigger = { is_alive = yes } effect = { WJ_rf_refresh_holder = yes } }
WJ_rf_before_death = { trigger = { WJ_rf_has_fief = yes } effect = { WJ_rf_capture_succession_liege = yes WJ_rf_capture_ministry = yes } }
WJ_rf_inherit = {
 trigger = { is_alive = yes exists = scope:title scope:title = { has_variable = wj_rf_rank } }
 effect = { WJ_rf_restore_succession_liege = yes WJ_rf_refresh_holder = yes WJ_rf_queue_ministry = yes }
}
WJ_rf_destroyed = {
 trigger = { exists = scope:landed_title scope:landed_title = { has_variable = wj_rf_rank } }
 effect = {
  scope:landed_title = {
   if = { limit = { var:wj_rf_rank = 1 } WJ_rf_restore_destroyed_fief = yes }
   else_if = { limit = { var:wj_rf_rank = 2 exists = var:wj_rf_constituent_fief }
    var:wj_rf_constituent_fief = {
     save_scope_as = wj_rf_orphan_fief
     if = { limit = { NOT = { exists = current_heir } }
      WJ_rf_restore_destroyed_fief = yes
      root = { destroy_title = scope:wj_rf_orphan_fief }
     }
    }
   }
  }
 }
}
