# Append named children: never replace native effect/events blocks.
on_war_started = { on_actions = { hd_submission_on_war_started } }
on_war_won_attacker = { on_actions = { hd_submission_on_attacker_victory } }
on_war_won_defender = { on_actions = { hd_submission_on_attacker_defeat } }
on_war_white_peace = { on_actions = { hd_submission_on_white_peace } }
yearly_playable_pulse = { on_actions = { hd_submission_yearly_maintenance } }

hd_submission_on_war_started = {
    trigger = { hd_submission_can_track_war_trigger = yes }
    effect = {
        scope:war = {
            set_variable = { name = hd_submission_war_actor value = scope:attacker }
            set_variable = { name = hd_submission_war_start_size value = scope:attacker.sub_realm_size }
        }
    }
}

hd_submission_on_attacker_victory = {
    trigger = { hd_submission_tracked_war_unsettled_trigger = yes }
    effect = {
        scope:war = { set_variable = hd_submission_war_scored }
        scope:attacker = {
            if = {
                limit = {
                    is_landed = yes
                    is_independent_ruler = yes
                    government_has_flag = government_is_celestial
                    sub_realm_size > scope:war.var:hd_submission_war_start_size
                }
                hd_submission_momentum_victory_effect = yes
            }
            else = {
                remove_character_flag = hd_submission_action_cooldown
                hd_submission_queue_followup_effect = { DELAY_MIN = 60 DELAY_MAX = 120 }
            }
        }
    }
}

hd_submission_on_attacker_defeat = {
    trigger = { hd_submission_tracked_war_unsettled_trigger = yes }
    effect = {
        scope:war = { set_variable = hd_submission_war_scored }
        scope:attacker = { hd_submission_momentum_defeat_effect = yes }
    }
}

hd_submission_on_white_peace = {
    trigger = { hd_submission_tracked_war_unsettled_trigger = yes }
    effect = {
        scope:war = { set_variable = hd_submission_war_scored }
        scope:attacker = { hd_submission_momentum_white_peace_effect = yes }
    }
}

hd_submission_yearly_maintenance = {
    trigger = {
        OR = {
            has_variable = hd_submission_momentum
            AND = {
                is_ai = yes
                is_independent_ruler = yes
                government_has_flag = government_is_celestial
            }
        }
    }
    effect = { hd_submission_yearly_maintenance_effect = yes }
}
