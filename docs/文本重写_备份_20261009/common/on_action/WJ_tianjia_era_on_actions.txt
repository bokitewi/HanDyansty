yearly_global_pulse = {
    on_actions = { 
        SanGong_NianHao_add_action
    }  
}

on_game_start_after_lobby = { on_actions = { WJ_era_184_start } }
WJ_era_184_start = {
 trigger = { game_start_date = 184.1.1 }
 effect = {
  title:h_china.holder ?= {
   if = { limit = { this = character:liu_hong_3 WJ_era_adopter_trigger = yes NOT = { has_variable = SG_NianHao_data } }
    # 光和七年: 中平 was proclaimed in the last lunar month, not at the January bookmark.
    remove_all_NianHao = yes
    set_variable = data_nianhao_to_SG
    set_variable = SG_nianhao_guanghe
    set_variable = { name = SG_NianHao_data value = 7 }
    set_global_variable = { name = SG_nianhao_guanghe years = 100 value = yes }
    add_character_flag = SanGong_NianHao_flag
   }
  }
 }
}

SanGong_NianHao_add_action = {#年号增加
    effect = {
    every_ruler = {
            if = {
                limit = {
                    WJ_era_adopter_trigger = yes
                    has_variable = SG_NianHao_data
                }
                change_variable = { name = SG_NianHao_data add = 1 }
            }
        }
    }
}
