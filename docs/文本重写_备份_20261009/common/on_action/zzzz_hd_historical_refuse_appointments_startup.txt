# HanDynasty: 184 开局给部分历史人物挂原版“辞谢新任命”状态（同 tgp_refuse_appointments_decision，150 年）
# 防止 5 年任期冷却（RECENT_APPOINTMENT_COOLDOWN）结束后被天朝任命调离本任。
# 玩家可用原版“恢复任命”决议解除。

on_game_start_after_lobby = {
	on_actions = {
		hd_historical_refuse_appointments_184_after_lobby
	}
}

hd_historical_refuse_appointments_184_after_lobby = {
	trigger = {
		game_start_date = 184.1.1
		current_date = 184.1.1
		NOT = { exists = global_var:hd_historical_refuse_appointments_184_initialized }
	}
	effect = {
		character:dong_zhuo_2 ?= { add_to_list = hd_refuse_appointments_list }	# 董卓
		character:gong_sun_zan ?= { add_to_list = hd_refuse_appointments_list }	# 公孙瓒
		character:han_fu ?= { add_to_list = hd_refuse_appointments_list }		# 韩馥
		character:liu_biao ?= { add_to_list = hd_refuse_appointments_list }		# 刘表
		character:ding_yuan ?= { add_to_list = hd_refuse_appointments_list }	# 丁原
		character:liu_dai ?= { add_to_list = hd_refuse_appointments_list }		# 刘岱（兖州）
		character:he_jin ?= { add_to_list = hd_refuse_appointments_list }		# 何进
		character:liu_yan_5 ?= { add_to_list = hd_refuse_appointments_list }	# 刘焉（益州）
		character:liu_yu_3 ?= { add_to_list = hd_refuse_appointments_list }		# 刘虞
		character:ma_teng ?= { add_to_list = hd_refuse_appointments_list }		# 马腾
		character:han_sui ?= { add_to_list = hd_refuse_appointments_list }		# 韩遂
		character:tao_qian ?= { add_to_list = hd_refuse_appointments_list }		# 陶谦
		character:chen_gong ?= { add_to_list = hd_refuse_appointments_list }	# 陈宫

		every_in_list = {
			list = hd_refuse_appointments_list
			limit = { is_alive = yes }
			set_appointment_timeout = {
				years = 150
				desc = appointment_timeout_desc_refusing
			}
		}

		set_global_variable = hd_historical_refuse_appointments_184_initialized
	}
}
