# 天朝循环·时代特色：紧张期套件（docs/天朝循环_时代特色重制计划.md §3.3）—— 只追加 on_actions 子项

# 乱象（仅紧张期，离开即清除）、中兴改革进度、终身辅政保全、党锢禁任清查：各阶段都运行
dynastic_cycle_yearly_pulse = { on_actions = { hd_dc_kit_ten_yearly } }
hd_dc_kit_ten_yearly = {
	effect = { hd_dc_kit_ten_yearly_effect = yes }
}

# 弹劾：伪造过罪证的弹劾被识破 → 罪行昭彰（root = 阴谋）
on_scheme_discovered = { on_actions = { hd_dc_kit_ten_on_scheme_discovered } }
hd_dc_kit_ten_on_scheme_discovered = {
	trigger = {
		scheme_type = hd_dc_ten_impeach
		has_variable = hd_dc_ten_forged
	}
	effect = { hd_dc_kit_ten_forgery_discovered_effect = yes }
}

# 清君侧所立终身辅政：辅政变动时排队补回（root = 君主）
on_diarch_change = { on_actions = { hd_dc_kit_ten_regent_watch } }
on_left_diarchy = { on_actions = { hd_dc_kit_ten_regent_watch } }
hd_dc_kit_ten_regent_watch = {
	trigger = {
		exists = title:h_china.var:hd_dc_ten_qjc_regent
	}
	effect = {
		title:h_china.holder ?= { hd_dc_kit_ten_regent_repair_check_effect = yes }
	}
}

# 终身辅政本人身故 / 登基：清除记录（root = 死者）
on_death = { on_actions = { hd_dc_kit_ten_on_death } }
hd_dc_kit_ten_on_death = {
	trigger = {
		title:h_china.var:hd_dc_ten_qjc_regent ?= root
	}
	effect = {
		title:h_china = { remove_variable = hd_dc_ten_qjc_regent }
	}
}
