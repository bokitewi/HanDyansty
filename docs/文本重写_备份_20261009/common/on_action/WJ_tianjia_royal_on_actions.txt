on_game_start_after_lobby = { on_actions = { WJ_royal_start } }
on_title_gain = { on_actions = { WJ_royal_accession } }
yearly_playable_pulse = { on_actions = { WJ_royal_yearly } }
on_vassal_change = { on_actions = { WJ_royal_identity_changed } }
on_government_change = { on_actions = { WJ_royal_identity_changed } }
on_liege_government_change = { on_actions = { WJ_royal_identity_changed } }
on_join_court = { on_actions = { WJ_royal_identity_changed } }
on_leave_court = { on_actions = { WJ_royal_identity_changed } }
on_death = { on_actions = { WJ_royal_prince_death } }
WJ_royal_start = {
 effect = { every_ruler = { limit = { WJ_inner_emperor_trigger = yes } trigger_event = { id = SanGong_DianZhang.015 days = 15 } } }
}
WJ_royal_accession = {
 trigger = { WJ_inner_emperor_trigger = yes scope:title = root.primary_title }
 effect = {
  WJ_royal_clear_prince_effect = yes
  if = { limit = { exists = scope:previous_holder }
   scope:previous_holder.var:WJ_royal_heir ?= { WJ_royal_clear_prince_effect = yes }
  }
  if = {
   limit = { exists = scope:previous_holder }
   scope:previous_holder = {
    every_child = {
     limit = { is_alive = yes has_trait = SanGong_Taizi var:WJ_royal_patron = prev }
     WJ_royal_clear_prince_effect = yes
    }
   }
  }
  trigger_event = { id = SanGong_DianZhang.015 days = 15 }
 }
}
WJ_royal_yearly = {
 effect = {
  if = { limit = { WJ_inner_emperor_trigger = yes } trigger_event = SanGong_DianZhang.015 }
  else_if = { limit = { has_trait = SanGong_Taizi } trigger_event = SanGong_DianZhang.901 }
 }
}
WJ_royal_identity_changed = {
 trigger = { has_trait = SanGong_Taizi NOT = { has_variable = WJ_royal_installing } }
 effect = { trigger_event = { id = SanGong_DianZhang.901 days = 1 } }
}
WJ_royal_prince_death = {
 trigger = { has_trait = SanGong_Taizi }
 effect = {
  # Capture before this independent on_death chain clears prince identity.
  WJ_name_before_death = yes
  save_scope_as = Taizi
  if = { limit = { exists = var:WJ_royal_patron var:WJ_royal_patron = { WJ_inner_emperor_trigger = yes } house = var:WJ_royal_patron.house }
   var:WJ_royal_patron = { save_scope_as = HuangDi
    if = { limit = { scope:Taizi = { is_adult = yes } } trigger_event = { id = ZongWang_ShiHao.020 days = 1 } }
    else = { trigger_event = { id = ZongWang_ShiHao.021 days = 1 } }
   }
  }
  WJ_royal_clear_prince_effect = yes
 }
}
