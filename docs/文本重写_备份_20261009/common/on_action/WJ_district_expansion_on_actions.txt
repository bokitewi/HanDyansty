on_war_started = { on_actions = { WJ_dd_rebellion_war_started } }
on_join_war_as_secondary = { on_actions = { WJ_dd_rebellion_joined } }
WJ_dd_rebellion_war_started = {
 effect = {
  scope:war = { every_war_participant = {
   limit = { WJ_dd_rebellion = yes }
   var:wj_dd_held_core = { WJ_dd_dissolve_core_effect = yes }
  } }
 }
}
WJ_dd_rebellion_joined = {
 trigger = { WJ_dd_rebellion = yes }
 effect = { var:wj_dd_held_core = { WJ_dd_dissolve_core_effect = yes } }
}
