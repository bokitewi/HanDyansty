# Initializes the Eastern Han Great Wall on the route selected for the Han Dynasty map.
# Keep this in a project-owned child on_action so the compatibility override of
# vanilla game_start.txt does not need to duplicate vanilla startup logic.
on_game_start = {
	on_actions = {
		hd_initialize_han_great_wall
	}
}

hd_initialize_han_great_wall = {
	trigger = {
		has_tgp_dlc_trigger = yes
		game_start_date = 184.1.1
		NOT = { exists = global_var:hd_han_great_wall_initialized }
	}

	effect = {
		# Hexi Corridor main line
		province:5097 = { add_great_building = the_great_wall_02 } # Yang Pass
		province:5007 = { add_great_building = the_great_wall_02 } # Dunhuang
		province:6769 = { add_great_building = the_great_wall_02 } # Yumen Pass
		province:5805 = { add_great_building = the_great_wall_02 } # Yumen
		province:6780 = { add_great_building = the_great_wall_02 } # Jiuquan
		province:1566 = { add_great_building = the_great_wall_02 } # Zhangye
		province:1564 = { add_great_building = the_great_wall_02 } # Guzang
		province:1551 = { add_great_building = the_great_wall_02 } # Lingju

		# Juyan branch
		province:5923 = { add_great_building = the_great_wall_02 } # Eastern Juyan

		# Hetao and Dai Commandery line
		province:1577 = { add_great_building = the_great_wall_02 } # Shuofang
		province:4725 = { add_great_building = the_great_wall_02 } # Wuyuan
		province:1332 = { add_great_building = the_great_wall_02 } # Yunzhong
		province:1338 = { add_great_building = the_great_wall_02 } # Dingxiang
		province:1321 = { add_great_building = the_great_wall_02 } # Wuzhou Fort
		province:1314 = { add_great_building = the_great_wall_02 } # Gaoliu

		# Yanshan and Liaoxi line
		province:1283 = { add_great_building = the_great_wall_02 } # Juyong Pass
		province:1245 = { add_great_building = the_great_wall_02 } # Yuyang
		province:1258 = { add_great_building = the_great_wall_02 } # Wuzhong
		province:2609 = { add_great_building = the_great_wall_02 } # Junmi
		province:2558 = { add_great_building = the_great_wall_02 } # Lingzhi
		province:3489 = { add_great_building = the_great_wall_02 } # Linyu
		province:3482 = { add_great_building = the_great_wall_02 } # Tuhe
		province:3480 = { add_great_building = the_great_wall_02 } # Yangle

		set_global_variable = hd_han_great_wall_initialized
	}
}
