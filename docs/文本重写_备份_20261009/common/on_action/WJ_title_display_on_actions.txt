on_game_start_after_lobby = { on_actions = { WJ_titles_start } }
on_title_gain = { on_actions = { WJ_titles_changed } }
on_title_lost = { on_actions = { WJ_titles_changed } }
on_vassal_change = { on_actions = { WJ_titles_changed } }
on_government_change = { on_actions = { WJ_titles_changed } }
on_liege_government_change = { on_actions = { WJ_titles_changed } }
on_character_faith_change = { on_actions = { WJ_titles_changed } }
yearly_global_pulse = { on_actions = { WJ_titles_start } }
WJ_titles_start = { effect = { every_ruler = { WJ_title_style_reconcile = yes } } }
WJ_titles_changed = {
 trigger = { is_alive = yes NOT = { has_character_flag = hd_title_style_reconcile_pending } }
 effect = {
  add_character_flag = hd_title_style_reconcile_pending
  trigger_event = { on_action = WJ_titles_after_change days = 1 }
 }
}
WJ_titles_after_change = {
 effect = {
  remove_character_flag = hd_title_style_reconcile_pending
  if = { limit = { is_alive = yes } WJ_title_style_reconcile_realm = yes }
 }
}
