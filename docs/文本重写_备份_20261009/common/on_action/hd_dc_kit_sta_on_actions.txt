# 天朝循环·时代特色：对峙期套件挂钩（只追加 on_actions 子项，不覆盖原版 effect）
# 见 docs/天朝循环_时代特色重制计划.md §3.6

# 质子：送子女做他国独立统治者的养子 → 双方停战 10 年（仿原版 apply_truce_when_sending_ward）
# Root = 养子，scope:target = 监护人
on_set_relation_guardian = { on_actions = { hd_dc_sta_on_set_relation_guardian } }
hd_dc_sta_on_set_relation_guardian = {
	trigger = { exists = situation:dynastic_cycle }
	effect = { hd_dc_sta_hostage_set_effect = yes }
}

# 质子离开：同原版，解除因送质而得的停战
on_remove_relation_guardian = { on_actions = { hd_dc_sta_on_remove_relation_guardian } }
hd_dc_sta_on_remove_relation_guardian = {
	trigger = { exists = var:hd_dc_sta_hostage_guardian }
	effect = { hd_dc_sta_hostage_removed_effect = yes }
}

# 策反阴谋败露。Root = 阴谋
on_scheme_discovered = { on_actions = { hd_dc_sta_on_scheme_discovered } }
hd_dc_sta_on_scheme_discovered = {
	trigger = {
		hd_dc_sta_is_subversion_scheme_trigger = yes
		NOT = { has_variable = hd_dc_sta_exposure_handled }
	}
	effect = {
		set_variable = {
			name = hd_dc_sta_exposure_handled
			value = yes
		}
		scheme_owner = { save_scope_as = hd_dc_sta_sub_owner }
		scheme_target_character = { save_scope_as = hd_dc_sta_sub_target }
		hd_dc_sta_subversion_exposed_effect = yes
	}
}
