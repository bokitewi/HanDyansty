# 天朝循环·时代特色：崩溃期套件钩子（只追加 on_actions 子项，不覆盖原版）

# 勤王诏：天子一方获胜 → 响应者奖励 + 勤王成功诱因。
# 必须写在 effect 中：原版注释说明 events 在下一 tick 才执行，届时战争（及其变量列表）已被销毁。
on_war_won_defender = { on_actions = { hd_dc_col_on_war_won_defender } }
hd_dc_col_on_war_won_defender = {
	trigger = {
		exists = scope:war
		scope:war = { has_variable = hd_dc_col_qinwang }
	}
	effect = {
		hd_dc_col_qinwang_victory_effect = yes
	}
}

# 授权募兵：天子每年向被授权封臣返还其上缴税额估计值的 50%
dynastic_cycle_yearly_pulse = { on_actions = { hd_dc_col_yearly } }
hd_dc_col_yearly = {
	effect = {
		hd_dc_col_militia_yearly_refund_effect = yes
	}
}
