# 瘴气开局挂载：
# 对 china_south_world（k_uuii_guangzhou、k_uuii_jiaozhou 两王国及十九公国）全部县
# 永久挂上 hd_miasma_county_modifier。模式照搬原版 on_game_start 的 GREECE BACKWATER COUNTIES 块，
# 改为项目自有子 on_action + 全局变量守卫（本 mod 惯例，不给原版 on_action 追加第二个 effect）。
on_game_start = {
	on_actions = {
		hd_miasma_counties_startup
	}
}

hd_miasma_counties_startup = {
	trigger = {
		NOT = { exists = global_var:hd_miasma_counties_initialized }
	}
	effect = {
		every_county_in_region = {
			region = china_south_world
			add_county_modifier = hd_miasma_county_modifier
		}
		set_global_variable = hd_miasma_counties_initialized
	}
}
