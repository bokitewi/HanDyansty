on_game_start_after_lobby = {
	on_actions = {
		WJ_minister_seat_titles_after_lobby
	}
}

WJ_minister_seat_titles_after_lobby = {
	trigger = {
		has_tgp_dlc_trigger = yes
		exists = title:h_china.holder
	}
	effect = {
		title:h_china.holder = {
			if = {
				limit = {
					OR = {
						government_has_flag = government_is_celestial
						government_has_flag = government_is_celestial
					}
				}
				WJ_minister_seat_fill_effect = yes
				WJ_minister_seat_titles_effect = yes
				trigger_event = { id = WJ_minister_seat.0100 days = 7 }
				trigger_event = { id = WJ_minister_seat.0100 days = 30 }
			}
		}
	}
}
