# 原版 cycle_vizierate_politicking_events_on_action 由 on_entered_diarchy 启动，
# vizierate.0051 等事件每 2-4 年无条件重排自身；每次重新进入共治都会多出一条永不终止的链。
# 原版此 on_action 无 trigger/effect（_on_actions.info：同名 on_action 只能有一个 trigger/effect），故在此追加。
# 1) 仅在维齐尔共治时保留：原版所有实际政治事件都要求 diarchy_is_vizierate，其余链只会空转 0051。
# 2) 去重：同一领主 700 天内只放行一次；单链间隔 >= 730 天，不会挡住自身，多余链自然断开。
cycle_vizierate_politicking_events_on_action = {
	trigger = {
		has_diarchy_active_parameter = diarchy_is_vizierate
		NOT = { has_variable = hd_vizierate_cycle_lock }
	}
	effect = {
		set_variable = {
			name = hd_vizierate_cycle_lock
			value = yes
			days = 700
		}
	}
}
