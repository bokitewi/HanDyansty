# 原版 on_title_gain 只枚举九个内阁头衔；太傅、宗正补充同样的授职回调。
on_title_gain = {
	on_actions = {
		WJ_tai_fu_title_gain
		WJ_zong_zheng_title_gain
		WJ_guang_lu_xun_title_gain
	}
}

WJ_tai_fu_title_gain = {
	trigger = {
		scope:title = title:e_WJ_minister_tai_fu
		is_alive = yes
		is_independent_ruler = no
		government_has_flag = government_is_celestial
		scope:title.holder = root
		exists = title:h_china.holder
		liege = top_liege
		top_liege = title:h_china.holder
		root != title:h_china.holder
		title:h_china.holder = { government_has_flag = government_is_celestial }
	}
	effect = {
		title:h_china.holder = {
			save_scope_as = councillor_liege
		}
		root = {
			got_minister_position_effect = { MINISTER_TITLE = e_WJ_minister_tai_fu MINISTER_POSITION = WJ_minister_tai_fu }
		}
	}
}
