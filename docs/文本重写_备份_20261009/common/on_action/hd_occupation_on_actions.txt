# 天朝循环大修：战时占领记录（只追加 on_actions 子项）

on_siege_completion = {
	on_actions = { hd_occupation_on_siege }
}
hd_occupation_on_siege = {
	trigger = {
		exists = scope:war
		scope:war = { hd_occupation_war_trigger = yes }
		scope:war = { is_attacker = root }
	}
	effect = { hd_occupation_record_effect = yes }
}

on_liberation_siege_completion = {
	on_actions = { hd_occupation_on_liberation }
}
hd_occupation_on_liberation = {
	trigger = {
		exists = scope:war
		scope:war = { hd_occupation_war_trigger = yes }
	}
	effect = { hd_occupation_liberate_effect = yes }
}
