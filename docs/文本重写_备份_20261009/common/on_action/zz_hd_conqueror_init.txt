# 原版只在开局为成吉思汗等史实人物创建征服者故事时才生成 global_var:active_conquerors，
# 而 AI 五年一次的征服者掷骰（ai_chance_to_become_conqueror_effect）要求该变量存在。
# 184 年开局没有这些人物，变量永远不存在，AI 征服者机制被卡死。这里在开局初始化为 0。
on_game_start = {
	on_actions = { hd_conqueror_init_on_game_start }
}

hd_conqueror_init_on_game_start = {
	effect = {
		if = {
			limit = {
				NOT = { exists = global_var:active_conquerors }
			}
			set_global_variable = {
				name = active_conquerors
				value = 0
			}
		}
	}
}
