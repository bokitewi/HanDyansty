on_game_start_after_lobby = {
	on_actions = {
		hd4_after_lobby
	}
}
quarterly_playable_pulse = {
	on_actions = {
		hd4_playable_recovery
	}
}
on_death = {
	on_actions = {
		hd4_before_death
	}
}
hd4_after_lobby = {
	trigger = {
		game_start_date = 184.1.1
	}
	effect = {
		hd4_bootstrap_effect = yes
	}
}
hd4_playable_recovery = {
	trigger = {
		game_start_date = 184.1.1
		OR = {
			this = character:tao_qian
			this = character:gong_sun_zan
			this = character:cao_cao
			this = character:yuan_shao
		}
	}
	effect = {
		hd4_bootstrap_effect = yes
	}
}
hd4_heartbeat = {
	effect = {
		title:h_china = {
			if = {
				limit = {
					NOT = {
						has_variable = hd4_tick_ran_today
					}
				}
				set_variable = {
					name = hd4_tick_ran_today
					value = yes
					days = 1
				}
				remove_variable = hd4_tick_queued
				hd4_tick_effect = yes
			}
			hd4_schedule_tick_effect = yes
		}
	}
}
hd4_gongsun_release_tick = {
	effect = {
		title:h_china = {
			hd4_gsz_release_tick_effect = yes
		}
	}
}
hd4_yuan_partition_tick = {
	effect = {
		title:h_china = {
			hd4_yuan_partition_tick_effect = yes
		}
	}
}
hd4_before_death = {
	trigger = {
		game_start_date = 184.1.1
		OR = {
			AND = {
				this = character:yuan_shao
				has_character_flag = hd4_yuan_partition_armed
			}
			AND = {
				this = character:gong_sun_zan
				has_character_flag = hd4_gsz_defeat_pending
			}
		}
	}
	effect = {
		if = {
			limit = {
				this = character:yuan_shao
			}
			hd4_yuan_snapshot_death_effect = yes
		}
		else_if = {
			limit = {
				this = character:gong_sun_zan
			}
			hd4_gsz_snapshot_effect = yes
		}
	}
}
