# 汉代宝物开局发放 - Han treasures startup distribution
# 184开局：传国玉玺/王莽头/和氏璧 → 汉帝（liu_hong_3）；太平要术 → 张角（zhang_jiao）
# 其余28件宝物的创建效果已就绪，等待后续事件链接入
on_game_start = {
	on_actions = {
		hd_han_treasures_startup
	}
}

hd_han_treasures_startup = {
	trigger = {
		game_start_date = 184.1.1
		current_date = 184.1.1
		NOT = { exists = global_var:hd_han_treasures_initialized }
	}
	effect = {
		character:liu_hong_3 ?= {
			if = {
				limit = { is_alive = yes }
				hd_han_create_imperial_seal_effect = { OWNER = this }
				hd_han_create_wang_mang_head_effect = { OWNER = this }
				hd_han_create_heshi_bi_effect = { OWNER = this }
			}
		}
		character:zhang_jiao ?= {
			if = {
				limit = { is_alive = yes }
				hd_han_create_taiping_yaoshu_effect = { OWNER = this }
			}
		}
		set_global_variable = hd_han_treasures_initialized
	}
}
