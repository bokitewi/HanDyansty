### A. 仕历：汉晋正式官员每年 +4 ###
random_yearly_everyone_pulse = {
	on_actions = {
		WJ_ziwang_shili_yearly_action
	}
}

WJ_ziwang_shili_yearly_action = {
	trigger = {
		government_has_flag = government_has_merit
		government_has_flag = government_is_celestial
		any_held_title = {
			tier >= tier_county
			is_noble_family_title = no
		}
	}
	effect = {
		WJ_ziwang_shili_xp_effect = { VALUE = 4 }
	}
}

### B. 武勋：仅奖励战斗与战争胜方 ###
on_combat_end_winner = {
	on_actions = {
		WJ_ziwang_combat_winner_action
	}
}

WJ_ziwang_combat_winner_action = {
	effect = {
		every_side_commander = {
			limit = { government_has_flag = government_is_celestial }
			WJ_ziwang_wuxun_xp_effect = { VALUE = 2 }
		}
		every_side_knight = {
			limit = { government_has_flag = government_is_celestial }
			WJ_ziwang_wuxun_xp_effect = { VALUE = 1 }
		}
	}
}

on_war_won_attacker = {
	on_actions = {
		WJ_ziwang_war_won_attacker_action
	}
}

WJ_ziwang_war_won_attacker_action = {
	effect = {
		scope:attacker = {
			save_scope_as = wj_ziwang_war_winner
			if = {
				limit = { government_has_flag = government_is_celestial }
				WJ_ziwang_wuxun_xp_effect = { VALUE = 12 }
			}
		}
		scope:war = {
			every_war_participant = {
				limit = {
					is_attacker_in_war = scope:war
					is_ruler = yes
					NOT = { this = scope:wj_ziwang_war_winner }
				}
				if = {
					limit = { government_has_flag = government_is_celestial }
					WJ_ziwang_wuxun_xp_effect = { VALUE = 4 }
				}
			}
		}
	}
}

on_war_won_defender = {
	on_actions = {
		WJ_ziwang_war_won_defender_action
	}
}

WJ_ziwang_war_won_defender_action = {
	effect = {
		scope:defender = {
			save_scope_as = wj_ziwang_war_winner
			if = {
				limit = { government_has_flag = government_is_celestial }
				WJ_ziwang_wuxun_xp_effect = { VALUE = 12 }
			}
		}
		scope:war = {
			every_war_participant = {
				limit = {
					is_defender_in_war = scope:war
					is_ruler = yes
					NOT = { this = scope:wj_ziwang_war_winner }
				}
				if = {
					limit = { government_has_flag = government_is_celestial }
					WJ_ziwang_wuxun_xp_effect = { VALUE = 4 }
				}
			}
		}
	}
}
