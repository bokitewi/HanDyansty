# Monthly pulse for the nine 184-196 historical regional revolts (phase 1).
# The dynastic-cycle situation's monthly pulse is the mod's global monthly heartbeat;
# the project-owned child on_action only appends to it.

dynastic_cycle_monthly_pulse = {
	on_actions = {
		hd_historical_revolt_monthly_check
		hd_historical_world_monthly_check
	}
}

			hd_historical_revolt_jiaozhi_effect = yes
		}

			hd_historical_revolt_beigong_effect = yes
		}

			hd_historical_revolt_heishan_effect = yes
		}
		if = {

			hd_historical_revolt_jiangxia_effect = yes
		}
		if = {
			limit = { hd_historical_revolt_wangguo_ready_trigger = yes }
			hd_historical_revolt_wangguo_effect = yes
		}
		if = {
			limit = { hd_historical_revolt_zhangchun_ready_trigger = yes }
			hd_historical_revolt_zhangchun_effect = yes
		}
		if = {

	}
