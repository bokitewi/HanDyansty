# ============================================================================
# WJ 事件驱动试守 —— 只监听实际头衔变化，不做全图周期扫描。
# 原版 on_title_gain/on_title_lost：root 分别为新持有者/旧持有者，scope:title 为变动头衔。
# ============================================================================

on_title_gain = {
	on_actions = {
		WJ_yinju_office_gain_action
		WJ_temp_official_title_gain_action
	}
}

WJ_yinju_office_gain_action = {
	trigger = {
		has_character_flag = wj_yinju_active
		scope:title = {
			tier >= tier_barony
			is_noble_family_title = no
			NOT = { this = title:h_china }
		}
	}
	effect = {
		remove_character_flag = wj_yinju_active
		WJ_clear_owned_appointment_timeout_effect = yes
	}
}

WJ_temp_official_title_gain_action = {
	trigger = {
		scope:title = {
			tier >= tier_barony
			is_noble_family_title = no
			NOT = { this = title:h_china }
		}
		WJ_is_valid_temp_official_trigger = yes
		NOT = { has_character_modifier = WJ_temp_official_modifier }
		NOT = { has_character_flag = wj_temp_official_timer }
		NOT = { has_character_flag = wj_temp_official_kaoke_ready }
	}
	effect = {
		WJ_temp_official_apply_effect = yes
	}
}

on_title_lost = {
	on_actions = {
		WJ_temp_official_title_lost_action
		WJ_yinju_eligibility_lost_action
	}
}

WJ_temp_official_title_lost_action = {
	trigger = {
		# 死亡移交也可能失去头衔；这里仅清理活人试守，死者任期另走死亡封存。
		is_alive = yes
		has_character_modifier = WJ_temp_official_modifier
		WJ_has_substantive_office_trigger = no
	}
	effect = {
		WJ_temp_official_clear_effect = yes
		WJ_clear_owned_appointment_timeout_effect = yes
	}
}

WJ_yinju_eligibility_lost_action = {
	trigger = {
		is_alive = yes
		has_character_flag = wj_yinju_active
		WJ_is_landless_shizu_without_office_trigger = no
	}
	effect = {
		remove_character_flag = wj_yinju_active
		WJ_clear_owned_appointment_timeout_effect = yes
	}
}

# 原版 on_join_court：root 为迁入角色。进入唐制时转换资格，进入外国时只清理。
on_join_court = {
	on_actions = {
		WJ_appointment_state_join_court_action
	}
}

WJ_appointment_state_join_court_action = {
	effect = {
		WJ_reconcile_han_jin_character_state_effect = yes
	}
}
