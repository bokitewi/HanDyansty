# Append only. Do not take ownership of shared engine-hook effect blocks.
on_game_start = {
    on_actions = { hd_audit_levy_startup }
}
on_government_change = {
    on_actions = { hd_audit_levy_character_refresh }
}
on_title_gain = {
    on_actions = { hd_audit_levy_character_refresh }
}
yearly_playable_pulse = {
    on_actions = { hd_audit_levy_character_refresh }
}

# One initialization scan. No recurrent global scan, timer or event queue.
hd_audit_levy_startup = {
    effect = {
        every_living_character = {
            hd_audit_refresh_levy_balance_effect = yes
        }
    }
}

# All three character hooks supply the character as ROOT.
hd_audit_levy_character_refresh = {
    trigger = { is_alive = yes }
    effect = {
        hd_audit_refresh_levy_balance_effect = yes
    }
}
