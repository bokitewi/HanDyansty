# ============================================================================
# WJ 三条入仕途径开局授予 —— 同名文件替换原版/主mod的 game_start.txt
# xinsanguo-MOD 2026-08-19
#
# 背景(已核实):
# 1) 原版 game_start.txt 里的"考试通过授予"段落, 在装主mod后不生效,
#    原因 = 主mod 的 common/on_action/game_start.txt 同名文件把原版整个文件替换掉了
#    (主mod该文件只有3行注释: "Deliberate compatibility override"),
#    所以原版 on_game_start 里的 every_courtier 授予代码不存在于主mod环境。
# 2) 键名 = on_game_start(原版 game_start.txt L2; 主mod的zz_saas/WJ_east文件
#    也用它注册启动子动作 = 该键在主mod环境确实被引擎调度; 同名键跨文件合并,
#    我们的效果与主mod的启动子动作并存)。
# 3) 效果: 只发给 WJ_starting_career_eligible 名单（与宦海沉浮同一批）：h_china 各层级天朝有实职封臣，
#    以及皇帝/天朝封臣的有内阁或宫廷职位廷臣；皇帝本人与无职廷臣不发。等概率辟召、举孝廉或举茂才。
# 4) 举茂才活动虽已退役，但举茂才入仕设定保留，开局与另两条入仕途径等概率生成。
# ============================================================================

on_game_start = {
	effect = {
		### h_china 体系开局入仕资格：与宦海沉浮共用 WJ_starting_career_eligible 名单 ###
		if = {
			limit = {
				exists = title:h_china.holder
				title:h_china.holder = { government_has_flag = government_is_celestial }
			}
			title:h_china.holder = {
				every_courtier = {
					limit = { WJ_starting_career_eligible = yes }
					hidden_effect = { WJ_grant_random_starting_road_effect = yes }
				}
				every_vassal_or_below = {
					if = {
						limit = { WJ_starting_career_eligible = yes }
						hidden_effect = { WJ_grant_random_starting_road_effect = yes }
					}
					every_courtier = {
						limit = { WJ_starting_career_eligible = yes }
						hidden_effect = { WJ_grant_random_starting_road_effect = yes }
					}
				}
			}
		}
	}
}
