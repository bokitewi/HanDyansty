# 天朝循环·时代特色：扩张期钩子（只追加 on_actions 子项，不覆盖原版 on_action）

# 四夷来朝：年度扫描天子的朝贡国（原版没有"成为朝贡国"的 on_action）
dynastic_cycle_yearly_pulse = { on_actions = { hd_dc_kit_exp_yearly } }
hd_dc_kit_exp_yearly = {
	effect = { hd_dc_exp_tributary_scan_effect = yes }
}

# 军功：记录扩张期天子与外邦交战时己方参战将领
on_combat_end_winner = { on_actions = { hd_dc_kit_exp_on_combat_end } }
on_combat_end_loser = { on_actions = { hd_dc_kit_exp_on_combat_end } }
hd_dc_kit_exp_on_combat_end = {
	trigger = {
		exists = title:h_china.holder
		hd_dc_phase_is_trigger = { PHASE = stability_expansion }
		exists = combat.combat_war
	}
	effect = { hd_dc_exp_record_commanders_effect = yes }
}

# 军功：天子战胜外邦，开放 1 年军功窗口
on_war_won_attacker = { on_actions = { hd_dc_kit_exp_on_war_won_attacker } }
hd_dc_kit_exp_on_war_won_attacker = {
	trigger = {
		exists = title:h_china.holder
		hd_dc_phase_is_trigger = { PHASE = stability_expansion }
		scope:attacker = title:h_china.holder
		scope:defender = { hd_dc_is_outsider_trigger = yes }
	}
	effect = {
		scope:attacker = { hd_dc_exp_open_merit_window_effect = yes }
	}
}
on_war_won_defender = { on_actions = { hd_dc_kit_exp_on_war_won_defender } }
hd_dc_kit_exp_on_war_won_defender = {
	trigger = {
		exists = title:h_china.holder
		hd_dc_phase_is_trigger = { PHASE = stability_expansion }
		scope:defender = title:h_china.holder
		scope:attacker = { hd_dc_is_outsider_trigger = yes }
	}
	effect = {
		scope:defender = { hd_dc_exp_open_merit_window_effect = yes }
	}
}
