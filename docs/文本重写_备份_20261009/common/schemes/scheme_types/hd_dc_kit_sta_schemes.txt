# 天朝循环·时代特色：对峙期“策反”阴谋（见 docs/天朝循环_时代特色重制计划.md §3.6，A 方案）
# 在原版“拉拢”(sway) 框架上改写：外交为本，基础阴谋（无密探/机遇），但为敌对、隐秘阴谋，可被对方密探发觉。
# 对象：敌国封臣。成功率主要受其对主君好感的影响。
#   hd_dc_sta_incite_independence：策反独立 —— 目标对其主君发动独立战争
#   hd_dc_sta_incite_defection   ：策反投靠 —— 目标带地投靠阴谋发起者
#   hd_dc_sta_incite_faction     ：煽动谋反 —— 目标加入/建立反对其主君的派系，派系不满 +30，目标兵员 +30%
# 败露（on_scheme_discovered，见 hd_dc_kit_sta_on_actions.txt）：双方关系恶化，计“背盟毁约”。成功：计“策反”。

hd_dc_sta_incite_independence = {
	skill = diplomacy
	desc = hd_dc_sta_incite_independence_desc_general
	success_desc = "HD_DC_STA_INCITE_INDEPENDENCE_SUCCESS_DESC"
	discovery_desc = "HD_DC_STA_SUBVERSION_DISCOVERY_DESC"
	icon = icon_scheme_seizure
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	target_type = character
	category = hostile
	is_secret = yes
	is_basic = yes
	maximum_breaches = 5
	cooldown = { years = 5 }

	speed_per_skill_point = -2.5
	spymaster_speed_per_skill_point = 0
	uses_resistance = no
	base_progress_goal = hd_dc_sta_subversion_phase_days
	base_maximum_success = 90
	minimum_success = 5

	freeze_scheme_when_traveling = yes
	freeze_scheme_when_traveling_target = no

	allow = {
		is_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		hd_dc_kit_sta_tools_trigger = yes
		scope:target = {
			is_adult = yes
			is_imprisoned = no
			hd_dc_sta_subversion_target_trigger = { OWNER = scope:owner }
		}
	}

	valid = {
		is_alive = yes
		is_incapable = no
		is_imprisoned = no
		is_independent_ruler = yes
		scope:target = {
			is_alive = yes
			is_imprisoned = no
			is_independent_ruler = no
			exists = liege
			NOT = { top_liege = scope:owner }
		}
	}

	odds_prediction = {
		add = base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_diplomacy_value
		min = 0
	}

	base_success_chance = {
		base = 10
		hd_dc_sta_subversion_success_chance_modifier = yes
		# 自身兵力足以与主君一战者更易动心
		modifier = {
			add = 10
			desc = hd_dc_sta_scheme_target_strong
			scope:target.max_military_strength >= scope:target.liege.max_military_strength
		}
	}

	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}

	on_start = {
		set_variable = {
			name = apply_countermeasures
			value = flag:political
		}
	}

	on_phase_completed = {
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = {
			save_scope_as = owner
			random = {
				chance = scope:scheme.scheme_success_chance
				save_scope_value_as = {
					name = scheme_successful
					value = yes
				}
			}
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = {
					id = hd_dc_sta.0101
					days = 1
				}
			}
			else = {
				trigger_event = {
					id = hd_dc_sta.0109
					days = 1
				}
			}
		}
	}

	on_monthly = {
		hostile_scheme_monthly_discovery_chance_effect = yes
	}

	on_invalidated = {
		hd_dc_sta_subversion_invalidated_effect = yes
	}
}

		base = 5
		hd_dc_sta_subversion_success_chance_modifier = yes
		# 新主君强于旧主，投靠者才有依靠
		modifier = {
			add = 10
			desc = hd_dc_sta_scheme_owner_stronger
			scope:owner.max_military_strength > scope:target.top_liege.max_military_strength
		}
		modifier = {
			add = -15
			desc = hd_dc_sta_scheme_owner_weaker
			scope:owner.max_military_strength < scope:target.top_liege.max_military_strength
		}
		# 带地来投须与阴谋者接壤
		modifier = {
			add = -30
			desc = hd_dc_sta_scheme_not_bordering
			NOT = {
				scope:target = {
					any_sub_realm_county = {
						any_neighboring_county = { holder.top_liege ?= scope:owner }
					}
				}
			}
		}
	}
