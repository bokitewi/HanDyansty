# Palace honor fragments: only the true imperial employee. Service names use actual employers.
WJ_NameCourtPreceptor = {
 type = character
 text = {
  trigger = { is_alive = yes government_has_flag = government_is_celestial top_liege = { WJ_inner_emperor_trigger = yes }
   has_court_position = grand_preceptor_court_position
   is_court_position_employer = { court_position = grand_preceptor_court_position who = top_liege }
  }
  localization_key = WJ_name_court_grand_preceptor
 }
 text = { fallback = yes localization_key = WJ_name_empty }
}

  setup_scope = {
   if = {
    limit = { WJ_name_court_service_allowed_trigger = yes has_court_position = court_physician_court_position }
    every_court_position_employer = {
     limit = { WJ_name_court_host_trigger = yes root = { is_court_position_employer = { court_position = court_physician_court_position who = prev } } }
     save_scope_as = wj_name_court_physician_employer
    }
   }
  }

   WJ_name_court_service_allowed_trigger = yes has_court_position = court_physician_court_position
