# Preserve the original Balhae name for every non-Fuyu holder.

	type = landed_title

	text = {

			}
			trigger_else = { always = no }
		}
		localization_key = cn_hd184_fuyu
	}

	text = {
		localization_key = cn_balhae
		fallback = yes
	}
}
