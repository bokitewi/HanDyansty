# 天朝循环大修·朋党：界面 scripted GUI（root = 玩家；scope:target = 选中的参与者组）

# "政治派别"页是否显示：只在帝国级朝廷（天子或乱世中的帝国级诸侯）内显示；四夷出身的帝王显示"不预朝政"
hd_party_tab_gui = {
	scope = character
	is_shown = {
		OR = {
			has_variable = hd_party_court
			var:hd_party_my_court ?= { has_variable = hd_party_court }
			AND = {
				hd_cycle_siyi_ruler_trigger = yes
				highest_held_title_tier >= tier_empire
			}
		}
	}
}

# 玩家是选中朋党所在朝廷的君主
hd_party_is_ruler_gui = {
	scope = character
	is_shown = {
		has_variable = hd_party_court
	}
}

# 政治运作：ban
hd_party_policy_ban_gui = {
	scope = character
	saved_scopes = { target }
	is_shown = {
		has_variable = hd_party_court
	}
	is_valid = {
		custom_tooltip = {
			text = hd_party_policy_cd_tt
			OR = {
			AND = {
				scope:target = { participant_group_type = undecided_movement }
				NOT = { has_variable = hd_party_huanguan_policy_cd }
			}
			AND = {
				scope:target = { participant_group_type = pro_hegemon_movement }
				NOT = { has_variable = hd_party_waiqi_policy_cd }
			}
			AND = {
				scope:target = { participant_group_type = expansion_movement }
				NOT = { has_variable = hd_party_wuxun_policy_cd }
			}
			AND = {
				scope:target = { participant_group_type = conservative_movement }
				NOT = { has_variable = hd_party_shizu_policy_cd }
			}
			AND = {
				scope:target = { participant_group_type = advancement_movement }
				NOT = { has_variable = hd_party_hanmen_policy_cd }
			}
			}
		}
	}
	effect = {
		save_scope_as = hd_party_court
		hd_party_save_group_code_effect = { GROUP = scope:target }
		hd_party_set_policy_by_code_effect = { POLICY = ban }
	}
}

# 要求效忠：打开事件 hd_party.0020
hd_party_demand_loyalty_gui = {
	scope = character
	saved_scopes = { target }
	is_shown = {
		has_variable = hd_party_court
	}
	is_valid = {
		custom_tooltip = {
			text = hd_party_loyalty_invalid_tt
			OR = {
			AND = {
				scope:target = { participant_group_type = undecided_movement }
				exists = var:hd_party_huanguan_leader
				NOT = { has_variable = hd_party_huanguan_loyalty_cd }
				NOT = { var:hd_party_huanguan_stance ?= flag:zunwang }
			}
			AND = {
				scope:target = { participant_group_type = pro_hegemon_movement }
				exists = var:hd_party_waiqi_leader
				NOT = { has_variable = hd_party_waiqi_loyalty_cd }
				NOT = { var:hd_party_waiqi_stance ?= flag:zunwang }
			}
			AND = {
				scope:target = { participant_group_type = expansion_movement }
				exists = var:hd_party_wuxun_leader
				NOT = { has_variable = hd_party_wuxun_loyalty_cd }
				NOT = { var:hd_party_wuxun_stance ?= flag:zunwang }
			}
			AND = {
				scope:target = { participant_group_type = conservative_movement }
				exists = var:hd_party_shizu_leader
				NOT = { has_variable = hd_party_shizu_loyalty_cd }
				NOT = { var:hd_party_shizu_stance ?= flag:zunwang }
			}
			AND = {
				scope:target = { participant_group_type = advancement_movement }
				exists = var:hd_party_hanmen_leader
				NOT = { has_variable = hd_party_hanmen_loyalty_cd }
				NOT = { var:hd_party_hanmen_stance ?= flag:zunwang }
			}
			}
		}
	}
	effect = {
		scope:target = { save_scope_as = hd_party_target_group }
		trigger_event = hd_party.0020
	}
}
