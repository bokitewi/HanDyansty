# Root is always CouncilWindow.GetCharacter, the displayed court owner.
# Read-only UI gates; appointment and candidate validation remain native.
WJ_inner_shizhong_has_holder = {
 scope = character
 is_shown = { employs_court_position = favored_minister_court_position }
}
WJ_inner_shizhong_can_open_appointment = {
 scope = character
 is_shown = {
  is_alive = yes
  highest_held_title_tier = tier_hegemony
  government_has_flag = government_is_celestial
  NOT = { employs_court_position = favored_minister_court_position }
 }
}

# Open the engine-owned appointment model directly, without the court-page GUI command.
# Candidate eligibility, replacement and confirmation remain native.
WJ_inner_shizhong_open_native = {
 scope = character
 is_shown = { is_alive = yes this = scope:actor }
 is_valid = {
  is_alive = yes
  this = scope:actor
  highest_held_title_tier = tier_hegemony
  government_has_flag = government_is_celestial
 }
 effect = { open_appoint_court_position_window = favored_minister_court_position }
}
