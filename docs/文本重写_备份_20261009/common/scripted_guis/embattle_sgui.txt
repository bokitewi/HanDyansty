# Preserve all public names used by existing army GUI files.
embattle_valid_commander_detection = {
    scope = army
    effect = {
        hd_embattle_normalize_army_effect = yes
        hd_embattle_refresh_army_bonus_effect = yes
        # Opening/closing a selection window must not revoke legal automatic deputies.
        clear_variable_list = armytest
        remove_variable = hd_embattle_gui_candidate_count
    }
}
embattle_set_army_commander_list = {
    scope = army
    effect = {
        save_scope_as = hd_embattle_army
        hd_embattle_normalize_army_effect = yes
        clear_variable_list = armytest
        set_variable = { name = hd_embattle_gui_candidate_count value = 0 }
        if = {
            limit = { exists = army_owner }
            army_owner = {
                every_knight = {
                    limit = {
                        is_alive = yes
                        exists = knight_army
                        knight_army = scope:hd_embattle_army
                        is_commanding_army = no
                    }
                    scope:hd_embattle_army = {
                        add_to_variable_list = { name = armytest target = prev }
                        change_variable = { name = hd_embattle_gui_candidate_count add = 1 }
                    }
                }
            }
        }
    }
}

embattle_set_army_left_flank_commander = {
    scope = army
    is_valid = { hd_embattle_manual_allowed_trigger = yes }
    effect = {
        save_scope_as = hd_embattle_army
        if = {
            limit = {
                hd_embattle_manual_allowed_trigger = yes
                exists = scope:knight
                scope:knight = { hd_embattle_candidate_trigger = yes }
            }
            set_variable = { name = embattle_left_flank_commander value = scope:knight }
            remove_variable = hd_embattle_manual_empty_left
            hd_embattle_normalize_army_effect = yes
            hd_embattle_refresh_army_bonus_effect = yes
        }
    }
}
embattle_is_invalid_left_flank_commander_trigger = {
    scope = army
    is_valid = {
        has_game_rule = embattle_side_commander_appointor_ai_only_except_own_troop
        exists = army_owner
        exists = army_commander
        NOT = { army_commander = army_owner }
    }
    is_shown = {
        has_variable = embattle_left_flank_commander
        hd_embattle_left_valid_trigger = no
    }
}
embattle_has_left_flank_commander_trigger = {
    scope = army
    is_shown = { has_variable = embattle_left_flank_commander }
}
embattle_clear_left_flank_commander = {
    scope = army
    is_valid = { hd_embattle_manual_allowed_trigger = yes }
    effect = {
        if = {
            limit = { hd_embattle_manual_allowed_trigger = yes }
            remove_variable = embattle_left_flank_commander
            set_variable = { name = hd_embattle_manual_empty_left value = yes }
            set_variable = { name = hd_embattle_auto_done value = yes }
            remove_variable = hd_embattle_complete
            hd_embattle_refresh_army_bonus_effect = yes
        }
    }
}

embattle_is_not_commanding_trigger = {
    scope = army
    is_shown = {
        OR = {
            AND = { exists = army_commander army_commander = scope:knight }
            AND = {
                exists = var:embattle_left_flank_commander
                var:embattle_left_flank_commander = scope:knight
            }
            AND = {
                exists = var:embattle_right_flank_commander
                var:embattle_right_flank_commander = scope:knight
            }
            NOT = { hd_embattle_manual_allowed_trigger = yes }
        }
    }
}
embattle_window_combat_sgui = {
    scope = character
    effect = {
        if = {
            limit = { exists = commanding_army }
            set_variable = { name = embattle_army value = commanding_army }
        }
        else = { remove_variable = embattle_army }
        if = {
            limit = { exists = scope:root2 }
            scope:root2 = {
                if = {
                    limit = { exists = commanding_army }
                    set_variable = { name = embattle_army value = commanding_army }
                }
                else = { remove_variable = embattle_army }
            }
        }
    }
}
embattle_is_trait = {
    scope = trait
    is_shown = { has_trait_category = commander }
}
