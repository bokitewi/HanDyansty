# 原版 important action 的决议入口写法：满足条件时在顶部提示，点击打开决议详情。
action_wj_temp_official_kaoke_ready = {
	type = alert
	icon = action_contract

	check_create_action = {
		if = {
			limit = {
				has_character_flag = wj_temp_official_kaoke_ready
				can_execute_decision = wj_temp_official_kaoke_decision
			}
			try_create_important_action = {
				important_action_type = action_wj_temp_official_kaoke_ready
				actor = root
			}
		}
	}

	effect = {
		decision:wj_temp_official_kaoke_decision = { open_view_data = decision_detail }
	}
}
