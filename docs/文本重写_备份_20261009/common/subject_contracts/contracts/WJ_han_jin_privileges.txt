# Native contract state only. The gameplay behind these new manual grants is reserved.
WJ_sanjie = {
	display_mode = radiobutton
	is_shown = { scope:subject = { WJ_contract_local_official_trigger = yes } }
	can_be_changed = {
		custom_tooltip = {
			text = WJ_privilege_sanjie_eligible_tt
			scope:subject = {
				OR = {
					WJ_contract_sanjie_eligible_trigger = yes
					vassal_contract_obligation_level:WJ_sanjie > 0
				}
			}
		}
	}
	obligation_levels = {
		# Default zero has no is_valid (engine requirement). It has no visible GUI action.
		# District creation must require a selected jie before committing the district.
		WJ_sanjie_none = {
			default = yes
			position = { 0 0 }
			ai_liege_desire = 0
			ai_subject_desire = 0
			score = 0
		}
		WJ_jiajie = {
			parent = WJ_sanjie_none
			position = { 1 0 }
			is_valid = { scope:subject = { WJ_contract_sanjie_eligible_trigger = yes } }
			flag = WJ_jiajie
			subject_modifier = { maa_damage_mult = 0.025 levy_reinforcement_rate = 0.025 }
			ai_liege_desire = 0
			ai_subject_desire = 0
			score = 1
		}
		WJ_chijie = {
			parent = WJ_sanjie_none
			position = { 2 0 }
			is_valid = { scope:subject = { WJ_contract_sanjie_eligible_trigger = yes } }
