# Boolean UI query only: preserve the original realm scan when no owned capital proves it true.
# The capital may be held by a vassal; a borrowed foreign capital must not qualify.
WJ_education_realm_has_university = {
 trigger_if = {
  limit = {
   exists = capital_province.province_owner
   capital_province.province_owner = { OR = { this = prev is_vassal_or_below_of = prev } }
   capital_province = { is_higher_studies_location = yes }
  }
  always = yes
 }
 trigger_else = {
  OR = {
   any_realm_province = { is_higher_studies_location = yes }
   any_realm_county = { has_county_modifier = pandidakterion_university_modifier }
  }
 }
}
