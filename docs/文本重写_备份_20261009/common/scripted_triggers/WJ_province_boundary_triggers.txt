# A.30: industrial and metropolitan are closed in Han/Jin; preserve petitioner/recipient realm checks.
WJ_blocked_province_petition_trigger = {
	OR = {
		AND = {
			exists = scope:hegemon
			scope:hegemon ?= { government_has_flag = government_is_celestial }
		}
		AND = {
			exists = scope:province_change_recipient
			scope:province_change_recipient ?= { government_has_flag = government_is_celestial }
		}
	}
	OR = {
		$PETITION$ ?= flag:change_province_industrial
		$PETITION$ ?= flag:change_province_metropolitan
		$PETITION$ ?= flag:change_province_protectorate
		$PETITION$ ?= flag:hold_examinations
	}
}
