# 天朝循环·时代特色：对峙期套件判定（见 docs/天朝循环_时代特色重制计划.md §3.6）

# Scope：角色。处于对峙期，且本人位于天下（h_china 法理）之内
hd_dc_kit_sta_tools_trigger = {
	hd_dc_phase_is_trigger = { PHASE = hd_standoff }
	hd_dc_in_china_trigger = yes
}

# Scope：角色（策反对象）。敌国封臣：有封地、非独立、位于天下之内，其最高领主不是 $OWNER$
hd_dc_sta_subversion_target_trigger = {
	is_ruler = yes
	is_landed = yes
	is_independent_ruler = no
	exists = liege
	highest_held_title_tier >= tier_county
	hd_dc_in_china_trigger = yes
	NOT = { top_liege = $OWNER$ }
	NOT = { primary_title ?= { is_mercenary_company = yes } }
}

# Scope：角色。$OWNER$ 是否已在策划任一策反阴谋（每人同时只许一项）
hd_dc_sta_has_subversion_scheme_trigger = {
	any_scheme = {
		OR = {
			scheme_type = hd_dc_sta_incite_independence
			scheme_type = hd_dc_sta_incite_defection
			scheme_type = hd_dc_sta_incite_faction
		}
	}
}

# Scope：阴谋。是否为本套件的策反阴谋
hd_dc_sta_is_subversion_scheme_trigger = {
	OR = {
		scheme_type = hd_dc_sta_incite_independence
		scheme_type = hd_dc_sta_incite_defection
		scheme_type = hd_dc_sta_incite_faction
	}
}

# Scope：战争。$ACTOR$ 与 $ENEMY$ 是对阵双方的首要首领（非内战）——割地求和用
hd_dc_sta_cede_land_war_trigger = {
	is_civil_war = no
	OR = {
		AND = {
			primary_attacker = $ACTOR$
			primary_defender = $ENEMY$
		}
		AND = {
			primary_defender = $ACTOR$
			primary_attacker = $ENEMY$
		}
	}
}

# Scope：角色。$WHO$ 在与 $ENEMY$ 首领对阵的战争中，自身一方战分 ≥ $SCORE$（可为负）
hd_dc_sta_war_score_at_least_trigger = {
	any_character_war = {
		hd_dc_sta_cede_land_war_trigger = {
			ACTOR = $WHO$
			ENEMY = $ENEMY$
		}
		OR = {
			AND = {
				primary_attacker = $WHO$
				attacker_war_score >= $SCORE$
			}
			AND = {
				primary_defender = $WHO$
				defender_war_score >= $SCORE$
			}
		}
	}
}

# Scope：角色。$WHO$ 在与 $ENEMY$ 首领对阵的战争中，自身一方战分 ≤ $SCORE$（可为负）
hd_dc_sta_war_score_at_most_trigger = {
	any_character_war = {
		hd_dc_sta_cede_land_war_trigger = {
			ACTOR = $WHO$
			ENEMY = $ENEMY$
		}
		OR = {
			AND = {
				primary_attacker = $WHO$
				attacker_war_score <= $SCORE$
			}
			AND = {
				primary_defender = $WHO$
				defender_war_score <= $SCORE$
			}
		}
	}
}

# Scope：角色。互市可签对象：接壤的天下内独立统治者
hd_dc_sta_border_market_partner_trigger = {
	is_independent_ruler = yes
	is_landed = yes
	hd_dc_in_china_trigger = yes
	character_is_land_realm_neighbor = $PARTNER$
}
