# Native range: council's court/vassals narrowed by approved direct king+ office rule.
WJ_inner_emperor_trigger = {
 is_alive = yes
 government_has_flag = government_is_celestial
 top_liege = this
 highest_held_title_tier >= tier_empire
}
WJ_dasima_candidate_trigger = {
 is_alive = yes
 highest_held_title_tier >= tier_kingdom
 liege = scope:wj_inner_emperor
 tgp_can_be_a_minister_trigger = { COURT_OWNER = scope:wj_inner_emperor }
 NOT = { this = scope:wj_inner_emperor }
}
# Holding the honor does not depend on temporary appointment availability.
WJ_dasima_retention_trigger = {
 is_alive = yes
 highest_held_title_tier >= tier_kingdom
 exists = liege
 liege = top_liege
 government_has_flag = government_is_celestial
 top_liege = {
  WJ_inner_emperor_trigger = yes
 }
 # The honor belongs to a person, not whichever title that person makes primary.
 exists = var:wj_dasima_emperor
 OR = {
  var:wj_dasima_emperor ?= top_liege
  AND = {
   exists = var:wj_dasima_emperor
   var:wj_dasima_emperor ?= { is_alive = no }
   exists = var:wj_dasima_realm.holder
   var:wj_dasima_realm.holder ?= top_liege
  }
 }
 # Never overwrite another valid personal appointment during succession.
 save_temporary_scope_as = wj_retained_dasima
 top_liege = {
  trigger_if = {
   limit = { exists = var:wj_dasima_holder }
   var:wj_dasima_holder = scope:wj_retained_dasima
  }
 }
}
# Display-only filters never change appointment eligibility.
WJ_dasima_candidate_filter_trigger = {
 trigger_if = {
  limit = { scope:wj_inner_emperor = { has_variable = wj_dasima_filter_faith var:wj_dasima_filter_faith = 1 } }
  faith = scope:wj_inner_emperor.faith
 }
 trigger_if = {
  limit = { scope:wj_inner_emperor = { has_variable = wj_dasima_filter_faith var:wj_dasima_filter_faith = 2 } }
  NOT = { faith = scope:wj_inner_emperor.faith }
 }
 trigger_if = {
  limit = { scope:wj_inner_emperor = { has_variable = wj_dasima_filter_culture var:wj_dasima_filter_culture = 1 } }
  culture = scope:wj_inner_emperor.culture
 }
 trigger_if = {
  limit = { scope:wj_inner_emperor = { has_variable = wj_dasima_filter_culture var:wj_dasima_filter_culture = 2 } }
  NOT = { culture = scope:wj_inner_emperor.culture }
 }
}
