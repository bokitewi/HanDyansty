# 天朝循环大修：崩溃期危机判定。
# 借鉴 All Under Heaven Crisis Overhaul（已获授权）的"先邀请、次日结算"派系模式。

# Scope：皇帝的直属封臣。集体独立的合格成员（遵守原版派系合法性检查）
hd_collapse_independence_vassal_eligible_trigger = {
	is_landed = yes
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	exists = liege
	primary_title.tier >= tier_county
	opinion = { target = liege value < 0 }
	NOT = { is_allied_to = liege }
	OR = {
		NOT = { exists = joined_faction }
		joined_faction = {
			faction_is_type = independence_faction
			faction_is_at_war = no
		}
	}
	can_join_or_create_faction_against = {
		who = liege
		faction = independence_faction
		check_in_a_faction = no
	}
}

# Scope：皇帝。可发动集体独立危机
hd_collapse_independence_target_valid_trigger = {
	is_alive = yes
	has_title = title:h_china
	NOT = { any_character_war = { using_cb = independence_faction_war } }
	NOT = { has_character_flag = hd_collapse_independence_pending }
	any_vassal = {
		count >= 2
		hd_collapse_independence_vassal_eligible_trigger = yes
	}
}

# Scope：皇帝的直属封臣。军阀称帝叛乱的拥立对象
hd_collapse_warlord_candidate_trigger = {
	is_landed = yes
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	primary_title.tier >= tier_duchy
	NOT = { dynasty ?= liege.dynasty }
	opinion = { target = liege value < 0 }
}

# Scope：皇帝的直属封臣。军阀称帝叛乱的响应者
hd_collapse_warlord_member_trigger = {
	is_landed = yes
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	opinion = { target = liege value < 0 }
	NOT = { is_allied_to = liege }
	OR = {
		NOT = { exists = joined_faction }
		joined_faction = { faction_is_type = hd_collapse_warlord_faction }
	}
}

# Scope：皇帝。可发动军阀称帝叛乱
hd_collapse_warlord_target_valid_trigger = {
	is_alive = yes
	has_title = title:h_china
	NOT = { any_character_war = { using_cb = hd_collapse_warlord_war } }
	NOT = { has_character_flag = hd_collapse_warlord_pending }
	any_vassal = { hd_collapse_warlord_candidate_trigger = yes }
	any_vassal = {
		count >= 2
		hd_collapse_warlord_member_trigger = yes
	}
}

# Scope：皇帝。可发动大规模民变
hd_collapse_peasant_target_valid_trigger = {
	is_alive = yes
	has_title = title:h_china
	NOT = { any_character_war = { using_cb = peasant_war } }
	any_realm_county = { county_opinion <= -25 }
}

# Scope：war。崩溃期危机类战争（皇帝为防守方）
hd_collapse_crisis_war_trigger = {
	OR = {
		hd_cycle_faction_war_trigger = yes
		using_cb = hd_collapse_warlord_war
	}
	exists = title:h_china.holder
	primary_defender = title:h_china.holder
}

# Scope：皇帝的直属封臣。文档 1.0：5 年 15% 强制建派系的高风险封臣
hd_collapse_faction_risk_vassal_trigger = {
	is_landed = yes
	is_adult = yes
	is_imprisoned = no
	NOT = { exists = joined_faction }
	OR = {
		has_trait = ambitious
		has_trait = zealous
		has_trait = cynical
		has_trait = greedy
		has_trait = deceitful
	}
	NOR = {
		has_trait = content
		has_trait = lazy
		has_trait = craven
		has_trait = loyal
	}
	OR = {
		opinion = { target = liege value < 0 }
		has_relation_nemesis = liege
		has_relation_rival = liege
	}
	# 不畏惧、不忠顺于上级领主
	NOT = {
		has_dread_level_towards = {
			target = liege
			level = 1
		}
	}
	NOT = { is_obedient_to = liege }
}
