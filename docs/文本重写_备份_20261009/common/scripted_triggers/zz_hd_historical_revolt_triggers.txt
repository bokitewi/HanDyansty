# Monthly-readiness triggers for the nine historical regional revolts (phase 1).
#
# A revolt fires when: the Yellow Turban uprising has begun, its global fired flag is absent,
# the current date is inside the fixed window, and at least one candidate duchy has a living
# holder. Nothing else (control, opinion, culture, ruler personality, ongoing wars, player
# status) may block it. Late starts beyond the window never back-fill.

# The Yellow Turban uprising has either fired on this map or started before game start.
hd_historical_revolt_yellow_turban_started_trigger = {
	OR = {
		character:zhang_jiao = { has_character_flag = hd_yellow_turban_rebellion_started }
		game_start_date >= 184.3.1
	}
}

# REV-01 184.6 Jiaozhi garrison revolt, window 184.6-186.12
hd_historical_revolt_jiaozhi_ready_trigger = {
	hd_historical_revolt_yellow_turban_started_trigger = yes
	NOT = { exists = global_var:hd_revolt_jiaozhi_fired }
	NOT = { exists = global_var:hd_revolt_jiaozhi_committed }
	current_date >= 184.6.1
	current_date <= 186.12.31
	exists = title:d_jiaozhi.holder
	title:d_jiaozhi.holder = { is_alive = yes }
}
