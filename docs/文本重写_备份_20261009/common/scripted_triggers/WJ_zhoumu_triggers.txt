# The contract is the only source of appointment state. The pointer is only opinion cleanup.
WJ_zhoumu_structural_eligible = {
 WJ_contract_local_official_trigger = yes
 highest_held_title_tier = tier_kingdom
 primary_title = { tier = tier_kingdom is_noble_family_title = no tgp_is_minister_title = no }
 top_liege = { WJ_inner_emperor_trigger = yes }
 exists = top_liege.capital_county.kingdom
 NOT = { has_title = top_liege.capital_county.kingdom }
}
WJ_zhoumu_eligible = {
 WJ_zhoumu_structural_eligible = yes
 vassal_contract_has_flag = celestial_province_military
}
WJ_zhoumu_active = {
 vassal_contract_has_flag = WJ_zhoumu_granted
 WJ_zhoumu_eligible = yes
}
WJ_zhoumu_has_state = {
 OR = { vassal_contract_obligation_level:WJ_zhoumu > 0 has_variable = wj_zhoumu_opinion_emperor }
}
# changed_obligations contains native obligation objects, not integer levels.
WJ_zhoumu_pending_grant = {
 any_in_list = { list = changed_obligations this = vassal_contract:WJ_zhoumu.obligation:WJ_zhoumu_grant }
}
WJ_zhoumu_batch_allowed = {
 trigger_if = {
  limit = { WJ_zhoumu_pending_grant = yes }
  $SUBJECT$ = { WJ_zhoumu_structural_eligible = yes top_liege = $RULER$ }
  OR = {
   $SUBJECT$ = { vassal_contract_has_flag = celestial_province_military }
   any_in_list = { list = changed_obligations this = vassal_contract:celestial_provinces.obligation:celestial_province_military }
  }
  # Reject a simultaneous switch away from military, regardless of list order.
  NOT = {
   any_in_list = {
    list = changed_obligations
    vassal_contract_type = vassal_contract:celestial_provinces
    NOT = { this = vassal_contract:celestial_provinces.obligation:celestial_province_military }
   }
  }
 }
}
WJ_zhoumu_request_hook = {
 trigger_if = {
  limit = { WJ_zhoumu_pending_grant = yes }
  always = scope:hook
  scope:actor = { has_usable_hook = scope:recipient }
 }
}
