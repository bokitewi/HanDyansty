# 天朝循环·时代特色：斗争期套件判定（见 docs/天朝循环_时代特色重制计划.md §3.5）
# 原版 FP2 工具以“卷入斗争”为门槛；本 mod 的天朝循环是局势而非斗争，故以“当前阶段 + 位于天下”替代“involvement = involved”。

# Scope：角色。处于斗争期，且本人位于天下（h_china 法理）之内
hd_dc_kit_cha_tools_trigger = {
	hd_dc_phase_is_trigger = { PHASE = chaos }
	hd_dc_in_china_trigger = yes
}

# Scope：角色。处于斗争期或对峙期，且本人位于天下之内（请求入侵等两期共用工具）
hd_dc_kit_cha_or_sta_tools_trigger = {
	OR = {
		hd_dc_phase_is_trigger = { PHASE = chaos }
		hd_dc_phase_is_trigger = { PHASE = hd_standoff }
	}
	hd_dc_in_china_trigger = yes
}

# Scope：头衔。可被“取得宣称”的头衔（郡/州；不含天下与帝国级）
hd_dc_cha_buy_claim_title_trigger = {
	OR = {
		tier = tier_duchy
		tier = tier_kingdom
	}
	is_landless_type_title = no
	is_head_of_faith = no
	is_noble_family_title = no
	NOT = { has_variable = adventurer_creation_reason }
}

# Scope：角色（被拉拢的封臣）。可作为“招纳叛将”的对象
hd_dc_cha_lure_target_trigger = {
	is_ruler = yes
	is_landed = yes
	is_independent_ruler = no
	exists = liege
	highest_held_title_tier >= tier_county
	hd_dc_in_china_trigger = yes
	NOT = { primary_title ?= { is_mercenary_company = yes } }
}
