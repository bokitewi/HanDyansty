# 天朝循环大修·朋党：身份（参与者组）判定。
# 组的归属顺序由局势文件 participant_groups 的排列决定：
#   天子 > 四夷 > 诸侯 > 宦官 > 外戚 > 武勋 > 世族 > 寒门（兜底）
# 简单条件放前，遍历放后。

# Scope：character。首都在 h_china 法理范围内
hd_party_capital_in_china_trigger = {
	capital_county ?= { target_is_de_jure_liege_or_above = title:h_china }
}

# Scope：character。是拥有朋党玩法的"朝廷"君主
# 治世：天子；乱世：首都在汉地、帝国级、非四夷的独立统治者
hd_party_court_ruler_trigger = {
	is_independent_ruler = yes
	highest_held_title_tier >= tier_empire
	trigger_if = {
		limit = { has_title = title:h_china }
		always = yes
	}
	trigger_else = {
		hd_cycle_is_luanshi_trigger = yes
		hd_cycle_non_han_trigger = no
		hd_party_capital_in_china_trigger = yes
	}
}

# Scope：character。所在政权可分配身份组（自身不是独立统治者）
# 治世：天子政权内（或带天朝授封的朝贡国）；乱世：首都在汉地的任意独立政权内
hd_party_identity_realm_trigger = {
	NOT = { has_title = title:h_china }
	OR = {
		AND = {
			exists = title:h_china.holder
			top_liege ?= title:h_china.holder
		}
		AND = {
			hd_cycle_is_zhishi_trigger = yes
			exists = title:h_china.holder
			is_tributary_of = title:h_china.holder
			subject_contract_has_flag = celestial_investiture_politics
		}
		AND = {
			hd_cycle_is_luanshi_trigger = yes
			is_independent_ruler = no
			top_liege ?= { hd_party_capital_in_china_trigger = yes }
		}
	}
}

# Scope：character。宦官
hd_party_is_huanguan_trigger = {
	is_eunuch_trigger = yes
}

# Scope：character。外戚：成年，宗族属于所在朝廷君主的后族
# 后族宗族列表 hd_party_waiqi_dynasties 缓存在君主身上，由 hd_party_refresh_waiqi_list_effect 维护
hd_party_is_waiqi_trigger = {
	is_adult = yes
	exists = dynasty
	OR = {
		AND = {
			exists = title:h_china.holder
			OR = {
				top_liege = title:h_china.holder
				is_tributary_of = title:h_china.holder
			}
			title:h_china.holder = {
				is_target_in_variable_list = { name = hd_party_waiqi_dynasties target = prev.dynasty }
			}
		}
		top_liege ?= {
			is_target_in_variable_list = { name = hd_party_waiqi_dynasties target = prev.dynasty }
		}
	}
}

# Scope：character。武勋：军事人员、军事教育 3 级以上、持武官军衔
hd_party_is_wuxun_trigger = {
	OR = {
		has_trait = education_martial_3
		has_trait = education_martial_4
		has_trait = education_martial_5
		WJ_mt_military_person = yes
		WJ_mr_has_rank = yes
	}
}

# Scope：character。世族：宗族威望 ≥7、豪族/主导家族、本人或家主持有世族头衔
hd_party_is_shizu_trigger = {
	OR = {
		dynasty ?= { dynasty_prestige_level >= 7 }
		house ?= {
			OR = {
				is_powerful_family = yes
				is_dominant_family = yes
			}
		}
		any_held_title = { is_noble_family_title = yes }
		house ?= {
			house_head ?= {
				any_held_title = { is_noble_family_title = yes }
			}
		}
	}
}

# Scope：character。四夷
hd_party_is_siyi_trigger = {
	hd_cycle_non_han_trigger = yes
}

# 184 年剧本开局（汉灵帝在位）
hd_party_is_184_start_trigger = {
	current_date >= 184.1.1
	current_date < 184.2.1
	exists = title:h_china.holder
	title:h_china.holder = character:liu_hong_3
}
