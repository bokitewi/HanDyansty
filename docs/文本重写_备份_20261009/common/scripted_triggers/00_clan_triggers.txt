### CLAN TRIGGERS ### (modded: also true for governments with House Unity, e.g. celestial)
is_in_clan_trigger = {
	exists = house
	dm_compat_uses_house_unity_trigger = yes
	top_liege = house.house_head
}

is_in_same_clan_as_trigger = {
	exists = house
	exists = $CHARACTER$.house
	OR = {
		dm_compat_uses_house_unity_trigger = yes
		$CHARACTER$ = { dm_compat_uses_house_unity_trigger = yes }
	}
	house = $CHARACTER$.house
	top_liege = $CHARACTER$.top_liege
}
