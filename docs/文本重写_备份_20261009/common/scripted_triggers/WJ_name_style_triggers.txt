# 汉式称谓适用人群: 汉文化(主mod 13个汉州文化实测均为 heritage_chinese)
WJ_han_name_style_trigger = {
	culture = { has_cultural_pillar = heritage_chinese }
}

# 皇室亲属判定(皇子/皇女/太子/太女/公子/公女/长公主等)
WJ_royal_relation_trigger = {
	OR = {
		AND = {
			exists = father
			father = { primary_title.tier >= tier_kingdom }
		}
		AND = {
			exists = mother
			mother = { primary_title.tier >= tier_kingdom }
		}
		AND = {
			exists = top_liege
			is_primary_heir_of = top_liege
		}
		AND = {
			is_female = yes
			any_sibling = { primary_title.tier >= tier_empire }
		}
	}
}
