# 天朝循环·时代特色：扩张期套件判定（见 docs/天朝循环_时代特色重制计划.md §3.1）

# Scope：角色。天子且处于扩张期
hd_dc_exp_emperor_trigger = {
	has_title = title:h_china
	hd_dc_phase_is_trigger = { PHASE = stability_expansion }
}

# Scope：县。边县：与 $RULER$ 疆域之外的县相邻
hd_dc_exp_border_county_trigger = {
	any_neighboring_county = {
		NOT = { holder.top_liege ?= $RULER$ }
	}
}

# Scope：角色（天子）。疆域内至少有一个边县（屯田可用）
hd_dc_exp_has_border_county_trigger = {
	any_sub_realm_county = {
		exists = kingdom
		hd_dc_exp_border_county_trigger = { RULER = $RULER$ }
	}
}

# Scope：角色。可授军功的将领：在世、属于 $RULER$ 疆域、非天子本人、未被囚
hd_dc_exp_merit_candidate_trigger = {
	is_alive = yes
	top_liege ?= $RULER$
	NOT = { this = $RULER$ }
	is_imprisoned = no
}
