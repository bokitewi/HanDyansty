# 天朝循环·时代特色：崩溃期套件判定（见 docs/天朝循环_时代特色重制计划.md §3.4）

# Scope：war。招安适用的叛乱战争：派系 / 民变战争，或危机"军阀称帝"战争。
# 不用 hd_collapse_crisis_war_trigger：它要求防守方必须是天子，而招安对任一主君开放。
hd_dc_col_rebel_war_trigger = {
	OR = {
		hd_cycle_faction_war_trigger = yes
		using_cb = hd_collapse_warlord_war
	}
}

# Scope：伯爵领（county title）。受灾县：带原版自然灾害（地震 / 洪水及其余波）或本 mod 饥荒的负面县修正。
# 修正清单照原版 natural_disaster_remove_realm_effect（events/situation_events/tgp_natural_disaster_events.txt）
hd_dc_col_county_has_disaster_trigger = {
	OR = {
		has_county_modifier = tgp_natural_disaster_earthquake_1_county_modifier
		has_county_modifier = tgp_natural_disaster_earthquake_2_county_modifier
		has_county_modifier = tgp_natural_disaster_earthquake_3_county_modifier
		has_county_modifier = tgp_natural_disaster_flood_1_county_modifier
		has_county_modifier = tgp_natural_disaster_flood_2_county_modifier
		has_county_modifier = tgp_natural_disaster_flood_3_county_modifier
		has_county_modifier = tgp_natural_disaster_fortification_county_modifier
		has_county_modifier = tgp_natural_disaster_exodus_1_county_modifier
		has_county_modifier = tgp_natural_disaster_exodus_2_county_modifier
		has_county_modifier = tgp_natural_disaster_exodus_3_county_modifier
		has_county_modifier = tgp_natural_disaster_fertility_1_county_modifier
		has_county_modifier = tgp_natural_disaster_fertility_2_county_modifier
		has_county_modifier = tgp_natural_disaster_fertility_3_county_modifier
		has_county_modifier = hd_famine_county_modifier
	}
}

# Scope：角色。其领内（含封臣）有受灾县
hd_dc_col_realm_has_disaster_trigger = {
	any_sub_realm_county = { hd_dc_col_county_has_disaster_trigger = yes }
}

# Scope：角色（主君）。正作为主防守方，被 $RECIPIENT$ 以叛乱战争进攻
hd_dc_col_is_rebel_war_target_trigger = {
	any_character_war = {
		primary_defender = prev
		primary_attacker = $RECIPIENT$
		hd_dc_col_rebel_war_trigger = yes
	}
}

# Scope：角色（天子）。有一场尚未颁布过勤王诏的防御战（天子为主防守方）
hd_dc_col_has_qinwang_war_trigger = {
	any_character_war = {
		primary_defender = prev
		NOT = { has_variable = hd_dc_col_qinwang }
	}
}

# Scope：角色（候选封臣）。可响应勤王：需先存 scope:target = 战争、scope:hd_dc_col_qinwang_emperor = 天子
hd_dc_col_qinwang_candidate_trigger = {
	is_landed = yes
	is_adult = yes
	is_imprisoned = no
	highest_held_title_tier >= tier_county
	opinion = {
		target = scope:hd_dc_col_qinwang_emperor
		value >= 0
	}
	NOR = {
		is_attacker_in_war = scope:target
		is_defender_in_war = scope:target
		is_at_war_with = scope:hd_dc_col_qinwang_emperor
		any_vassal_or_below = { is_attacker_in_war = scope:target }
		any_liege_or_above = { is_attacker_in_war = scope:target }
	}
}
