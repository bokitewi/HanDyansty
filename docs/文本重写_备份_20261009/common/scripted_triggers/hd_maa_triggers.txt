# --- 地域兵源：至少一处直辖县、控制力 ≥ 50，且当地文化传承匹配 ---
# 兵源只看角色直接持有的县；封臣的县不会为上级领主提供兵源。
hd_valid_youzhou_tuqi_source_trigger = {
	any_held_county = {
		title_province = { geographical_region = hd_region_youzhou_tuqi }
		county_control >= 50
		culture = { has_cultural_pillar = heritage_chinese }
	}
}

hd_has_established_youzhou_tuqi_trigger = {
	any_held_title = { has_variable = hd_established_youzhou_tuqi }
}

hd_can_recruit_youzhou_tuqi_trigger = {
	hd_has_established_youzhou_tuqi_trigger = yes
	hd_valid_youzhou_tuqi_source_trigger = yes
}

# --- 刷新守卫：是否已建立任一地域军制 ---
# 军制标记存于建立时的头衔；兵源旗标仍是当前持有者的角色缓存。

		hd_has_established_shanyue_jiashi_trigger = yes
		hd_has_established_danyang_jingbing_trigger = yes
		hd_has_established_fuling_liannushi_trigger = yes
		hd_has_established_qiang_qi_trigger = yes
		hd_has_established_bingzhou_bianqi_trigger = yes

	}
}

	OR = {

	}

# HD 其余7种精锐兵：训练者触发器（骑士scope，ROOT为主君）
# 南蛮象兵的南方/南中军事背景暂定为：主君领域含宁州(南中)县（在决议is_shown中检查）

hd_hubaoqi_trainer_trigger = {
	is_knight_of = root
	is_alive = yes
		martial >= 18
		prowess >= 18
		OR = {
			has_trait = organizer
			has_trait = aggressive_attacker
		}
		NOT = {
			has_trait = craven
		}
}

hd_baier_trainer_trigger = {
	is_knight_of = root
	is_alive = yes
		martial >= 16
		prowess >= 18
		OR = {
			has_trait = brave
			has_trait = unyielding_defender
		}
}

hd_jiefanbing_trainer_trigger = {
	is_knight_of = root
	is_alive = yes
		martial >= 16
		prowess >= 15
		OR = {
			has_trait = flexible_leader
			has_trait = organizer
		}
}

hd_quyi_xiandeng_trainer_trigger = {

hd_suwei_hushi_trainer_trigger = {
	is_knight_of = root
	is_alive = yes
		martial >= 12
		prowess >= 22

			value >= 80
		}
