### Repaired Confederations & Leagues Expanded triggers ###

hd_uses_defensive_leagues = {
	NOR = {
		government_has_flag = government_is_nomadic
		government_has_flag = government_is_herder
	}
}

hd_valid_defensive_league_member_trigger = {
	$CHARACTER$ = {
		is_alive = yes
		is_landed = yes
		is_imprisoned = no
		is_independent_ruler = yes
		highest_held_title_tier <= tier_duchy
		hd_uses_defensive_leagues = yes
	}
}

hd_valid_anti_regent_member_trigger = {
	$CHARACTER$ = {
		# Admission restrictions do not expel an existing member after promotion or capture.
		is_alive = yes
		is_landed = yes
		is_independent_ruler = yes
	}
}

# Rank filter for invitations/responses only. Jun = duchy; kingdoms remain eligible.
# Do not reuse this to validate existing members or select the historical core.
hd_anti_regent_invitation_rank_allowed_trigger = {
	is_landed = yes
	OR = {
		highest_held_title_tier = tier_duchy
		highest_held_title_tier = tier_kingdom
	}
}

# Shared eligibility before independence and after joining.
hd_anti_regent_invitee_eligible_trigger = {
	is_playable_character = yes
	is_tributary = no
	is_alive = yes
	is_landed = yes
	is_imprisoned = no
	hd_anti_regent_invitation_rank_allowed_trigger = yes
	NOT = { has_character_flag = hd_anti_regent_league_excluded }
}

# Invitation-only restriction; existing confederation membership is unaffected.
hd_anti_regent_invitation_liege_allowed_trigger = {
	trigger_if = {
		limit = { exists = title:k_uuii_sili.holder }
		NOT = { is_vassal_or_below_of = title:k_uuii_sili.holder }
	}
	trigger_if = {
		limit = { exists = title:k_uuii_bingzhou.holder }
		NOT = { is_vassal_or_below_of = title:k_uuii_bingzhou.holder }
	}
	trigger_if = {
		limit = { exists = title:k_uuii_liangzhou.holder }
		NOT = { is_vassal_or_below_of = title:k_uuii_liangzhou.holder }
	}
}

hd_anti_regent_successor_eligible_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = yes
	is_independent_ruler = yes
	is_imprisoned = no
	NOT = { has_character_flag = hd_anti_regent_leader_dying }
	is_member_of_confederation = global_var:hd_anti_regent_active_confederation
}

hd_anti_regent_can_restore_liege_trigger = {
	exists = var:hd_anti_regent_original_liege
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	var:hd_anti_regent_original_liege = {
		is_alive = yes
		is_landed = yes
		highest_held_title_tier > prev.highest_held_title_tier
		NOT = { is_vassal_or_below_of = prev }
	}
}

hd_anti_regent_core_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	NOT = { this = character:dong_zhuo_2 }
	NOT = { has_title = title:e_han }
	NOT = { has_title = title:h_china }
	OR = {
		this = character:yuan_shao
		this = character:cao_cao
		AND = {
			is_landed = yes
			highest_held_title_tier >= tier_county
			highest_held_title_tier < tier_empire
		}
	}
}

hd_is_forced_anti_regent_member_trigger = {

		this = character:han_fu
		this = character:kong_zhou
		this = character:liu_dai
		this = character:wang_kuang
		this = character:zhang_miao
		this = character:qiao_mao
		this = character:bao_xin
		this = character:yuan_yi
		this = character:sun_jian
	}
}

hd_is_hd_defensive_league = {
	exists = confederation
	confederation = {
		confederation_type = confederation_type:hd_defensive_league_type
	}
}

hd_is_anti_regent_league_member = {
	exists = global_var:hd_anti_regent_league_leader
	global_var:hd_anti_regent_league_leader = {
		is_alive = yes
		has_variable_list = hd_anti_regent_members
		is_target_in_variable_list = {
			name = hd_anti_regent_members
			target = prev
		}
	}
}

hd_anti_regent_real_confederation_ready_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = yes
	is_independent_ruler = yes
	is_imprisoned = no
	NOT = { has_character_flag = hd_anti_regent_leader_dying }
	has_variable = hd_anti_regent_confederation
	exists = confederation
	confederation = {
		this = prev.var:hd_anti_regent_confederation
		confederation_type = confederation_type:hd_anti_regent_league_type
		any_confederation_member = { count >= 2 }
	}
}

hd_can_call_league_member_to_war_trigger = {
	exists = confederation
	has_variable = hd_league_leader
	var:hd_league_leader = root
	is_at_war_as_attacker = yes
	any_character_war = {
		primary_attacker = root
	}
}
