# Minimal CK3 1.20.0.3 native Rite filter; original Faith helper retained.
rite_is_interesting_heresy_to_state_rite_trigger = {
	faith.religion = root.top_liege.primary_title.state_rite.faith.religion
	this != root.top_liege.primary_title.state_rite
	num_rite_county_followers >= 3
	save_temporary_scope_as = temp_heresy
	root.top_liege.primary_title.state_rite = {
		rite_hostility_level = {
			target = scope:temp_heresy
			value >= faith_hostile_level
		}
	}
}
