# 天朝循环·时代特色：紧张期套件（docs/天朝循环_时代特色重制计划.md §3.3）判定

# Scope：角色。紧张期的天子
hd_dc_kit_ten_emperor_trigger = {
	has_title = title:h_china
	hd_dc_phase_is_trigger = { PHASE = instability }
}

# Scope：角色。天子朝中的"权臣"：天朝大臣、天子朝廷的朋党领袖，或天子的辅政
hd_dc_kit_ten_is_power_minister_trigger = {
	is_alive = yes
	is_adult = yes
	exists = title:h_china.holder
	NOT = { this = title:h_china.holder }
	OR = {
		top_liege = title:h_china.holder
		is_courtier_of = title:h_china.holder
	}
	OR = {
		tgp_has_minister_title = yes
		hd_party_is_any_leader_of_court_trigger = { COURT = title:h_china.holder }
		is_diarch_of_target = title:h_china.holder
	}
}

# Scope：角色。可以发起弹劾：天子治下（不含天子本人）的重要人物
hd_dc_kit_ten_can_impeach_trigger = {
	is_adult = yes
	is_imprisoned = no
	exists = title:h_china.holder
	NOT = { this = title:h_china.holder }
	OR = {
		top_liege = title:h_china.holder
		is_courtier_of = title:h_china.holder
	}
}

# Scope：角色。可以勒兵入朝（清君侧）：天子的有地直属封臣
hd_dc_kit_ten_can_qingjunce_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_landed = yes
	exists = title:h_china.holder
	liege = title:h_china.holder
	NOT = { is_at_war_with = title:h_china.holder }
}

# Scope：天子。当前乱象为 $STATE$（consort 外戚专权 / eunuch 宦官干政 / warlord 权臣藩镇 / partisan 党争内耗）
hd_dc_kit_ten_chaos_is_trigger = {
	has_character_modifier = hd_dc_kit_ten_chaos_$STATE$_modifier
}

# Scope：天子。$G$ 党为本年最强且不属"各党势均"
hd_dc_kit_ten_dominant_party_trigger = {
	hd_dc_strongest_party_is_trigger = { G = $G$ }
	var:hd_party_$G$_share >= hd_dc_ten_chaos_even_share
}
