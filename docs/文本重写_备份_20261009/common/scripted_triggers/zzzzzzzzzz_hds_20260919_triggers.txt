# 孙策继承判定放宽：只要持有家族头衔即可（不再要求营地等于孙坚原营地）。
# 强制继承脚本（hds_sun_force_inherit_effect）已打 hds_sc_inherited 时也视为成立，避免死锁。
hds_sun_estate_inherited_trigger = {
    has_global_variable = wj_hchar_sun_ce
    AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
    is_alive = yes
    OR = {
        any_held_title = { is_noble_family_title = yes }
        has_character_flag = hds_sc_inherited
    }
}

# A campaign is completed by actual control, never by issuing a war command.
hds_sc_wu_control_trigger = {
    has_global_variable = wj_hchar_sun_ce
    AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
    is_alive = yes is_landed = yes
    has_title = title:d_danyang
    title:d_danyang = {
        any_de_jure_county = { always = yes }
        NOT = {
            any_de_jure_county = {
                NOT = {
                    AND = {
                        exists = holder
                        holder ?= {
                            is_alive = yes
                            OR = { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } is_vassal_or_below_of = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } }
                        }
                    }
                }
            }
        }
    }
}
# Sun Ce's historical invasion is not blocked by a peacetime strength-ratio heuristic.
# Actual war legality, truce, alliance, imprisonment and adulthood gates remain in place.
hds_candidate_enemy_trigger = {
    is_alive = yes is_landed = yes is_independent_ruler = yes
    exists = scope:hds_campaign_actor
    scope:hds_campaign_actor ?= { is_alive = yes }
    NOT = { this = scope:hds_campaign_actor }
    OR = {
        scope:hds_campaign_actor ?= { AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } } is_alive = yes }
        current_military_strength <= hds_strength_ceiling
    }
    NOT = { is_allied_to = scope:hds_campaign_actor }
    NOT = { is_at_war_with = scope:hds_campaign_actor }
    scope:hds_campaign_actor ?= {
        NOT = { has_truce = prev }
        NOT = { is_allied_in_war = prev }
    }
}
