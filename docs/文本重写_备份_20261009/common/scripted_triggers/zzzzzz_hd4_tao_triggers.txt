hd4_tao_complete_trigger = {
	this = character:tao_qian
	title:k_uuii_xuzhou = {
		any_in_de_jure_hierarchy = {
			tier = tier_duchy
		}
		NOT = {
			any_in_de_jure_hierarchy = {
				tier = tier_duchy
				NOT = {
					AND = {
						exists = holder
						holder = {
							OR = {
								this = character:tao_qian
								is_vassal_or_below_of = character:tao_qian
							}
						}
					}
				}
			}
		}
	}
}
hd4_tao_can_campaign_trigger = {
	this = character:tao_qian
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	is_imprisoned = no
	has_character_flag = hd4_tao_activated
	NOT = {
		has_character_flag = hd4_tao_complete
	}
	NOT = {
		any_character_war = {
			count >= 3
			always = yes
		}
	}
	NOT = {
		has_character_flag = hd4_tao_split_grace
	}
}
hd4_tao_target_duchy_trigger = {
	tier = tier_duchy
	target_is_de_jure_liege_or_above = title:k_uuii_xuzhou
	NOT = {
		AND = {
			exists = holder
			holder = {
				OR = {
					this = character:tao_qian
					is_vassal_or_below_of = character:tao_qian
				}
			}
		}
	}
}
