# 是否显示"隐居养望"决议：持有家族头衔、有上级、尚未隐居。
WJ_yinshi_can_enter_shown_trigger = {
 NOT = { has_trait = WJ_yangwang_trait }
 any_held_title = { is_noble_family_title = yes }
 exists = liege
 is_independent_ruler = no
}

# 决议可用条件（失败时显示）。
WJ_yinshi_can_enter_valid_trigger = {
 is_adult = yes
 is_imprisoned = no
 is_incapable = no
 custom_tooltip = {
  text = WJ_yinshi_cooldown_tt
  NOT = { has_variable = wj_yinshi_cooldown_end }
 }
}

# 雅号事件入口：已是隐士（改雅号），或正在进入隐居。
WJ_yinshi_naming_ok_trigger = {
 OR = {
  has_trait = WJ_yangwang_trait
  AND = {
   WJ_yinshi_can_enter_shown_trigger = yes
   WJ_yinshi_can_enter_valid_trigger = yes
  }
 }
}

# 故事事件能否发生在本人身上。
WJ_yinshi_able_trigger = {
 is_alive = yes
 has_trait = WJ_yangwang_trait
 is_imprisoned = no
 is_incapable = no
}

# 宫廷职位、议会、骑士、统兵的屏蔽条件（带提示）。
WJ_yinshi_not_hermit_trigger = {
 custom_tooltip = {
  text = WJ_yinshi_blocked_tt
  NOT = { has_trait = WJ_yangwang_trait }
 }
}

# 持有男爵级以上非家族头衔（授官事件判定）。
WJ_yinshi_holds_office_title_trigger = {
 any_held_title = { is_noble_family_title = no }
}

# 朋友类事件的人选。
WJ_yinshi_companion_ok_trigger = {
 is_alive = yes
 is_adult = yes
 is_imprisoned = no
 is_incapable = no
 this != scope:wj_yinshi_subject
}

# 当地人事件里用作肖像的县持有者。
WJ_yinshi_local_lord_ok_trigger = {
 is_alive = yes
 is_adult = yes
 this != scope:wj_yinshi_subject
}

# ---------------------------------------------------------------------------
# 邀请类发邀者（沿用旧养望条件；只在隐士本人的上级链/玩家中找，不做全图搜索）
# 发邀者作用域；scope:wj_yinshi_subject = 隐士。
# ---------------------------------------------------------------------------
WJ_yinshi_xiaolian_actor_trigger = {
 is_alive = yes
 is_adult = yes
 is_imprisoned = no
 is_incapable = no
 government_has_flag = government_is_celestial
 is_landed = yes
 OR = { primary_title.tier = tier_duchy primary_title.tier = tier_kingdom }
 this != scope:wj_yinshi_subject
 top_liege = scope:wj_yinshi_subject.top_liege
 scope:wj_yinshi_subject = { WJ_can_be_recommended_xiaolian_candidate_trigger = { RECOMMENDER = prev } }
}
WJ_yinshi_maocai_actor_trigger = {
 WJ_langguan_real_emperor_trigger = no
 scope:wj_yinshi_subject = { WJ_langguan_real_emperor_trigger = no }
 is_alive = yes
 is_adult = yes
 is_imprisoned = no
 is_incapable = no
 government_has_flag = government_is_celestial
 WJ_has_substantive_office_trigger = yes
 highest_held_title_tier >= tier_kingdom
 this != top_liege
 this != scope:wj_yinshi_subject
 top_liege = scope:wj_yinshi_subject.top_liege
 scope:wj_yinshi_subject = { WJ_can_be_recommended_maocai_candidate_trigger = { RECOMMENDER = prev } }
}
# 入幕邀请只由玩家府主发出（AI 府主走 WJ_ai_bizhao_01）。
WJ_yinshi_mufu_host_trigger = {
 is_ai = no
 WJ_mufu_owner_eligible = yes
 WJ_mufu_empty = yes
 is_imprisoned = no
 is_incapable = no
 NOT = { has_pending_interaction_of_type = wj_request_bizhao_interaction }
 NOT = { has_variable = wj_yangwang_mufu_pending }
 influence >= WJ_mufu_base_influence_cost
 scope:wj_yinshi_subject = { WJ_yinshi_mufu_candidate_trigger = { HOST = prev } }
}
# 隐士作为幕府候选：直接复用五岗幕府候选条件（含拒绝名单与环路索引）。
WJ_yinshi_mufu_candidate_trigger = {
 WJ_yinshi_able_trigger = yes
 WJ_mufu_candidate = { HOST = $HOST$ }
}
# 故事里的入幕邀请：scope:actor = 府主，scope:recipient = 隐士。
WJ_yinshi_mufu_story_valid_trigger = {
 scope:actor = { is_alive = yes is_imprisoned = no is_incapable = no WJ_mufu_owner_eligible = yes WJ_mufu_empty = yes }
 scope:recipient = { WJ_yinshi_mufu_candidate_trigger = { HOST = scope:actor } }
 WJ_mufu_selected_empty = yes
 scope:actor.influence >= WJ_mufu_base_influence_cost
}
WJ_yinshi_mufu_story_reply_valid_trigger = {
 WJ_yinshi_mufu_story_valid_trigger = yes
 scope:actor = { has_variable = wj_yangwang_mufu_pending }
 scope:actor.var:wj_yangwang_mufu_pending = scope:recipient
}

# 对隐士发起交互时的 AI 意愿折扣、自动转事件判定。
WJ_yinshi_recipient_is_hermit_trigger = {
 scope:recipient = { has_trait = WJ_yangwang_trait }
}

# 自省事件：能否再得一个性格特质（性格特质 <3 且至少一项无相反特质）。
WJ_yinshi_can_gain_personality_trigger = {
 number_of_personality_traits < 3
 OR = {
  NOR = { has_trait = patient has_trait = impatient }
  NOR = { has_trait = diligent has_trait = lazy }
  NOR = { has_trait = content has_trait = ambitious }
  NOR = { has_trait = humble has_trait = arrogant }
  NOR = { has_trait = calm has_trait = wrathful }
  NOR = { has_trait = temperate has_trait = gluttonous }
 }
}
