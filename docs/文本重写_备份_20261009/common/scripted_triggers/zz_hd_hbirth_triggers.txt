# 按年出场的历史人物没有 history ID，用出场时保存的全局指针识别：hd_hb_is = { ID = shi_lei }
hd_hb_is = {
	trigger_if = {
		limit = { has_global_variable = hd_hb_p_$ID$ }
		this = global_var:hd_hb_p_$ID$
	}
	trigger_else = { always = no }
}

# 该人物已出场且在世。
hd_hb_alive = {
	trigger_if = {
		limit = { has_global_variable = hd_hb_p_$ID$ }
		global_var:hd_hb_p_$ID$ = { is_alive = yes }
	}
	trigger_else = { always = no }
}
