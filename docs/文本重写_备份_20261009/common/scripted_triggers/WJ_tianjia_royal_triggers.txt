# Candidate scope; EMPEROR is a real character scope, not a player/global flag.
WJ_royal_prince_candidate = {
 is_alive = yes is_male = yes is_child_of = $EMPEROR$ top_liege = $EMPEROR$
 NOT = { has_trait = SanGong_Taizi }
 NOT = { AND = { has_variable = wj_peer_rank var:wj_peer_rank >= 4 } }
}
WJ_royal_crown_prince = {
 is_alive = yes has_trait = SanGong_Taizi
 exists = var:WJ_royal_patron
 var:WJ_royal_patron = { WJ_inner_emperor_trigger = yes }
 top_liege = var:WJ_royal_patron
 is_child_of = var:WJ_royal_patron
}
WJ_royal_can_install_prince = {
 WJ_inner_emperor_trigger = yes
 save_temporary_scope_as = wj_prince_emperor
 exists = scope:taizi
 scope:taizi = { WJ_royal_prince_candidate = { EMPEROR = scope:wj_prince_emperor } }
 NOT = { any_child = { is_alive = yes has_trait = SanGong_Taizi } }
 NOT = { AND = { exists = var:WJ_royal_heir var:WJ_royal_heir = { WJ_royal_crown_prince = yes } } }
}
