# No situation phase, participation-group, culture, faith or wealth gate here.
# Geographic membership = at least one OWNED subrealm county de jure in h_china.
# Occupation does not change the holder and is therefore not membership.
hd_unification_county_in_china_trigger = {
	tier = tier_county
	exists = hegemony
	hegemony = title:h_china
}

hd_unification_realm_in_china_trigger = {
	is_alive = yes
	is_ruler = yes
	is_landed = yes
	exists = primary_title
	any_sub_realm_county = {
		hd_unification_county_in_china_trigger = yes
	}
}

hd_unification_celestial_attacker_trigger = {
	is_independent_ruler = yes
	government_has_flag = government_is_celestial
	hd_unification_realm_in_china_trigger = yes
}

# New full-annex CB is an empire-tier privilege, not an AI-only privilege.
# Other governments do NOT receive the dedicated aggressive AI story.
hd_unification_empire_attacker_trigger = {
	is_independent_ruler = yes
	highest_held_title_tier >= tier_empire
	hd_unification_realm_in_china_trigger = yes
}

hd_unification_ai_ready_trigger = {
	is_ai = yes
	hd_unification_celestial_attacker_trigger = yes
	is_physically_able_adult = yes
	current_military_strength > 0
	NOT = { has_character_flag = hd_unification_ai_cooldown }
	# Defensive wars are deliberately not counted. One ongoing offensive war
	# still leaves a slot. This is NOT an is_at_war = no condition.
	NOT = {
		any_character_war = {
			count >= hd_unification_ai_max_offensive_wars
			primary_attacker = prev
		}
	}
}

# Parameters: ATTACKER and DEFENDER must both be CHARACTER scopes.
# Native CB checks supply scope:attacker and scope:defender, i.e. the actual
# war participants, not the holder of a frontier county. Missing parameters
# fail closed BEFORE any character scope switch or military-strength read.
hd_unification_ai_pair_allowed_trigger = {
	trigger_if = {
		limit = {
			exists = $ATTACKER$
			exists = $DEFENDER$
		}
		$ATTACKER$ = {
			hd_unification_ai_ready_trigger = yes
			NOT = { this = $DEFENDER$ }
			NOT = { has_truce = $DEFENDER$ }
			NOT = { is_allied_to = $DEFENDER$ }
			NOT = { is_at_war_with = $DEFENDER$ }
		}
		$DEFENDER$ = {
			is_independent_ruler = yes
			hd_unification_realm_in_china_trigger = yes
			current_military_strength <= $ATTACKER$.hd_unification_enemy_ceiling_value
		}
	}
	trigger_else = { always = no }
}

# Also apply the cap when the native background AI selects either CB.
# This gate is for STARTING only; never call it from should_invalidate.
hd_unification_ai_cb_start_allowed_trigger = {
	trigger_if = {
		limit = {
			exists = $ATTACKER$
			exists = $DEFENDER$
		}
		trigger_if = {
			limit = {
				$ATTACKER$ = {
					is_ai = yes
					government_has_flag = government_is_celestial
				}
			}
			hd_unification_ai_pair_allowed_trigger = {
				ATTACKER = $ATTACKER$
				DEFENDER = $DEFENDER$
			}
		}
	}
	trigger_else = { always = no }
}

# Title scope: requires a real, still-held enemy county under the target kingdom.
hd_unification_target_has_defender_land_trigger = {
	any_in_de_jure_hierarchy = {
		tier = tier_county
		exists = holder
		holder = {
			OR = {
				this = scope:defender
				target_is_liege_or_above = scope:defender
			}
		}
	}
}
