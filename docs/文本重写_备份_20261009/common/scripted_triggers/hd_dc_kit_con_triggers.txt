# 天朝循环·时代特色：新朝征服期套件判定（见 docs/天朝循环_时代特色重制计划.md §3.8）

# Scope：角色。天子且处于新朝征服期
hd_dc_con_emperor_trigger = {
	has_title = title:h_china
	hd_dc_phase_is_trigger = { PHASE = instability_conquest }
}

# Scope：角色。本族：$CULTURE$ 文化且非中华传承
hd_dc_con_native_trigger = {
	culture = $CULTURE$
	hd_dc_is_chinese_heritage_trigger = no
}

# Scope：县。可圈之地：$RULER$ 直辖、汉人（中华传承）文化、非都城
hd_dc_con_quandi_county_trigger = {
	holder = $RULER$
	culture ?= { has_cultural_pillar = heritage_chinese }
	NOT = { this = $RULER$.capital_county }
}

# Scope：角色。前朝遗民候选（$RULER$ = 天子）
hd_dc_con_remnant_candidate_trigger = {
	is_alive = yes
	NOT = { this = $RULER$ }
	NOT = { dynasty ?= $RULER$.dynasty }
}

# Scope：角色。前朝复辟人选：在世、成年、有地、在天子境内、对天子好感 < 0、未被囚、不在战
hd_dc_con_restoration_candidate_trigger = {
	hd_dc_con_remnant_candidate_trigger = { RULER = $RULER$ }
	is_adult = yes
	is_landed = yes
	top_liege ?= $RULER$
	is_imprisoned = no
	is_at_war = no
	opinion = {
		target = $RULER$
		value < 0
	}
}

# Scope：角色。可受圈地之赐的本族廷臣
hd_dc_con_favorite_candidate_trigger = {
	is_adult = yes
	is_landed = no
	is_imprisoned = no
	culture = $RULER$.culture
	NOT = { has_trait = incapable }
	can_be_granted_titles_by = { RULER = $RULER$ }
}

# Scope：天子。可创建胡汉混合文化：DLC 与游戏规则允许、都城为汉地文化、原版 can_hybridize 通过
hd_dc_con_can_hybridize_trigger = {
	has_dlc_feature = hybridize_culture
	game_rule_can_create_hybird_culture_trigger = yes
	capital_county.culture ?= { has_cultural_pillar = heritage_chinese }
	can_hybridize = capital_county.culture
}
