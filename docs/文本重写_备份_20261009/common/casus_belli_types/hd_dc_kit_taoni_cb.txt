# 天朝循环·时代特色：讨逆（2026-10-08 用户指定）
# 原主君讨伐「要求独立」被拒后仍自立的封臣。结构参照原版 vassalization_cb（00_vassalization.txt）。
# - 2026-10-08 泛化：不再限天子。叛离者主头衔带"叛离"标记 hd_dc_taoni_rebel（自立后 10 年，标记跟着头衔走，继承人同样可被讨），
#   标记值为原主君当时的主头衔，该头衔的现任持有者可用
# - 免费；胜利：对方连同全部领地重新臣服，本人下狱；停战按原版内战 1825 天（胜/和/败）
# - AI 积极性：一般（不加权）

hd_dc_taoni_cb = {
	icon = vassalization
	group = vassalization

	allowed_for_character = {
		is_landed = yes
	}

	allowed_against_character = {
		scope:defender = {
			is_independent_ruler = yes
			primary_title = {
				exists = var:hd_dc_taoni_rebel
				var:hd_dc_taoni_rebel = { holder ?= scope:attacker }
			}
		}
	}

	# 标记到期不作废（与原设定一致）；原主君头衔易手或叛离者不再独立则作废
	should_invalidate = {
		OR = {
			scope:defender = { is_independent_ruler = no }
			scope:defender.primary_title = {
				exists = var:hd_dc_taoni_rebel
				var:hd_dc_taoni_rebel = { NOT = { holder ?= scope:attacker } }
			}
		}
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	cost = {
		prestige = 0
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = hd_dc_taoni_cb_victory_desc
	on_victory = {
		hd_dc_kit_taoni_victory_effect = yes
	}

	on_white_peace_desc = hd_dc_taoni_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = root.war
				result = white_peace
			}
		}
	}

	on_defeat_desc = hd_dc_taoni_cb_defeat_desc
	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_truce_both_ways = {
				character = scope:defender
				days = 1825
				war = root.war
				result = defeat
			}
		}
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = hd_dc_taoni_cb_war_name
	interface_priority = 80
}
