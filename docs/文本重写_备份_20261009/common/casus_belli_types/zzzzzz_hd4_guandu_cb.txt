hd4_guandu_cb = {
	icon = conquest
	group = conquest
	combine_into_one = no
	target_titles = all
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	target_top_liege_if_outside_realm = no
	allowed_for_character = {
		scope:attacker = {
			this = character:cao_cao
			is_alive = yes
			is_landed = yes
			has_character_flag = hd4_guandu_authorized
		}
		title:h_china = {
			has_variable = hd4_northern_two_announced
			NOT = {
				has_variable = hd4_guandu_done
			}
			NOT = {
				has_variable = hd4_legacy_guandu
			}
		}
	}
	allowed_against_character = {
		scope:defender = {
			this = character:yuan_shao
			is_alive = yes
			is_landed = yes
		}
	}
	valid_to_start = {
		scope:attacker = character:cao_cao
		scope:defender = character:yuan_shao
		scope:target = {
			tier = tier_duchy
		}
		character:cao_cao = {
			is_target_in_variable_list = {
				name = hd4_guandu_targets
				target = scope:target
			}
		}
	}
	cost = { }
	on_declaration = {
		on_declared_war = yes
		set_variable = {
			name = hd_world_war_kind
			value = flag:guandu
		}
		set_global_variable = hd_guandu_world_active
		character:yuan_shao = {
			remove_opinion = {
				modifier = hd_cao_yuan_historical_peace_opinion
				target = character:cao_cao
			}
		}
		character:cao_cao = {
			remove_opinion = {
				modifier = hd_cao_yuan_historical_peace_opinion
				target = character:yuan_shao
			}
		}
	}
	should_invalidate = {
		OR = {
			NOT = {
				scope:attacker = character:cao_cao
			}
			NOT = {
				scope:defender = character:yuan_shao
			}
			character:cao_cao = {
				is_alive = no
			}
			character:yuan_shao = {
				is_alive = no
			}
			NOT = {
				any_in_list = {
					list = target_titles
					any_de_jure_county = {
						AND = {
							exists = holder
							holder = {
								OR = {
									this = scope:defender
									is_vassal_or_below_of = scope:defender
								}
							}
						}
					}
				}
			}
		}
	}
	on_victory_desc = {
		desc = hd4_guandu_victory_desc
	}
	on_victory = {
		hd4_guandu_victory_effect = yes
		add_truce_attacker_victory_effect = yes
	}
	on_defeat_desc = {
		desc = hd4_guandu_defeat_desc
	}
	on_defeat = {
		hd4_guandu_defeat_effect = yes
		add_truce_attacker_defeat_effect = yes
	}
	on_white_peace_desc = {
		desc = hd4_guandu_white_peace_desc
	}
	on_white_peace = {
		hd4_guandu_finish_effect = yes
		add_truce_white_peace_effect = yes
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {
		hd4_guandu_invalidation_effect = yes
	}
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	transfer_behavior = transfer
	attacker_allies_inherit = no
	defender_allies_inherit = no
	war_name = HD4_GUANDU_WAR_NAME
	interface_priority = 90
	use_de_jure_wargoal_only = yes
}
