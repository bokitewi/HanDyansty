# 夺州：乱世（斗争/对峙）中王国级及以上独立统治者之间的一生一次战争。
# 战争目标为防守方一个公国（州），胜利夺取防守方全部头衔、领地与直属封臣。

hd_kingdom_seize_cb = {
	icon = celestial
	group = celestial
	combine_into_one = yes
	should_show_war_goal_subview = yes
	ai_only_against_neighbors = yes

	allowed_for_character = {
		hd_cycle_is_divided_trigger = yes
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		NOT = { has_character_flag = used_lifetime_kingdom_seize }
	}
	allowed_for_character_display_regardless = {
		custom_tooltip = {
			text = hd_kingdom_seize_cb_lifetime_tt
			NOT = { has_character_flag = used_lifetime_kingdom_seize }
		}
	}
	allowed_against_character = {
		scope:defender = {
			is_independent_ruler = yes
			highest_held_title_tier >= tier_kingdom
			hd_cycle_in_china_trigger = yes
		}
	}

	target_titles = all
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			any_in_de_jure_hierarchy = {
				tier = tier_county
				holder.top_liege ?= scope:defender
			}
		}
	}

	should_invalidate = {
		OR = {
			hd_cycle_is_divided_trigger = no
			scope:defender = { is_independent_ruler = no }
		}
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {
		scope:attacker = {
			if = {
				limit = { is_alive = yes }
				remove_character_flag = used_lifetime_kingdom_seize	# 战争被无效化则返还
			}
		}
	}

	ai_score_mult = { value = 1 }

	cost = {
		prestige = {
			add = {
				value = 300
				desc = CB_BASE_COST
			}
			multiply = common_cb_prestige_cost_multiplier
			multiply = hd_dc_sta_conquest_cost_mult_value	# 天朝循环·时代特色：对峙期征服类 CB 花费 ×2
		}
	}

	on_declaration = {
		on_declared_war = yes
		scope:attacker = { add_character_flag = used_lifetime_kingdom_seize }
	}

	on_victory_desc = hd_kingdom_seize_cb_victory_desc
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		# 直属封臣转投胜者
		scope:defender = {
			every_vassal = {
				change_liege = {
					liege = scope:attacker
					change = scope:change
				}
			}
			# 全部头衔（含主头衔与全部领地）
			every_held_title = {
				change_title_holder = {
					holder = scope:attacker
					change = scope:change
					take_baronies = yes
				}
			}
		}
		resolve_title_and_vassal_change = scope:change

		hd_cycle_fire_catalyst_effect = { CATALYST = catalyst_hd_big_conquest }
		scope:attacker = { add_prestige_experience = major_prestige_value }

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = hd_kingdom_seize_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = {
			add_prestige = {
				value = miniscule_prestige_value
				multiply = -1.0
			}
		}
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = hd_kingdom_seize_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_defeat_effect = yes
		scope:attacker = {
			pay_short_term_gold_reparations_effect = { GOLD_VALUE = 3 }
			add_prestige = {
				value = medium_prestige_value
				multiply = -1.0
			}
		}
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = hd_kingdom_seize_cb_war_name
	cb_name = hd_kingdom_seize_cb_name
	interface_priority = 115
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 100
	defender_score_from_battles_scale = 100
	max_defender_score_from_battles = 100
	max_attacker_score_from_battles = 100
	max_defender_score_from_occupation = 100
	max_attacker_score_from_occupation = 150
	ticking_war_score_targets_entire_realm = no
}
