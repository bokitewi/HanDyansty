# 天朝循环·时代特色：扩张期 · 开边 CB（见 docs/天朝循环_时代特色重制计划.md §3.1，参考 AUHCO frontier_expansion）
# - 仅天子在扩张期可用
# - 对象：天下（h_china 法理）以外、与天子疆域接壤的独立统治者（不含天子自己的朝贡国）
# - 战果：对方所有与天子疆域接壤的郡（公爵领）中属于对方疆域的县（胜利时结算）
# - 花费威望 500；停战 10 年（胜、和、败皆然）；胜利发射 catalyst_hd_frontier_victory

hd_dc_exp_frontier_cb = {
	icon = duchy_expansion
	group = celestial
	should_show_war_goal_subview = no
	ai_only_against_neighbors = yes

	allowed_for_character = {
		hd_dc_exp_emperor_trigger = yes
	}
	allowed_against_character = {
		scope:defender = {
			is_independent_ruler = yes
			hd_dc_is_outsider_trigger = yes
			NOT = { is_tributary_of = scope:attacker }
		}
	}
	valid_to_start = {
		scope:defender = {
			any_sub_realm_county = {
				any_neighboring_county = { holder.top_liege ?= scope:attacker }
			}
		}
	}
	should_invalidate = {
		NOT = { scope:attacker = { has_title = title:h_china } }
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	cost = {
		prestige = hd_dc_exp_frontier_prestige_cost
	}

	ai_score = {
		value = hd_dc_exp_frontier_ai_score_value
	}
	ai_score_mult = {
		value = 1
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = hd_dc_exp_frontier_cb_victory_desc_attacker
			}
			desc = hd_dc_exp_frontier_cb_victory_desc
		}
	}
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		hd_dc_exp_frontier_victory_effect = yes
		add_truce_attacker_victory_effect = yes
		# 停战 10 年（覆盖默认停战）
		scope:attacker = {
			add_truce_both_ways = {
				character = scope:defender
				days = 3650
				war = root.war
				result = victory
			}
		}
	}

	on_white_peace_desc = hd_dc_exp_frontier_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
		# 停战 10 年
		scope:attacker = {
			add_truce_both_ways = {
				character = scope:defender
				days = 3650
				war = root.war
				result = white_peace
			}
		}
	}

	on_defeat_desc = hd_dc_exp_frontier_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { pay_short_term_gold_reparations_effect = { GOLD_VALUE = 3 } }
		add_truce_attacker_defeat_effect = yes
		# 停战 10 年
		scope:attacker = {
			add_truce_both_ways = {
				character = scope:defender
				days = 3650
				war = root.war
				result = defeat
			}
		}
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = hd_dc_exp_frontier_cb_war_name
	cb_name = hd_dc_exp_frontier_cb_name
	interface_priority = 105

	ticking_war_score_targets_entire_realm = yes	# 无战争目标头衔：照原版降服 CB 以全境计算持续战争分
	attacker_wargoal_percentage = 0.8
	attacker_score_from_occupation_scale = 120
	attacker_score_from_battles_scale = 100
	defender_score_from_battles_scale = 100
	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100
	max_attacker_score_from_occupation = 120
	max_defender_score_from_occupation = 100
}
