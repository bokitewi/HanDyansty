WJ_kaoke_malpractice_secret = {
 category = civil
 is_valid = {
  exists = scope:secret_owner
  exists = scope:secret_target
  scope:secret_owner = { is_alive = yes }
  scope:secret_target = { is_alive = yes }
 }
 # 原生major寻秘与勒索会以secret_owner作为评价者；不能只让皇帝判断犯罪。
 # 专属抓捕/剥夺理由由下方曝光opinion单独授予受损皇帝。
 is_shunned = { always = no }
 is_criminal = { always = yes }
 on_owner_death = { remove_secret = yes }
 on_expose = {
  if = {
   limit = {
    NOT = { has_variable = wj_kaoke_exposed }
    exists = scope:secret_owner
    exists = scope:secret_target
    scope:secret_owner = { is_alive = yes }
    scope:secret_target = { is_alive = yes }
   }
   set_variable = { name = wj_kaoke_exposed value = 1 }
   save_scope_as = wj_kaoke_exposed_secret
   save_scope_value_as = { name = wj_kaoke_case_influence value = var:wj_kaoke_influence }
   scope:secret_target = { save_scope_as = wj_kaoke_injured_emperor }
   scope:secret_owner = {
    save_scope_as = wj_kaoke_offender
    # Public, lifelong and non-stacking; pardon only clears the separate case opinion.
    add_trait = WJ_kaoke_malpractice
    if = { limit = { has_variable = wj_kaoke_case_epoch } change_variable = { name = wj_kaoke_case_epoch add = 1 } }
    else = { set_variable = { name = wj_kaoke_case_epoch value = 1 } }
    save_scope_value_as = { name = wj_kaoke_case_epoch value = var:wj_kaoke_case_epoch }
    set_variable = { name = wj_kaoke_case_pending value = 1 }
    set_variable = { name = wj_kaoke_case_victim value = scope:wj_kaoke_injured_emperor }
    set_variable = { name = wj_kaoke_case_realm value = scope:wj_kaoke_injured_emperor.primary_title }
   }
   scope:wj_kaoke_injured_emperor = {
    add_opinion = { target = scope:wj_kaoke_offender modifier = WJ_kaoke_malpractice_opinion }
    trigger_event = { id = wj_kaoke_central.2000 }
   }
  }
 }
}
