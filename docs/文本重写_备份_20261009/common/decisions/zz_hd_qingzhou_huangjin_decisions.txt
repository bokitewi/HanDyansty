#   决议 → hd_huangjin.1(军事) → .2(管理) → .3(外交+总结算)
# 关闭规则: 任一鉴定大成功→当场永久关闭;三验全过→结算时永久关闭;
#           其余情形全局点击计数达3次→永久关闭
# 阈值来源: 01_landed_titles.txt 实数 青州70县/兖州80县,40%即 28/32/60

hd_subdue_qingzhou_huangjin_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_invite_knights.dds"
	}
	title = hd_subdue_qingzhou_huangjin_decision
	desc = hd_subdue_qingzhou_huangjin_decision_desc
	selection_tooltip = hd_subdue_qingzhou_huangjin_decision_tooltip
	confirm_text = hd_subdue_qingzhou_huangjin_decision_confirm
	decision_group_type = major
	sort_order = 2109

	is_shown = {
		is_playable_character = yes
		is_landed = yes
		current_date > 190.12.31
		NOT = { exists = global_var:hd_hj_closed }
		NOT = { has_character_flag = hd_hj_done }
		OR = {
			any_sub_realm_county = {
				count >= 28
				title_province = { geographical_region = uuii_han_qingzhou_region }
			}
			any_sub_realm_county = {
				count >= 32
				title_province = { geographical_region = uuii_han_yanzhou_region }
			}
			any_sub_realm_county = {
				count >= 60
				OR = {
					title_province = { geographical_region = uuii_han_qingzhou_region }
					title_province = { geographical_region = uuii_han_yanzhou_region }
				}
			}
		}
	}

	effect = {
		add_character_flag = hd_hj_done
		hidden_effect = {
			# 全局尝试计数: 第三次点击即彻底关闭
			if = {
				limit = { NOT = { exists = global_var:hd_hj_click_count } }
				set_global_variable = { name = hd_hj_click_count value = 0 }
			}
			change_global_variable = { name = hd_hj_click_count add = 1 }
			if = {
				limit = { global_var:hd_hj_click_count >= 3 }
				set_global_variable = { name = hd_hj_closed value = root }
			}
			# 鉴定链临时变量(链末 hd_huangjin.3 结算后清除)
			set_variable = { name = hd_hj_successes value = 0 }
			set_variable = { name = hd_hj_crit value = 0 }
			trigger_event = { id = hd_huangjin.1 days = 1 }
		}
	}

	ai_check_interval = 12
	ai_will_do = {
		base = 100
		# 安于现状/胆怯的AI永不招降
		modifier = { factor = 0 has_trait = content }
		modifier = { factor = 0 has_trait = craven }
		# 既无野心又无勇的AI几乎不会尝试
		modifier = {
			factor = 0.05
			NOT = { has_trait = ambitious }
			NOT = { has_trait = brave }
		}
	}
}
