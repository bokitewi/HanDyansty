# 清除瘴气决议：移除规则照搬原版 EP3 restore_backwater_counties_decision（拜占庭"死气沉沉"）。
# 经确认的差异：不限头衔等级——直辖瘴气县即可见（holder = root）；发展度阈值 30（原版 50）。
# 不照搬部分：EP3 庆祝事件 3000/3001、总督奖励修正、建筑锁定。
hd_clear_miasma_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_black_plague.dds"
	}
	title = hd_clear_miasma_decision
	desc = hd_clear_miasma_decision_desc
	selection_tooltip = hd_clear_miasma_decision_tooltip
	confirm_text = hd_clear_miasma_decision_confirm
	decision_group_type = major
	sort_order = 2010

	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 120
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	is_shown = {
		any_county_in_region = {
			region = china_south_world
			has_county_modifier = hd_miasma_county_modifier
			holder = { this = root }
		}
	}

	is_valid = {
		is_at_war = no
		custom_tooltip = {
			text = hd_miasma_county_eligible_removal_tt
			any_county_in_region = {
				region = china_south_world
				has_county_modifier = hd_miasma_county_modifier
				holder = { this = root }
				development_level >= 30
			}
		}
	}

	is_valid_showing_failures_only = {
		is_alive = yes
		is_imprisoned = no
		NOT = { exists = involved_activity }
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
				}
				add = medium_gold_value
				multiply = {
					value = 1
					every_county_in_region = {
						region = china_south_world
						limit = {
							has_county_modifier = hd_miasma_county_modifier
							holder = { this = root }
							development_level >= 30
						}
						add = 1
					}
				}
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = medium_gold_value
				multiply = {
					value = 1
					every_county_in_region = {
						region = china_south_world
						limit = {
							has_county_modifier = hd_miasma_county_modifier
							holder = { this = root }
							development_level >= 30
						}
						add = 1
					}
				}
			}
		}
	}

	effect = {
		custom_tooltip = hd_miasma_removal_tt
		if = {
			limit = {
				NOT = {
					any_county_in_region = {
						region = china_south_world
						has_county_modifier = hd_miasma_county_modifier
						holder = { this = root }
						development_level >= 30
					}
				}
			}
			custom_tooltip = hd_miasma_no_counties_meet_criteria_tt
		}
		else = {
			every_county_in_region = {
				region = china_south_world
				limit = {
					has_county_modifier = hd_miasma_county_modifier
					holder = { this = root }
					development_level >= 30
				}
				add_to_list = hd_miasma_cleared_counties
			}
		}
		send_interface_message = {
			type = event_generic_good
			title = hd_miasma_cleared.t
			desc = hd_miasma_cleared.desc
			left_icon = root
			right_icon = root.primary_title
			every_in_list = {
				list = hd_miasma_cleared_counties
				remove_county_modifier = hd_miasma_county_modifier
			}
		}
	}
}
