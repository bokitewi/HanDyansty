		piety = {
			if = {
				limit = { petition_head_of_faith_has_usable_hook_on_religious_authority_trigger = yes }
				value = 0
			}
			else_if = {
				limit = { scope:petition_head_of_faith_repentance = yes }
				value = major_piety_value
			}
			else_if = {
				limit = { scope:petition_head_of_faith_excommunication = yes }
				value = massive_piety_value
			}
			else = {
				value = 0
			}
		}
	}

	widget = {
		gui = "decision_view_widget_petition_head_of_faith"
		controller = decision_option_list_controller
		decision_to_second_step_button = "petition_head_of_faith_decision_confirm"

		item = {
			value = petition_head_of_faith_repentance
			localization = petition_head_of_faith_decision_option_repentance
			current_description = petition_head_of_faith_decision_option_repentance_desc
			icon = "gfx/interface/icons/traits/excommunicated.dds"
			is_valid = {
				petition_head_of_faith_repentance_available_trigger = yes
				custom_description = {
					text = petition_head_of_faith_decision_cost_major_piety
					petition_head_of_faith_can_afford_major_piety_cost_trigger = yes
				}
			}
			ai_chance = { value = 0 }
		}
		item = {
			value = petition_head_of_faith_excommunication
			localization = petition_head_of_faith_decision_option_excommunication
			current_description = petition_head_of_faith_decision_option_excommunication_desc
			icon = "gfx/interface/icons/character_interactions/excommunication.dds"
			is_valid = {
				petition_head_of_faith_excommunication_decision_available_trigger = yes
				custom_description = {
					text = petition_head_of_faith_decision_cost_massive_piety
					petition_head_of_faith_can_afford_massive_piety_cost_trigger = yes
				}
			}
			ai_chance = { value = 0 }
		}
	}

	effect = {
		save_scope_as = petitioner
		petition_head_of_faith_save_religious_authority_effect = { SCOPE = petition_recipient }
		if = {
			limit = { scope:petition_head_of_faith_repentance = yes }
			custom_tooltip = petition_head_of_faith_decision_effects_major_piety
		}
		else = {
			custom_tooltip = petition_head_of_faith_decision_effects_massive_piety
		}
		petition_head_of_faith_display_recipient_disposition_effect = yes
		if = {
			limit = { scope:petition_head_of_faith_excommunication = yes }
			custom_tooltip = petition_head_of_faith_decision_excommunication_target_hint
		}
		if = {
			limit = { has_usable_hook = scope:petition_recipient }
			custom_tooltip = petition_head_of_faith_decision_hook_available
		}
		if = {
			limit = { scope:petition_head_of_faith_repentance = yes }
			open_interaction_window = {
				interaction = petition_head_of_faith_repentance_setup_interaction
				actor = root
				recipient = scope:petition_recipient
			}
		}
		else_if = {
			limit = { scope:petition_head_of_faith_excommunication = yes }
			open_interaction_window = {
				interaction = petition_head_of_faith_excommunication_setup_interaction
				actor = root
				recipient = scope:petition_recipient
			}
		}
	}

	ai_potential = { always = no }
}
