hd_train_xianzhenying_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_invite_knights.dds"
	}
	title = hd_train_xianzhenying_decision
	desc = hd_train_xianzhenying_decision_desc
	selection_tooltip = hd_train_xianzhenying_decision_tooltip
	confirm_text = hd_train_xianzhenying_decision_confirm
	decision_group_type = major
	sort_order = 2101

	is_shown = {
		is_playable_character = yes
		is_landed = yes
		hd_has_xianzhenying_accolade_trigger = yes
	}

	is_valid = {
		gold >= 100
		prestige >= 150
		any_knight = { hd_xianzhenying_trainer_trigger = yes }
	}

	cooldown = { years = 1 }

	effect = {
		hd_select_xianzhenying_trainer_effect = yes
		if = {
			limit = { exists = scope:hd_maa_trainer }
			trigger_event = { id = hd_maa_training.1000 }
		}
		else = {
			custom_tooltip = hd_maa_training_no_valid_trainer
		}
	}

	ai_check_interval = 12
	ai_will_do = {
		base = 60
		modifier = { factor = 0 gold < 300 }
		modifier = { factor = 0.5 gold < 600 }
	}
}
