# UUII excludes Japan from the playable map and final static database.
# This Japan-only gameplay entry point is intentionally disabled for new games.
#
# The two keys below must stay defined as hidden stubs: window_decisions.gui runs
# GetDecision('...') in a per-frame visible binding for both keys, so removing the
# keys entirely made every rendered decision row log a database lookup error
# (~300 errors/sec while the decisions window was open).

tgp_japan_become_shogun_decision = {
	picture = { reference = "gfx/interface/illustrations/event_story/tgp_japanese_shogunate.dds" }

	is_shown = { always = no }

	is_valid = { always = no }
}

	picture = { reference = "gfx/interface/illustrations/event_story/tgp_japanese_shogunate.dds" }

	is_shown = { always = no }

	is_valid = { always = no }
}
