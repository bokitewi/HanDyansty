	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	# 无影响力的政体改付威望（影响力 ×2）
	cost = {
		gold = hd_dc_adv_reform_gold
		influence = {
			value = 0
			if = {
				limit = { government_has_flag = government_has_influence }
				add = hd_dc_adv_reform_influence
			}
		}
		prestige = {
			value = 0
			if = {
				limit = { NOT = { government_has_flag = government_has_influence } }
				add = {
					value = hd_dc_adv_reform_influence
					multiply = 2
				}
			}
		}
	}

	effect = {
		custom_tooltip = hd_dc_kit_adv_reform_decision_effect_tt
		hidden_effect = {
			hd_dc_kit_adv_reform_start_effect = yes
			trigger_event = hd_dc_adv.0021
		}
	}

	ai_check_interval = 60
	ai_potential = {
		is_ai = yes
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			exists = var:hd_party_hanmen_share
			exists = var:hd_party_shizu_share
			var:hd_party_hanmen_share > var:hd_party_shizu_share
		}
		modifier = {
			add = 15
			stewardship >= 15
		}
		modifier = {
			add = 10
			OR = {
				has_trait = ambitious

			}

		}
	}
