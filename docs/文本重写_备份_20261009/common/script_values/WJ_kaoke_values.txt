WJ_kaoke_days_value = {
 value = 0
 if = {
  limit = { has_variable = wj_kaoke_start }
  value = current_date
  subtract = var:wj_kaoke_start
  min = 0
 }
}
WJ_kaoke_remaining_value = {
 value = 1825
 subtract = WJ_kaoke_days_value
 min = 0
}
WJ_kaoke_military_value = {
 value = var:wj_kaoke_military_points
 min = -60
 max = 60
}
WJ_kaoke_score_value = {
 value = 50
 add = var:wj_kaoke_civil_points
 add = WJ_kaoke_military_value
 add = var:wj_kaoke_choice_points
 if = { limit = { has_variable = wj_kaoke_central_points } add = var:wj_kaoke_central_points }
 if = { limit = { has_variable = wj_kaoke_superior_points } add = var:wj_kaoke_superior_points }
}
WJ_kaoke_grade_value = {
 value = 1
 if = { limit = { WJ_kaoke_score_value >= 100 } value = 5 }
 else_if = { limit = { WJ_kaoke_score_value >= 80 } value = 4 }
 else_if = { limit = { WJ_kaoke_score_value >= 50 } value = 3 }
 else_if = { limit = { WJ_kaoke_score_value >= 30 } value = 2 }
}
WJ_kaoke_candidate_value = {
 value = 0
 if = {
  limit = { WJ_kaoke_grade_usable_trigger = yes }
  if = {
   limit = { NOT = { has_variable = wj_kaoke_schema_version } }
   if = { limit = { var:wj_kaoke_last_grade = 4 } value = 60 }
   else_if = { limit = { var:wj_kaoke_last_grade = 3 } value = 30 }
   else_if = { limit = { var:wj_kaoke_last_grade = 1 } value = -30 }
  }
  else_if = { limit = { var:wj_kaoke_last_grade = 1 } value = -30 }
  else_if = { limit = { var:wj_kaoke_last_grade = 2 } value = -15 }
  else_if = {
   limit = { var:wj_kaoke_reward_choice = 1 }
   value = WJ_kaoke_reward_candidate_value
  }
 }
}
WJ_kaoke_reward_candidate_value = {
 value = 0
 if = { limit = { var:wj_kaoke_last_grade = 5 } value = 60 }
 else_if = { limit = { var:wj_kaoke_last_grade = 4 } value = 30 }
}
WJ_kaoke_reward_merit_value = {
 value = 75
 if = { limit = { var:wj_kaoke_last_grade = 5 } value = 150 }
}
WJ_kaoke_reward_treasury_value = {
 value = 150
 if = { limit = { var:wj_kaoke_last_grade = 5 } value = 300 }
}
WJ_kaoke_reward_xp_input_value = {
 value = 4
 if = { limit = { var:wj_kaoke_last_grade = 5 } value = 6 }
}
WJ_kaoke_reward_xp_value = {
 value = WJ_kaoke_reward_xp_input_value
 multiply = 0.5
}
