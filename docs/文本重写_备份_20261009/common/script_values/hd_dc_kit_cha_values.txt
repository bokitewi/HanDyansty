# 取得宣称：郡 = 200 金 + 200 威望 + 100 影响力；州 = ×3
############################################################
hd_dc_cha_buy_claim_gold_base = 200
hd_dc_cha_buy_claim_prestige_base = 200
hd_dc_cha_buy_claim_influence_base = 100
hd_dc_cha_buy_claim_kingdom_mult = 3

# 互动 scope（需 scope:target = 所选头衔）。郡 ×1，州 ×3
hd_dc_cha_buy_claim_tier_mult_value = {
	value = 1
	if = {
		limit = { scope:target ?= { tier = tier_kingdom } }
		value = hd_dc_cha_buy_claim_kingdom_mult
	}
}
hd_dc_cha_buy_claim_gold_value = {
	value = hd_dc_cha_buy_claim_gold_base
	multiply = hd_dc_cha_buy_claim_tier_mult_value
}
hd_dc_cha_buy_claim_prestige_value = {
	value = hd_dc_cha_buy_claim_prestige_base
	multiply = hd_dc_cha_buy_claim_tier_mult_value
}
hd_dc_cha_buy_claim_influence_value = {
	value = hd_dc_cha_buy_claim_influence_base
	multiply = hd_dc_cha_buy_claim_tier_mult_value
}
