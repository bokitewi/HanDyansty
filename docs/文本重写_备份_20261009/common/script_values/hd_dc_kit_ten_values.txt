# Scope：天子。中兴改革成功率
# 基础 20；天子五维均值每超过 10 一点 +2；丞相外交 / 户部内政 / 大将军军事 ≥20 各 +5、≥30 各 +10；
# 尊王诸党势力占比之和 ×0.3；事件链调整；5~90
hd_dc_kit_ten_restoration_chance_value = {
	value = 20
	add = {
		value = diplomacy
		add = martial
		add = stewardship
		add = intrigue
		add = learning
		divide = 5
		subtract = 10
		multiply = 2
	}
	if = {
		limit = { title:e_minister_chancellor.holder ?= { diplomacy >= 30 } }
		add = 10
	}
	else_if = {
		limit = { title:e_minister_chancellor.holder ?= { diplomacy >= 20 } }
		add = 5
	}
	if = {
		limit = { title:e_minister_of_revenue.holder ?= { stewardship >= 30 } }
		add = 10
	}
	else_if = {
		limit = { title:e_minister_of_revenue.holder ?= { stewardship >= 20 } }
		add = 5
	}
	if = {
		limit = { title:e_minister_grand_marshal.holder ?= { martial >= 30 } }
		add = 10
	}
	else_if = {
		limit = { title:e_minister_grand_marshal.holder ?= { martial >= 20 } }
		add = 5
	}
	add = {
		value = hd_dc_kit_ten_zunwang_share_value
		multiply = 0.3
	}
	if = {
		limit = { exists = var:hd_dc_ten_restoration_bonus }
		add = var:hd_dc_ten_restoration_bonus
	}
	min = 5
	max = 90
}
