WJ_official_assessment_score_value = {
	value = 0

	if = {
		limit = { has_character_flag = wj_assessment_temp_official }
		add = {
			value = stewardship
			multiply = 1.2
			ceiling = yes
			max = 30
			desc = WJ_ASSESSMENT_STEWARDSHIP_REASON
		}
	}
	else = {
		add = {
			value = learning
			multiply = 1.2
			ceiling = yes
			max = 30
			desc = WJ_ASSESSMENT_LEARNING_REASON
		}
	}

	add = {
		value = merit_level
		multiply = 3
		max = 15
		desc = WJ_ASSESSMENT_MERIT_REASON
	}

	add = {
		if = {
			limit = { var:wj_assessment_safe_answers > 0 }
			value = var:wj_assessment_safe_answers
			multiply = 5
			max = 15
		}
		desc = WJ_ASSESSMENT_SAFE_REASON
	}

	add = {
		if = {
			limit = { var:wj_assessment_round_bonus > 0 }
			value = var:wj_assessment_round_bonus
			max = 60
		}
		desc = WJ_ASSESSMENT_ROUND_REASON
	}

	max = 100
}
