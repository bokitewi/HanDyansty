# 义军来投 · 队数（1 队 = 100 人）
# 基线 8 队（800 人）：寻常人亦有乡党来投
# 威望与美名（忠义/仁德/生祠香火/乡里立祠）增其众；恶名（野心/千夫所指/附逆遗臭）大减其众，乡里铸有逆像者再削
zgrc_yijun_stacks = {
	value = 8
	add = prestige_level
	if = {
		limit = { has_character_modifier = zgrc_zhongyi_modifier }
		add = 3
	}
	if = {
		limit = { has_character_modifier = zgrc_rende_modifier }
		add = 3
	}
	if = {
		limit = { has_character_modifier = zgrc_shengci_modifier }
		add = 2
	}
	if = {
		limit = {
			capital_county = {
				OR = {
					has_county_modifier = zgrc_ci_zhong
					has_county_modifier = zgrc_ci_ren
				}
			}
		}
		add = 2
	}
	# 恶名：择最重者一击，不叠乘
	if = {
		limit = { has_character_modifier = zgrc_tuozhi_ni_modifier }
		multiply = 0.35
	}
	else_if = {
		limit = { has_character_modifier = zgrc_tuozhi_modifier }
		multiply = 0.5
	}
	else_if = {
		limit = { has_character_modifier = zgrc_yexin_modifier }
		multiply = 0.65
	}
	# 乡里铸有逆像：再削 2 队
	if = {
		limit = {
			capital_county = {
				OR = {
					has_county_modifier = zgrc_xiang_zui
					has_county_modifier = zgrc_xiang_ni
				}
			}
		}
		subtract = 2
	}
	floor = yes
	max = 16
}

# ============================================================
# 教化 CB —— AI 目标偏好（名托教化，实为兼并 → 专挑「蛮夷中的蛮夷」）
# 消费方：casus_belli_types/zgrc_cb_types.txt → ai_score_mult（县级 / 公国共用）
# 【四轴 · 轻重不一致】正 = 更想打，负 = 不想打
#   ① 关系  仇敌 / 朋友       → 最重档（±150；档比 2:1 照原版 chinese_consolidation）
#   ② 汉言  会说汉语          → 最轻档（个人习得项）
#   ③ 信仰  仅「邪恶」档      → 中重档（教化正当性）
#   ④ 文教  中华传承 / 慕华   → 最重档（文化已归华）
# ============================================================
zgrc_jiaohua_ai_score_value = {
	value = 0
	scope:attacker = {
		# —— ① 关系（分组判定用原版现成触发器，定义见 common/scripted_triggers/00_relation_triggers.txt:353-500）——
		if = {
			limit = { has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:defender } }
			add = zgrc_jiaohua_unfriendly_value
		}
		else_if = {
			limit = { has_any_good_relationship_with_character_trigger = { CHARACTER = scope:defender } }
			add = zgrc_jiaohua_friendly_value
		}
		# —— ② 汉言：会说汉语者＝已沾华化 → 不伐 ——
		if = {
			limit = { scope:defender = { knows_language = language_chinese } }
			add = zgrc_jiaohua_speaks_chinese_value
		}
		# —— ③ 信仰：仅「邪恶」档 → 先伐（原版常量 02_religion_values.txt:5-8）——
		if = {
			limit = { faith = { faith_hostility_level = { target = scope:defender.faith value >= faith_evil_level } } }
			add = zgrc_jiaohua_evil_faith_value
		}
		# —— ④ 文教：中华传承 / 慕华 → 不伐 ——
		if = {
			limit = {
				scope:defender.culture = {
					OR = {
						has_cultural_pillar = heritage_chinese
						has_cultural_tradition = tradition_sinophilic
					}
				}
			}
			add = zgrc_jiaohua_sinified_value
		}
	}
}

zgrc_jiaohua_unfriendly_value     = 50
zgrc_jiaohua_friendly_value       = -50
zgrc_jiaohua_speaks_chinese_value = -100
zgrc_jiaohua_evil_faith_value     = 100
zgrc_jiaohua_sinified_value       = -150
