# 陷阵营训练者评分（在骑士 scope 内求值，ROOT 为主君）：
# 军事×3 + 武勇×2 + 组织者 20 + 勇敢 10 + 对主君好感分档（≥80:+8 / ≥50:+5 / ≥20:+2）
# 好感取值无连续 script value，按原版惯例用阈值分桶近似"好感÷10"。
hd_xianzhenying_trainer_score_value = {
	value = martial
	add = martial
	add = martial
	add = prowess
	add = prowess
	if = {
		limit = { has_trait = organizer }
		add = 20
	}
	if = {
		limit = { has_trait = brave }
		add = 10
	}
	if = {
		limit = {
			opinion = {
				target = root
				value >= 80
			}
		}
		add = 8
	}
	else_if = {
		limit = {
			opinion = {
				target = root
				value >= 50
			}
		}
		add = 5
	}
	else_if = {
		limit = {
			opinion = {
				target = root
				value >= 20
			}
		}
		add = 2
	}
}
