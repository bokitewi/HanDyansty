# Character scope; one highest XP band, never a cumulative sum.
WJ_wuxun_appointment_value = {
 value = 0
 if = {
  limit = { government_has_flag = government_is_celestial has_trait = WJ_mingshi_trait has_trait_xp = { trait = WJ_mingshi_trait track = jungong value >= 0 } }
  value = 15
  if = { limit = { has_trait_xp = { trait = WJ_mingshi_trait track = jungong value >= 75 } } value = 50 }
  else_if = { limit = { has_trait_xp = { trait = WJ_mingshi_trait track = jungong value >= 50 } } value = 35 }
  else_if = { limit = { has_trait_xp = { trait = WJ_mingshi_trait track = jungong value >= 25 } } value = 25 }
 }
}

}
