hd_dc_adv_taixue_event_chance = 30		# 进取期太学年度联动时触发专属事件的概率

# Scope：天子。变法成功率（基础 50；寒门势力大于世族 +20；主持大臣内政 ≥20 +10；事件链调整）
hd_dc_kit_adv_reform_chance_value = {
	value = hd_dc_adv_reform_base_chance
	if = {
		limit = {
			exists = var:hd_party_hanmen_share
			exists = var:hd_party_shizu_share
			var:hd_party_hanmen_share > var:hd_party_shizu_share
		}
		add = hd_dc_adv_reform_hanmen_bonus
	}
	if = {
		limit = {
			var:hd_dc_adv_reform_minister ?= {
				is_alive = yes
				stewardship >= hd_dc_adv_reform_minister_skill
			}
		}
		add = hd_dc_adv_reform_minister_bonus
	}
	if = {
		limit = { exists = var:hd_dc_adv_reform_bonus }
		add = var:hd_dc_adv_reform_bonus
	}
	min = 5
	max = 95
}

# Scope：辩论参与者。进取期额外威望 = 本次辩论所得威望 × 0.5（奖励前后差额）
hd_dc_kit_adv_debate_prestige_bonus_value = {
	value = 0
	if = {
		limit = { exists = var:hd_dc_adv_pre_prestige }
		add = prestige
		subtract = var:hd_dc_adv_pre_prestige
		multiply = hd_dc_adv_debate_bonus_mult
	}
	min = 0
}

# Scope：辩论参与者。进取期额外儒学经验 = 原版 debate_guest_rewards 公式 × 0.5
hd_dc_kit_adv_debate_xp_bonus_value = {
	value = lifestyle_confucian_education_xp_gain_minor_value
	if = {
		limit = {
			has_variable = tgp_debate_success_score
			var:tgp_debate_success_score >= 75
		}
		multiply = 3
	}
	else_if = {
		limit = {
			has_variable = tgp_debate_success_score
			var:tgp_debate_success_score >= 50
		}
		multiply = 1.5
	}
	else_if = {
		limit = { has_variable = tgp_debate_success_score }
		multiply = 0.8
	}
	multiply = hd_dc_adv_debate_bonus_mult
}
