# 天朝循环·时代特色：对峙期“策反”阴谋共用成功率修正（见 docs/天朝循环_时代特色重制计划.md §3.6）
# Scope：阴谋（scope:owner = 策反者，scope:target = 敌国封臣）

hd_dc_sta_subversion_success_chance_modifier = {
	# 策反者外交（同原版拉拢：每点 ×2）
	modifier = {
		add = {
			value = scope:owner.diplomacy
			multiply = 2
			floor = yes
		}
		desc = SCHEME_YOUR_DIPLOMACY
	}
	# 目标对其主君的好感（核心：越不满越易策反）
	modifier = {
		add = {
			value = "scope:target.opinion(scope:target.liege)"
			multiply = -0.5
			floor = yes
		}
		desc = hd_dc_sta_scheme_target_liege_opinion
	}
	# 目标对策反者的好感
	modifier = {
		add = {
			value = "scope:target.opinion(scope:owner)"
			multiply = 0.25
			floor = yes
		}
		desc = hd_dc_sta_scheme_target_owner_opinion
	}
	# 性格
	modifier = {
		add = 10
		desc = hd_dc_sta_scheme_target_ambitious
		scope:target = { has_trait = ambitious }
	}
	modifier = {
		add = -15
		desc = hd_dc_sta_scheme_target_content
		scope:target = { has_trait = content }
	}
	modifier = {
		add = -30
		desc = hd_dc_sta_scheme_target_loyal
		scope:target = { has_trait = loyal }
	}
	modifier = {
		add = 20
		desc = hd_dc_sta_scheme_target_disloyal
		scope:target = { has_trait = disloyal }
	}
	modifier = {
		add = 8
		desc = SCHEME_BEFRIEND_MY_LIFESTYLE
		scope:owner = { has_trait = diplomat }
	}
	# 与主君的关系
	modifier = {
		add = 20
		desc = hd_dc_sta_scheme_target_rival_of_liege
		scope:target = { has_relation_rival = scope:target.liege }
	}
	modifier = {
		add = -25
		desc = hd_dc_sta_scheme_target_liege_family
		scope:target = { is_close_family_of = scope:target.liege }
	}
	modifier = {
		add = -20
		desc = hd_dc_sta_scheme_target_liege_friend
		scope:target = {
			OR = {
				has_relation_friend = scope:target.liege
				has_relation_lover = scope:target.liege
			}
		}
	}
	# 主君握有其把柄
	modifier = {
		add = -15
		desc = hd_dc_sta_scheme_target_hooked
		scope:target.liege = { has_hook = scope:target }
	}
}

# Scope：策反互动的 ai_will_do（root = scope:actor = 策反者，scope:recipient = 敌国封臣）
hd_dc_sta_subversion_ai_will_do_modifier = {
	modifier = {
		add = 30
		scope:recipient = {
			opinion = {
				target = scope:recipient.liege
				value < 0
			}
		}
	}
	modifier = {
		add = 20
		scope:recipient = {
			opinion = {
				target = scope:recipient.liege
				value < -30
			}
		}
	}
	modifier = {
		add = 10
		diplomacy >= 12
	}
	modifier = {
		add = 15
		OR = {
			has_trait = deceitful
			has_trait = schemer
		}
	}
	modifier = {
		add = -20
		has_trait = honest
	}
	modifier = {
		add = 20
		scope:recipient.top_liege = {
			OR = {
				has_relation_rival = scope:actor
				has_relation_nemesis = scope:actor
			}
		}
	}
	modifier = {
		factor = 0
		scope:recipient = {
			opinion = {
				target = scope:recipient.liege
				value >= 30
			}
		}
	}
	modifier = {
		factor = 0
		scope:recipient = { has_trait = loyal }
	}
	modifier = {
		factor = 0
		scope:recipient.top_liege = { is_allied_to = scope:actor }
	}
	modifier = {
		factor = 0.5
		has_truce = scope:recipient.top_liege
	}
}
