# 地方官个人考期内向本朝现任尚书令请求；不调用荐举请求/答复流程。
wj_kaoke_request_interaction = {
 icon = scroll_scales
 category = interaction_category_friendly
 common_interaction = yes
 interface_priority = 25
 needs_recipient_to_open = yes
 name = wj_kaoke_request_interaction
 desc = wj_kaoke_request_interaction_desc
 send_name = wj_kaoke_request_interaction_send
 greeting = positive
 notification_text = WJ_kaoke_request_notification
 # AI走与荐举相同的原生ai_accept显示及答复，立即处理、不掷随机成功率。
 ai_maybe = no
 ai_instant_response = yes
 # 玩家尚书令保留既定“符合条件即可成功”，不新增手动审批。
 auto_accept = { scope:recipient = { is_ai = no } }
 # 与原版请求军事援助相同：皇帝是入口，实际收件人仍为本朝现任尚书令。
 redirect = {
  scope:recipient = {
   if = {
    limit = { this = scope:actor.top_liege }
    WJ_kaoke_find_reviewer = yes
   }
  }
 }
 is_shown = {
  scope:actor = { is_ai = no }
  OR = {
   WJ_kaoke_request_participants_trigger = yes
   AND = {
    exists = scope:actor
    exists = scope:recipient
    scope:actor = {
     WJ_kaoke_eligible_trigger = yes
     WJ_kaoke_table_viewer_trigger = yes
     this != scope:recipient
     top_liege = scope:recipient
    }
   }
  }
 }
 is_valid_showing_failures_only = {
    WJ_kaoke_request_participants_trigger = yes
 }
 can_send = { WJ_kaoke_request_can_send_trigger = yes }
 options_heading = WJ_kaoke_request_options
 send_options_exclusive = no
 send_option = {
  flag = wj_kaoke_request_20
  localization = WJ_kaoke_request_20
  can_invalidate_interaction = yes
  is_valid = { scope:wj_kaoke_request_30 = no scope:wj_kaoke_request_50 = no }
 }
 send_option = {
  flag = wj_kaoke_request_30
  localization = WJ_kaoke_request_30
  can_invalidate_interaction = yes
  is_valid = { scope:wj_kaoke_request_20 = no scope:wj_kaoke_request_50 = no }
 }
 send_option = {
  flag = wj_kaoke_request_50
  localization = WJ_kaoke_request_50
  can_invalidate_interaction = yes
  is_valid = { scope:wj_kaoke_request_20 = no scope:wj_kaoke_request_30 = no }
 }
 send_option = {
  flag = gift
  localization = WJ_REQUEST_RECOMMENDATION_GOLD_OPTION
  can_invalidate_interaction = yes
  is_valid = { WJ_kaoke_request_gift_valid_trigger = yes }
  current_description = { triggered_desc = { trigger = { WJ_kaoke_request_gift_valid_trigger = yes } desc = WJ_REQUEST_RECOMMENDATION_GOLD_VALID } }
 }
 send_option = {
  flag = offer_hook
  localization = SCHEME_AGENT_HOOK
  can_invalidate_interaction = yes
  is_valid = { WJ_kaoke_request_offer_hook_valid_trigger = yes }
 }
 send_option = {
  flag = offer_hook_strong
  localization = SCHEME_AGENT_HOOK_STRONG
  can_invalidate_interaction = yes
  is_valid = { WJ_kaoke_request_offer_hook_strong_valid_trigger = yes }
  is_shown = { scope:actor = { has_trait_with_flag = can_offer_strong_hook_to_agents } }
 }
 send_option = {
  flag = hook
  localization = SCHEME_HOOK
  can_invalidate_interaction = yes
  is_valid = { WJ_kaoke_request_hook_valid_trigger = yes }
 }
 send_option = {
  flag = prestige
  localization = SCHEME_AGENT_PRESTIGE
  can_invalidate_interaction = yes
  is_valid = { WJ_kaoke_request_prestige_valid_trigger = yes }
  current_description = { triggered_desc = { trigger = { WJ_kaoke_request_prestige_valid_trigger = yes } desc = SCHEME_AGENT_PRESTIGE_VALID } }
 }
 send_option = {
  flag = influence
  localization = SCHEME_AGENT_INFLUENCE
  can_invalidate_interaction = yes
  is_valid = { WJ_kaoke_request_influence_valid_trigger = yes }
  is_shown = { WJ_kaoke_request_influence_access_trigger = yes }
  current_description = { triggered_desc = { trigger = { WJ_kaoke_request_influence_valid_trigger = yes } desc = SCHEME_AGENT_INFLUENCE_VALID } }
 }
 on_send = {
  if = {
   limit = { WJ_kaoke_request_can_send_trigger = yes }
   scope:actor = {
    save_scope_value_as = { name = wj_kaoke_request_epoch value = var:wj_kaoke_window_epoch }
    save_scope_value_as = { name = wj_kaoke_request_start value = var:wj_kaoke_start }
    var:wj_kaoke_realm = { save_scope_as = wj_kaoke_request_realm }
    top_liege = { save_scope_as = wj_kaoke_request_emperor }
    liege = { save_scope_as = wj_kaoke_request_liege }
   }
  }
 }
 on_accept = {
  # 原版adopt_interaction同样将预览与带最终校验的实际结算分开。
  # 预览发生在on_send之前，不能要求其拥有发送后才捕获的考期快照。
  show_as_tooltip = {
   WJ_settle_recommendation_movement_options_effect = yes
   scope:actor = {
    if = { limit = { scope:wj_kaoke_request_20 = yes } custom_tooltip = WJ_kaoke_request_gain_20_tt }
    if = { limit = { scope:wj_kaoke_request_30 = yes } custom_tooltip = WJ_kaoke_request_gain_30_tt }
    if = { limit = { scope:wj_kaoke_request_50 = yes } custom_tooltip = WJ_kaoke_request_gain_50_tt }
    custom_tooltip = WJ_kaoke_request_secret_known_tt
   }
   scope:recipient = { custom_tooltip = WJ_kaoke_request_secret_owner_tt }
  }
  hidden_effect = { WJ_kaoke_request_success_effect = yes }
 }
 ai_accept = {
  WJ_dd_jie_acceptance_modifier = yes
  base = 0
  modifier = { NOT = { WJ_kaoke_request_personality_value = 0 } add = WJ_kaoke_request_personality_value desc = WJ_ai_kaoke_personality }
  modifier = { add = WJ_kaoke_request_burden_value desc = WJ_kaoke_request_burden_reason }
  modifier = { scope:gift = yes add = WJ_kaoke_request_gift_willingness_value desc = WJ_REQUEST_RECOMMENDATION_GOLD_REASON }
  modifier = { scope:offer_hook = yes add = WJ_kaoke_request_offer_hook_willingness_value desc = SCHEME_WITH_OFFER_HOOK }
  modifier = { scope:offer_hook_strong = yes add = WJ_kaoke_request_offer_hook_strong_willingness_value desc = SCHEME_WITH_OFFER_HOOK_STRONG }
  modifier = { scope:hook = yes add = WJ_kaoke_request_hook_willingness_value desc = SCHEME_WEAK_HOOK_USED }
  modifier = { scope:prestige = yes add = WJ_kaoke_request_prestige_willingness_value desc = SCHEME_WITH_PRESTIGE }
  modifier = { scope:influence = yes add = WJ_kaoke_request_influence_willingness_value desc = SCHEME_WITH_INFLUENCE }
 }
}
