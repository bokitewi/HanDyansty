# 天朝循环·时代特色：紧张期套件（docs/天朝循环_时代特色重制计划.md §3.3）互动
# 弹劾（发起阴谋 hd_dc_ten_impeach）；勒兵入朝·清君侧（封臣指名权臣，对天子宣战）

hd_dc_ten_start_impeach_interaction = {
	icon = icon_scheme_slander
	interface_priority = 90
	common_interaction = no
	category = interaction_category_hostile
	filter_tags = { dynastic_cycle }

	send_name = START_SCHEME

	scheme = hd_dc_ten_impeach
	ignores_pending_interaction_block = yes

	ai_targets = {
		ai_recipients = peer_vassals
		max = 10
	}
	ai_targets = {
		ai_recipients = scripted_relations
		max = 5
	}
	ai_frequency = 36

	is_available = {
		hd_dc_kit_ten_can_impeach_trigger = yes
		trigger_if = {
			limit = { is_ai = yes }
			hd_dc_is_vip_trigger = yes
			NOT = {
				any_scheme = {
					is_scheme_category = political
				}
			}
		}
	}

	is_shown = {
		hd_dc_phase_is_trigger = { PHASE = instability }
		scope:actor = { hd_dc_kit_ten_can_impeach_trigger = yes }
		scope:recipient = {
			this != scope:actor
			hd_dc_kit_ten_is_power_minister_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = hd_dc_ten_impeach_vip_tt
				hd_dc_is_vip_trigger = yes
			}
			can_start_scheme = {
				type = hd_dc_ten_impeach
				target_character = scope:recipient
			}
		}
		scope:recipient = {
			is_imprisoned = no
			NOT = { has_strong_hook = scope:actor }
		}
	}

	desc = hd_dc_ten_start_impeach_interaction_desc

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = start_hd_dc_ten_impeach_notification
				left_icon = scope:actor
				right_icon = scope:recipient
				begin_scheme_with_agents_effect = {
					SCHEME_TYPE = hd_dc_ten_impeach
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
					AGENT_1 = agent_intermediary
					AGENT_2 = agent_bureaucrat
					AGENT_3 = agent_political_socialite
					AGENT_4 = agent_amanuensis
					AGENT_5 = agent_courtesan
				}
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
		modifier = {
			add = 15
			scope:actor = {
				opinion = {
					target = scope:recipient
					value <= -25
				}
			}
		}
		modifier = {
			add = 25
			scope:actor = { has_relation_rival = scope:recipient }
		}
		modifier = {
			add = 10
			scope:actor = {
				OR = {
					has_trait = ambitious
					has_trait = deceitful
					has_trait = arbitrary
					has_trait = vengeful
				}
			}
		}
		modifier = {
			add = 10
			scope:actor = { tgp_has_minister_title = yes }	# 言官 / 同僚互讦
		}
		modifier = {
			factor = 0
			scope:actor = {
				OR = {
					has_trait = content
					has_trait = honest
				}
			}
		}
		modifier = {
			factor = 0
			scope:actor = {
				opinion = {
					target = scope:recipient
					value >= 0
				}
			}
		}
	}
}

hd_dc_ten_qingjunce_interaction = {
	icon = icon_declare_me_regent
	interface_priority = 85
	common_interaction = no
	category = interaction_category_hostile
	filter_tags = { dynastic_cycle }

	ai_targets = {
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 60

	is_available = {
		hd_dc_kit_ten_can_qingjunce_trigger = yes
		trigger_if = {
			limit = { is_ai = yes }
			highest_held_title_tier >= tier_duchy
			is_at_war = no
		}
	}

	is_shown = {
		hd_dc_phase_is_trigger = { PHASE = instability }
		scope:actor = { hd_dc_kit_ten_can_qingjunce_trigger = yes }
		scope:recipient = {
			this != scope:actor
			hd_dc_kit_ten_is_power_minister_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		scope:actor = {
			custom_tooltip = {
				text = hd_dc_ten_qjc_peace_tt
				NOT = { is_at_war_with = title:h_china.holder }
			}
		}
	}

	desc = hd_dc_ten_qingjunce_interaction_desc

	on_accept = {
		scope:actor = {
			custom_tooltip = hd_dc_ten_qingjunce_interaction_tt
			hidden_effect = {
				hd_dc_kit_ten_start_qingjunce_effect = { TARGET = scope:recipient }
			}
		}
	}

	auto_accept = yes

	# 勒兵入朝是大事：只在兵强、恨权臣而不恨天子时偶尔为之
	ai_will_do = {
		base = 0
		modifier = {
			add = 10
			scope:actor = {
				max_military_strength >= title:h_china.holder.hd_dc_kit_half_strength_value
			}
		}
		modifier = {
			add = 10
			scope:actor = {
				opinion = {
					target = scope:recipient
					value <= -50
				}
			}
		}
		modifier = {
			add = 5
			scope:actor = {
				OR = {
					has_trait = ambitious
					has_trait = wrathful
					has_trait = brave
				}
			}
		}
		modifier = {
			factor = 0
			scope:actor = {
				max_military_strength < title:h_china.holder.hd_dc_kit_half_strength_value
			}
		}
		modifier = {
			factor = 0
			scope:actor = {
				opinion = {
					target = scope:recipient
					value > -50
				}
			}
		}
	}
}
