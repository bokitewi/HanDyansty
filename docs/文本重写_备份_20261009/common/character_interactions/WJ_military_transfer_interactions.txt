# Card-menu selection only: native filter_tags / needs_confirmation, no army or money changes.
WJ_mt_pick_title_source = {
 category = interaction_category_wj_military_transfer
 icon = icon_soldier_survivor
 filter_tags = { wj_district_management wj_military_transfer }
 common_interaction = yes interface_priority = 100
 use_diplomatic_range = no
 needs_confirmation = { always = no }
 auto_accept = yes
 is_available = { is_ai = no government_has_flag = government_is_celestial }
 # Stable eligibility; the native GUI limits these rows to filtered card menus.
 is_shown = { scope:recipient = { is_alive = yes } }
is_valid_showing_failures_only = {
  scope:actor = {
   WJ_mt_authority = yes
   custom_tooltip = { text = WJ_mt_already_picked_tt NOT = { AND = { has_variable = wj_mt_title_source var:wj_mt_title_source = scope:recipient } } }
  }
  scope:recipient = {
   WJ_mt_person_allowed = { OP = scope:actor }
   custom_tooltip = { text = WJ_mt_title_tier_tt highest_held_title_tier >= tier_duchy }
  }
 }
 on_accept = {
  scope:actor = {
   if = { limit = {
    WJ_mt_authority = yes
    NOT = { AND = { has_variable = wj_mt_title_source var:wj_mt_title_source = scope:recipient } }
    scope:recipient = { WJ_mt_person_allowed = { OP = scope:actor } highest_held_title_tier >= tier_duchy }
   }
    if = { limit = { has_variable = wj_mt_open } WJ_mt_clear_draft = yes remove_variable = wj_mt_mode }
    if = { limit = { has_variable = wj_mt_title_receiver var:wj_mt_title_receiver = scope:recipient } remove_variable = wj_mt_title_receiver }
     set_variable = { name = wj_mt_title_source value = scope:recipient }
   }
  }
 }
 ai_will_do = { base = 0 }
}
