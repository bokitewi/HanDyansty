# 天朝循环大修·朋党：立场辩论（草案 4.4）
# 发起人：朋党领袖或前三名；目标：当前立场以外的四种立场之一。
# 阶段：准备（30 天）→ 辩论初期（15 天）→ 辩论后期（15 天）→ 结局（后期结束时结算）。
# 进度：scope:activity.var:hd_debate_progress（−100 ~ 100），结局时加势力修正，≥ +30 成功。
activity_hd_party_debate = {
	is_shown = {
		hd_party_can_start_debate_trigger = yes
	}

	can_start_showing_failures_only = {
		is_available_adult = yes
		prestige >= hd_party_debate_prestige_cost
	}

	can_always_plan = no

	is_valid = {
		scope:host = {
			is_physically_able = yes
		}
	}

	on_invalidated = {
		scope:host = { hd_party_debate_cleanup_effect = yes }
	}

	on_host_death = {
		scope:host = { hd_party_debate_cleanup_effect = yes }
	}

	province_description = {
		desc = hd_party_debate_province_desc
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { has_variable = hd_party_ai_debate_wish }
			add = 100
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	ai_will_select_province = {
		add = 5
	}

	is_single_location = yes
	cooldown = { years = 1 }
	max_province_icons = 5
	max_route_deviation_mult = 3.0
	province_filter = top_realm
	ai_province_filter = top_realm

	is_location_valid = {
		custom_tooltip = {
			text = hd_party_debate_location_desc
			OR = {
				this = scope:host.capital_province
				this = scope:host.var:hd_party_my_court.capital_province
			}
		}
	}

	province_score = {
		value = 0
		if = {
			limit = { this = scope:host.var:hd_party_my_court.capital_province }
			add = 15
		}
	}

	cost = {
		gold = {
			add = hd_party_debate_gold_cost
		}
		prestige = {
			add = hd_party_debate_prestige_cost
		}
	}
	ui_predicted_cost = {
		gold = {
			add = hd_party_debate_gold_cost
		}
		prestige = {
			add = hd_party_debate_prestige_cost
		}
	}

	conclusion_description = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:activity.var:hd_debate_result ?= flag:success }
				desc = hd_party_debate_conclusion_success
			}
			triggered_desc = {
				trigger = { scope:activity.var:hd_debate_result ?= flag:fail }
				desc = hd_party_debate_conclusion_fail
			}
			desc = hd_party_debate_conclusion_default
		}
	}

	options = {
		hd_debate_target = {
			hd_target_zunwang = {
				is_valid = { NOT = { var:hd_party_my_stance ?= flag:zunwang } }
				ai_will_do = { value = hd_party_stance_weight_zunwang_value }
			}
			hd_target_kaituo = {
				is_valid = { NOT = { var:hd_party_my_stance ?= flag:kaituo } }
				ai_will_do = { value = hd_party_stance_weight_kaituo_value }
			}
			hd_target_jinqu = {
				is_valid = { NOT = { var:hd_party_my_stance ?= flag:jinqu } }
				ai_will_do = { value = hd_party_stance_weight_jinqu_value }
			}
			hd_target_shoujiu = {
				is_valid = { NOT = { var:hd_party_my_stance ?= flag:shoujiu } }
				ai_will_do = { value = hd_party_stance_weight_shoujiu_value }
			}
			hd_target_zhongli = {
				is_valid = { NOT = { var:hd_party_my_stance ?= flag:zhongli } }
				ai_will_do = { value = hd_party_stance_weight_zhongli_value }
			}
		}
	}

	special_option_category = hd_debate_target

	phases = {
		hd_debate_phase_prep = {
			order = 1
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = {
						progress_activity_phase_after = { days = 30 }
					}
					hd_party_debate_sort_guests_effect = yes
					trigger_event = { id = hd_party_debate.0010 days = 3 }
				}
				else = {
					hd_party_debate_guest_event_effect = yes
				}
			}
		}
		hd_debate_phase_early = {
			order = 2
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = {
						progress_activity_phase_after = { days = 15 }
					}
					hd_party_debate_sort_guests_effect = yes
					hd_party_debate_early_events_effect = yes
				}
				else = {
					hd_party_debate_guest_event_effect = yes
				}
			}
		}
		hd_debate_phase_late = {
			order = 3
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = {
						progress_activity_phase_after = { days = 15 }
					}
					hd_party_debate_sort_guests_effect = yes
					hd_party_debate_late_events_effect = yes
					# 请圣裁：君主无论是否出席都会收到
					var:hd_party_my_court ?= {
						set_variable = { name = hd_party_debate_activity value = scope:activity days = 30 }
						trigger_event = { id = hd_party_debate.0050 days = 5 }
					}
				}
				else = {
					hd_party_debate_guest_event_effect = yes
				}
			}

			on_end = {
				if = {
					limit = { this = scope:host }
					hd_party_debate_resolve_effect = yes
				}
			}
		}
	}

	guest_subsets = {
		hd_debate_support
		hd_debate_neutral
		hd_debate_oppose
	}

	open_invite = yes

	max_guests = {
		value = 60
	}

	# 本党成员、本朝廷其他朋党的领袖、君主
	can_be_activity_guest = {
		is_available_adult = yes
		in_diplomatic_range = scope:host
		exists = scope:host.var:hd_party_my_court
		OR = {
			this = scope:host.var:hd_party_my_court
			AND = {
				var:hd_party_my_court ?= scope:host.var:hd_party_my_court
				var:hd_party_my_group ?= scope:host.var:hd_party_my_group
			}
			AND = {
				var:hd_party_my_court ?= scope:host.var:hd_party_my_court
				hd_party_is_any_leader_of_court_trigger = { COURT = scope:host.var:hd_party_my_court }
			}
		}
	}

	host_intents = {
		intents = { hd_debate_host_intent }
		default = hd_debate_host_intent
	}

	guest_intents = {
		intents = { hd_debate_support_intent hd_debate_neutral_intent hd_debate_oppose_intent }
		default = hd_debate_neutral_intent
	}

	guest_join_chance = {
		base = 10
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
		activity_guest_health_ai_accept_modifier = yes
		modifier = {
			add = 20
			var:hd_party_my_group ?= scope:host.var:hd_party_my_group
		}
	}

	wait_time_before_start = {
		months = 1
	}

	max_guest_arrival_delay_time = { months = 1 }

	guest_invite_rules = {
		defaults = {
			1 = activity_invite_rule_close_family
		}
	}

	on_start = {
		hd_party_debate_start_effect = yes
	}

	on_complete = {
	}

	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_debate.dds"

	background = {
		texture = "gfx/interface/illustrations/event_scenes/tgp_study_asia.dds"
		environment = "environment_tgp_study_asia"
		ambience = "event:/DLC/EP4/SFX/Events/Event_Backgrounds/tgp_study_asia"
	}

	activity_window_widgets = {
		hd_party_debate_widget = "window_activity_contestants_right_bar_container"
	}

	window_characters = {
		host = {
			camera = camera_body
			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}
			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { debating storyteller }
				}
			}
		}
		guest = {
			camera = camera_body_right
			effect = {
				ordered_attending_character = {
					limit = {
						location = scope:activity.activity_location
						NOT = { this = scope:host }
					}
					order_by = hd_party_debate_sort_value
					add_to_list = characters
				}
			}
			scripted_animation = {
				triggered_animation = {
					trigger = { always = yes }
					animation = { interested personality_honorable }
				}
			}
		}
	}
}
