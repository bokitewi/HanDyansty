hd4_guandu.0002 = {
	type = character_event
	title = hd4_guandu_invite.t
	desc = hd4_guandu_invite.desc
	theme = war
	trigger = {
		this = character:cao_cao
		is_alive = yes
		title:h_china = {
			has_variable = hd4_northern_two_announced
			NOT = {
				has_variable = hd4_guandu_done
			}
		}
		AND = {
			exists = character:yuan_shao
			character:yuan_shao = {
				is_alive = yes
			}
		}
		NOT = {
			has_character_flag = hd4_guandu_authorized
		}
	}
	on_trigger_fail = {
		remove_character_flag = hd4_guandu_invite_lease
	}
	option = {
		name = hd4_guandu_accept
		if = {
			limit = {
				NOT = {
					title:h_china = {
						has_variable = hd4_guandu_done
					}
				}
			}
			add_character_flag = hd4_guandu_authorized
			remove_character_flag = hd4_guandu_invite_lease
			hd4_guandu_prepare_effect = yes
		}
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = hd4_wait
		add_character_flag = {
			flag = hd4_guandu_invite_lease
			days = 180
		}
		ai_chance = {
			base = 0
		}
	}
}
