namespace = WJ_rf_ministry

# One delayed qualification check; underage heirs are refused, not deferred.
WJ_rf_ministry.1 = {
 type = character_event hidden = yes
 trigger = { is_alive = yes exists = scope:wj_rf_inherited_office exists = scope:wj_rf_inherited_court }
 immediate = {
  clear_saved_scope = councillor_liege
  scope:wj_rf_inherited_court.holder ?= { save_scope_as = councillor_liege }
  if = { limit = { WJ_rf_ministry_can_restore = yes } trigger_event = WJ_rf_ministry.2 }
  else = { trigger_event = WJ_rf_ministry.3 }
 }
}
WJ_rf_ministry.2 = {
 type = character_event title = WJ_rf_ministry_grant desc = WJ_rf_ministry_grant_desc theme = administrative
 left_portrait = { character = root }
 right_portrait = { character = scope:councillor_liege }
 option = {
  name = WJ_rf_ministry_accept
  if = { limit = { WJ_rf_ministry_can_restore = yes } WJ_rf_restore_ministry = yes }
  else = { trigger_event = WJ_rf_ministry.3 }
 }
}
WJ_rf_ministry.3 = {
 type = character_event title = WJ_rf_ministry_refused desc = WJ_rf_ministry_refused_desc theme = administrative
 left_portrait = { character = root }
 option = { name = WJ_rf_ministry_refused_accept }
}
# No visible puppet event and no special type; player emperors skip this effect.
WJ_rf_ministry.4 = {
 type = character_event hidden = yes
 trigger = { is_alive = yes exists = scope:councillor_liege exists = scope:wj_rf_inherited_office scope:wj_rf_inherited_office = { holder = root } }
 immediate = { WJ_rf_ministry_after_grant = yes }
}
