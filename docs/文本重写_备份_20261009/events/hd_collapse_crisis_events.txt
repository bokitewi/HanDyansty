# 集体独立
############################################################
hd_collapse.1001 = {
	type = character_event
	title = hd_collapse.1001.t
	desc = hd_collapse.1001.desc
	theme = realm
	left_portrait = { character = root animation = worry }
	trigger = { hd_collapse_emperor_still_valid_trigger = yes }

	immediate = {
		save_scope_value_as = {
			name = hd_collapse_rebel_strength
			value = hd_collapse_independence_rebel_strength_value
		}
	}

	option = {	# 允其自立
		name = hd_collapse.1001.a
		hd_collapse_accept_independence_effect = yes
		ai_chance = {
			base = 20
			modifier = {
				add = 50
				scope:hd_collapse_rebel_strength > root.current_military_strength
			}
		}
	}
	option = {	# 严词拒绝
		name = hd_collapse.1001.b
		hd_collapse_refuse_independence_effect = yes
		ai_chance = {
			base = 80
			modifier = {
				add = -50
				scope:hd_collapse_rebel_strength > root.current_military_strength
			}
		}
	}
}
