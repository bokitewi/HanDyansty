WJ_rf.4 = {
 type = character_event hidden = yes
 immediate = {
  if = { limit = { exists = scope:wj_rf_law_title scope:wj_rf_law_title = { has_variable = wj_rf_rank holder = root } }
   scope:wj_rf_law_title = {
    set_destroy_if_invalid_heir = yes
    clear_title_laws = yes
    add_title_law = noble_family_succession_law
   }
   if = { limit = { exists = scope:wj_rf_capital_county scope:wj_rf_capital_county = { has_variable = wj_rf_capital_of holder = root } }
    scope:wj_rf_capital_county = {
     clear_title_laws = yes
     add_title_law = noble_family_succession_law
     }
   }
  }
 }
