namespace = WJ_yangwang_timeout

# Scheduled at the existing timeout's expiry, not at a fixed polling interval.
# If extended, follow its new expiry without taking ownership of that source.
WJ_yangwang_timeout.0001 = {
 type = character_event
 hidden = yes
 immediate = {
  if = {
   limit = {
    is_alive = yes
    has_trait = WJ_yangwang_trait
    appointment_timeout_days = 0
   }
   set_appointment_timeout = { years = 150 desc = appointment_timeout_desc_refusing }
   add_character_flag = wj_yangwang_timeout_owned
  }
  else_if = {
   limit = {
    is_alive = yes
    has_trait = WJ_yangwang_trait
    appointment_timeout_days > 0
   }
   trigger_event = { id = WJ_yangwang_timeout.0001 days = WJ_yangwang_timeout_handoff_days }
  }
 }
}
