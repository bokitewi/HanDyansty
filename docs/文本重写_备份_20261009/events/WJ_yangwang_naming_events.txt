 type = character_event
 title = WJ_yangwang.naming.title
 desc = WJ_yangwang.custom.desc
 theme = learning
 trigger = { is_ai = no WJ_yinshi_naming_ok_trigger = yes NOT = { var:wj_yangwang_custom_slot = 1 } }
 immediate = { remove_variable = wj_yangwang_text_1 remove_localized_text = wj_yangwang_text_1 }
 widget = {
  gui = "event_window_widget_enter_text"
  container = "dynamic_birth_name"
  controller = { type = text data = { key = wj_yangwang_text_1 } }
  setup_scope = { root = { save_scope_as = text_target } }
 }

  name = CONFIRM
  set_variable = { name = wj_yangwang_custom_slot value = 1 }
  store_localized_text_in_death = wj_yangwang_text_1
  WJ_yangwang_confirm_name_effect = yes
 }
 option = { name = CANCEL is_cancel_option = yes }
}
