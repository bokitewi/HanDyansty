# 大赦天下：二选一。Scope：天子
############################################################
hd_dc_rec.0001 = {
	type = character_event
	title = hd_dc_rec.0001.t
	desc = hd_dc_rec.0001.desc
	theme = court
	left_portrait = {
		character = root
		animation = personality_honorable
	}

	option = {	# 只赦天子之囚
		name = hd_dc_rec.0001.a
		trigger = {
			any_prisoner = { always = yes }
		}
		hd_dc_rec_release_own_prisoners_effect = yes
		hd_dc_rec_amnesty_vassal_opinion_effect = yes
		if = {
			limit = { is_valid_for_legitimacy_change = yes }
			add_legitimacy = medium_legitimacy_gain
		}
		hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_reconciliation ACTOR = root }
		ai_chance = {
			base = 60
			modifier = {
				add = 20
				has_trait = arbitrary
			}
			modifier = {
				add = 20
				has_trait = paranoid
			}
		}
	}

	option = {	# 连封臣之囚一并大赦
		name = hd_dc_rec.0001.b
		trigger = {
			any_vassal_or_below = {
				any_prisoner = { always = yes }
			}
		}
		hd_dc_kit_fire_effect = { CATALYST = catalyst_hd_grand_amnesty_full ACTOR = root }
		hd_dc_rec_release_own_prisoners_effect = yes
		hd_dc_rec_release_vassal_prisoners_effect = yes
		hd_dc_rec_amnesty_vassal_opinion_effect = yes
		if = {
			limit = { is_valid_for_legitimacy_change = yes }
			add_legitimacy = major_legitimacy_gain
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = forgiving
			}
			modifier = {
				add = 20
				has_trait = just
			}
			modifier = {
				add = 20
				has_trait = compassionate
			}
		}
	}

	option = {	# 收回成命（两项皆不可用时的兜底）
		name = hd_dc_rec.0001.c
		trigger = {
			NOR = {
				any_prisoner = { always = yes }
				any_vassal_or_below = {
					any_prisoner = { always = yes }
				}
			}
		}
		ai_chance = { base = 1 }
	}
}

# 求贤诏：1~3 位人才应诏入朝。Scope：天子
############################################################
hd_dc_rec.0002 = {
	type = character_event
	title = hd_dc_rec.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:hd_dc_rec_talent_count >= 3 }
				desc = hd_dc_rec.0002.desc_3
			}
			triggered_desc = {
				trigger = { scope:hd_dc_rec_talent_count >= 2 }
				desc = hd_dc_rec.0002.desc_2
			}
			desc = hd_dc_rec.0002.desc_1
		}
	}
	theme = court
	left_portrait = {
		character = root
		animation = happiness
	}
	right_portrait = {
		trigger = { exists = scope:hd_dc_rec_talent_1 }
		character = scope:hd_dc_rec_talent_1
		animation = personality_rational
	}
	lower_left_portrait = {
		trigger = { exists = scope:hd_dc_rec_talent_2 }
		character = scope:hd_dc_rec_talent_2
	}
	lower_right_portrait = {
		trigger = { exists = scope:hd_dc_rec_talent_3 }
		character = scope:hd_dc_rec_talent_3
	}

	immediate = {
		random_list = {
			1 = { save_scope_value_as = { name = hd_dc_rec_talent_count value = 1 } }
			1 = { save_scope_value_as = { name = hd_dc_rec_talent_count value = 2 } }
			1 = { save_scope_value_as = { name = hd_dc_rec_talent_count value = 3 } }
		}
		hd_dc_rec_create_talent_effect = { N = 1 }
		if = {
			limit = { scope:hd_dc_rec_talent_count >= 2 }
			hd_dc_rec_create_talent_effect = { N = 2 }
		}
		if = {
			limit = { scope:hd_dc_rec_talent_count >= 3 }
			hd_dc_rec_create_talent_effect = { N = 3 }
		}
	}

	option = {
		name = hd_dc_rec.0002.a
	}
}
