namespace = wj_temp_official

# 一年到期：玩家开放考课决议，NPC 按相同三轮规则直接模拟结算。
wj_temp_official.0100 = {
	trigger = { government_has_flag = government_is_celestial WJ_kaoke_eligible_trigger = no }
	hidden = yes
	immediate = {
		top_liege = { save_scope_as = wj_assessment_emperor }
		if = {
			limit = {
				has_character_modifier = WJ_temp_official_modifier
				WJ_is_valid_temp_official_trigger = yes
			}
			remove_character_flag ?= wj_temp_official_timer
			if = {
				limit = { is_ai = yes }
				# AI 不进三轮事件链，直接逐轮掷骰模拟并结算。
				WJ_temp_official_ai_simulate_effect = yes
			}
			else = {
				add_character_flag = wj_temp_official_kaoke_ready
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = wj_temp_official_kaoke_ready_toast
					left_icon = root
					right_icon = scope:wj_assessment_emperor
				}
			}
		}
		else = {
			WJ_temp_official_clear_effect = yes
		}
	}
}

wj_temp_official.1000 = {
	trigger = { WJ_is_valid_temp_official_trigger = yes WJ_kaoke_eligible_trigger = no }
	type = character_event
	title = wj_temp_official.1000.t
	desc = wj_temp_official.1000.desc
	theme = education
	override_background = { reference = study }

	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:wj_assessment_examiner
		animation = personality_rational
	}

	immediate = {
		WJ_set_up_temp_official_examiner_effect = yes
		set_variable = {
			name = wj_official_assessment_score
			value = WJ_official_assessment_score_value
		}
	}

	widget = { gui = "WJ_official_assessment_score" container = "custom_widgets_container" }

	option = {
		name = wj_temp_official.1000.a
		duel = {
			skill = stewardship
			value = 1
			50 = {
				desc = WJ_temp_official_round_success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				change_variable = {
					name = wj_assessment_round_bonus
					add = 15
				}
				custom_tooltip = WJ_assessment_round_1_success_tt
			}
			50 = {
				desc = WJ_temp_official_round_failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				stress_impact = { base = medium_stress_impact_gain }
			}
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_boldness = 1
				ai_energy = 1
			}
			modifier = { add = 30 stewardship > medium_skill_rating }
		}
	}
	option = {
		name = wj_temp_official.1000.b
		change_variable = {
			name = wj_assessment_safe_answers
			add = 1
		}
		custom_tooltip = WJ_assessment_safe_answer_tt
		add_prestige = miniscule_prestige_loss
		ai_chance = {
			base = 100
			modifier = {
				add = 40
				OR = {
					has_trait = content
					has_trait = lazy
					has_trait = craven
				}
			}
		}
	}

	after = {
		set_variable = {
			name = wj_official_assessment_score
			value = WJ_official_assessment_score_value
		}
		trigger_event = wj_temp_official.1001
	}
}

wj_temp_official.1005 = {
	trigger = { WJ_is_valid_temp_official_trigger = yes WJ_kaoke_eligible_trigger = no }
	type = character_event
	title = wj_temp_official.1005.t
	desc = {
		desc = wj_temp_official.1005.desc
		first_valid = {
			triggered_desc = {
				trigger = { var:wj_official_assessment_score >= 90 }
				desc = wj_temp_official.1005.desc_high
			}
			triggered_desc = {
				trigger = { var:wj_official_assessment_score >= 60 }
				desc = wj_temp_official.1005.desc_good
			}
			triggered_desc = {
				trigger = { var:wj_official_assessment_score >= 40 }
				desc = wj_temp_official.1005.desc_pass
			}
			desc = wj_temp_official.1005.desc_fail
		}
	}
	theme = education
	override_background = { reference = study }

	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { var:wj_official_assessment_score >= 40 }
			animation = happiness
		}
		animation = disappointed
	}
	right_portrait = {
		character = scope:wj_assessment_examiner
		animation = personality_rational
	}

	immediate = {
		if = {
			limit = {
				var:wj_official_assessment_score >= 40
				WJ_is_valid_temp_official_trigger = yes
			}
			WJ_temp_official_kaoke_pass_effect = yes
		}
		else = {
			WJ_temp_official_kaoke_fail_effect = yes
		}
	}

	widget = { gui = "WJ_official_assessment_score" container = "custom_widgets_container" }

	option = {
		name = wj_temp_official.1005.a
		trigger = {
			var:wj_official_assessment_score >= 40
			WJ_has_entry_road_trigger = yes
		}
		ai_chance = { base = 100 }
	}
	option = {
		name = wj_temp_official.1005.b
		trigger = {
			OR = {
				var:wj_official_assessment_score < 40
				WJ_has_entry_road_trigger = no
			}
		}
		ai_chance = { base = 100 }
	}

	after = {
		if = {
			limit = { WJ_has_entry_road_trigger = no }
			WJ_official_assessment_cleanup_effect = yes
			if = {
				limit = { WJ_is_valid_temp_official_trigger = yes }
				WJ_temp_official_retry_effect = yes
			}
			else = {
				WJ_temp_official_clear_effect = yes
			}
		}
		else = {
			WJ_official_assessment_cleanup_effect = yes
		}
	}
}
