namespace = hd_cao_chain
hd_cao_chain.0001 = {
	type = character_event
	title = hd_cao_chain.0001.t
	desc = hd_cao_chain.0001.desc
	theme = intrigue
	trigger = {
		this = character:cao_cao
		is_alive = yes
		NOT = {
			has_character_flag = hd_cao_early_story_completed
		}
		OR = {
			NOT = {
				has_variable = hd4_cao_stage
			}
			var:hd4_cao_stage = 1
		}
		OR = {
			has_character_flag = hd4_cao_early_requested
			has_character_flag = hd_cao_early_story_pending
			has_character_flag = hd_cao_early_story_started
		}
	}
	on_trigger_fail = {
		if = {
			limit = {
				this = character:cao_cao
				is_alive = yes
				has_variable = hd4_cao_stage
				var:hd4_cao_stage = 1
			}
			remove_character_flag = hd4_cao_stage_wait
		}
	}
	immediate = {
		remove_character_flag = hd_cao_early_story_pending
		add_character_flag = hd_cao_early_story_started
		set_variable = {
			name = hd_cao_early_story_phase
			value = flag:hd_cao_phase_prepare
		}
		character:wu_fu ?= {
			if = {
				limit = {
					is_alive = yes
				}
				death = {
					death_reason = death_execution
				}
			}
		}
		set_variable = {
			name = hd4_cao_stage
			value = 2
		}
		add_character_flag = {
			flag = hd4_cao_stage_wait
			days = 30
		}
		trigger_event = {
			id = hd_cao_chain.0002
			days = 31
		}
	}
	option = {
		name = hd_cao_chain.0001.a
	}
}
