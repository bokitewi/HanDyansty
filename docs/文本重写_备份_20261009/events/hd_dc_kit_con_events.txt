# 前朝遗民（每 5 年；由 hd_dc_con_yearly_effect 触发，scope:hd_dc_con_remnant = 前朝宗室）
############################################################
hd_dc_con.0300 = {
	type = character_event
	title = hd_dc_con.0300.t
	desc = {
		desc = hd_dc_con.0300.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:hd_dc_con_remnant = { is_landed = yes } }
				desc = hd_dc_con.0300.desc.landed
			}
			desc = hd_dc_con.0300.desc.landless
		}
	}
	theme = intrigue
	override_background = { reference = tgp_asia_throne_room }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:hd_dc_con_remnant
		animation = worry
	}
	trigger = {
		has_title = title:h_china
		exists = scope:hd_dc_con_remnant
		scope:hd_dc_con_remnant = { is_alive = yes }
	}

	immediate = {
		save_scope_as = hd_dc_con_ruler
		# 优待时可封之县：天子直辖、非都城、发展度最低者
		ordered_held_county = {
			limit = { NOT = { this = root.capital_county } }
			order_by = {
				value = 0
				subtract = development_level
			}
			save_scope_as = hd_dc_con_honor_county
		}
	}

	option = {	# 优待
		name = hd_dc_con.0300.a
		hd_dc_con_honor_remnant_effect = yes
		ai_chance = {
			base = 50
			modifier = {
				add = 30
				OR = {
					has_trait = compassionate
					has_trait = just
					has_trait = generous
				}
			}
			modifier = {
				add = -30
				OR = {
					has_trait = paranoid
					has_trait = callous
				}
			}
		}
	}
	option = {	# 监视
		name = hd_dc_con.0300.b
		trigger = { gold >= hd_dc_con_watch_gold }
		remove_short_term_gold = hd_dc_con_watch_gold
		scope:hd_dc_con_remnant = {
			custom_tooltip = hd_dc_con_watched_tt
			add_character_flag = {
				flag = hd_dc_con_watched
				years = 10
			}
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 20
				has_trait = paranoid
			}
		}
	}
	option = {	# 诛杀（只能对境内或狱中的宗室）
		name = hd_dc_con.0300.c
		trigger = {
			scope:hd_dc_con_remnant = {
				OR = {
					top_liege ?= root
					is_imprisoned_by = root
				}
			}
		}
		hd_dc_con_kill_remnant_effect = yes
		ai_chance = {
			base = 10
			modifier = {
				add = 40
				OR = {
					has_trait = paranoid
					has_trait = callous
					has_trait = sadistic
				}
			}
			modifier = {
				add = -10
				OR = {
					has_trait = compassionate
					has_trait = just
				}
			}
		}
	}
}
