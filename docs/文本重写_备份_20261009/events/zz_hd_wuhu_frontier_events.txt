hd_wuhu.2001 = {
	type = letter_event
	opening = hd_wuhu.2001_opening
	desc = hd_wuhu.2001_letter
	sender = scope:upset_vassal

	trigger = {

		NOR = {
			has_character_flag = had_event_hd_wuhu.2001
			has_character_flag = had_event_hd_wuhu.2002
		}

		exists = scope:story.var:upset_vassal

		exists = scope:story.var:turkic_leader

		scope:story = {
			var:upset_vassal = {
				is_alive = yes
				is_imprisoned = no
				NOT = {
					has_trait = forgiving
				}
			}
		}
	}

	immediate = {

		add_character_flag = {
			flag = had_event_hd_wuhu.2001
		}
		scope:story = {
			var:upset_vassal = {
				save_scope_as = upset_vassal
			}
			var:turkic_leader = {
				save_scope_as = turkic_leader
			}
		}
	}

	option = {
		name = hd_wuhu.2001.a

		custom_tooltip = children_arriving_shortly.tt

		trigger_event = {
			id = hd_wuhu.2004
			days = { 5 30 }
		}
	}
}
