hd_yuan_jizhou.0012 = {
	type = character_event
	title = hd4_gsz_invite.t
	desc = hd4_gsz_invite.desc
	theme = war
	trigger = {
		this = character:gong_sun_zan
		is_alive = yes
		has_title = title:k_uuii_youzhou
		NOT = {
			exists = global_var:hd4_jizhou_finished
		}
		NOT = {
			has_character_flag = hd4_jizhou_authorized
		}
		AND = {
			exists = character:yuan_shao
			character:yuan_shao = {
				is_alive = yes
			}
		}
	}
	on_trigger_fail = {
		remove_character_flag = hd4_jizhou_invitation_lease
	}
	option = {
		name = hd4_gsz_accept
		if = {
			limit = {
				has_title = title:k_uuii_youzhou
				NOT = {
					exists = global_var:hd4_jizhou_finished
				}
			}
			add_character_flag = hd4_jizhou_authorized
			remove_character_flag = hd4_jizhou_invitation_lease
			hd4_jizhou_try_start_effect = yes
		}
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = hd4_wait
		add_character_flag = {
			flag = hd4_jizhou_invitation_lease
			days = 180
		}
		ai_chance = {
			base = 0
		}
	}
}
