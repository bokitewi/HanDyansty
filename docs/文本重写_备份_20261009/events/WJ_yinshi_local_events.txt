WJ_yinshi.3001 = {
 type = character_event
 title = WJ_yinshi.3001.t
 desc = WJ_yinshi.3001.desc
 theme = diplomacy

 right_portrait = { trigger = { exists = scope:wj_yinshi_local_lord } character = scope:wj_yinshi_local_lord }
 trigger = {
  WJ_yinshi_able_trigger = yes
  NOT = { has_character_flag = wj_yinshi_ev_3001 }
 }
 immediate = { add_character_flag = { flag = wj_yinshi_ev_3001 years = 10 } }
 option = {
  name = WJ_yinshi.3001.a
  WJ_yinshi_safe_local_effect = yes
  ai_chance = { base = 50 modifier = { add = 20 has_trait = content } modifier = { add = 20 has_trait = craven } }
 }
 option = {
  name = WJ_yinshi.3001.b
  WJ_yinshi_local_risk_effect = { SKILL = stewardship }
  ai_chance = { base = 50 modifier = { add = 20 has_trait = ambitious } modifier = { add = 20 has_trait = brave } }
 }
 option = {
  name = WJ_yinshi.3001.c
  trigger = { has_trait = just }

  ai_chance = { base = 100 }
 }
}
