# 废立：原版"废黜"阴谋得手（scheme_critical_moments.1191 的成功事件改指向此处）
# 原版 grand_ambitions.0200 在本 mod 中被置空，此事件复刻其核心结算（不含宏图故事链），
# 并在头衔易手之前判定被废者是否天子、计"废立"
# root = 废黜者；scope:target = 被废者；scope:scheme = 阴谋
############################################################
hd_dc_ten.0051 = {
	type = character_event
	title = hd_dc_ten.0051.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:hd_dc_ten_deposed_huangdi }
				desc = hd_dc_ten.0051.desc_huangdi
			}
			desc = hd_dc_ten.0051.desc
		}
	}
	theme = administrative
	left_portrait = {
		character = root
		animation = celebrate_dagger
	}
	right_portrait = {
		character = scope:target
		animation = fear
	}
	trigger = {
		exists = scope:target
		scope:target = { is_alive = yes }
	}
	immediate = {
		if = {
			limit = { title:h_china.holder ?= scope:target }
			save_scope_value_as = {
				name = hd_dc_ten_deposed_huangdi
				value = yes
			}
		}
	}
	option = {
		name = hd_dc_ten.0051.a
		# 先计诱因：头衔易手后可能立即换期
		if = {
			limit = { exists = scope:hd_dc_ten_deposed_huangdi }
			hd_dc_kit_fire_global_effect = { CATALYST = catalyst_hd_deposed }
		}
		imprison = {
			target = scope:target
			type = dungeon
		}
		if = {
			limit = {
				scope:target = { has_realm_law_flag = appointment_type_succession }
			}
			scope:target.primary_title = {
				current_heir = { save_scope_as = next_heir }
				change_appointment_investment = {
					target = root
					value = {
						value = "scope:next_heir.appointment_candidate_score(scope:target.primary_title)"
						add = 100
					}
				}
			}
			scope:target = { depose = yes }
		}
		else = {
			scope:target = {
				every_vassal = { add_to_list = hd_dc_ten_target_vassals }
				add_to_list = hd_dc_ten_target_vassals
				every_held_title = {
					limit = {
						is_noble_family_title = no
						tier >= tier_county
					}
					add_to_list = hd_dc_ten_target_titles
				}
				capital_county = { save_temporary_scope_as = hd_dc_ten_new_capital }
			}
			create_title_and_vassal_change = {
				type = usurped
				save_scope_as = change
				add_claim_on_loss = yes
			}
			scope:target.primary_title = {
				change_title_holder_include_vassals = {
					holder = root
					change = scope:change
				}
			}
			hidden_effect = {
				every_in_list = {
					list = hd_dc_ten_target_titles
					change_title_holder_include_vassals = {
						holder = root
						change = scope:change
					}
				}
				every_in_list = {
					list = hd_dc_ten_target_vassals
					limit = {
						NOT = { this = root }
					}
					change_liege = {
						liege = root
						change = scope:change
					}
				}
				if = {
					limit = { exists = scope:hd_dc_ten_new_capital }
					set_realm_capital = scope:hd_dc_ten_new_capital
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		if = {
			limit = { government_has_flag = government_has_influence }
			change_influence_level = 1
		}
		reverse_add_opinion = {
			target = scope:target
			modifier = deposed_me_opinion
		}
		scope:scheme ?= { end_scheme = yes }
	}
}
