namespace = hds

# Sun Jian hard event chain (around 200 AD).
# 5000 land hunger -> independence + Danyang invasion.
# 5001-5003 only when Sun Jian holds the Imperial Seal: Yuan Shu and Liu Biao plot his murder.

hds.5000 = {
	type = character_event
	title = hds.5000.t
	desc = hds.5000.desc
	theme = war
	left_portrait = { character = root }
	trigger = { hds_sj_land_hunger_ready_trigger = yes }
	on_trigger_fail = { remove_character_flag = hds_lease_5000 }
	option = {
		name = hds.5000.a
		ai_chance = { base = 100 }
		hds_sj_independence_effect = yes
		remove_character_flag = hds_lease_5000
	}
	option = {
		name = hds.5000.b
		ai_chance = { base = 0 }
		add_character_flag = { flag = hds_sj_land_deferred days = 365 }
		remove_character_flag = hds_lease_5000
	}
}

hds.5001 = {
	type = character_event
	title = hds.5001.t
	desc = hds.5001.desc
	theme = intrigue
	left_portrait = { character = root }
	right_portrait = { character = character:sun_jian }
	trigger = {
		this = character:yuan_shu
		hds_sj_plot_actors_ready_trigger = yes
		character:sun_jian ?= { has_character_flag = hds_sj_seal_plot }
	}
	option = {
		name = hds.5001.a
		ai_chance = { base = 100 }
		hds_sj_plot_begin_effect = yes
	}
}

hds.5002 = {
	type = character_event
	title = hds.5002.t
	desc = hds.5002.desc
	theme = intrigue
	left_portrait = { character = root }
	trigger = {
		this = character:sun_jian
		is_alive = yes
		has_character_flag = hds_sj_seal_plot
		hds_sj_plot_actors_ready_trigger = yes
	}
	option = {
		name = hds.5002.a
		ai_chance = { base = 100 }
		hds_sj_plot_warned_effect = yes
	}
}

hds.5003 = {
	type = character_event
	title = hds.5003.t
	desc = hds.5003.desc
	theme = death
	left_portrait = { character = root }
	right_portrait = { character = character:liu_biao }
	trigger = {
		this = character:sun_jian
		is_alive = yes
		has_character_flag = hds_sj_seal_plot
		hds_sj_plot_actors_ready_trigger = yes
	}
	# If a plotter died meanwhile the plot is cancelled and the invasion resumes.
	on_trigger_fail = { remove_character_flag = hds_sj_seal_plot }
	option = {
		name = hds.5003.a
		ai_chance = { base = 100 }
		hds_sj_plot_death_effect = yes
	}
}
