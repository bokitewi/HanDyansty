# 朋友类：稳妥 贤望75+双方好感+15+压力-10；冒险成功 贤望150+外交经验300+好感+25%结友；失败 对方好感-15。
# scope:wj_yinshi_friend 由 WJ_yinshi_find_friend_effect 选定。

WJ_yinshi.4001 = {
 type = character_event
 title = WJ_yinshi.4001.t
 desc = WJ_yinshi.4001.desc
 theme = friendly
 left_portrait = { character = root animation = happiness }
 right_portrait = { character = scope:wj_yinshi_friend animation = happiness }
 trigger = {
  WJ_yinshi_able_trigger = yes
  NOT = { has_character_flag = wj_yinshi_ev_4001 }
  exists = scope:wj_yinshi_friend
  scope:wj_yinshi_friend = { is_alive = yes }
 }
 immediate = { add_character_flag = { flag = wj_yinshi_ev_4001 years = 10 } }
 option = {
  name = WJ_yinshi.4001.a
  WJ_yinshi_safe_friend_effect = yes
  ai_chance = { base = 50 modifier = { add = 20 has_trait = content } modifier = { add = 20 has_trait = craven } }
 }
 option = {
  name = WJ_yinshi.4001.b
  WJ_yinshi_friend_risk_effect = yes
  ai_chance = { base = 50 modifier = { add = 20 has_trait = ambitious } modifier = { add = 20 has_trait = brave } }
 }
}
