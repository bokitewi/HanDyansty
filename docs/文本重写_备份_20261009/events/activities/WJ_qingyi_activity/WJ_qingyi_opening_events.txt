# 0001 候客。结构对照原版 feast.7002。
WJ_qingyi.0001 = {
	type = activity_event
	title = WJ_qingyi.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					this = scope:host
					has_activity_intent = qingyi_yuedan_intent
				}
				desc = WJ_qingyi.0001.desc.host_yuedan
			}
			triggered_desc = {
				trigger = { this = scope:host }
				desc = WJ_qingyi.0001.desc.host
			}
			desc = WJ_qingyi.0001.desc.guest
		}
	}
	theme = feast_activity
	cooldown = { years = 1 }

	left_portrait = { character = root animation = personality_rational }
	center_portrait = {
		trigger = { this != scope:host }
		character = scope:host
		animation = toast_goblet
	}
	right_portrait = {
		trigger = { exists = scope:wj_qingyi_waiting_companion }
		character = scope:wj_qingyi_waiting_companion
		animation = drink_goblet
	}

	trigger = {
		scope:activity = { has_activity_type = activity_wj_qingyi }
	}

	immediate = {
		play_music_cue = mx_cue_banquet
		scope:activity = {
			random_attending_character = {
				limit = {
					this != root
					this != scope:host
					is_adult = yes
					is_travelling = no
				}
				save_scope_as = wj_qingyi_waiting_companion
			}
		}
	}

	option = {
		name = WJ_qingyi.0001.a
		if = {
			limit = { this = scope:host }
			scope:activity = {
				add_activity_log_entry = {
					key = WJ_qingyi_log_0001_host
					tags = { positive }
					score = 5
					character = root
					root = { add_prestige = miniscule_prestige_gain }
				}
			}
		}
		else = {
			scope:activity = {
				add_activity_log_entry = {
					key = WJ_qingyi_log_0001_arrival
					tags = { positive }
					score = 5
					character = root
					target = scope:host
					root = { add_prestige = miniscule_prestige_gain }
				}
			}
		}
	}
}

# 0002 主人开席。结构对照原版 feast.2001。
WJ_qingyi.0002 = {
	type = activity_event
	title = WJ_qingyi.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_activity_intent = qingyi_yuedan_intent }
				desc = WJ_qingyi.0002.desc.yuedan
			}
			triggered_desc = {
				trigger = { has_activity_intent = qingyi_guest_learning_intent }
				desc = WJ_qingyi.0002.desc.learning
			}
			desc = WJ_qingyi.0002.desc
		}
	}
	theme = feast_activity

	left_portrait = { character = root animation = toast_goblet }
	center_portrait = {
		trigger = { exists = scope:wj_qingyi_opening_guest }
		character = scope:wj_qingyi_opening_guest
		animation = personality_honorable
	}

	trigger = {
		this = scope:host
		scope:activity = { has_activity_type = activity_wj_qingyi }
	}

	immediate = {
		play_music_cue = grandfeast_cue
		if = {
			limit = {
				scope:activity = {
					exists = special_guest:ming_shi
					special_guest:ming_shi = {
						is_alive = yes
						is_participant_in_activity = scope:activity
					}
				}
			}
			scope:activity.special_guest:ming_shi = { save_scope_as = wj_qingyi_opening_guest }
		}
		else = {
			scope:activity = {
				random_attending_character = {
					limit = { this != root is_alive = yes }
					save_scope_as = wj_qingyi_opening_guest
				}
			}
		}
	}

	# 开宗明义，亲自出题
	option = {
		name = WJ_qingyi.0002.a
		scope:activity = {
			add_activity_log_entry = {
				key = WJ_qingyi_log_0002
				tags = { positive }
				score = 10
				character = root
				root = { add_prestige = miniscule_prestige_gain }
			}
		}
		ai_chance = { base = 60 }
	}

	# 先请座上宾开题
	option = {
		name = WJ_qingyi.0002.b
		trigger = { exists = scope:wj_qingyi_opening_guest }
		scope:activity = {
			add_activity_log_entry = {
				key = WJ_qingyi_log_0002_b
				tags = { positive }
				score = 10
				character = root
				target = scope:wj_qingyi_opening_guest
				scope:wj_qingyi_opening_guest = {
					add_prestige = miniscule_prestige_gain
					add_opinion = { target = root modifier = pleased_opinion opinion = 10 }
				}
			}
		}
		ai_chance = { base = 40 modifier = { add = 30 has_trait = humble } }
	}
}
