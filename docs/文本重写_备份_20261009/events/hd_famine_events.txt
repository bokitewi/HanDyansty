namespace = hd_famine

# 天朝循环大修：饥荒抉择。Scope：受灾独立统治者
hd_famine.0001 = {
	type = character_event
	title = hd_famine.0001.t
	desc = hd_famine.0001.desc
	theme = realm
	left_portrait = { character = root animation = worry }

	trigger = {
		any_realm_county = { has_county_modifier = hd_famine_county_modifier }
	}

	immediate = {
		save_scope_value_as = {
			name = hd_famine_relief_cost
			value = hd_famine_relief_cost_value
		}
	}

	option = {	# 开仓赈济
		name = hd_famine.0001.a
		trigger = { gold >= scope:hd_famine_relief_cost }
		remove_short_term_gold = scope:hd_famine_relief_cost
		every_realm_county = {
			limit = { has_county_modifier = hd_famine_county_modifier }
			remove_county_modifier = hd_famine_county_modifier
			add_county_modifier = { modifier = hd_famine_relieved_county_modifier years = 2 }
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = compassionate }
			modifier = { add = 20 has_trait = just }
			modifier = { add = -40 has_trait = greedy }
		}
	}
	option = {	# 强征
		name = hd_famine.0001.b
		add_gold = scope:hd_famine_relief_cost
		every_realm_county = {
			limit = { has_county_modifier = hd_famine_county_modifier }
			add_county_modifier = { modifier = hd_famine_requisition_county_modifier years = 5 }
		}
		ai_chance = {
			base = 20
			modifier = { add = 30 has_trait = greedy }
			modifier = { add = 20 has_trait = callous }
			modifier = { add = 20 has_trait = sadistic }
		}
	}
	option = {	# 不理
		name = hd_famine.0001.c
		add_prestige = minor_prestige_loss
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = lazy }
		}
	}
}
