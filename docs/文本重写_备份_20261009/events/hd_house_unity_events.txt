namespace = hd_unity

# Celestial house unity flavour events. Root = house head under a government with house unity.
# Fired from hd_celestial_house_unity_yearly_effect (common/scripted_effects/zz_hd_celestial_house_unity_effects.txt).

# 族内倾轧 (unity <= low_house_unity)
hd_unity.0001 = {
	type = character_event
	title = hd_unity.0001.t
	desc = hd_unity.0001.desc
	theme = realm
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:hd_unity_branch
		animation = anger
	}

	trigger = {
		is_house_head = yes
		house = {
			has_house_unity = yes
			any_house_member = {
				is_house_head = no
				is_adult = yes
				is_alive = yes
			}
		}
	}

	immediate = {
		house = { # random_house_member 只能在家族作用域里用
			random_house_member = {
				limit = {
					is_house_head = no
					is_adult = yes
					is_alive = yes
				}
				weight = {
					base = 1
					modifier = {
						add = 10
						is_landed = yes
					}
				}
				save_scope_as = hd_unity_branch
			}
		}
	}

	option = { # Mediate
		name = hd_unity.0001.a
		ai_chance = {
			base = 60
		}
		remove_treasury_or_gold = minor_treasury_or_gold_value
		house = {
			add_unity_value = {
				value = medium_unity_gain
				character = root
				desc = hd_unity_discord_mediated.desc
			}
		}
		add_stress = minor_stress_gain
	}

	option = { # Favour one side
		name = hd_unity.0001.b
		ai_chance = {
			base = 25
		}
		trigger = { exists = scope:hd_unity_branch }
		scope:hd_unity_branch = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}
		house = {
			add_unity_value = {
				value = minor_unity_loss
				character = root
				desc = hd_unity_discord_favouritism.desc
			}
		}
	}

	option = { # Ignore it
		name = hd_unity.0001.c
		ai_chance = {
			base = 15
		}
		house = {
			add_unity_value = {
				value = medium_unity_loss
				character = root
				desc = hd_unity_discord_ignored.desc
			}
		}
	}
}

# 树大招风 (unity >= high_house_unity)
hd_unity.0002 = {
	type = character_event
	title = hd_unity.0002.t
	desc = hd_unity.0002.desc
	theme = realm
	left_portrait = {
		character = root
		animation = personality_cynical
	}
	right_portrait = {
		character = scope:hd_unity_liege
		animation = paranoia
	}

	trigger = {
		is_house_head = yes
		house = { has_house_unity = yes }
		exists = liege
		NOT = { liege = this }
	}

	immediate = {
		liege = { save_scope_as = hd_unity_liege }
	}

	option = { # Restrain the house
		name = hd_unity.0002.a
		ai_chance = {
			base = 35
		}
		add_prestige = minor_prestige_loss
	}

	option = { # Lavish gifts
		name = hd_unity.0002.b
		ai_chance = {
			base = 50
		}
		remove_treasury_or_gold = medium_treasury_or_gold_value
		scope:hd_unity_liege = {
			add_opinion = {
				target = root
				modifier = hd_unity_goodwill_opinion
			}
		}
	}

	option = { # Brush off the rumours
		name = hd_unity.0002.c
		ai_chance = {
			base = 15
		}
		add_prestige = minor_prestige_gain
		scope:hd_unity_liege = {
			add_opinion = {
				target = root
				modifier = hd_unity_suspicion_opinion
			}
		}
	}
}

# 分门立户 (unity <= critical_house_unity)
hd_unity.0003 = {
	type = character_event
	title = hd_unity.0003.t
	desc = hd_unity.0003.desc
	theme = realm
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:hd_unity_branch
		animation = personality_bold
	}

	trigger = {
		is_house_head = yes
		house = {
			has_house_unity = yes
			any_house_member = {
				is_house_head = no
				is_adult = yes
				is_alive = yes
				is_landed = yes
			}
		}
	}

	immediate = {
		house = { # random_house_member 只能在家族作用域里用
			random_house_member = {
				limit = {
					is_house_head = no
					is_adult = yes
					is_alive = yes
					is_landed = yes
				}
				save_scope_as = hd_unity_branch
			}
		}
	}

	option = { # Let the branch found its own house
		name = hd_unity.0003.a
		ai_chance = {
			base = 25
		}
		trigger = { exists = scope:hd_unity_branch }
		# Same path as the cadet-branch decision (cadet naming, noble family set-up, etc.)
		scope:hd_unity_branch = {
			house.house_head ?= { save_scope_as = old_head }
			found_cadet_house_decision_effect = {
				CHARACTER = scope:hd_unity_branch
				PRESTIGE = minor_prestige_gain
			}
		}
		house = {
			add_unity_value = {
				value = medium_unity_gain
				character = root
				desc = hd_unity_split_relief.desc
			}
		}
	}

	option = { # Keep them by force
		name = hd_unity.0003.b
		ai_chance = {
			base = 50
			modifier = {
				add = 25
				OR = {
					has_trait_malicious_trigger = yes
					has_trait_dominant_trigger = yes
				}
			}
		}
		trigger = { exists = scope:hd_unity_branch }
		add_stress = minor_stress_gain
		scope:hd_unity_branch = {
			add_opinion = {
				target = root
				modifier = hd_unity_forced_stay_opinion
			}
		}
	}

	option = { # Nothing can be done
		name = hd_unity.0003.c
		ai_chance = {
			base = 25
		}
		house = {
			add_unity_value = {
				value = minor_unity_loss
				character = root
				desc = hd_unity_discord_ignored.desc
			}
		}
	}
}
