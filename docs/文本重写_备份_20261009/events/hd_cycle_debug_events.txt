namespace = hd_cycle_debug

# 天朝循环大修：测试用调试事件（控制台：event hd_cycle_debug.0001 / 0002 / 0003）

scripted_effect hd_cycle_debug_set_phase_effect = {
	situation:dynastic_cycle ?= {
		situation_top_sub_region = {
			change_phase = { phase = situation_dynastic_cycle_phase_$PHASE$ }
		}
	}
}

# 切换时代（1/2）
hd_cycle_debug.0001 = {
	type = character_event
	title = hd_cycle_debug.0001.t
	desc = hd_cycle_debug.0001.desc
	theme = realm
	orphan = yes

	option = { name = hd_cycle_debug.phase.recovery hd_cycle_debug_set_phase_effect = { PHASE = hd_recovery } }
	option = { name = hd_cycle_debug.phase.expansion hd_cycle_debug_set_phase_effect = { PHASE = stability_expansion } }
	option = { name = hd_cycle_debug.phase.advancement hd_cycle_debug_set_phase_effect = { PHASE = stability_advancement } }
	option = { name = hd_cycle_debug.phase.instability hd_cycle_debug_set_phase_effect = { PHASE = instability } }
	option = { name = hd_cycle_debug.next trigger_event = hd_cycle_debug.0002 }
	option = { name = hd_cycle_debug.close }
}

# 功能测试
hd_cycle_debug.0003 = {
	type = character_event
	title = hd_cycle_debug.0003.t
	desc = hd_cycle_debug.0003.desc
	theme = realm
	orphan = yes

	option = {	# 皇帝立即抽一次崩溃期危机
		name = hd_cycle_debug.crisis
		title:h_china.holder ?= { hd_collapse_select_crisis_effect = yes }
	}
	option = {	# 在自己首都所在王国生成饥荒
		name = hd_cycle_debug.famine
		capital_county.kingdom ?= { save_scope_as = hd_debug_kingdom }
		if = {
			limit = { exists = scope:hd_debug_kingdom }
			spawn_famine_in_kingdom_effect = { KINGDOM = scope:hd_debug_kingdom }
		}
	}
	option = {	# 在皇帝国内生成冒险者
		name = hd_cycle_debug.adventurer
		title:h_china.holder ?= { hd_collapse_spawn_adventurer_effect = yes }
	}
	option = {	# 发放战略 AI story，并让全部持有者立即脉冲一次
		name = hd_cycle_debug.strat
		hd_strat_grant_all_effect = yes
		every_independent_ruler = {
			limit = { owns_story_of_type = hd_strategic_ai_story }
			hd_strat_pulse_effect = yes
		}
	}
	option = {	# 立即运行局势年度诱因与年度内容
		name = hd_cycle_debug.yearly
		situation:dynastic_cycle ?= {
			hd_cycle_yearly_catalysts_effect = yes
			hd_cycle_siyi_refresh_effect = yes
			if = {
				limit = { situation_current_phase = situation_dynastic_cycle_phase_hd_collapse }
				hd_collapse_yearly_effect = yes
			}
			if = {
				limit = { situation_current_phase = situation_dynastic_cycle_phase_chaos }
				hd_struggle_yearly_effect = yes
			}
		}
	}
	option = { name = hd_cycle_debug.close }
}
