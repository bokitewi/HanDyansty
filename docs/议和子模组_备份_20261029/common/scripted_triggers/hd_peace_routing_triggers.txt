# Character scope, proposer. The war, not the UI tab, owns routing.
hd_peace_no_ordinary_primary_war_trigger = {
	NOT = {
		any_character_war = {
			hd_peace_is_ordinary_war_trigger = yes
			OR = { primary_attacker = prev primary_defender = prev }
		}
	}
}

hd_peace_ordinary_context_trigger = {
	exists = var:hd_peace_treaty_war
	var:hd_peace_treaty_war = { hd_peace_is_ordinary_war_trigger = yes }
}

hd_peace_independence_war_trigger = {
	OR = { using_cb = independence_war using_cb = hd_dc_demand_independence_war using_cb = tributary_independence_war }
}

hd_peace_subjugation_war_trigger = {
	OR = { using_cb = vassalization_cb using_cb = tribal_subjugation_cb using_cb = leg_demand_fealty_cb using_cb = silk_road_vassalization_cb using_cb = hd_dc_taoni_cb }
}

# A territory can only be demanded forcibly if it is in the declared war goal.
# A mutually accepted exchange may include other counties in the two realms.
hd_peace_forced_territories_valid_trigger = {
	NOT = {
		any_in_list = {
			variable = hd_peace_demand_counties
			NOT = { root = { is_target_in_variable_list = { name = hd_peace_cb_reserved_counties target = prev } } }
		}
	}
	# A whole title is forced only when all of the land it takes is a war target.
	NOT = {
		any_in_list = {
			variable = hd_peace_demand_titles
			NOT = { hd_peace_demand_title_forced_valid_trigger = yes }
		}
	}
}

hd_peace_force_allowed_trigger = {
	hd_peace_ordinary_context_trigger = yes
	hd_peace_actor_warscore_value >= 100
	hd_peace_demand_cost_value <= 100
	hd_peace_offer_cost_value = 0
	hd_peace_forced_territories_valid_trigger = yes
	hd_peace_forced_clause_types_valid_trigger = yes
	hd_peace_all_clauses_valid_trigger = yes
}

hd_peace_forced_clause_types_valid_trigger = {
	# Independence of third parties and a new alliance require mutual consent.
	hd_peace_demand_release_count_value = 0
	NOT = { has_variable = hd_peace_forced_alliance }
	trigger_if = {
		limit = { has_variable = hd_peace_demand_conversion }
		var:hd_peace_treaty_war = { OR = { using_cb = minor_religious_war using_cb = religious_war using_cb = major_religious_war } }
	}
	trigger_if = {
		limit = { OR = { has_variable = hd_peace_demand_suzerainty has_variable = hd_peace_demand_tribute } }
		var:hd_peace_treaty_war = { OR = { using_cb = make_tributary_cb using_cb = domination_cb } }
	}
	trigger_if = {
		limit = { OR = { hd_peace_demand_county_count_value > 0 hd_peace_demand_title_count_value > 0 } }
		var:hd_peace_treaty_war = {
			NOT = { hd_peace_independence_war_trigger = yes }
			NOT = { hd_peace_subjugation_war_trigger = yes }
			NOT = { OR = { using_cb = claim_cb using_cb = crusading_claim_cb } }
		}
	}
}

	hd_peace_selected_releases_valid_trigger = yes
	hd_peace_selected_territories_valid_trigger = yes
	var:hd_peace_offer_gold >= 0
	var:hd_peace_demand_gold >= 0
	var:hd_peace_offer_gold <= hd_peace_max_offer_gold_value
