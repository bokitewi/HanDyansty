on_war_started = { on_actions = { hd_peace_on_war_started } }
on_war_transferred = { on_actions = { hd_peace_on_war_started } }
on_game_start_after_lobby = { on_actions = { hd_peace_after_lobby } }
on_death = { on_actions = { hd_peace_on_leader_death } }
on_war_won_attacker = { on_actions = { hd_peace_on_war_ended } }
on_war_won_defender = { on_actions = { hd_peace_on_war_ended } }
on_war_white_peace = { on_actions = { hd_peace_on_war_ended } }
on_war_invalidated = { on_actions = { hd_peace_on_war_ended } }

hd_peace_on_war_started = {
	effect = {
		scope:war ?= { hd_peace_start_monitor_effect = yes }
	}
}
hd_peace_after_lobby = {
	effect = {
		every_living_character = {
			limit = { is_at_war = yes }
			every_character_war = {
				limit = { primary_attacker = prev }
				hd_peace_start_monitor_effect = yes
			}
		}
	}
}
hd_peace_on_leader_death = {
	effect = {
		every_character_war = {
			limit = { primary_attacker = root }
			save_scope_as = war
			# The defender survives the attacker's death and rebinds the monitor
			# to the actual new attacker after inheritance has completed.
			primary_defender = { trigger_event = { id = hd_peace.0501 days = 1 } }
		}
	}
}
hd_peace_on_war_ended = {
	effect = {
		scope:attacker ?= { hd_peace_cleanup_ended_war_treaty_effect = yes }
		scope:defender ?= { hd_peace_cleanup_ended_war_treaty_effect = yes }
		scope:war ?= { remove_variable = hd_peace_ai_offer_pending }
	}
}
