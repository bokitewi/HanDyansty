# Whole-title demands. ROOT is always the proposer, as in every treaty value.

# Landed title scope (duchy or above). The clause takes every county of the
# de jure title held inside the opponent's realm; occupied ones cost half.
# A title below the proposer's own highest rank costs 10% less.
hd_peace_demand_title_price_value = {
	value = 0
	every_de_jure_county = {
		limit = { hd_peace_county_in_opponent_realm_trigger = yes }
		add = hd_peace_occupied_county_price_value
	}
	if = { limit = { hd_peace_title_below_proposer_tier_trigger = yes } multiply = 0.9 }
	round = yes
}

# Landed title scope: counties the clause would actually transfer.
hd_peace_demand_title_county_count_value = {
	value = 0
	every_de_jure_county = {
		limit = { hd_peace_county_in_opponent_realm_trigger = yes }
		add = 1
	}
}

hd_peace_demand_title_count_value = { value = 0 if = { limit = { has_variable_list = hd_peace_demand_titles } every_in_list = { variable = hd_peace_demand_titles add = 1 } } }
hd_peace_demand_title_candidate_count_value = { value = 0 if = { limit = { has_variable_list = hd_peace_demand_title_candidates } every_in_list = { variable = hd_peace_demand_title_candidates add = 1 } } }

# Display-only subtotals.
hd_peace_demand_title_cost_value = { value = 0 if = { limit = { has_variable_list = hd_peace_demand_titles } every_in_list = { variable = hd_peace_demand_titles add = hd_peace_demand_title_price_value } } }

# The recipient event names five counties per side and three titles; land
# clauses have no count cap, so the remainder is shown as a number.
hd_peace_offer_county_extra_count_value = { value = hd_peace_offer_county_count_value subtract = 5 min = 0 }
hd_peace_demand_county_extra_count_value = { value = hd_peace_demand_county_count_value subtract = 5 min = 0 }
hd_peace_demand_title_extra_count_value = { value = hd_peace_demand_title_count_value subtract = 3 min = 0 }

# GUI rows: ROOT = player, scope:county = the listed county or title.
hd_peace_demand_county_row_price_value = {
	value = 0
	if = { limit = { exists = scope:county } scope:county = { add = hd_peace_demand_county_price_value } }
}
hd_peace_demand_county_row_base_price_value = {
	value = 0
	if = { limit = { exists = scope:county } scope:county = { add = hd_peace_county_price_value } }
}
hd_peace_demand_title_row_price_value = {
	value = 0
	if = { limit = { exists = scope:county } scope:county = { add = hd_peace_demand_title_price_value } }
}
hd_peace_demand_title_row_county_count_value = {
	value = 0
	if = { limit = { exists = scope:county } scope:county = { add = hd_peace_demand_title_county_count_value } }
}
