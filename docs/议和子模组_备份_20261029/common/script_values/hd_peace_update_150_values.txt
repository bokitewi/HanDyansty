# Negotiation values. ROOT remains the proposer unless stated otherwise.
# County scope: developed counties and realm capitals carry a higher price.
hd_peace_county_price_value = {
 value = 12
 add = { value = development_level multiply = 0.3 }
 if = { limit = { exists = holder this = holder.capital_county } add = 8 }
 max = 40
 round = yes
}
# County scope, ROOT = proposer. Land the proposer's side already occupies in
# this war costs half when demanded. Offers keep the plain county price.
hd_peace_occupied_county_price_value = {
 value = hd_peace_county_price_value
 if = { limit = { hd_peace_county_occupied_by_proposer_side_trigger = yes } multiply = 0.5 }
}
# County scope, ROOT = proposer. One individually demanded county; a ruler of
# duchy rank or above demands a lower-rank title at 10% less.
hd_peace_demand_county_price_value = {
 value = hd_peace_occupied_county_price_value
 if = { limit = { root = { highest_held_title_tier > tier_county } } multiply = 0.9 }
 round = yes
}
# Character scope: the vassal's sub-realm size and military strength.
hd_peace_release_vassal_price_value = {
 value = 15
 add = { value = realm_size multiply = 3 max = 45 }
 add = { value = current_military_strength divide = 1000 max = 15 }
 max = 75
 round = yes
}
hd_peace_offer_release_count_value = { value = 0 if = { limit = { has_variable_list = hd_peace_offer_released_vassals } every_in_list = { variable = hd_peace_offer_released_vassals add = 1 } } }
hd_peace_demand_release_count_value = { value = 0 if = { limit = { has_variable_list = hd_peace_demand_released_vassals } every_in_list = { variable = hd_peace_demand_released_vassals add = 1 } } }

# Outcome price is separate from net clause balance to avoid circular evaluation.
hd_peace_war_outcome_adjustment_value = {
 value = 0
}
hd_peace_war_duration_value = {
 value = 0
 if = { limit = { exists = var:hd_peace_treaty_war } var:hd_peace_treaty_war = { add = war_days } }
}
# Capacity to keep fighting: duration and the two leaders' current armies.
# At a complete military result, this additional reluctance disappears.
hd_peace_war_resistance_value = {
 value = 30
 subtract = { value = hd_peace_war_duration_value divide = 73 max = 20 }
 if = { limit = { hd_peace_war_duration_value < 365 } add = 15 }
 if = {
  limit = { exists = var:hd_peace_treaty_opponent }
  add = {
   value = var:hd_peace_treaty_opponent.current_military_strength
   divide = { value = current_military_strength min = 1 }
   subtract = 1
   multiply = 10
   min = -10
   max = 15
  }
 }
 min = 0
 if = { limit = { OR = { hd_peace_actor_warscore_value >= 100 hd_peace_actor_warscore_value <= -100 } } multiply = 0 }
 round = yes
}
hd_peace_gold_search_mid_value = {
 value = var:hd_peace_gold_search_low
 add = var:hd_peace_gold_search_high
 divide = 2
 floor = yes
}

hd_peace_demand_release_candidate_count_value = { value = 0 if = { limit = { has_variable_list = hd_peace_demand_release_candidates } every_in_list = { variable = hd_peace_demand_release_candidates add = 1 } } }
