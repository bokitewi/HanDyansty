# 253「姜维北伐」剧本史实战争（诸葛恪围新城）。照抄 218 的 hd_sg218_war_cb：只能由脚本宣战，胜利时只转目标头衔。
# 守方只是被攻打的边将本人（target_top_liege_if_outside_realm = no），不牵动魏国全国。
# 战争变量 hd253_war_kind = flag:<场次>（由 hd253_start_history_war_effect 设）；结局设全局标记 hd253_<场次>_won / _lost / _white / _invalid。
hd253_history_war_cb = {
	icon = conquest
	group = conquest
	combine_into_one = no
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no
	allowed_for_character = {}
	allowed_against_character = {}
	target_titles = all
	target_title_tier = all
	target_de_jure_regions_above = yes
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder
	should_check_for_interface_availability = no
	valid_to_start = { always = no }
	cost = {}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = { OR = { this = scope:defender target_is_liege_or_above = scope:defender } }
				}
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }

		hd253_war_result_effect = { RESULT = invalid }
	
		}
		else = { hd_peace_cb_invalidated_effect = yes }
	}
	on_declaration = { on_declared_war = yes }

	on_victory_desc = { desc = hd253_history_war_victory_desc }
	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }

		create_title_and_vassal_change = { type = conquest save_scope_as = hd253_conquest_change add_claim_on_loss = no }
		every_in_list = {
			list = target_titles
			change_title_holder = { holder = scope:attacker change = scope:hd253_conquest_change }
		}
		resolve_title_and_vassal_change = scope:hd253_conquest_change
		add_truce_attacker_victory_effect = yes
		hd253_war_result_effect = { RESULT = won }
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_white_peace_desc = { desc = hd253_history_war_white_peace_desc }
	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }

		add_truce_white_peace_effect = yes
		hd253_war_result_effect = { RESULT = white }
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_defeat_desc = { desc = hd253_history_war_defeat_desc }
	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }

		add_truce_attacker_defeat_effect = yes
		hd253_war_result_effect = { RESULT = lost }
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	transfer_behavior = transfer
	attacker_allies_inherit = no
	defender_allies_inherit = yes
	war_name = "HD253_HISTORY_WAR_NAME"
	cb_name = "HD253_HISTORY_WAR_CB_NAME"
	interface_priority = 80
	attacker_wargoal_percentage = 0.8
	landless_attacker_needs_armies = no
	full_occupation_by_attacker_gives_victory = no
	full_occupation_by_defender_gives_victory = no
	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 50
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	max_ai_diplo_distance_to_title = 500
}
