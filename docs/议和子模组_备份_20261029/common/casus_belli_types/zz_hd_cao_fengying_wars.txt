hd_cao_fengying_cb = {
	icon = claim_cb
	group = civil_war
	target_titles = none
	target_top_liege_if_outside_realm = no
	ticking_war_score_targets_entire_realm = yes
	allow_hostages = no

	allowed_for_character = {
		scope:attacker = {
			hd_cao_fengying_actor_valid_trigger = yes
			has_character_flag = hd_cao_fengying_authorized
			hd_dong_transition_pending_trigger = no
		}
	}
	allowed_against_character = {
		scope:defender = {
			hd_cao_fengying_target_trigger = yes
		}
	}

	# The target is whoever holds the emperor (regent), or the emperor himself.
	# If that changes or the defender dies, this attempt is invalidated but Cao's
	# authorization stays so he can challenge the new target.
	should_invalidate = {
		OR = {
			scope:defender = { hd_cao_fengying_target_trigger = no }
			scope:attacker = { hd_cao_fengying_actor_valid_trigger = no }
		}
	}

	on_declaration = { on_declared_war = yes }

	on_victory_desc = { desc = hd_cao_fengying_victory_desc }
	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }

		scope:attacker = { hd_cao_fengying_victory_effect = yes }
		# 全局通报：天子东归
		scope:attacker = {
			save_scope_as = hd_news_a
			capital_province ?= { save_scope_as = hd_news_place }
		}
		title:h_china.holder ?= { save_scope_as = hd_news_b }
		if = { limit = { NOT = { exists = scope:hd_news_b } } scope:attacker = { save_scope_as = hd_news_b } }
		hd_world_news_send_effect = { EVENT = hd_world_news.0012 }
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_white_peace_desc = { desc = hd_cao_fengying_no_victory_desc }
	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }

		add_truce_white_peace_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_defeat_desc = { desc = hd_cao_fengying_no_victory_desc }
	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }

		add_truce_attacker_defeat_effect = yes
	
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }
 
		}
		else = { hd_peace_cb_invalidated_effect = yes }
	}
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	interface_priority = 75
	war_name = HD_CAO_FENGYING_WAR_NAME
}
