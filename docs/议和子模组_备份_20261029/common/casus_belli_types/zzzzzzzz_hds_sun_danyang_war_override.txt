# Legacy CB key retained for save compatibility; this is now Sun Ce's Danyang invasion.
hds_wu_invasion_cb = {
	icon = invasion
	group = conquest
	target_titles = all
	target_title_tier = duchy
	target_top_liege_if_outside_realm = no
	allowed_against_character = {
		scope:defender = { is_alive = yes is_landed = yes is_independent_ruler = yes }
		NOT = { scope:attacker = scope:defender }
	}
	cost = { prestige = { value = 0 } }
	ai_score_mult = { value = 0 }
	use_de_jure_wargoal_only = yes
	should_show_war_goal_subview = yes
	ignore_effect = change_title_holder
	on_declaration = { on_declared_war = yes }
	on_victory_desc = { desc = hds_war_victory_desc }
	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }
 hds_cb_victory_effect = yes add_truce_attacker_victory_effect = yes 
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_white_peace_desc = { desc = hds_war_no_conquest_desc }
	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }
 hds_cb_retry_effect = yes add_truce_white_peace_effect = yes 
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_defeat_desc = { desc = hds_war_no_conquest_desc }
	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }
 hds_cb_retry_effect = yes add_truce_attacker_defeat_effect = yes 
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }
 hds_cb_retry_effect = yes 
		}
		else = { hd_peace_cb_invalidated_effect = yes }
	}
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	transfer_behavior = transfer
	# 孙策/孙坚以无封地家族身份开战：没有已集结部队时不自动判防守方 100% 战争分。
	landless_attacker_needs_armies = no
	interface_priority = 75

	# The inherited Sun-family estate remains a legal non-landed campaign identity.
	allowed_for_character = {
		scope:attacker = {
			is_alive = yes
			is_ruler = yes
			is_independent_ruler = yes
			OR = {
				AND = {
					AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } this = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
					has_character_flag = hds_sc_departed
					has_character_flag = hds_sc_wu_authorized
					NOT = { has_character_flag = hds_sc_wu_complete }
					OR = { is_landed = yes hds_sun_estate_inherited_trigger = yes }
				}
				AND = {
					this = character:sun_jian
					has_character_flag = hds_sj_departed
					has_character_flag = hds_sj_wu_authorized
					NOT = { has_character_flag = hds_sj_wu_complete }
					OR = { is_landed = yes hds_estate_ruler_trigger = yes has_character_flag = hds_sj_camp_converted }
				}
				# 2026-10-10：接续孙策的孙权。
				AND = {
					hds_sq_leader_trigger = yes
					has_character_flag = hds_sq_departed
					NOT = { has_character_flag = hds_sq_wu_complete }
				}
			}
		}
	}
	valid_to_start = {
		scope:attacker = { has_character_flag = hds_launch_authorized }
		scope:target = title:d_danyang
		scope:target = scope:attacker.var:hds_requested_duchy
		title:d_danyang = { hds_enemy_holds_target_trigger = yes }
	}
	should_invalidate = {
		OR = {
			NOR = {
				AND = { trigger_if = { limit = { has_global_variable = wj_hchar_sun_ce } scope:attacker = global_var:wj_hchar_sun_ce } trigger_else = { always = no } }
				scope:attacker = character:sun_jian
				AND = { trigger_if = { limit = { has_global_variable = hds_sq_leader } scope:attacker = global_var:hds_sq_leader } trigger_else = { always = no } }
			}
			scope:attacker = { is_alive = no }
			scope:defender = { is_alive = no }
			NOT = { title:d_danyang = { hds_enemy_holds_target_trigger = yes } }
		}
	}
	cb_name = hds_wu_invasion_cb_name
	war_name_base = hds_wu_invasion_cb_name
	war_name = HDS_WU_INVASION_WAR_NAME
}
