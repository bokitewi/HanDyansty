# Retired CB compatibility override.
# Intentionally keep this vanilla identifier registered: other loaded scripts
# and pre-patch saves may still reference it. No player or AI can select it.
# Existing wars with this CB invalidate without granting victory/peace rewards.
# Do NOT rename this file: it must load after 10_tgp_china_wars.txt.
# chinese_consolidation_cb and the mod's empire-unification CB are untouched.

chinese_reunification_cb = {
	icon = celestial
	group = celestial
	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = { always = no }
	allowed_for_character_display_regardless = { always = no }
	allowed_against_character = { always = no }
	allowed_against_character_display_regardless = { always = no }
	valid_to_start = { always = no }
	ai_score_mult = { value = 0 }

	# Retain the target representation for already-existing saved wars.
	target_title_tier = kingdom
	target_titles = neighbor_land_or_water
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder
	ai_only_against_neighbors = yes
	cost = { prestige = { value = 0 } }

	should_invalidate = { always = yes }
	on_declaration = { }
	on_invalidated_desc = hd_removed_chinese_reunification_cb_desc
	on_invalidated = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes } }
 
		}
		else = { hd_peace_cb_invalidated_effect = yes }
	}
	on_victory_desc = hd_removed_chinese_reunification_cb_desc
	on_victory = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }
 
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_white_peace_desc = hd_removed_chinese_reunification_cb_desc
	on_white_peace = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }
 
		}
		else = { hd_peace_cb_result_effect = yes }
	}
	on_defeat_desc = hd_removed_chinese_reunification_cb_desc
	on_defeat = {
		# HD peace: the old payload belongs only to an authorized scripted war.
		if = {
			limit = { exists = root.war root.war = { hd_peace_is_scripted_war_trigger = yes has_variable = hd_peace_script_authorized } }
 
		}
		else = { hd_peace_cb_result_effect = yes }
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = "CHINESE_CONSOLIDATION_WAR_NAME"
	war_name_base = "CHINESE_CONSOLIDATION_WAR_NAME_BASE"
	cb_name = "CHINESE_CONSOLIDATION_CB_NAME"
	interface_priority = 0
}
