# $ATTACKER$ 对 $DEFENDER$ 宣战，主目标 $TARGET$（头衔），场次 $KIND$。成功则存为 scope:hd_nbc_war_$KIND$
hd_nbc_start_war_effect = {
	$ATTACKER$ = {
		if = {
			limit = { is_allied_to = $DEFENDER$ }
			break_alliance_due_to_attack_effect = { TARGET = $DEFENDER$ }
		}
		start_war = {
			cb = hd_nbc_war_cb
			target = $DEFENDER$
			target_title = $TARGET$
		}
		random_character_war = {
			limit = {
				using_cb = hd_nbc_war_cb
				primary_attacker = $ATTACKER$
				primary_defender = $DEFENDER$
				NOT = { has_variable = hd_nbc_war_kind }
			}
			set_variable = { name = hd_nbc_war_kind value = flag:$KIND$ }
			save_scope_as = hd_nbc_war_$KIND$
		}
	}
	if = {
		limit = { exists = scope:hd_nbc_war_$KIND$ }
		set_global_variable = hd_nbc_$KIND$_active
		debug_log = "HD_NBC WAR_CONFIRMED $KIND$"
	}
	else = {
		debug_log = "HD_NBC WAR_START_FAILED $KIND$"
	}
}

# 封臣出征：先由本人宣战，不成则由其最高领主宣战、本人参战。$WHO$ 是人物 ID
hd_nbc_vassal_war_effect = {
	hd_nbc_start_war_effect = { ATTACKER = character:$WHO$ DEFENDER = $DEFENDER$ TARGET = $TARGET$ KIND = $KIND$ }
	if = {
		limit = {
			NOT = { exists = scope:hd_nbc_war_$KIND$ }
			character:$WHO$ = { is_independent_ruler = no }
		}
		character:$WHO$.top_liege = { save_scope_as = hd_nbc_att_$KIND$ }
		hd_nbc_start_war_effect = { ATTACKER = scope:hd_nbc_att_$KIND$ DEFENDER = $DEFENDER$ TARGET = $TARGET$ KIND = $KIND$ }
		scope:hd_nbc_war_$KIND$ ?= { hd_nbc_add_attacker_effect = { WHO = $WHO$ } }
	}
}

# 战争结局。Root：战争（hd_nbc_war_cb 的 on_victory 等）
hd_nbc_war_result_effect = {
	hd_nbc_war_result_kind_effect = { KIND = fashu RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = yannan RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = beifa RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = feishui RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = restore RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = luxun RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = nanyan RESULT = $RESULT$ }
	hd_nbc_war_result_kind_effect = { KIND = houqin RESULT = $RESULT$ }
}
hd_nbc_war_result_kind_effect = {
	if = {
		limit = { var:hd_nbc_war_kind ?= flag:$KIND$ }
		set_global_variable = hd_nbc_$KIND$_$RESULT$
		if = {
			limit = { has_global_variable = hd_nbc_$KIND$_active }
			remove_global_variable = hd_nbc_$KIND$_active
		}
		debug_log = "HD_NBC WAR_RESULT $KIND$ $RESULT$"
	}
}

# 把 $COUNTY$ 转给 $TO$（scope），附属男爵领一并转
hd_nbc_give_county_effect = {
	if = {
		limit = { NOT = { $COUNTY$.holder ?= $TO$ } }
		create_title_and_vassal_change = { type = usurped save_scope_as = hd_nbc_give_change add_claim_on_loss = no }
		$COUNTY$ = {
			change_title_holder = {
				holder = $TO$
				change = scope:hd_nbc_give_change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:hd_nbc_give_change
	}
}

# Character：脱离领主独立
hd_nbc_become_independent_effect = {
	if = {
		limit = {
			is_landed = yes
			is_independent_ruler = no
		}
		create_title_and_vassal_change = { type = independency save_scope_as = hd_nbc_indep_change add_claim_on_loss = no }
		becomes_independent = { change = scope:hd_nbc_indep_change }
		resolve_title_and_vassal_change = scope:hd_nbc_indep_change
	}
}

# Character：改投 $LIEGE$
hd_nbc_swear_fealty_effect = {
	if = {
		limit = {
			is_landed = yes
			NOT = { liege ?= $LIEGE$ }
			NOT = { this = $LIEGE$ }
		}
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = hd_nbc_fealty_change add_claim_on_loss = no }
		change_liege = { liege = $LIEGE$ change = scope:hd_nbc_fealty_change }
		resolve_title_and_vassal_change = scope:hd_nbc_fealty_change
	}
}

# Character：建国号、定称谓（新政权）。FIXED = yes 时记到王朝上
hd_nbc_found_realm_effect = {
	if = {
		limit = { NOT = { has_government = celestial_government } }
		change_government = celestial_government
	}
	primary_title = {
		set_title_name = $NAME$
		set_variable = { name = hd_nbc_style value = flag:$STYLE$ }
	}
	if = {
		limit = { always = $FIXED$ }
		dynasty ?= {
			remove_variable = hd_fixed_realm_title
			set_variable = { name = hd_fixed_realm_name value = flag:$NAME$ }
		}
	}
}

# A0 史实卒年：电脑角色到日病逝；玩家角色弹"病笃"，可"强撑"一年
############################################################
hd_nbc_hist_death_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = hd_nbc_death_$WHO$_done }
			current_date >= $DATE$
		}
		if = {
			limit = { NOT = { character:$WHO$ ?= { is_alive = yes } } }
			set_global_variable = hd_nbc_death_$WHO$_done
		}
		else = {
			character:$WHO$ = {
				if = {
					limit = { is_ai = yes }
					death = { death_reason = death_natural_causes }
					set_global_variable = hd_nbc_death_$WHO$_done
					debug_log = "HD_NBC HIST_DEATH $WHO$"
				}
				else_if = {
					limit = { NOT = { has_variable = hd_nbc_death_delay } }
					trigger_event = hd_nbc.0100
				}
			}
		}
	}
}

# A1 灭蜀：桓温伐蜀得胜。巴蜀帝国头衔交还天子，李势归建康
############################################################
hd_nbc_check_fashu_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = hd_nbc_fashu_done }
			OR = {
				has_global_variable = hd_nbc_fashu_won
				has_global_variable = hd_nbc_fashu_lost
				has_global_variable = hd_nbc_fashu_white
				has_global_variable = hd_nbc_fashu_invalid
			}
		}
		set_global_variable = hd_nbc_fashu_done
		if = {
			limit = {
				has_global_variable = hd_nbc_fashu_won
				character:longkang_huan_017 ?= { is_alive = yes }
			}
			character:longkang_huan_017 = { trigger_event = hd_nbc.0110 }
		}
	}
}

# A6 淝水之战：383 年起，苻坚独立、持帝国级头衔，晋（天子）在
############################################################
hd_nbc_check_feishui_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = hd_nbc_feishui_done }
			current_date >= 383.6.1
		}
		if = {
			limit = {
				exists = title:h_china.holder
				character:lueyang_fu_020 ?= {
					is_alive = yes
					is_independent_ruler = yes
					realm_size >= 100	# 苻坚统一北方的程度：领土不少于 100 县（前秦只有郡级头衔，不按头衔等级判）
					is_at_war = no
					NOT = { this = title:h_china.holder }
				}
			}
			set_global_variable = hd_nbc_feishui_done
			character:lueyang_fu_020 = { trigger_event = hd_nbc.0160 }
		}
		else_if = {
			limit = { current_date >= 386.1.1 }
			set_global_variable = hd_nbc_feishui_done
			debug_log = "HD_NBC SKIP feishui"
		}
	}
	# 淝水结局：苻坚败（或白和），前秦瓦解
	if = {
		limit = {
			has_global_variable = hd_nbc_feishui_done
			NOT = { has_global_variable = hd_nbc_feishui_result_done }
			OR = {
				has_global_variable = hd_nbc_feishui_won
				has_global_variable = hd_nbc_feishui_lost
				has_global_variable = hd_nbc_feishui_white
				has_global_variable = hd_nbc_feishui_invalid
			}
		}
		set_global_variable = hd_nbc_feishui_result_done
		character:lueyang_fu_020 ?= {
			if = {
				limit = {
					is_alive = yes
					OR = {
						has_global_variable = hd_nbc_feishui_lost
						has_global_variable = hd_nbc_feishui_white
					}
				}
				trigger_event = hd_nbc.0161
			}
			else_if = {
				limit = { has_global_variable = hd_nbc_feishui_won }
				save_scope_as = hd_news_a
				title:h_china.holder ?= { save_scope_as = hd_news_b }
				hd_world_news_send_all_effect = { EVENT = hd_nbc.0902 }
			}
		}
	}
}
