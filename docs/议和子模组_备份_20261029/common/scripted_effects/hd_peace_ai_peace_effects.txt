# ROOT = AI proposer. Candidate lists were created by hd_peace_open_treaty_effect.
# Claims and counties bordering ROOT's realm receive a strong preference.
hd_peace_ai_select_demand_county_effect = {
	random_in_list = {
		variable = hd_peace_demand_candidates
		limit = {
			NOT = { this = scope:hd_peace_ai_opponent.capital_county }
		}
		weight = {
			base = 10
			modifier = {
				add = 90
				root = { has_claim_on = prev }
			}
			modifier = {
				add = 50
				any_neighboring_county = {
					exists = holder
					holder = { top_liege = root }
				}
			}
		}
		save_scope_as = hd_peace_ai_selected_county
	}
	if = {
		limit = { exists = scope:hd_peace_ai_selected_county }
		add_to_variable_list = {
			name = hd_peace_demand_counties
			target = scope:hd_peace_ai_selected_county
		}
	}
}

hd_peace_ai_select_offer_county_effect = {
	random_in_list = {
		variable = hd_peace_offer_candidates
		weight = {
			base = 10
			modifier = {
				add = 90
				scope:hd_peace_ai_opponent = { has_claim_on = prev }
			}
			modifier = {
				add = 50
				any_neighboring_county = {
					exists = holder
					holder = { top_liege = scope:hd_peace_ai_opponent }
				}
			}
		}
		save_scope_as = hd_peace_ai_selected_county
	}
	if = {
		limit = { exists = scope:hd_peace_ai_selected_county }
		add_to_variable_list = {
			name = hd_peace_offer_counties
			target = scope:hd_peace_ai_selected_county
		}
	}
}

hd_peace_ai_compose_overwhelming_demands_effect = {
	set_variable = { name = hd_peace_extended_truce value = 1 }
	random_list = {
		20 = {
			trigger = { hd_peace_opponent_can_become_tributary_trigger = yes }
			set_variable = { name = hd_peace_demand_suzerainty value = 1 }
		}
		25 = {
			trigger = { hd_peace_ai_can_demand_county_trigger = yes }
			hd_peace_ai_select_demand_county_effect = yes
			if = {
				limit = { hd_peace_ai_can_demand_tribute_trigger = yes }
				set_variable = { name = hd_peace_demand_tribute value = 1 }
			}
			set_variable = { name = hd_peace_demand_humiliation value = 1 }
		}
		20 = {
			trigger = { hd_peace_ai_can_demand_conversion_trigger = yes }
			set_variable = { name = hd_peace_demand_conversion value = 1 }
			set_variable = { name = hd_peace_demand_disarmament value = 1 }
		}
		20 = {
			set_variable = { name = hd_peace_demand_disarmament value = 1 }
			set_variable = { name = hd_peace_demand_humiliation value = 1 }
			if = {
				limit = { hd_peace_ai_can_demand_claims_trigger = yes }
				set_variable = { name = hd_peace_demand_claims value = 1 }
			}
		}
		15 = {
			hd_peace_ai_set_max_demand_resource_effect = yes
			if = {
				limit = { hd_peace_ai_can_demand_claims_trigger = yes }
				set_variable = { name = hd_peace_demand_claims value = 1 }
			}
		}
	}
}

# ROOT = AI proposer with scope:war and scope:hd_peace_ai_opponent available.
# A leading attacker first asks for its declared war goal alone, when that is
# already acceptable; otherwise it falls back to the packages below.
hd_peace_ai_compose_treaty_effect = {
	if = {
		limit = {
			hd_peace_actor_warscore_value >= 15
			hd_peace_objective_available_trigger = yes
		}
		set_variable = { name = hd_peace_objective value = 1 }
		set_variable = { name = hd_peace_long_truce value = 1 }
		if = {
			limit = { NOT = { hd_peace_ai_accepts_treaty_trigger = yes } }
			remove_variable = hd_peace_objective
			remove_variable = hd_peace_long_truce
		}
	}
	if = {
		limit = { NOT = { has_variable = hd_peace_objective } }
		if = {
			limit = { hd_peace_actor_warscore_value >= 75 }
			hd_peace_ai_compose_overwhelming_demands_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value >= 50 }
			hd_peace_ai_compose_major_demands_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value >= 30 }
			hd_peace_ai_compose_moderate_demands_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value >= 15 }
			hd_peace_ai_compose_minor_demands_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value <= -75 }
			hd_peace_ai_compose_desperate_concessions_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value <= -50 }
			hd_peace_ai_compose_major_concessions_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value <= -30 }
			hd_peace_ai_compose_moderate_concessions_effect = yes
		}
		else_if = {
			limit = { hd_peace_actor_warscore_value <= -15 }
			hd_peace_ai_compose_minor_concessions_effect = yes
		}
	}
}
