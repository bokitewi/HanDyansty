	random_list = {
		20 = {
			trigger = { hd_peace_can_become_tributary_of_opponent_trigger = yes }
			set_variable = { name = hd_peace_offer_suzerainty value = 1 }
			set_variable = { name = hd_peace_offer_humiliation value = 1 }
		}
		25 = {
			trigger = { hd_peace_ai_can_offer_county_trigger = yes }
			hd_peace_ai_select_offer_county_effect = yes
			set_variable = { name = hd_peace_offer_disarmament value = 1 }

			if = {
				limit = { hd_peace_ai_can_offer_tribute_trigger = yes }
				set_variable = { name = hd_peace_offer_tribute value = 1 }
			}
		}
		20 = {
			trigger = { hd_peace_ai_can_offer_conversion_trigger = yes }
			set_variable = { name = hd_peace_offer_conversion value = 1 }
			set_variable = { name = hd_peace_offer_disarmament value = 1 }

				limit = { hd_peace_ai_can_offer_tribute_trigger = yes }
				set_variable = { name = hd_peace_offer_tribute value = 1 }
			}
		}
		20 = {

			set_variable = { name = hd_peace_offer_humiliation value = 1 }

				limit = { hd_peace_ai_can_offer_tribute_trigger = yes }
				set_variable = { name = hd_peace_offer_tribute value = 1 }
			}
		}
		15 = {
			hd_peace_ai_set_max_offer_resource_effect = yes
			set_variable = { name = hd_peace_offer_humiliation value = 1 }
			if = {
				limit = { hd_peace_ai_can_offer_tribute_trigger = yes }
				set_variable = { name = hd_peace_offer_tribute value = 1 }

			if = {

			}
