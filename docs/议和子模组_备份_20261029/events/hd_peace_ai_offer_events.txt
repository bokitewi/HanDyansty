namespace = hd_peace

# Execute an accepted offer with ROOT = proposer. Both player and AI offers use
# this handoff because the visible recipient event has ROOT = recipient.
hd_peace.0202 = {
 type = character_event
 hidden = yes
 immediate = {
  # No signature without a matching, explicit acceptance from the recipient.
  if = {
   limit = {
    has_variable = hd_peace_treaty_locked
    exists = var:hd_peace_consent_war
    exists = var:hd_peace_consent_recipient
    has_variable = hd_peace_consent_nonce
    var:hd_peace_consent_war = var:hd_peace_treaty_war
    var:hd_peace_consent_recipient = var:hd_peace_treaty_opponent
    var:hd_peace_consent_nonce = var:hd_peace_offer_nonce
    var:hd_peace_consent_recipient = { is_ai = no }
    hd_peace_treaty_context_is_valid_trigger = yes
    hd_peace_has_any_treaty_clause_trigger = yes
    hd_peace_selected_releases_valid_trigger = yes hd_peace_selected_territories_valid_trigger = yes
			 hd_peace_all_clauses_valid_trigger = yes
    var:hd_peace_offer_gold <= hd_peace_max_offer_gold_value
    var:hd_peace_demand_gold <= hd_peace_max_demand_gold_value
    var:hd_peace_offer_herd <= hd_peace_max_offer_herd_value
    var:hd_peace_demand_herd <= hd_peace_max_demand_herd_value
   }
   var:hd_peace_treaty_war ?= { remove_variable = hd_peace_ai_offer_pending }
   hd_peace_end_negotiated_war_effect = yes
  }
  else_if = {
   limit = {
    exists = var:hd_peace_consent_recipient
    has_variable = hd_peace_consent_nonce
    var:hd_peace_consent_nonce = var:hd_peace_offer_nonce
   }
   var:hd_peace_consent_recipient = { trigger_event = { id = hd_peace.0203 } }
   var:hd_peace_treaty_war ?= { remove_variable = hd_peace_ai_offer_pending }
   hd_peace_clear_treaty_state_effect = yes
  }
 }
}

hd_peace.0203 = {
 type = character_event
 title = hd_peace_expired_offer_title
 desc = hd_peace_expired_offer_desc
 theme = diplomacy_family_focus
 option = { name = hd_peace_expired_offer_close }
}
