# Annual hidden cash payment. ROOT is the payer; the amount was fixed when the
# peace treaty was signed and may drive the payer into ordinary CK3 debt.
hd_peace.0100 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_alive = yes
				has_variable = hd_peace_tribute_receiver
				has_variable = hd_peace_tribute_years
				has_variable = hd_peace_tribute_annual_amount
				var:hd_peace_tribute_years > 0
				var:hd_peace_tribute_receiver = { is_alive = yes }
			}
			pay_short_term_gold = {
				target = var:hd_peace_tribute_receiver
				gold = var:hd_peace_tribute_annual_amount
			}
			change_variable = { name = hd_peace_tribute_years subtract = 1 }
			if = {
				limit = { var:hd_peace_tribute_years > 0 }
				trigger_event = { id = hd_peace.0100 years = 1 }
			}
			else = { hd_peace_end_money_tribute_effect = yes }
		}
		else = { hd_peace_end_money_tribute_effect = yes }
	}
}

# Form the negotiated alliance after the war result has fully resolved. Vanilla's
# negotiated-alliance opinion marker gives the direct ruler-to-ruler bond the
# same support used by CK3's own diplomatic interaction.
hd_peace.0301 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				exists = scope:hd_peace_opponent
				is_alive = yes
				is_independent_ruler = yes
				scope:hd_peace_opponent = {
					is_alive = yes
					is_independent_ruler = yes
				}
				NOT = { is_allied_to = scope:hd_peace_opponent }
			}
			scope:hd_peace_opponent = {
				add_opinion = {
					target = root
					modifier = event_negotiated_alliance_opinion
					years = 10
				}
			}
			create_alliance = {
				target = scope:hd_peace_opponent
				allied_through_owner = root
				allied_through_target = scope:hd_peace_opponent
			}
		}
	}
}
