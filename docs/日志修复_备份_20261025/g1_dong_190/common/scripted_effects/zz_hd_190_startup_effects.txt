# 指定朝官入席。头衔已由 190 头衔历史交给本人，got_minister_position_effect 见已持头衔就只补席位、居所。
# 照 hd_nbc_minister_effect 的写法，判定改用 NOT has_council_position。需 scope:councillor_liege。
hd190_minister_effect = {
	character:$WHO$ ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { has_council_position = $POS$ }
				NOT = { this = scope:councillor_liege }
			}
			if = {
				limit = {
					is_landed = no
					NOT = { employer ?= scope:councillor_liege }
				}
				scope:councillor_liege = { add_courtier = character:$WHO$ }
			}
			got_minister_position_effect = { MINISTER_TITLE = $TITLE$ MINISTER_POSITION = $POS$ }
		}
	}
}
