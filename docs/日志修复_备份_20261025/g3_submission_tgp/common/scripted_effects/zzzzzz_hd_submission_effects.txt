# NEW effects only. Existing HD war effects/CBs are neither copied nor replaced.

# Interaction scope. Locks are applied only after the engine actually sends.
hd_submission_request_sent_effect = {
    scope:actor = {
        add_character_flag = { flag = hd_submission_outgoing_pending days = 120 }
        add_character_flag = { flag = hd_submission_request_issued days = 1 }
        set_variable = { name = hd_submission_outgoing_target value = scope:recipient days = 120 }
    }
    scope:recipient = {
        add_character_flag = { flag = hd_submission_incoming_pending days = 120 }
        set_variable = { name = hd_submission_incoming_actor value = scope:actor days = 120 }
        if = {
            limit = { is_ai = no }
            add_character_flag = { flag = hd_submission_player_inbox_cooldown days = 365 }
        }
    }
}

# Only clear locks belonging to THIS pair. Old cancelled messages must not
# unlock somebody else's later pending request.
hd_submission_clear_request_effect = {
    if = {
        limit = { exists = scope:actor }
        scope:actor = {
            if = {
                limit = {
                    exists = var:hd_submission_outgoing_target
                    var:hd_submission_outgoing_target = scope:recipient
                }
                remove_variable = hd_submission_outgoing_target
                remove_character_flag = hd_submission_outgoing_pending
            }
        }
    }
    if = {
        limit = { exists = scope:recipient }
        scope:recipient = {
            if = {
                limit = {
                    exists = var:hd_submission_incoming_actor
                    var:hd_submission_incoming_actor = scope:actor
                }
                remove_variable = hd_submission_incoming_actor
                remove_character_flag = hd_submission_incoming_pending
            }
        }
    }
}

hd_submission_request_accepted_effect = {
    if = {
        limit = {
            exists = scope:actor
            exists = scope:recipient
            scope:recipient = {
                hd_submission_recipient_legal_trigger = { LIEGE = scope:actor }
                NOT = { hd_submission_ai_refuses_trigger = { LIEGE = scope:actor } }
                exists = var:hd_submission_incoming_actor
                var:hd_submission_incoming_actor = scope:actor
            }
            scope:actor = {
                exists = var:hd_submission_outgoing_target
                var:hd_submission_outgoing_target = scope:recipient
            }
        }
        # The native effect creates ONE swear_fealty title/vassal transaction,
        # preserves the subject's titles/sub-vassals and handles contract options.
        offer_vassalization_interaction_effect = yes
        if = {
            limit = { scope:recipient = { exists = liege liege = scope:actor } }
            hd_submission_post_accept_effect = yes
        }
    }
    else = {
        if = {
            limit = { exists = scope:actor }
            scope:actor = {
                if = {
                    limit = { is_alive = yes is_ai = no }
                    send_interface_toast = {
                        type = event_toast_effect_neutral
                        title = hd_sub_cancelled_title
                        desc = hd_sub_cancelled_desc
                        left_icon = scope:actor
                    }
                }
                hd_submission_queue_followup_effect = { DELAY_MIN = 90 DELAY_MAX = 180 }
            }
        }
    }
    hd_submission_clear_request_effect = yes
}

# 2026-10-25 归附谈判：归附成功后的善后（原在上面的成交效果里，事件路径成交也要用）。
# scope:actor = 主君，scope:recipient = 新封臣
hd_submission_post_accept_effect = {
    scope:recipient = {
        add_character_modifier = { modifier = hd_submission_recently_submitted years = 5 }
        if = { limit = { is_ai = yes } add_faction_cooldown_effect = { YEARS = 5 } }
    }
    scope:actor = {
        add_prestige = 50
        # A submission maintains existing momentum, but NEVER adds a level.
        if = {
            limit = { has_variable = hd_submission_momentum var:hd_submission_momentum > 0 }
            add_character_flag = { flag = hd_submission_recent_success years = 3 }
        }
        remove_character_flag = hd_submission_action_cooldown
        hd_submission_queue_followup_effect = { DELAY_MIN = 90 DELAY_MAX = 150 }
        if = {
            limit = { is_ai = no }
            send_interface_toast = {
                type = event_toast_effect_good
                title = hd_sub_success_title
                left_icon = scope:recipient
                right_icon = scope:actor
            }
        }
    }
}

                    left_icon = scope:recipient
                    right_icon = scope:actor

}

# Character scope. The single queued event has a finite watchdog lock. Its event
# has no restrictive top-level trigger, so changing liege/government clears it.
hd_submission_queue_followup_effect = {
    if = {
        limit = {
            is_alive = yes
            is_ai = yes
            is_landed = yes
            is_independent_ruler = yes
            government_has_flag = government_is_celestial
            NOT = { has_character_flag = hd_submission_followup_pending }
        }
        add_character_flag = { flag = hd_submission_followup_pending days = 240 }
        trigger_event = { id = hd_submission.0001 days = { $DELAY_MIN$ $DELAY_MAX$ } }
    }
}

# Actor scope. Never pass hd_submission_actor into an ai_start_best_war callback.
# The new system invokes only the existing, scope-safe HD war-selection effect.
hd_submission_try_offer_effect = {
    if = {
        limit = {
            hd_submission_world_active_trigger = yes
            hd_submission_liege_trigger = yes
        }
        save_scope_as = hd_submission_actor
        # 2026-10-25 归附谈判：预开价与复算接受度要用 scope:actor / scope:recipient
        save_scope_as = actor
        # A local neighbor snapshot: no every_ruler / province / army scan here.
        every_neighboring_top_liege_realm_owner = {
            limit = {
                is_landed = yes
                is_independent_ruler = yes
                highest_held_title_tier < scope:hd_submission_actor.highest_held_title_tier
                hd_submission_inbox_free_trigger = { LIEGE = scope:hd_submission_actor }
                hd_submission_recipient_legal_trigger = { LIEGE = scope:hd_submission_actor }
                NOT = { hd_submission_ai_refuses_trigger = { LIEGE = scope:hd_submission_actor } }
                # 正在请求归附别人：等那边谈完
                NOT = { has_character_flag = hd_petition_outgoing_pending }
                # 历史剧本保护：曹操与袁绍互不招降；陶谦激活后不被招降。
                NOT = { AND = { this = character:cao_cao scope:hd_submission_actor = character:yuan_shao } }
                NOT = { AND = { this = character:yuan_shao scope:hd_submission_actor = character:cao_cao } }
                NOT = { AND = { this = character:tao_qian has_character_flag = hd4_tao_activated } }
                OR = {
                    is_ai = yes
                    sub_realm_size <= 5
                }
            }
            # 2026-10-25 归附谈判（boki 拍板）：AI 对象本来肯降，或差 60 以内且付得起预开价，才先招降
            if = {
                limit = { is_ai = yes }
                save_scope_as = recipient
                hd_neg_ai_prepare_offer_effect = yes
                if = {
                    limit = { var:hd_neg_viable ?= 1 }
                    add_to_list = hd_submission_candidates
                }
                else = { hd_neg_ai_offer_cleanup_effect = yes }
            }
            else = { add_to_list = hd_submission_candidates }
        }
        if = {
            limit = { any_in_list = { list = hd_submission_candidates always = yes } }
            ordered_in_list = {
                list = hd_submission_candidates
                order_by = { value = sub_realm_size multiply = -1 }
                max = 3
                check_range_bounds = no
                # Native cooldown checks may reject a candidate. At most three
                # valid candidates are attempted, with at most ONE actual request.
                if = {
                    limit = { scope:hd_submission_actor = { NOT = { has_character_flag = hd_submission_request_issued } } }
                    save_scope_as = recipient
                    if = {
                        # 2026-10-25 归附谈判：AI 劝玩家归附改走事件，玩家可开价
                        limit = { is_ai = no }
                        hd_neg_offer_to_player_effect = yes
                    }
                    else = {
                        run_interaction = {
                            interaction = hd_warlord_offer_vassalization_interaction
                            actor = scope:hd_submission_actor
                            recipient = this
                            send_threshold = accept
                        }
                    }
                }
            }
            # 没发出去的候选撤掉预开价
            every_in_list = {
                list = hd_submission_candidates
                limit = { is_ai = yes }
                save_scope_as = recipient
                hd_neg_ai_offer_cleanup_effect = yes
            }
        }
        # 借用的作用域用完即清，不带进调用方随后的宣战
        clear_saved_scope = actor
        if = {
            limit = { exists = scope:recipient }
            clear_saved_scope = recipient
        }
    }
}

# Actor scope. 开战前先招降：若本次实际发出了招降（或上一份仍在等待回复），
# 之后 hd_submission_request_issued 为真，调用方应跳过本次宣战。
hd_submission_offer_before_war_effect = {
    remove_character_flag = hd_submission_request_issued
    if = {
        limit = { is_ai = yes }
        if = {
            limit = { has_character_flag = hd_submission_outgoing_pending }
            add_character_flag = { flag = hd_submission_request_issued days = 1 }
        }
        else = {
            hd_submission_try_offer_effect = yes
        }
    }
}

hd_submission_dispatch_effect = {
    if = {
        limit = { hd_submission_ai_can_scan_trigger = yes }
        # Counts keep their old war route; only dukes+ can ask for vassals.
        hd_submission_try_offer_effect = yes
        if = {
            limit = { NOT = { has_character_flag = hd_submission_request_issued } }
            if = {
                # 战略 AI 持有者已由自己的 story 脉冲开战，这里不再重复触发。
                limit = {
                    hd_submission_war_ready_trigger = yes
                    NOT = { owns_story_of_type = hd_strategic_ai_story }
                }
                # Keeps current local CBs and all historical war-management guards.
                hd_unification_ai_attempt_effect = yes
            }
            # This also backs off when the engine has no legal/affordable CB.
            add_character_flag = { flag = hd_submission_action_cooldown days = 180 }
        }
    }
}

hd_submission_refresh_momentum_modifier_effect = {
    remove_character_modifier = hd_submission_momentum_1
    remove_character_modifier = hd_submission_momentum_2
    remove_character_modifier = hd_submission_momentum_3
    remove_character_modifier = hd_submission_momentum_4
    remove_character_modifier = hd_submission_momentum_5
    if = {
        limit = { has_variable = hd_submission_momentum }
        if = { limit = { var:hd_submission_momentum >= 5 } set_variable = { name = hd_submission_momentum value = 5 } add_character_modifier = { modifier = hd_submission_momentum_5 } }
        else_if = { limit = { var:hd_submission_momentum = 4 } add_character_modifier = { modifier = hd_submission_momentum_4 } }
        else_if = { limit = { var:hd_submission_momentum = 3 } add_character_modifier = { modifier = hd_submission_momentum_3 } }
        else_if = { limit = { var:hd_submission_momentum = 2 } add_character_modifier = { modifier = hd_submission_momentum_2 } }
        else_if = { limit = { var:hd_submission_momentum = 1 } add_character_modifier = { modifier = hd_submission_momentum_1 } }
        else = { remove_variable = hd_submission_momentum remove_character_flag = hd_submission_recent_success }
    }
}

hd_submission_momentum_victory_effect = {
    if = { limit = { NOT = { has_variable = hd_submission_momentum } } set_variable = { name = hd_submission_momentum value = 0 } }
    if = { limit = { var:hd_submission_momentum < 5 } change_variable = { name = hd_submission_momentum add = 1 } }
    add_character_flag = { flag = hd_submission_recent_success years = 3 }
    hd_submission_refresh_momentum_modifier_effect = yes
    remove_character_flag = hd_submission_action_cooldown
    hd_submission_queue_followup_effect = { DELAY_MIN = 60 DELAY_MAX = 120 }
}

hd_submission_momentum_defeat_effect = {
    if = {
        limit = { has_variable = hd_submission_momentum }
        if = { limit = { var:hd_submission_momentum >= 2 } change_variable = { name = hd_submission_momentum add = -2 } }
        else = { set_variable = { name = hd_submission_momentum value = 0 } }
        hd_submission_refresh_momentum_modifier_effect = yes
    }
    remove_character_flag = hd_submission_action_cooldown
    hd_submission_queue_followup_effect = { DELAY_MIN = 120 DELAY_MAX = 180 }
}

hd_submission_yearly_maintenance_effect = {
    if = {
        limit = { has_variable = hd_submission_momentum }
        if = {
            limit = { NOT = { has_character_flag = hd_submission_recent_success } var:hd_submission_momentum > 0 }
            change_variable = { name = hd_submission_momentum add = -1 }
        }
        hd_submission_refresh_momentum_modifier_effect = yes
    }
    # One annual lightweight recovery for old saves/new independent rulers.
    # The existing helper prevents duplicate stories.
    if = {
        limit = {
            is_ai = yes
            is_independent_ruler = yes
            government_has_flag = government_is_celestial
        }
        hd_unification_ai_ensure_story_effect = yes
    }
