@admin_law_cooldown_value = 5

han_province_title_law = {
	law_group_type = province_title_flavour_law
	index = 0
	flag = province_title_flavour_law
	flag = han_province_title
	flag = hanzhi

	modifier = {
		courtly_opinion = 5 # Pro-hegemon movement
		glory_hound_opinion = 5 # Conservative movement
		barons_and_minor_landholders_opinion = 5
		celestial_government_tax_contribution_add = 0.05
	}
	can_keep = {
		realm_law_use_celestial_bureaucracy = yes
		trigger_if = {
			limit = {
				is_independent_ruler = no
			}
			liege = { has_realm_law = han_province_title_law }
		}
	}

	can_pass = {
		NOT = { has_realm_law = tang_province_title_law }
		trigger_if = {
			limit = {
				is_independent_ruler = no
				liege = { has_realm_law = celestial_bureaucracy_0 }
			}
			custom_tooltip = {
				text = imperial_bureaucracy_pending_law_change
				always = no
			}
		}
		custom_tooltip = {
			text = admin_law_cooldown_desc
			NOT = { has_variable = admin_law_cooldown }
		}
	}

	pass_cost = {
		prestige = {
			if = {
				limit = {
					NOR = {
						has_realm_law = celestial_bureaucracy_2
						has_realm_law = celestial_bureaucracy_3
					}
				}
				add = increase_crown_authority_prestige_cost
			}
		}
	}

	on_pass = {
            trigger_event = { on_action = WJ_titles_after_change days = 1 } # on_pass still sees old law
		## Set cooldown
		set_variable = {
			name = admin_law_cooldown
			years = @admin_law_cooldown_value
		}
		## 全局切换: 所有封臣同步改法(隐藏效果, 浮窗不逐条列出封臣)
		hidden_effect = {
			every_vassal_or_below = {
				limit = {
					highest_held_title_tier >= tier_county
					realm_law_use_celestial_bureaucracy = yes
				}
				add_realm_law_skip_effects = han_province_title_law
			}
		}
	}

	ai_will_do = {
		value = 0
	}
}

			limit = {
				is_independent_ruler = no
			}

		}
	}

	can_pass = {
		OR = {
			has_realm_law = han_province_title_law
			has_realm_law = jin_province_title_law
		}
		trigger_if = {
			limit = {
				is_independent_ruler = no

			}

				text = imperial_bureaucracy_pending_law_change

			}
		}
		custom_tooltip = {
			text = admin_law_cooldown_desc
			NOT = { has_variable = admin_law_cooldown }
		}
	}

	pass_cost = {
		prestige = {
			if = {
				limit = {
					NOR = {
						has_realm_law = celestial_bureaucracy_2
						has_realm_law = celestial_bureaucracy_3
					}
				}
				add = increase_crown_authority_prestige_cost
			}
		}
	}

	on_pass = {
            trigger_event = { on_action = WJ_titles_after_change days = 1 } # on_pass still sees old law
		set_variable = {
			name = admin_law_cooldown
			years = @admin_law_cooldown_value
		}
		## 全局切换: 所有封臣同步改法(隐藏效果)

			every_vassal_or_below = {

				}
				add_realm_law_skip_effects = tang_province_title_law
			}
			# 批量转换不能在法律确认框逐人展开；仍在确认改制时正常执行。
			WJ_convert_han_jin_realm_to_tang_effect = yes
		}
		custom_tooltip = WJ_han_jin_to_tang_conversion_tt
	}

	# on_pass 仍是旧法律；必须等唐法生效，才能阻止撤职回调补位。
	on_after_pass = {
		hidden_effect = {
			WJ_retire_han_jin_ministries_effect = yes
			WJ_sili_reconcile_subrealm_effect = yes
		}
	}

	ai_will_do = {
		value = 0
	}
}
