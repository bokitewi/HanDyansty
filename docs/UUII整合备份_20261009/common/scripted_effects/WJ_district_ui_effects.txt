WJ_dd_refresh_view_effect = {

  clear_variable_list = wj_dd_ui_cores
  clear_variable_list = wj_dd_ui_candidates
  save_scope_as = wj_dd_viewer
  if = {
   limit = { WJ_han_jin_law_trigger = yes }
   top_liege = {
    every_vassal_or_below = {
     if = {
      limit = { WJ_dd_head_trigger = yes }
      WJ_dd_ensure_kaifu_effect = yes
      primary_title = {
       save_scope_as = wj_dd_ui_core
       WJ_dd_refresh_name_effect = yes
       WJ_dd_refresh_removal_preview_effect = yes
      }
      WJ_dd_refresh_district_people_effect = yes
      scope:wj_dd_viewer = { add_to_variable_list = { name = wj_dd_ui_cores target = scope:wj_dd_ui_core } }
     }
     else_if = {
      limit = {
       WJ_dd_head_candidate_trigger = yes
       NOT = { primary_title = { has_variable = wj_dd_core } }
       scope:wj_dd_viewer = { WJ_zw_central_viewer = yes }
      }
      save_scope_as = wj_dd_ui_candidate
      scope:wj_dd_viewer = { add_to_variable_list = { name = wj_dd_ui_candidates target = scope:wj_dd_ui_candidate } }
     }
    }
   }
  }
  # Remove only the retired display experiment from older saves; never recreate it.
  WJ_dd_map_end_prototype_effect = yes
}

# Title-scope cache for disabled buttons only. The committing transaction runs
# the connectivity traversal again and never trusts this cache as authority.
WJ_dd_refresh_removal_preview_effect = {
 save_scope_as = wj_dd_preview_core
 # Display only subordinate members below the head card. The authoritative
 # wj_dd_members still includes the core for map coverage and connectivity.
 clear_variable_list = wj_dd_ui_members
 every_in_list = {
  variable = wj_dd_members
  remove_variable = wj_dd_ui_can_remove
 }
 every_in_list = {
  variable = wj_dd_members
  limit = { this != scope:wj_dd_preview_core }
  save_scope_as = wj_dd_preview_member
  scope:wj_dd_preview_core = {
   add_to_variable_list = { name = wj_dd_ui_members target = scope:wj_dd_preview_member }
   set_variable = { name = wj_dd_excluded value = scope:wj_dd_preview_member }
   WJ_dd_check_connectivity_effect = yes
   if = {
    limit = { NOT = { has_variable = wj_dd_graph_disconnected } }
    scope:wj_dd_preview_member = { set_variable = { name = wj_dd_ui_can_remove value = yes } }
   }
   remove_variable = wj_dd_excluded
   remove_variable = wj_dd_graph_disconnected
   clear_variable_list = wj_dd_unreachable
  }
 }
