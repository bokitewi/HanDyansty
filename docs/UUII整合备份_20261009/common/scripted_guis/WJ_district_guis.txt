WJ_dd_ui_rename = {
 scope = character
 is_valid = { WJ_zw_not_superseded = yes }
 is_shown = { is_ai = no scope:head = { WJ_dd_head_trigger = yes OR = { this = root top_liege = root AND = { top_liege = root.top_liege root = { WJ_zw_valid = yes } } } } }
 effect = {
  scope:head = { save_scope_as = wj_dd_rename_head primary_title = { save_scope_as = wj_dd_rename_core } }
  if = { limit = { WJ_dd_rename_valid = yes }
   if = { limit = { scope:wj_dd_rename_core = { has_variable = wj_dd_custom_name_slot var:wj_dd_custom_name_slot = 1 } } trigger_event = WJ_district_expansion.2 }
   else = { trigger_event = WJ_district_expansion.1 }
  }
 }
}

WJ_dd_ui_officers_access = {
 scope = character
 is_valid = { WJ_zw_not_superseded = yes }
 is_shown = {
  scope:head = {
   WJ_dd_head_trigger = yes
   OR = { this = root any_liege_or_above = { this = root } AND = { top_liege = root.top_liege root = { WJ_zw_valid = yes } } }
  }
 }
 effect = {
  if = {
   limit = { WJ_zw_not_superseded = yes scope:head = { WJ_dd_head_trigger = yes OR = { this = root any_liege_or_above = { this = root } AND = { top_liege = root.top_liege root = { WJ_zw_valid = yes } } } } }
   set_variable = { name = wj_dd_officers_view value = scope:head }
   scope:head = { WJ_dd_refresh_district_people_effect = yes }
  }
 }
}
WJ_dd_ui_officers_visible = {
	is_shown = {
		trigger_if = {
			limit = { exists = scope:hd_gui_player }
			scope:hd_gui_player = {
				has_variable = wj_dd_officers_view
				var:wj_dd_officers_view = {
					WJ_dd_head_trigger = yes
					OR = { this = scope:hd_gui_player any_liege_or_above = { this = scope:hd_gui_player } AND = { top_liege = scope:hd_gui_player.top_liege scope:hd_gui_player = { WJ_zw_valid = yes } } }
				}
			}
		}
		trigger_else = { always = no }
	}
}
