# Character scope. Only current Han/Jin celestial officials; no marker creates eligibility.
WJ_contract_privilege_realm_trigger = {
	is_alive = yes
	is_independent_ruler = no
	government_has_flag = government_is_celestial
	WJ_han_jin_law_trigger = yes
}

WJ_contract_minister_eligible_trigger = {
	WJ_contract_privilege_realm_trigger = yes
	tgp_is_any_minister = yes
}

WJ_contract_six_offices_trigger = {
	WJ_contract_minister_eligible_trigger = yes
	OR = {
		has_title = title:e_minister_grand_marshal
		has_title = title:e_WJ_minister_tai_fu
		has_title = title:e_WJ_minister_cheng_xiang
		has_title = title:e_minister_chancellor
		has_title = title:e_minister_censor
		has_title = title:e_minister_of_war
	}
}

# A local title, not a noble-family or ministry title. Used in title scope.
WJ_contract_local_title_trigger = {
	is_noble_family_title = no
	tgp_is_minister_title = no
	tier >= tier_county
}

WJ_contract_local_official_trigger = {
	WJ_contract_privilege_realm_trigger = yes
	is_landed = yes
	tgp_has_minister_title = no
	any_held_title = { WJ_contract_local_title_trigger = yes }
	custom_description = { text = WJ_failure_9
OR = {
		vassal_contract_has_flag = celestial_province_standard
		vassal_contract_has_flag = celestial_province_military
		vassal_contract_has_flag = WJ_celestial_province_sili
	}
}
}

WJ_contract_sanjie_visible_trigger = {
	WJ_contract_local_official_trigger = yes
	custom_description = { text = WJ_failure_10
OR = {
		vassal_contract_has_flag = celestial_province_military
		vassal_contract_has_flag = WJ_celestial_province_sili
	}
}
}

# Real core office only; a military contract or cached reference is insufficient.
WJ_contract_dudu_trigger = {
	WJ_dd_head_trigger = yes
}

WJ_contract_sanjie_eligible_trigger = {
	WJ_contract_sanjie_visible_trigger = yes
	WJ_contract_dudu_trigger = yes
}

WJ_contract_kaifu_visible_trigger = {
	OR = {
		WJ_contract_minister_eligible_trigger = yes
		WJ_contract_local_official_trigger = yes
	}
}

WJ_contract_kaifu_eligible_trigger = {
	custom_description = { text = WJ_failure_0
OR = {
		WJ_contract_minister_eligible_trigger = yes
		AND = {
			WJ_contract_local_official_trigger = yes
			any_held_title = {
				WJ_contract_local_title_trigger = yes
				tier >= tier_kingdom
			}
		}
	}
}
}

# Read the two native sources separately. Six-office eligibility alone is not a grant.
WJ_has_kaifu_privilege_trigger = {
	WJ_contract_kaifu_eligible_trigger = yes
	custom_description = { text = WJ_failure_1
OR = {
		vassal_contract_has_flag = WJ_kaifu_granted
		AND = {
			WJ_contract_six_offices_trigger = yes
			vassal_contract_has_flag = WJ_kaifu_by_office
		}
	}
}
}

# Raw obligation levels also find invalid states that need clearing after an identity change.
WJ_has_contract_privilege_state_trigger = {
	OR = {
		vassal_contract_obligation_level:WJ_sanjie > 0
		vassal_contract_obligation_level:WJ_kaifu > 0
		vassal_contract_obligation_level:WJ_kaifu_office > 0
		vassal_contract_obligation_level:WJ_huangyue > 0
	}
}
