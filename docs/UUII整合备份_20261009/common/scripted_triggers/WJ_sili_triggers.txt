# Character scope: actual capital kingdom, not necessarily the primary title.
WJ_sili_eligible_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = yes
	is_independent_ruler = no
	government_has_flag = government_is_celestial
	WJ_han_jin_law_trigger = yes
	WJ_has_celestial_province_contract_trigger = yes
	exists = top_liege.capital_county.kingdom
	has_title = top_liege.capital_county.kingdom
}

# One slot per court. Exclude the contract subject by identity, not its draft flags.
WJ_sili_slot_available_trigger = {
	exists = scope:subject
	scope:subject.top_liege = {
		NOT = {
			any_vassal_or_below = {
				this != scope:subject
				vassal_contract_has_flag = WJ_celestial_province_sili
			}
		}
	}
}
