# Character scope: Cao Cao. These conditions also apply after the invitation.
hd_cao_fengying_actor_valid_trigger = {
	this = character:cao_cao
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	character:liu_xie = { is_alive = yes }
	title:h_china.holder = character:liu_xie
}

hd_cao_fengying_invitation_trigger = {
	hd_cao_fengying_actor_valid_trigger = yes
	current_date > 196.1.1
	current_military_strength > 10000
	hd_cao_controls_xuchang_trigger = yes
	hd_dong_transition_pending_trigger = no
}

hd_cao_fengying_can_declare_trigger = {
	hd_cao_fengying_actor_valid_trigger = yes
	has_character_flag = hd_cao_fengying_authorized
	NOT = { exists = global_var:hd_fengying_world_resolved }
	hd_dong_transition_pending_trigger = no
	is_at_war = no
	is_imprisoned = no
	exists = title:d_jingzhao.holder
	title:d_jingzhao.holder = { is_alive = yes NOT = { this = character:cao_cao } }
	NOT = { has_truce = title:d_jingzhao.holder }
	NOT = { is_allied_to = title:d_jingzhao.holder }
	can_declare_war = { defender = title:d_jingzhao.holder casus_belli = hd_cao_fengying_cb }
}

hd_fengying_world_ready_trigger = {
	hd_dong_transition_pending_trigger = no
	exists = global_var:hd_anti_regent_chain_resolved
	NOT = { exists = global_var:hd_fengying_world_open }
	NOT = { exists = global_var:hd_fengying_world_claimed }
	NOT = { exists = global_var:hd_fengying_world_resolved }
	character:liu_xie = { is_alive = yes }
	title:h_china.holder = character:liu_xie
	character:cao_cao = { hd_cao_fengying_invitation_trigger = yes }
}

hd_guandu_world_ready_trigger = {

    NOT = { exists = global_var:hd_guandu_world_active }

hd_cao_historical_war_leader_trigger = {
	OR = {
		is_independent_ruler = yes
		AND = {
			liege = title:h_china.holder
			title:h_china.holder = { has_variable = hd_dong_fixed_regent var:hd_dong_fixed_regent = character:cao_cao }
		}
	}
}

# Chibi remains a shared Cao/Sun/Liu world-history opportunity.  It is not part
# of the removed Liu Bei personal chain and is intentionally preserved.
hd_chibi_world_ready_trigger = {
	exists = global_var:hd_guandu_world_resolved
	NOT = { exists = global_var:hd_chibi_world_open }
	NOT = { exists = global_var:hd_chibi_world_active }
	NOT = { exists = global_var:hd_chibi_world_resolved }
	character:cao_cao = { is_alive = yes }
	character:sun_quan = { is_alive = yes }
	character:liu_bei = { is_alive = yes }
}

hd_chibi_historical_war_possible_trigger = {
	character:cao_cao = { is_alive = yes is_landed = yes hd_cao_historical_war_leader_trigger = yes realm_size >= 25 NOT = { is_allied_to = character:sun_quan } }
	character:sun_quan = { is_alive = yes is_landed = yes is_independent_ruler = yes realm_size >= 8 NOT = { is_allied_to = character:cao_cao } primary_title = { tier >= tier_county } }
	character:liu_bei = { is_alive = yes is_landed = yes realm_size >= 1 }
	NOT = { character:cao_cao.top_liege = character:sun_quan.top_liege }
	OR = { exists = global_var:hd_chibi_sun_liu_alliance character:sun_quan = { is_allied_to = character:liu_bei } }
	OR = { title:k_uuii_yangzhou.holder.top_liege = character:sun_quan character:sun_quan = { realm_size >= 18 } }
}
