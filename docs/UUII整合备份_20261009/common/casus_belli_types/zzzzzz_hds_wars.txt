 target_title_tier = duchy

 ignore_effect = change_title_holder
 on_declaration = { on_declared_war = yes }
 on_victory_desc = { desc = hds_war_victory_desc }
 on_victory = { hds_cb_victory_effect = yes add_truce_attacker_victory_effect = yes }
 on_white_peace_desc = { desc = hds_war_no_conquest_desc }
 on_white_peace = { hds_cb_retry_effect = yes add_truce_white_peace_effect = yes }
 on_defeat_desc = { desc = hds_war_no_conquest_desc }
 on_defeat = { hds_cb_retry_effect = yes add_truce_attacker_defeat_effect = yes }
 on_invalidated_desc = msg_invalidate_war_title
 on_invalidated = { hds_cb_retry_effect = yes }
 on_primary_attacker_death = invalidate
 on_primary_defender_death = invalidate
 attacker_allies_inherit = yes
 defender_allies_inherit = yes
 transfer_behavior = transfer
 interface_priority = 75

 allowed_for_character = { scope:attacker = { is_alive = yes is_landed = yes is_independent_ruler = yes hds_regional_actor_trigger = yes } }
 valid_to_start = {
  scope:attacker = { has_character_flag = hds_launch_authorized }
  scope:target = scope:attacker.var:hds_requested_duchy
  scope:target = { hds_cb_target_region_trigger = yes hds_enemy_holds_target_trigger = yes }
 }
 should_invalidate = {
  OR = {
   scope:attacker = { is_alive = no }
   scope:defender = { is_alive = no }
   NOT = { scope:attacker = { hds_regional_actor_trigger = yes } }
   NOT = { any_in_list = { list = target_titles hds_enemy_holds_target_trigger = yes } }
  }
 }
 cb_name = hds_regional_cb_name
 war_name_base = hds_regional_cb_name

}
