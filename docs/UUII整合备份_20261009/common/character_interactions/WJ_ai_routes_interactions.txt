WJ_ai_bizhao_01 = {
 category = interaction_category_friendly icon = mentor
 name = WJ_mufu_recruit desc = WJ_bizhao_recruit_desc
 auto_accept = yes use_diplomatic_range = no
 is_available = { WJ_ai_bizhao_available = yes }
 is_shown = { WJ_ai_bizhao_valid = yes }
 is_valid_showing_failures_only = { WJ_ai_bizhao_valid = yes }
 on_accept = {
  if = {
   limit = { WJ_ai_bizhao_valid = yes scope:actor = { WJ_ai_bizhao_available = yes } }
   WJ_yangwang_select_mufu_slot_effect = yes
   if = {
    # 隐士：转为"辟书叩衡门"事件，由其决定是否出山入幕。
    limit = { WJ_yinshi_recipient_is_hermit_trigger = yes }
    # 隐士被囚或无行为能力时不受理，也不锁府主。
    if = {
     limit = { scope:recipient = { WJ_yinshi_able_trigger = yes } }
     save_scope_value_as = { name = wj_yinshi_via value = flag:ai_route }
     scope:recipient = { save_scope_as = wj_yinshi_subject }
     WJ_yinshi_mufu_lock_effect = yes
     scope:recipient = { trigger_event = WJ_yinshi.5001 }
    }
   }
   else_if = { limit = { scope:recipient = { is_ai = yes } } WJ_ai_bizhao_complete = yes }
   else = {
    WJ_ai_choose_offer = yes
    scope:actor = {
     if = { limit = { NOT = { has_variable = wj_ai_bizhao_serial } } set_variable = { name = wj_ai_bizhao_serial value = 0 } }
     change_variable = { name = wj_ai_bizhao_serial add = 1 }
     save_scope_value_as = { name = wj_ai_serial value = var:wj_ai_bizhao_serial }
     set_variable = { name = wj_ai_bizhao_pending value = scope:recipient days = 61 }
    }
    scope:recipient = { trigger_event = WJ_ai_routes.0001 }
   }
  }
 }
 ai_targets = { ai_recipients = courtiers ai_recipients = vassals max = 12 }
 ai_frequency_by_tier = { barony = 0 county = 0 duchy = 0 kingdom = 1 empire = 1 hegemony = 1 }
 ai_will_do = {
  base = 100
  modifier = { factor = 0.1 WJ_yinshi_recipient_is_hermit_trigger = yes }
 }
}
