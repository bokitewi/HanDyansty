hd_story_historical_ai_controller = {
	visible = no
	icon = { trigger = { always = yes } reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds" }
	on_owner_death = { scope:story = { end_story = yes } }
	effect_group = {
		days = { 15 30 }
		triggered_effect = {
			trigger = { story_owner = { is_alive = yes } }
			effect = {
				story_owner = {
					# Cao and Yuan use separate single-region expansion schedulers.
					# Other historical actors retain the original dispatcher unchanged.
					if = {
						limit = { this = character:cao_cao }
						hd_cao_expansion_legacy_tick_effect = yes
					}
					else_if = {
						limit = { this = character:yuan_shao }
						hd_yuan_expansion_legacy_tick_effect = yes
					}
					else = { hd_historical_actor_story_tick_effect = yes }
				}
			}
		}
	}
}

hd_story_sun_jian_life = {
	visible = yes
	icon = { trigger = { always = yes } reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds" }
	on_setup = { set_variable = { name = hd_sun_life_phase value = flag:hd_sun_life_sun_jian } }
	visualization = { custom_string_key = "HD_SUN_JIAN_LIFE_STORY_STATUS" }
	on_owner_death = {
		if = {
			limit = { story_owner = { this = character:sun_jian } character:sun_ce ?= { is_alive = yes } }
			make_story_owner = character:sun_ce
			set_variable = { name = hd_sun_life_phase value = flag:hd_sun_life_sun_ce }
			# The death hook snapshots assets; native inheritance finishes first.
			title:h_china = { trigger_event = { on_action = hds_succession_tick days = 2 } }
		}
		else = { scope:story = { end_story = yes } }
	}
}
