# 追加原生人物入口。换职仅在交接完毕次日复核身份，不按具体职位截断。
on_title_gain = { on_actions = { WJ_kaoke_title_changed } }
on_title_lost = { on_actions = { WJ_kaoke_title_changed } }
on_government_change = { on_actions = { WJ_kaoke_identity_changed } }
on_liege_government_change = { on_actions = { WJ_kaoke_identity_changed } }
on_death = { on_actions = { WJ_kaoke_death } }
on_game_start_after_lobby = { on_actions = { WJ_kaoke_new_game } }
# 当前本体 yearly_on_actions.txt 的实际代码入口；不沿用过时.info示例名。
yearly_playable_pulse = { on_actions = { WJ_kaoke_yearly_check } }
WJ_kaoke_title_changed = {
 trigger = { OR = { has_variable = wj_kaoke_start WJ_kaoke_eligible_trigger = yes } }
 effect = { trigger_event = { on_action = WJ_kaoke_identity_changed days = 1 } }
}
WJ_kaoke_identity_changed = {
 trigger = { OR = { has_variable = wj_kaoke_start WJ_kaoke_eligible_trigger = yes } }
 effect = { WJ_kaoke_reconcile_effect = yes }
}
WJ_kaoke_death = {
 trigger = { OR = { has_variable = wj_kaoke_start has_variable = wj_kaoke_window_epoch } }
 effect = {
  WJ_kaoke_clear_effect = yes
  # 原生on_death开始时可能尚活着，明确清除最后一个窗口废止编号。
  remove_variable ?= wj_kaoke_window_epoch
 }
}
# 原生生日入口兜底资格变化／旧档首次初始化，不另设全图轮询。
WJ_kaoke_yearly_check = {
 trigger = { OR = { has_variable = wj_kaoke_start WJ_kaoke_eligible_trigger = yes } }
 effect = { WJ_kaoke_due_effect = yes }
}
WJ_kaoke_new_game = {
 trigger = {
  exists = title:h_china.holder
  title:h_china.holder = { WJ_han_jin_law_trigger = yes }
 }
 effect = {
  title:h_china.holder = {
   every_vassal_or_below = {
    limit = { WJ_kaoke_eligible_trigger = yes }
    WJ_kaoke_reconcile_effect = yes
   }
  }
 }
}
