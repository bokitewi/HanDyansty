# 太尉摄政法律下，天朝最高头衔换手后按现任太尉重建尚书台。
on_title_gain = {
	on_actions = {
		WJ_taiwei_regency_title_gain
	}
}

WJ_taiwei_regency_title_gain = {
	trigger = {
		scope:title = title:h_china
		root = { has_realm_law = WJ_grand_taiwei_law }
		exists = title:e_minister_of_war.holder
	}
	effect = {
		root = {
			if = {
				limit = { NOT = { has_diarchy_type = grand_secretariat } }
				if = {
					limit = { has_active_diarchy = yes }
					end_diarchy = yes
				}
				try_start_diarchy = grand_secretariat
			}
			set_diarch = title:e_minister_of_war.holder
		}
	}
}
